FORM GET_DATA.

  CLEAR: GT_UPLOAD, GT_UPLOAD[], GT_TAB, GT_TAB[].

  DATA : LF_RETRN TYPE CHAR1.

  DATA : LT_EXCEL_TAB  TYPE ALSMEX_TABLINE OCCURS 0 WITH HEADER LINE.
  DATA : LV_INDEX    LIKE SY-TABIX.

  FIELD-SYMBOLS : <FS> TYPE ANY.

  CALL FUNCTION 'WS_QUERY' " 파일 존재여부 체크
    EXPORTING
      FILENAME = P_FILE
      QUERY    = 'FE'
    IMPORTING
      RETURN   = LF_RETRN.      " '0' = 없다,  '1' = 존재한다

  IF LF_RETRN = '0'.
    EXIT.
  ENDIF.


  IF P_FILE NE SPACE.
*___Upload the Excel File at Excel Table
    CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
      EXPORTING
        FILENAME                = P_FILE
        I_BEGIN_COL             = GC_BEGIN_COL
        I_BEGIN_ROW             = GC_BEGIN_ROW
        I_END_COL               = GC_END_COL
        I_END_ROW               = GC_END_ROW
      TABLES
        INTERN                  = LT_EXCEL_TAB
      EXCEPTIONS
        INCONSISTENT_PARAMETERS = 1
        UPLOAD_OLE              = 2
        OTHERS                  = 3.
  ENDIF.

  LOOP AT LT_EXCEL_TAB WHERE ROW GE 2.
    MOVE : LT_EXCEL_TAB-COL TO LV_INDEX.
    ASSIGN COMPONENT LV_INDEX OF STRUCTURE GT_UPLOAD TO <FS>.
    MOVE : LT_EXCEL_TAB-VALUE TO <FS>.

    AT END OF ROW.

      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          INPUT  = GT_UPLOAD-MATNR
        IMPORTING
          OUTPUT = GT_UPLOAD-MATNR.

      PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING GT_UPLOAD-PREVE.

      APPEND GT_UPLOAD.
      CLEAR  GT_UPLOAD.
    ENDAT.
  ENDLOOP.

  LOOP AT GT_UPLOAD.
    MOVE-CORRESPONDING GT_UPLOAD TO GT_TAB.
    SELECT SINGLE MAKTX INTO GT_TAB-MAKTX
      FROM MAKT
      WHERE MATNR = GT_TAB-MATNR.

    APPEND GT_TAB.
  ENDLOOP.

ENDFORM. " GET_DATA

FORM SET_DATA .

  DATA : BEGIN OF LT_DB OCCURS 0,
          PREVE TYPE ZPMT0080-PREVE,
          MATNR TYPE ZPMT0080-MATNR,
          ERNAM TYPE ZPMT1000-ERNAM,
          ERDAT TYPE ZPMT1000-ERDAT,
          ERZET TYPE ZPMT1000-ERZET,
          ERPSN TYPE ZPMT1000-ERPSN,
         END   OF LT_DB.

  CLEAR: LT_DB, LT_DB[].

  CHECK GT_TAB[] IS NOT INITIAL.

  SELECT PREVE MATNR ERNAM ERDAT ERZET ERPSN
    INTO TABLE LT_DB
    FROM ZPMT0080
     FOR ALL ENTRIES IN GT_TAB
   WHERE PREVE = GT_TAB-PREVE
     AND MATNR = GT_TAB-MATNR.

  LOOP AT GT_TAB.

    GT_TAB-AENAM = SY-UNAME.
    GT_TAB-AEDAT = SY-DATUM.
    GT_TAB-AEZET = SY-UZEIT.
    call function 'ZPM_GET_EMPNO'
      IMPORTING
        E_PERNR = GT_TAB-AEPSN.

    READ TABLE LT_DB WITH KEY PREVE = GT_TAB-PREVE.
    IF SY-SUBRC = 0.
      GT_TAB-ERNAM = LT_DB-ERNAM.
      GT_TAB-ERDAT = LT_DB-ERDAT.
      GT_TAB-ERZET = LT_DB-ERZET.
      GT_TAB-ERPSN = LT_DB-ERPSN.
    ELSE.
      GT_TAB-ERNAM = SY-UNAME.
      GT_TAB-ERDAT = SY-DATUM.
      GT_TAB-ERZET = SY-UZEIT.
      call function 'ZPM_GET_EMPNO'
        IMPORTING
          E_PERNR = GT_TAB-ERPSN.
    ENDIF.

    MODIFY GT_TAB.
  ENDLOOP.

ENDFORM. " SET_DATA
