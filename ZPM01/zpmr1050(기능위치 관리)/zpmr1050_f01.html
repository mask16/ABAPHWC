FORM POPUP_TO_CONFIRM USING P_VAL
                    CHANGING    P_ANSWER.
  DATA : L_TITLE(30),
         L_QUESTION(100).

  CLEAR : P_ANSWER, L_TITLE, L_QUESTION.

  CASE P_VAL.
    WHEN 'SAVE'.
      L_TITLE    = TEXT-A01.   "저 장
      L_QUESTION = TEXT-A02.   "변경하신 내용을 저장 하시겠습니까?
    WHEN 'EXIT'.
      L_TITLE    = TEXT-A05.   "확 인
      L_QUESTION = TEXT-A06.   "변경된 내용이 있습니다. 저장하시겠습니까?
  ENDCASE.

  IF P_VAL = 'EXIT'.
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
      EXPORTING
        DEFAULTOPTION = 'N'
        TITEL         = L_TITLE
        TEXTLINE1     = L_QUESTION
      IMPORTING
        ANSWER        = P_ANSWER.
  ELSE.
    CALL FUNCTION 'POPUP_TO_CONFIRM'
      EXPORTING
        TITLEBAR              = L_TITLE
        TEXT_QUESTION         = L_QUESTION
        TEXT_BUTTON_1         = '예'    " '
        ICON_BUTTON_1         = 'ICON_OKAY'
        TEXT_BUTTON_2         = '아니오'    " '
        ICON_BUTTON_2         = 'ICON_CANCEL'
        DEFAULT_BUTTON        = '2'
        DISPLAY_CANCEL_BUTTON = ''
      IMPORTING
        ANSWER                = P_ANSWER
      EXCEPTIONS
        TEXT_NOT_FOUND        = 1
        OTHERS                = 2.
  ENDIF.

ENDFORM. " POPUP_TO_CONFIRM

FORM GET_DATA.

  DATA : LV_PLTXT TYPE STRING.
  DATA : BEGIN OF LT_CSKT OCCURS 0,
           KOSTL LIKE CSKT-KOSTL,
           KTEXT LIKE CSKT-KTEXT,
         END OF LT_CSKT.

  CLEAR : GT_TAB, GT_TAB[].

*  CONCATENATE 'PLTXT LIKE ''%'P_PLTXT '%'''  INTO LV_PLTXT.

  RANGES: R_PLTXT FOR ZPMT0010-PLTXT.
  CLEAR : R_PLTXT , R_PLTXT[].

  IF P_PLTXT IS NOT INITIAL.
    R_PLTXT = 'ICP'. R_PLTXT-LOW = P_PLTXT.
    TRANSLATE R_PLTXT-LOW TO UPPER CASE.
    APPEND R_PLTXT.
  ENDIF.

  SELECT * INTO TABLE GT_TAB
    FROM ZPMT0010
    WHERE BUKRS IN S_BUKRS
      AND WERKS IN S_WERKS
      AND TPLNR IN S_TPLNR
      AND PLTXT IN R_PLTXT.
*      AND (LV_PLTXT).

  SORT GT_TAB BY TPLNR.

  LOOP AT GT_TAB.

    PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING GT_TAB-KOSTL.
    SELECT SINGLE KTEXT INTO GT_TAB-KTEXT
      FROM CSKT
      WHERE KOSTL = GT_TAB-KOSTL
        AND SPRAS = '3'
        AND DATBI >= SY-DATUM.

    READ TABLE GT_PLTYP WITH KEY PLTYP = GT_TAB-PLTYP
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      GT_TAB-PLTYP_TX = GT_PLTYP-PLTYP_TX.
    ENDIF.

    PERFORM GET_NAME_TEXT USING 'PERNR' GT_TAB-ERPSN
                       CHANGING GT_TAB-ERPSN_TX.

    PERFORM GET_NAME_TEXT USING 'PERNR' GT_TAB-AEPSN
                       CHANGING GT_TAB-AEPSN_TX.

    MODIFY GT_TAB.

  ENDLOOP.

ENDFORM. " GET_DATA

FORM SAVE_PROCESS .

  DATA : L_LEVEL TYPE I.
  CLEAR: L_LEVEL.

  DATA: L_OK TYPE C.
  CLEAR L_OK.
  PERFORM CK_TPLNR_FORMAT USING GS_TAB-TPLNR '1'
                       CHANGING L_OK G_LEVEL.
  CHECK L_OK IS INITIAL.

  CLEAR L_OK.
  PERFORM CK_TPLNR_FORMAT USING GS_TAB-TPLma '2'
                       CHANGING L_OK L_LEVEL.
  CHECK L_OK IS INITIAL.

  L_LEVEL = L_LEVEL + 1.

  IF G_LEVEL <> L_LEVEL.
    MESSAGE E025.
    EXIT.
  ENDIF.

* GV_TNEW 가 'Y' 이면 기능위치생성으로 들어온 경우이고, 아니면 기능위치 변경으로 들어온 경우
  IF GV_TNEW = 'Y'.
    SELECT SINGLE *
      FROM ZPMT0010
     WHERE BUKRS = GS_TAB-BUKRS
       AND WERKS = GS_TAB-WERKS
       AND TPLNR = GS_TAB-TPLNR.
    IF SY-SUBRC = 0.
      MESSAGE E001 WITH GS_TAB-TPLNR.
      EXIT.
    ENDIF.
    GS_TAB-ERNAM = GS_TAB-AENAM = SY-UNAME.
    GS_TAB-ERDAT = GS_TAB-AEDAT = SY-DATUM.
    GS_TAB-ERZET = GS_TAB-AEZET = SY-UZEIT.
    call function 'ZPM_GET_EMPNO'
      IMPORTING
        E_PERNR = GS_TAB-ERPSN.
    call function 'ZPM_GET_EMPNO'
      IMPORTING
        E_PERNR = GS_TAB-AEPSN.

    MODIFY ZPMT0010 FROM GS_TAB.
    IF SY-SUBRC = 0.
      MESSAGE S007 WITH '기능위치' GS_TAB-TPLNR.
      LEAVE TO SCREEN 0.
    ENDIF.
    LEAVE TO SCREEN 0.

  ELSE.

    GS_TAB-AENAM = SY-UNAME.
    GS_TAB-AEDAT = SY-DATUM.
    GS_TAB-AEZET = SY-UZEIT.
    call function 'ZPM_GET_EMPNO'
      IMPORTING
        E_PERNR = GS_TAB-AEPSN.

    MODIFY ZPMT0010 FROM GS_TAB.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING GS_TAB TO GT_TAB.
      IF GS_ROW-INDEX IS NOT INITIAL.
        MODIFY GT_TAB INDEX GS_ROW-INDEX.
      ENDIF.
      PERFORM SET_ORIGINAL_DATA.
      MESSAGE S007 WITH '기능위치' GS_TAB-TPLNR.
      LEAVE TO SCREEN 0.
    ENDIF.

  ENDIF.

ENDFORM. " SAVE_PROCESS

FORM GET_NAME_TEXT USING P_FIELDNAME
                             P_VALUE
                    CHANGING P_TEXT.
  DATA : L_TABLE TYPE TABNAME,
         L_WHERE TYPE STRING,
         L_SELECT TYPE STRING.

  CASE P_FIELDNAME.
    WHEN 'BUKRS'.
      L_TABLE  = 'T001'.
      L_WHERE  = 'BUKRS = p_value'.
      L_SELECT = 'BUTXT'.
    WHEN 'WERKS'.
      L_TABLE = 'T001W'.
      L_WHERE = 'WERKS = p_value'.
      L_SELECT = 'NAME1'.
    WHEN 'EKORG'.
      L_TABLE  = 'T024E'.
      L_WHERE  = 'EKORG = p_value'.
      L_SELECT = 'EKOTX'.
    WHEN 'EKGRP'.
      L_TABLE  = 'T024'.
      L_WHERE  = 'EKGRP = p_value'.
      L_SELECT = 'EKNAM'.
    WHEN 'PERNR'.
      L_TABLE = 'PA0001'.
      L_WHERE = 'PERNR = p_value AND endda = ''99991231'''.
      L_SELECT = 'SNAME'.
    WHEN 'ORGEH'.
      L_TABLE = 'T527X'.
      L_WHERE = 'SPRSL = SY-LANGU and ORGEH = p_value AND endda = ''99991231'''.
      L_SELECT = 'ORGTX'.
    WHEN 'LIFNR'.
      L_TABLE = 'LFA1'.
      L_WHERE = 'LIFNR = p_value'.
      L_SELECT = 'NAME1'.
    WHEN 'KNTTP'.
      L_TABLE = 'T163I'.
      L_WHERE = 'SPRAS = SY-LANGU and KNTTP = p_value '.
      L_SELECT = 'KNTTX'.
    WHEN 'KOSTL'.
      L_TABLE = 'CSKT'.
      L_WHERE = 'KOKRS = ''1000'' and KOSTL = p_value AND datbi = ''99991231'''.
      L_SELECT = 'MCTXT'.
    WHEN 'SAKNR'.
      L_TABLE   = 'SKAT'.
      L_WHERE   = 'SPRAS = SY-LANGU AND KTOPL = ''1000'' and SAKNR = p_value '.
      L_SELECT  = 'TXT20'.
    WHEN 'MATKL'.
      L_TABLE = 'T023T'.
      L_WHERE = 'SPRAS = SY-LANGU and MATKL = p_value'.
      L_SELECT = 'WGBEZ'.
    WHEN OTHERS.
      EXIT.
  ENDCASE.

  SELECT SINGLE (L_SELECT)
    INTO P_TEXT
    FROM (L_TABLE)
   WHERE (L_WHERE).

ENDFORM. " GET_NAME_TEXT
