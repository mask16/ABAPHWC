INITIALIZATION.
  call function 'ZPM_CHECK_AUTHORITY .'
* ZPMR1000 공무부 작업현황 프로그램에서 T-CODE 호출할때 값설정
  GET PARAMETER ID 'ZCR' FIELD gv_tnew.
  GET PARAMETER ID 'ZTP' FIELD gv_ttplnr.
  GET PARAMETER ID 'BUK' FIELD gv_bukrs.
  GET PARAMETER ID 'WRK' FIELD gv_werks.
  GET PARAMETER ID 'ZVIEW' FIELD gv_view.


* PARAMETER 값 CLEAR
  SET PARAMETER ID 'ZCR' FIELD ''.
  SET PARAMETER ID 'ZTP' FIELD ''.
  SET PARAMETER ID 'ZVIEW' FIELD ''.

  PERFORM get_domvalues_pltyp USING 'ZDPLTYP'.

  SET TITLEBAR  '0100'.
*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN
*&---------------------------------------------------------------------*

START-OF-SELECTION.
  DATA : lv_str(2) TYPE n.
  CLEAR: gt_tab, gt_tab[], gv_col_pos, lv_str.

* 생성
  IF gv_tnew = 'Y'.

    CLEAR : gs_tab.
    gs_tab-bukrs = gv_bukrs.
    gs_tab-werks = gv_werks.
    gs_tab-tplma = gv_ttplnr.
    gs_tab-tplnr = gv_ttplnr.
    gs_tab-ernam = gs_tab-aenam = sy-uname.
    gs_tab-erdat = gs_tab-aedat = sy-datum.
    gs_tab-erzet = gs_tab-aezet = sy-uzeit.

    lv_str = strlen( gs_tab-tplma ).  "상위 기능위치의 LENGTH에 따라
    CASE lv_str.
      WHEN 5.  gs_tab-pltyp = 'B'.
      WHEN 9.  gs_tab-pltyp = 'C'.
      WHEN 12. gs_tab-pltyp = 'D'.
      WHEN 15. gs_tab-pltyp = 'E'.
      WHEN 18. gs_tab-pltyp = 'F'.
    ENDCASE.

    PERFORM get_bukrs_text.
    PERFORM get_werks_text.
    PERFORM get_pltyp_text.
    PERFORM get_tplma_text.

    CALL SCREEN 200.

* 변경,조회
  ELSE.

* DATA GATHERING
    PERFORM get_data.

    IF p_loevm = 'X'.
      DELETE gt_tab WHERE loevm NE 'X'.
    ENDIF.
    IF gt_tab[] IS INITIAL.
      MESSAGE s005.
      EXIT.
    ELSE.
      IF gv_view = 'X'.
        PERFORM call_view.
      ELSE.
        CALL SCREEN 100.
      ENDIF.
    ENDIF.
  ENDIF.
