   WHERE devclass = p_devcls

    ELSE.

    ENDIF.

    SELECT SINGLE subc into i_prog-subc
      from reposrc
     where progname = i_prog-prog.
    if sy-subrc <> 0.
      continue.
    endif.

    append i_prog.

  ENDLOOP.

*  SELECT a~devclass a~obj_name b~subc
*    INTO (i_prog-devclass, i_prog-prog, i_prog-subc)
*    FROM tadir AS a INNER JOIN reposrc AS b
*      ON a~obj_name EQ b~progname
*   WHERE a~devclass = p_devcls
*     AND a~object IN ('PROG')
*     AND ( b~subc   = '1' OR b~subc  = 'M' )
*     AND a~obj_name IN p_objnam.
*    APPEND i_prog.
*
*  ENDSELECT.
*
*  SELECT a~devclass a~obj_name b~subc
*    INTO (i_prog-devclass, i_prog-prog, i_prog-subc)
*    FROM tadir AS a INNER JOIN reposrc AS b
*      ON a~obj_name EQ b~progname
*   WHERE a~devclass = p_devcls
*     AND a~object IN ('FUGR')
*     AND b~subc  = 'F'

*    APPEND i_prog.
*
*  ENDSELECT.

*ENLFDIR
