FORM GET_DATA.

  CLEAR: GT_UPLOAD, GT_UPLOAD[], GT_TAB, GT_TAB[].

  DATA : LF_RETRN TYPE CHAR1.
  DATA : LT_EXCEL_TAB  TYPE ALSMEX_TABLINE OCCURS 0 WITH HEADER LINE.
  DATA : LV_INDEX    LIKE SY-TABIX.

  FIELD-SYMBOLS : <FS> TYPE ANY.

  CALL FUNCTION 'WS_QUERY' " 파일 존재여부 체크
    EXPORTING
      FILENAME = P_FILE
      QUERY    = 'FE'
    IMPORTING
      RETURN   = LF_RETRN.      " '0' = 없다,  '1' = 존재한다

  IF LF_RETRN = '0'.
    EXIT.
  ENDIF.

  IF P_FILE NE SPACE.
*___Upload the Excel File at Excel Table
    CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
      EXPORTING
        FILENAME                = P_FILE
        I_BEGIN_COL             = GC_BEGIN_COL
        I_BEGIN_ROW             = GC_BEGIN_ROW
        I_END_COL               = GC_END_COL
        I_END_ROW               = GC_END_ROW
      TABLES
        INTERN                  = LT_EXCEL_TAB
      EXCEPTIONS
        INCONSISTENT_PARAMETERS = 1
        UPLOAD_OLE              = 2
        OTHERS                  = 3.
  ENDIF.


  LOOP AT LT_EXCEL_TAB WHERE ROW GE 2.
    MOVE : LT_EXCEL_TAB-COL TO LV_INDEX.
    ASSIGN COMPONENT LV_INDEX OF STRUCTURE GT_UPLOAD TO <FS>.
    MOVE : LT_EXCEL_TAB-VALUE TO <FS>.

    AT END OF ROW.
      PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING GT_UPLOAD-DOCNO.

      APPEND GT_UPLOAD.
      CLEAR  GT_UPLOAD.
    ENDAT.
  ENDLOOP.

  LOOP AT GT_UPLOAD.
    MOVE-CORRESPONDING GT_UPLOAD TO GT_TAB.
    APPEND GT_TAB.
  ENDLOOP.

ENDFORM. " GET_DATA
