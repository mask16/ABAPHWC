FORM POPUP_TO_CONFIRM USING P_VAL
                    CHANGING    P_ANSWER.
  DATA : L_TITLE(30),
         L_QUESTION(100).

  CLEAR : P_ANSWER, L_TITLE, L_QUESTION.

  CASE P_VAL.
    WHEN 'SAVE'.
      L_TITLE    = TEXT-A01.   "저 장
      L_QUESTION = TEXT-A02.   "변경하신 내용을 저장 하시겠습니까?
    WHEN 'DELE'.
      L_TITLE    = TEXT-A03.   "저 장
      L_QUESTION = TEXT-A04.   "선택하신 행을 삭제하시겠습니까?

  ENDCASE.

  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      TITLEBAR              = L_TITLE
      TEXT_QUESTION         = L_QUESTION
      TEXT_BUTTON_1         = '예'    " '
      ICON_BUTTON_1         = 'ICON_OKAY'
      TEXT_BUTTON_2         = '아니오'    " '
      ICON_BUTTON_2         = 'ICON_CANCEL'
      DEFAULT_BUTTON        = '2'
      DISPLAY_CANCEL_BUTTON = ''
    IMPORTING
      ANSWER                = P_ANSWER
    EXCEPTIONS
      TEXT_NOT_FOUND        = 1
      OTHERS                = 2.

ENDFORM. " POPUP_TO_CONFIRM

FORM GET_SELECTED_ROW USING PT_ROW TYPE LVC_T_ROW
                              PS_ROW TYPE LVC_S_ROW
                              P_GRID TYPE REF TO CL_GUI_ALV_GRID
                     CHANGING P_NOTSEL.

*선택된 행 읽어오기
  CALL METHOD P_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = PT_ROW.

  IF PT_ROW[] IS INITIAL.

  READ TABLE GT_TAB INDEX PS_ROW-INDEX.

ENDFORM. " GET_SELECTED_ROW

FORM GET_DATA.

  DATA : LT_TAB LIKE GT_TAB OCCURS 0 WITH HEADER LINE.
  CLEAR : GT_TAB, GT_TAB[].

  SELECT * INTO TABLE LT_TAB
    FROM ZPMT1000
    WHERE FGBN IN S_FGBN
      AND EQTYP IN S_EQTYP
      AND FCODE IN S_FCODE
      AND FTEXT IN S_FTEXT.


  LOOP AT LT_TAB.

    MOVE-CORRESPONDING LT_TAB TO GT_TAB.
    APPEND GT_TAB.

  ENDLOOP.



ENDFORM. " GET_DATA

FORM INIT .

  CLEAR : G_TITLE.
  CASE SY-TCODE.
    WHEN 'ZPMR1090'.
      CONCATENATE TEXT-G01 '(' SY-TCODE ')' INTO G_TITLE.
    WHEN 'ZPMR1091'.
      CONCATENATE TEXT-G02 '(' SY-TCODE ')' INTO G_TITLE.
    WHEN OTHERS.
      CONCATENATE SY-REPID '(' SY-TCODE ')' INTO G_TITLE.
  ENDCASE.

  SET TITLEBAR  '0100' WITH       G_TITLE.

ENDFORM. " INIT
