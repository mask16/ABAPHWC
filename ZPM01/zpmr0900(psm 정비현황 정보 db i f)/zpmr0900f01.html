FORM p002_disconn_db .
  TRY.
      EXEC SQL.
        DISCONNECT :DB_CONN
      ENDEXEC.

    CATCH cx_sy_native_sql_error INTO gv_err.
      IF gv_retry LE 3.
        CLEAR gv_err.
        ADD 1 TO gv_retry.
        WAIT UP TO 3 SECONDS.
        RETRY.

      ELSE.
        CLEAR: gv_retry.
        gv_dbmsg = gv_err->get_text( ).

        MESSAGE e000(su) WITH 'DB discon error:'  db_conn INTO gv_symsg.
*        PERFORM p003_interface_log USING '' ''.
        EXIT.
      ENDIF.
  ENDTRY.

  EXEC SQL.
    SET CONNECTION DEFAULT
  ENDEXEC.

ENDFORM. " P002_DISCONN_DB
