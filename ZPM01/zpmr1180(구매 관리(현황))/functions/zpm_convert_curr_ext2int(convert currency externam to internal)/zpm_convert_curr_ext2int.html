Description: convert CURRENCY Externam to Internal
FUNCTION ZPM_CONVERT_CURR_EXT2INT.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_EXTVAL)
*" REFERENCE(I_CUKY)
*" EXPORTING
*" REFERENCE(E_INTVAL)
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_STR TYPE CHAR50,
         L_DCPFM TYPE USR01-DCPFM.

  WRITE I_EXTVAL TO L_STR CURRENCY I_CUKY.

  SELECT SINGLE DCPFM
    INTO L_DCPFM
    FROM USR01
   WHERE BNAME EQ SY-UNAME.

  CASE L_DCPFM.
    WHEN 'X'." N,NNN.NN
      REPLACE ALL OCCURRENCES OF ',' IN  L_STR  WITH ''.
    WHEN 'Y'." NNN NNN,NN
      REPLACE ALL OCCURRENCES OF ',' IN  L_STR  WITH '.'.
    WHEN OTHERS." N.NNN,NN
      REPLACE ALL OCCURRENCES OF '.' IN  L_STR  WITH ''.
      REPLACE ALL OCCURRENCES OF ',' IN  L_STR  WITH '.'.
  ENDCASE.

  CONDENSE L_STR NO-GAPS.

  MOVE L_STR TO E_INTVAL.

ENDFUNCTION.
