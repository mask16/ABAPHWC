FORM data_select .

  DATA : lv_pltxt LIKE zpmt0010-pltxt.
  DATA : lv_tplma LIKE zpmt0010-tplma.

  CLEAR: v_pland.
  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      date      = sy-datum
      days      = 6
      months    = 0
      signum    = '+'
      years     = 0
    IMPORTING
      calc_date = v_pland.

  CLEAR: gt_data.

  SELECT *
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zpmt0050
  WHERE ( pland = sy-datum
        OR ( pland = v_pland AND wplanchk = 'X' ) )
    AND werks IN s_werks
    AND loevm = space.

  SORT gt_data BY tplnr equnr preve.

  LOOP AT gt_data INTO gs_data.
    "예방부서내역명
    SELECT SINGLE orgtx
    INTO gs_data-ppdep_tx
    FROM  t527x
    WHERE orgeh = gs_data-ppdep.

    "위치내역
    SELECT SINGLE pltxt tplma
    INTO (gs_data-pltxt, lv_tplma)
    FROM zpmt0010
    WHERE tplnr = gs_data-tplnr.

    IF sy-subrc = 0.
      SELECT SINGLE pltxt
      INTO lv_pltxt
      FROM zpmt0010
      WHERE tplnr = lv_tplma.

      CONCATENATE lv_pltxt '-' gs_data-pltxt
        INTO gs_data-pltxt SEPARATED BY space.
    ENDIF.

    "설비내역
    SELECT SINGLE eqktx
    INTO gs_data-eqktx
    FROM zpmt0020
    WHERE equnr = gs_data-equnr.

    PERFORM domvalue_text_get USING 'ZDPTYPE' gs_data-ptype
                              CHANGING gs_data-ptype_tx.

    PERFORM get_name_text USING 'WERKS' gs_data-werks
                       CHANGING gs_data-werks_tx.

    PERFORM get_name_text USING 'PERNR' gs_data-erpsn
                       CHANGING gs_data-erpsn_tx.

    PERFORM get_name_text USING 'PERNR' gs_data-aepsn
                       CHANGING gs_data-aepsn_tx.

    MODIFY gt_data FROM gs_data.
  ENDLOOP.

ENDFORM. " DATA_SELECT

FORM get_name_text USING p_fieldname
                             p_value
                    CHANGING p_text.
  DATA : l_table TYPE tabname,
         l_where TYPE string,
         l_select TYPE string.

  CASE p_fieldname.
    WHEN 'BUKRS'.
      l_table  = 'T001'.
      l_where  = 'BUKRS = p_value'.
      l_select = 'BUTXT'.
    WHEN 'WERKS'.
      l_table = 'T001W'.
      l_where = 'WERKS = p_value'.
      l_select = 'NAME1'.
    WHEN 'EKORG'.
      l_table  = 'T024E'.
      l_where  = 'EKORG = p_value'.
      l_select = 'EKOTX'.
    WHEN 'EKGRP'.
      l_table  = 'T024'.
      l_where  = 'EKGRP = p_value'.
      l_select = 'EKNAM'.
    WHEN 'PERNR'.
      l_table = 'PA0001'.
      l_where = 'PERNR = p_value AND endda = ''99991231'''.
      l_select = 'SNAME'.
    WHEN 'ORGEH'.
      l_table = 'T527X'.
      l_where
      = 'SPRSL = SY-LANGU and ORGEH = p_value AND endda = ''99991231'''.
      l_select = 'ORGTX'.
    WHEN 'LIFNR'.
      l_table = 'LFA1'.
      l_where = 'LIFNR = p_value'.
      l_select = 'NAME1'.
    WHEN 'KNTTP'.
      l_table = 'T163I'.
      l_where = 'SPRAS = SY-LANGU and KNTTP = p_value '.
      l_select = 'KNTTX'.
    WHEN 'KOSTL'.
      l_table = 'CSKT'.
      l_where
      = 'KOKRS = ''1000'' and KOSTL = p_value AND datbi = ''99991231'''.
      l_select = 'MCTXT'.
    WHEN 'SAKNR'.
      l_table   = 'SKAT'.
      l_where
      = 'SPRAS = SY-LANGU AND KTOPL = ''1000'' and SAKNR = p_value '.
      l_select  = 'TXT20'.
    WHEN 'MATKL'.
      l_table = 'T023T'.
      l_where = 'SPRAS = SY-LANGU and MATKL = p_value'.
      l_select = 'WGBEZ'.
    WHEN OTHERS.
      EXIT.
  ENDCASE.

  SELECT SINGLE (l_select)
    INTO p_text
    FROM (l_table)
   WHERE (l_where).


ENDFORM. " GET_NAME_TEXT

FORM send_mail .

  DATA: t_mail TYPE TABLE OF zpmt1060,
        s_mail TYPE zpmt1060.

  DATA: t_string TYPE TABLE OF string,
        v_string TYPE string.

  DEFINE _reference.
    if &1 is not initial.
      i_reclist-receiver = &1.
      i_reclist-rec_type = 'U'.

      case s_mail-mtype.
        when 'T'.
          i_reclist-copy = space.
        when 'C'.
          i_reclist-copy = 'X'.
      endcase.

      translate i_reclist-receiver to lower case.

      append i_reclist.
      clear i_reclist.
    endif.
  END-OF-DEFINITION.

*  SORT gt_data BY pland wplanchk.

  LOOP AT gt_data INTO gs_data.

    "Mail Title
    wa_doc_chng-obj_name = 'PMWARN'.
    IF gs_data-wplanchk = 'X'.
      IF gs_data-pland <> sy-datum.
        wa_doc_chng-obj_descr = '(외주)예방정비일 도래 안내(D-6일)'.
      ELSE.
        wa_doc_chng-obj_descr = '(외주)예방정비일 도래 안내'.
      ENDIF.

    ELSE.
      wa_doc_chng-obj_descr = '(자체)예방정비일 도래 안내'.
    ENDIF.
    wa_doc_chng-sensitivty = 'F'.


    CLEAR: t_mail.
    SELECT *
    INTO CORRESPONDING FIELDS OF TABLE t_mail
    FROM zpmt1060.

    "Main Text
    PERFORM text_build TABLES t_string.

    CLEAR: i_objtxt, i_objtxt[].
    MOVE t_string TO i_objtxt[].

    CLEAR: i_objpack, i_objpack[].
    i_objpack-transf_bin = space.
    i_objpack-head_start = 1.
    i_objpack-head_num = 0.
    i_objpack-body_start = 1.
    DESCRIBE TABLE i_objtxt LINES i_objpack-body_num.
    i_objpack-doc_type = 'RAW'.
    APPEND i_objpack.
    CLEAR i_objpack.


    "Sender Address
    CLEAR v_sender.
    v_sender = 'no_reply@sap.admin'.
    TRANSLATE v_sender TO LOWER CASE.


    "Receiver Address
    CLEAR: i_reclist, i_reclist[].
    LOOP AT t_mail INTO s_mail.
      _reference: s_mail-email.
    ENDLOOP.


    CALL FUNCTION 'SO_DOCUMENT_SEND_API1'
      EXPORTING
        document_data              = wa_doc_chng
        put_in_outbox              = 'X'
        sender_address             = v_sender
        sender_address_type        = 'INT'
        commit_work                = 'X'
      IMPORTING
        new_object_id              = v_object_id
      TABLES
        packing_list               = i_objpack
        object_header              = wa_objhead
        contents_bin               = i_objbin
        contents_txt               = i_objtxt
        receivers                  = i_reclist
      EXCEPTIONS
        too_many_receivers         = 1
        document_not_sent          = 2
        document_type_not_exist    = 3
        operation_no_authorization = 4
        parameter_error            = 5
        x_error                    = 6
        enqueue_error              = 7
        OTHERS                     = 8.

  ENDLOOP.

ENDFORM. " SEND_MAIL
