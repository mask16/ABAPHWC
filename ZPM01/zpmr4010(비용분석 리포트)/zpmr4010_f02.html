FORM HANDLE_DOUBLE_CLICK USING P_ROW-INDEX
                                   P_COLUMN
                                   P_ROW_NO.

  CASE SY-DYNNR.
    WHEN '0100'.
      READ TABLE GT_TAB INDEX P_ROW-INDEX.
    WHEN '0200'.
      READ TABLE GT_DETAIL INDEX P_ROW-INDEX.
    WHEN OTHERS.
      EXIT.
  ENDCASE.

  CASE P_COLUMN.
    WHEN 'RESULTN'.
      CHECK GT_DETAIL-RESULTN IS NOT INITIAL.
      SET PARAMETER ID 'YRN' FIELD GT_DETAIL-RESULTN.
      SET PARAMETER ID 'ZVIEW' FIELD 'X'.
      CALL TRANSACTION 'ZPMR1160' AND SKIP FIRST SCREEN.
      SET PARAMETER ID 'YRN' FIELD ''.
    WHEN 'WPLANN'.
      CHECK GT_DETAIL-WPLANN IS NOT INITIAL.
      SET PARAMETER ID 'YWP' FIELD GT_DETAIL-WPLANN.
      SET PARAMETER ID 'ZVIEW' FIELD 'X'.
      CALL TRANSACTION 'ZPMR1150' AND SKIP FIRST SCREEN.
      SET PARAMETER ID 'YWP' FIELD ''.
    WHEN 'MATNR'.

      CHECK GT_DETAIL-MATNR IS NOT INITIAL.
      SET PARAMETER ID  'MAT' FIELD GT_DETAIL-MATNR.
      SET PARAMETER ID  'WRK' FIELD GT_DETAIL-WERKS.
      SET PARAMETER ID  'MXX' FIELD 'K'.
      CALL  TRANSACTION 'MM03' AND SKIP FIRST SCREEN.
      SET PARAMETER ID  'MAT' FIELD ''.
      SET PARAMETER ID  'WRK' FIELD ''.
      SET PARAMETER ID  'MXX' FIELD ''.

    WHEN 'DOCNO' OR 'DOCYR'.

      CHECK GT_DETAIL-DOCNO IS NOT INITIAL.
      IF GT_DETAIL-DOCTY = 'MSEG'.
        SET PARAMETER ID  'MBN' FIELD GT_DETAIL-DOCNO.
        SET PARAMETER ID  'MJA' FIELD GT_DETAIL-DOCYR.
        CALL  TRANSACTION 'MB03' AND SKIP FIRST SCREEN.
        SET PARAMETER ID  'MBN' FIELD ''.
        SET PARAMETER ID  'MJA' FIELD ''.
     elseIF GT_DETAIL-DOCTY = 'RSEG'.
        SET PARAMETER ID  'RBN' FIELD GT_DETAIL-DOCNO.
        SET PARAMETER ID  'RJA' FIELD GT_DETAIL-DOCYR.
        CALL  TRANSACTION 'MIR4' AND SKIP FIRST SCREEN.
        SET PARAMETER ID  'RBN' FIELD ''.
        SET PARAMETER ID  'RJA' FIELD ''.
      ENDIF.

    WHEN OTHERS.
      CASE P_COLUMN+0(1).

        WHEN 'M'.
          PERFORM MOVE_TO_DETAIL USING P_COLUMN.
      ENDCASE.
