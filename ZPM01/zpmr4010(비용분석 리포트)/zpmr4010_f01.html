    QMDEP TYPE ZPMT0040-QMDEP,

    PREVE TYPE ZPMT0050-PREVE,
    PTYPE TYPE ZPMT0050-PTYPE,
    PPDEP TYPE ZPMT0050-PPDEP,
  END OF LT_0050.

  RANGES : R_BUDAT FOR ZPMT0170-BUDAT.
  RANGES : R_ORDWF FOR ZPMT0140-ORDWF.

  IF P_YEAR IS NOT INITIAL.

    CONCATENATE P_YEAR '0101' INTO R_BUDAT-LOW.
    CONCATENATE P_YEAR '1231' INTO R_BUDAT-HIGH.
    APPEND R_BUDAT.

         B~PREVE    A~OTYPE   C~WCENTER   C~OSTYPE  C~FAILC
         C~PROBLEM  C~CAUSE   C~ACTION    D~COSTCAT
         D~BUDAT AS ERDAT

*     AND A~ORDWF IN R_ORDWF.

      INTO TABLE LT_0040
      FROM ZPMT0040

    ELSE.

    UNASSIGN <FS_M>.

  ENDLOOP.

ENDFORM. " MAKE_GT_DATA

FORM MAKE_GT_TAB .
  CLEAR : GT_TAB, GT_TAB[].

  DATA : L_STR TYPE STRING.
  DATA : L_NUM(2) TYPE N.
  FIELD-SYMBOLS : <FS_TMP>.

  CLEAR : L_STR, L_NUM.

  SORT GT_DATA BY (G_FNAME).

  LOOP AT GT_DATA.

    MOVE-CORRESPONDING GT_DATA TO GT_TAB.
    CONCATENATE 'GT_DATA-' G_FNAME INTO L_STR.
    ASSIGN (L_STR) TO <FS_TMP>.
    GT_TAB-GUBUN = <FS_TMP>.

    IF G_FNAME = C_14.
      GT_TAB-GROUP  = GT_TAB-GUBUN+0(2).
    ENDIF.

    COLLECT GT_TAB.
    UNASSIGN <FS_TMP>.

  ENDLOOP.

  LOOP AT GT_TAB.
    CLEAR : GT_TAB-M13, L_NUM.
    DO 12 TIMES.
      L_NUM =  L_NUM + 1.
      CONCATENATE 'GT_TAB-M' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_TMP>.
      GT_TAB-M13 = GT_TAB-M13 + <FS_TMP>.
      UNASSIGN <FS_TMP>.
    ENDDO.

    PERFORM GET_GUBUN_TEXT USING G_FNAME GT_TAB-GUBUN
                        CHANGING GT_TAB-GUBUN_TX.

    MODIFY GT_TAB.

  ENDLOOP.

  DELETE GT_TAB WHERE M13 = 0.

ENDFORM. " MAKE_GT_TAB

FORM GET_GUBUN_TEXT USING P_FIELDNAME
                              P_VALUE
                     CHANGING P_TEXT.

  DATA : L_TABLE  TYPE TABNAME,
         L_WHERE  TYPE STRING,
         L_SELECT TYPE STRING,
         L_VALUE  TYPE STRING.

  CASE P_FIELDNAME.
    WHEN 'WERKS'.
      L_TABLE  = 'T001W'.
      L_WHERE  = 'WERKS = p_value'.
      L_SELECT = 'NAME1'.
    WHEN 'TPLNR'.
      L_TABLE  = 'ZPMT0010'.
      L_WHERE  = 'TPLNR = p_value'.
      L_SELECT = 'PLTXT'.
    WHEN 'EQUNR'.
      L_TABLE  = 'ZPMT0020'.
      L_WHERE  = 'EQUNR = p_value'.
      L_SELECT = 'EQKTX'.
    WHEN 'ORGEH' OR 'WPDEPT'.
      L_TABLE  = 'T527X'.
      L_WHERE  = 'SPRSL = SY-LANGU and ORGEH = p_value AND endda = ''99991231'''.
      L_SELECT = 'ORGTX'.
    WHEN 'KOSTL'.
      L_TABLE = 'CSKT'.
      L_WHERE = 'KOKRS = ''1000'' and KOSTL = p_value AND datbi = ''99991231'''.
      L_SELECT = 'MCTXT'.
    WHEN 'WPTYPE'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDWTYPE' P_VALUE
                                CHANGING P_TEXT.
      IF P_TEXT IS INITIAL.
        PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDPTYPE' P_VALUE
                                  CHANGING P_TEXT.
      ENDIF.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'OTYPE'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDOTYPE' P_VALUE
                                CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'OSTYPE'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDOSTYPE' P_VALUE
                                CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'WCENTER'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDGRP' P_VALUE
                                CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'FAILC' OR 'PROBLEM' OR 'CAUSE' OR 'ACTION'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDEQTYP' P_VALUE+0(1)
                                CHANGING L_SELECT.

      PERFORM SET_FCODE_TEXT USING P_FIELDNAME+0(1) P_VALUE+0(1) P_VALUE+1(4)
                          CHANGING P_TEXT.
      CONCATENATE '[' P_VALUE+0(1) '-' L_SELECT ']' P_VALUE+1(4) '-' P_TEXT
             INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'COSTCAT'.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDCOSTCAT' P_VALUE
                                CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN OTHERS.
      EXIT.
  ENDCASE.

  SELECT SINGLE (L_SELECT)
    INTO P_TEXT
    FROM (L_TABLE)
   WHERE (L_WHERE).

  IF P_FIELDNAME <> 'TPLNR'.
    CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
  ENDIF.

ENDFORM. " GET_GUBUN_TEXT

FORM MOVE_TO_DETAIL USING P_COLUMN.

  DATA : L_EP_LOW LIKE SY-DATUM,

  ELSE.
    CONCATENATE P_YEAR P_COLUMN+1(2) '01' INTO L_EP_LOW.

  CLEAR: LT_KEY, LT_KEY[].

  DATA : L_STR TYPE STRING.
  FIELD-SYMBOLS : <FS_TMP>.

  CLEAR : L_STR.

  LOOP AT GT_DATA WHERE ERDAT >= L_EP_LOW AND ERDAT <= L_EP_HIGH.

    CONCATENATE 'GT_DATA-' G_FNAME INTO L_STR.
    ASSIGN (L_STR) TO <FS_TMP>.

    IF GT_TAB-GUBUN = <FS_TMP>.
      MOVE-CORRESPONDING GT_DATA TO LT_KEY.
      APPEND LT_KEY.
    ENDIF.

  ENDLOOP.

  CLEAR : GT_DETAIL, GT_DETAIL[].

  IF LT_KEY[] IS NOT INITIAL.

    SELECT A~BUKRS    A~WERKS   A~DOCNO A~BUDAT A~WAERS

           MSEG     AS C
        ON B~MBLNR = C~MBLNR
       AND B~MJAHR = C~MJAHR INNER JOIN
           ZPMT0140 AS D
        ON D~WPLANN = A~WPLANN INNER JOIN
           ZPMT0090 AS E
        ON E~WPLANN = A~WPLANN
       AND E~WPITEM = A~WPITEM
       AND E~RSNUM  = C~RSNUM INNER JOIN
           MAKT AS F
        ON C~MATNR = F~MATNR
       AND F~SPRAS = SY-LANGU
       FOR ALL ENTRIES IN LT_KEY
     WHERE A~WPLANN   = LT_KEY-WPLANN
       AND A~WPITEM   = LT_KEY-WPITEM
       AND A~COSTCAT  = LT_KEY-COSTCAT
       AND A~COSTSEQ  = LT_KEY-COSTSEQ
       AND A~DOCTY    = 'MSEG'.

           A~COSTCAT  A~DOCYR   A~DOCTY

           E~RESULTN  E~WPLANN  E~WPITEM
           C~MATNR    C~MENGE AS ERFMG   F~MEINS AS ERFME

           C~SHKZG

FORM GET_SAKTO USING P_DOCNO
                         P_DOCYR
                CHANGING P_SAKNR.

  DATA : L_AWKEY TYPE BKPF-AWKEY,
         LS_BKPF TYPE BKPF.
  CLEAR: L_AWKEY, LS_BKPF.

  SELECT SINGLE BUKRS BELNR GJAHR

    FROM BKPF

    FROM BSEG
   WHERE BUKRS = LS_BKPF-BUKRS
     AND BELNR = LS_BKPF-BELNR
     AND GJAHR = LS_BKPF-GJAHR
