  DATA : LV_NOTSEL(1).
  CLEAR OK_CODE.

  CASE OK_CODE.

    WHEN 'DELE'.

      PERFORM GET_SELECTED_ROW USING GT_ROW
                                     GS_ROW
                                     G_GRID
                            CHANGING LV_NOTSEL.

      IF LV_NOTSEL = 'N'.
        MESSAGE S001 WITH '행을 선택하세요.'.
        EXIT.
      ENDIF.

      CLEAR : LV_ISOK.
      PERFORM CHECK_STATUS_FOR_CHANGE CHANGING LV_ISOK.
      IF LV_ISOK = 'N'.
        MESSAGE E001 WITH '변경할수 있는 상태가 아닙니다.'.
        EXIT.
      ENDIF.

      PERFORM DELETE_PR_ITEM USING 'A'.

* REFRESH  DATA 다시 GATHERING
      CLEAR : GT_TAB, GT_TAB[].
      PERFORM GET_DATA.


* 삭제취소
    WHEN 'UNDO'.

      CLEAR : LV_NOTSEL, GT_ROW, GT_ROW[], GS_ROW.
      PERFORM GET_SELECTED_ROW USING GT_ROW
                                     GS_ROW
                                     G_GRID
                            CHANGING LV_NOTSEL.

      IF LV_NOTSEL = 'N'.
        MESSAGE S001 WITH '행을 선택하세요.'.
        EXIT.
      ENDIF.

      CLEAR : LV_ISOK.
      PERFORM CHECK_STATUS_FOR_CHANGE CHANGING LV_ISOK.
      IF LV_ISOK = 'N'.
        MESSAGE E001 WITH '변경할수 있는 상태가 아닙니다.'.
        EXIT.
      ENDIF.


      PERFORM DELETE_PR_ITEM USING 'B'.

* REFRESH  DATA 다시 GATHERING
      CLEAR : GT_TAB, GT_TAB[].
