    RESULTN TYPE ZPMT0090-RESULTN,

  END OF LT_0090.

    SELECT MATNR MTART INTO TABLE LT_MARA
      FROM MARA FOR ALL ENTRIES IN GT_DATA
     WHERE MATNR = GT_DATA-MATNR.

    SORT LT_MARA BY MATNR.

    SELECT A~WPLANN A~WPITEM A~OSTYPE A~RESULTN B~ORDWF

      FROM ZPMT0090 AS A INNER JOIN ZPMT0140 AS B

       AND A~WPITEM = GT_DATA-WPITEM.

                                BINARY SEARCH.

FORM GET_GT_DATA .
  CLEAR : GT_DATA, GT_DATA[].

  DATA : LT_DATA LIKE GT_DATA OCCURS 0 WITH HEADER LINE.

  SELECT * INTO CORRESPONDING FIELDS OF TABLE LT_DATA
    FROM RBKP AS A INNER JOIN RSEG AS B
      ON A~BELNR = B~BELNR
     AND A~GJAHR = B~GJAHR INNER JOIN EKPO AS C
      ON B~EBELN = C~EBELN
     AND B~EBELP = C~EBELP INNER JOIN EBAN AS D
      ON C~BANFN = D~BANFN
     AND C~BNFPO = D~BNFPO
   WHERE A~CPUDT IN S_CPUDT
     AND B~WERKS IN S_WERKS
     AND D~BADAT IN S_BADAT
     AND D~BSART = 'PM'
     AND D~EKGRP IN ('116', '132').

    BANFN TYPE EBAN-BANFN,

  DATA : BEGIN OF LT_MARA OCCURS 0,
    MATNR TYPE MARA-MATNR,

  END OF LT_MARA.

    CLEAR : LT_EBKN, LT_EBKN[].

     WHERE MATNR = LT_DATA-MATNR.

    SORT LT_MARA BY MATNR.

  ENDIF.

      CONTINUE.
    ENDIF.

*    ENDIF.

    APPEND GT_DATA.

  ENDLOOP.

  DATA : L_INDEX TYPE SY-TABIX.
  CLEAR: LT_0190, LT_0190[], L_INDEX.
