FUNCTION ZPM_ATTACH_CHECK.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_DOCTY)
*" REFERENCE(I_DOCNO)
*" EXPORTING
*" REFERENCE(E_COUNT) TYPE I
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_DOCTY TYPE ZPMT0150-DOCTY,
         L_DOCNO TYPE ZPMT0150-DOCNO,
         L_CNT   TYPE I.
  CLEAR: L_DOCTY, L_DOCNO, L_CNT.

  MOVE I_DOCTY TO L_DOCTY.
  MOVE I_DOCNO TO L_DOCNO.

  PERFORM CONVERSION_EXIT_ALPHA_INPUT  CHANGING L_DOCNO.

  SELECT COUNT( DISTINCT DOCNO )
    INTO L_CNT
    FROM ZPMT0150
   WHERE DOCTY = L_DOCTY
     AND DOCNO = L_DOCNO.

  IF SY-SUBRC = 0.
    E_COUNT = L_CNT.
  ELSE.
    E_COUNT = 0.
  ENDIF.

ENDFUNCTION.
