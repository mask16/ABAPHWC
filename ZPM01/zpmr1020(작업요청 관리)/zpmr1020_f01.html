FORM popup_to_confirm USING p_val
                    CHANGING    p_answer.
  DATA : l_title(30),
         l_question(100).

  CLEAR : p_answer, l_title, l_question.

  CASE p_val.
    WHEN 'SAVE'.
      l_title    = text-a01.   "저 장
      l_question = text-a02.   "변경하신 내용을 저장 하시겠습니까?
    WHEN 'ALLO'.
      l_title    = text-a03.   "승 인
      l_question = text-a04.   "선택한 작업요청에 대해 승인하시겠습니까?
    WHEN 'REJE'.
      l_title    = text-a05.   "반 려
      l_question = text-a06.   "선택한 작업요청에 대해 반려하시겠습니까?
    WHEN 'ORDE'.
      l_title    = text-a07.   "작업지시
      l_question = text-a08.   "선택한 작업요청에 대해 작업지시 하시겠습니까?
    WHEN 'EXIT'.
      l_title    = text-a09.   "종 료
      l_question = text-a10.   "먼저 문서를 저장하시겠습니까?
    WHEN 'DELE'.
      l_title    = text-a11.   "삭 제
      l_question = text-a12.   "선택한 작업요청에 대해 삭제하시겠습니까?
  ENDCASE.


  IF p_val = 'EXIT'.
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
      EXPORTING
        defaultoption = 'N'
        titel         = l_title
        textline1     = l_question
      IMPORTING
        answer        = p_answer.
  ELSE.
    CALL FUNCTION 'POPUP_TO_CONFIRM'
      EXPORTING
        titlebar              = l_title
        text_question         = l_question
        text_button_1         = '예'    " '
        icon_button_1         = 'ICON_OKAY'
        text_button_2         = '아니오'    " '
        icon_button_2         = 'ICON_CANCEL'
        default_button        = '2'
        display_cancel_button = ''
      IMPORTING
        answer                = p_answer
      EXCEPTIONS
        text_not_found        = 1
        OTHERS                = 2.
  ENDIF.

ENDFORM. " POPUP_TO_CONFIRM
