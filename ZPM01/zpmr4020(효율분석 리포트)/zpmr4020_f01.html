FORM MAKE_GT_DATA .

  DATA : BEGIN OF LT_KEY OCCURS 0,
    PREVE   TYPE ZPMT0090-PREVE,
    PTYPE   TYPE ZPMT0050-PTYPE,
    QMNUM   TYPE ZPMT0090-QMNUM,
    WTYPE   TYPE ZPMT0040-WTYPE,
    ERDAT   TYPE ZPMT0090-ERDAT,
    ORDWF   TYPE ZPMT0140-ORDWF,
  END OF LT_KEY,
  LT_PRE LIKE LT_KEY OCCURS 0 WITH HEADER LINE.

  RANGES : R_BUDAT FOR ZPMT0170-BUDAT.

  CLEAR: LT_KEY, LT_KEY[], LT_PRE, LT_PRE[].

  IF P_YEAR IS NOT INITIAL.
    R_BUDAT = 'IBT'.
    CONCATENATE P_YEAR '0101' INTO R_BUDAT-LOW.
    CONCATENATE P_YEAR '1231' INTO R_BUDAT-HIGH.
    APPEND R_BUDAT.
  ENDIF.

  SELECT PREVE PTYPE PLAND AS ERDAT
    APPENDING CORRESPONDING FIELDS OF TABLE LT_PRE
    FROM ZPMT0050
   WHERE BUKRS IN S_BUKRS
     AND WERKS IN S_WERKS
     AND PLAND IN R_BUDAT
     AND LOEVM = ''              "삭제지시자
     AND WPLANCHK = 'X'.         "정비계획 포함

  SELECT A~PREVE A~PTYPE B~PLNDT AS ERDAT C~ORDWF
    APPENDING CORRESPONDING FIELDS OF TABLE LT_KEY
    FROM ZPMT0050 AS A INNER JOIN
         ZPMT0090 AS B
      ON A~PREVE  = B~PREVE INNER JOIN
         ZPMT0140 AS C
      ON B~WPLANN = C~WPLANN
   WHERE A~BUKRS IN S_BUKRS
     AND A~WERKS IN S_WERKS
     AND B~PLNDT IN R_BUDAT.

  LOOP AT LT_PRE.
    READ TABLE LT_KEY WITH KEY PREVE = LT_PRE-PREVE
                               PTYPE = LT_PRE-PTYPE
                               ERDAT = LT_PRE-ERDAT.
    IF SY-SUBRC <> 0.
      MOVE-CORRESPONDING LT_PRE TO LT_KEY.
      APPEND LT_KEY.
    ENDIF.
  ENDLOOP.

  SELECT A~QMNUM A~WTYPE B~QMDAT AS ERDAT C~ORDWF
    APPENDING CORRESPONDING FIELDS OF TABLE LT_KEY
    FROM ZPMT0040 AS A INNER JOIN
         ZPMT0090 AS B
      ON A~QMNUM  = B~QMNUM INNER JOIN
         ZPMT0140 AS C
      ON B~WPLANN = C~WPLANN
   WHERE A~BUKRS IN S_BUKRS
     AND A~WERKS IN S_WERKS
     AND B~QMDAT IN R_BUDAT.

  FIELD-SYMBOLS : <FS_M>.
  DATA : L_FNAME TYPE STRING.

  SORT LT_KEY.

  LOOP AT LT_KEY.

    CLEAR : GT_DATA.
    MOVE-CORRESPONDING LT_KEY TO GT_DATA.

    CONCATENATE 'GT_DATA-P' LT_KEY-ERDAT+4(2) INTO L_FNAME.
    ASSIGN (L_FNAME) TO <FS_M>.
    <FS_M> = 1.

    CONCATENATE 'GT_DATA-A' LT_KEY-ERDAT+4(2) INTO L_FNAME.
    ASSIGN (L_FNAME) TO <FS_M>.

    IF LT_KEY-ORDWF = 'Z'.
      <FS_M> = 1.
    ELSE.
      <FS_M> = 0.
    ENDIF.

    COLLECT : GT_DATA.

    CONCATENATE P_YEAR LT_KEY-ERDAT+4(2) '01' INTO GT_DATA-ERDAT.
    IF LT_KEY-PREVE IS NOT INITIAL.
      GT_DATA-PTYPE = 'ZZ'.
    ELSE.
      GT_DATA-WTYPE = 'ZZ'.
    ENDIF.
    COLLECT : GT_DATA.
    UNASSIGN <FS_M>.

  ENDLOOP.

ENDFORM. " MAKE_GT_DATA

FORM MAKE_GT_TAB .
  CLEAR : GT_TAB, GT_TAB[].

  DATA : L_STR TYPE STRING.
  DATA : L_NUM(2) TYPE N.
  FIELD-SYMBOLS : <FS_TMP>, <FS_P>, <FS_A>, <FS_MP>, <FS_MA>, <FS_M>.

  CLEAR : L_STR, L_NUM.

  SORT GT_DATA BY (G_FNAME).

  LOOP AT GT_DATA.

    MOVE-CORRESPONDING GT_DATA TO GT_TAB.
    CONCATENATE 'GT_DATA-' G_FNAME INTO L_STR.
    ASSIGN (L_STR) TO <FS_TMP>.
    GT_TAB-GUBUN = <FS_TMP>.

    IF GT_TAB-GUBUN IS INITIAL.
      CONTINUE.
    ENDIF.

    CLEAR : GT_TAB-P13, GT_TAB-A13, L_NUM.
    DO 12 TIMES.
      L_NUM =  L_NUM + 1.
      CONCATENATE 'GT_TAB-P' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_MP>.

      CONCATENATE 'GT_DATA-P' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_P>.

      <FS_MP> = <FS_P>.
      GT_TAB-P13 = GT_TAB-P13 + <FS_P>.

      CONCATENATE 'GT_DATA-A' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_A>.

      CONCATENATE 'GT_TAB-A' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_MA>.

      <FS_MA> = <FS_A>.
      GT_TAB-A13 = GT_TAB-A13 + <FS_A>.

    ENDDO.

    COLLECT GT_TAB.
    UNASSIGN <FS_TMP>.

  ENDLOOP.

  IF G_FNAME = C_02.
    DATA : LS_TAB LIKE GT_TAB.
    CLEAR: LS_TAB.
    READ TABLE GT_TAB INTO LS_TAB WITH KEY GUBUN = 'ZZ'.

    LOOP AT GT_TAB.

      CLEAR : L_NUM.
      DO 13 TIMES.

        L_NUM =  L_NUM + 1.

        CONCATENATE 'GT_TAB-P' L_NUM INTO L_STR.
        ASSIGN (L_STR) TO <FS_MP>.

        CONCATENATE 'GT_TAB-A' L_NUM INTO L_STR.
        ASSIGN (L_STR) TO <FS_MA>.
        <FS_MA> = <FS_MP>.

        CONCATENATE 'LS_TAB-P' L_NUM INTO L_STR.
        ASSIGN (L_STR) TO <FS_P>.
        <FS_MP> = <FS_P>.

        IF GT_TAB-GUBUN = 'ZZ'.
          CONCATENATE 'GT_TAB-A' L_NUM INTO L_STR.
          ASSIGN (L_STR) TO <FS_MA>.
          <FS_MA> = <FS_P>.
        ENDIF.

      ENDDO.

      MODIFY GT_TAB.

    ENDLOOP.

  ENDIF.

  LOOP AT GT_TAB.
    CLEAR : L_NUM.
    DO 13 TIMES.

      L_NUM =  L_NUM + 1.
      CONCATENATE 'GT_TAB-M' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_M>.

      CONCATENATE 'GT_TAB-P' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_MP>.

      CONCATENATE 'GT_TAB-A' L_NUM INTO L_STR.
      ASSIGN (L_STR) TO <FS_MA>.

      <FS_M> = <FS_MA>.
*      <FS_M> = <FS_MP>.
      PERFORM BUILD_PERCENTAGE_FORMAT USING <FS_M> <FS_MP>.

    ENDDO.

    PERFORM GET_GUBUN_TEXT USING G_FNAME GT_TAB-GUBUN
                        CHANGING GT_TAB-GUBUN_TX.
    IF GT_TAB-GUBUN = 'ZZ'.
      GT_TAB-LINECOLOR = 'C310'.
    ENDIF.

    MODIFY GT_TAB.

  ENDLOOP.

ENDFORM. " MAKE_GT_TAB

FORM GET_GUBUN_TEXT USING P_FIELDNAME
                              P_VALUE
                     CHANGING P_TEXT.

  DATA : L_TABLE  TYPE TABNAME,
         L_WHERE  TYPE STRING,
         L_SELECT TYPE STRING,
         L_VALUE  TYPE STRING.

  CASE P_FIELDNAME.
    WHEN 'WTYPE'.
      IF P_VALUE = 'ZZ'.
        P_TEXT = 'Total'.
        EXIT.
      ENDIF.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDWTYPE' P_VALUE
                                CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN 'PTYPE'.
      IF P_VALUE = 'ZZ'.
        P_TEXT = 'Total'.
        EXIT.
      ENDIF.
      PERFORM DD_DOMVALUE_TEXT_GET USING 'ZDPTYPE' P_VALUE
                                  CHANGING P_TEXT.
      CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
      EXIT.
    WHEN OTHERS.
      EXIT.
  ENDCASE.

  SELECT SINGLE (L_SELECT)
    INTO P_TEXT
    FROM (L_TABLE)
   WHERE (L_WHERE).

  IF P_FIELDNAME <> 'TPLNR'.
    CONCATENATE P_VALUE '-' P_TEXT INTO P_TEXT SEPARATED BY SPACE.
  ENDIF.

ENDFORM. " GET_GUBUN_TEXT

FORM GET_SAKTO USING P_DOCNO
                         P_DOCYR
                CHANGING P_SAKNR.

  DATA : L_AWKEY TYPE BKPF-AWKEY,
         LS_BKPF TYPE BKPF.
  CLEAR: L_AWKEY, LS_BKPF.

  SELECT SINGLE BUKRS BELNR GJAHR
    INTO CORRESPONDING FIELDS OF LS_BKPF
    FROM BKPF

  SELECT SINGLE SAKNR

    FROM BSEG
   WHERE BUKRS = LS_BKPF-BUKRS
     AND BELNR = LS_BKPF-BELNR

    'S_WCENTR-LOW' ' ',
    'S_QMDEP-LOW' ' ',
    'S_QMNAM-LOW' ' ',
    'S_QMNAM-LOW' ' '.

         L_EP_HIGH LIKE SY-DATUM.

    CONCATENATE P_YEAR '1231' INTO L_EP_HIGH.
  ELSE.
    CONCATENATE P_YEAR P_COLUMN+1(2) '01' INTO L_EP_LOW.
    CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
      EXPORTING
        DAY_IN            = L_EP_LOW
      IMPORTING
        LAST_DAY_OF_MONTH = L_EP_HIGH.
  ENDIF.

  CASE G_FNAME.
    WHEN 'PTYPE'.
      IF GT_TAB-GUBUN <> 'ZZ'.

      IF GT_TAB-GUBUN <> 'ZZ'.
        PERFORM P_FIELD  USING 'S_WTYPE-LOW'  GT_TAB-GUBUN.
      ENDIF.

      PERFORM P_FIELD  USING 'S_PRDAT-HIGH' ''.
      PERFORM P_FIELD  USING 'S_QMDAT-LOW'  L_EP_LOW.
      PERFORM P_FIELD  USING 'S_QMDAT-HIGH' L_EP_HIGH.
  ENDCASE.

  PERFORM P_FIELD  USING : 'BDC_OKCODE'           '/EE'.

  CALL TRANSACTION 'ZPMR1150' USING         BDCDATA
                              OPTIONS FROM  I_OPTIONS
                              MESSAGES INTO MESSTAB.

ENDFORM. " MOVE_TO_ZPMR1150
