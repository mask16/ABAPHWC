FORM POPUP_TO_CONFIRM USING P_VAL
                    CHANGING    P_ANSWER.
  DATA : L_TITLE(30),
         L_QUESTION(100).

  CLEAR : P_ANSWER, L_TITLE, L_QUESTION.

  CASE P_VAL.
    WHEN 'SAVE'.
      L_TITLE    = TEXT-A01.   "저 장
      L_QUESTION = TEXT-A02.   "변경하신 내용을 저장 하시겠습니까?
    WHEN 'DELE'.
      L_TITLE    = TEXT-A03.   "저 장
      L_QUESTION = TEXT-A04.   "선택하신 행을 삭제하시겠습니까?
    WHEN 'EXIT'.
      L_TITLE    = TEXT-A05.   "종 료
      L_QUESTION = TEXT-A06.   "먼저 문서를 저장하시겠습니까?
    WHEN 'RSET'.
      L_TITLE    = TEXT-A07.   "패스워드초기화
      L_QUESTION = TEXT-A08.   "패스워드를 초기화하시겠습니까?
  ENDCASE.

  IF P_VAL = 'EXIT'.
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
      EXPORTING
        DEFAULTOPTION = 'N'
        TITEL         = L_TITLE

    CALL FUNCTION 'POPUP_TO_CONFIRM'
      EXPORTING
        TITLEBAR              = L_TITLE
        TEXT_QUESTION         = L_QUESTION

FORM GET_DATA.

  DATA : LT_TAB LIKE GT_TAB OCCURS 0 WITH HEADER LINE.

  CLEAR : GT_TAB, GT_TAB[].

  SELECT * INTO TABLE LT_TAB
    FROM ZPMT1020
   WHERE USERID IN S_USERID
     AND ROLEID IN S_ROLEID.

  LOOP AT LT_TAB.
    MOVE-CORRESPONDING LT_TAB TO GT_TAB.

    PERFORM GET_ROLE_TEXT USING GT_TAB-ROLEID
                       CHANGING GT_TAB-ROLETX.

    PERFORM GET_USER_TEXT USING GT_TAB-USERID
                       CHANGING GT_TAB-USERNM.

    APPEND GT_TAB.

  ENDLOOP.

ENDFORM. " GET_DATA
