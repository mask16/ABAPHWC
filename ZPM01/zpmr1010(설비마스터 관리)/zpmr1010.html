INITIALIZATION.

* ZPMR1000 설비관리팀 작업현황 프로그램에서 T-CODE 호출할때 값설정
  GET PARAMETER ID 'ZCR'   FIELD GV_TNEW.
  GET PARAMETER ID 'ZTP'   FIELD GV_TTPLNR.
  GET PARAMETER ID 'ZOT'   FIELD GV_OTHER.
  GET PARAMETER ID 'ZVIEW' FIELD GV_VIEW.

* PARAMETER 값 CLEAR
  SET PARAMETER ID 'ZCR' FIELD ''.
  SET PARAMETER ID 'ZTP' FIELD ''.
  SET PARAMETER ID 'ZOT' FIELD ''.
  SET PARAMETER ID 'ZVIEW' FIELD ''.

  SET TITLEBAR  '0100'.
*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN OUTPUT
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& START-OF-SELECTION
*&---------------------------------------------------------------------*
START-OF-SELECTION.
  CLEAR : GT_TAB, GT_TAB[], GV_COL_POS.

* 생성
  IF GV_TNEW = 'Y'.

    CLEAR : GS_TAB.
    GS_TAB-TPLNR = GV_TTPLNR.

* 기능위치 내역
    SELECT SINGLE BUKRS WERKS PLTXT
      INTO (GS_TAB-bukrs, GS_TAB-werks, GV_PLTXT)
      FROM ZPMT0010
      WHERE TPLNR = GS_TAB-TPLNR.

    CALL SCREEN 200.

* 변경,조회
  ELSE.
* DATA GATHERING
    PERFORM GET_DATA.

    IF GT_TAB[] IS INITIAL.
      MESSAGE S005.
      EXIT.
    ELSE.
      IF GV_VIEW = 'X'.
        PERFORM CALL_VIEW.
      ELSE.
        CALL SCREEN 100.
      ENDIF.
    ENDIF.
  ENDIF.
