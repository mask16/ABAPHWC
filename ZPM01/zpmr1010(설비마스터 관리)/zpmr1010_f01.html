  CLEAR: LT_RESULTN, LT_RESULTN[], LT_0060, LT_0060[].

  SELECT * APPENDING TABLE LT_0060
    FROM ZPMT0060
   WHERE EQUNR = GS_TAB-EQUNR.

  SELECT RESULTN
    INTO TABLE LT_RESULTN
    FROM ZPMT0110 AS A INNER JOIN
         ZPMT0090 AS B
      ON A~WPLANN = B~WPLANN
     AND A~WPITEM = B~WPITEM
   WHERE A~EQUNR = GS_TAB-EQUNR.

  IF LT_RESULTN[] IS NOT INITIAL.
    SELECT * APPENDING TABLE LT_0060
      FROM ZPMT0060
       FOR ALL ENTRIES IN LT_RESULTN
     WHERE RESULTN = LT_RESULTN-RESULTN.
  ENDIF.

  SORT LT_0060 BY RESULTN.
  DELETE ADJACENT DUPLICATES FROM LT_0060.

  "->>MAPPING
  DATA : BEGIN OF LT_QMNUM OCCURS 0,
    QMNUM TYPE ZPMT0040-QMNUM,
    QMTXT TYPE ZPMT0040-QMTXT,
  END OF LT_QMNUM,

  END OF LT_WPLANN,

    WPITEM TYPE ZPMT0090-WPITEM,
    LIFNR  TYPE LFA1-LIFNR,
  END OF LT_LIFNR.
  DATA : LT_T0170 TYPE ZPMT0170 OCCURS 0 WITH HEADER LINE.

  CLEAR: LT_QMNUM, LT_QMNUM[], LT_WPLANN, LT_WPLANN[], LT_T0170 , LT_T0170[],
  LT_LIFNR, LT_LIFNR[].

  IF LT_0060[] IS NOT INITIAL.

    SELECT QMNUM QMTXT
      INTO TABLE LT_QMNUM
      FROM ZPMT0040
       FOR ALL ENTRIES IN LT_0060
     WHERE QMNUM = LT_0060-QMNUM.

    SELECT WPLANN PMTXT
      INTO TABLE LT_WPLANN
      FROM ZPMT0140
       FOR ALL ENTRIES IN LT_0060
     WHERE WPLANN = LT_0060-WPLANN.

    SELECT A~WPLANN A~WPITEM C~LIFNR
      INTO TABLE LT_LIFNR
      FROM ZPMT0090 AS A INNER JOIN
           EBAN     AS B
        ON A~BANFN = B~BANFN INNER JOIN
*       AND B~BNFPO = '010' INNER JOIN

       FOR ALL ENTRIES IN LT_0060
     WHERE A~WPLANN = LT_0060-WPLANN
       AND A~WPITEM = LT_0060-WPITEM.

    SELECT *
      INTO TABLE LT_T0170
      FROM ZPMT0170
       FOR ALL ENTRIES IN LT_0060
     WHERE WPLANN = LT_0060-WPLANN
       AND WPITEM = LT_0060-WPITEM.

    SORT LT_QMNUM  BY QMNUM.
    SORT LT_WPLANN BY WPLANN.
    SORT LT_T0170  BY WPLANN WPITEM.
    SORT LT_LIFNR  BY WPLANN WPITEM.
    DELETE ADJACENT DUPLICATES FROM LT_LIFNR.
  ENDIF.

  LOOP AT LT_0060.
    CLEAR GT_ZPMT0060.
    MOVE-CORRESPONDING LT_0060 TO GT_ZPMT0060.

    READ TABLE LT_QMNUM WITH KEY QMNUM = GT_ZPMT0060-QMNUM BINARY SEARCH.

      GT_ZPMT0060-PMTXT = LT_WPLANN-PMTXT.
    ENDIF.

                                 WPITEM = GT_ZPMT0060-WPITEM  BINARY SEARCH.
