FUNCTION ZPM_GET_STATUS_HISTORY.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_QMNUM) TYPE ZEQMNUM OPTIONAL
*" REFERENCE(I_WPLANN) TYPE ZEWPLANN OPTIONAL
*" EXPORTING
*" REFERENCE(E_HIST) TYPE STRING
*"----------------------------------------------------------------------
* Global data declarations

  data : l_qmnum  type zpmt0040-qmnum,
         l_wplann type zpmt0040-wplann,
         l_str type string.

  clear: l_qmnum, l_wplann.

  l_qmnum  = I_QMNUM.
  l_wplann = I_WPLANN.

  check l_qmnum is not initial or l_wplann is not initial.

  perform make_status_string using 'ZPMT1040' l_qmnum
                          changing l_str.
  if l_str is initial.
    perform make_status_string using 'ZPMT1050' l_wplann
                            changing l_str.
  endif.

  e_hist = l_str.

ENDFUNCTION.
