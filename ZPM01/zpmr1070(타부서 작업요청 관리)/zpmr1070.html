INITIALIZATION.
  GET PARAMETER ID 'ZCR' FIELD GV_TNEW.

  GET PARAMETER ID 'ZEQ' FIELD GV_TEQUNR.

  SET PARAMETER ID 'ZCR' FIELD ''.
  SET PARAMETER ID 'ZTP' FIELD ''.
  SET PARAMETER ID 'ZEQ' FIELD ''.
  SET PARAMETER ID 'ZON' FIELD ''.

*  CASE SY-UNAME.
**    WHEN 'PPUSER01'. S_WERKS-LOW = '2200'. APPEND S_WERKS.  "합판
**    WHEN 'PPUSER02'. S_WERKS-LOW = '2100'. APPEND S_WERKS.  "마루
**    WHEN 'PPUSER03'. S_WERKS-LOW = '3100'. APPEND S_WERKS.  "보드
*    WHEN OTHERS.
*      S_QMNAM-LOW = SY-UNAME.
*      APPEND S_QMNAM.
*  ENDCASE.

  S_QMDAT-LOW = SY-DATUM - 15.
  S_QMDAT-HIGH = SY-DATUM.
  APPEND S_QMDAT.

  SET TITLEBAR  '0100'.
*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN OUTPUT
*&---------------------------------------------------------------------*
AT SELECTION-SCREEN OUTPUT.
  PERFORM SCREEN_MODIFY.

*&---------------------------------------------------------------------*
*& START-OF-SELECTION
*&---------------------------------------------------------------------*
START-OF-SELECTION.
  CLEAR : GT_TAB, GT_TAB[], GV_COL_POS.

  IF GV_TNEW = 'Y'.

    CLEAR : GS_TAB.
    GS_TAB-TPLNR = GV_TTPLNR.
    GS_TAB-EQUNR = GV_TEQUNR.

* 기능위치 내역
    SELECT SINGLE BUKRS WERKS PLTXT
      INTO (GS_TAB-bukrs, GS_TAB-werks, GS_TAB-PLTXT)
      FROM ZPMT0010
      WHERE TPLNR = GS_TAB-TPLNR.

* 설비내역
    SELECT SINGLE EQKTX INTO GS_TAB-EQKTX
      FROM ZPMT0020
      WHERE EQUNR = GS_TAB-EQUNR.

    GS_TAB-QMTXT = GS_TAB-EQKTX.

    GS_TAB-QMDAT = SY-DATUM.
    GS_TAB-MZEIT = SY-UZEIT.

    GS_TAB-AUSVN = SY-DATUM.
    GS_TAB-AUZTV = SY-UZEIT.
    GS_TAB-AUSBS = SY-DATUM.
    GS_TAB-AUZTB = SY-UZEIT.

    GS_TAB-STRMN = SY-DATUM.
    GS_TAB-STRUR = SY-UZEIT.
    GS_TAB-LTRMN = SY-DATUM.
    GS_TAB-LTRUR = SY-UZEIT.

    CALL SCREEN 200.

  ELSE.

* DATA GATHERING
    PERFORM GET_DATA.

    IF GT_TAB[] IS INITIAL.
      MESSAGE S005.
      EXIT.
    ELSE.
      CALL SCREEN 100.
    ENDIF.
  ENDIF.
