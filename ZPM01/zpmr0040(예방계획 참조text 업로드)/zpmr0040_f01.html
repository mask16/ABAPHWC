FORM MOVE_TEXT TABLES PT_TEXT STRUCTURE RSSOURCE
                USING    P_TEXT.
  DATA : L_INPUT TYPE F,
         L_TIMES TYPE I,
         L_LIMIT TYPE I VALUE 70,
         L_CNT   TYPE I,
         L_TEXT  LIKE GT_TAB-ZTEXT.

  CLEAR: L_CNT, L_TIMES, L_CNT, L_TEXT.

  L_TEXT  = P_TEXT.
  L_INPUT = STRLEN( L_TEXT ) / L_LIMIT.

  CALL FUNCTION 'FIMA_NUMERICAL_VALUE_ROUND'
    EXPORTING
      I_RTYPE     = '+'
      I_RUNIT     = '1'
      I_VALUE     = L_INPUT
    IMPORTING
      E_VALUE_RND = L_TIMES.

  DO L_TIMES TIMES.
    L_CNT = L_CNT + 1.
    CALL FUNCTION 'CUT_2BYTES_STRINGS'
      EXPORTING
        I_STR = L_TEXT
        I_LEN = L_LIMIT
      IMPORTING
        O_STR = PT_TEXT-LINE.
    APPEND PT_TEXT.

    IF L_CNT < L_TIMES.
      SHIFT L_TEXT BY L_LIMIT PLACES.
    ENDIF.
  ENDDO.

ENDFORM. " MOVE_TEXT
