FORM get_data.

         END   OF lt_equnr,
         BEGIN OF lt_doktl OCCURS 0,
           docno TYPE zpmt0150-docno,
           doktl TYPE zpmt0150-doktl,
         END   OF lt_doktl,
         BEGIN OF lt_matnr OCCURS 0,
          wplann TYPE zpmt0070-wplann,
          wpitem TYPE zpmt0070-wpitem,
          matnr  TYPE zpmt0070-matnr,
          quant  TYPE zpmt0070-quant,
          erfmg  TYPE zpmt0070-erfmg,
         END   OF lt_matnr.
  DATA : l_cnt TYPE numc10.
  CLEAR: gt_tab, gt_tab[], lt_tplnr, lt_tplnr[],
         lt_equnr, lt_equnr[], lt_doktl, lt_doktl[],
         lt_matnr, lt_matnr[].

  CLEAR: l_cnt.

  IF s_self[] IS NOT INITIAL.
    IF s_self-low = 'X'.
      r_othyn = 'EEQ'.
      r_othyn-low = 'X'.
      APPEND r_othyn.
    ELSE.
      r_othyn = 'IEQ'.
      r_othyn-low = 'X'.
      APPEND r_othyn.
    ENDIF.
  ENDIF.

  PERFORM get_range_status.
  IF gv_4000 = 'X'.
    PERFORM get_zpmt0060.
  ELSE.
    PERFORM get_zpmt0090.
  ENDIF.

  PERFORM get_zpmt0170.

  IF gt_tab[] IS NOT INITIAL.

    SELECT tplnr pltxt
      INTO TABLE lt_tplnr
      FROM zpmt0010
       FOR ALL ENTRIES IN gt_tab
     WHERE tplnr = gt_tab-tplnr.

    SORT lt_tplnr BY tplnr.
    DELETE ADJACENT DUPLICATES FROM lt_tplnr.

    SELECT equnr eqktx
      INTO TABLE lt_equnr
      FROM zpmt0020
       FOR ALL ENTRIES IN gt_tab
     WHERE equnr = gt_tab-equnr.

    LOOP AT lt_equnr.
      lt_equnr-docno = lt_equnr-equnr.
      PERFORM conversion_exit_alpha_input CHANGING lt_equnr-docno.
      MODIFY lt_equnr.
    ENDLOOP.

    SORT lt_equnr BY equnr.
    DELETE ADJACENT DUPLICATES FROM lt_equnr.

    SELECT wplann wpitem matnr quant erfmg
      INTO TABLE lt_matnr
      FROM zpmt0070
       FOR ALL ENTRIES IN gt_tab
     WHERE wplann = gt_tab-wplann

    SELECT docno doktl
      INTO TABLE lt_doktl
      FROM zpmt0150
       FOR ALL ENTRIES IN lt_equnr
     WHERE docty = 'E'
       AND docno = lt_equnr-docno.
    LOOP AT lt_equnr.
      CLEAR : l_cnt.
      LOOP AT lt_doktl WHERE docno = lt_equnr-docno.
        l_cnt = l_cnt + 1.
      ENDLOOP.
      MOVE l_cnt TO lt_equnr-docct.
      MODIFY lt_equnr.
    ENDLOOP.
  ENDIF.

  LOOP AT gt_tab.

    READ TABLE lt_tplnr WITH KEY tplnr = gt_tab-tplnr
                                 BINARY SEARCH.
    IF sy-subrc = 0.
      gt_tab-pltxt = lt_tplnr-pltxt.
    ENDIF.

    READ TABLE lt_equnr WITH KEY equnr = gt_tab-equnr
                                 BINARY SEARCH.
    IF sy-subrc = 0.
      gt_tab-eqktx = lt_equnr-eqktx.
      gt_tab-docct = lt_equnr-docct.
    ENDIF.

    CLEAR gt_tab-matyn.
    LOOP AT lt_matnr WHERE wplann = gt_tab-wplann
                       AND wpitem = gt_tab-wpitem.
      IF gt_tab-resultn IS INITIAL.
        IF lt_matnr-quant <> 0.
          gt_tab-matyn = 'X'.
          EXIT.
        ENDIF.
      ELSE.
        IF lt_matnr-erfmg <> 0.
          gt_tab-matyn = 'X'.
          EXIT.
        ENDIF.
      ENDIF.
    ENDLOOP.

    CLEAR: gt_tab-amount.
    LOOP AT gt_pmt0170 WHERE wplann = gt_tab-wplann
                          AND wpitem = gt_tab-wpitem.
      gt_tab-amount = gt_tab-amount + gt_pmt0170-price.
      gt_tab-waers  = 'KRW'.
    ENDLOOP.

    PERFORM dd_domvalue_text_get USING 'ZDGRP' gt_tab-wcenter
                              CHANGING gt_tab-wcenter_tx.

    PERFORM dd_domvalue_text_get USING 'ZDOTYPE' gt_tab-otype
                              CHANGING gt_tab-otype_tx.

    PERFORM dd_domvalue_text_get USING 'ZDWTYPE' gt_tab-wtype
                              CHANGING gt_tab-wtype_tx.

    PERFORM get_name_text USING 'WERKS' gt_tab-werks

    PERFORM get_name_text USING 'PERNR' gt_tab-qmnam

FORM set_work_flow USING p_wplann p_reqwf.

  DATA : BEGIN OF lt_40_key OCCURS 0,
          qmnum TYPE zpmt0040-qmnum,
         END   OF lt_40_key,
         BEGIN OF lt_50_key OCCURS 0,
          preve TYPE zpmt0050-preve,
          wendd TYPE zpmt0060-wendd,
          ptype TYPE zpmt0060-ptype,
         END   OF lt_50_key,
         BEGIN OF lt_0050 OCCURS 0,
          preve   TYPE zpmt0050-preve,
          workd   TYPE zpmt0050-workd,
          period  TYPE zpmt0050-period,
          periodu TYPE zpmt0050-periodu,
          pland   TYPE zpmt0050-pland,
         END   OF lt_0050.

  CLEAR: lt_40_key, lt_40_key[], lt_50_key, lt_50_key[], lt_0050, lt_0050[].

  SELECT qmnum
    INTO TABLE lt_40_key
    FROM zpmt0090
   WHERE wplann = p_wplann
     AND qmnum <> ''.

  SELECT a~preve b~wendd b~ptype
    INTO TABLE lt_50_key
    FROM zpmt0090 AS a INNER JOIN
         zpmt0060 AS b
      ON a~wplann = b~wplann
     AND a~wpitem = b~wpitem
   WHERE a~wplann = p_wplann
     AND a~preve <> ''.

  SORT lt_40_key BY qmnum.
  SORT lt_50_key BY preve.

  IF lt_40_key[] IS NOT INITIAL.
    LOOP AT lt_40_key.
      UPDATE zpmt0040
         SET reqwf = p_reqwf
             aenam = sy-uname
             aedat = sy-datum
             aezet = sy-uzeit
       WHERE qmnum = lt_40_key-qmnum.
      PERFORM set_req_status USING lt_40_key-qmnum p_reqwf.
    ENDLOOP.
  ENDIF.

  IF lt_50_key[] IS NOT INITIAL.
    SELECT preve workd period periodu pland
      INTO TABLE lt_0050
      FROM zpmt0050
       FOR ALL ENTRIES IN lt_50_key
     WHERE preve = lt_50_key-preve.

    DATA : l_workday TYPE sy-datum.

    LOOP AT lt_0050.
      READ TABLE lt_50_key WITH KEY preve = lt_0050-preve
                                    BINARY SEARCH.
      CLEAR: l_workday.
      IF lt_50_key-ptype = 'MAIN'.
        l_workday     = lt_50_key-wendd.
      ELSE.
        l_workday     = lt_0050-pland.
      ENDIF.

      lt_0050-workd = lt_50_key-wendd.

      call function 'ZPM_GET_NEXT_PLAN_DATE'
        EXPORTING
          i_workd   = l_workday
          i_period  = lt_0050-period
          i_periodu = lt_0050-periodu
        IMPORTING
          e_pland   = lt_0050-pland.

      IF lt_50_key-ptype = 'INSP'.
        lt_0050-pland = l_workday.
      ENDIF.

      MODIFY lt_0050.

      UPDATE zpmt0050
         SET workd = lt_0050-workd
             pland = lt_0050-pland
             aenam = sy-uname
             aedat = sy-datum
             aezet = sy-uzeit
       WHERE preve = lt_0050-preve.

    ENDLOOP.
  ENDIF.

ENDFORM. " SET_WORK_FLOW
