FUNCTION ZPM_GET_EMPNO.
*"----------------------------------------------------------------------
*"*"Local interface:
*" EXPORTING
*" REFERENCE(E_PERNR) TYPE PERSNO
*"----------------------------------------------------------------------

* Global data declarations

  DATA : L_PAR_ID  TYPE USR05-PARID,
         L_PAR_VAL TYPE USR05-PARVA,
         L_PERNR   TYPE PA0001-PERNR.

  CLEAR: L_PAR_ID, L_PAR_VAL, L_PERNR.

  CHECK SY-BATCH <> 'X'.

  SELECT SINGLE * FROM ZPMT1021 WHERE USNAM = SY-UNAME.
  CHECK SY-SUBRC = 0.

  L_PAR_ID  = 'ZEMPNO'.

  GET PARAMETER ID L_PAR_ID FIELD L_PAR_VAL.

  MOVE L_PAR_VAL TO L_PERNR.

  E_PERNR = L_PERNR.

ENDFUNCTION.
