FUNCTION ZPM_GET_NEXT_PLAN_DATE.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_WORKD) TYPE DATUM
*" REFERENCE(I_PERIOD) LIKE ZPMT0050-PERIOD
*" REFERENCE(I_PERIODU) LIKE ZPMT0050-PERIODU
*" EXPORTING
*" REFERENCE(E_PLAND) TYPE DATUM
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_DAY   LIKE T5A4A-DLYDY.
  DATA : LV_WEEK   LIKE T5A4A-DLYDY.
  DATA : LV_MONTH LIKE T5A4A-DLYMO.
  DATA : LV_YEAR  LIKE T5A4A-DLYYR.

* 날짜 주기, 주기단위 값이 없으면 EXIT
  IF I_WORKD = '' OR I_PERIODU = '' OR I_PERIOD = 0.
    EXIT.
  ENDIF.

* 주기단위에 따라 주기일 입력
  CASE I_PERIODU.
    WHEN 'DAY'.
      LV_DAY = I_PERIOD.
    WHEN 'WEEK'.             "주기가 WEEK 일때는 주기 곱하기 7 을 해서 날짜로 계산
      LV_DAY = I_PERIOD * 7.
    WHEN 'MONTH'.
      LV_MONTH = I_PERIOD.
    WHEN 'YEAR'.
      LV_YEAR = I_PERIOD.
  ENDCASE.


  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = I_WORKD
      DAYS      = LV_DAY
      MONTHS    = LV_MONTH
      SIGNUM    = '+'
      YEARS     = LV_YEAR
    IMPORTING
      CALC_DATE = E_PLAND.


ENDFUNCTION.
