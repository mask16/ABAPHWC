        SCREEN-ACTIVE = ''.
        SCREEN-INPUT = ''.
        MODIFY SCREEN.
      ENDIF.
    ENDIF.

    IF SCREEN-NAME = 'BTN_EDOC'.

      DATA : L_CNT TYPE I.

      ENDIF.

    ENDIF.
