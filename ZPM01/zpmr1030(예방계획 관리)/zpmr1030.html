*&---------------------------------------------------------------------*
*& INITIALIZATION

INITIALIZATION.
  call function 'ZPM_CHECK_AUTHORITY .'
* ZPMR1000 설비관리팀 작업현황 프로그램에서 T-CODE 호출할때 값설정
  GET PARAMETER ID 'ZCR' FIELD GV_TNEW.
  GET PARAMETER ID 'ZTP' FIELD GV_TTPLNR.
  GET PARAMETER ID 'ZEQ' FIELD GV_TEQUNR.
  GET PARAMETER ID 'ZVIEW' FIELD GV_VIEW.

  CLEAR: GT_0020, GT_0020[].
  IMPORT MEMORY_ABAP TO GT_0020 FROM MEMORY ID 'ZT_0020'.

* PARAMETER 값 CLEAR
  SET PARAMETER ID 'ZCR' FIELD ''.
  SET PARAMETER ID 'ZTP' FIELD ''.
  SET PARAMETER ID 'ZEQ' FIELD ''.
  SET PARAMETER ID 'ZVIEW' FIELD ''.

  IF GV_VIEW = 'X'.
    P_CHK1 = 'X'.
  ELSE.
  ENDIF.

  SET TITLEBAR  '0100'.
*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& AT SELECTION-SCREEN OUTPUT
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& START-OF-SELECTION
*&---------------------------------------------------------------------*
START-OF-SELECTION.
  CLEAR : GT_TAB, GT_TAB[], GV_COL_POS.

* 생성
  IF GV_TNEW = 'Y'.

*    설비리스트
    PERFORM GET_GT_0020.

    CLEAR : GS_TAB.
    IF GT_203[] IS NOT INITIAL.
      READ TABLE GT_203 INDEX 1.
      GS_TAB-TPLNR = GT_203-TPLNR.
      GS_TAB-EQUNR = GT_203-EQUNR.
      GS_TAB-PTYPE = 'INSP'.
    ELSE.
      GS_TAB-TPLNR = GV_TTPLNR.
      GS_TAB-EQUNR = GV_TEQUNR.
      GS_TAB-PTYPE = 'MAIN'.
    ENDIF.
* 기능위치 내역
    SELECT SINGLE BUKRS WERKS PLTXT
      INTO (GS_TAB-BUKRS, GS_TAB-WERKS, GS_TAB-PLTXT)
      FROM ZPMT0010
      WHERE TPLNR = GS_TAB-TPLNR.

* 설비내역
    SELECT SINGLE EQKTX INTO GS_TAB-EQKTX
      FROM ZPMT0020
      WHERE EQUNR = GS_TAB-EQUNR.

    CALL SCREEN 200.

* 변경,조회
  ELSE.
* DATA GATHERING
    PERFORM GET_DATA.

    IF GT_TAB[] IS INITIAL.
      MESSAGE S005.
      EXIT.
    ELSE.
      IF GV_VIEW = 'X'.
        PERFORM CALL_VIEW.
      ELSE.
        CALL SCREEN 100.
      ENDIF.
    ENDIF.

  ENDIF.
