FORM SELECT_BILLING USING P_STEP P_SORT.
  DATA : V_STEP(2) TYPE N.

  SELECT DISTINCT A~VBELN
    INTO TABLE IT_VBRK
    FROM VBRK AS A JOIN VBRP AS B
      ON A~VBELN EQ B~VBELN
   WHERE A~FKSTO NE 'X'
     AND ( A~SFAKN IS NULL OR A~SFAKN EQ '' )
     AND B~VGTYP EQ IT_SLSDOC_HIST-AUTYP
     AND B~VGBEL EQ IT_SLSDOC_HIST-AUBEL.

  LOOP AT IT_VBRK.
    SELECT SINGLE *
      FROM VBRK
     WHERE VBELN = IT_VBRK-VBELN.

    MOVE : P_STEP        TO IT_SLSDOC_HIST-STEP,
           'Billing'     TO IT_SLSDOC_HIST-STATUS,
           IT_VBRK-VBELN TO IT_SLSDOC_HIST-DN,
           ' '           TO IT_SLSDOC_HIST-NO,
           VBRK-FKDAT    TO IT_SLSDOC_HIST-DT,
           'M'           TO IT_SLSDOC_HIST-AUTYP,
           IT_VBRK-VBELN TO IT_SLSDOC_HIST-AUBEL.
    CONCATENATE P_SORT 'X'  IT_VBRK-VBELN
                       INTO IT_SLSDOC_HIST-SORT.
    APPEND IT_SLSDOC_HIST.

    V_STEP = P_STEP + 1.
    PERFORM SELECT_BILLING_RETRO USING V_STEP IT_SLSDOC_HIST-SORT.
  ENDLOOP.

ENDFORM. "SELECT_BILLING

FORM SELECT_BILLING_RETRO USING P_STEP P_SORT.
  SELECT DISTINCT A~VBELN
    INTO TABLE IT_VBRK_R
    FROM VBRK AS A JOIN VBRP AS B
      ON A~VBELN EQ B~VBELN
   WHERE A~FKSTO NE 'X'
     AND ( A~SFAKN IS NULL OR A~SFAKN EQ '' )
     AND B~VGTYP EQ 'M'
     AND B~VGBEL EQ IT_SLSDOC_HIST-AUBEL.

  LOOP AT IT_VBRK_R.
    SELECT SINGLE *
      FROM VBRK
     WHERE VBELN = IT_VBRK_R-VBELN.

    MOVE : P_STEP          TO IT_SLSDOC_HIST-STEP,
           'Retro-Billing' TO IT_SLSDOC_HIST-STATUS,
           IT_VBRK_R-VBELN TO IT_SLSDOC_HIST-DN,
           ' '             TO IT_SLSDOC_HIST-NO,
           VBRK-FKDAT      TO IT_SLSDOC_HIST-DT,
           'M'             TO IT_SLSDOC_HIST-AUTYP,
           IT_VBRK_R-VBELN TO IT_SLSDOC_HIST-AUBEL.
    CONCATENATE P_SORT 'X' IT_VBRK_R-VBELN
                         INTO IT_SLSDOC_HIST-SORT.
    APPEND IT_SLSDOC_HIST.
  ENDLOOP.
ENDFORM. "SELECT_BILLING_RETRO
