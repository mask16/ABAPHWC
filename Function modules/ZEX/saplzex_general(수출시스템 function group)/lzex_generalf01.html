FORM conv_string_1 USING p_conv p_result.

  CASE p_conv.
    WHEN 1.
      p_result = 'ONE'.
    WHEN 2.
      p_result = 'TWO'.
    WHEN 3.
      p_result = 'THREE'.
    WHEN 4.
      p_result = 'FOUR'.
    WHEN 5.
      p_result = 'FIVE'.
    WHEN 6.
      p_result = 'SIX'.
    WHEN 7.
      p_result = 'SEVEN'.
    WHEN 8.
      p_result = 'EIGHT'.
    WHEN 9.
      p_result = 'NINE'.
    WHEN 0.
      CLEAR p_result.
    WHEN OTHERS.
      RAISE non_number.
  ENDCASE.

ENDFORM. " CONV_STRING_1
