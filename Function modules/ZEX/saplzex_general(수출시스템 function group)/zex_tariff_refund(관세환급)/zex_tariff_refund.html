FUNCTION ZEX_TARIFF_REFUND.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(P_ZEXDCL) LIKE ZEX415-ZEXDCL
*" REFERENCE(P_ZISEQN) LIKE ZEX415-ZISEQN
*" TABLES
*" RETURN STRUCTURE BAPIRET2
*" EXCEPTIONS
*" POST_ERROR
*"----------------------------------------------------------------------

* Global data declarations

  REFRESH RETURN.
  CLEAR  ZEX406.
  SELECT SINGLE *
    FROM ZEX406
   WHERE ZEXDCL EQ P_ZEXDCL.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '수출신고 관리번호를 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  CLEAR  ZEX415.
  SELECT SINGLE *
    FROM ZEX415
   WHERE ZEXDCL EQ P_ZEXDCL
     AND ZISEQN EQ P_ZISEQN.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '환급 내역을 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  IF NOT ZEX415-ZCOMPL IS INITIAL.
    MESSAGE S000 WITH '전기완료 상태입니다.'
                RAISING POST_ERROR.
    EXIT.
  ENDIF.

  CLEAR  ZEX005.
  SELECT SINGLE *
    FROM ZEX005
   WHERE BUKRS = ZEX406-BUKRS.
  MOVE 'N' TO CTU_PARAMS-DISMODE.
  IF SY-UNAME = ZEX005-ZSYMGR.
    IF ZEX005-DISMODE IS INITIAL.
      MOVE 'N' TO CTU_PARAMS-DISMODE.
    ELSE.
      MOVE ZEX005-DISMODE TO CTU_PARAMS-DISMODE.
    ENDIF.
  ENDIF.
  CTU_PARAMS-UPDMODE  = 'S'.
  CTU_PARAMS-CATTMODE = ' '.
  CTU_PARAMS-DEFSIZE  = ' '.
  CTU_PARAMS-RACOMMIT = 'X'.
  CTU_PARAMS-NOBINPT  = 'X'.
  CTU_PARAMS-NOBIEND  = 'X'.

* REFRESH IT_006.
* SELECT *
* FROM ZEX006
* INTO CORRESPONDING FIELDS OF TABLE IT_006
* WHERE BUKRS EQ ZEX406-BUKRS
* AND ZFUSYN EQ 'X'.
* SORT IT_006 BY ZFPAGE ZFSEQU.

  REFRESH  BDCDATA.
  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0100',
      ' ' 'BKPF-BLDAT'    ZEX415-ZREFDT,        " Document Date
      ' ' 'BKPF-BLART'   'DR',                  " Type 전표유형
      ' ' 'BKPF-BUKRS'    ZEX406-BUKRS,         " Company Code
      ' ' 'BKPF-BUDAT'    ZEX415-ZREFDT,        " Posting Date
      ' ' 'BKPF-BKTXT'   '관세환급',            " 전표헤더텍스트
      ' ' 'BKPF-WAERS'    ZEX415-ZCRKRW,        " Currency
      ' ' 'RF05A-NEWBS'   '01',                 " Posting Key
      ' ' 'RF05A-NEWKO'  '9100001',             " 고객
      ' ' 'BDC_OKCODE'   '/00'.                 " Enter

  WRITE ZEX415-ZREFAM CURRENCY ZEX415-ZCRKRW
                            TO V_WRBTR.         " 환급액

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0301',
      ' ' 'BSEG-HKONT'   '1130160',             " G/L 계정
      ' ' 'BSEG-WRBTR'    V_WRBTR,              " 환급액
      ' ' 'BSEG-ZFBDT'    ZEX415-ZREFDT,        " 기준일
      ' ' 'RF05A-NEWBS'  '50',                  " Posting Key
      ' ' 'RF05A-NEWKO'  '1130161',             " 미수금-환급관세(추정)
      ' ' 'BDC_OKCODE'   '/00'.                 " Enter

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0300',
      ' ' 'BSEG-WRBTR'    V_WRBTR,              " 환급액
      ' ' 'BDC_OKCODE'   '=AB'.                 " Enter

  PERFORM SET_COPA_SEG_REFUND.

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0700',
      ' ' 'BDC_OKCODE'   '=BU'.                 " Save

  SET PARAMETER ID 'BLN' FIELD ''.
  SET PARAMETER ID 'GJR' FIELD ''.

  REFRESH : MESSTAB.
  CALL TRANSACTION 'FB01'  USING    BDCDATA
                           OPTIONS  FROM CTU_PARAMS
                           MESSAGES INTO MESSTAB.
  LOOP AT MESSTAB.
    MOVE : MESSTAB-MSGTYP  TO     RETURN-TYPE,
           MESSTAB-MSGID   TO     RETURN-ID,
           MESSTAB-MSGNR   TO     RETURN-NUMBER,
           MESSTAB-MSGV1   TO     RETURN-MESSAGE_V1,
           MESSTAB-MSGV2   TO     RETURN-MESSAGE_V2,
           MESSTAB-MSGV3   TO     RETURN-MESSAGE_V3,
           MESSTAB-MSGV4   TO     RETURN-MESSAGE_V4.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = RETURN-ID
        MSGNR               = RETURN-NUMBER
        MSGV1               = RETURN-MESSAGE_V1
        MSGV2               = RETURN-MESSAGE_V2
        MSGV3               = RETURN-MESSAGE_V3
        MSGV4               = RETURN-MESSAGE_V4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = RETURN-MESSAGE.
    APPEND  RETURN.
  ENDLOOP.

  GET PARAMETER ID 'BLN' FIELD BELNR.
  GET PARAMETER ID 'GJR' FIELD GJAHR.   " 회계년도.

  IF BELNR IS INITIAL OR GJAHR IS INITIAL.
    MESSAGE E000 WITH 'Data 전기 도중 오류가 발생하였습니다.'
                 RAISING POST_ERROR.
    EXIT.
  ENDIF.


  UPDATE ZEX415
     SET BELNR  = BELNR
         GJAHR  = GJAHR
         BLDAT  = ZEX415-ZREFDT
         BUDAT  = ZEX415-ZREFDT
         ZCOMPL = 'X'
 WHERE ZEXDCL EQ P_ZEXDCL
   AND ZISEQN EQ P_ZISEQN.

  IF SY-SUBRC = 0.
    COMMIT WORK.
  ELSE.
    ROLLBACK WORK.
    MESSAGE S000 WITH 'Data 전기후 저장 도중 오류가 발생하였습니다.'
                   RAISING POST_ERROR.
  ENDIF.

ENDFUNCTION.
