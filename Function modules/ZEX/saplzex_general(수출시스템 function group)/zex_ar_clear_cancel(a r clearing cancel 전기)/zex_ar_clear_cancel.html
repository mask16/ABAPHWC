Description: A/R Clearing Cancel 전기
FUNCTION ZEX_AR_CLEAR_CANCEL.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(P_ZNGRNO) LIKE ZEX502-ZNGRNO
*" REFERENCE(P_ZINPSQ) LIKE ZEX502-ZINPSQ
*" REFERENCE(P_STGRD) LIKE UF05A-STGRD
*" REFERENCE(P_ZBUDAT) LIKE ZEX502-ZBUDAT
*" TABLES
*" RETURN STRUCTURE BAPIRET2
*" EXCEPTIONS
*" POST_ERROR
*"----------------------------------------------------------------------

* Global data declarations

  REFRESH RETURN.
  CLEAR  ZEX500.
  SELECT SINGLE *
    FROM ZEX500
   WHERE ZNGRNO EQ P_ZNGRNO.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH 'Nego 관리번호를 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  CLEAR  ZEX502.
  SELECT SINGLE *
    FROM ZEX502
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '입금 내역을 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  IF ZEX502-ZCOMPL IS INITIAL.
    MESSAGE E000 WITH '전기미완료 상태입니다.'
                RAISING POST_ERROR.
  ENDIF.

  CLEAR  ZEX005.
  SELECT SINGLE *
    FROM ZEX005
   WHERE BUKRS = ZEX500-BUKRS.
  MOVE 'N' TO CTU_PARAMS-DISMODE.
  IF SY-UNAME = ZEX005-ZSYMGR.
    IF ZEX005-DISMODE IS INITIAL.
      MOVE 'N' TO CTU_PARAMS-DISMODE.
    ELSE.
      MOVE ZEX005-DISMODE TO CTU_PARAMS-DISMODE.
    ENDIF.
  ENDIF.
  CTU_PARAMS-UPDMODE  = 'S'.
  CTU_PARAMS-CATTMODE = ' '.
  CTU_PARAMS-DEFSIZE  = ' '.
  CTU_PARAMS-RACOMMIT = 'X'.
  CTU_PARAMS-NOBINPT  = 'X'.
  CTU_PARAMS-NOBIEND  = 'X'.

  REFRESH  BDCDATA.
  PERFORM  BDCDATA_APPEND USING :
           'X' 'SAPMF05R'    '0100',
           ' ' 'RF05R-AUGBL' ZEX502-BELNR,     " 반제전표
           ' ' 'RF05R-BUKRS' ZEX500-BUKRS,     " 회사코드
           ' ' 'RF05R-GJAHR' ZEX502-GJAHR,     " 회계연도
           ' ' 'BDC_OKCODE'  '=RAGL'.

  DATA : V_WAERS_O LIKE BKPF-WAERS,
         V_WAERS_C LIKE BKPF-WAERS.

  CLEAR ZEX503.
  SELECT SINGLE *
    FROM ZEX503
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.
  SELECT SINGLE *
     FROM BKPF
    WHERE AWKEY = ZEX503-VBELN
      AND AWTYP = 'VBRK'
      AND STBLG = ''
      AND STGRD = ''.
  MOVE BKPF-WAERS TO V_WAERS_O.

  SELECT SINGLE WAERS
    INTO V_WAERS_C
    FROM BKPF
   WHERE BUKRS EQ ZEX500-BUKRS
     AND BELNR EQ ZEX502-BELNR
     AND GJAHR EQ ZEX502-GJAHR.

  IF V_WAERS_O NE V_WAERS_C. " 반제전표 통화 NE 원전표 통화
    PERFORM BDCDATA_APPEND USING :
           'X' 'SAPLSPO1'    '0300',
           ' ' 'BDC_OKCODE'  '=YES'.
  ELSE.
    SELECT SINGLE *
      FROM BSEG
     WHERE BUKRS EQ ZEX500-BUKRS
       AND BELNR EQ ZEX502-BELNR
       AND GJAHR EQ ZEX502-GJAHR
       AND WRBTR EQ 0
       AND DMBTR GT 0.
    IF SY-SUBRC EQ 0. " 환차
      PERFORM BDCDATA_APPEND USING :
             'X' 'SAPLSPO1'    '0300',
             ' ' 'BDC_OKCODE'  '=YES'.
    ELSE.
      PERFORM BDCDATA_APPEND USING :
             'X' 'SAPLSPO2'    '0100',
             ' ' 'BDC_OKCODE'  '=OPT2'.
    ENDIF.
  ENDIF.

  PERFORM  BDCDATA_APPEND USING :
           'X' 'SAPMF05R'    '0300',
           ' ' 'RF05R-STGRD' P_STGRD,
           ' ' 'RF05R-BUDAT' P_ZBUDAT,
           ' ' 'BDC_OKCODE'  '=ENTR'.

  SET PARAMETER ID 'BLN' FIELD ''.
  SET PARAMETER ID 'GJR' FIELD ''.

  REFRESH : MESSTAB.
  CALL TRANSACTION 'FBRA'  USING    BDCDATA
                           OPTIONS  FROM CTU_PARAMS
                           MESSAGES INTO MESSTAB.

  LOOP AT MESSTAB.
    MOVE : MESSTAB-MSGTYP  TO     RETURN-TYPE,
           MESSTAB-MSGID   TO     RETURN-ID,
           MESSTAB-MSGNR   TO     RETURN-NUMBER,
           MESSTAB-MSGV1   TO     RETURN-MESSAGE_V1,
           MESSTAB-MSGV2   TO     RETURN-MESSAGE_V2,
           MESSTAB-MSGV3   TO     RETURN-MESSAGE_V3,
           MESSTAB-MSGV4   TO     RETURN-MESSAGE_V4.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = RETURN-ID
        MSGNR               = RETURN-NUMBER
        MSGV1               = RETURN-MESSAGE_V1
        MSGV2               = RETURN-MESSAGE_V2
        MSGV3               = RETURN-MESSAGE_V3
        MSGV4               = RETURN-MESSAGE_V4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = RETURN-MESSAGE.
    APPEND  RETURN.
  ENDLOOP.

  GET PARAMETER ID 'BLN' FIELD BELNR.
  GET PARAMETER ID 'GJR' FIELD GJAHR.

* --
  IF BELNR IS INITIAL OR GJAHR IS INITIAL.
      MESSAGE E000 WITH 'Data 전기 도중 오류가 발생하였습니다.'
                   RAISING POST_ERROR.
    EXIT.
  ENDIF.
  SELECT SINGLE *
    FROM ZEX502
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.
  IF ZEX502-BELNR = BELNR AND
     ZEX502-GJAHR = GJAHR.
      MESSAGE E000 WITH 'Data 전기 도중 오류가 발생하였습니다.'
                   RAISING POST_ERROR.
    EXIT.
  ENDIF.
* --

  UPDATE ZEX502
     SET ZBELNR = BELNR
         ZGJAHR = GJAHR
         ZBUDAT = P_ZBUDAT
         ZCOMPL = ' '
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.

  IF SY-SUBRC = 0.
    COMMIT WORK.
  ELSE.
    ROLLBACK WORK.
    MESSAGE E000 WITH 'Data 전기후 저장 도중 오류가 발생하였습니다.'
                    RAISING POST_ERROR.
  ENDIF.


  SELECT SUM( ZIPAMT ) INTO V_ZIPAMT
    FROM ZEX502
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZCOMPL EQ 'X'.
  IF V_ZIPAMT < ZEX500-ZNGRMT.
    UPDATE ZEX500
       SET ZCOMPL = ' '
   WHERE ZNGRNO EQ P_ZNGRNO.
  ELSE.
    UPDATE ZEX500
       SET ZCOMPL = 'X'
   WHERE ZNGRNO EQ P_ZNGRNO.
  ENDIF.

ENDFUNCTION.
