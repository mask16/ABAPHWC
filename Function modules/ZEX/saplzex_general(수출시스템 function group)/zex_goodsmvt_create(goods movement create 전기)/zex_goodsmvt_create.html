*" VALUE(MATDOCUMENTYEAR) TYPE BAPI2017_GM_HEAD_RET-DOC_YEAR
*" TABLES
*" RETURN STRUCTURE BAPIRET2
*" EXCEPTIONS

* Global data declarations

  DATA : BEGIN OF RETURN_1 OCCURS 0.
          INCLUDE STRUCTURE   BAPIRET2.
  DATA : END   OF RETURN_1.

  CLEAR  ZEX650.
  SELECT SINGLE *
    FROM ZEX650
   WHERE ZMIGON EQ P_ZMIGON.
  IF SY-SUBRC NE 0.
    IF SY-LANGU EQ '3'.
      MESSAGE E001 WITH '입고문서 관리번호를 확인하시오.'
              RAISING GR1_ERROR.
    ELSE.
      MESSAGE E001 WITH 'Invalid data : G/R document no.'
              RAISING GR1_ERROR.
    ENDIF.
  ENDIF.

  IF NOT ZEX650-ZCOMPL IS INITIAL.
    IF SY-LANGU EQ '3'.
      MESSAGE E001 WITH 'G/R 완료 상태입니다.'
                  RAISING GR2_ERROR.
    ELSE.
      MESSAGE E001 WITH 'Already G/R posted data.'
                  RAISING GR2_ERROR.
    ENDIF.
  ENDIF.

  LOOP AT RETURN_1.
    CLEAR RETURN.
    MOVE-CORRESPONDING RETURN_1 TO RETURN.
    APPEND RETURN.
  ENDLOOP.

  READ TABLE RETURN_1 WITH KEY TYPE = 'E'.
  IF SY-SUBRC = 0.
    EXIT.
  ENDIF.

* ITEM
  REFRESH IT_651.
  SELECT *
    FROM ZEX651
    INTO CORRESPONDING FIELDS OF TABLE IT_651
   WHERE ZMIGON EQ P_ZMIGON.

*GOODSMVT_TRANSACTION.
*01: MB01 : 구매오더에 대한 입고.
*02: MB31 : 오더에 대한 입고.
*03: MB1A : 출고전기.
*04: MB1B : 이전전기.
*05: MB1C : 기타입고.
*06: MB11 : 자재이동 입력.

  REFRESH : GOODSMVT_ITEM,    GOODSMVT_SERIALNUMBER.
  CLEAR  :  GOODSMVT_ITEM,    GOODSMVT_SERIALNUMBER,
            MATERIALDOCUMENT, MATDOCUMENTYEAR,
            GOODSMVT_HEADER,  GOODSMVT_CODE.

  CLEAR V_COUNT.
  LOOP AT IT_651.

    CLEAR : GOODSMVT_ITEM.
    ADD    1                 TO  V_COUNT.
    MOVE : IT_651-MATNR      TO  GOODSMVT_ITEM-MATERIAL,
           IT_651-WERKS      TO  GOODSMVT_ITEM-PLANT,
           IT_651-LGORT      TO  GOODSMVT_ITEM-STGE_LOC,
           IT_651-ABLAD      TO  GOODSMVT_ITEM-UNLOAD_PT,
           ZEX650-BWART      TO  GOODSMVT_ITEM-MOVE_TYPE,
           IT_651-LFIMG      TO  GOODSMVT_ITEM-ENTRY_QNT,
           IT_651-VRKME      TO  GOODSMVT_ITEM-ENTRY_UOM,
           IT_651-VGBEL      TO  GOODSMVT_ITEM-PO_NUMBER,
           IT_651-VGPOS      TO  GOODSMVT_ITEM-PO_ITEM,
          '구매오더 입고'    TO  GOODSMVT_ITEM-ITEM_TEXT,
          'B'                TO  GOODSMVT_ITEM-MVT_IND.

    APPEND GOODSMVT_ITEM.

    GOODSMVT_SERIALNUMBER-MATDOC_ITM = V_COUNT.
    GOODSMVT_SERIALNUMBER-SERIALNO   = V_COUNT.
    APPEND   GOODSMVT_SERIALNUMBER.
  ENDLOOP.

  MOVE : ZEX650-BUDAT  TO   GOODSMVT_HEADER-PSTNG_DATE,
         ZEX650-BLDAT  TO   GOODSMVT_HEADER-DOC_DATE,
         ZEX650-ZMIGON TO   GOODSMVT_HEADER-REF_DOC_NO,
         ZEX650-FRBNR  TO   GOODSMVT_HEADER-BILL_OF_LADING,
         SY-UNAME      TO   GOODSMVT_HEADER-PR_UNAME,
         ZEX650-BKTXT  TO   GOODSMVT_HEADER-HEADER_TXT,
        '01'           TO   GOODSMVT_CODE-GM_CODE.

  CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
    EXPORTING
      GOODSMVT_HEADER       = GOODSMVT_HEADER
      GOODSMVT_CODE         = GOODSMVT_CODE
    IMPORTING
      GOODSMVT_HEADRET      = GOODSMVT_HEADRET
      MATERIALDOCUMENT      = MATERIALDOCUMENT
      MATDOCUMENTYEAR       = MATDOCUMENTYEAR
    TABLES
      GOODSMVT_ITEM         = GOODSMVT_ITEM
      GOODSMVT_SERIALNUMBER = GOODSMVT_SERIALNUMBER
      RETURN                = RETURN_1.

  REFRESH RETURN.
  LOOP AT RETURN_1.
    CLEAR RETURN.
    MOVE-CORRESPONDING RETURN_1 TO RETURN.
    APPEND RETURN.
  ENDLOOP.

  READ TABLE RETURN WITH KEY TYPE = 'E'.
  IF SY-SUBRC = 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    RAISE GR6_ERROR.
    EXIT.
  ENDIF.

  SELECT MAX( ZISEQN ) INTO V_ZISEQN
    FROM ZEX652
   WHERE ZMIGON = ZEX650-ZMIGON.
  CLEAR ZEX652.
  ADD    10               TO V_ZISEQN.
  MOVE : ZEX650-ZMIGON    TO ZEX652-ZMIGON,
         V_ZISEQN         TO ZEX652-ZISEQN,
         MATERIALDOCUMENT TO ZEX652-MBLNR,
         MATDOCUMENTYEAR  TO ZEX652-MJAHR,
         ZEX650-BUDAT     TO ZEX652-BUDAT,
         ZEX650-BLDAT     TO ZEX652-BLDAT,
         SY-UNAME         TO ZEX652-ZERNAM,
         SY-DATUM         TO ZEX652-ZERDAT,
         SY-UZEIT         TO ZEX652-ZERZET,
         SY-UNAME         TO ZEX652-ZAENAM,
         SY-DATUM         TO ZEX652-ZAEDAT,
         SY-UZEIT         TO ZEX652-ZAEZET.
  INSERT ZEX652.

  IF SY-SUBRC = 0.
    UPDATE ZEX650 SET ZCOMPL = 'X'
                      ZAENAM = SY-UNAME
                      ZAEDAT = SY-DATUM
                      ZAEZET = SY-UZEIT
     WHERE ZMIGON = ZEX650-ZMIGON.

    IF SY-SUBRC = 0.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          WAIT = 'X'.
      MESSAGE S012(MIGO) WITH  MATERIALDOCUMENT.
      MOVE : SY-MSGTY   TO     RETURN-TYPE,
             SY-MSGID   TO     RETURN-ID,
             SY-MSGNO   TO     RETURN-NUMBER,
             SY-MSGV1   TO     RETURN-MESSAGE_V1,
             SY-MSGV2   TO     RETURN-MESSAGE_V2,
             SY-MSGV3   TO     RETURN-MESSAGE_V3,
             SY-MSGV4   TO     RETURN-MESSAGE_V4.

      CALL FUNCTION 'MESSAGE_TEXT_BUILD'
        EXPORTING
          MSGID               = RETURN-ID
          MSGNR               = RETURN-NUMBER
          MSGV1               = RETURN-MESSAGE_V1
          MSGV2               = RETURN-MESSAGE_V2
          MSGV3               = RETURN-MESSAGE_V3
          MSGV4               = RETURN-MESSAGE_V4
        IMPORTING
          MESSAGE_TEXT_OUTPUT = RETURN-MESSAGE.
      APPEND  RETURN.

    ELSE.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      IF SY-LANGU EQ '3'.
        MESSAGE E011 WITH '전기후 저장' RAISING GR5_ERROR.
      ELSE.
        MESSAGE E011 WITH 'save after posting' RAISING GR5_ERROR.
      ENDIF.
    ENDIF.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

    ELSE.
      MESSAGE E011 WITH 'save after posting' RAISING GR5_ERROR.
    ENDIF.
  ENDIF.


ENDFUNCTION.
