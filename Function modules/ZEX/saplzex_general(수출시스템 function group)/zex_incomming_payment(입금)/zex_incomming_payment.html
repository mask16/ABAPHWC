FUNCTION ZEX_INCOMMING_PAYMENT.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(P_ZNGRNO) LIKE ZEX502-ZNGRNO
*" REFERENCE(P_ZINPSQ) LIKE ZEX502-ZINPSQ
*" TABLES
*" RETURN STRUCTURE BAPIRET2
*" EXCEPTIONS
*" POST_ERROR
*"----------------------------------------------------------------------
* Global data declarations

  DATA : V_XREF3 LIKE ZEX502-XREF3,
         V_VKGRP LIKE VBRP-VKGRP,
         V_VKBUR LIKE VBRP-VKBUR.

  REFRESH RETURN.
  CLEAR  ZEX500.
  SELECT SINGLE *
    FROM ZEX500
   WHERE ZNGRNO EQ P_ZNGRNO.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '입금의뢰 관리번호를 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  CLEAR  ZEX502.
  SELECT SINGLE *
    FROM ZEX502
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '입금 내역을 확인하시오.'
            RAISING POST_ERROR.
  ENDIF.

  IF NOT ZEX502-ZCOMPL IS INITIAL.
    MESSAGE S000 WITH '전기완료 상태입니다.'
                RAISING POST_ERROR.
    EXIT.
  ENDIF.

  CLEAR  ZEX005.
  SELECT SINGLE *
    FROM ZEX005
   WHERE BUKRS = ZEX500-BUKRS.
  MOVE 'N' TO CTU_PARAMS-DISMODE.
  IF SY-UNAME = ZEX005-ZSYMGR.
    IF ZEX005-DISMODE IS INITIAL.
      MOVE 'N' TO CTU_PARAMS-DISMODE.
    ELSE.
      MOVE ZEX005-DISMODE TO CTU_PARAMS-DISMODE.
    ENDIF.
  ENDIF.
  CTU_PARAMS-UPDMODE  = 'S'.
  CTU_PARAMS-CATTMODE = ' '.
  CTU_PARAMS-DEFSIZE  = ' '.
  CTU_PARAMS-RACOMMIT = 'X'.
  CTU_PARAMS-NOBINPT  = 'X'.
  CTU_PARAMS-NOBIEND  = 'X'.

  REFRESH IT_503.
  SELECT *
    FROM ZEX503
    INTO CORRESPONDING FIELDS OF TABLE IT_503
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.

  REFRESH IT_505.
  SELECT *
    FROM ZEX505
    INTO CORRESPONDING FIELDS OF TABLE IT_505
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ.

  REFRESH IT_006.
  SELECT *
    FROM ZEX006
    INTO CORRESPONDING FIELDS OF TABLE IT_006
   WHERE BUKRS  EQ ZEX500-BUKRS
     AND ZFUSYN EQ 'X'.
  SORT IT_006 BY ZFPAGE ZFSEQU.

  CLEAR  V_WRBTR_1.
  SELECT SUM( WRBTR ) INTO V_WRBTR_1
    FROM ZEX504
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZINPSQ EQ P_ZINPSQ
     AND WAERS  EQ ZEX502-WAERS.

  READ TABLE IT_503 WITH KEY ZINPSQ = P_ZINPSQ.
  CLEAR VBRK.
  SELECT SINGLE *
    FROM VBRK
   WHERE VBELN EQ IT_503-VBELN.
* 사업영역
  CLEAR : V_GSBER, V_PRCTR, V_VKGRP, V_VKBUR.
  SELECT MAX( GSBER ) MAX( PRCTR ) MAX( VKGRP ) MAX( VKBUR )
    INTO (V_GSBER, V_PRCTR, V_VKGRP, V_VKBUR)
    FROM VBRP
   WHERE VBELN EQ IT_503-VBELN.
  CONCATENATE VBRK-VKORG VBRK-VTWEG VBRK-SPART V_VKBUR V_VKGRP INTO V_XREF3.
* 사업장
  SELECT MAX( BUPLA )
    INTO V_BUPLA
    FROM VBRK
   WHERE VBELN EQ IT_503-VBELN.
  MOVE '1000' TO V_BUPLA.

  REFRESH  BDCDATA.
  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0100',
      ' ' 'BKPF-BLDAT'    ZEX502-ZIPDAT,        " Document Date
      ' ' 'BKPF-BLART'   'D3',                  " Type 전표유형
      ' ' 'BKPF-BUKRS'    ZEX500-BUKRS,         " Company Code
      ' ' 'BKPF-BUDAT'    ZEX502-ZIPDAT,        " Posting Date
      ' ' 'BKPF-BKTXT'   '수출대금입금',        " 전표헤더텍스트
      ' ' 'BKPF-WAERS'    ZEX502-WAERS,         " Currency
      ' ' 'RF05A-NEWBS'   '40',                 " Posting Key
      ' ' 'RF05A-NEWKO'   ZEX502-ZACCN1,        " 외화당좌
      ' ' 'BDC_OKCODE'   '/00'.                 " Enter

* 실입금액 = 입금액 - SUM(비용).
  V_ZIPAMT = ZEX502-ZIPAMT - V_WRBTR_1.
  WRITE V_ZIPAMT CURRENCY ZEX502-WAERS
                          TO V_WRBTR.           " 실입금액

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0300',
      ' ' 'BSEG-WRBTR'    V_WRBTR,              " 실입금액
      ' ' 'BSEG-VALUT'    ZEX502-ZIPDAT.        " 기준일
*--------
* 비용
  LOOP AT IT_505.

    WRITE IT_505-WRBTR  CURRENCY IT_505-WAERS
                                 TO V_WRBTR.
    PERFORM  BDCDATA_APPEND USING :
        ' ' 'RF05A-NEWKO'   IT_505-NEWKO,       " 계정
        ' ' 'RF05A-NEWBS'   IT_505-NEWBS,       " Posting Key
        ' ' 'BDC_OKCODE'   '/00'.

*--------------------------------------------------------------------
* Coding Block 주의.....
* PERFORM BDCDATA_APPEND USING :
* 'X' 'SAPLKACB' '0002',
* ' ' 'BDC_OKCODE' '/00'.
*- Coding Block ----

* CostCenter비용 & 일반비용
* IF IT_504-ZCHRTY EQ 'C'.
    PERFORM  BDCDATA_APPEND USING :
        'X' 'SAPMF05A'     '0300',
        ' ' 'BSEG-WRBTR'    V_WRBTR,            " 비용
        ' ' 'BSEG-MWSKZ'    IT_505-MWSKZ,       " Tax code
        ' ' 'BSEG-BUPLA'    V_BUPLA,            " 사업장
        ' ' 'COBL-KOSTL'    IT_505-KOSTL,       " Cost Center.
        ' ' 'BSEG-SGTXT'    IT_505-SGTXT.       " Text
    IF IT_505-ZCHRTY EQ 'D'.
      PERFORM BDCDATA_APPEND USING :
              ' ' 'BDC_OKCODE' '=COBL_XERGO'.
      PERFORM SET_COPA_SEG_NEGO.
    ENDIF.
    PERFORM  BDCDATA_APPEND USING :
        'X' 'SAPMF05A'     '0300',
        ' ' 'BDC_OKCODE'   '/00'.

*--------------------------------------------------------------------
* Coding Block 주의.....
* PERFORM BDCDATA_APPEND USING :
* 'X' 'SAPLKACB' '0002',
* ' ' 'BDC_OKCODE' '/00'.
*- Coding Block ----
* ENDIF.

  ENDLOOP.

*----
  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0300',
      ' ' 'RF05A-NEWBS'   '11',                 " Posting Key
      ' ' 'RF05A-NEWKO'   VBRK-KUNRG,           " Customer
      ' ' 'BDC_OKCODE'   '/00'.                 " Enter

* PERFORM BDCDATA_APPEND USING :
* 'X' 'SAPLKACB' '0002',
* ' ' 'BDC_OKCODE' '=ENTE'. " Enter

* 입금액
  WRITE ZEX502-ZIPAMT CURRENCY ZEX502-WAERS
                          TO V_WRBTR.           " 입금액

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0301',
      ' ' 'BSEG-WRBTR'    V_WRBTR,              " 입금액
      ' ' 'BSEG-GSBER'    V_GSBER,              " 사업영역
      ' ' 'BDC_OKCODE'   '=ZK'.                 "

  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0331',
* ' ' 'BSEG-XREF2' ZEX502-XREF2, " 참조키2
* ' ' 'BSEG-XREF3' ZEX502-XREF3, " 참조키3
      ' ' 'BSEG-XREF3'   V_XREF3,                           " 참조키3
* ' ' 'BSEG-FDLEV' ZEX502-FDLEV, " 계획레벨
* ' ' 'BSEG-FDTAG' ZEX502-ZIPDAT, " 계획일
      ' ' 'BDC_OKCODE'   '=ZK'.                 "


* 저장.
  PERFORM  BDCDATA_APPEND USING :
      'X' 'SAPMF05A'     '0301',
      ' ' 'BSEG-HKONT'   ZEX502-HKONT,          " G/L 계정
      ' ' 'BDC_OKCODE'   '=BU'.

  SET PARAMETER ID 'BLN' FIELD ''.
  SET PARAMETER ID 'GJR' FIELD ''.

  REFRESH : MESSTAB.
  CALL TRANSACTION 'FB01'  USING    BDCDATA
                           OPTIONS  FROM CTU_PARAMS
                           MESSAGES INTO MESSTAB.

  LOOP AT MESSTAB.
    MOVE : MESSTAB-MSGTYP  TO     RETURN-TYPE,
           MESSTAB-MSGID   TO     RETURN-ID,
           MESSTAB-MSGNR   TO     RETURN-NUMBER,
           MESSTAB-MSGV1   TO     RETURN-MESSAGE_V1,
           MESSTAB-MSGV2   TO     RETURN-MESSAGE_V2,
           MESSTAB-MSGV3   TO     RETURN-MESSAGE_V3,
           MESSTAB-MSGV4   TO     RETURN-MESSAGE_V4.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = RETURN-ID
        MSGNR               = RETURN-NUMBER
        MSGV1               = RETURN-MESSAGE_V1
        MSGV2               = RETURN-MESSAGE_V2
        MSGV3               = RETURN-MESSAGE_V3
        MSGV4               = RETURN-MESSAGE_V4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = RETURN-MESSAGE.
    APPEND  RETURN.
  ENDLOOP.

  GET PARAMETER ID 'BLN' FIELD BELNR.
  GET PARAMETER ID 'GJR' FIELD GJAHR.   " 회계년도.
*
  IF BELNR  IS INITIAL OR  GJAHR  IS INITIAL.
    MESSAGE E000 WITH 'Data 전기 도중 오류가 발생하였습니다.'
                 RAISING POST_ERROR.
    EXIT.
  ENDIF.

  UPDATE ZEX502
     SET BELNR  = BELNR
         GJAHR  = GJAHR
         BLDAT  = ZEX502-ZIPDAT
         BUDAT  = ZEX502-ZIPDAT
         ZCOMPL = 'X'
 WHERE ZNGRNO EQ P_ZNGRNO
   AND ZINPSQ EQ P_ZINPSQ.

  IF SY-SUBRC = 0.
    COMMIT WORK.
  ELSE.
    ROLLBACK WORK.
    MESSAGE S000 WITH 'Data 전기후 저장 도중 오류가 발생하였습니다.'
                   RAISING POST_ERROR.
  ENDIF.

  SELECT SUM( ZIPAMT ) INTO V_ZIPAMT
    FROM ZEX502
   WHERE ZNGRNO EQ P_ZNGRNO
     AND ZCOMPL EQ 'X'.
  IF V_ZIPAMT < ZEX500-ZNGRMT.
    UPDATE ZEX500
       SET ZCOMPL = ' '
   WHERE ZNGRNO EQ P_ZNGRNO.
  ELSE.
    UPDATE ZEX500
       SET ZCOMPL = 'X'
   WHERE ZNGRNO EQ P_ZNGRNO.
  ENDIF.

ENDFUNCTION.
