FORM CUSTOMS_RETURN_FILE_SAVE USING P_ERRX P_ERRTEXT.
  IF IT_DOWN_EXP[] IS INITIAL.
    MOVE 'X' TO P_ERRX.
    MOVE '처리할 데이타가 없습니다' TO P_ERRTEXT.
    EXIT.
  ENDIF.

* FILE DOWNLOAD & TABLE UPDATE
  REFRESH IT_DOWN_EXF.
  LOOP AT IT_DOWN_EXP.
    MOVE     IT_DOWN_EXP TO IT_DOWN_EXF.
    CONDENSE IT_DOWN_EXF.
    REPLACE  ALL OCCURRENCES OF ' |'  IN IT_DOWN_EXF WITH '|'.
    REPLACE  ALL OCCURRENCES OF '| 0' IN IT_DOWN_EXF WITH '|0'.
    REPLACE  ALL OCCURRENCES OF '| 1' IN IT_DOWN_EXF WITH '|1'.
    REPLACE  ALL OCCURRENCES OF '| 2' IN IT_DOWN_EXF WITH '|2'.
    REPLACE  ALL OCCURRENCES OF '| 3' IN IT_DOWN_EXF WITH '|3'.
    REPLACE  ALL OCCURRENCES OF '| 4' IN IT_DOWN_EXF WITH '|4'.
    REPLACE  ALL OCCURRENCES OF '| 5' IN IT_DOWN_EXF WITH '|5'.
    REPLACE  ALL OCCURRENCES OF '| 6' IN IT_DOWN_EXF WITH '|6'.
    REPLACE  ALL OCCURRENCES OF '| 7' IN IT_DOWN_EXF WITH '|7'.
    REPLACE  ALL OCCURRENCES OF '| 8' IN IT_DOWN_EXF WITH '|8'.
    REPLACE  ALL OCCURRENCES OF '| 9' IN IT_DOWN_EXF WITH '|9'.
    APPEND   IT_DOWN_EXF.
  ENDLOOP.

  CALL FUNCTION 'WS_DOWNLOAD'
    EXPORTING
      FILENAME  = CUSTOMS_RETURN_FILDID
      FILETYPE  = 'ASC'
    TABLES
      DATA_TAB  = IT_DOWN_EXF.

  IF SY-SUBRC <> 0.
    MOVE 'X' TO P_ERRX.
    CONCATENATE SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4
           INTO P_ERRTEXT SEPARATED BY SPACE.
  ELSE.
    LOOP AT IT_CUSTOMS_EXP WHERE CHECK = 'X'.
      IF IT_CUSTOMS_EXP-PROCX = 'X'.
        CONTINUE.
      ENDIF.

      MOVE 'X' TO IT_CUSTOMS_EXP-ZCUSTX.
      MODIFY IT_CUSTOMS_EXP.

      UPDATE ZEX406
         SET ZCUSTX = 'X'
             ZCUSTF = CUSTOMS_RETURN_FILDID
       WHERE ZEXDCL = IT_CUSTOMS_EXP-ZEXDCL.

      IF SY-SUBRC NE 0.
        MOVE 'X' TO P_ERRX.
        CONCATENATE 'ZEX406 수출신고필증 UPDATE ERROR' IT_CUSTOMS_EXP-ZEXDCL
               INTO P_ERRTEXT SEPARATED BY SPACE.
        ROLLBACK WORK.
      ENDIF.
    ENDLOOP.
  ENDIF.
ENDFORM.
