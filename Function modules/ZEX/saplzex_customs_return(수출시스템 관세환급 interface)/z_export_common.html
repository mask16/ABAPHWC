FORM SELECT_ALL_SALES_DOC_COM.

  REFRESH : IT_VBAK, IT_LIKP, IT_VTTK, IT_VBRK, IT_VBRK_R, IT_ALLDOC.
  CLEAR   : IT_VBAK, IT_LIKP, IT_VTTK, IT_VBRK, IT_VBRK_R, IT_ALLDOC.

  LOOP AT IT_SLSDOC.
    CLEAR : IT_VBAK, IT_LIKP, IT_VTTK, IT_VBRK, IT_VBRK_R.

    CASE  IT_SLSDOC-AUTYP.
*     S/O : 오더,대변메모,차변메모,수수료없는오더
      WHEN 'C' OR 'K' OR 'L' OR 'l'.
        SELECT SINGLE VBELN
          INTO IT_VBAK-VGBEL
          FROM VBAK
         WHERE VBELN = IT_SLSDOC-AUBEL.

        IF SY-SUBRC EQ 0.
          MOVE : IT_SLSDOC-AUTYP TO IT_VBAK-AUTYP,
                 IT_SLSDOC-AUBEL TO IT_VBAK-AUBEL.
          APPEND IT_VBAK.
        ENDIF.

        SELECT DISTINCT VBELN
        APPENDING TABLE IT_LIKP
          FROM LIPS
         WHERE VGBEL = IT_SLSDOC-AUBEL.

        SELECT DISTINCT B~TKNUM
        APPENDING TABLE IT_VTTK
          FROM LIPS AS A JOIN VTTP AS B
            ON A~VBELN = B~VBELN
         WHERE A~VGBEL = IT_SLSDOC-AUBEL.

*     Delivery
      WHEN 'J'.
        SELECT SINGLE VBELN
          INTO IT_LIKP-VBELN
          FROM LIKP
         WHERE VBELN = IT_SLSDOC-AUBEL.
        IF SY-SUBRC EQ 0.
          APPEND IT_LIKP.
        ENDIF.

        SELECT DISTINCT VGBEL VGTYP
          INTO (IT_VBAK-VGBEL, IT_VBAK-VGTYP)
          FROM LIPS
         WHERE VBELN = IT_SLSDOC-AUBEL.

          MOVE : IT_SLSDOC-AUTYP TO IT_VBAK-AUTYP,
                 IT_SLSDOC-AUBEL TO IT_VBAK-AUBEL.
          COLLECT IT_VBAK.
        ENDSELECT.

        SELECT DISTINCT TKNUM
        APPENDING TABLE IT_VTTK
          FROM VTTP
         WHERE VBELN = IT_SLSDOC-AUBEL.

*     Shipment
      WHEN '8'.
        SELECT SINGLE TKNUM
          INTO IT_VTTK-TKNUM
          FROM VTTK
         WHERE TKNUM = IT_SLSDOC-AUBEL.
        IF SY-SUBRC EQ 0.
          APPEND IT_VTTK.
        ENDIF.

        SELECT DISTINCT B~VGBEL
          INTO IT_VBAK-VGBEL
          FROM VTTP AS A JOIN LIPS AS B
            ON A~VBELN = B~VBELN
         WHERE A~TKNUM = IT_SLSDOC-AUBEL
           AND B~VGTYP = 'C'.

          MOVE : IT_SLSDOC-AUTYP TO IT_VBAK-AUTYP,
                 IT_SLSDOC-AUBEL TO IT_VBAK-AUBEL.
          COLLECT IT_VBAK.
        ENDSELECT.

        SELECT DISTINCT VBELN
        APPENDING TABLE IT_LIKP
          FROM VTTP
         WHERE TKNUM = IT_SLSDOC-AUBEL.

*     Billing
      WHEN 'M'.
        PERFORM SELECT_ALL_SALES_DOC_COM_B.

      WHEN OTHERS.

    ENDCASE.
  ENDLOOP.

  SORT IT_VBAK BY VGBEL.
  SORT IT_LIKP BY VBELN.
  SORT IT_VTTK BY TKNUM.
  DELETE ADJACENT DUPLICATES FROM IT_VBAK.
  DELETE ADJACENT DUPLICATES FROM IT_LIKP.
  DELETE ADJACENT DUPLICATES FROM IT_VTTK.

* 오더
  LOOP AT IT_VBAK.
    CLEAR  IT_ALLDOC.
    MOVE IT_VBAK-VGBEL  TO IT_ALLDOC-AUBEL.
    SELECT SINGLE VBTYP INTO IT_ALLDOC-AUTYP
      FROM VBAK
     WHERE VBELN = IT_ALLDOC-AUBEL.
    APPEND IT_ALLDOC.
  ENDLOOP.

* 납품
  LOOP AT IT_LIKP.
    CLEAR  IT_ALLDOC.
    MOVE 'J'           TO IT_ALLDOC-AUTYP.
    MOVE IT_LIKP-VBELN TO IT_ALLDOC-AUBEL.
    APPEND IT_ALLDOC.
  ENDLOOP.

* 선적
  LOOP AT IT_VTTK.
    CLEAR  IT_ALLDOC.
    MOVE '8'           TO IT_ALLDOC-AUTYP.
    MOVE IT_VTTK-TKNUM TO IT_ALLDOC-AUBEL.
    APPEND IT_ALLDOC.
  ENDLOOP.

* TR
* LOOP AT IT_OIGS.
*   CLEAR  IT_ALLDOC.
*   MOVE 'r'              TO IT_ALLDOC-AUTYP.
*   MOVE IT_OIGS-SHNUMBER TO IT_ALLDOC-AUBEL.
*   APPEND IT_ALLDOC.
* ENDLOOP.

  READ TABLE IT_ALLDOC INDEX 1.
  IF SY-SUBRC NE 0.
    EXIT.
  ENDIF.

  IF IT_SLSDOC-AUTYP EQ 'C' OR
     IT_SLSDOC-AUTYP EQ 'K' OR
     IT_SLSDOC-AUTYP EQ 'k' OR
     IT_SLSDOC-AUTYP EQ 'L' OR
     IT_SLSDOC-AUTYP EQ 'l' OR
     IT_SLSDOC-AUTYP EQ 'J' OR
     IT_SLSDOC-AUTYP EQ 'r' OR
     IT_SLSDOC-AUTYP EQ '8'.
* Billing
    SELECT DISTINCT A~VBELN
      INTO TABLE IT_VBRK
      FROM VBRK AS A JOIN VBRP AS B
        ON A~VBELN EQ B~VBELN
    FOR ALL ENTRIES IN IT_ALLDOC
     WHERE A~FKSTO NE 'X'
       AND ( A~SFAKN IS NULL OR A~SFAKN EQ '' )
       AND B~VGTYP EQ IT_ALLDOC-AUTYP
       AND B~VGBEL EQ IT_ALLDOC-AUBEL.
    SORT IT_VBRK BY VBELN.
    DELETE ADJACENT DUPLICATES FROM IT_VBRK.
    LOOP AT IT_VBRK.
      CLEAR  IT_ALLDOC.
      MOVE 'M'              TO IT_ALLDOC-AUTYP.
      MOVE IT_VBRK-VBELN    TO IT_ALLDOC-AUBEL.
      APPEND IT_ALLDOC.
    ENDLOOP.
  ENDIF.

* Retro-Billing
  LOOP AT IT_VBRK.
    SELECT DISTINCT A~VBELN
      INTO TABLE IT_VBRK_R
      FROM VBRK AS A JOIN VBRP AS B
        ON A~VBELN EQ B~VBELN
    FOR ALL ENTRIES IN IT_VBRK
     WHERE A~FKSTO NE 'X'
       AND ( A~SFAKN IS NULL OR A~SFAKN EQ '' )
       AND B~VGTYP EQ 'M'
       AND B~VGBEL EQ IT_VBRK-VBELN.
    SORT IT_VBRK_R BY VBELN.
    DELETE ADJACENT DUPLICATES FROM IT_VBRK_R.
    LOOP AT IT_VBRK_R.
      CLEAR  IT_ALLDOC.
      MOVE 'M'              TO IT_ALLDOC-AUTYP.
      MOVE IT_VBRK_R-VBELN  TO IT_ALLDOC-AUBEL.
      APPEND IT_ALLDOC.
    ENDLOOP.
  ENDLOOP.

ENDFORM. " SELECT_ALL_SALES_DOC_COM
