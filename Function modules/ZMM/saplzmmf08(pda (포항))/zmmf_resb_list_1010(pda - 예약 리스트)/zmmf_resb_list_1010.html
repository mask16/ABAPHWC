FUNCTION ZMMF_RESB_LIST_1010.
*"--------------------------------------------------------------------
*"*"로컬인터페이스:
*" IMPORTING
*" VALUE(I_WERKS) TYPE WERKS_D
*" VALUE(I_LGORT) TYPE LGORT_D
*" VALUE(I_RSNUM_F) TYPE RSNUM
*" VALUE(I_RSNUM_T) TYPE RSNUM OPTIONAL
*" VALUE(I_BDTER_F) TYPE BDTER
*" VALUE(I_BDTER_T) TYPE BDTER OPTIONAL
*" EXPORTING
*" VALUE(E_RET) TYPE ZERETURN
*" VALUE(E_MSG) TYPE ZEMESSAGE
*" TABLES
*" ET_TAB STRUCTURE ZMMS0005
*"--------------------------------------------------------------------
* Global data declarations

  TABLES resb.
  RANGES ra_rsnum FOR resb-rsnum.
  RANGES ra_bdter FOR resb-bdter.
  CLEAR : ra_rsnum, ra_rsnum[],
          ra_bdter, ra_bdter[].

  IF i_rsnum_f IS NOT INITIAL
    AND i_rsnum_t IS INITIAL.

    ra_rsnum-sign = 'I'.
    ra_rsnum-option =  'EQ'.
    ra_rsnum-low =  i_rsnum_f.
    APPEND ra_rsnum.

  ENDIF.

  IF i_rsnum_f IS NOT INITIAL
    AND i_rsnum_t IS NOT INITIAL.
    ra_rsnum-sign = 'I'.
    ra_rsnum-option =  'BT'.
    ra_rsnum-low =  i_rsnum_f.
    ra_rsnum-high =  i_rsnum_t.
    APPEND ra_rsnum.

  ENDIF.

  IF i_rsnum_f IS  INITIAL
    AND i_rsnum_t IS NOT INITIAL.
    ra_rsnum-sign = 'I'.
    ra_rsnum-option =  'BT'.
    ra_rsnum-high =  i_rsnum_t.
    APPEND ra_rsnum.

  ENDIF.



  IF i_bdter_f IS NOT INITIAL
    AND i_bdter_t IS INITIAL.

    ra_bdter-sign = 'I'.
    ra_bdter-option =  'EQ'.
    ra_bdter-low =  i_bdter_f.
    APPEND ra_bdter.

  ENDIF.

  IF i_bdter_f IS NOT INITIAL
    AND i_bdter_t IS NOT INITIAL.
    ra_bdter-sign = 'I'.
    ra_bdter-option =  'BT'.
    ra_bdter-low =  i_bdter_f.
    ra_bdter-high =  i_bdter_t.
    APPEND ra_bdter.

  ENDIF.

  IF i_bdter_f IS  INITIAL
    AND i_bdter_t IS NOT INITIAL.
    ra_bdter-sign = 'I'.
    ra_bdter-option =  'BT'.
    ra_bdter-high =  i_bdter_t.
    APPEND ra_bdter.

  ENDIF.

  SELECT a~rsnum a~rspos
         a~matnr
         a~bdmng
         a~enmng
         a~meins
         a~bdter
         a~wempf
         a~kzear
    INTO CORRESPONDING FIELDS OF TABLE et_tab
    FROM resb AS a
   WHERE werks = i_werks
     AND lgort = i_lgort
     AND bwart = '311'
     AND rsnum IN ra_rsnum
     AND bdter IN ra_bdter
     AND kzear  = ''.

  IF sy-subrc = 0.
    LOOP AT et_tab.

      SELECT SINGLE maktx
        INTO et_tab-maktx
        FROM makt
       WHERE matnr = et_tab-matnr
         AND spras = sy-langu.


      et_tab-menge_re = et_tab-bdmng - et_tab-enmng.

      MODIFY et_tab.
      CLEAR et_tab.
    ENDLOOP.

    e_ret = 'S'.
    e_msg  = text-009.
  ELSE.
    e_ret = 'W'.
    e_msg  = text-002.
  ENDIF.

ENDFUNCTION.
