FUNCTION ZMMF02_GET_STOCK.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_MATNR) TYPE MATNR
*" REFERENCE(I_WERKS) TYPE WERKS_D
*" REFERENCE(I_LGORT) TYPE LGORT_D
*" REFERENCE(I_SDATE) TYPE SPMON
*" REFERENCE(I_MTART) TYPE MTART OPTIONAL
*" EXPORTING
*" REFERENCE(O_SPEME) TYPE LABST
*" REFERENCE(O_VPRSV) TYPE VPRSV
*" REFERENCE(O_VERPR) TYPE VERPR
*" REFERENCE(O_STPRS) TYPE STPRS
*" REFERENCE(O_DMBTR) TYPE DMBTR
*" REFERENCE(O_SALK3) TYPE DMBTR
*" REFERENCE(O_LBKUM) TYPE LBKUM
*" TABLES
*" B_MARD STRUCTURE MARDH OPTIONAL
*"----------------------------------------------------------------------


* Global data declarations

  DATA tmp_spmon TYPE spmon.
  DATA : p_matnr TYPE matnr,
         p_werks TYPE werks_d,
         p_lgort TYPE lgort_d,
         p_sdate TYPE spmon.
  DATA : v_sdate  TYPE sy-datum.

  DATA v_icnt0 TYPE i.

  CLEAR o_speme.
  p_matnr   = i_matnr.
  p_werks   = i_werks.
  p_lgort   = i_lgort.
  p_sdate   = i_sdate.

  v_sdate =  p_sdate.
  v_sdate+6(2) = '01'.

* 구하는 월 >= 현재월 일때는 MARD 에서만 읽음 된다....
  IF i_sdate >= sy-datum+0(6).
    SELECT SINGLE speme INTO o_speme
        FROM      mard
       WHERE matnr = p_matnr
        AND werks = p_werks
        AND lgort = p_lgort.
  ELSE.
    SELECT lfgja lfmon speme
     APPENDING CORRESPONDING FIELDS OF TABLE b_mard
        FROM      mard
         WHERE matnr = p_matnr
           AND werks = p_werks
           AND lgort = p_lgort.
    SELECT lfgja lfmon speme
      APPENDING CORRESPONDING FIELDS OF TABLE b_mard
       FROM      mardh
         WHERE matnr = p_matnr
            AND werks = p_werks
            AND lgort = p_lgort.

    SORT b_mard BY lfgja DESCENDING  lfmon DESCENDING.


* 구하는 월이 존재할 경우 해당 Record 를 읽는다

    READ TABLE b_mard WITH KEY lfgja = p_sdate+0(4)
                               lfmon = p_sdate+4(2).

    IF sy-subrc = 0.
* o_labst = t_mard-labst.
      o_speme =  b_mard-speme.
    ELSE.

** 구하는 월이 MARD 날짜 최신 날짜 보다 크면 그 Record 를 가져온다
      READ TABLE b_mard INDEX 1.
      CONCATENATE b_mard-lfgja   b_mard-lfmon INTO tmp_spmon .

      IF i_sdate > tmp_spmon .
* o_labst = t_mard-labst.
        o_speme = b_mard-speme.
        EXIT.
      ENDIF.
* 구하는 월에 해당하는 Record 가 없으면 + 1개월씩하면서 구한다
      DO 12 TIMES.
        CALL FUNCTION 'MONTH_PLUS_DETERMINE'
          EXPORTING
            months  = 1
            olddate = v_sdate
          IMPORTING
            newdate = v_sdate.


        IF v_sdate+0(6) = sy-datum+0(6).
          READ TABLE b_mard INDEX 1.
* o_labst = t_mard-labst.
          o_speme = b_mard-speme.
          EXIT.
        ENDIF.

        READ TABLE b_mard WITH KEY lfgja = v_sdate+0(4)
                                   lfmon = v_sdate+4(2)  .

        IF sy-subrc = 0.

* o_labst = t_mard-labst.
          o_speme = b_mard-speme.
          EXIT.
        ENDIF.

      ENDDO.

    ENDIF.

  ENDIF.



ENDFUNCTION.
