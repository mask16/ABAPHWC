FUNCTION zmmf_sh_exit.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------
* Select the Bill to party customer based on the select options
* Global data declarations

  CASE callcontrol-step.
    WHEN 'SELECT'.

      select MWSKZ

         INTO TABLE i_mwskz
        from T007AC
       where VORGG  = 'MMIV'.


      CHECK sy-subrc = 0.
      DELETE ADJACENT DUPLICATES FROM i_mwskz.

      LOOP AT i_mwskz INTO wa_mwskz.
        v_tabix = sy-tabix.
        SELECT SINGLE TEXT1 FROM T007S
        INTO wa_mwskz-TEXT1

        WHERE MWSKZ = wa_mwskz-MWSKZ and spras = sy-langu.


        MOVE wa_mwskz TO record_tab-string.
        APPEND record_tab.
        CLEAR record_tab.
        MODIFY i_mwskz FROM wa_mwskz INDEX v_tabix.
        CLEAR wa_mwskz.
      ENDLOOP.
      rc = 0.
      IF rc = 0.
        callcontrol-step = 'DISP'.
      ELSE.
        callcontrol-step = 'EXIT'.
      ENDIF.
      EXIT.
  ENDCASE.
ENDFUNCTION.
