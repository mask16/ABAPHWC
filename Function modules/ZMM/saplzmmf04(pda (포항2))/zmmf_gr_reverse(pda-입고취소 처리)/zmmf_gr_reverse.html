FUNCTION zmmf_gr_reverse.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BARCODE) TYPE ZEBARCODE
*" VALUE(I_FLAG) TYPE CHAR1
*" VALUE(I_FRDAT) TYPE DATUM
*" VALUE(I_TODAT) TYPE DATUM
*" EXPORTING
*" VALUE(E_RET) TYPE ZERETURN
*" VALUE(E_MSG) TYPE ZEMESSAGE
*" TABLES
*" ET_TAB STRUCTURE ZMMS0016 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA: v_matnr TYPE matnr.

  DATA: v_docno TYPE bapi2017_gm_head_02-mat_doc,
        v_gjahr TYPE bapi2017_gm_head_02-doc_year,
        t_return TYPE TABLE OF bapiret2,
        s_return TYPE bapiret2.

  CASE i_flag.
    WHEN 'D'.
      CHECK i_barcode IS NOT INITIAL.

      CLEAR: v_matnr.
      SELECT SINGLE matnr
      INTO (v_matnr)
      FROM zmmt0090
      WHERE barcode = i_barcode.


      IF sy-subrc = 0.
        CLEAR: et_tab, et_tab[].

        SELECT a~mblnr a~mjahr a~budat a~cpudt
               b~zeile b~lifnr b~bwart b~menge b~meins
               b~ebeln b~ebelp b~lgort b~matnr
               c~maktx
        INTO CORRESPONDING FIELDS OF TABLE et_tab
        FROM mkpf AS a
        JOIN mseg AS b ON a~mblnr = b~mblnr
                       AND a~mjahr = b~mjahr
        LEFT JOIN makt AS c ON b~matnr = c~matnr
                            AND c~spras = sy-langu
        WHERE b~matnr = v_matnr
          AND ( a~budat GE i_frdat AND a~budat LE i_todat ).

       SORT et_tab BY mblnr ASCENDING.

        LOOP AT et_tab.
         CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
         EXPORTING
            input  = et_tab-matnr
            IMPORTING
                output = et_tab-matnr.

         MODIFY et_tab.
        ENDLOOP.

      ELSE.


      ENDIF.

    WHEN 'R'.
      LOOP AT et_tab.
        IF et_tab-mblnr IS INITIAL.
          e_ret = 'E'.
          e_msg  = text-002.
          EXIT.
        ENDIF.

        CLEAR: v_docno, v_gjahr, t_return.

        v_docno = et_tab-mblnr.
        v_gjahr = et_tab-mjahr.
        CALL FUNCTION 'BAPI_GOODSMVT_CANCEL'
          EXPORTING
            materialdocument = v_docno
            matdocumentyear  = v_gjahr
          TABLES
            return           = t_return.

        IF t_return IS INITIAL.
          CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
            EXPORTING
              wait = 'X'.
        ELSE.
          CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

          LOOP AT t_return INTO s_return.

          ENDLOOP.
        ENDIF.

      ENDLOOP.

  ENDCASE.


ENDFUNCTION.
