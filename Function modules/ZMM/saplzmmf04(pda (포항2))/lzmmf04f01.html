                           p_gi-rspos

                           p_gi-wempf

           werks
           lgort
           umlgo
           wempf

            p_gi-rspos,
            p_gi-bwart,
            p_gi-matnr,
          "  p_gi-bdmng,
            p_gi-meins,

            p_gi-lgort,
            p_gi-umlgo,
            p_gi-wempf
            )

     WHERE rsnum = p_rsnum
