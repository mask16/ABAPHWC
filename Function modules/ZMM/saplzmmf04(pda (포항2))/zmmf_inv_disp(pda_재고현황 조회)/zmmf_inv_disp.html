FUNCTION zmmf_inv_disp.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BARCODE) TYPE ZEBARCODE
*" EXPORTING
*" VALUE(E_RET) TYPE ZERETURN
*" VALUE(E_MSG) TYPE ZEMESSAGE
*" TABLES
*" ET_TAB STRUCTURE ZMMS0017
*"----------------------------------------------------------------------
* Global data declarations

  DATA: v_matnr TYPE matnr.

  CHECK i_barcode IS NOT INITIAL.

  CLEAR: v_matnr.
  SELECT SINGLE matnr
  INTO (v_matnr)
  FROM zmmt0090
  WHERE barcode = i_barcode.


  IF sy-subrc = 0.
    CLEAR: et_tab, et_tab[].

    SELECT a~matnr a~werks a~lgort a~labst
           b~meins
           c~maktx
           d~lgobe
    INTO CORRESPONDING FIELDS OF TABLE et_tab
    FROM mard AS a
    JOIN mara AS b ON a~matnr = b~matnr
    LEFT JOIN makt AS c ON b~matnr = c~matnr
                        AND c~spras = sy-langu
    LEFT JOIN t001l AS d ON d~werks = a~werks
                         AND d~lgort = a~lgort
    WHERE a~matnr = v_matnr
      AND a~werks = '1040'
      AND a~labst <> 0.


    LOOP AT et_tab.
      CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
        EXPORTING
          input  = et_tab-matnr
        IMPORTING
          output = et_tab-matnr.

      MODIFY et_tab.
    ENDLOOP.

  ELSE.
    e_ret = 'W'.
    e_msg  = text-006.

  ENDIF.



ENDFUNCTION.
