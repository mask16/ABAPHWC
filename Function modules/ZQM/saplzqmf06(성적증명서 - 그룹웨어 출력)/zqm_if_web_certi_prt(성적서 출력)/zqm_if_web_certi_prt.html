FUNCTION zqm_if_web_certi_prt.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_DATA) LIKE ZQMS0440 STRUCTURE ZQMS0440 OPTIONAL
*" VALUE(I_LANGU) TYPE CHAR2 OPTIONAL
*" TABLES
*" PDF_DATA STRUCTURE HRB2A_RAW255 OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : s_itab TYPE TABLE OF rsparams WITH HEADER LINE.
  DATA : p_langu TYPE langu.

  IF i_data IS NOT INITIAL.
    IF i_data-bukrs IS NOT INITIAL.
      CLEAR s_itab.
      s_itab-selname = 'P_BUKRS'.
      s_itab-kind = 'P'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-bukrs.
      APPEND s_itab.
    ENDIF.

    IF i_data-kunnr IS NOT INITIAL.

      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = i_data-kunnr
        IMPORTING
          output = i_data-kunnr.

      CLEAR s_itab.
      s_itab-selname = 'S_KUNNR'.
      s_itab-kind = 's'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-kunnr.
      APPEND s_itab.
    ENDIF.

    IF i_data-zqmnr IS NOT INITIAL.
      CLEAR s_itab.
      s_itab-selname = 'S_ZQMNR'.
      s_itab-kind = 'S'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-zqmnr.
      APPEND s_itab.
    ENDIF.

    IF i_data-matnr IS NOT INITIAL.

      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input  = i_data-matnr
        IMPORTING
          output = i_data-matnr.


      CLEAR s_itab.
      s_itab-selname = 'S_MATNR'.
      s_itab-kind = 'S'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-matnr.
      APPEND s_itab.
    ENDIF.

    IF i_data-matnr2 IS NOT INITIAL.
      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input  = i_data-matnr2
        IMPORTING
          output = i_data-matnr2.

      CLEAR s_itab.
      s_itab-selname = 'S_MATNR2'.
      s_itab-kind = 'S'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-matnr2.
      APPEND s_itab.
    ENDIF.

    IF i_data-reportno IS NOT INITIAL.
      CLEAR s_itab.
      s_itab-selname = 'S_REPORT'.
      s_itab-kind = 'S'.
      s_itab-sign = 'I'.
      s_itab-option = 'EQ'.
      s_itab-low = i_data-reportno.
      APPEND s_itab.
    ENDIF.

    CLEAR s_itab.
    s_itab-selname = 'P_R1'.
    s_itab-kind = 'P'.
    s_itab-sign = 'I'.
    s_itab-option = 'EQ'.
    s_itab-low = ''.
    APPEND s_itab.

    CLEAR s_itab.
    s_itab-selname = 'P_R2'.
    s_itab-kind = 'P'.
    s_itab-sign = 'I'.
    s_itab-option = 'EQ'.
    s_itab-low = 'X'.
    APPEND s_itab.

    IF i_langu EQ 'EN'.
      p_langu = 'E'.
    ELSEIF i_langu EQ 'KO'.
      p_langu = '3'.
    ELSEIF i_langu EQ 'ZH'.
      p_langu = '1'.
    ENDIF.

    CLEAR s_itab.
    s_itab-selname = 'P_LANG'.
    s_itab-kind = 'P'.
    s_itab-sign = 'I'.
    s_itab-option = 'EQ'.
    s_itab-low = p_langu.
    APPEND s_itab.

    CLEAR s_itab.
    s_itab-selname = 'P_CALL'.
    s_itab-kind = 'P'.
    s_itab-sign = 'I'.
    s_itab-option = 'EQ'.
    s_itab-low = 'X'.
    APPEND s_itab.

    SUBMIT zqmr0240 WITH SELECTION-TABLE s_itab AND RETURN.

    IMPORT pdf_data[] FROM MEMORY ID 'ZQMR0240_PRT'.

  ENDIF.

ENDFUNCTION.
