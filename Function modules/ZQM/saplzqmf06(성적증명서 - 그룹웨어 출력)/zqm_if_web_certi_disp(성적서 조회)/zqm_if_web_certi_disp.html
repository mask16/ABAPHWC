FUNCTION ZQM_IF_WEB_CERTI_DISP.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_KUNNR) TYPE ZQMT0090-KUNNR OPTIONAL
*" VALUE(I_FRGIDAT) TYPE ZQMT0090-GIDATE OPTIONAL
*" VALUE(I_TOGIDAT) TYPE ZQMT0090-GIDATE OPTIONAL
*" VALUE(I_LOTNO) TYPE ZQMT0090-LOTNO OPTIONAL
*" TABLES
*" T_DATA STRUCTURE ZQMS0440 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_KUNNR LIKE ZQMS0440-KUNNR.

  IF I_KUNNR IS NOT INITIAL.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        INPUT  = I_KUNNR
      IMPORTING
        OUTPUT = I_KUNNR.
  ENDIF.

  PERFORM CREATE_RANGES  USING I_KUNNR
                               I_FRGIDAT
                               I_TOGIDAT
                               I_LOTNO.

  INIT T_DATA.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE T_DATA
    FROM ZQMT0440 AS A
         INNER JOIN ZQMT0420 AS B
              ON A~BUKRS EQ B~BUKRS
             AND A~ZQMNR EQ B~ZQMNR
      WHERE A~BUKRS    EQ '1000'   "그룹웨어 조건 없음.
        AND A~KUNNR    IN GR_KUNNR
        AND A~GIDATE   IN GR_GIDATE
        AND A~APPROVE  EQ 'X'      "그룹웨어는 승인완료건만
        AND ( A~ZLOT_NO1  IN GR_LOTNO OR A~ZLOT_NO2 IN GR_LOTNO ).

  LOOP AT T_DATA.
* 고객이름
    SELECT SINGLE NAME1
       INTO T_DATA-NAME1
       FROM KNA1
        WHERE KUNNR EQ T_DATA-KUNNR.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        INPUT  = T_DATA-KUNNR
      IMPORTING
        OUTPUT = T_DATA-KUNNR.

    CALL FUNCTION 'CONVERSION_EXIT_MATN1_OUTPUT'
      EXPORTING
        INPUT  = T_DATA-MATNR
      IMPORTING
        OUTPUT = T_DATA-MATNR.

    MODIFY T_DATA.
    CLEAR T_DATA.
  ENDLOOP.

  SORT T_DATA BY GIDATE.


ENDFUNCTION.
