FUNCTION ZQM_IF_WEB_KUNNR.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BUKRS) TYPE BUKRS DEFAULT '1000'
*" VALUE(I_KUNNR) TYPE KUNNR OPTIONAL
*" VALUE(I_NAME1) TYPE NAME1 OPTIONAL
*" TABLES
*" T_DATA STRUCTURE ZQMS9998 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  INIT GR_BUKRS.
  IF I_BUKRS IS NOT INITIAL.
    GR_BUKRS     = 'IEQ'.
    GR_BUKRS-LOW = I_BUKRS.
    APPEND GR_BUKRS.
    CLEAR GR_BUKRS.
  ENDIF.

  IF I_KUNNR IS NOT INITIAL.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        INPUT  = I_KUNNR
      IMPORTING
        OUTPUT = I_KUNNR.
  ENDIF.

  INIT GR_KUNNR.
  IF I_KUNNR IS NOT INITIAL.
    GR_KUNNR     = 'IEQ'.
    GR_KUNNR-LOW = I_KUNNR.
    APPEND GR_KUNNR.
    CLEAR GR_KUNNR.
  ENDIF.

  INIT GR_NAME1.
  IF I_NAME1 IS NOT INITIAL.
    GR_NAME1     = 'ICP'.
    CONCATENATE '*' I_NAME1 '*' INTO GR_NAME1-LOW.

    APPEND GR_NAME1.
    CLEAR GR_NAME1.
  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE T_DATA
    FROM KNB1 AS A
       INNER JOIN KNA1 AS B
               ON A~KUNNR EQ B~KUNNR
       WHERE A~BUKRS IN GR_BUKRS
         AND B~KUNNR IN GR_KUNNR
         AND B~NAME1 IN GR_NAME1.

  DELETE T_DATA WHERE KUNNR CP 'A*'.

  LOOP AT T_DATA.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        INPUT  = T_DATA-KUNNR
      IMPORTING
        OUTPUT = T_DATA-KUNNR.

    MODIFY T_DATA.
    CLEAR T_DATA.
  ENDLOOP.



ENDFUNCTION.
