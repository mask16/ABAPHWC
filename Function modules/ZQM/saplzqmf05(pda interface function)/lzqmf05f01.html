FORM create_ranges USING p_kunnr p_frgidat p_togidat
                         p_vkbur p_matnr p_zmaktx p_lotno.

  CLEAR : gr_kunnr, gr_kunnr[],
          gr_gidate, gr_gidate[],
          gr_vkbur, gr_vkbur[],
          gr_matnr, gr_matnr[],
          gr_zmaktx, gr_zmaktx[],
          gr_lotno, gr_lotno[].

  IF p_kunnr <> ''.
    gr_kunnr-sign = 'I'. gr_kunnr-option = 'EQ'. gr_kunnr-low = p_kunnr.
    APPEND gr_kunnr. CLEAR gr_kunnr.
  ENDIF.
  IF p_frgidat <> ''.
    IF p_togidat <> ''.
      gr_gidate-sign = 'I'. gr_gidate-option = 'BT'.
      gr_gidate-low = p_frgidat. gr_gidate-high = p_togidat.
      APPEND gr_gidate. CLEAR gr_gidate.
    ELSE.
      gr_gidate-sign = 'I'. gr_gidate-option = 'EQ'. gr_gidate-low = p_frgidat.
      APPEND gr_gidate. CLEAR gr_gidate.
    ENDIF.
  ENDIF.
  IF p_vkbur <> ''.
    gr_vkbur-sign = 'I'. gr_vkbur-option = 'EQ'.   gr_vkbur-low = p_vkbur.
    APPEND gr_vkbur. CLEAR gr_vkbur.
  ENDIF.
  IF p_matnr <> ''.
    gr_matnr-sign = 'I'. gr_matnr-option = 'EQ'.   gr_matnr-low = p_matnr.
    APPEND gr_matnr. CLEAR gr_matnr.
  ENDIF.
  IF p_lotno <> ''.
    gr_lotno-sign = 'I'. gr_lotno-option = 'EQ'.   gr_lotno-low = p_lotno.
    APPEND gr_lotno. CLEAR gr_lotno.
  ENDIF.
  IF p_zmaktx <> ''.
    gr_zmaktx-sign = 'I'. gr_zmaktx-option = 'CP'.
    CONCATENATE '*' p_zmaktx '*' INTO gr_zmaktx-low.
    APPEND gr_zmaktx. CLEAR gr_zmaktx.
  ENDIF.

ENDFORM. " CREATE_RANGES

FORM base_select_data TABLES pt_tab STRUCTURE zqms0210.

  DATA : lt_base TYPE STANDARD TABLE OF zqms0211,
         ls_base TYPE zqms0211,
         ls_tab TYPE zqms0210.


  SELECT kunnr name1 gidate vkbur bezei kunn2 name2
         matnr maktx zmaktx lotno groes1
         matnr2 maktx2 zmaktx2 lotno2 groes2
         spec spec2 entype cmtrtype oemtype
         revitype reportno purorderno issuedate
         vbeln vbelnno quantity unit
         text1 text2 text3
    FROM zqmt0090
    INTO CORRESPONDING FIELDS OF TABLE lt_base
   WHERE kunnr IN gr_kunnr AND gidate IN gr_gidate
     AND vkbur IN gr_vkbur AND matnr IN gr_matnr
     AND zmaktx IN gr_zmaktx AND lotno IN gr_lotno.

  IF lt_base[] IS NOT INITIAL.
    LOOP AT lt_base INTO ls_base.
      MOVE-CORRESPONDING ls_base TO ls_tab.
      WRITE ls_base-quantity TO ls_tab-quantity UNIT ls_base-unit.
      APPEND ls_tab TO pt_tab. CLEAR ls_tab.
    ENDLOOP.
  ENDIF.

ENDFORM. " BASE_SELECT_DATA
