FUNCTION zpp_if_web_fg_tf.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_FLAG) TYPE ZEWEBFLAG OPTIONAL
*" VALUE(I_MTNUM) LIKE ZSDT5010-MTNUM OPTIONAL
*" VALUE(I_USERID) LIKE ZSDT5000-USERID OPTIONAL
*" VALUE(I_POSDT) LIKE LIKP-ERDAT DEFAULT SY-DATUM
*" VALUE(I_FRSLOC) LIKE ZSDT5010-FRSLOC OPTIONAL
*" VALUE(I_TOSLOC) LIKE ZSDT5010-TOSLOC OPTIONAL
*" VALUE(I_BARCOD) LIKE ZSDT5010-BARCODE OPTIONAL
*" EXPORTING
*" VALUE(E_MTNUM) LIKE ZSDT5010-MTNUM
*" VALUE(E_RET) LIKE ZSDS0001-RET
*" VALUE(E_MSG) LIKE ZSDS0001-MSG
*" TABLES
*" T_TAB STRUCTURE ZSDS0230
*"----------------------------------------------------------------------
* Global data declarations

  DATA : l_gubun  LIKE zsdt5000-gubun,
         l_plant1 LIKE zsdt5000-plant1.

  DATA : lt_tab TYPE STANDARD TABLE OF zsds0231.

  CLEAR : e_mtnum, e_ret, e_msg, g_bzgbn.

  SELECT SINGLE gubun plant1 INTO (l_gubun, l_plant1)
    FROM zsdt5000 WHERE userid EQ i_userid.

  g_bzgbn = text-007.

  IF l_gubun = text-002.
    e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e01.
  ELSEIF l_gubun = text-001.
    IF l_plant1 <> text-003.
      e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e01.
    ELSE.
      CASE i_flag.
*"----------------------------------------------------------------------
*" Display
*"----------------------------------------------------------------------
        WHEN 'D'.
          PERFORM display_tf_data TABLES t_tab
                                   USING i_barcod i_userid i_mtnum
                                         i_frsloc i_tosloc
                                         e_mtnum e_ret e_msg.
*"----------------------------------------------------------------------
*" Reading
*"----------------------------------------------------------------------
        WHEN 'R'.
          IF i_barcod IS INITIAL.
            e_mtnum = i_mtnum. e_msg = text-e02. e_ret = 'E'.
          ELSE.
            PERFORM reading_tf_data TABLES t_tab
                                     USING i_barcod i_userid i_mtnum
                                           i_frsloc i_tosloc
                                           e_mtnum e_ret e_msg.
          ENDIF.
          PERFORM refresh_tf_data TABLES t_tab
                                   USING i_barcod i_userid i_mtnum
                                         i_frsloc i_tosloc.
*"----------------------------------------------------------------------
*" Saving
*"----------------------------------------------------------------------
        WHEN 'S'.
          IF t_tab[] IS INITIAL.
            e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e13.
          ELSE.
            PERFORM save_tf_data TABLES t_tab
                                    USING i_barcod i_userid i_mtnum
                                          i_frsloc i_tosloc
                                          e_mtnum e_ret e_msg.
          ENDIF.
          PERFORM refresh_tf_data TABLES t_tab
                                   USING i_barcod i_userid i_mtnum
                                         i_frsloc i_tosloc.

*"----------------------------------------------------------------------
*" Deleting
*"----------------------------------------------------------------------
        WHEN 'X'.
          IF t_tab[] IS INITIAL.
            e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e06.
          ELSE.
            PERFORM delete_tf_data TABLES t_tab
                                    USING i_barcod i_userid i_mtnum
                                          i_frsloc i_tosloc
                                          e_mtnum e_ret e_msg.
          ENDIF.
          PERFORM refresh_tf_data TABLES t_tab
                                   USING i_barcod i_userid i_mtnum
                                         i_frsloc i_tosloc.
*"----------------------------------------------------------------------
*" ALL Deleting
*"----------------------------------------------------------------------
        WHEN 'A'.
          IF t_tab[] IS INITIAL.
            e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e06.
          ELSE.
            PERFORM delete_tf_data TABLES t_tab
                                    USING i_barcod i_userid i_mtnum
                                          i_frsloc i_tosloc
                                          e_mtnum e_ret e_msg.
          ENDIF.
          PERFORM refresh_tf_data TABLES t_tab
                                   USING i_barcod i_userid i_mtnum
                                         i_frsloc i_tosloc.
*"----------------------------------------------------------------------
*" 이전처리 - PT
*"----------------------------------------------------------------------
        WHEN 'T'.
          IF t_tab[] IS INITIAL.
            e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e09.
          ELSE.
            PERFORM pt_data TABLES t_tab
                             USING i_barcod i_userid i_mtnum
                                   i_frsloc i_tosloc i_posdt
                                   e_mtnum e_ret e_msg.
          ENDIF.
          IF e_msg = text-e10.
            PERFORM refresh_tf_data TABLES t_tab
                           USING i_barcod i_userid i_mtnum
                                 i_frsloc i_tosloc.
          ELSE.
            PERFORM refresh_fg_data TABLES t_tab
                                     USING i_barcod i_userid i_mtnum
                                           i_frsloc i_tosloc.
          ENDIF.
        WHEN OTHERS.
          e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e11.
      ENDCASE.
    ENDIF.
  ELSE.
    e_mtnum = i_mtnum. e_ret = 'E'. e_msg = text-e07.
  ENDIF.

ENDFUNCTION.
