                      USING p_barcod p_userid p_mtnum
                            p_frsloc p_tosloc
                            p_etnum p_ret p_msg.

  DATA : lt_tab  TYPE STANDARD TABLE OF zsds0231,
         ls_tab  TYPE zsds0231.

  DATA : l_recod TYPE zsdt5010-recod,
         l_mtnum LIKE  zsdt5010-mtnum,
         l_ret LIKE  zsds0001-ret,
         l_msg LIKE  zsds0001-msg.

  IF strlen( p_barcod ) = 19.
*-- 생산 바코드 라벨 관리 TABLE
    SELECT a~matnr a~barcode a~pbarcode
           c~maktx a~menge a~meins a~werks
           a~pcharg AS lotmat a~qcharg AS qalot
           a~charg b~name1
      FROM zppt0170 AS a LEFT OUTER JOIN t001w AS b
                           ON a~werks = b~werks
                         LEFT OUTER JOIN makt AS c
                           ON a~matnr = c~matnr
                          AND c~spras = sy-langu
      INTO CORRESPONDING FIELDS OF TABLE lt_tab
     WHERE a~barcode EQ p_barcod AND a~werks EQ text-003.

      ls_tab-tosloc = p_tosloc.
      MODIFY lt_tab FROM ls_tab. CLEAR ls_tab.

     WHERE barcode EQ p_barcod AND werks EQ text-003
       AND bzgbn EQ g_bzgbn.

    IF l_recod = 'S'.
      CLEAR : lt_tab, lt_tab[].
      l_mtnum = p_mtnum. l_ret = 'E'. l_msg = text-e15.
    ENDIF.

*-- PDA LOG TABLE
    SELECT a~mtnum a~barcode a~pbarcode
           a~lotmat a~matnr a~charg
           a~bzgbn c~maktx a~menge
           a~meins a~refno a~repos
           a~werks a~frsloc a~prddt
           a~qalot b~name1 a~frsloc a~tosloc
           a~userid
      FROM zsdt5010 AS a LEFT OUTER JOIN t001w AS b
                           ON a~werks = b~werks
                         LEFT OUTER JOIN makt AS c
                           ON a~matnr = c~matnr
                          AND c~spras = sy-langu
  APPENDING CORRESPONDING FIELDS OF TABLE lt_tab
      WHERE a~werks EQ text-003 AND a~recod EQ '' AND a~bzgbn EQ g_bzgbn.
*-- 중복 된 DATA 제거
    SORT lt_tab BY mtnum DESCENDING barcode ASCENDING.

  ENDIF.

  IF l_ret = 'E'.

FORM saving_tf_sd TABLES pt_tab STRUCTURE zsds0231
                   USING p_barcod p_userid p_mtnum
                         p_frsloc p_tosloc
                         p_etnum p_ret p_msg.

  DATA : l_nrran      LIKE inri-nrrangenr,
         l_object     LIKE inri-object,
         l_retcd      LIKE inri-returncode,
         l_number(13) TYPE c,
         l_erdat      TYPE sy-datum,
         l_erzet      TYPE sy-uzeit,
         l_ernam      TYPE sy-uname.

  DATA : lt_tab TYPE STANDARD TABLE OF zsdt5010,
         ls_tab TYPE zsdt5010,
         ls_231 TYPE zsds0231.

  DATA : l_mtnum TYPE zsdt5010-mtnum.

  LOOP AT pt_tab INTO ls_231.
    IF ls_231-mtnum = ''.
      ls_231-mtnum = p_mtnum = l_mtnum.
      IF p_mtnum = ''.
        l_object = 'ZSD_MTNUM'.
        l_nrran = '01'.

        CALL FUNCTION 'NUMBER_GET_NEXT'
          EXPORTING
            nr_range_nr             = l_nrran
            object                  = l_object
          IMPORTING
            number                  = l_number
            returncode              = l_retcd
          EXCEPTIONS
            interval_not_found      = 1
            number_range_not_intern = 2
            object_not_found        = 3
            quantity_is_0           = 4
            quantity_is_not_1       = 5
            interval_overflow       = 6
            buffer_overflow         = 7
            OTHERS                  = 8.

        IF sy-subrc <> 0.
          MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                  WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
        ELSE.
          CONCATENATE g_bzgbn l_number INTO p_mtnum.
          l_mtnum = p_mtnum.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.

  CLEAR : ls_231.

  LOOP AT pt_tab INTO ls_231.
    MOVE-CORRESPONDING ls_231 TO ls_tab.
    ls_tab-userid = p_userid.
    ls_tab-mandt = sy-mandt.
    ls_tab-frsloc = p_frsloc.
    ls_tab-tosloc = p_tosloc.
    IF ls_231-mtnum = ''.
      ls_tab-mtnum = p_mtnum.
      ls_231-mtnum = p_mtnum.
      ls_tab-erdat = sy-datum.
      ls_tab-erzet = sy-uzeit.
      ls_tab-ernam = sy-uname.
    ELSE.
      SELECT SINGLE erdat erzet ernam INTO (l_erdat, l_erzet, l_ernam)
        FROM zsdt5010 WHERE mtnum EQ ls_231-mtnum.
      ls_tab-erdat = l_erdat.
      ls_tab-erzet = l_erzet.
      ls_tab-ernam = l_ernam.
    ENDIF.
    ls_tab-updat = sy-datum.
    ls_tab-uptim = sy-uzeit.
    ls_tab-upnam = sy-uname.
    APPEND ls_tab TO lt_tab. CLEAR : ls_tab.
    MODIFY pt_tab FROM ls_231. CLEAR : ls_231.
  ENDLOOP.

  MODIFY zsdt5010 FROM TABLE lt_tab.
  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.
    p_etnum = p_mtnum. p_ret = 'S'. p_msg = text-s01.
  ELSE.
    ROLLBACK WORK.
    p_etnum = p_mtnum. p_ret = 'E'. p_msg = text-e05.
  ENDIF.

ENDFORM. " SAVING_TF_SD

FORM pt_check TABLES pt_tab STRUCTURE zsds0231
               USING p_barcod p_userid p_mtnum
                     p_frsloc p_tosloc
                     p_etnum p_ret p_msg.

  TYPES : BEGIN OF typ_data_eac,
            matnr     TYPE matnr   ,
            werks     TYPE werks_d ,
            frsloc    TYPE zefrsloc,
            menge     TYPE menge_d ,
          END OF typ_data_eac.

  TYPES : BEGIN OF typ_data_com,
            matnr     TYPE matnr   ,
            werks     TYPE werks_d ,
            frsloc    TYPE zefrsloc,
            charg     TYPE charg_d ,
            menge     TYPE menge_d ,
          END OF typ_data_com.

  TYPES : BEGIN OF typ_data_sum,
            matnr     TYPE matnr   ,
            menge     TYPE menge_d ,
          END OF typ_data_sum.

  DATA : ls_tab   TYPE zsds0231,
         lt_spa   TYPE STANDARD TABLE OF zsds0231,
         ls_spa   TYPE zsds0231,
         lt_cha   TYPE STANDARD TABLE OF zsds0231,
         ls_cha   TYPE zsds0231,
         lt_mar   TYPE STANDARD TABLE OF mard,
         ls_mar   TYPE mard,
         lt_mch   TYPE STANDARD TABLE OF mchb,
         ls_mch   TYPE mchb,
         lt_com   TYPE STANDARD TABLE OF typ_data_com,
         ls_com   TYPE typ_data_com,
         lt_eac   TYPE STANDARD TABLE OF typ_data_eac,
         ls_eac   TYPE typ_data_eac,
         lt_sum   TYPE STANDARD TABLE OF typ_data_sum,
         ls_sum   TYPE typ_data_sum,
         lt_mum   TYPE STANDARD TABLE OF typ_data_sum,
         ls_mum   TYPE typ_data_sum,
         lt_cum   TYPE STANDARD TABLE OF typ_data_sum,
         ls_cum   TYPE typ_data_sum.

  DATA : l_menge   TYPE menge_d,
         l_flag(1) TYPE c.

*-- Batch 존재 유무에 따른 그룹핑
  LOOP AT pt_tab INTO ls_tab.
    IF ls_tab-charg = ''.
      MOVE-CORRESPONDING ls_tab TO ls_spa.
      APPEND ls_spa TO lt_spa. CLEAR : ls_tab, ls_spa.
    ELSE.
      MOVE-CORRESPONDING ls_tab TO ls_cha.
      APPEND ls_cha TO lt_cha. CLEAR : ls_tab, ls_cha.
    ENDIF.
  ENDLOOP.
*-- Batch 존재시 적합성 점검
  IF lt_cha[] IS NOT INITIAL.
    SELECT *
     INTO CORRESPONDING FIELDS OF TABLE lt_mch
     FROM mchb
      FOR ALL ENTRIES IN lt_cha
    WHERE matnr EQ lt_cha-matnr AND werks EQ text-003
      AND lgort EQ p_frsloc AND charg EQ lt_cha-charg.

    SORT lt_mch BY matnr werks lgort charg.
    CLEAR : l_menge, ls_mch.

    LOOP AT lt_mch INTO ls_mch.
      MOVE-CORRESPONDING ls_mch TO ls_cum.
      l_menge = ls_mch-clabs + l_menge.
      AT END OF charg.
        ls_cum-menge = l_menge.
        APPEND ls_cum TO lt_cum. CLEAR : ls_mch, ls_cum, l_menge.
      ENDAT.
    ENDLOOP.

    CLEAR : ls_cha, ls_com, lt_com, lt_com[],
            ls_sum, lt_sum, lt_sum[], l_menge.

    LOOP AT lt_cha INTO ls_cha.
      MOVE-CORRESPONDING ls_cha TO ls_com.
      APPEND ls_com TO lt_com. CLEAR : ls_com, ls_cha.
    ENDLOOP.

    SORT lt_com BY matnr werks frsloc charg.
    CLEAR : l_menge, ls_com.
    LOOP AT lt_com INTO ls_com.
      MOVE-CORRESPONDING ls_com TO ls_sum.
      l_menge = ls_com-menge + l_menge.
      AT END OF charg.
        ls_sum-menge = l_menge.
        APPEND ls_sum TO lt_sum. CLEAR : ls_com, ls_sum, l_menge.
      ENDAT.
    ENDLOOP.

    CLEAR l_flag.
    SORT lt_cum BY matnr. SORT lt_sum BY matnr.
