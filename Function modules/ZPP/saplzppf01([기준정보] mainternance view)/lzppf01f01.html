FORM CHLOG_ZPPV0050 .

  CLEAR ZPPV0050-KTEXT.

* 작업장내역
  IF ZPPV0050-ARBPL IS NOT INITIAL AND
     ZPPV0050-WERKS IS NOT INITIAL.
    SELECT SINGLE B~KTEXT_UP INTO ZPPV0050-KTEXT
           FROM CRHD AS A INNER JOIN CRTX AS B ON A~OBJTY = B~OBJTY AND
                                                  A~OBJID = B~OBJID
           WHERE A~OBJTY = 'A' AND
                 A~WERKS = ZPPV0050-WERKS AND
                 A~ARBPL = ZPPV0050-ARBPL AND
                 B~SPRAS = SY-LANGU.
  ENDIF.

* 생성이력
  IF ZPPV0050-CREATED_BY IS INITIAL.
    ZPPV0050-CREATED_DATE = SY-DATUM.
    ZPPV0050-CREATED_TIME = SY-UZEIT.
    ZPPV0050-CREATED_BY   = SY-UNAME.
  ENDIF.

* 변경이력
  ZPPV0050-CHANGED_DATE = SY-DATUM.
  ZPPV0050-CHANGED_TIME = SY-UZEIT.
  ZPPV0050-CHANGED_BY   = SY-UNAME.

FORM CHECK_ZPPV0050 .

  TABLES CRHD.

* 작업장 체크

         WHERE OBJTY = 'A' AND

FORM CHLOG_ZPPV0180.
