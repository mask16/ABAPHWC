MODULE ETC_UPDATE_LISTE INPUT.

  IF ZPPT0003-KOKRS IS INITIAL.
    ZPPT0003-KOKRS = '1000'.
  ENDIF.
  IF ZPPT0003-KOSTL IS INITIAL.
    MESSAGE E001(ZPPM) WITH TEXT-M01. "코스트센터를 입력하세요.
  ENDIF.

  IF ZPPT0003-KOKRS IS NOT INITIAL AND ZPPT0003-KOSTL IS NOT INITIAL.
    SELECT SINGLE KTEXT INTO ZPPT0003-KTEXT
      FROM CSKT
     WHERE SPRAS = SY-LANGU
       AND KOKRS = ZPPT0003-KOKRS
       AND KOSTL = ZPPT0003-KOSTL
       AND DATBI > SY-DATLO.
  ENDIF.

  IF ZPPT0003-CREATED_BY IS INITIAL.
    ZPPT0003-CREATED_DATE = SY-DATUM.
    ZPPT0003-CREATED_TIME = SY-UZEIT.
    ZPPT0003-CREATED_BY   = SY-UNAME.

  ENDIF.
  ZPPT0003-CHANGED_DATE = SY-DATUM.
  ZPPT0003-CHANGED_TIME = SY-UZEIT.
  ZPPT0003-CHANGED_BY   = SY-UNAME.

ENDMODULE. " ETC_UPDATE_LISTE INPUT
