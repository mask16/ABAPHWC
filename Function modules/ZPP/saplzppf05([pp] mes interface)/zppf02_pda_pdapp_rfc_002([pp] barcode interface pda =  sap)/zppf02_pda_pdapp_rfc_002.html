FUNCTION ZPPF02_PDA_PDAPP_RFC_002.
*"----------------------------------------------------------------------
*"*"Local interface:
*" EXPORTING
*" VALUE(E_RET) TYPE ZERETURN
*" VALUE(E_MSG) TYPE ZEMESSAGE
*" TABLES
*" T_TAB STRUCTURE ZPPS0170
*"----------------------------------------------------------------------
*-----------------------------------------------------------------------
* PROGRAM : ZPPF02_PDA_PDAPP_RFC_002
* DESCRIPTION : [PP] Barcode Interface PDA => SAP
* DEVELOPER : AB01
* CREATED DATE : 2013.03.03
*-----------------------------------------------------------------------

*-----------------------------------------------------------------------

* Global data declarations

  DATA : LT_T170     LIKE TABLE OF ZPPT0170 WITH HEADER LINE,
         LV_MODE.

* 테이블에 한 건씩 들어옴
  IF T_TAB[] IS INITIAL.
    E_RET = 'E'. E_MSG = TEXT-006. EXIT.
  ENDIF.

  LOOP AT T_TAB.
    MOVE-CORRESPONDING T_TAB TO LT_T170.
    LV_MODE = T_TAB-FLAG.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        INPUT  = LT_T170-MATNR
      IMPORTING
        OUTPUT = LT_T170-MATNR.

    CASE LV_MODE.
      WHEN 'C'.
* 생성로그
        LT_T170-CREATED_DATE = SY-DATUM.
        LT_T170-CREATED_TIME = SY-UZEIT.
        LT_T170-CREATED_BY = SY-UNAME.
      WHEN 'U'.
* 변경로그
        LT_T170-CHANGED_DATE = SY-DATUM.
        LT_T170-CHANGED_TIME = SY-UZEIT.
        LT_T170-CHANGED_BY = SY-UNAME.
    ENDCASE.

    APPEND LT_T170.
  ENDLOOP.

  CASE LV_MODE.
    WHEN 'C'.
      INSERT ZPPT0170 FROM TABLE LT_T170.
    WHEN 'U'.
      MODIFY ZPPT0170 FROM TABLE LT_T170.
    WHEN 'D'.
      DELETE ZPPT0170 FROM TABLE LT_T170.
  ENDCASE.

  IF SY-SUBRC EQ 0.
    E_RET = 'S'. E_MSG = TEXT-009.
  ELSE.
    CASE LV_MODE.
      WHEN 'C'.
        E_RET = 'E'. E_MSG = TEXT-002.
      WHEN 'U'.
        E_RET = 'E'. E_MSG = TEXT-001.
      WHEN 'D'.
        E_RET = 'E'. E_MSG = TEXT-003.
    ENDCASE.
  ENDIF.

ENDFUNCTION.
