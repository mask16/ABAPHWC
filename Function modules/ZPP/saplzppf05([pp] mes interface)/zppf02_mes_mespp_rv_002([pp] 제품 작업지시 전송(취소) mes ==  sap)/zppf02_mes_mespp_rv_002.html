FUNCTION ZPPF02_MES_MESPP_RV_002.
*"----------------------------------------------------------------------
*"*"Local interface:
*" EXPORTING
*" VALUE(E_RET) TYPE ZERETURN
*" VALUE(E_MSG) TYPE ZEMESSAGE
*" TABLES
*" T_TAB STRUCTURE ZPPS0070
*"----------------------------------------------------------------------
*-----------------------------------------------------------------------
* PROGRAM : ZPPF02_MES_MESPP_RV_002
* DESCRIPTION : [생산계획] 공정별 생산실적 전송(취소) MES ==> SAP
* DEVELOPER : AB01
* CREATED DATE : 2012.12.07
*-----------------------------------------------------------------------
* HISTORY :
* DEVELOPER DATE DESCRIPTION
* AB01 2012.12.07 CREATION
*
*-----------------------------------------------------------------------

* Global data declarations

  DATA : LT_T110      LIKE TABLE OF ZPPT0110 WITH HEADER LINE,
         LT_T111      LIKE TABLE OF ZPPT0111 WITH HEADER LINE,
         LT_T112      LIKE TABLE OF ZPPT0112 WITH HEADER LINE.

  CHECK T_TAB[] IS NOT INITIAL.

  LOOP AT T_TAB.
    MOVE-CORRESPONDING T_TAB TO LT_T111.

    MOVE : T_TAB-PLNUM   TO LT_T111-ZPLNUM.

* 생성로그
    LT_T111-CREATED_DATE = SY-DATUM.
    LT_T111-CREATED_TIME = SY-UZEIT.
    LT_T111-CREATED_BY   = SY-UNAME.

    APPEND LT_T111.
  ENDLOOP.

  IF LT_T111[] IS NOT INITIAL.
    MODIFY ZPPT0111 FROM TABLE LT_T111.
    IF SY-SUBRC EQ 0.
      E_RET = 'S'.
    ELSE.
      E_RET = 'E'.
      E_MSG = TEXT-001.
    ENDIF.
  ENDIF.

  IF E_RET = 'S'.
    SORT LT_T111.

    SELECT * INTO TABLE LT_T110
           FROM ZPPT0110
           FOR ALL ENTRIES IN LT_T111
           WHERE  WERKS   = LT_T111-WERKS AND
                  ZPLNUM  = LT_T111-ZPLNUM AND
                  ARBPL   = LT_T111-ARBPL AND
                  HOGI    = LT_T111-HOGI AND
                  SHIFT   = LT_T111-SHIFT AND
                  T_TIME  = LT_T111-T_TIME.
    SORT LT_T110.

    SELECT * INTO TABLE LT_T112
           FROM ZPPT0112
           FOR ALL ENTRIES IN LT_T111
           WHERE  WERKS   = LT_T111-WERKS AND
                  ZPLNUM  = LT_T111-ZPLNUM AND
                  ARBPL   = LT_T111-ARBPL AND
                  HOGI    = LT_T111-HOGI AND
                  SHIFT   = LT_T111-SHIFT AND
                  T_TIME  = LT_T111-T_TIME.
    SORT LT_T112.

    LOOP AT LT_T111.
      IF LT_T111-GUBUN = 'X'.  "코일 투입정보
        READ TABLE LT_T112 WITH KEY WERKS   = LT_T111-WERKS
                                    ZPLNUM  = LT_T111-ZPLNUM
                                    ARBPL   = LT_T111-ARBPL
                                    HOGI    = LT_T111-HOGI
                                    SHIFT   = LT_T111-SHIFT
                                    T_TIME  = LT_T111-T_TIME
                                    BINARY SEARCH.
        IF SY-SUBRC EQ 0.
* 처리전/에러건은 MES 삭제로 변경
          IF LT_T112-STATUS = 'A' OR LT_T112-STATUS = 'E'.
            UPDATE ZPPT0112 SET
                   STATUS = 'X'
                   WHERE ZPLNUM = LT_T111-ZPLNUM AND
                         ARBPL  = LT_T111-ARBPL AND
                         HOGI   = LT_T111-HOGI AND
                         SHIFT  = LT_T111-SHIFT AND
                         T_TIME = LT_T111-T_TIME.
* 처리성공건은 실적취소 대상으로 변경
          ELSEIF LT_T112-STATUS = 'S'.
            UPDATE ZPPT0112 SET
                   STATUS = 'Z'
                   WHERE ZPLNUM = LT_T111-ZPLNUM AND
                         ARBPL  = LT_T111-ARBPL AND
                         HOGI   = LT_T111-HOGI AND
                         SHIFT  = LT_T111-SHIFT AND
                         T_TIME = LT_T111-T_TIME.
          ENDIF.
        ENDIF.
      ELSE.
        READ TABLE LT_T110 WITH KEY WERKS   = LT_T111-WERKS
                                    ZPLNUM  = LT_T111-ZPLNUM
                                    ARBPL   = LT_T111-ARBPL
                                    HOGI    = LT_T111-HOGI
                                    SHIFT   = LT_T111-SHIFT
                                    T_TIME  = LT_T111-T_TIME
                                    BINARY SEARCH.
        IF SY-SUBRC EQ 0.
* 처리전/에러건은 MES 삭제로 변경
          IF LT_T110-STATUS = 'A' OR LT_T110-STATUS = 'E'.
            UPDATE ZPPT0110 SET
                   STATUS = 'X'
                   WHERE ZPLNUM = LT_T111-ZPLNUM AND
                         ARBPL  = LT_T111-ARBPL AND
                         HOGI   = LT_T111-HOGI AND
                         SHIFT  = LT_T111-SHIFT AND
                         T_TIME = LT_T111-T_TIME.
* 처리성공건은 실적취소 대상으로 변경
          ELSEIF LT_T110-STATUS = 'S'.
            UPDATE ZPPT0110 SET
                   STATUS = 'Z'
                   WHERE ZPLNUM = LT_T111-ZPLNUM AND
                         ARBPL  = LT_T111-ARBPL AND
                         HOGI   = LT_T111-HOGI AND
                         SHIFT  = LT_T111-SHIFT AND
                         T_TIME = LT_T111-T_TIME.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
