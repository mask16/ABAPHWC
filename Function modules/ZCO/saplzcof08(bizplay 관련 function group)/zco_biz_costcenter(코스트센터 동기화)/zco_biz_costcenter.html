FUNCTION ZCO_BIZ_COSTCENTER.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BUKRS) TYPE BUKRS
*" VALUE(I_DATE) TYPE DATUM
*" EXPORTING
*" VALUE(E_MSGTY) TYPE CHAR01
*" VALUE(E_MSGTX) TYPE CHAR200
*" TABLES
*" T_DATA STRUCTURE ZCOS007 OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA: LV_KOSTL TYPE CSKS-KOSTL,
        LV_LINE TYPE I.

  " 회사코드 존재유무 체크.
  CLEAR GV_BUKRS.
  SELECT SINGLE BUKRS
    INTO GV_BUKRS
    FROM T001
    WHERE BUKRS = I_BUKRS.

  IF SY-SUBRC <> 0.
    "유효한 회사코드가 아닙니다.
    E_MSGTX = TEXT-001.
    E_MSGTY = GC_ERROR.
  ELSE.
    SELECT A~BUKRS A~KOSTL B~KTEXT
      FROM CSKS AS A JOIN CSKT AS B
      ON A~KOKRS = B~KOKRS
      AND A~KOSTL = B~KOSTL
      AND A~DATBI = B~DATBI
      INTO CORRESPONDING FIELDS OF TABLE T_DATA
      WHERE A~KOKRS  = I_BUKRS
        AND A~DATBI >= I_DATE
        AND A~DATAB <= I_DATE
        AND B~SPRAS  = SY-LANGU.

    IF LINES( T_DATA ) IS INITIAL.
      "조회된 데이터가 없습니다.
      E_MSGTX = TEXT-002.
      E_MSGTY = GC_ERROR.
    ELSE.
      LOOP AT T_DATA.
        CLEAR LV_KOSTL.
        SELECT SINGLE KOSTL
          FROM CSKS
          INTO LV_KOSTL
          WHERE KOSTL = T_DATA-KOSTL
            AND KOKRS = T_DATA-BUKRS
            AND DATBI >= I_DATE
            AND DATAB <= I_DATE
            AND BKZKP = 'X'
            AND PKZKP = 'X'.

        PERFORM CONVERSION_OUTPUT USING T_DATA-KOSTL T_DATA-KOSTL.

        IF LV_KOSTL IS NOT INITIAL.
          T_DATA-USE_YN = 'N'.
        ELSE.
          T_DATA-USE_YN = 'Y'.
        ENDIF.

        MODIFY T_DATA TRANSPORTING KOSTL USE_YN.
      ENDLOOP.

      LV_LINE = LINES( T_DATA ).

      E_MSGTX = LV_LINE && TEXT-003.
      E_MSGTY = GC_SUCCESS.

    ENDIF.
  ENDIF.

ENDFUNCTION.
