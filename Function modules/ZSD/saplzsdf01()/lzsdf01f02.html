*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
FORM set_user_log.

  DATA: l_index      LIKE sy-tabix,
        ls_zsdt9100  TYPE zsdt9100.

  LOOP AT total.
*   신규생성, 수정일 경우
    CHECK <action> EQ neuer_eintrag OR
          <action> EQ aendern.

    READ TABLE extract WITH KEY <vim_xtotal_key>.
    IF sy-subrc EQ 0.
      l_index = sy-tabix.
    ELSE.
      CLEAR l_index.
    ENDIF.
    MOVE-CORRESPONDING total TO ls_zsdt9100.
*   신규생성일 경우
    IF <action> EQ neuer_eintrag.
      ls_zsdt9100-ernam = sy-uname.
      ls_zsdt9100-erdat = sy-datum.
      ls_zsdt9100-erzet = sy-uzeit.
      ls_zsdt9100-aenam = sy-uname.
      ls_zsdt9100-aedat = sy-datum.
      ls_zsdt9100-aezet = sy-uzeit.
    ENDIF.
*   수정일 경우
    IF <action> EQ aendern.
      ls_zsdt9100-aenam = sy-uname.
      ls_zsdt9100-aedat = sy-datum.
      ls_zsdt9100-aezet = sy-uzeit.
    ENDIF.
    MOVE-CORRESPONDING ls_zsdt9100 TO total.
    MODIFY total.
    CHECK l_index GT 0.
    extract = total.
    MODIFY extract INDEX l_index.
  ENDLOOP.
  sy-subrc = 0.

ENDFORM. "SET_USER_LOG
