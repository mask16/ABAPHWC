FUNCTION zsd_bill_payable.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BUKRS) TYPE VBRK-BUKRS
*" REFERENCE(I_FKDAT) TYPE SY-DATUM
*" EXPORTING
*" VALUE(E_PAYABLE) TYPE CHAR01
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_t001b   TYPE STANDARD TABLE OF t001b,
        ls_t001b   LIKE LINE OF lt_t001b,
        lv_version TYPE t001-periv,
        lv_period  TYPE coep-perio,
        lv_year    TYPE t009b-bdatj,
        lv_perbl   TYPE ce21000-perbl,
        lv_ferbl   TYPE ce21000-perbl,
        lv_terbl   TYPE ce21000-perbl.

  CLEAR: e_payable.
  lv_version = 'K4'.

  CALL FUNCTION 'DETERMINE_PERIOD'
    EXPORTING
      date    = i_fkdat
      version = lv_version
    IMPORTING
      period  = lv_period
      year    = lv_year.

  CONCATENATE lv_year lv_period
         INTO lv_perbl.

  SELECT frye1 frpe1 toye1 tope1
    INTO CORRESPONDING FIELDS OF TABLE lt_t001b
    FROM t001b
   WHERE bukrs EQ i_bukrs
     AND mkoar IN ('+', 'D', 'S').

  LOOP AT lt_t001b INTO ls_t001b.
    CLEAR: lv_ferbl, lv_terbl.
    CONCATENATE ls_t001b-frye1 ls_t001b-frpe1
           INTO lv_ferbl.
    CONCATENATE ls_t001b-toye1 ls_t001b-tope1
           INTO lv_terbl.
    IF lv_perbl >= lv_ferbl AND lv_perbl <= lv_terbl.
    ELSE.
      e_payable = 'X'.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
