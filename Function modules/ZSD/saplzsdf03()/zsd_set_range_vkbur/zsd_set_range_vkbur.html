FUNCTION zsd_set_range_vkbur.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_UNAME) TYPE ZSDT9010-UNAME
*" TABLES
*" T_VKBUR TYPE TDT_RG_VKBUR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_zsdt9010 TYPE STANDARD TABLE OF zsdt9010,
        lt_vkbur    TYPE tdt_rg_vkbur,
        ls_zsdt9010 LIKE LINE OF lt_zsdt9010,
        ls_vkbur    LIKE LINE OF lt_vkbur.

  SELECT *
    INTO TABLE lt_zsdt9010
    FROM zsdt9010
   WHERE uname EQ i_uname.

  READ TABLE lt_zsdt9010 WITH KEY vkbur = '*'
    TRANSPORTING NO FIELDS.

  IF sy-subrc EQ 0.
    IF t_vkbur[] IS NOT INITIAL.
      EXIT.
    ENDIF.
    ls_vkbur-low = '*'.
    ls_vkbur-sign = 'I'.
    ls_vkbur-option = 'CP'.
    APPEND ls_vkbur TO lt_vkbur.
  ELSE.
    ls_vkbur-sign = 'I'.
    ls_vkbur-option = 'EQ'.
    LOOP AT lt_zsdt9010 INTO ls_zsdt9010 WHERE vkbur IN t_vkbur.
      ls_vkbur-low = ls_zsdt9010-vkbur.
      APPEND ls_vkbur TO lt_vkbur.
    ENDLOOP.
  ENDIF.

  CLEAR: t_vkbur.
  REFRESH: t_vkbur.
  t_vkbur[] = lt_vkbur.

ENDFUNCTION.
