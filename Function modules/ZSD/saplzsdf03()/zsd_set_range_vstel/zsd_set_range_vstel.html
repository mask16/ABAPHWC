FUNCTION zsd_set_range_vstel.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_UNAME) TYPE ZSDT9010-UNAME
*" TABLES
*" T_VSTEL TYPE TDT_RG_VSTEL
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_zsdt9020 TYPE STANDARD TABLE OF zsdt9020,
        lt_vstel    TYPE tdt_rg_vstel,
        ls_zsdt9020 LIKE LINE OF lt_zsdt9020,
        ls_vstel    LIKE LINE OF lt_vstel.

  SELECT *
    INTO TABLE lt_zsdt9020
    FROM zsdt9020
   WHERE uname EQ i_uname.

  READ TABLE lt_zsdt9020 WITH KEY vstel = '*'
    TRANSPORTING NO FIELDS.

  IF sy-subrc EQ 0.
    IF t_vstel[] IS NOT INITIAL.
      EXIT.
    ENDIF.
    ls_vstel-low = '*'.
    ls_vstel-sign = 'I'.
    ls_vstel-option = 'CP'.
    APPEND ls_vstel TO lt_vstel.
  ELSE.
    ls_vstel-sign = 'I'.
    ls_vstel-option = 'EQ'.
    LOOP AT lt_zsdt9020 INTO ls_zsdt9020 WHERE vstel IN t_vstel.
      ls_vstel-low = ls_zsdt9020-vstel.
      APPEND ls_vstel TO lt_vstel.
    ENDLOOP.
  ENDIF.

  CLEAR: t_vstel.
  REFRESH: t_vstel.
  t_vstel[] = lt_vstel.

ENDFUNCTION.
