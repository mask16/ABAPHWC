FUNCTION zsd_if_gw_claimc.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_INTNO) TYPE ZSDT0150-INTNO
*" VALUE(I_STATUS) TYPE ZSDT0150-STATUS
*" VALUE(I_APNUM) TYPE ZSDT0150-APNUM
*" VALUE(I_APDAT) TYPE ZSDT0150-APDAT
*" VALUE(I_CFDAT) TYPE ZSDT0150-CFDAT
*" EXPORTING
*" VALUE(E_RET) TYPE BAPIRET2-TYPE
*" VALUE(E_MSG) TYPE CHAR100
*"----------------------------------------------------------------------

* Global data declarations

  UPDATE zsdt0150
     SET apnum  = i_apnum
         status = i_status
         apdat  = i_apdat
         cfdat  = i_cfdat
         aenam  = sy-uname
         aedat  = sy-datum
         aezet  = sy-uzeit
   WHERE intno EQ i_intno
     AND apnum EQ i_apnum.

  IF sy-subrc EQ 0.
    COMMIT WORK.
    e_ret = 'S'.
  ELSE.
    e_ret = 'E'.
    e_msg = '내부번호 및 품의번호가 존재하지 않습니다.'.
  ENDIF.

ENDFUNCTION.
