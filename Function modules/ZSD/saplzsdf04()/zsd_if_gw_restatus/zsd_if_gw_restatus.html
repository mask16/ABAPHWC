FUNCTION zsd_if_gw_restatus.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_APNUM) TYPE ZSDT9520-APNUM OPTIONAL
*" VALUE(I_TITLE) TYPE ZSDT9520-TITLE OPTIONAL
*" VALUE(I_EMPNM) TYPE ZSDT9520-EMPNM OPTIONAL
*" VALUE(I_APDAT) TYPE ZSDT9520-APDAT OPTIONAL
*" VALUE(I_CFDAT) TYPE ZSDT9520-CFDAT OPTIONAL
*" VALUE(I_STATUS) TYPE ZSDT9520-STATUS OPTIONAL
*" VALUE(I_APTYP) TYPE ZSDT9520-APTYP
*" VALUE(I_INTNO) TYPE ZSDT9520-INTNO
*" EXPORTING
*" VALUE(E_RET) TYPE BAPIRET2-TYPE
*" VALUE(E_MSG) TYPE CHAR100
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_zsdt9520 TYPE zsdt9520,
        lv_aedat    TYPE sy-datum,
        lv_aezet    TYPE sy-uzeit.

  ls_zsdt9520-apnum = i_apnum.
  ls_zsdt9520-title = i_title.
  ls_zsdt9520-empnm = i_empnm.
  ls_zsdt9520-apdat = i_apdat.
  ls_zsdt9520-cfdat = i_cfdat.
  ls_zsdt9520-status = i_status.
  ls_zsdt9520-aptyp = i_aptyp.
  ls_zsdt9520-intno = i_intno.

  SELECT SINGLE *
    INTO ls_zsdt9520
    FROM zsdt9520
   WHERE apnum EQ i_apnum.

* 저장일자
  lv_aedat = sy-datum.
* 저장시각
  lv_aezet = sy-uzeit.

  IF sy-subrc EQ 0.
    ls_zsdt9520-aenam = sy-uname.
    ls_zsdt9520-aedat = lv_aedat.
    ls_zsdt9520-aezet = lv_aezet.
  ELSE.
    ls_zsdt9520-ernam = sy-uname.
    ls_zsdt9520-erdat = lv_aedat.
    ls_zsdt9520-erzet = lv_aezet.
    ls_zsdt9520-aenam = sy-uname.
    ls_zsdt9520-aedat = lv_aedat.
    ls_zsdt9520-aezet = lv_aezet.
  ENDIF.

  MODIFY zsdt9520 FROM ls_zsdt9520.
  IF sy-subrc EQ 0.
    COMMIT WORK.
    e_ret = 'S'.
  ELSE.
    e_ret = 'E'.
    e_msg = '저장중 오류발생'.
  ENDIF.

ENDFUNCTION.
