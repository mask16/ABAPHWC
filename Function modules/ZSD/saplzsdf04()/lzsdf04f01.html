FORM get_domain_text USING p_value
                           p_domname
                  CHANGING p_text.

  DATA: lt_dd07v TYPE STANDARD TABLE OF dd07v,
        ls_dd07v LIKE LINE OF lt_dd07v.

  CLEAR: p_text.
  CHECK p_value IS NOT INITIAL.
  CALL METHOD cl_reca_ddic_doma=>get_values
    EXPORTING
      id_name   = p_domname
    IMPORTING
      et_values = lt_dd07v.

  IF sy-subrc NE 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
           WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    EXIT.
  ENDIF.

  READ TABLE lt_dd07v INTO ls_dd07v WITH KEY domvalue_l = p_value.
  IF sy-subrc EQ 0.
    p_text = ls_dd07v-ddtext.
  ENDIF.

ENDFORM. "GET_DOMAIN_TEXT

FORM get_matnm_text USING p_prdha

                     CHANGING c_matnm.
  data : l_vtext type t179t-vtext,
