FUNCTION zsd_if_gw_claima.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_INTNO) TYPE ZSDT0150-INTNO
*" EXPORTING
*" VALUE(E_CUSTNM) TYPE KNA1-NAME1
*" VALUE(E_ASTYP) TYPE ZSDT0150-ASTYP
*" VALUE(E_TOTQTY) TYPE ZSDT0150-TOTQTY
*" VALUE(E_NUSQTY) TYPE ZSDT0150-NUSQTY
*" VALUE(E_PALQTY) TYPE ZSDT0150-PALQTY
*" VALUE(E_RET) TYPE BAPIRET2-TYPE
*" VALUE(E_MSG) TYPE CHAR100
*" TABLES
*" ET_LIST STRUCTURE ZSDS0150
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_zsdt0151   TYPE STANDARD TABLE OF zsdt0151,
        ls_zsdt0151   LIKE LINE OF lt_zsdt0151,
        ls_zsdt0150   TYPE zsdt0150,
        ls_list       TYPE zsds0150.

  SELECT SINGLE *
    INTO ls_zsdt0150
    FROM zsdt0150
   WHERE intno EQ i_intno.

  IF sy-subrc NE 0.
    e_ret = 'E'.
    e_msg = '데이터가 존재하지 않습니다.'.
    EXIT.
  ENDIF.

* 거래처명
  SELECT SINGLE name1
    INTO e_custnm
    FROM kna1
   WHERE kunnr EQ ls_zsdt0150-kunnr.

* 기술영업 A/S 여부
  e_astyp = ls_zsdt0150-astyp.
* 총수량(KG)
  e_totqty = ls_zsdt0150-totqty.
* 미사용품중량(KG)
  e_nusqty = ls_zsdt0150-nusqty.
* 미사용품수량(PAL)
  e_palqty = ls_zsdt0150-palqty.

  SELECT *
    INTO TABLE lt_zsdt0151
    FROM zsdt0151
   WHERE intno EQ i_intno.

  LOOP AT lt_zsdt0151 INTO ls_zsdt0151.
    CLEAR: ls_list.
    MOVE-CORRESPONDING ls_zsdt0151 TO ls_list.
* 불만유형
    SELECT SINGLE ddtext
      INTO ls_list-fecod
      FROM dd07t
     WHERE domname    EQ 'ZDSDFECOD'
       AND ddlanguage EQ sy-langu
       AND domvalue_l EQ ls_zsdt0151-fecod.
* 플랜트
    SELECT SINGLE name1
      INTO ls_list-plant
      FROM t001w
     WHERE werks EQ ls_zsdt0151-werks.
* 모델명, 규격
    SELECT SINGLE zmaktx groes
      INTO (ls_list-model, ls_list-groes)
      FROM mara
     WHERE matnr EQ ls_zsdt0151-matnr.

    APPEND ls_list TO et_list.
  ENDLOOP.
  e_ret = 'S'.

ENDFUNCTION.
