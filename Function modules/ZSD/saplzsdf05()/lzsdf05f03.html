FORM display_gr_data TABLES pt_tab STRUCTURE zsds0180
                      USING p_mtnum p_vbeln p_gubun p_plant.

  TYPES : BEGIN OF typ_min,
            refno TYPE zsdt5010-refno,
            matnr TYPE zsdt5010-matnr,
            werks TYPE zsdt5010-werks,
            menge TYPE zsdt5010-menge,
          END OF typ_min.

  DATA : lt_tab   TYPE STANDARD TABLE OF zsds0190,
         lt_tab1  TYPE STANDARD TABLE OF zsds0190,
         ls_tab1  TYPE zsds0190,
         lt_tab2  TYPE STANDARD TABLE OF zsds0190,
         ls_tab2  TYPE zsds0190,
         ls_tab   TYPE zsds0180,
         lt_min   TYPE STANDARD TABLE OF typ_min,
         ls_min   TYPE typ_min.

  DATA : l_menge LIKE zsdt5010-menge,
         l_senge LIKE zsdt5010-menge,
         l_mtart LIKE mara-mtart,
         l_matnr LIKE lips-matnr,
         l_werks LIKE lips-werks,
         l_lfimg LIKE lips-lfimg.

  RANGES : lr_mtart FOR mara-mtart.

  IF p_gubun = text-002.
    CLEAR : p_plant.
    SELECT SINGLE werks INTO p_plant FROM lips WHERE vbeln EQ p_vbeln.
  ENDIF.

  CLEAR : pt_tab, pt_tab[].
*-- 잔량이 존재하는 납품문서의 경우
  SELECT a~mtnum a~barcode a~pbarcode
         a~lotmat a~matnr a~charg
         a~bzgbn c~maktx a~menge
         a~meins a~refno a~repos
         a~werks a~frsloc a~prddt
         a~qalot b~name1
    FROM zsdt5010 AS a LEFT OUTER JOIN t001w AS b
                         ON a~werks = b~werks
                       LEFT OUTER JOIN makt AS c
                         ON a~matnr = c~matnr
                        AND c~spras = sy-langu
      INTO CORRESPONDING FIELDS OF TABLE lt_tab
  WHERE a~refno EQ p_vbeln AND a~recod NE 'S'
    AND a~bzgbn EQ g_bzgbn.

  IF p_plant = text-009 OR p_plant = text-010 OR p_plant = text-011
  OR p_plant = text-016 OR p_plant = text-017.
    LOOP AT lt_tab INTO ls_tab1.
      CLEAR l_mtart.
      SELECT SINGLE mtart INTO l_mtart
        FROM mara WHERE matnr EQ ls_tab1-matnr.
      IF l_mtart = text-015.
        APPEND ls_tab1 TO lt_tab1. CLEAR ls_tab1.
      ENDIF.
    ENDLOOP.
  ELSEIF p_plant = text-003.
    LOOP AT lt_tab INTO ls_tab1.
      APPEND ls_tab1 TO lt_tab1. CLEAR ls_tab1.
    ENDLOOP.
  ENDIF.

  SORT lt_tab1 BY matnr lotmat.
  CLEAR : ls_tab, ls_tab1, ls_tab2, l_menge.

  LOOP AT lt_tab1 INTO ls_tab1.
*-- Mother와 Child를 구분
    MOVE-CORRESPONDING ls_tab1 TO ls_tab.
    MOVE-CORRESPONDING ls_tab1 TO ls_tab2.

    l_menge = ls_tab1-menge + l_menge.
    l_senge = ls_tab1-menge + l_senge.

    AT END OF lotmat.
      ls_tab-mcflg = text-005. ls_tab-bzgbn = g_bzgbn.
      ls_tab1-menge = l_senge.
      WRITE ls_tab1-menge TO ls_tab-menge UNIT ls_tab-meins.
      APPEND ls_tab TO pt_tab. CLEAR : l_senge, ls_tab.
    ENDAT.

    AT END OF matnr.
      MOVE-CORRESPONDING ls_tab2 TO ls_tab.
      ls_tab2-menge = l_menge.
      ls_tab-mcflg = text-004. ls_tab-bzgbn = g_bzgbn.
      ls_tab-barcode = ''. ls_tab-pbarcode = ''.
      ls_tab-lotmat = ''. ls_tab-charg = ''.
      ls_tab-qalot = ''.
      WRITE ls_tab2-menge TO ls_tab-menge UNIT ls_tab-meins.
      APPEND ls_tab TO pt_tab. CLEAR : l_menge, ls_tab.
    ENDAT.
    CLEAR : ls_tab1, ls_tab2.
  ENDLOOP.
*-- 납품문서에서 Data가지고 오기
  SELECT a~matnr a~werks
         a~lfimg AS menge a~meins
         c~maktx a~vbeln AS refno
         a~posnr AS repos b~name1
    FROM lips AS a LEFT OUTER JOIN t001w AS b
                     ON a~werks = b~werks
                   LEFT OUTER JOIN makt AS c
                     ON a~matnr = c~matnr
                    AND c~spras = sy-langu
    INTO CORRESPONDING FIELDS OF TABLE lt_tab2
   WHERE a~vbeln EQ p_vbeln.

  SELECT refno matnr werks SUM( menge ) AS menge
    FROM zsdt5010
    INTO CORRESPONDING FIELDS OF TABLE lt_min
   WHERE refno EQ p_vbeln AND bzgbn EQ g_bzgbn
     AND recod EQ 'S' GROUP BY refno matnr werks.

  SORT lt_tab2 BY matnr. CLEAR : ls_tab, ls_tab2.

  LOOP AT lt_tab2 INTO ls_tab2.
    MOVE-CORRESPONDING ls_tab2 TO ls_tab.
    ls_tab-mcflg = text-004. ls_tab-bzgbn = g_bzgbn.
    ls_tab-menge = 0.
    ls_tab-barcode = ''. ls_tab-pbarcode = ''.
    ls_tab-lotmat = ''. ls_tab-charg = ''.
    ls_tab-qalot = ''. ls_tab-mtnum = p_mtnum.
    APPEND ls_tab TO pt_tab. CLEAR ls_tab.
  ENDLOOP.

*-- 중복제거
  SORT pt_tab BY matnr barcode pbarcode charg bzgbn mcflg
     DESCENDING lotmat ASCENDING.

  DELETE ADJACENT DUPLICATES FROM pt_tab
        COMPARING matnr barcode pbarcode charg bzgbn mcflg lotmat.

  SORT lt_min BY refno matnr werks.

  LOOP AT pt_tab INTO ls_tab.
    SELECT SUM( lfimg ) INTO l_lfimg FROM lips
     WHERE vbeln EQ p_vbeln AND matnr EQ ls_tab-matnr
       AND werks EQ ls_tab-werks
     GROUP BY vbeln matnr werks.
    ENDSELECT.
    READ TABLE lt_min INTO ls_min WITH KEY refno = ls_tab-refno
                                           matnr = ls_tab-matnr
                                           werks = ls_tab-werks
                                           BINARY SEARCH.
    IF sy-subrc = 0.
      l_lfimg = l_lfimg - ls_min-menge.
    ENDIF.
    IF l_lfimg <> 0.
      WRITE l_lfimg TO ls_tab-senge UNIT ls_tab-meins.
    ENDIF.
    MODIFY pt_tab FROM ls_tab. CLEAR : l_matnr, l_werks, l_lfimg.
  ENDLOOP.

  LOOP AT pt_tab INTO ls_tab WHERE mcflg = text-004.
    CLEAR : l_lfimg.
    SELECT SUM( menge ) INTO l_lfimg FROM zsdt5010
     WHERE refno EQ p_vbeln AND matnr EQ ls_tab-matnr
       AND werks EQ ls_tab-werks AND bzgbn EQ g_bzgbn
       AND recod EQ ''
     GROUP BY refno matnr werks.
    ENDSELECT.
    WRITE l_lfimg TO ls_tab-menge UNIT ls_tab-meins.

    MODIFY pt_tab FROM ls_tab. CLEAR : l_matnr, l_werks, l_lfimg.
  ENDLOOP.

  DELETE pt_tab WHERE senge = ''.

  LOOP AT pt_tab INTO ls_tab.
    p_mtnum = ls_tab-mtnum.
    IF p_mtnum <> ''. EXIT. ENDIF.
  ENDLOOP.

  SORT pt_tab BY matnr mcflg DESCENDING pbarcode barcode lotmat.

ENDFORM. " DISPLAY_GR_DATA

FORM gr_select_lgort TABLES pt_tab STRUCTURE zsds0180
                  USING p_barcod p_userid
                        p_mtnum p_vbeln
                        p_etnum p_ealue
                        p_ret p_msg.
*-- 관리구분자
* 1: 바코드존재하지않음
* 2: 바코드 8자리
* 3: 이천공장아니면서 바코드 8째자리가 P
* 4: 이천공장이면서 바코드 19자리
  DATA : l_flag TYPE char1.

  DATA : lt_tab  TYPE STANDARD TABLE OF zsds0190,
         ls_tab  TYPE zsds0190,
         ls_180  TYPE zsds0180.

  LOOP AT pt_tab INTO ls_180.
    MOVE-CORRESPONDING ls_180 TO ls_tab.

    CALL FUNCTION 'STRING_REPLACE'
      EXPORTING
        pattern             = ','
        substitute          = ''
      CHANGING
        text                = ls_180-menge
      EXCEPTIONS
        wrong_string_length = 1
        OTHERS              = 2.

    ls_tab-menge = ls_180-menge.

    APPEND ls_tab TO lt_tab. CLEAR : ls_tab, ls_180.
  ENDLOOP.

*-- PDA LOG TABLE
  IF p_mtnum = ''.
    IF strlen( p_barcod ) = 11.
      SELECT SINGLE mtnum INTO p_mtnum FROM zsdt5010
       WHERE bzgbn EQ g_bzgbn AND pbarcode EQ p_barcod AND recod NE 'S'
        AND refno EQ p_vbeln.
    ELSE.
      SELECT SINGLE mtnum INTO p_mtnum FROM zsdt5010
       WHERE bzgbn EQ g_bzgbn AND barcode EQ p_barcod AND recod NE 'S'
         AND refno EQ p_vbeln.
    ENDIF.
  ENDIF.

*-- 바코드가 8자리인 경우 상품으로 처리
  IF strlen( p_barcod ) = 8.
    PERFORM matnr_gr_data TABLES lt_tab USING p_mtnum p_barcod p_vbeln.
    l_flag = 2.
  ELSE.
*-- 바코드가 11자리 Pallet barcode
    IF strlen( p_barcod ) = 11.
*-- 바코드 8번째 문자가 P인경우 Pallet barcode 아니면 소포장 barcode
      IF p_barcod+7(1) = text-006. " Pallet barcode
        PERFORM pallet_gr_data TABLES lt_tab USING p_mtnum p_barcod p_vbeln.
        l_flag = 3.
      ENDIF.
*-- 바코드가 26자리 소포장 barcode
    ELSEIF strlen( p_barcod ) = 26.
      IF p_barcod+7(1) <> text-006. " 소포장 barcode
        PERFORM box_gr_data TABLES lt_tab USING p_mtnum p_barcod p_vbeln.
        l_flag = 3.
      ENDIF.
*-- 바코드가 19자리
    ELSEIF strlen( p_barcod ) = 19.
      PERFORM len19_gr_data TABLES lt_tab USING p_mtnum p_barcod p_vbeln.
      l_flag = 4.
    ENDIF.
  ENDIF.

  DELETE lt_tab WHERE menge = 0.

  IF lt_tab[] IS INITIAL.
    p_etnum = p_mtnum.
    p_ealue = p_vbeln.
    p_ret = 'E'.
    p_msg = text-e00.
  ELSE.
*-- PDA 처리 테이블에 저장
    CASE l_flag.
      WHEN 1.
        PERFORM saving_sd TABLES lt_tab
                           USING p_userid p_mtnum p_ret p_msg.
      WHEN 2.
        PERFORM saving_sd TABLES lt_tab
                           USING p_userid p_mtnum p_ret p_msg.
      WHEN 3.
        PERFORM saving_sd TABLES lt_tab
                           USING p_userid p_mtnum p_ret p_msg.
      WHEN 4.
        PERFORM saving_sd TABLES lt_tab
                           USING p_userid p_mtnum p_ret p_msg.
    ENDCASE.
    p_etnum = p_mtnum. p_ealue = p_vbeln.
  ENDIF.

ENDFORM. " GR_SELECT_LGORT

FORM matnr_gr_data TABLES pt_tab STRUCTURE zsds0190
                    USING p_mtnum
                          p_barcod
                          p_vbeln.

  TYPES : BEGIN OF typ_data_lip,
            vbeln TYPE lips-vbeln,
            posnr TYPE lips-posnr,
            matnr TYPE lips-matnr,
            maktx TYPE makt-maktx,
            werks TYPE lips-werks,
            lgort TYPE lips-lgort,
            charg TYPE lips-charg,
            menge TYPE lips-lfimg,
            meins TYPE lips-meins,
          END OF typ_data_lip.

  DATA : lt_lip  TYPE STANDARD TABLE OF typ_data_lip,
         ls_lip  TYPE typ_data_lip,
         lt_main TYPE STANDARD TABLE OF zsdt5010,
         ls_main TYPE zsdt5010,
         ls_tab TYPE zsds0190.

  CLEAR : gr_mtnum, gr_mtnum[].

  IF p_mtnum NE ''.
    gr_mtnum-low = p_mtnum. gr_mtnum-sign = 'I'.
    gr_mtnum-option = 'EQ'. APPEND gr_mtnum.
  ENDIF.

  SELECT a~vbeln a~posnr a~matnr a~werks a~lgort
         lfimg as a~menge a~meins a~charg b~maktx
    FROM lips AS a LEFT OUTER JOIN makt AS b
                           ON a~matnr = b~matnr
                          AND b~spras = sy-langu
    INTO CORRESPONDING FIELDS OF TABLE lt_lip
   WHERE a~vbeln EQ p_vbeln AND a~matnr EQ p_barcod.

  IF lt_lip[] IS NOT INITIAL.
    SELECT *
      FROM zsdt5010
      INTO CORRESPONDING FIELDS OF TABLE lt_main
       FOR ALL ENTRIES IN lt_lip
     WHERE refno EQ lt_lip-vbeln AND repos EQ lt_lip-posnr
       AND recod EQ text-007 AND mtnum IN gr_mtnum
       AND bzgbn EQ g_bzgbn.

    IF lt_main[] IS NOT INITIAL.
      SORT lt_main BY refno repos.
      LOOP AT lt_lip INTO ls_lip.
        READ TABLE lt_main INTO ls_main WITH KEY refno = ls_lip-vbeln
                                                 repos = ls_lip-posnr
                                                 BINARY SEARCH.
        IF sy-subrc <> 0.
          ls_tab-mtnum = p_mtnum.
          ls_tab-matnr = ls_lip-matnr.
          ls_tab-barcode = p_barcod.
          ls_tab-pbarcode = p_barcod.
          ls_tab-charg = ls_lip-charg.
          ls_tab-bzgbn = g_bzgbn.
          ls_tab-maktx = ls_lip-maktx.
          ls_tab-meins = ls_lip-meins.
          ls_tab-refno = p_vbeln.
          ls_tab-werks = ls_lip-werks.
          ls_tab-menge = ls_lip-menge.
          ls_tab-repos  = ls_lip-posnr.
          ls_tab-frsloc = ls_lip-lgort.

          APPEND ls_tab TO pt_tab. CLEAR ls_tab.
        ENDIF.
      ENDLOOP.
    ELSE.
      LOOP AT lt_lip INTO ls_lip.
        ls_tab-mtnum = p_mtnum.
        ls_tab-matnr = ls_lip-matnr.
        ls_tab-barcode = p_barcod.
        ls_tab-pbarcode = p_barcod.
        ls_tab-charg = ls_lip-charg.
        ls_tab-bzgbn = g_bzgbn.
        ls_tab-maktx = ls_lip-maktx.
        ls_tab-meins = ls_lip-meins.
        ls_tab-refno = p_vbeln.
        ls_tab-werks = ls_lip-werks.
        ls_tab-menge = ls_lip-menge.
        ls_tab-repos  = ls_lip-posnr.
        ls_tab-frsloc = ls_lip-lgort.

        APPEND ls_tab TO pt_tab. CLEAR ls_tab.
      ENDLOOP.
    ENDIF.
  ENDIF.

ENDFORM. " MATNR_GR_DATA

*      -->P_P_BARCOD  text

*----------------------------------------------------------------------*
FORM pallet_gr_data TABLES pt_tab STRUCTURE zsds0190
                  USING p_mtnum
                        p_barcod
                        p_vbeln.

  TYPES : BEGIN OF typ_data_lip,
            vbeln TYPE lips-vbeln,
            posnr TYPE lips-posnr,
            matnr TYPE lips-matnr,
            werks TYPE lips-werks,
            lgort TYPE lips-lgort,
          END OF typ_data_lip.

  DATA : lt_sub TYPE STANDARD TABLE OF zppt0170,
         ls_sub TYPE zppt0170,
         ls_tab TYPE zsds0190,
         lt_gr  TYPE STANDARD TABLE OF zsdt5010,
         ls_gr  TYPE zsdt5010,
         lt_lip TYPE STANDARD TABLE OF typ_data_lip,
         ls_lip TYPE typ_data_lip.

  DATA : l_flag TYPE c.

  CLEAR : gr_mtnum, gr_mtnum[].

  IF p_mtnum NE ''.
    gr_mtnum-low = p_mtnum. gr_mtnum-sign = 'I'.
    gr_mtnum-option = 'EQ'. APPEND gr_mtnum.
  ENDIF.

*-- 잔량 테이블에서 DATA를 Select
  SELECT a~werks a~barcode a~pbarcode a~bbarcode a~itqty a~remng
         a~confm a~matnr a~menge a~meins a~pcharg a~qcharg a~charg a~zmaktx
         a~groes a~icode1 a~icode2 a~icode3 a~icode4 a~icode5
         a~idesc1 a~idesc2 a~idesc3 a~idesc4 a~idesc5 a~aufnr
         a~ebeln a~ebelp a~lgubun a~iscnt a~zwerks
         a~created_date a~created_time a~created_by
         a~changed_date a~changed_time a~changed_by
         b~maktx
    FROM zppt0170 AS a LEFT OUTER JOIN makt AS b
                         ON a~matnr = b~matnr
                        AND b~spras = sy-langu
    INTO CORRESPONDING FIELDS OF TABLE lt_sub
   WHERE a~pbarcode EQ p_barcod.

  IF lt_sub[] IS NOT INITIAL.
*-- PDA 처리 테이블에서 GR DATA를 Select
    SELECT *
      FROM zsdt5010
      INTO CORRESPONDING FIELDS OF TABLE lt_gr
       FOR ALL ENTRIES IN lt_sub
     WHERE pbarcode EQ lt_sub-pbarcode AND barcode EQ lt_sub-barcode

    MOVE-CORRESPONDING ls_180 TO ls_tab.
    CALL FUNCTION 'STRING_REPLACE'

        pattern             = ','

           c~maktx a~vbeln AS refno a~posnr AS repos b~name1
      FROM lips AS a INNER JOIN mara AS d

  ENDIF.

                            USING p_mtnum p_vbeln p_lgort p_posdt p_userid
                                  p_etnum p_ebeln p_ret p_msg l_doc.
