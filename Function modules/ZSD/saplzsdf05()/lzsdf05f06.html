  IF p_lgort <> ''.

        APPEND gr_lgort. CLEAR gr_lgort.
        gr_lgort-sign = 'I'. gr_lgort-option = 'EQ'. gr_lgort-low = '1200'.

        APPEND gr_lgort. CLEAR gr_lgort.
        gr_lgort-sign = 'I'. gr_lgort-option = 'EQ'. gr_lgort-low = '1600'.
        APPEND gr_lgort. CLEAR gr_lgort.
        gr_lgort-sign = 'I'. gr_lgort-option = 'EQ'. gr_lgort-low = '1900'.
        APPEND gr_lgort. CLEAR gr_lgort.
    ELSE.

    APPEND gr_bzirk. CLEAR gr_bzirk.
  ENDIF.
  IF p_werks <> ''.
    gr_werks-sign = 'I'. gr_werks-option = 'EQ'. gr_werks-low = p_werks.
    APPEND gr_werks. CLEAR gr_werks.
  ENDIF.
  IF p_frlfdat <> '' AND p_tolfdat = ''.
    gr_lfdat-sign = 'I'. gr_lfdat-option = 'EQ'. gr_lfdat-low = p_frlfdat.
    APPEND gr_lfdat. CLEAR gr_lfdat.
  ELSEIF p_frlfdat <> '' AND p_tolfdat <> ''.
    gr_lfdat-sign = 'I'. gr_lfdat-option = 'BT'. gr_lfdat-low = p_frlfdat. gr_lfdat-high = p_tolfdat.
    APPEND gr_lfdat. CLEAR gr_lfdat.
  ENDIF.
  IF p_matnr <> ''.
    gr_matnr-sign = 'I'. gr_matnr-option = 'EQ'.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = p_matnr
      IMPORTING
        output = gr_matnr-low.
    APPEND gr_matnr. CLEAR gr_matnr.
  ENDIF.
  IF p_maktx <> ''.
    gr_maktx-sign = 'I'. gr_maktx-option = 'CP'. CONCATENATE '*'  p_maktx  '*' INTO gr_maktx-low .
    APPEND gr_maktx. CLEAR gr_maktx.
  ENDIF.
  IF p_spart <> ''.

    APPEND gr_vkgrp. CLEAR gr_vkgrp.
  ENDIF.

FORM base_gi_data TABLES pt_tab STRUCTURE zsds0260
                   USING p_auart p_wbstk.

  TYPES : BEGIN OF typ_base,
            auart   TYPE vbak-auart,
            lfart   TYPE likp-lfart,
            wadat   TYPE likp-wadat,
            lfdat   TYPE likp-lfdat,
            vbeln   TYPE likp-vbeln,
            lgort   TYPE lips-lgort,
            lgobe   TYPE t001l-lgobe,
            vstel   TYPE likp-vstel,
            kunnr   TYPE likp-kunnr,
            werks   TYPE lips-werks,
            wame1   TYPE t001w-name1,
            vtweg   TYPE vbak-vtweg,
            matnr   TYPE lips-matnr,
            maktx   TYPE makt-maktx,
            ormng   TYPE lips-ormng,
            lfimg   TYPE lips-lfimg,
            meins   TYPE lips-meins,
            sakunnr TYPE vbpa-kunnr,
            vkgrp   TYPE lips-vkgrp,
            bezei   TYPE tvgrt-bezei,
            wbstk   TYPE vbuk-wbstk,
          END OF typ_base.

  TYPES : BEGIN OF typ_tvstt,
            vstel TYPE likp-vstel,
            vtext TYPE tvstt-vtext,
          END OF typ_tvstt.

  TYPES : BEGIN OF typ_kna1,
            kunnr TYPE kna1-kunnr,
            name1 TYPE kna1-name1,
          END OF typ_kna1.

  TYPES : BEGIN OF typ_tvtwt,
            vtweg TYPE tvtwt-vtweg,
            vtxtk TYPE tvtwt-vtext,
          END OF typ_tvtwt.

  TYPES : BEGIN OF typ_adrc,
            vbeln   TYPE likp-vbeln,
            kunnr   TYPE vbpa-kunnr,
            name1   TYPE adrc-name1,
            adrda   TYPE vbpa-adrda,
            adrnr   TYPE tvst-adrnr,
          END OF typ_adrc.

  TYPES : BEGIN OF typ_kdrc,
            vbeln   TYPE likp-vbeln,
            kunnr   TYPE vbpa-kunnr,
            name1   TYPE adrc-name1,
            adrnr   TYPE tvst-adrnr,
          END OF typ_kdrc.

  DATA : lt_main TYPE STANDARD TABLE OF typ_base,
         ls_main TYPE typ_base,
         lt_base TYPE STANDARD TABLE OF zsds0261,
         ls_base TYPE zsds0261,
         lt_sum  TYPE STANDARD TABLE OF zsds0261,
         ls_sum  TYPE zsds0261,
         lt_tvstt TYPE STANDARD TABLE OF typ_tvstt,
         ls_tvstt TYPE typ_tvstt,
         lt_kna1 TYPE STANDARD TABLE OF typ_kna1,
         ls_kna1 TYPE typ_kna1,
         lt_knak TYPE STANDARD TABLE OF typ_kna1,
         ls_knak TYPE typ_kna1,
         lt_adrc TYPE STANDARD TABLE OF typ_adrc,
         ls_adrc TYPE typ_adrc,
         lt_kdrc TYPE STANDARD TABLE OF typ_kdrc,
         ls_kdrc TYPE typ_kdrc,
         lt_tvtwt TYPE STANDARD TABLE OF typ_tvtwt,
         ls_tvtwt TYPE typ_tvtwt,
         ls_tab  TYPE zsds0260.

  DATA : l_adrda TYPE vbpa-adrda,
         l_adrnr TYPE adrc-addrnumber.

  SELECT  a~wadat a~lfdat a~vbeln a~vstel a~kunnr a~lfart
          b~werks b~lgort b~matnr b~arktx AS maktx
          b~vkgrp b~ormng b~lfimg b~meins
          d~name1 AS wame1
          e~bezei f~lgobe g~auart g~vtweg
          i~kunnr AS sakunnr h~wbstk
     FROM likp AS a INNER JOIN lips AS b
                       ON a~vbeln = b~vbeln
                    INNER JOIN mara AS c
                       ON b~matnr = c~matnr
                     LEFT OUTER JOIN t001w AS d
                       ON b~werks = d~werks
                     LEFT OUTER JOIN tvgrt AS e
                       ON b~vkgrp = e~vkgrp
                      AND e~spras = sy-langu
                     LEFT OUTER JOIN t001l AS f
                       ON b~werks = f~werks
                      AND b~lgort = f~lgort
                     LEFT OUTER JOIN vbak AS g
                       ON b~vgbel = g~vbeln
                     LEFT OUTER JOIN  vbuk AS h
                       ON b~vbeln = h~vbeln
                     LEFT OUTER JOIN vbpa AS i
                       ON b~vbeln = i~vbeln
                      AND i~parvw = 'S1'
       INTO CORRESPONDING FIELDS OF TABLE lt_main
      WHERE a~kunnr IN gr_kunnr AND a~bzirk IN gr_bzirk
        AND a~lfdat IN gr_lfdat AND a~vkorg IN gr_vkorg
        AND b~lgort IN gr_lgort AND b~werks IN gr_werks
        AND b~matnr IN gr_matnr AND b~arktx IN gr_maktx
        AND b~vtweg IN gr_vtweg AND b~vkgrp IN gr_vkgrp
        AND c~spart IN gr_spart AND c~matkl IN gr_matkl.

  IF p_auart <> ''.
    DELETE lt_main WHERE auart NE p_auart.
  ENDIF.

  IF p_wbstk <> ''.
    DELETE lt_main WHERE wbstk NE p_wbstk.
  ENDIF.

  IF lt_main[] IS NOT INITIAL.
*-- 출하지점
    SELECT vstel vtext
      FROM tvstt
      INTO CORRESPONDING FIELDS OF TABLE lt_tvstt
     WHERE spras EQ sy-langu.
*-- 유통 경로
    SELECT vtweg vtext AS vtxtk
      FROM tvtwt
      INTO CORRESPONDING FIELDS OF TABLE lt_tvtwt
     WHERE spras EQ sy-langu.
*-- 납품처
    SELECT a~vbeln a~kunnr a~adrda c~name1 c~addrnumber AS adrnr
      FROM vbpa AS a INNER JOIN adrc AS c
                        ON a~adrnr = c~addrnumber
      INTO CORRESPONDING FIELDS OF TABLE lt_adrc
       FOR ALL ENTRIES IN lt_main
     WHERE a~vbeln EQ lt_main-vbeln AND a~parvw EQ 'WE'.
*-- 판매처
    SELECT a~vbeln a~kunnr c~name1 c~addrnumber AS adrnr
      FROM vbpa AS a INNER JOIN adrc AS c
                        ON a~adrnr = c~addrnumber
      INTO CORRESPONDING FIELDS OF TABLE lt_kdrc
       FOR ALL ENTRIES IN lt_main
     WHERE a~vbeln EQ lt_main-vbeln AND a~parvw EQ 'AG'.

    SELECT kunnr name1 FROM kna1
      INTO CORRESPONDING FIELDS OF TABLE lt_knak
       FOR ALL ENTRIES IN lt_main
     WHERE kunnr EQ lt_main-kunnr.
*-- 사원번호
    SELECT kunnr name1
      FROM kna1
      INTO CORRESPONDING FIELDS OF TABLE lt_kna1
       FOR ALL ENTRIES IN lt_main
     WHERE kunnr EQ lt_main-sakunnr.

    SORT lt_kna1 BY kunnr.
    DELETE ADJACENT DUPLICATES FROM lt_kna1.
*----------------------------------------------------------------------*
* 메인화면 구성하기
*----------------------------------------------------------------------*
    SORT lt_tvstt BY vstel. SORT lt_kna1 BY kunnr. SORT lt_adrc BY vbeln.
    SORT lt_kdrc BY vbeln. SORT lt_knak BY kunnr.
    CLEAR : l_adrnr, l_adrda.
    LOOP AT lt_main INTO ls_main.
      MOVE-CORRESPONDING ls_main TO ls_base.
      ls_base-remng = ls_base-ormng - ls_base-lfimg.
*-- 출하지점
      READ TABLE lt_tvstt INTO ls_tvstt WITH KEY vstel = ls_base-vstel
                                             BINARY SEARCH.
      IF sy-subrc = 0. ls_base-vtext = ls_tvstt-vtext. ENDIF.
*-- 영업사원
      READ TABLE lt_kna1 INTO ls_kna1 WITH KEY kunnr = ls_base-sakunnr
                                           BINARY SEARCH.
      IF sy-subrc = 0. ls_base-saname1 = ls_kna1-name1. ENDIF.
*-- 유통경로
      READ TABLE lt_tvtwt INTO ls_tvtwt WITH KEY vtweg = ls_base-vtweg
                                             BINARY SEARCH.
      IF sy-subrc = 0. ls_base-vtxtk = ls_tvtwt-vtxtk. ENDIF.
*-- 납품처
      READ TABLE lt_adrc INTO ls_adrc WITH KEY vbeln = ls_base-vbeln
                                           BINARY SEARCH.
      IF sy-subrc = 0.
        ls_base-kunnr  = ls_adrc-kunnr. ls_base-name1 = ls_adrc-name1.
        l_adrda = ls_adrc-adrda. l_adrnr = ls_adrc-adrnr.
      ENDIF.
*----------------------------------------------------------------------*
* 납품문서가 NL인 경우
*----------------------------------------------------------------------*
*-- 판매처
      IF ls_main-lfart = 'NL'.
        READ TABLE lt_knak INTO ls_knak WITH KEY kunnr = ls_base-kunnr
                                             BINARY SEARCH.
        IF sy-subrc = 0.
          ls_base-kunnr12  = ls_knak-kunnr . ls_base-name12 = ls_knak-name1.
*-- 판매처코드하고 납품처코드가 불일치
          IF ls_base-kunnr12 <> ls_base-kunnr.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
*-- 판매처명하고 납품처명이 불일치
          ELSEIF ls_base-name12 <> ls_base-name1.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
*-- 판매처코드하고 납품처코드가 일치시 수작업필드(ADRDA)가 'E'인 경우
          ELSEIF l_adrda = 'E'.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
          ENDIF.
        ENDIF.
      ELSE.
*----------------------------------------------------------------------*
* 납품문서가 NL이 아닌 경우
*----------------------------------------------------------------------*
        READ TABLE lt_kdrc INTO ls_kdrc WITH KEY vbeln = ls_base-vbeln
                                             BINARY SEARCH.
        IF sy-subrc = 0.
          ls_base-kunnr12  = ls_kdrc-kunnr . ls_base-name12 = ls_kdrc-name1.
*-- 판매처코드하고 납품처코드가 불일치
          IF ls_base-kunnr12 <> ls_base-kunnr.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
*-- 판매처명하고 납품처명이 불일치
          ELSEIF ls_base-name12 <> ls_base-name1.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
*-- 판매처코드하고 납품처코드가 일치시 수작업필드(ADRDA)가 'E'인 경우
          ELSEIF l_adrda = 'E'.
            SELECT SINGLE city1 tel_number INTO (ls_base-city1, ls_base-tel1)
              FROM adrc
             WHERE addrnumber EQ l_adrnr.
          ENDIF.
        ENDIF.
      ENDIF.
      APPEND ls_base TO lt_base. CLEAR : ls_base, l_adrnr, l_adrda.
    ENDLOOP.

*-- T_TAB에 옮기기
    SORT lt_base BY vbeln matnr werks lgort meins.
    LOOP AT lt_base INTO ls_base.
      MOVE-CORRESPONDING ls_base TO ls_sum.
      COLLECT ls_sum INTO lt_sum. CLEAR ls_sum.
    ENDLOOP.

    SORT lt_sum BY vbeln matnr werks lgort meins.
    LOOP AT lt_sum INTO ls_sum.
      MOVE-CORRESPONDING ls_sum TO ls_tab.
      WRITE ls_sum-ormng TO ls_tab-ormng UNIT ls_sum-meins.
      WRITE ls_sum-lfimg TO ls_tab-lfimg UNIT ls_sum-meins.
      WRITE ls_sum-remng TO ls_tab-remng UNIT ls_sum-meins.

      CALL FUNCTION 'CONVERSION_EXIT_MATN2_OUTPUT'
       EXPORTING
        input  = ls_tab-matnr
       IMPORTING
        output = ls_tab-matnr.

      APPEND ls_tab TO pt_tab. CLEAR ls_tab.
    ENDLOOP.
  ENDIF.
