FORM STO_SELECT_DATA TABLES PT_TAB STRUCTURE ZSDS0270.

  DATA : LT_BASE TYPE STANDARD TABLE OF ZSDS0271,
         LS_BASE TYPE ZSDS0271,
         LS_TAB  TYPE ZSDS0270.

  SELECT A~MATNR A~WERKS A~LGORT A~UMLME
         A~INSME A~LABST A~SPEME A~RETME
         B~MAKTX C~NAME1 D~LGOBE E~MEINS
    FROM MARD AS A INNER JOIN  MARA AS E
                      ON A~MATNR = E~MATNR
                   INNER JOIN MAKT AS B
                      ON E~MATNR = B~MATNR
                     AND B~SPRAS = SY-LANGU
                    LEFT OUTER JOIN T001W AS C
                      ON A~WERKS = C~WERKS
                    LEFT OUTER JOIN T001L AS D
                      ON A~WERKS = D~WERKS
                     AND A~LGORT = D~LGORT
      INTO CORRESPONDING FIELDS OF TABLE LT_BASE
     WHERE A~LGORT IN GR_LGORT AND A~MATNR IN GR_MATNR
       AND B~MAKTX IN GR_MAKTX AND A~WERKS IN GR_WERKS
       AND E~SPART IN GR_SPART AND E~MATKL IN GR_MATKL.

  IF LT_BASE[] IS NOT INITIAL.
    DELETE LT_BASE WHERE UMLME = '0' AND LABST = '0' AND SPEME = '0' AND RETME = '0'.
    LOOP AT LT_BASE INTO LS_BASE.
      MOVE-CORRESPONDING LS_BASE TO LS_TAB.
      WRITE LS_BASE-UMLME TO LS_TAB-UMLME UNIT LS_BASE-MEINS.
      WRITE LS_BASE-INSME TO LS_TAB-INSME UNIT LS_BASE-MEINS.
      WRITE LS_BASE-LABST TO LS_TAB-LABST UNIT LS_BASE-MEINS.
      WRITE LS_BASE-SPEME TO LS_TAB-SPEME UNIT LS_BASE-MEINS.
      WRITE LS_BASE-RETME TO LS_TAB-RETME UNIT LS_BASE-MEINS.

      CALL FUNCTION 'CONVERSION_EXIT_MATN2_OUTPUT'
        EXPORTING
          INPUT  = LS_TAB-MATNR
        IMPORTING
          OUTPUT = LS_TAB-MATNR.

      APPEND LS_TAB TO PT_TAB. CLEAR LS_TAB.

    ENDLOOP.
  ENDIF.

  SORT PT_TAB BY MAKTX.

ENDFORM. " STO_SELECT_DATA
