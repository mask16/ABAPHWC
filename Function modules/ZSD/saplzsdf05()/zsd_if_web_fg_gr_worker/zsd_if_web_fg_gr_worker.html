FUNCTION ZSD_IF_WEB_FG_GR_WORKER.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_USERID) LIKE ZSDT5000-USERID OPTIONAL
*" VALUE(I_FDATE) TYPE ZPRODGRDATE OPTIONAL
*" VALUE(I_TDATE) TYPE ZPRODGRDATE OPTIONAL
*" VALUE(I_WERKS) TYPE ZEWERKS OPTIONAL
*" EXPORTING
*" VALUE(E_RET) LIKE ZSDS0001-RET
*" VALUE(E_MSG) LIKE ZSDS0001-MSG
*" TABLES
*" T_TAB STRUCTURE ZSDS0243
*"----------------------------------------------------------------------

* Global data declarations

  TYPES: BEGIN OF TY_DATA,
      PBARCODE        LIKE ZPPT0110-PBARCODE,                 " 팔레트 바코드
      ZPLNUM          LIKE ZPPT0110-ZPLNUM,                   " 작업지시번호
      PROD_GR_DATE    LIKE ZPPT0110-PROD_GR_DATE,             " 생산 입고날짜
      SHIP_GR_DATE    LIKE ZPPT0110-SHIP_GR_DATE,             " 출하 입고 날짜
      QCHARG          LIKE ZPPT0170-QCHARG,                   " 품질 LOT ID
      PRODTYPE        LIKE ZPPT0170-PRODTYPE,                 " 생산 타입
      ARBPL           LIKE ZPPT0110-ARBPL,                    " 작업장
      MATNR           LIKE ZPPT0170-MATNR,                    " 자재번호
      MAKTX           LIKE MAKT-MAKTX,                        " 자재내역
      VTWEG           LIKE ZPPT0090-VTWEG,                    " 유통경로
      LGORT           TYPE LGORT_D,                           " 저장위치
      MENGE           LIKE ZPPT0170-MENGE,                    " 중량
      MEINS           LIKE ZPPT0170-MEINS,                    " 단위
      WERKS           LIKE ZPPT0110-WERKS,                    " 플랜트
    END OF TY_DATA.

  DATA: LT_DATA   TYPE STANDARD TABLE OF TY_DATA,
        LS_DATA   TYPE TY_DATA,
        LV_VTWEG  TYPE ZPPT0090-VTWEG,                        " 유통경로
        LV_MENGE  TYPE ZPPT0170-REMNG,                        " 잔량 합산 계산
        LV_PLANT1 TYPE ZSDT5000-PLANT1.                       " 플랜트 공장.

*--- 플랜트 변환
  IF I_WERKS = 'P2SUB'.         " SUB
    LV_PLANT1 = '1020'.
  ELSEIF I_WERKS = 'P1STSW'.    " STS
    LV_PLANT1 = '1010'.
  ELSE.
    SELECT SINGLE WERKS INTO LV_PLANT1
      FROM ZPPT0180
     WHERE ZWERKS = I_WERKS.
  ENDIF.

*--- ZPPT0110 기본 데이터 조회
  SELECT PBARCODE ZPLNUM PDA_SHIFT WERKS
         PROD_GR_DATE SHIP_GR_DATE ARBPL
    INTO CORRESPONDING FIELDS OF TABLE LT_DATA
    FROM ZPPT0110
    WHERE PROD_GR_DATE    BETWEEN I_FDATE AND I_TDATE   " 생산 입고일 조회
      AND WERKS           = LV_PLANT1.

*--- 중복 데이터 값 삭제
  SORT LT_DATA BY PBARCODE.
  DELETE ADJACENT DUPLICATES FROM LT_DATA COMPARING PBARCODE.

*--- 필요한 데이터 보강 (ZPPT0170, MAKT)
  LOOP AT LT_DATA INTO LS_DATA.

    CLEAR LV_VTWEG.

    " 2공장 SUB 조건
    IF LV_PLANT1 = '1020' AND I_WERKS = 'P2SUB' AND LS_DATA-ARBPL <> 'PG11'.
      DELETE LT_DATA INDEX SY-TABIX.
      CONTINUE.
    ENDIF.

    " 2공장 STS 조건
    IF LV_PLANT1 = '1010' AND I_WERKS = 'P1STSW' AND ( LS_DATA-ARBPL <> 'PG05' AND LS_DATA-ARBPL <> 'PG06' AND LS_DATA-ARBPL <> 'PG07' ).
      DELETE LT_DATA INDEX SY-TABIX.
      CONTINUE.
    ENDIF.

    " 'PG11'.  'PG05' 'PG06' 'PG07'  이 아닌 경우
    IF  I_WERKS <> 'P1STSW' AND  I_WERKS <> 'P2SUB' .
      IF  LS_DATA-ARBPL <> 'PG11' AND LS_DATA-ARBPL <> 'PG05' AND LS_DATA-ARBPL <> 'PG06' AND LS_DATA-ARBPL <> 'PG07' .
      ELSE.
        DELETE LT_DATA INDEX SY-TABIX.
        CONTINUE.
      ENDIF.
    ENDIF.

    " ZPPT0170: PBarcode 기준으로 잔량 합산 조회
    SELECT SUM( REMNG ) INTO LV_MENGE
      FROM ZPPT0170
     WHERE PBARCODE = LS_DATA-PBARCODE.
    LS_DATA-MENGE = LV_MENGE.

    " ZPPT0170: 품질LOT, 생산타입, 자재번호, 단위 조회
    IF I_WERKS = 'P2SUB' OR I_WERKS = 'P1STSW'.
      SELECT SINGLE QCHARG PRODTYPE MATNR MEINS LGORT
        INTO (LS_DATA-QCHARG, LS_DATA-PRODTYPE, LS_DATA-MATNR, LS_DATA-MEINS, LS_DATA-LGORT)
        FROM ZPPT0170
        WHERE PBARCODE  = LS_DATA-PBARCODE.
    ELSE.
      SELECT SINGLE QCHARG PRODTYPE MATNR MEINS LGORT
        INTO (LS_DATA-QCHARG, LS_DATA-PRODTYPE, LS_DATA-MATNR, LS_DATA-MEINS, LS_DATA-LGORT)
        FROM ZPPT0170
        WHERE PBARCODE  = LS_DATA-PBARCODE
            AND ZWERKSP = I_WERKS.             " 공장구분.
    ENDIF.

    IF SY-SUBRC <> 0.
      DELETE LT_DATA INDEX SY-TABIX.                     " 검색한 공장구분이 아니면 삭제
      CONTINUE.
    ENDIF.

    " MAKT: 자재 내역 조회
    SELECT SINGLE MAKTX
      INTO (LS_DATA-MAKTX)
      FROM MAKT
     WHERE MATNR = LS_DATA-MATNR
       AND SPRAS = SY-LANGU.

    " 최종 결과 T_TAB에 반영
    MOVE-CORRESPONDING LS_DATA TO T_TAB.
    APPEND T_TAB.

  ENDLOOP.



ENDFUNCTION.
