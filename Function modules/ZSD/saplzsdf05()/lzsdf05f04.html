FORM GET_PBARCODE_DATA USING P_BARCOD P_TYPE P_RET P_MSG.

  DATA: LT_ZPPT0170 TYPE TABLE OF ZPPT0170,
            LS_ZPPT0170 TYPE ZPPT0170,
            LS_MAKT     TYPE MAKT.

  DATA: LV_TOTAL_MENGE TYPE ZPPT0170-REMNG.

  CLEAR: GS_RESULT, GT_RESULT, LV_TOTAL_MENGE.

* ZPPT0170 데이터 한 번에 조회
  SELECT *
    INTO TABLE LT_ZPPT0170
    FROM ZPPT0170
    WHERE WERKS IN ('1010', '1020')
       AND   PBARCODE = P_BARCOD.

* 데이터가 없으면 에러 처리
  IF SY-SUBRC <> 0.
    P_RET = 'E'.
    P_MSG = '데이터 조회 오류(PBARCODE 미존재-ZPPT0170)'.
    RETURN.
  ENDIF.

* ABAP 메모리 상에서 중량 합계 계산 (잔량)
  LOOP AT LT_ZPPT0170 INTO LS_ZPPT0170.
    LV_TOTAL_MENGE = LV_TOTAL_MENGE + LS_ZPPT0170-REMNG.
  ENDLOOP.

* 파렛트 중량이 0이면
  IF LV_TOTAL_MENGE IS INITIAL.
    P_RET = 'E'.
    P_MSG = '중량 계산할 데이터가 없습니다.'.
    RETURN.
  ENDIF.

* 대표 데이터 1건 읽기 (첫 번째 레코드 기준)
  READ TABLE LT_ZPPT0170 INTO LS_ZPPT0170 INDEX 1.

* ZPPT0170 데이터 매핑
  MOVE-CORRESPONDING LS_ZPPT0170 TO GS_RESULT.

* 자재명 조회 (MAKT) - PK 조회가 빠름
  SELECT SINGLE MAKTX
    INTO GS_RESULT-MAKTX
    FROM MAKT
    WHERE MATNR = LS_ZPPT0170-MATNR
      AND SPRAS = SY-LANGU.

* 생산실적 정보 조회 (ZPPT0110)
  SELECT SINGLE PROD_GR_DATE SHIP_GR_DATE ZPLNUM HOGI SHIFT T_TIME
                PRTNR SHIP_WK_DATE STATUS CREATED_TIME CREATED_DATE
    INTO CORRESPONDING FIELDS OF GS_RESULT
    FROM ZPPT0110
    WHERE PBARCODE = LS_ZPPT0170-PBARCODE.

* 매핑이 끝난 후, 계산해둔 합계를 최종 할당
  GS_RESULT-MENGE = LV_TOTAL_MENGE.

* 추가 로직 처리
  IF P_TYPE = '시험품'.
    GS_RESULT-LGORT = '5900'.
  ENDIF.

ENDFORM. " reading_pbarcode_data

        LV_DEPT         TYPE ZSDT5000-DEPT,
        LV_TIME_SEC     TYPE I,                " 테이블 생성 시간 비교
        LV_TIME             TYPE T,
        LV_DATETIME     TYPE C LENGTH 14,
        LV_NOW            TYPE C LENGTH 14,
        LV_WERKS(4)     TYPE C,
        LV_VTWEG(2)     TYPE C,
        LV_LGORT(4)     TYPE C.

  " 바코드 정보 조회
  PERFORM GET_PBARCODE_DATA USING I_BARCOD '' E_RET E_MSG.
  IF E_RET = 'E'.
    RETURN.
  ENDIF.

  IF GS_RESULT-STATUS = 'S'
    AND GS_RESULT-PROD_GR_DATE IS INITIAL
    AND GS_RESULT-SHIP_GR_DATE IS INITIAL.

* 해당 ID의 부서 조회
  SELECT SINGLE DEPT
    INTO LV_DEPT

  IF SY-SUBRC <> 0.

* 부서
  CASE LV_DEPT.

*"----------------------------------------------------------------------
*" 생산팀 입고 'PRODUCTION2'
*"----------------------------------------------------------------------
    WHEN 'PRODUCTION1' OR 'PRODUCTION2' OR 'PRODUCTION3' OR 'PRODUCTION4'.
      IF GS_RESULT-PROD_GR_DATE IS INITIAL.

        " 1분 더하기: 초 단위 계산
        LV_TIME_SEC = ( GS_RESULT-CREATED_TIME+0(2) * 3600 )

                    + ( GS_RESULT-CREATED_TIME+4(2) ) + 60.
