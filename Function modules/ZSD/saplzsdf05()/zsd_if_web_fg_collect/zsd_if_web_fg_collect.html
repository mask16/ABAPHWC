FUNCTION ZSD_IF_WEB_FG_COLLECT.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_USERID) LIKE ZSDT5000-USERID OPTIONAL
*" VALUE(I_FDATE) TYPE ZPRODGRDATE OPTIONAL
*" VALUE(I_TDATE) TYPE ZPRODGRDATE OPTIONAL
*" VALUE(I_WERKS) TYPE ZEWERKS OPTIONAL
*" VALUE(I_SHIFT) TYPE ZPDASHIFT OPTIONAL
*" VALUE(I_MARKET) TYPE VTWEG OPTIONAL
*" VALUE(I_MRP) TYPE DISPO OPTIONAL
*" EXPORTING
*" VALUE(E_RET) LIKE ZSDS0001-RET
*" VALUE(E_MSG) LIKE ZSDS0001-MSG
*" TABLES
*" T_TAB STRUCTURE ZSDS0243
*"----------------------------------------------------------------------
* Global data declarations

" (ZPPT0110)의 결과
  TYPES:
    BEGIN OF ty_product,
      werks       TYPE werks_d,
      pbarcode  TYPE zppt0110-pbarcode,
      zplnum     TYPE zppt0110-zplnum, " MAX(ZPLNUM)을 구하기 위한 원본
    END OF ty_product,

    "  (ZPPT0170)의 집계 결과
    BEGIN OF ty_barcode_agg,
      werks       TYPE werks_d,
      pbarcode TYPE zppt0170-pbarcode,
      matnr       TYPE matnr,
      qcharg     TYPE zppt0170-qcharg,
      menge     TYPE zppt0170-menge, " SUM(MENGE)
    END OF ty_barcode_agg,

    " 최종 집계
    BEGIN OF ty_final_agg,
        PBARCODE      LIKE ZPPT0170-PBARCODE,       " 팔레트 바코드
        MAKTX            TYPE MAKTX,                            " 자재내역
        MENGE            LIKE ZPPT0170-MENGE,             " 수량
        QCHARG          LIKE ZPPT0170-QCHARG,           " 품질 LOT ID
        PCOUNT          TYPE I,                                      " 파렛트 개수 카운트용
    END OF ty_final_agg.

*--- 데이터 변수(임시 테이블) 선언 ---*
  DATA:
    lt_prod_raw    TYPE STANDARD TABLE OF ty_product,
    ls_prod_raw    TYPE ty_product,

    " (ZPPT0170) - 필터링 전 원본
    lt_barcode_raw TYPE STANDARD TABLE OF zppt0170,
    ls_barcode_raw TYPE zppt0170,

    " (ZPPT0170) - 필터링 및 집계 후
    lt_barcode_agg TYPE STANDARD TABLE OF ty_barcode_agg,
    ls_barcode_agg TYPE ty_barcode_agg,

    " Join 대상 마스터 데이터
    lt_makt        TYPE STANDARD TABLE OF makt,
    ls_makt       TYPE  makt,
    lt_marc        TYPE STANDARD TABLE OF marc,
    ls_marc       TYPE  marc,
    lt_t024d       TYPE STANDARD TABLE OF t024d,
    ls_t024d      TYPE t024d,
    lt_zppt0090  TYPE STANDARD TABLE OF zppt0090,
    ls_zppt0090  TYPE zppt0090,

    " 최종 집계 테이블
    lt_final_agg   TYPE STANDARD TABLE OF ty_final_agg,
    ls_final_agg   TYPE ty_final_agg,

    "  필요한 변수
    lv_plant1               TYPE werks_d,
    lv_werks                TYPE ZEWERKS,
    lv_check_werks     TYPE zppt0170-zwerks,
    lv_shift_pattern     TYPE char40.   " LIKE 패턴용

  RANGES: lr_mrp_pattern      FOR ls_marc-dispo,  " <--- RANGES로 선언
                lr_market_pattern  FOR ls_zppt0090-vtweg.

  DATA: ls_mrp_range    LIKE LINE OF lr_mrp_pattern,
           ls_market_range LIKE LINE OF lr_market_pattern.

  DATA: ls_result TYPE ZSDS0243. " T_TAB의 구조체

  "DATA: lv_total_menge LIKE ZPPT0170-MENGE,   " 총계 계산 변수
  "         lv_total_count TYPE I.                             " 총계 변수 (파렛트 개수)

  CLEAR: T_TAB, T_TAB[].

*--- 2. 플랜트 코드 변환 ---*
  lv_werks = I_WERKS.  " 공정을 임시로 저장한다.

  IF I_WERKS = 'P2SUB'.
      lv_plant1  = '1020'.
      lv_werks  = 'P2CO2W'.   " 2공장 CO2공정에 SUB가 존재한다
  ELSEIF I_WERKS = 'P1STSW'.
      lv_plant1  = '1010'.
      IF I_MRP = '402'.  " STS SUB 일 경우
          lv_werks = 'P1CO2W'.   " 1공장 CO2
      ENDIF.
  ELSE.
    SELECT SINGLE werks INTO lv_plant1
      FROM zppt0180
      WHERE zwerks = I_WERKS.
  ENDIF.

  IF lv_plant1 IS INITIAL.
    e_ret = 'E'.
    e_msg = '플랜트 코드가 존재하지 않습니다.'.
    RETURN.
  ENDIF.

  IF i_shift IS INITIAL OR i_shift  = 'A'.
      lv_shift_pattern  = '%'.
  ELSE.
      CONCATENATE i_shift '%' INTO  lv_shift_pattern.
  ENDIF.

*--- ZPPT0110 조회 및 MAX(ZPLNUM) 구현 ---*
  SELECT werks pbarcode zplnum
    FROM zppt0110
    INTO TABLE lt_prod_raw
   WHERE  werks = lv_plant1
          AND prod_gr_date BETWEEN i_fdate AND i_tdate
          AND pda_shift LIKE  lv_shift_pattern.      " 전체 / 주간 / 야간

  IF lt_prod_raw IS INITIAL.
    e_ret = 'E'.
    e_msg = '조회 기간에 생산 실적이 없습니다.'.
    RETURN.
  ENDIF.

  " ABAP으로 MAX(ZPLNUM) 구현 (정렬 -> 중복제거)
  SORT lt_prod_raw BY werks pbarcode zplnum DESCENDING.
  DELETE ADJACENT DUPLICATES FROM lt_prod_raw COMPARING werks pbarcode.

  " DB에서 ZPPT0170 원시 데이터 조회
  SELECT *
    FROM zppt0170
    INTO TABLE lt_barcode_raw
    FOR ALL ENTRIES IN lt_prod_raw
   WHERE  werks = lt_prod_raw-werks
         AND pbarcode = lt_prod_raw-pbarcode.

  IF lt_barcode_raw IS INITIAL.
    e_ret = 'E'.
    e_msg = '바코드 정보가  없습니다.'.
    RETURN.
  ENDIF.

  "  LOOP - 170 Total  Col / 공정이 아닌것은 삭제 (P1STSW등)
  LOOP AT lt_barcode_raw INTO ls_barcode_raw.

    IF ls_barcode_raw-pbarcode+6(1) = '1'.
       lv_check_werks = ls_barcode_raw-zwerks.
    ELSE.
      lv_check_werks = ls_barcode_raw-zwerksp.
    ENDIF.

    " 필터 조건과 맞지 않으면 데이터 삭제
    IF lv_check_werks <> lv_werks.
      DELETE lt_barcode_raw.
    ENDIF.

  ENDLOOP.

  " 170 Total  Col ->  PBARCODE MAKTX MENGE QCHARG PCOUNT 로 집계
  LOOP AT lt_barcode_raw INTO ls_barcode_raw.
    CLEAR ls_barcode_agg.
    MOVE-CORRESPONDING ls_barcode_raw TO ls_barcode_agg.

    " COLLECT: 키(문자필드)가 같은 항목을 찾아 숫자필드(itqty, menge, pcount)를 자동 합산
    COLLECT ls_barcode_agg INTO lt_barcode_agg.

  ENDLOOP.

  IF lt_barcode_agg IS INITIAL.
    e_ret = 'E'.
    e_msg = '해당 공정의 자료가 없습니다.'.
    RETURN.
  ENDIF.

  " (B) MAKT 자재 - 자재내역 임시테이블
  SELECT matnr maktx FROM makt INTO CORRESPONDING FIELDS OF TABLE lt_makt
    FOR ALL ENTRIES IN lt_barcode_agg
   WHERE  matnr = lt_barcode_agg-matnr
        AND spras = SY-LANGU.

  " (C) MARC - 자재+ 공정 MRP 관리자 임시테이블
  SELECT matnr werks dispo FROM marc INTO CORRESPONDING FIELDS OF TABLE lt_marc
    FOR ALL ENTRIES IN lt_barcode_agg
   WHERE  matnr = lt_barcode_agg-matnr
         AND werks = lt_barcode_agg-werks.

  " (F) T024D   MRP내역
  IF lt_marc IS NOT INITIAL.
    SELECT werks dispo dsnam FROM t024d INTO CORRESPONDING FIELDS OF TABLE lt_t024d
      FOR ALL ENTRIES IN lt_marc
     WHERE  werks = lt_marc-werks
          AND dispo = lt_marc-dispo.
  ENDIF.

  " (E) ZPPT0090 유통경로
  RANGES: lr_zplnum FOR zppt0090-zplnum.
  DATA: ls_zplnum LIKE LINE OF lr_zplnum.

  CLEAR: lr_zplnum[].

 "  작업지시번호가 같은 걸 배열에 넣는다
  LOOP AT lt_prod_raw INTO ls_prod_raw.
    CHECK ls_prod_raw-zplnum IS NOT INITIAL. " 빈 값은 제외

    ls_zplnum-sign    = 'I'.
    ls_zplnum-option = 'EQ'.
    ls_zplnum-low      = ls_prod_raw-zplnum.

    COLLECT ls_zplnum INTO lr_zplnum. " 중복 제거하며 Range 생성
  ENDLOOP.

  " 작업지시번호가 존재하면
  IF lr_zplnum[] IS NOT INITIAL.
    SELECT zplnum matnr vtweg
      FROM zppt0090
      INTO CORRESPONDING FIELDS OF TABLE lt_zppt0090
     WHERE zplnum IN lr_zplnum.
  ENDIF.

  " .NET에서 넘어온 SQL 와일드카드(%)를 ABAP 와일드카드(*)로 변환
  REPLACE ALL OCCURRENCES OF '%' IN i_mrp WITH '*'.
  REPLACE ALL OCCURRENCES OF '_' IN i_mrp WITH '+'.

*--- MRP 필터 패턴 생성 ---*
  IF i_mrp IS INITIAL OR i_mrp = 'AAA' OR i_mrp = '*'.
    CLEAR lr_mrp_pattern[].

  ELSEIF i_mrp = '302'.      " CO2일 경우
    ls_mrp_range-sign   = 'I'.
    ls_mrp_range-option = 'BT'.
    ls_mrp_range-low    = '301'.
    ls_mrp_range-high   = '302'.
    APPEND ls_mrp_range TO lr_mrp_pattern.
  ELSE.
    ls_mrp_range-sign   = 'I'.
    ls_mrp_range-option = 'CP'.
    ls_mrp_range-low    = i_mrp.
    APPEND ls_mrp_range TO lr_mrp_pattern.
  ENDIF.

*--- MARKET 필터 패턴 생성 ---*
  REPLACE ALL OCCURRENCES OF '%' IN i_market WITH '*'.
  REPLACE ALL OCCURRENCES OF '_' IN i_market WITH '+'.

  IF i_market IS INITIAL OR i_market = 'AA' OR i_market = '*'.
    CLEAR lr_market_pattern[].
  ELSE.
    ls_market_range-sign   = 'I'.
    ls_market_range-option = 'CP'.
    ls_market_range-low    = i_market.
    APPEND ls_market_range TO lr_market_pattern.
  ENDIF.

  " 메인 루프 (Subquery A와 D 조인)
  LOOP AT lt_prod_raw INTO ls_prod_raw. " (A 테이블 기준)
    " (A와 D 조인)
    LOOP AT lt_barcode_agg INTO ls_barcode_agg
        WHERE werks    = ls_prod_raw-werks
          AND pbarcode = ls_prod_raw-pbarcode.

      " (D와 C 조인) mrp관리자 없으면 skip
      READ TABLE lt_marc INTO ls_marc
        WITH KEY   matnr = ls_barcode_agg-matnr
                         werks = ls_barcode_agg-werks.

      IF sy-subrc <> 0. CONTINUE. ENDIF. " (MARC 데이터 없음)

      " (C와 F 조인)  MRP내역
      READ TABLE lt_t024d INTO ls_t024d
        WITH KEY werks = ls_marc-werks
                       dispo = ls_marc-dispo.
      " (T024D는 데이터가 없을 수 있으므로 sy-subrc 체크 생략)

      " (A와 E 조인) 유통경로 없으면 skip
      READ TABLE lt_zppt0090 INTO ls_zppt0090
        WITH KEY zplnum = ls_prod_raw-zplnum
                       matnr  = ls_barcode_agg-matnr.

      IF sy-subrc <> 0. CONTINUE. ENDIF. " (ZPPT0090 데이터 없음)

      " (D와 B 조인) 자재내역
      READ TABLE lt_makt INTO ls_makt
        WITH KEY matnr = ls_barcode_agg-matnr.

      " lr_mrp_pattern이 비어있고 'AAA' - 전체
      " 전체 - 2공장 CO2 - MRP (SUB 제외)
      IF lr_mrp_pattern[] IS  INITIAL AND  lv_werks  = 'P2CO2W' AND  ls_marc-dispo = '401'.
          CONTINUE. " (MRP 조건 불일치)
      ENDIF.

      " lr_mrp_pattern이 비어있고 'AAA' - 전체
      " 전체 - 1공장 CO2 - MRP (STS SUB 제외)
      IF lr_mrp_pattern[] IS  INITIAL AND  lv_werks  = 'P1CO2W' AND  ls_marc-dispo = '402'.
          CONTINUE. " (MRP 조건 불일치)
      ENDIF.

      " lr_mrp_pattern이 비어있지 않고(즉, '전체'가 아니고)
      " ls_marc-dispo가 lr_mrp_pattern 조건 바구니에 포함되지 않으면(NOT IN)
      IF lr_mrp_pattern[] IS NOT INITIAL AND
         ls_marc-dispo NOT IN lr_mrp_pattern.
        CONTINUE. " (MRP 조건 불일치)
      ENDIF.

      " (SQL의 'E.VTWEG LIKE I_MARKET' 구현)
      IF lr_market_pattern[] IS NOT INITIAL AND
         ls_zppt0090-vtweg NOT IN lr_market_pattern.
        CONTINUE. " (MARKET 조건 불일치)
      ENDIF.

      " 최종 집계
      CLEAR ls_final_agg.

      " ls_final_agg-pbarcode = ls_barcode_agg-pbarcode.
      CLEAR LS_FINAL_AGG-PBARCODE.

      ls_final_agg-maktx      = ls_makt-maktx.
      ls_final_agg-menge     = ls_barcode_agg-menge.
      ls_final_agg-qcharg     = ls_barcode_agg-qcharg.
      ls_final_agg-pcount     = 1.

      " COLLECT: 키(MAKTX, QCHARG)가 같으면 숫자(PCOUNT, MENGE)를 합산
      COLLECT ls_final_agg INTO lt_final_agg.

    ENDLOOP. " (lt_barcode_agg)
  ENDLOOP. " (lt_prod_raw)

*--- 최종 결과를 T_TAB으로 전송 ---*
  IF lt_final_agg IS INITIAL.
    e_ret = 'E'.
    e_msg = '모든 조건을 만족하는 데이터가 없습니다.'.
    RETURN.
  ENDIF.

  LOOP AT lt_final_agg INTO ls_final_agg.
    CLEAR ls_result.

    " ls_result-pbarcode  = ls_final_agg-pbarcode.
    ls_result-pbarcode  = ls_final_agg-pcount.
    ls_result-maktx       = ls_final_agg-maktx.
    ls_result-menge      = ls_final_agg-menge.
    ls_result-qcharg      = ls_final_agg-qcharg.

    APPEND ls_result TO  T_TAB.

  ENDLOOP.

  e_ret = 'S'.
  e_msg = '데이터 조회가 완료되었습니다.'.

ENDFUNCTION.
