*&      Form  STRUCTURE_COMPARE
*&---------------------------------------------------------------------*
FORM structure_compare USING p_tname
                                 p_org
                                 p_trs
                        CHANGING p_same.

  CLEAR p_same.
  FIELD-SYMBOLS : <fs_org>, <fs_trs>.

*  ASSIGN P_ORG TO <F_ORG>.
*  ASSIGN P_TRS TO <F_TRS>.
*
*  CLEAR: <F_ORG>-ERDAT, <F_ORG>-ERZET, <F_ORG>-ERNAM,
*         <F_ORG>-UPDAT, <F_ORG>-UPTIM, <F_ORG>-UPNAM.
*  CLEAR: <F_TRS>-ERDAT, <F_TRS>-ERZET, <F_TRS>-ERNAM,
*         <F_TRS>-UPDAT, <F_TRS>-UPTIM, <F_TRS>-UPNAM.
*  IF <F_ORG> = <F_TRS>.
*    P_SAME = 'X'.
*  ENDIF.
  DATA : lt_fcat TYPE lvc_t_fcat,
         ls_fcat TYPE lvc_s_fcat.
  DATA : l_diff TYPE c.

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name       = p_tname
    CHANGING
      ct_fieldcat            = lt_fcat
    EXCEPTIONS
      inconsistent_interface = 1
      program_error          = 2
      OTHERS                 = 3.

  WHILE sy-subrc = 0.
    ASSIGN COMPONENT sy-index OF STRUCTURE p_org  TO <fs_org>.
    ASSIGN COMPONENT sy-index OF STRUCTURE p_trs  TO <fs_trs>.
    IF <fs_org> <> <fs_trs>.
      READ TABLE lt_fcat INTO ls_fcat WITH KEY col_pos = sy-index.
      IF ls_fcat-fieldname <> 'MANDT' AND ls_fcat-fieldname <> 'ERNAM' AND
         ls_fcat-fieldname <> 'ERDAT' AND ls_fcat-fieldname <> 'ERZET' AND
         ls_fcat-fieldname <> 'UPDAT' AND ls_fcat-fieldname <> 'UPTIM' AND
        ls_fcat-fieldname <> 'UPNAM'.
        l_diff = 'X'.
      ENDIF.
    ENDIF.

    IF l_diff = 'X'.
      EXIT.
    ENDIF.
  ENDWHILE.

  IF l_diff = 'X'.
    CLEAR p_same.
  ELSE.
    p_same = 'X'.
  ENDIF.

ENDFORM. " STRUCTURE_COMPARE
