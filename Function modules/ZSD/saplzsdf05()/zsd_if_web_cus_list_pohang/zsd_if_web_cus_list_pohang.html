FUNCTION ZSD_IF_WEB_CUS_LIST_POHANG .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_WADAT) LIKE SY-DATUM DEFAULT SY-DATUM
*" VALUE(I_TODAT) LIKE SY-DATUM DEFAULT SY-DATUM
*" VALUE(I_FLAG) TYPE CHAR01 OPTIONAL
*" EXPORTING
*" VALUE(E_RET) LIKE ZSDS0001-RET
*" VALUE(E_MSG) LIKE ZSDS0001-MSG
*" TABLES
*" T_TAB STRUCTURE ZSDS0160
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_GUBUN  LIKE ZSDT5000-GUBUN,
         L_PLANT1 LIKE ZSDT5000-PLANT1,
         L_PLANT2 LIKE ZSDT5000-PLANT2,
         L_VSTEL  LIKE ZSDT5000-VSTEL,
         L_CNTNO  LIKE ZSDS0160-CNTNO,
         L_USER   TYPE CHAR2.

  TYPES : BEGIN OF TYP_DATA_160,
            KUNNR TYPE LIKP-KUNNR,
            NAME1 TYPE KNA1-NAME1,
            VBELN TYPE VBUK-VBELN,
            WBSTK TYPE VBUK-WBSTK,
            EBELN TYPE EKPO-EBELN,
            ADRNR TYPE VBPA-ADRNR,
            LGORT TYPE LIPS-LGORT,
          END OF TYP_DATA_160.

  DATA : LS_160 TYPE ZSDS0160,
         LT_TAB TYPE STANDARD TABLE OF TYP_DATA_160,
         LS_TAB TYPE TYP_DATA_160.

  DATA : LV_LGORT TYPE T001L-LGORT.

  DATA : LS_5071 LIKE ZSDT5071,
             LT_5071 LIKE TABLE OF LS_5071.

  CLEAR : E_RET, E_MSG.
  CLEAR : T_TAB, T_TAB[].

  IF I_WADAT IS INITIAL.
    E_RET = 'E'.
    E_MSG = '출하일자를 입력해주세요'.
    EXIT.
  ENDIF.

  CLEAR : GR_WERKS, GR_WERKS[], GR_LGORT, GR_LGORT[], GR_BUDAT, GR_BUDAT[], GR_VSTEL, GR_VSTEL[].

  "2026.03.06 NYG 플랜트 -> 저장위치 I/F 바뀌어 기존 로직 주석처리
* GR_LGORT-LOW = '9000'. GR_LGORT-SIGN = 'E'. GR_LGORT-OPTION = 'EQ'. APPEND GR_LGORT. CLEAR GR_LGORT.
* GR_LGORT-LOW = 'CY00'. GR_LGORT-SIGN = 'E'. GR_LGORT-OPTION = 'EQ'. APPEND GR_LGORT. CLEAR GR_LGORT.

  "저장위치 구분
  CASE I_FLAG.
    WHEN '1'. "내수
      "포항1공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1010'.
      "포항2공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1020'.
      "포항1공장
      PERFORM APPEND_WERKS USING '1010'.
      "포항2공장
      PERFORM APPEND_WERKS USING '1020'.

    WHEN '2'. "수출
      "포항1공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1011'.
      "포항2공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1021'.
      "포항1공장 수출
      PERFORM APPEND_WERKS USING '1011'.
      "포항2공장 수출
      PERFORM APPEND_WERKS USING '1021'.

    WHEN OTHERS. "전체(A) or 빈 값일 때
      "포항1공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1010'.
      "포항2공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1020'.
      "포항1공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1011'.
      "포항2공장 내수 저장위치
      PERFORM APPEND_LGORT USING '1021'.
      "포항1공장
      PERFORM APPEND_WERKS USING '1010'.
      "포항2공장
      PERFORM APPEND_WERKS USING '1020'.
      "포항1공장 수출
      PERFORM APPEND_WERKS USING '1011'.
      "포항2공장 수출
      PERFORM APPEND_WERKS USING '1021'.

  ENDCASE.

  IF I_TODAT = 00000000.
    GR_BUDAT-LOW = I_WADAT. GR_BUDAT-SIGN = 'I'. GR_BUDAT-OPTION = 'EQ'. APPEND GR_BUDAT. CLEAR GR_BUDAT.
  ELSE.
    GR_BUDAT-LOW = I_WADAT. GR_BUDAT-HIGH = I_TODAT. GR_BUDAT-SIGN = 'I'.
    GR_BUDAT-OPTION = 'BT'. APPEND GR_BUDAT. CLEAR GR_BUDAT.
  ENDIF.

  SELECT A~KUNNR F~NAME1 C~VBELN C~WBSTK E~EBELN D~ADRNR
    FROM LIKP AS A INNER JOIN LIPS AS B
                      ON A~VBELN = B~VBELN
                   INNER JOIN VBPA AS D
                      ON A~VBELN = D~VBELN
                     AND D~PARVW = 'WE' "납품처
                   INNER JOIN ADRC AS F
                      ON D~ADRNR = F~ADDRNUMBER
              LEFT OUTER JOIN VBUK AS C
                      ON A~VBELN = C~VBELN
                     AND C~WBSTK EQ 'A'       " 전체 제품이동상태 : 아직 처리되지 않음
              LEFT OUTER JOIN EKPO AS E
                      ON B~VGBEL = E~EBELN
   INTO CORRESPONDING FIELDS OF TABLE LT_TAB
  WHERE A~WADAT IN GR_BUDAT AND A~VSTEL IN GR_WERKS AND A~LFART NOT IN ('LR', 'RL') "반품
    AND B~LFIMG GT 0 AND B~LGORT IN GR_LGORT.

  DELETE LT_TAB WHERE WBSTK NE 'A'.

  LOOP AT LT_TAB INTO LS_TAB.
    IF LS_TAB-KUNNR <> '0000000001'. LS_TAB-ADRNR = ''.  ENDIF.
    MODIFY LT_TAB FROM LS_TAB.
  ENDLOOP.

  SORT LT_TAB BY KUNNR NAME1 VBELN ADRNR.
  DELETE ADJACENT DUPLICATES FROM LT_TAB COMPARING KUNNR NAME1 VBELN ADRNR.

  LOOP AT LT_TAB INTO LS_TAB.
    MOVE-CORRESPONDING LS_TAB TO LS_160.
    L_CNTNO = L_CNTNO + 1.
    AT END OF NAME1.
      LS_160-CNTNO = L_CNTNO.
      APPEND LS_160 TO T_TAB. CLEAR : LS_160, L_CNTNO.
    ENDAT.
  ENDLOOP.

  IF T_TAB[] IS NOT INITIAL.
    SELECT * INTO CORRESPONDING FIELDS OF TABLE LT_5071
       FROM ZSDT5071
          FOR ALL ENTRIES IN T_TAB
     WHERE KUNNR EQ T_TAB-KUNNR.

    SORT T_TAB BY KUNNR.
    SORT LT_5071 BY KUNNR.
    LOOP AT T_TAB.
      READ TABLE LT_5071 INTO LS_5071 WITH KEY KUNNR = T_TAB-KUNNR.
      IF SY-SUBRC EQ 0.
        T_TAB-ZGROUP = LS_5071-ZGROUP.
      ELSE.
        T_TAB-ZGROUP = 'Z'. "해당되지 않는 것
      ENDIF.
      MODIFY T_TAB.
    ENDLOOP.
  ENDIF.

  SORT T_TAB BY ZGROUP KUNNR.

  IF T_TAB[] IS INITIAL.
    E_RET = 'E'.
    E_MSG = '출하일자에 해당하는 거래처가 없습니다'.
  ELSE.
    E_RET = 'S'.
    E_MSG = TEXT-S01.
  ENDIF.

ENDFUNCTION.
