FUNCTION ZFI_ACC_DOC_READ.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BUKRS) LIKE BKPF-BUKRS
*" REFERENCE(I_BELNR) TYPE BKPF-BELNR
*" REFERENCE(I_GJAHR) TYPE BKPF-GJAHR
*" EXPORTING
*" REFERENCE(E_MESSAGE) TYPE BAPI_MSG
*" TABLES
*" T_BKPF STRUCTURE BKPF OPTIONAL
*" T_BSEG STRUCTURE BSEG OPTIONAL
*" T_BSET STRUCTURE BSET OPTIONAL
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LT_VBKPF LIKE FVBKPF OCCURS 0 WITH HEADER LINE,
           LT_VBSEG LIKE FVBSEG OCCURS 0 WITH HEADER LINE,
           LT_VBSEC LIKE FVBSEC OCCURS 0 WITH HEADER LINE,
           LT_VBSET LIKE FVBSET OCCURS 0 WITH HEADER LINE.

  DATA : LS_ITXDAT LIKE ITXDAT,
         L_BUZEI   TYPE BUZEI,
         L_MWSKZ   TYPE MWSKZ,
         L_MWSKV   TYPE T001-MWSKV, " 세금코드 무관 매입
         L_MWSKA   TYPE T001-MWSKA. " 세금코드 무관 매출

  DATA : EX_MWSKZ TYPE RANGE OF MWSKZ.    " 세금코드점검용

  DATA : BEGIN OF XBKPF OCCURS 0.
          INCLUDE STRUCTURE BKPF.
  DATA: END OF XBKPF,
  LS_BKPF LIKE BKPF.

  DATA : BEGIN OF XBSEG OCCURS 0.
          INCLUDE STRUCTURE BSEG.
  DATA: END OF XBSEG,
  LS_BSEG LIKE BSEG.

  DATA : BEGIN OF XBSET OCCURS 0.
          INCLUDE STRUCTURE BSET.
  DATA: END OF XBSET,
  LS_BSET LIKE BSET.

  DATA : L_CALCULATE(1),
         L_FWSTE        TYPE BSET-FWSTE,
         L_BSTAT        TYPE BKPF-BSTAT,
         L_XBSET        TYPE VBKPF-XBSET,
         L_XMWST        TYPE VBKPF-XMWST,
         L_DSUM         TYPE WERTV12,
         L_WSUM         TYPE WERTV12.

  DATA : LS_T001 LIKE T001.
*----------------------------------------------------------------------*

  SELECT SINGLE * FROM T001 INTO LS_T001 WHERE BUKRS = I_BUKRS.

* 임시전표 여부
  SELECT SINGLE BSTAT FROM BKPF INTO L_BSTAT
                               WHERE BUKRS = I_BUKRS
                                 AND BELNR = I_BELNR
                                 AND GJAHR = I_GJAHR.

  IF L_BSTAT = 'V'.
* 임시전표정보
    SELECT SINGLE XBSET XMWST FROM VBKPF
                              INTO (L_XBSET, L_XMWST)
                             WHERE BUKRS = I_BUKRS
                               AND BELNR = I_BELNR
                               AND GJAHR = I_GJAHR.
* 임시전표
    CLEAR : LT_VBKPF, LT_VBSEG, LT_VBSEC, LT_VBSET,
            LT_VBKPF[], LT_VBSEG[], LT_VBSEC[], LT_VBSET[].

    CALL FUNCTION 'PRELIMINARY_POSTING_DOC_READ'
      EXPORTING
        BELNR                   = I_BELNR
        BUKRS                   = I_BUKRS
        GJAHR                   = I_GJAHR
      TABLES
        T_VBKPF                 = LT_VBKPF
        T_VBSEC                 = LT_VBSEC
        T_VBSEG                 = LT_VBSEG
        T_VBSET                 = LT_VBSET
      EXCEPTIONS
        DOCUMENT_LINE_NOT_FOUND = 1
        DOCUMENT_NOT_FOUND      = 2
        INPUT_INCOMPLETE        = 3
        OTHERS                  = 4.
    IF SY-SUBRC <> 0.
* Error
      MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
              INTO E_MESSAGE
              WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      RAISE ERROR.
    ENDIF.

    CLEAR : LS_ITXDAT, L_BUZEI, L_MWSKZ.

    READ TABLE LT_VBKPF  INDEX 1.
    LOOP AT LT_VBSEG  WHERE KOART = 'K' OR KOART = 'D'.
      CASE LT_VBSEG-SHKZG.
        WHEN 'S'.
          LS_ITXDAT-SHKZG = 'H'.
        WHEN 'H'.
          LS_ITXDAT-SHKZG = 'S'.
      ENDCASE.
      LS_ITXDAT-FWSTE = LS_ITXDAT-HWSTE = LT_VBSEG-MWSTS.
      L_BUZEI = LT_VBSEG-BUZEI.
      L_MWSKZ = LT_VBSEG-MWSKZ.
    ENDLOOP.

* Tax info.
    L_MWSKV = LS_T001-MWSKV.
    L_MWSKA = LS_T001-MWSKA.

    CALL FUNCTION 'SET_TAX_AMOUNT_DK'
      EXPORTING
        I_BUKRS  = I_BUKRS
        I_BUZEI  = L_BUZEI
        I_ITXDAT = LS_ITXDAT
        I_MWSKZ  = L_MWSKZ
        I_XMWST  = LT_VBKPF-XMWST.

* Set BSET
    REFRESH : XBKPF, XBSEG, XBSET.
    LOOP AT LT_VBKPF.
      CLEAR LS_BKPF.
      MOVE-CORRESPONDING LT_VBKPF TO LS_BKPF.
      CLEAR LS_BKPF-BSTAT.
      LS_BKPF-GLVOR = 'RFBV'.
      APPEND LS_BKPF TO XBKPF.
      CLEAR LS_BKPF.
    ENDLOOP.

    LOOP AT LT_VBSEG.
      CLEAR LS_BSEG.
      MOVE-CORRESPONDING LT_VBSEG TO LS_BSEG.
      LS_BSEG-PSWBT = LS_BSEG-DMBTR.
      LS_BSEG-PSWSL = LS_T001-WAERS.
      APPEND LS_BSEG TO XBSEG.
      CLEAR LS_BSEG.
    ENDLOOP.

    IF E_MESSAGE <> SPACE.
* Error
      RAISE ERROR.
    ENDIF.
*
    EXPORT XBKPF XBSEG TO MEMORY ID '%BKPF%'.

    READ TABLE LT_VBKPF INDEX 1.

    CALL FUNCTION 'CALCULATE_TAX_DOCUMENT'
      EXPORTING
        I_BUKRS                    = I_BUKRS
        I_NOERR                    = 'X'
        I_XNOMS                    = 'X'
      IMPORTING
        E_ITXDAT                   = LS_ITXDAT
        E_XSTVR                    = LT_VBKPF-XMWST
      TABLES
        T_BKPF                     = XBKPF
        T_BSEG                     = XBSEG
* T_DBSEG =
      EXCEPTIONS
        ERROR_CALCULATE_DISCOUNTB  = 1
        MWSKZ_NOT_DEFINED          = 2
        USER_EXIT                  = 3
        MWSKZ_EINHEITLICH_VORGEBEN = 4
        STEUERBETRAG_FALSCH        = 5
        OTHERS                     = 6.

    IF SY-SUBRC <> 0.
* Error
      MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
              INTO E_MESSAGE
              WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      RAISE ERROR.
    ENDIF.

    CLEAR L_FWSTE.

    CALL FUNCTION 'CREATE_BSET_ITEM'
      IMPORTING
        STEUER_FW         = L_FWSTE
      TABLES
        T_BKPF            = XBKPF
        T_BSEG            = XBSEG
        T_BSET            = XBSET
      EXCEPTIONS
        ACCOUNT_NOT_FOUND = 1
        BSCHL_NOT_FOUND   = 2
        TOO_MANY_ITEMS    = 3
        OTHERS            = 4.

    IF SY-SUBRC <> 0.
* Error
      MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
              INTO E_MESSAGE
              WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      RAISE ERROR.
    ENDIF.

    IF L_XBSET = 'X'.
* 전표에 세금입력
      LOOP AT XBSET.
        CLEAR LT_VBSET.
        READ TABLE LT_VBSET WITH KEY MWSKZ = XBSET-MWSKZ.
        IF SY-SUBRC = 0.
          XBSET-HWSTE = LT_VBSET-HWSTE.
          XBSET-FWSTE = LT_VBSET-FWSTE.
          MODIFY XBSET.
        ELSE.
          DELETE XBSET.
        ENDIF.
      ENDLOOP.
    ENDIF.

* 세금 0 제외
    DELETE XBSET WHERE HWSTE = 0.

** 세금 자동계산 -> 동일 TaxGroup내에서 안분

    LOOP AT XBSET.
      READ TABLE XBSEG WITH KEY MWSKZ = XBSET-MWSKZ
                                HKONT = XBSET-HKONT
                                DMBTR = XBSET-HWSTE
                            TRANSPORTING NO FIELDS.
      IF SY-SUBRC = 0.
        DELETE XBSET.
      ENDIF.
    ENDLOOP.

  ELSE.
* 전기전표
    SELECT * FROM BKPF INTO CORRESPONDING FIELDS OF TABLE XBKPF
                      WHERE BUKRS = I_BUKRS
                        AND BELNR = I_BELNR
                        AND GJAHR = I_GJAHR.

    SELECT * FROM BSEG INTO CORRESPONDING FIELDS OF TABLE XBSEG
                     WHERE BUKRS = I_BUKRS
                       AND BELNR = I_BELNR
                       AND GJAHR = I_GJAHR.

    SORT XBSEG BY BUKRS BELNR GJAHR BUZEI.

  ENDIF.

* Return
  CHECK E_MESSAGE = SPACE.

  T_BKPF[] = XBKPF[].
  T_BSEG[] = XBSEG[].


  IF  XBSET[] IS NOT INITIAL.
    DELETE XBSET WHERE HKONT = ''.
  ENDIF.

  T_BSET[] = XBSET[].



ENDFUNCTION.
