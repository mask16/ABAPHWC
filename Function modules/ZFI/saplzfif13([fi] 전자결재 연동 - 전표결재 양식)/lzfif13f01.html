FORM set_html CHANGING e_html.

  DATA : i_header TYPE TABLE OF w3head   WITH HEADER LINE,
         i_fields TYPE TABLE OF w3fields WITH HEADER LINE,
         i_html   TYPE TABLE OF w3html   WITH HEADER LINE.

* Set Header
  CLEAR : i_header, i_header[], i_fields, i_fields[], i_html, i_html[].

  DO 6 TIMES.
    CLEAR : i_header, i_fields, i_html.
    i_header-nr = i_fields-nr = sy-index.
    CASE i_header-nr.
      WHEN 1.
        i_header-just = '"5"'.
        i_header-text = '상세'(h00).  " ' '.
        i_fields-just = '"5B"'.
      WHEN 2.
        i_header-just = '"15"'.
        i_header-text = '전표번호'(h01).
        i_fields-just = '"15B"'.
      WHEN 3.
        i_header-just = '"15"'.
        i_header-text = '전기일'(h02).
        i_fields-just = '"15B"'.
      WHEN 4.
        i_header-just = '"5"'.
        i_header-text = '통화'(h03).
        i_fields-just = '"5B"'.
      WHEN 5.
        i_header-just = '"15"'.
        i_header-text = '금액'(h04).
        i_fields-just = '"15R"'.
      WHEN 6.
        i_header-just = '"45"'.
        i_header-text = '전표 적요'(h05).
        i_fields-just = '"45B"'.
    ENDCASE.
    i_header-bg_color  = '"#F6F6F6"'.

* Append
    APPEND : i_header, i_fields.
  ENDDO.

  IF gv_head = space.
    gv_head = 'X'.                 " head flag (head는 한번만)
  ELSE.
    CLEAR : i_header, i_header[].
  ENDIF.

* Create HTML
  PERFORM itab_to_html TABLES i_header i_fields i_html gt_head
                       USING 'H'.

* Append
  APPEND LINES OF i_html TO gt_html.

* Set Item
  CLEAR : i_header, i_header[], i_fields, i_fields[], i_html, i_html[].

  DO 8 TIMES.
    CLEAR : i_header, i_fields, i_html.
    i_header-nr = i_fields-nr = sy-index.
    CASE i_header-nr.
      WHEN 1.
        i_header-just = '"15"'.
        i_header-text = '계정과목명'(i01).
        i_fields-just = '"center"'.
      WHEN 2.
        i_header-just = '"20"'.
        i_header-text = '거래처명'(i02).
        i_fields-just = '"center"'.
      WHEN 3.
        i_header-just = '"12"'.
        i_header-text = '지급일'(i03).
        i_fields-just = '"center"'.
      WHEN 4.
        i_header-just = '"12"'.
        i_header-text = '차변금액 '(i04).
        i_fields-just = '"right"'.
      WHEN 5.
        i_header-just = '"12"'.
        i_header-text = '대변금액 '(i05).
        i_fields-just = '"right"'.
      WHEN 6.
        i_header-just = '"4"'.
        i_header-text = '세금코드'(i06).
        i_fields-just = '"center"'.
      WHEN 7.
        i_header-just = '"10"'.
        i_header-text = '코스트센터'(i07).
        i_fields-just = '"center"'.
      WHEN 8.
        i_header-just = '"15"'.
        i_header-text = '적요'(i08).
        i_fields-just = '"center"'.
    ENDCASE.
    i_header-bg_color  = '"#F6F6F6"'.

* Append
    APPEND : i_header, i_fields.
  ENDDO.

* Create HTML
  PERFORM itab_to_html TABLES i_header i_fields i_html gt_item
                       USING 'I'.

* Append
  APPEND LINES OF i_html TO gt_html.

ENDFORM.

*& Form ITAB_TO_HTML
*&---------------------------------------------------------------------*
FORM itab_to_html TABLES i_header STRUCTURE w3head
                         i_fields STRUCTURE w3fields
                         i_html   STRUCTURE w3html
                         lt_itab
                   USING pf_hi.

  DATA : ls_str TYPE string,
         s_html TYPE w3html.

  s_html-line = `border="1" cellspacing="0" cellpadding="1" `
             && `class="sap_fi"`.

* itab -> html
  CALL FUNCTION 'WWW_ITAB_TO_HTML'
    EXPORTING
      table_attributes = s_html
*     table_header     =
      all_fields       = 'X'
    TABLES
      html             = i_html
      fields           = i_fields
      row_header       = i_header
      itable           = lt_itab.

* html -> string
  PERFORM html_to_string TABLES i_html CHANGING ls_str.

  CLEAR : i_html, i_html[].

  SPLIT ls_str AT '<' INTO TABLE i_html IN CHARACTER MODE.

  LOOP AT i_html.
    IF i_html-line IS INITIAL OR
       i_html-line CP 'script>*' OR
       i_html-line CP '/script>*' OR
       i_html-line CP 'span *' OR
       i_html-line CP '/span>*'.
      DELETE i_html.
    ELSEIF i_html-line = '/table>' AND pf_hi = 'H'.
      DELETE i_html.           " Header인 경우, </table>을 Item뒤로
    ELSE.
      i_html-line = '<' && i_html-line.
      REPLACE 'align="4"' IN i_html-line
         WITH 'align="center" width="4%"'.
      REPLACE 'align="5"' IN i_html-line
         WITH 'align="center" width="5%"'.
      REPLACE 'align="5B"' IN i_html-line
         WITH 'align="center" width="5%"><b'.
      REPLACE 'align="10"' IN i_html-line
         WITH 'align="center" width="10%"'.
      REPLACE 'align="10B"' IN i_html-line
         WITH 'align="center" width="10%"><b'.
      REPLACE 'align="12"' IN i_html-line
         WITH 'align="center" width="12%"'.
      REPLACE 'align="15"' IN i_html-line
         WITH 'align="center" width="15%"'.
      REPLACE 'align="15B"' IN i_html-line
         WITH 'align="center" width="15%"><b'.
      REPLACE 'align="15R"' IN i_html-line
         WITH 'align="right"  width="15%"><b'.
      REPLACE 'align="20"' IN i_html-line
         WITH 'align="center" width="20%"'.
      REPLACE 'align="20R"' IN i_html-line
         WITH 'align="right"  width="20%"><b'.
      REPLACE 'align="45"' IN i_html-line
         WITH 'align="center" width="45%"'.
      REPLACE 'align="45B"' IN i_html-line
         WITH 'align="center" width="45%"><b'.
      REPLACE 'align="50"' IN i_html-line
         WITH 'align="center" width="50%"'.
      REPLACE 'align="50B"' IN i_html-line
         WITH 'align="center" width="50%"><b'.
* Set link
      IF i_html-line CP `*□*`.
        i_html-line = `<td class="fi_maintr" tabindex="0"`
                      && ` `
                      && `align="center" width="5%" >`
                      && `<a href="#" onclick="return false;">□</a>`.
      ENDIF.

* header / bold
      IF pf_hi = 'H' AND i_html-line = '</td>'.
        i_html-line = '</b></td>'.
      ENDIF.

* 금액필드
      IF i_html-line CP '*align="right"*'.
        i_html-line = i_html-line && ` `.
      ENDIF.

      MODIFY i_html.
    ENDIF.
  ENDLOOP.

  IF pf_hi = 'I'.
* Item인 경우, 속성 추가
*    i_html-line = `<tr style="display:none"><td colspan="8">`.
    i_html-line = `<tr style="display:table-row;"><td colspan="8">`.
    INSERT i_html INDEX 1.

    i_html-line = `<br></td></tr></table>`.
    APPEND i_html.
  ENDIF.

ENDFORM.
