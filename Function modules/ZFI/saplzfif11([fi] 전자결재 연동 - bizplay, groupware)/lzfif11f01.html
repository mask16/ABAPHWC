FORM ZVALUE_INPUT.

  SELECT SINGLE A~NAME_TEXT
    INTO ZFIV0040-NAME_TEXT
    FROM ADRP AS A
    INNER JOIN USR21 AS B
    ON A~PERSNUMBER EQ B~PERSNUMBER
    WHERE B~BNAME EQ ZFIV0040-BNAME.

  ZFIV0040-ERNAM = SY-UNAME.
  ZFIV0040-ERDAT = SY-DATUM.
  ZFIV0040-ERZET = SY-UZEIT.

ENDFORM.
