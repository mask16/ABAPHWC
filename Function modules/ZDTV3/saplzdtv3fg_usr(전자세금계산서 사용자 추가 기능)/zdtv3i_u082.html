DATA: L_KUNNR TYPE KUNNR.

WAIT UP TO 1 SECONDS.

*
LOOP AT INV_HD WHERE AMEND_CODE NE SPACE.
  CLEAR: L_KUNNR.

  SELECT SINGLE B~KUNNR INTO L_KUNNR
    FROM ZDTV3T_AR_HEAD AS A INNER JOIN ZDTV3T_AR_EXT AS B
      ON A~BUKRS       = B~BUKRS
     AND A~ISSUE_DATE  = B~ISSUE_DATE
     AND A~BUPLA       = B~BUPLA
     AND A~INV_SEQ     = B~INV_SEQ
   WHERE ISSUE_ID EQ INV_HD-ISSUE_ID_OLD.

  IF L_KUNNR NE SPACE.
    UPDATE ZDTV3T_AR_EXT SET KUNNR   = L_KUNNR
                       WHERE INV_SEQ = INV_HD-INV_SEQ.

  ENDIF.

ENDLOOP.
