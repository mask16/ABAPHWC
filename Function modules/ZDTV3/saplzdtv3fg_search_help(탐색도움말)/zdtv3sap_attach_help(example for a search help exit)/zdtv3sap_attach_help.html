Description: Example for a search help exit
FUNCTION ZDTV3SAP_ATTACH_HELP.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" VALUE(SHLP) TYPE SHLP_DESCR
*" VALUE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : PATH TYPE TEXT1000.

  IF CALLCONTROL-STEP EQ 'SELONE'.

    CALL FUNCTION 'WS_FILENAME_GET'
      EXPORTING
        DEF_PATH         = 'C:\'
        MASK             = ',*.*,*.*.'
        MODE             = 'O'
      IMPORTING
        FILENAME         = PATH
      EXCEPTIONS
        INV_WINSYS       = 1
        NO_BATCH         = 2
        SELECTION_CANCEL = 3
        SELECTION_ERROR  = 4
        OTHERS           = 5.

     IF SY-SUBRC EQ 0.
       CLEAR RECORD_TAB[].
       RECORD_TAB-STRING = PATH.
       APPEND RECORD_TAB.
     ENDIF.

    CALLCONTROL-STEP = 'RETURN'.
    EXIT.
  ENDIF.

ENDFUNCTION.
