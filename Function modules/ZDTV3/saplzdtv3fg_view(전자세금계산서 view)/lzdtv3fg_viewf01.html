FORM SELECT_KUNNR.
  CASE VIEW_NAME.
    WHEN 'ZDTV3V_CUST_MIN'.
      CHECK NOT ZDTV3V_CUST_MIN-MIN_STCD2  IS INITIAL.
      SELECT SINGLE NAME1 INTO KNA1-NAME1
        FROM KNA1
       WHERE STCD2 = ZDTV3V_CUST_MIN-MIN_STCD2.
    WHEN 'ZDTV3V_CUST_PERS'.
      CHECK NOT ZDTV3V_CUST_PERS-STCD2 IS INITIAL.
      SELECT SINGLE NAME1 INTO KNA1-NAME1
        FROM KNA1
       WHERE STCD2 = ZDTV3V_CUST_PERS-STCD2.
    WHEN 'ZDTV3V_VEND_MIN'.
      CHECK NOT ZDTV3V_VEND_MIN-MIN_STCD2  IS INITIAL.
      SELECT SINGLE NAME1 INTO LFA1-NAME1
        FROM LFA1
       WHERE STCD2 = ZDTV3V_VEND_MIN-MIN_STCD2.
    WHEN 'ZDTV3V_VEND_PERS'.
      CHECK NOT ZDTV3V_VEND_PERS-STCD2 IS INITIAL.
      SELECT SINGLE NAME1 INTO LFA1-NAME1
        FROM LFA1
       WHERE STCD2 = ZDTV3V_VEND_PERS-STCD2.
  ENDCASE.

FORM UPDATE_ZDTV3V_NTS_CLOSE_AA.
  DATA : CNT(2) TYPE N ,
         DAY    TYPE C ,
         DATE   TYPE DATUM.

  PERFORM VIM_FILL_WHERETAB.
*.read data from database.............................................*
  REFRESH TOTAL.
  CLEAR   TOTAL.
  SELECT * FROM ZDTV3T_NTS_CLOSE
    INTO ZDTV3V_NTS_CLOSE
   WHERE (VIM_WHERETAB) .
    <VIM_TOTAL_STRUC> = ZDTV3V_NTS_CLOSE.
    APPEND TOTAL.
  ENDSELECT.
  IF TOTAL IS INITIAL.
    READ TABLE DBA_SELLIST INDEX 1.
    DO 12 TIMES.
      ADD 1 TO CNT.
      ZDTV3V_NTS_CLOSE-MANDT = SY-MANDT.
      ZDTV3V_NTS_CLOSE-GJAHR = DBA_SELLIST-VALUE.
      ZDTV3V_NTS_CLOSE-MONAT = CNT.
      ZDTV3V_NTS_CLOSE-ERDAT = SY-DATUM.
      ZDTV3V_NTS_CLOSE-ERZET = SY-UZEIT.
      ZDTV3V_NTS_CLOSE-ERNAM = SY-UNAME.
      CONCATENATE DBA_SELLIST-VALUE CNT '01' INTO DATE.
      CALL FUNCTION 'LAST_DAY_OF_MONTHS'
        EXPORTING
          DAY_IN            = DATE
        IMPORTING
          LAST_DAY_OF_MONTH = DATE.
*
      DATE = DATE + 11.
*
      DO 7 TIMES.
*
        CALL FUNCTION 'DATE_COMPUTE_DAY'
          EXPORTING
            DATE = DATE
          IMPORTING
            DAY  = DAY.
*
        IF DAY EQ '6' OR DAY EQ '7'.
          DATE = DATE + 1.
        ELSE.
          ZDTV3V_NTS_CLOSE-DATUM = DATE.
          EXIT.
        ENDIF.
*
      ENDDO.
*
      <VIM_TOTAL_STRUC> = ZDTV3V_NTS_CLOSE.
      TOTAL+43(1) = 'U'.
      APPEND TOTAL.
    ENDDO.
  ENDIF.
*
  SORT TOTAL BY <VIM_XTOTAL_KEY>.
  <STATUS>-ALR_SORTED = 'R'.
*.check dynamic selectoptions (not in DDIC)...........................*

    PERFORM CHECK_DYNAMIC_SELECT_OPTIONS.

  ENDIF.

                 <aenam> TYPE ANY.


  CHECK  <status>-upd_flag =  'X'.
