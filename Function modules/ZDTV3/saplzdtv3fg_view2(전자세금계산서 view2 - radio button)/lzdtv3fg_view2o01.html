MODULE GET_FIELDTEXT OUTPUT.
  DATA: LV_LABEL  TYPE STRING.

  CLEAR G_FIELDTEXT.
  CALL FUNCTION 'DDIF_FIELDLABEL_GET'
    EXPORTING
      TABNAME        = ZDTV3V_PROG_FIED-TABNAME
      FIELDNAME      = ZDTV3V_PROG_FIED-FIELDNAME
    IMPORTING
      LABEL          = LV_LABEL
    EXCEPTIONS
      NOT_FOUND      = 1
      INTERNAL_ERROR = 2
      OTHERS         = 3.

  IF SY-SUBRC = 0.
    MOVE LV_LABEL TO G_FIELDTEXT.
  ENDIF.

ENDMODULE. " GET_FIELDTEXT OUTPUT
