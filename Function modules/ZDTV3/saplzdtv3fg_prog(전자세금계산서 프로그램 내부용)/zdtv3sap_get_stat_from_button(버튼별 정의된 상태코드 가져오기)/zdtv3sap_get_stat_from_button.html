*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUTTON) TYPE ZDTV3E_BUTTON
*" EXPORTING
*" VALUE(STATUS) TYPE ZDTV3E_STATUS
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*"----------------------------------------------------------------------
*
* Global data declarations

  SELECT SINGLE STATUS INTO STATUS
    FROM ZDTV3T_BUTTON
   WHERE BUTTON EQ BUTTON.
  IF SY-SUBRC NE 0 OR STATUS IS INITIAL.
    DS_RETURN RETURN 'E' '11003' '' '' '' '' ''.
  ELSE.
    DS_RETURN RETURN 'S' '00000' '' '' '' '' ''.
  ENDIF.
*
ENDFUNCTION.
