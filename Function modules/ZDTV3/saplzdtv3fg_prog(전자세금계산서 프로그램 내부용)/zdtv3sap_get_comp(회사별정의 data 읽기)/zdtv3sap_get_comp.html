FUNCTION ZDTV3SAP_GET_COMP .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(INV_SEQ) TYPE ZDTV3E_INV_SEQ
*" VALUE(INV_TYPE) TYPE CHAR2
*" EXPORTING
*" VALUE(COMPANY) LIKE ZDTV3T_COMPANY STRUCTURE ZDTV3T_COMPANY
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_BUKRS TYPE BUKRS.
*
  CASE INV_TYPE.
    WHEN 'AR'.
      SELECT SINGLE BUKRS INTO L_BUKRS
        FROM ZDTV3T_AR_HEAD
       WHERE INV_SEQ EQ INV_SEQ.
      IF NOT L_BUKRS IS INITIAL.
        SELECT SINGLE * INTO COMPANY
          FROM ZDTV3T_COMPANY
         WHERE BUKRS EQ L_BUKRS.
      ENDIF.
    WHEN 'AP'.
      SELECT SINGLE BUKRS INTO L_BUKRS
        FROM ZDTV3T_AP_HEAD
       WHERE INV_SEQ EQ INV_SEQ.
      IF NOT L_BUKRS IS INITIAL.
        SELECT SINGLE * INTO COMPANY
          FROM ZDTV3T_COMPANY
         WHERE BUKRS EQ L_BUKRS.
      ENDIF.
  ENDCASE.
*
ENDFUNCTION.
