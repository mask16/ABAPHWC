FUNCTION ZDTV3SAP_SEL_COMP_MIN.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) TYPE BUKRS
*" VALUE(BUPLA) TYPE BUPLA
*" VALUE(ISSUE_DATE) TYPE ZDTV3E_ISSUE_DATE
*" EXPORTING
*" VALUE(SU_MIN) TYPE ZDTV3E_SU_MIN
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*"----------------------------------------------------------------------

* 전자세금계산서회사정의확인.
* Global data declarations

  SELECT SINGLE *
    FROM ZDTV3T_COMPANY
   WHERE BUKRS EQ BUKRS.
  IF SY-SUBRC EQ 0 AND ZDTV3T_COMPANY-SU_MIN_FLAG EQ 'X'.
* 시행일자 존재확인.
    IF ZDTV3T_COMPANY-APP_SPMON IS INITIAL.
      DS_RETURN RETURN 'E' '12007' '' BUKRS '' '' ''.
      EXIT.
    ENDIF.
* 적용년월 CHECK.
    IF ISSUE_DATE+0(6) >= ZDTV3T_COMPANY-APP_SPMON.
* 우리회사 종사업장 관리
      SELECT SINGLE *
        FROM ZDTV3T_COMP_MIN
       WHERE BUKRS EQ BUKRS
         AND BUPLA EQ BUPLA.
      IF ZDTV3T_COMP_MIN-SU_MIN IS INITIAL.
        DS_RETURN RETURN 'E' '12001' '' BUKRS BUPLA '' ''.
      ELSE.
        SU_MIN = ZDTV3T_COMP_MIN-SU_MIN.
      ENDIF.
    ELSE.
      EXIT.
    ENDIF.
  ENDIF.

ENDFUNCTION.
