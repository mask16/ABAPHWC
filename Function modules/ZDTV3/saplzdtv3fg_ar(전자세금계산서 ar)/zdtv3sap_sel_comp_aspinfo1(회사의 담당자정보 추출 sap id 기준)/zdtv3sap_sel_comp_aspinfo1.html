FUNCTION ZDTV3SAP_SEL_COMP_ASPINFO1 .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) TYPE BUKRS
*" VALUE(BNAME) TYPE XUBNAME
*" VALUE(ASP_CODE) TYPE ZDTV3E_ASP_CODE
*" EXPORTING
*" VALUE(OUT_SU_PERS) TYPE ZDTV3_ASP_LOGIN
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*"----------------------------------------------------------------------
* Global data declarations

  CLEAR : OUT_SU_PERS.
** 입력값 확인.
* IF BNAME IS INITIAL.
* DS_RETURN RETURN 'E' '12008' '' '' '' '' ''.
* EXIT.
* ENDIF.
  SELECT SINGLE *
    FROM ZDTV3T_ASP_CODE
   WHERE ASP_CODE  EQ ASP_CODE
     AND ASP_LOCAL EQ 'X'.
  IF SY-SUBRC EQ 0.
    DS_RETURN RETURN 'S' '00000' '' '' ASP_CODE '' ''.
    EXIT.
  ENDIF.

  SELECT SINGLE SU_ASPID SU_ASPPW
    INTO (OUT_SU_PERS-ASPID , OUT_SU_PERS-ASPPW)
    FROM ZDTV3T_COMP_PASP
   WHERE BUKRS    EQ BUKRS
     AND BNAME    EQ BNAME
     AND ASP_CODE EQ ASP_CODE.
  IF SY-SUBRC NE 0.
    SELECT SINGLE SU_ASPID SU_ASPPW
      INTO (OUT_SU_PERS-ASPID , OUT_SU_PERS-ASPPW)
      FROM ZDTV3T_COMP_PASP
     WHERE BUKRS    EQ BUKRS
       AND BNAME    EQ G_ALL
       AND ASP_CODE EQ ASP_CODE.
    IF OUT_SU_PERS-ASPID IS INITIAL OR
       OUT_SU_PERS-ASPPW IS INITIAL.
      DS_RETURN RETURN 'E' '12004' '' BNAME ASP_CODE '' ''.
      EXIT.
    ENDIF.
  ELSE.
    IF OUT_SU_PERS-ASPID IS INITIAL OR
       OUT_SU_PERS-ASPPW IS INITIAL.
      DS_RETURN RETURN 'E' '12004' '' BNAME ASP_CODE '' ''.
      EXIT.
    ENDIF.
  ENDIF.

ENDFUNCTION.
