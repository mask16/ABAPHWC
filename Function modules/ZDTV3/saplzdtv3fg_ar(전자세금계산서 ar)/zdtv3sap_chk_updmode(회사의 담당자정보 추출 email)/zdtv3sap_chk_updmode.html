FUNCTION ZDTV3SAP_CHK_UPDMODE .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(INV_SEQ) TYPE ZDTV3E_INV_SEQ
*" VALUE(SEND) TYPE CHAR1 OPTIONAL
*" EXPORTING
*" REFERENCE(UPDMODE) TYPE CTU_UPDATE
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_MODE ,
         L_ASP_CODE TYPE ZDTV3E_ASP_CODE.

  SELECT SINGLE  MANDT INTO SY-MANDT
    FROM ZDTV3T_HOST
  WHERE SYSID = SY-SYSID.

  IF SY-SUBRC = 0.
    IF SEND EQ 'X'.
      UPDMODE = 'S'. "동기
    ELSE.
      UPDMODE = 'A'. "비동기
    ENDIF.

  ELSE.

    SELECT SINGLE ASP_CODE INTO L_ASP_CODE
      FROM ZDTV3T_AR_HEAD
     WHERE INV_SEQ EQ INV_SEQ.

    SELECT SINGLE ASP_LOCAL INTO L_MODE
      FROM ZDTV3T_ASP_CODE
     WHERE ASP_CODE EQ L_ASP_CODE .

    IF L_MODE EQ 'X'.
      UPDMODE = 'S'. "동기
    ELSE.
      UPDMODE = 'A'. "비동기
    ENDIF.

  ENDIF.

ENDFUNCTION.
