FUNCTION ZDTV3SAP_SEL_CUST_PERS1 .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) TYPE BUKRS
*" VALUE(STCD1) TYPE CHAR13
*" VALUE(KUNNR) TYPE KUNNR OPTIONAL
*" EXPORTING
*" VALUE(OUT_IP_PERS1) TYPE ZDTV3_IP_PERS1
*"----------------------------------------------------------------------
* Global data declarations

  CLEAR : GS_ZDTV3T_COMPANY.
* 회사코드정의중 고객의 담당자읽기방법...
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF GS_ZDTV3T_COMPANY
    FROM ZDTV3T_COMPANY
   WHERE BUKRS EQ BUKRS.
*
  CHECK SY-SUBRC EQ 0.
* 고객마스터의 담당자를 사용하는경우.
  IF GS_ZDTV3T_COMPANY-IP_PERS0_FLAG EQ 'X'.
    SELECT SINGLE *
      FROM KNB1
     WHERE KUNNR EQ KUNNR
       AND BUKRS EQ BUKRS .
    IF SY-SUBRC EQ 0.
      OUT_IP_PERS1-IP_DEPTNAME1 = KNB1-KVERM.
      OUT_IP_PERS1-IP_PERSNAME1 = KNB1-ZSABE.
      OUT_IP_PERS1-IP_TELNO1    = KNB1-TLFNS.
      OUT_IP_PERS1-IP_HPNO1     = KNB1-TLFXS.
      OUT_IP_PERS1-IP_EMAIL1    = KNB1-INTAD.
    ENDIF.
  ENDIF.
*
ENDFUNCTION.
