FUNCTION ZDTV3SAP_CHK_STATUS .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUTTON) TYPE ZDTV3E_BUTTON
*" VALUE(PROGRAM) TYPE ZDTV3E_PROGRAM
*" TABLES
*" INPUT STRUCTURE ZDTV3S_INFSEND OPTIONAL
*" RETURN STRUCTURE ZDTV3S_RETURN OPTIONAL
*"----------------------------------------------------------------------
*변수초기화.
* Global data declarations

  PERFORM INIT_DATA.
  G_BUTTON     = BUTTON.
  G_PROGRAM    = PROGRAM.
  GT_INFSEND[] = INPUT[].
*
* 처리자료존재확인.
  IF INPUT[] IS INITIAL.
    DT_RETURN RETURN 'E' '13001' '' '' '' '' ''.
    EXIT.
  ELSE.
    READ TABLE INPUT INDEX 1.
    G_BUKRS = INPUT-INV_SEQ+8(4).
  ENDIF.

* C.상태점검.
* C.1.BC 서버연결확인.
  PERFORM CHECK_CONNECT USING G_BUKRS.
  IF GS_RETURN-TYPE EQ 'E'.
    APPEND GS_RETURN TO RETURN.
    EXIT.
  ENDIF.
  LOOP AT GT_INFSEND.
    CLEAR : GS_IN_STATUS.
*
    CASE GT_INFSEND-STATUS+0(1).
      WHEN '1'. "매출상태점검.
        SELECT SINGLE *
          INTO CORRESPONDING FIELDS OF GT_INFSEND
          FROM ZDTV3T_AR_HEAD
         WHERE INV_SEQ EQ GT_INFSEND-INV_SEQ.
      WHEN '2'. "매입상태점검.
        SELECT SINGLE *
          INTO CORRESPONDING FIELDS OF GT_INFSEND
          FROM ZDTV3T_AP_HEAD
         WHERE INV_SEQ EQ GT_INFSEND-INV_SEQ.
    ENDCASE.
*

    MOVE-CORRESPONDING GT_INFSEND TO GS_IN_STATUS .
* B.2. BC전송.
    PERFORM SEND_BC_CHKSTATUS.
* 받은결과가 없는경우.
    IF GS_INFRETURN IS INITIAL.
      DT_RETURN RETURN 'E' '10027'
            GT_INFSEND-INV_SEQ '' '' '' ''.
    ELSE.
* B.3 RETURN 받은 결과기록
      PERFORM RESULT_UPDATE_CHKSTATUS USING 'CHK'
                                            GT_INFSEND-INV_SEQ
                                            GT_INFSEND-STATUS
                                            GS_INFRETURN-ERROR_CODE
                                           GS_INFRETURN-ERROR_CODE_TEXT.
    ENDIF.
*
    RETURN[] = GT_RETURN[].
  ENDLOOP.
*
ENDFUNCTION.
