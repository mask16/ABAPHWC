Description: ASP CODE 점검 및 회사 대표 ASP 코드 RETURN
FUNCTION ZDTV3SAP_CHK_ASP_CODE .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(ASP_CODE) TYPE ZDTV3E_ASP_CODE
*" VALUE(ASP_AP) TYPE ZDTV3E_ASP_AP OPTIONAL
*" EXPORTING
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*"----------------------------------------------------------------------

* Global data declarations

  SELECT SINGLE *
    FROM ZDTV3T_ASP_CODE
   WHERE ASP_CODE EQ ASP_CODE.
  IF SY-SUBRC NE 0.
    DS_RETURN RETURN 'E' '10025' '' ASP_CODE '' '' ''.
    EXIT.
  ENDIF.
*
  IF ASP_AP <> 'X'.
* 매출기능사용유무.
    IF ZDTV3T_ASP_CODE-ASP_AR NE 'Y'.
      DS_RETURN RETURN 'E' '10026' '' ASP_CODE '' '' ''.
      EXIT.
    ENDIF.

  ELSE.
    IF ZDTV3T_ASP_CODE-ASP_AP NE 'Y'.
      DS_RETURN RETURN 'E' '15010' '' ASP_CODE '' '' ''.
      EXIT.
    ENDIF.
  ENDIF.

ENDFUNCTION.
