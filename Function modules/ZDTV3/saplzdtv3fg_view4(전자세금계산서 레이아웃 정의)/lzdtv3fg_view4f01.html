FORM CALL_TRANSACTION_GEN_PROGRAM.
  DATA: LV_FIELD(60).
  FIELD-SYMBOLS: <FS> TYPE ANY.
  CONCATENATE X_HEADER-VIEWNAME '-ZPROGRAM'
              INTO LV_FIELD.

      CALL TRANSACTION 'ZDTV3_P10'.

      CALL TRANSACTION 'ZDTV3_P20'.

      CALL TRANSACTION 'ZDTV3_P11'.
    WHEN 'ZDTV3_AP_LIST'.
      CALL TRANSACTION 'ZDTV3_P21'.
  ENDCASE.

ENDFORM. "CALL_TRANSACTION_GEN_PROGRAM

FORM GET_TABLETEXT USING P_TABNAME.

  CLEAR G_TABLETEXT.
  SELECT SINGLE DDTEXT   INTO G_TABLETEXT
                         FROM DD02T
                        WHERE TABNAME    = P_TABNAME
                          AND DDLANGUAGE = SY-LANGU
                          AND AS4LOCAL   = 'A'.

ENDFORM. " GET_TABLETEXT

FORM GET_FIELDTEXT USING P_TABNAME TYPE DDOBJNAME
                             P_FIELDNAME TYPE DFIES-FIELDNAME.

  DATA: LV_LABEL  TYPE STRING.

  CLEAR G_FIELDTEXT.
  CALL FUNCTION 'DDIF_FIELDLABEL_GET'
    EXPORTING
      TABNAME        = P_TABNAME
      FIELDNAME      = P_FIELDNAME
    IMPORTING
      LABEL          = LV_LABEL
    EXCEPTIONS
      NOT_FOUND      = 1
      INTERNAL_ERROR = 2
      OTHERS         = 3.

  IF SY-SUBRC = 0.
    MOVE LV_LABEL TO G_FIELDTEXT.
  ENDIF.

ENDFORM. " GET_FIELDTEXT
