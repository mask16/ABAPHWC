FUNCTION ZDTV3SAP_USER_INPUT .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) TYPE BUKRS OPTIONAL
*" VALUE(BUPLA) TYPE BUPLA OPTIONAL
*" VALUE(MOD) TYPE CHAR1
*" VALUE(NO_POPUP) TYPE CHAR1 OPTIONAL
*" VALUE(PAP_FLAG) TYPE CHAR1 OPTIONAL
*" VALUE(INV_SEQ) TYPE ZDTV3E_INV_SEQ OPTIONAL
*" VALUE(INV_TYPE) TYPE ZDTV3E_INV_TYPE OPTIONAL
*" VALUE(ISSUE_ID) TYPE ZDTV3E_SSUE_ID OPTIONAL
*" EXPORTING
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*" TABLES
*" IT_HEAD STRUCTURE ZDTV3S_AP_HD OPTIONAL
*" IT_ITEM STRUCTURE ZDTV3S_AP_IM OPTIONAL
*" IT_PAYM STRUCTURE ZDTV3S_AP_PM OPTIONAL
*" IT_EXT STRUCTURE ZDTV3S_AP_EX OPTIONAL
*" IT_EXT_D STRUCTURE ZDTV3S_AP_EX_D OPTIONAL
*" IT_AR_HEAD STRUCTURE ZDTV3S_AR_HD OPTIONAL
*" IT_AR_ITEM STRUCTURE ZDTV3S_AR_IM OPTIONAL
*" IT_AR_PAYM STRUCTURE ZDTV3S_AR_PM OPTIONAL
*" IT_AR_EXT STRUCTURE ZDTV3S_AR_EX OPTIONAL
*" IT_AR_EXT_D STRUCTURE ZDTV3S_AR_EX_D OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  CLEAR :
* 매입
          GS_AP_HEAD  , GS_AP_PAYM   ,
          GS_AP_ITEM  , GT_AP_ITEM[] ,
          GS_AP_EXT   , GT_AP_EXT[]  ,
          GS_AP_EXT_D , GT_AP_EXT_D[].

  CASE MOD.
    WHEN 'I' OR'C'.
      G_INPUT = '1'.
    WHEN 'D'.
      G_INPUT = '0'.
  ENDCASE.
* 전역변수로 값이동.
  G_NO_POPUP = NO_POPUP.
  G_ISSUE_ID = ISSUE_ID.
*
  IF MOD EQ 'I'.
* 회사코드확인.
    SELECT SINGLE *
      FROM T001
     WHERE BUKRS EQ BUKRS.
    IF SY-SUBRC NE 0.
      DS_RETURN RETURN 'E' '19999' '' '회사코드' BUKRS
                '가 존재하지 않습니다' ''.
      EXIT.
    ENDIF.
* 사업장확인.
    SELECT SINGLE *
      FROM J_1BBRANCH
     WHERE BUKRS  EQ BUKRS
       AND BRANCH EQ BUPLA.
    IF SY-SUBRC NE 0.
      DS_RETURN RETURN 'E' '19999' '' '사업장' BUPLA
                '가 존재하지 않습니다' ''.
      EXIT.
    ENDIF.
  ENDIF.

  CASE PAP_FLAG.
    WHEN ''. "종이
      R3 = 'X'.
      R4 = ' '.
    WHEN 'X'."지로
      R3 = ' '.
      R4 = 'X'.
  ENDCASE.

* 기본정보.
  CASE MOD.
    WHEN 'I'.
      PERFORM MAKE_INT_DATA USING J_1BBRANCH.
      READ TABLE IT_HEAD INDEX 1.
      IF IT_HEAD-GRANDTOTAL IS INITIAL.
* MOVE-CORRESPONDING IT_HEAD TO GS_AP_HEAD.
        APPEND INITIAL LINE TO GT_AP_EXT_D.
        APPEND INITIAL LINE TO GT_AP_EXT.
        APPEND INITIAL LINE TO GT_AP_ITEM.
      ELSE.
        READ TABLE IT_HEAD INDEX 1.
        MOVE-CORRESPONDING IT_HEAD TO GS_AP_HEAD.
* 회사코드확인.
        CLEAR : GT_AP_ITEM[] ,GT_AP_EXT[] , GT_AP_EXT_D[].
        READ TABLE IT_HEAD INDEX 1.
        SELECT SINGLE *
          FROM T001
         WHERE BUKRS EQ IT_HEAD-BUKRS.
        IF SY-SUBRC NE 0.
          DS_RETURN RETURN 'E' '19999' '' '회사코드' BUKRS
                    '가 존재하지 않습니다' ''.
          EXIT.
        ENDIF.
        LOOP AT IT_ITEM.
          MOVE-CORRESPONDING GS_AP_HEAD TO GT_AP_ITEM.
          MOVE-CORRESPONDING IT_ITEM TO GT_AP_ITEM.
          APPEND GT_AP_ITEM.
        ENDLOOP.
        IF SY-SUBRC EQ 4.
          APPEND INITIAL LINE TO GT_AP_ITEM.
        ENDIF.
        READ TABLE IT_PAYM INDEX 1.
        MOVE-CORRESPONDING IT_PAYM TO GS_AP_PAYM.
        LOOP AT IT_EXT.
          MOVE-CORRESPONDING GS_AP_HEAD TO GT_AP_EXT.
          MOVE-CORRESPONDING IT_EXT TO GT_AP_EXT.
          APPEND GT_AP_EXT.
        ENDLOOP.
        IF SY-SUBRC EQ 4.
          APPEND INITIAL LINE TO GT_AP_EXT.
        ENDIF.
        LOOP AT IT_EXT_D.
          MOVE-CORRESPONDING GS_AP_HEAD TO GT_AP_EXT_D.
          MOVE-CORRESPONDING IT_EXT_D TO GT_AP_EXT_D.
          APPEND GT_AP_EXT_D.
        ENDLOOP.
        IF SY-SUBRC EQ 4.
          APPEND GT_AP_EXT_D.
        ENDIF.
      ENDIF.

    WHEN 'D'.
      PERFORM SELECT_DATA USING INV_SEQ.
      IF GS_AP_HEAD IS INITIAL.
        DS_RETURN RETURN 'E' '19999' '' '매입' INV_SEQ
                  '가 존재하지 않습니다' ''.
        EXIT.
      ENDIF.
* 회사코드확인.
      SELECT SINGLE *
        FROM T001
       WHERE BUKRS EQ GS_AP_HEAD-BUKRS.

    WHEN 'C'.
      PERFORM SELECT_DATA USING INV_SEQ.
      IF GS_AP_HEAD IS INITIAL.
        DS_RETURN RETURN 'E' '19999' '' '매입' INV_SEQ
                  '가 존재하지 않습니다' ''.
        EXIT.
      ENDIF.

* 회사코드확인.
      SELECT SINGLE *
       FROM T001
      WHERE BUKRS EQ GS_AP_HEAD-BUKRS.
      IF SY-SUBRC NE 0.
        DS_RETURN RETURN 'E' '19999' '' '회사코드' BUKRS
                  '가 존재하지 않습니다' ''.
        EXIT.
      ENDIF.
* MOVE-CORRESPONDING IT_HEAD TO GS_AP_HEAD.
      LOOP AT IT_ITEM.
        MOVE-CORRESPONDING IT_ITEM TO GT_AP_ITEM.
        APPEND GT_AP_ITEM.
      ENDLOOP.
      READ TABLE IT_PAYM INDEX 1.
      MOVE-CORRESPONDING IT_PAYM TO GS_AP_PAYM.
      LOOP AT IT_EXT.
        MOVE-CORRESPONDING IT_EXT TO GT_AP_EXT.
        APPEND GT_AP_EXT.
      ENDLOOP.
      LOOP AT IT_EXT_D.
        MOVE-CORRESPONDING IT_EXT_D TO GT_AP_EXT_D.
        APPEND GT_AP_EXT_D.
      ENDLOOP.
  ENDCASE.
*
  TAB-ACTIVETAB = 'ITEM'.

  CLEAR : G_CHANGE.
  G_REPID  = SY-CPROG.
  SY-CPROG = 'SAPLZDTV3FG_USR01'.

  CLEAR  G_MOD .
  G_MOD = MOD.

  IF G_NO_POPUP EQ SPACE.

    CALL SCREEN 9100 STARTING AT 02  002
                     ENDING   AT 110 029.
  ELSE.
    PERFORM CHECK_DATA_9100.
    CHECK G_MSG_GB NE 'E'.
    PERFORM SAVE_DATA_9100.

  ENDIF.


  CASE G_MSG_GB.
    WHEN '1'.
      RETURN-TYPE     = 'S'.
      RETURN-INV_SEQ  = GS_AP_HEAD-INV_SEQ.
      RETURN-MSGV1    = GS_AP_HEAD-STATUS.
      RETURN-MESSAGE = '변경하였습니다'.
    WHEN '2'.
      RETURN-TYPE     = 'S'.
      RETURN-INV_SEQ  = GS_AP_HEAD-INV_SEQ.
      RETURN-MSGV1    = GS_AP_HEAD-STATUS.
      RETURN-MESSAGE = '변경된 내용이 없습니다.'.
    WHEN '3'.
      RETURN-TYPE     = 'E'.
      RETURN-INV_SEQ  = GS_AP_HEAD-INV_SEQ.
      RETURN-MSGV1    = GS_AP_HEAD-STATUS.
      RETURN-MESSAGE = '작업을 취소하였습니다.'.
  ENDCASE.



*
  SY-CPROG = G_REPID.

ENDFUNCTION.
