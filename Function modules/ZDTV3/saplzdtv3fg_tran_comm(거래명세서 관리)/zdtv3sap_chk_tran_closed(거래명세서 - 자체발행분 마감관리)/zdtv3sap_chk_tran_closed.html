FUNCTION ZDTV3SAP_CHK_TRAN_CLOSED.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) LIKE ZDTV3T_TRAN_HD-BUKRS OPTIONAL
*" VALUE(ISSUE_DATE) LIKE ZDTV3T_TRAN_HD-ISSUE_DATE OPTIONAL
*" VALUE(BUPLA) LIKE ZDTV3T_TRAN_HD-BUPLA OPTIONAL
*" VALUE(TRAN_SEQ) LIKE ZDTV3T_TRAN_HD-TRAN_SEQ OPTIONAL
*" VALUE(AMEND_CODE) LIKE ZDTV3T_TRAN_HD-AMEND_CODE OPTIONAL
*" EXPORTING
*" VALUE(CLOSED) TYPE ZDTV3E_CLOSED
*"----------------------------------------------------------------------

* Global data declarations

  CLEAR : GT_CLOSED, GT_CLOSED[].
* 마감확인.
  SELECT *
    INTO CORRESPONDING FIELDS OF GT_CLOSED
    FROM ZDTV3T_CLOSED.
*
    APPEND GT_CLOSED.
* 마감사업장별.
    SELECT *
      INTO CORRESPONDING FIELDS OF GT_CLOSED
      FROM ZDTV3T_CLOSED_SU
     WHERE BUKRS EQ GT_CLOSED-BUKRS.
      APPEND GT_CLOSED.
    ENDSELECT.
  ENDSELECT.
* 마감 CHECK
  SELECT SINGLE AMEND_CODE
    INTO AMEND_CODE
    FROM ZDTV3T_TRAN_HD
    WHERE TRAN_SEQ EQ TRAN_SEQ.

  PERFORM CLOSED_CHECK TABLES GT_CLOSED
                       USING  BUKRS
                              ISSUE_DATE
                              BUPLA
                              TRAN_SEQ
                              AMEND_CODE
                     CHANGING CLOSED.

ENDFUNCTION.
