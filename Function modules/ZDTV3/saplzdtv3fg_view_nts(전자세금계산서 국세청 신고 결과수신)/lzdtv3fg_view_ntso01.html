MODULE GET_DDTEXT OUTPUT.
  DATA: L_TABNAME(60),
        L_FIELDNAME(60).
  FIELD-SYMBOLS: <FS_TABNAME>   TYPE ANY,
                 <FS_FIELDNAME> TYPE ANY.

  CONCATENATE X_HEADER-VIEWNAME '-TABNAME'   INTO L_TABNAME.
  ASSIGN (L_TABNAME)   TO <FS_TABNAME>.
  CONCATENATE X_HEADER-VIEWNAME '-FIELDNAME' INTO L_FIELDNAME.
  ASSIGN (L_FIELDNAME) TO <FS_FIELDNAME>.

  PERFORM GET_DDTEXT  USING <FS_TABNAME>
                            <FS_FIELDNAME>.

ENDMODULE. "GET_DDTEXT OUTPUT

FORM GET_TABLETEXT USING P_TABNAME.

  CLEAR G_TABLETEXT.
  SELECT SINGLE DDTEXT   INTO G_TABLETEXT
                         FROM DD02T
                        WHERE TABNAME    = P_TABNAME
                          AND DDLANGUAGE = SY-LANGU
                          AND AS4LOCAL   = 'A'.

ENDFORM. " GET_TABLETEXT

FORM GET_FIELDTEXT USING P_TABNAME TYPE DDOBJNAME
                             P_FIELDNAME TYPE DFIES-FIELDNAME.

  DATA: LV_LABEL  TYPE STRING.

  CLEAR G_FIELDTEXT.
  CALL FUNCTION 'DDIF_FIELDLABEL_GET'
    EXPORTING
      TABNAME        = P_TABNAME
      FIELDNAME      = P_FIELDNAME
    IMPORTING
      LABEL          = LV_LABEL
    EXCEPTIONS
      NOT_FOUND      = 1
      INTERNAL_ERROR = 2
      OTHERS         = 3.

  IF SY-SUBRC = 0.
    MOVE LV_LABEL TO G_FIELDTEXT.
  ENDIF.

ENDFORM. " GET_FIELDTEXT
