FUNCTION ZDTV3SAP_DEL_INVOICE.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_INVNUM) LIKE BAPI_INCINV_FLD-INV_DOC_NO OPTIONAL
*" VALUE(I_YEAR) LIKE BAPI_INCINV_FLD-FISC_YEAR OPTIONAL
*" TABLES
*" RETURN STRUCTURE ZDTV3S_RETURN OPTIONAL
*"----------------------------------------------------------------------
*
* Global data declarations

  CALL FUNCTION 'BAPI_INCOMINGINVOICE_DELETE'
    EXPORTING
      INVOICEDOCNUMBER = I_INVNUM
      FISCALYEAR       = I_YEAR
    TABLES
      RETURN           = GT_RETURN.
*
  READ TABLE GT_RETURN INTO GS_RETURN WITH KEY TYPE = 'E'.
  IF SY-SUBRC EQ 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT' .
  ENDIF.
*
  LOOP AT GT_RETURN INTO GS_RETURN.
    PERFORM GET_MESSAGE_TEXT USING    GS_RETURN-ID
                                      GS_RETURN-NUMBER
                                      GS_RETURN-MESSAGE_V1
                                      GS_RETURN-MESSAGE_V2
                                      GS_RETURN-MESSAGE_V3
                                      GS_RETURN-MESSAGE_V4
                             CHANGING GS_RETURN-MESSAGE.

    RETURN-TYPE     = GS_RETURN-TYPE.
    RETURN-MESSAGE  = GS_RETURN-MESSAGE.
    APPEND RETURN. CLEAR RETURN.
  ENDLOOP.
*
ENDFUNCTION.
