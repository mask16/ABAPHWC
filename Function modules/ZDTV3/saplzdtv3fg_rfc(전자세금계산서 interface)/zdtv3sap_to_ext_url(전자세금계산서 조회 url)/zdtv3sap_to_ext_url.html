FUNCTION ZDTV3SAP_TO_EXT_URL.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(BUKRS) TYPE BUKRS
*" VALUE(INV_SEQ) TYPE ZDTV3E_INV_SEQ
*" VALUE(ASP_CODE) TYPE ZDTV3E_ASP_CODE
*" VALUE(STATUS) TYPE ZDTV3E_STATUS
*" VALUE(ISSUEID) TYPE ZDTV3E_SSUE_ID OPTIONAL
*" EXPORTING
*" VALUE(URL) TYPE TEXT255
*" VALUE(RETURN) LIKE ZDTV3S_INFRETURN STRUCTURE ZDTV3S_INFRETURN
*" EXCEPTIONS
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  SELECT SINGLE *
    INTO GS_ZDTV3T_COMPANY
    FROM ZDTV3T_COMPANY
   WHERE BUKRS EQ BUKRS.
*
  CALL FUNCTION 'ZDTV3INF_SAPTOBC_GETXML'
    DESTINATION GS_ZDTV3T_COMPANY-RFCDEST
    EXPORTING
      INV_SEQ   = INV_SEQ
      ASP_CODE  = ASP_CODE
      STATUS    = STATUS
      ISSUEID   = ISSUEID
    IMPORTING
      URL       = URL
      RETURN    = RETURN
    EXCEPTIONS
     COMMUNICATION_FAILURE = 1
     SYSTEM_FAILURE        = 2 .

  include ZDTV3I_UA10 IF FOUND.


ENDFUNCTION.
