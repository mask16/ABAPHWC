FUNCTION zdtv3sap_shlp_exit_asp_code.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" VALUE(SHLP) TYPE SHLP_DESCR
*" VALUE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : l_record  TYPE seahlpres.
  DATA : BEGIN OF ls_record,
   mandt         LIKE sy-mandt,
   asp_code      TYPE zdtv3e_asp_code,
   asp_code_text TYPE zdtv3e_asp_code_text,
  END OF ls_record.

  RANGES : r_asp_code FOR zdtv3t_asp_code-asp_code.
  IF callcontrol-step = 'DISP'.
    IMPORT r_asp_code FROM MEMORY ID 'ZDTV3_R_ASP'.
    FREE MEMORY ID 'ZDTV3_R_ASP'.
    IF NOT r_asp_code[] IS INITIAL.
      LOOP AT record_tab INTO ls_record.
        IF ls_record-asp_code NOT IN r_asp_code.
          DELETE record_tab.
        ENDIF.
      ENDLOOP.
    ENDIF.
    l_record-string+3(11)  = '00000000'.
    l_record-string+12(20) = '전체'.
    APPEND l_record TO record_tab.
    SORT record_tab BY string.
    EXIT.
  ENDIF.
ENDFUNCTION.
