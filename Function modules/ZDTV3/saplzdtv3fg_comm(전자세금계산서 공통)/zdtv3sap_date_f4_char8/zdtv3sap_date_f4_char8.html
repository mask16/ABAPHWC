FUNCTION ZDTV3SAP_DATE_F4_CHAR8.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCR_TAB_T
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR_T
*" REFERENCE(CALLCONTROL) TYPE DDSHF4CTRL
*"----------------------------------------------------------------------

* Local data.
* Global data declarations

  DATA LS_INTERFACE LIKE LINE OF SHLP_TAB-INTERFACE.
  DATA LV_DATE      TYPE WORKFLDS-GKDAY.
  DATA LV_DATE_CHAR TYPE CHAR10.

* EXIT immediately, if you do not want to handle this step
  IF CALLCONTROL-STEP <> 'SELECT'.
    EXIT.
  ENDIF.

*"----------------------------------------------------------------------
* STEP DISP (Display values)
*"----------------------------------------------------------------------
* This step is called, before the selected data is displayed.
* You can e.g. modify or reduce the data in RECORD_TAB
* according to the users authority.
* If you want to get the standard display dialog afterwards, you
* should not change CALLCONTROL-STEP.
* If you want to overtake the dialog on you own, you must return
* the following values in CALLCONTROL-STEP:
* - "RETURN" if one line was selected. The selected line must be
* the only record left in RECORD_TAB. The corresponding fields of
* this line are entered into the screen.
* - "EXIT" if the values request should be aborted
* - "PRESEL" if you want to return to the selection dialog

* Standard function modules F4UT_PARAMETER_VALUE_GET and
* F4UT_PARAMETER_RESULTS_PUT may be very helpfull in this step.
  READ TABLE SHLP_TAB INDEX 1.
  READ TABLE SHLP_TAB-INTERFACE INTO LS_INTERFACE
                                WITH KEY SHLPFIELD = 'DATE_CHAR'.
  LV_DATE_CHAR = LS_INTERFACE-VALUE.

  CALL FUNCTION 'CONVERT_DATE_TO_INTERNAL'
    EXPORTING
      DATE_EXTERNAL = LV_DATE_CHAR
    IMPORTING
      DATE_INTERNAL = LV_DATE
    EXCEPTIONS
      OTHERS        = 1.
  IF SY-SUBRC <> 0.
    LV_DATE = SY-DATLO.
  ENDIF.

  CALL FUNCTION 'F4_DATE'
       EXPORTING
         DATE_FOR_FIRST_MONTH               = LV_DATE
* DISPLAY = ' '
* FACTORY_CALENDAR_ID = ' '
* GREGORIAN_CALENDAR_FLAG = ' '
* HOLIDAY_CALENDAR_ID = ' '
* PROGNAME_FOR_FIRST_MONTH = ' '
       IMPORTING
         SELECT_DATE                        = LV_DATE
* SELECT_WEEK =
* SELECT_WEEK_BEGIN =
* SELECT_WEEK_END =
       EXCEPTIONS
         OTHERS                             = 1.

  IF SY-SUBRC IS INITIAL AND NOT LV_DATE IS INITIAL.

    CALLCONTROL-STEP = 'RETURN'.
* WRITE LV_DATE TO LV_DATE_CHAR.
     MOVE LV_DATE TO LV_DATE_CHAR.
    CALL FUNCTION 'F4UT_PARAMETER_RESULTS_PUT'
      EXPORTING
        PARAMETER               = 'DATE_CHAR'
* OFF_SOURCE = 0
* LEN_SOURCE = 0
        VALUE                   = LV_DATE_CHAR
* FIELDNAME = ' '
      TABLES
        SHLP_TAB                = SHLP_TAB
        RECORD_TAB              = RECORD_TAB
* SOURCE_TAB =
      CHANGING
        SHLP                    = SHLP
        CALLCONTROL             = CALLCONTROL
      EXCEPTIONS OTHERS = 1.
    IF SY-SUBRC = 0.
      APPEND RECORD_TAB.
    ENDIF.

  ELSE.
    CALLCONTROL-STEP = 'EXIT'.
    EXIT.
  ENDIF.

ENDFUNCTION.
