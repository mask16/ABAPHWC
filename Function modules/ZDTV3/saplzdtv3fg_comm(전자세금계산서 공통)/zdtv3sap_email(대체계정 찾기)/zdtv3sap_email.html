FUNCTION ZDTV3SAP_EMAIL.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" VALUE(SHLP) TYPE SHLP_DESCR
*" VALUE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

  RANGES : R_BUKRS    FOR ZDTV3T_AR_HEAD-BUKRS,
           R_EMAIL    FOR ZDTV3T_AR_HEAD-IP_EMAIL1,
           R_PERSNAME FOR ZDTV3T_AR_HEAD-IP_PERSNAME1.
*
  DATA: BEGIN OF T_EDIDC,
      BUKRS    LIKE ZDTV3T_AR_HEAD-BUKRS,
      EMAIL    LIKE ZDTV3T_AR_HEAD-IP_EMAIL1,
      PERSNAME LIKE ZDTV3T_AR_HEAD-IP_PERSNAME1.
  DATA: END OF T_EDIDC.

  DATA: LT_RECORD LIKE T_EDIDC OCCURS 20,
        LS_RECORD  LIKE LINE OF LT_RECORD,
        LS_SELOPT  LIKE DDSHSELOPT.

  CHECK CALLCONTROL-STEP EQ 'SELECT'.

* clear create button on search help
  CLEAR CALLCONTROL-CUSTTAB.
  LOOP AT SHLP-SELOPT INTO LS_SELOPT.
    CASE LS_SELOPT-SHLPFIELD.
      WHEN 'BUKRS'.
        MOVE-CORRESPONDING LS_SELOPT TO R_BUKRS .
        APPEND R_BUKRS .
      WHEN 'EMAIL'.
        MOVE-CORRESPONDING LS_SELOPT TO R_EMAIL .
        APPEND R_EMAIL .
      WHEN 'PERSNAME'.
        MOVE-CORRESPONDING LS_SELOPT TO R_PERSNAME .
        APPEND R_PERSNAME .
    ENDCASE.
  ENDLOOP.
*
  REFRESH LT_RECORD.
*
  include ZDTV3I_U010H IF FOUND.
*
* move to the table with help-values
  CALL FUNCTION 'F4UT_RESULTS_MAP'
    TABLES
      SHLP_TAB          = SHLP_TAB
      RECORD_TAB        = RECORD_TAB
      SOURCE_TAB        = LT_RECORD
    CHANGING
      SHLP              = SHLP
      CALLCONTROL       = CALLCONTROL
    EXCEPTIONS
      ILLEGAL_STRUCTURE = 1
      OTHERS            = 2.
  IF SY-SUBRC <> 0.
    MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
  CALLCONTROL-STEP = 'DISP'.

ENDFUNCTION.
