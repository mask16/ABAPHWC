FUNCTION ZDTV3SAP_CRD_USER_INV .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(RETURN) TYPE ZDTV3S_RETURN
*" TABLES
*" INPUT OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  ASSIGN INPUT[] TO <GT_TABLE>.

  SELECT *
    INTO TABLE GT_0002
    FROM ZDTV3T_PROG_0002
    ORDER BY TABNAME.

  LOOP AT <GT_TABLE> INTO <GS_TABLE>.
    LOOP AT GT_0002.
      ASSIGN COMPONENT GT_0002-FIELD OF STRUCTURE
             <GS_TABLE> TO <GS_FIELD>.

      CASE GT_0002-TABNAME+10(4).
        WHEN 'HEAD'.
          PERFORM MOVE_DATA USING 'GT_HEAD'.
        WHEN 'ITEM'.
          PERFORM MOVE_DATA USING 'GT_ITEM'.
        WHEN 'PAYM'.
          PERFORM MOVE_DATA USING 'GT_PAYM'.
        WHEN 'EXT'.
          PERFORM MOVE_DATA USING 'GT_EXT'.
        WHEN 'EXT_D'.
          PERFORM MOVE_DATA USING 'GT_EXT_D'.
        WHEN OTHERS.
          MESSAGE E000(ZDTV3) WITH '정의되지 않은 TABLE 입니다.'.
      ENDCASE.

      ON CHANGE OF GT_0002-TABNAME.
        COLLECT <GS_VALUE> INTO <GT_VALUE>.
      ENDON.
*
    ENDLOOP.

  ENDLOOP.

  call function 'ZDTV3SAP_CRD_AR_INV'
    EXPORTING
      SEND       = 'X'
    TABLES
      INV_HD     = GT_HEAD
      INV_IM     = GT_ITEM
      INV_PM     = GT_PAYM
      INV_EX     = GT_EXT
      INV_EX_D   = GT_EXT_D
      RETURN     = GT_RETURN
    EXCEPTIONS
      FAIL_ISSUE = 1
      OTHERS     = 2.








ENDFUNCTION.
