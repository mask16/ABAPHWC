FUNCTION ZDTV3INF_SAPTOBC_VIEW.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(IN_INV_SEQ) TYPE ZDTV3E_INV_SEQ OPTIONAL
*" VALUE(IN_ISSUE_ID) TYPE ZDTV3E_SSUE_ID
*" VALUE(IN_STATUS) TYPE ZDTV3E_STATUS OPTIONAL
*" TABLES
*" RETURN STRUCTURE ZDTV3S_INFRETURN OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : G_URL TYPE TEXT255.

*BC Interface 구조정의
DATA :
* OUT_HEAD LIKE ZDTV3S_HEAD_INF OCCURS 0 WITH HEADER LINE ,
* OUT_ITEM LIKE ZDTV3S_ITEM_INF OCCURS 0 WITH HEADER LINE ,
* OUT_PAYM LIKE ZDTV3S_PAYM_INF OCCURS 0 WITH HEADER LINE ,
* OUT_EXT LIKE ZDTV3S_EXT_INF OCCURS 0 WITH HEADER LINE .

 OUT_HEAD  LIKE ZDTV3T_AP_HEAD   OCCURS 0 WITH HEADER LINE ,
 OUT_ITEM  LIKE ZDTV3T_AP_ITEM   OCCURS 0 WITH HEADER LINE ,
 OUT_PAYM  LIKE ZDTV3T_AP_PAYM   OCCURS 0 WITH HEADER LINE ,
 OUT_EXT   LIKE ZDTV3T_AP_EXT    OCCURS 0 WITH HEADER LINE ,
 OUT_EXT_D LIKE ZDTV3T_AP_EXT_D  OCCURS 0 WITH HEADER LINE .

  IF IN_INV_SEQ IS INITIAL.
    DT_RETURN IN_INV_SEQ IN_STATUS '' '' IN_ISSUE_ID '19991'
    '해당 값은 필수입니다' '' ''.
    RETURN[] = GT_RETURN[].
    EXIT.
  ENDIF.

  IF IN_STATUS IS INITIAL.
    DT_RETURN IN_INV_SEQ IN_STATUS '' '' IN_ISSUE_ID '19992'
    '해당 값은 필수입니다' '' ''.
    RETURN[] = GT_RETURN[].
    EXIT.
  ENDIF.


* 1.매입/매출구분.
  CASE IN_STATUS+0(1).
    WHEN '1'. "매출


* HEAD DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_HEAD
        FROM ZDTV3T_AR_HEAD
       WHERE INV_SEQ EQ IN_INV_SEQ.

      IF OUT_HEAD[] IS INITIAL.
        DT_RETURN IN_INV_SEQ IN_STATUS '' '' IN_ISSUE_ID '19993'
        '데이터가 존재하지 않습니다' '' ''.
        RETURN[] = GT_RETURN[].

        EXIT.
      ENDIF.

* ITEM DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_ITEM
        FROM ZDTV3T_AR_ITEM FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* PAYM DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_PAYM
        FROM ZDTV3T_AR_PAYM FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_EXT
        FROM ZDTV3T_AR_EXT FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT_D DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_EXT_D
        FROM ZDTV3T_AR_EXT_D FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

      DT_RETURN IN_INV_SEQ  '' '' '' '' '00000' '데이터가 검색되었습니다' '' ''.
      RETURN[] = GT_RETURN[].


    WHEN '2'. "매입


* HEAD DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_HEAD
        FROM ZDTV3T_AP_HEAD
       WHERE INV_SEQ EQ IN_INV_SEQ .
      IF OUT_HEAD[] IS INITIAL.
        DT_RETURN IN_INV_SEQ IN_STATUS '' '' IN_ISSUE_ID '19994'
        '데이터가 존재하지 않습니다' '' ''.
        EXIT.
      ENDIF.
* ITEM DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_ITEM
        FROM ZDTV3T_AP_ITEM FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* PAYM DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_PAYM
        FROM ZDTV3T_AP_PAYM FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_EXT
        FROM ZDTV3T_AP_EXT FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT_D DATA
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE OUT_EXT_D
        FROM ZDTV3T_AP_EXT_D FOR ALL ENTRIES IN OUT_HEAD
       WHERE BUKRS      EQ OUT_HEAD-BUKRS
         AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
         AND BUPLA      EQ OUT_HEAD-BUPLA
         AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

      DT_RETURN IN_INV_SEQ  '' '' '' '' '00000' '데이터가 검색되었습니다' '' ''.

  ENDCASE.


* HEAD DATA (금액x100)
  READ TABLE OUT_HEAD INDEX 1.
  IF NOT OUT_HEAD-CHARGETOTAL IS INITIAL.
    OUT_HEAD-CHARGETOTAL = OUT_HEAD-CHARGETOTAL * 100.
  ELSE.
    OUT_HEAD-CHARGETOTAL = 0.
  ENDIF.
  IF NOT OUT_HEAD-TAXTOTAL IS INITIAL.
    OUT_HEAD-TAXTOTAL = OUT_HEAD-TAXTOTAL * 100.
  ELSE.
    OUT_HEAD-TAXTOTAL = 0.
  ENDIF.
  IF NOT OUT_HEAD-GRANDTOTAL IS INITIAL.
    OUT_HEAD-GRANDTOTAL = OUT_HEAD-GRANDTOTAL * 100.
  ELSE.
    OUT_HEAD-GRANDTOTAL = 0.
  ENDIF.
  MODIFY OUT_HEAD INDEX 1.

* ITEM DATA (금액x100)
  READ TABLE OUT_ITEM INDEX 1.
  IF NOT OUT_ITEM-GOOD_QUAN IS INITIAL.
  ELSE.
    OUT_ITEM-GOOD_QUAN = 0.
  ENDIF.
  IF NOT OUT_ITEM-GOOD_UNIAMOUNT IS INITIAL.
  ELSE.
    OUT_ITEM-GOOD_UNIAMOUNT = 0.
  ENDIF.
  IF NOT OUT_ITEM-GOOD_INVAMOUNT IS INITIAL.
    OUT_ITEM-GOOD_INVAMOUNT = OUT_ITEM-GOOD_INVAMOUNT * 100.
  ELSE.
    OUT_ITEM-GOOD_INVAMOUNT = 0.
  ENDIF.
  IF NOT OUT_ITEM-GOOD_TAXAMOUNT IS INITIAL.
    OUT_ITEM-GOOD_TAXAMOUNT = OUT_ITEM-GOOD_TAXAMOUNT * 100.
  ELSE.
    OUT_ITEM-GOOD_TAXAMOUNT = 0.
  ENDIF.
  MODIFY OUT_ITEM INDEX 1.

* PAYM DATA (금액x100)
  READ TABLE OUT_PAYM INDEX 1.
  IF NOT OUT_PAYM-PAID_AMOUNT_10 IS INITIAL.
    OUT_PAYM-PAID_AMOUNT_10 = OUT_PAYM-PAID_AMOUNT_10 * 100.
  ELSE.
    OUT_PAYM-PAID_AMOUNT_10 = 0.
  ENDIF.
  IF NOT OUT_PAYM-PAID_AMOUNT_20 IS INITIAL.
    OUT_PAYM-PAID_AMOUNT_20 = OUT_PAYM-PAID_AMOUNT_20 * 100.
  ELSE.
    OUT_PAYM-PAID_AMOUNT_20 = 0.
  ENDIF.
  IF NOT OUT_PAYM-PAID_AMOUNT_30 IS INITIAL.
    OUT_PAYM-PAID_AMOUNT_30 = OUT_PAYM-PAID_AMOUNT_30 * 100.
  ELSE.
    OUT_PAYM-PAID_AMOUNT_30 = 0.
  ENDIF.
  IF NOT OUT_PAYM-PAID_AMOUNT_40 IS INITIAL.
    OUT_PAYM-PAID_AMOUNT_40 = OUT_PAYM-PAID_AMOUNT_40 * 100.
  ELSE.
    OUT_PAYM-PAID_AMOUNT_40 = 0.
  ENDIF.
  MODIFY OUT_PAYM INDEX 1.


  RETURN[] = GT_RETURN[].


* IF NOT OUT_HEAD IS INITIAL.
* CALL FUNCTION 'ZDTV3INF_SAPTOBC_VIEW_DATA'
* DESTINATION 'ZDTV3INF'
* IMPORTING
* URL = G_URL
* RETURN = RETURN
* TABLES
* HEAD = OUT_HEAD
* ITEM = OUT_ITEM
* PAYM = OUT_PAYM
* EXT = OUT_EXT
* EXCEPTIONS
* OTHERS = 4.
* ENDIF.



  CHECK NOT OUT_HEAD[] IS INITIAL.
  CALL FUNCTION 'ZDTV3INF_SAPTOBC_VIEW_DATA'
    DESTINATION 'ZDTV3INF'
    TABLES
      OUT_HEAD    = OUT_HEAD
      OUT_ITEM    = OUT_ITEM
      OUT_PAYM    = OUT_PAYM
      OUT_EXT     = OUT_EXT
      OUT_EXT_D   = OUT_EXT_D
      RETURN      = RETURN
    EXCEPTIONS
      OTHERS  = 4.


ENDFUNCTION.
