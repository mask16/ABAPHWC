FORM SAVE_LONG_TEXT USING P_DOCNO.

  LOOP AT IT_LTEXT.
    CLEAR   : IT_THEAD, IT_TLINE.
    REFRESH : IT_THEAD, IT_TLINE.

    CONCATENATE P_DOCNO IT_LTEXT-ZPOSN1 IT_LTEXT-ZPOSN2 INTO V_DOCNO.
    CONDENSE V_DOCNO.

    LOOP AT IT_LTEXT-ITEMS INTO V_ITEMS.
      IF SY-TABIX = 1.
        IT_TLINE-TDFORMAT = '*'.
      ELSE.
        IT_TLINE-TDFORMAT = '/'.
      ENDIF.

      IT_TLINE-TDLINE = V_ITEMS-TEXT.
      APPEND IT_TLINE.
    ENDLOOP.

    IT_THEAD-TDOBJECT = 'ZIMPORT'.
    IT_THEAD-TDNAME   = V_DOCNO.
    IT_THEAD-TDID     = IT_LTEXT-ZTXTID.
    IT_THEAD-TDSPRAS  = SY-LANGU.

    CALL FUNCTION 'SAVE_TEXT'
         EXPORTING
              CLIENT          = SY-MANDT
              HEADER          = IT_THEAD
              SAVEMODE_DIRECT = 'X'
         IMPORTING
              NEWHEADER       = IT_THEAD
         TABLES
              LINES           = IT_TLINE
         EXCEPTIONS
              ID              = 1
              LANGUAGE        = 2
              NAME            = 3
              OBJECT          = 4
              OTHERS          = 5.
  ENDLOOP.

ENDFORM. " SAVE_LONG_TEXT

FORM READ_LONG_TEXT TABLES P_ITEMS
                    USING    P_DOCNO

  CONDENSE V_DOCNO.

  CLEAR   : IT_THEAD, IT_TLINE.

       EXPORTING
            CLIENT                  = SY-MANDT
            ID                      = P_ZTXTID
            LANGUAGE                = SY-LANGU
            NAME                    = V_DOCNO

       EXCEPTIONS
            ID                      = 1
            LANGUAGE                = 2
            NAME                    = 3
            NOT_FOUND               = 4
            OBJECT                  = 5

            OTHERS                  = 8.

FORM SAVE_LONG_TEXT_TEMP USING P_ZPOSN1
                               P_ZPOSN2
                               P_ZTXTID
                               P_EDITOR TYPE REF TO CL_GUI_TEXTEDIT.

  REFRESH : IT_EDIT.

  CALL METHOD P_EDITOR->GET_TEXT_AS_R3TABLE
        IMPORTING
            TABLE =  IT_EDIT
        EXCEPTIONS
            OTHERS = 1.

  LOOP AT IT_EDIT INTO V_ITEMS.
    TRANSLATE V_ITEMS TO UPPER CASE.
    MODIFY IT_EDIT INDEX SY-TABIX FROM V_ITEMS.
  ENDLOOP.

  READ TABLE IT_LTEXT WITH KEY ZPOSN1 = P_ZPOSN1
                               ZPOSN2 = P_ZPOSN2
                               ZTXTID = P_ZTXTID.
  IF SY-SUBRC = 0.
    DESCRIBE TABLE IT_EDIT LINES V_LINES.

    IF V_LINES > IT_LTEXT-ZLNCNT.
      MESSAGE I019(ZIM01) WITH IT_LTEXT-ZLTEXT IT_LTEXT-ZLNCNT
                        IT_LTEXT-ZLNCNT.
      V_LINES = IT_LTEXT-ZLNCNT + 1.
      DELETE IT_EDIT FROM V_LINES.
    ENDIF.

    MOVE IT_EDIT[] TO IT_LTEXT-ITEMS[].
    MODIFY IT_LTEXT INDEX SY-TABIX.
  ELSE.
    READ TABLE IT_LTEXT WITH KEY ZPOSN1 = P_ZPOSN1
                                 ZTXTID = P_ZTXTID.
    IF SY-SUBRC = 0.
      MOVE : P_ZPOSN2  TO IT_LTEXT-ZPOSN2,
             IT_EDIT[] TO IT_LTEXT-ITEMS[].
      APPEND IT_LTEXT.
    ENDIF.
  ENDIF.

ENDFORM. " SAVE_LONG_TEXT_TEMP
