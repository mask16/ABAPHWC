FORM GET_PERNR_INFO USING VALUE(P_PERNR)
                           VALUE(P_IDATE)
                  CHANGING VALUE(P_INSA) TYPE ZHR_INSA.

*  DATA : L_ORGDT TYPE DATUM.
*
*  "날자에 -1 한 것을 다시 +1 함 (조직을 READ 시에만 사용)
*  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
*    EXPORTING
*      DATE      = P_IDATE   "기준일
**     DAYS      = '1'
*      DAYS      = '2'       "기준일 + 1일
*      MONTHS    = '00'
*      SIGNUM    = '+'
*      YEARS     = '00'
*    IMPORTING
*      CALC_DATE = L_ORGDT.

  READ TABLE IT_0000 WITH KEY PERNR = P_PERNR.
  IF SY-SUBRC = 0.
    READ TABLE GT_STAT2 WITH KEY STATV = IT_0000-STAT2.
    IF SY-SUBRC = 0.
      P_INSA-STAT2 = IT_0000-STAT2.
      P_INSA-STATX = GT_STAT2-TEXT1.
    ENDIF.
  ENDIF.

  "성명 / 소속부서
  READ TABLE IT_0001 WITH KEY PERNR = P_PERNR.
  IF SY-SUBRC = 0.
    P_INSA-ENAME = IT_0001-ENAME.
    P_INSA-ORGEH = IT_0001-ORGEH.
    PERFORM GET_OBJECT_TEXT USING '01'
                                  'O'
                                  P_INSA-ORGEH
*                                  L_ORGDT
                                  P_IDATE
                         CHANGING P_INSA-ORGTX.

    IF NOT P_INSA-ORGEH IS INITIAL AND
           P_INSA-ORGTX IS INITIAL.
      SELECT SINGLE ORGTX FROM T527X
        INTO P_INSA-ORGTX
       WHERE SPRSL = SY-LANGU
         AND ORGEH = P_INSA-ORGEH
*         AND ENDDA >= L_ORGDT
*         AND BEGDA <= L_ORGDT.
         AND ENDDA >= P_IDATE
         AND BEGDA <= P_IDATE.
    ENDIF.

    READ TABLE GT_PERSK WITH KEY PERSK = IT_0001-PERSK.
    IF SY-SUBRC = 0.
      P_INSA-PERSK = IT_0001-PERSK.
      P_INSA-PKTXT = GT_PERSK-PTEXT.
    ENDIF.
  ENDIF.

  READ TABLE IT_0002 WITH KEY PERNR = P_PERNR.
  IF SY-SUBRC = 0.
    P_INSA-TITEL = IT_0002-TITEL.
    P_INSA-TITL2 = IT_0002-TITL2.
    CONCATENATE IT_0002-NACHN IT_0002-VORNA
           INTO P_INSA-EENAME SEPARATED BY SPACE.
  ENDIF.

  LOOP AT IT_0105 WHERE PERNR = P_PERNR.
    CASE IT_0105-SUBTY.
      WHEN '0010'.
        P_INSA-EMAIL = IT_0105-USRID_LONG.
      WHEN 'CELL'.
        P_INSA-TCELL = IT_0105-USRID.
      WHEN '0020'.
        P_INSA-T0020 = IT_0105-USRID.
    ENDCASE.
  ENDLOOP.

ENDFORM. " GET_PERNR_INFO

FORM GET_OBJECT_TEXT USING VALUE(P_PLVAR) TYPE PLVAR
                               VALUE(P_OTYPE) TYPE OTYPE
                               VALUE(P_OBJID) TYPE HROBJID
                               VALUE(P_BEGDA) TYPE DATUM
                      CHANGING VALUE(P_STEXT).

  DATA : L_SHORT LIKE OBJEC-SHORT,
         L_STEXT LIKE OBJEC-STEXT.

  CALL FUNCTION 'RH_READ_OBJECT'
    EXPORTING
      BEGDA     = P_BEGDA
      ENDDA     = P_BEGDA
      ISTAT     = '1'
      LANGU     = SY-LANGU
      OBJID     = P_OBJID
      OINTERVAL = 'X'
      OTYPE     = P_OTYPE
      PLVAR     = P_PLVAR
    IMPORTING
      SHORT     = L_SHORT
      STEXT     = L_STEXT
    EXCEPTIONS
      NOT_FOUND = 01.
  IF SY-SUBRC = 0.
    IF L_STEXT <> SPACE.
      P_STEXT = L_STEXT.
    ELSE.
      P_STEXT = L_SHORT.
    ENDIF.
  ELSE.
    CLEAR P_STEXT.
  ENDIF.

ENDFORM. " GET_OBJECT_TEXT
