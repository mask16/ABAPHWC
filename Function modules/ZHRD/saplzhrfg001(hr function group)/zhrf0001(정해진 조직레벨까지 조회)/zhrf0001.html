             lv_im_orgeh TYPE orgeh.

  RANGES : ra_orgeh   FOR hrp1000-objid.

  lv_keydate = i_keydate.

* 인사하위영역/최하위소속

    CLEAR p0001.

        leavingdate            = lv_keydate

  IF i_orgeh IS INITIAL.

  "인사하위영역
  e_btrtl = p0001-btrtl.

  "최하위소속
  SELECT SINGLE objid stext
    INTO (e_orgeh, e_orgtx)

    WHERE plvar = '01'

      AND langu = sy-langu
      AND objid = lv_im_orgeh
      AND begda <= lv_keydate
      AND endda => lv_keydate.

* 조직
  CALL FUNCTION 'RHPH_STRUCTURE_READ'
    EXPORTING
      plvar             = '01'
      otype             = 'O'
      objid             = lv_im_orgeh
      wegid             = 'O-O'
      begda             = lv_keydate
      endda             = lv_keydate
      pup_info          = 'X'
      with_stext        = 'X'
    TABLES
      stru_tab          = org_result
    EXCEPTIONS
      catalogue_problem = 1

  SELECT *
  INTO TABLE lt_org
  FROM hrp9800 CLIENT SPECIFIED
  WHERE mandt = sy-mandt

    AND otype = 'O'

    AND begda <= lv_keydate
    AND endda => lv_keydate.

  DATA : lv_orgeh(20),
         lv_orgtx(20).
  FIELD-SYMBOLS: <fs> TYPE any.

  LOOP AT lt_org.
    READ TABLE org_result WITH KEY objid = lt_org-objid.
    IF sy-subrc = 0.
      lt_orgeh-scode = lt_org-scode.

      SELECT SINGLE sctext INTO  lt_orgeh-sctext
                                   FROM zhrt9001t
                                  WHERE code = 'A001'
                                        AND scode = lt_org-scode
                                        AND spras = sy-langu.

      lt_orgeh-orgeh = org_result-objid.
      lt_orgeh-orgtx = org_result-stext.
      APPEND lt_orgeh.

    ENDIF.
  ENDLOOP.

  CHECK  NOT i_prog IS INITIAL.  " 추가
  SELECT SINGLE org_lev INTO lv_org
                                   FROM zhrt9002
                                   WHERE prog = i_prog.
  IF sy-subrc = 0.
    DELETE  lt_orgeh WHERE  scode > lv_org.
  ELSE.
    CLEAR : lt_orgeh[], lt_orgeh.
  ENDIF.
ENDFUNCTION.
