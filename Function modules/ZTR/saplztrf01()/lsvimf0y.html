FORM vim_get_global_field_value USING value(name_of_field_to_get) TYPE c
                                      value(type_of_field_to_get) TYPE c

                                      vggfv_return LIKE sy-subrc.

  ASSIGN (name_of_field_to_get) TO <field>

  ENDIF.
