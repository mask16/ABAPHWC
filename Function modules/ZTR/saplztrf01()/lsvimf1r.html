FORM TRANSPORTIEREN.
  DATA: lv_text TYPE string.
  CONCATENATE svim_text_018 svim_text_021 svim_text_019 INTO lv_text SEPARATED BY space.
  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar      = svim_text_020  "Eintr#ge transportieren
      text_question = lv_text
    IMPORTING
      answer        = answer.

  CASE answer.
    WHEN '1'.         "'J'.
      sy-subrc = 0.
    WHEN '2'.         "'N'.
      CLEAR <STATUS>-UPD_FLAG.
      SY-SUBRC = 8.
    WHEN 'A'.
      SY-SUBRC = 12.
  ENDCASE.
  CLEAR lv_text.
ENDFORM.
