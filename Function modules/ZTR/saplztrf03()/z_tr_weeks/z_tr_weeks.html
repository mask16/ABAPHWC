FUNCTION z_tr_weeks.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_SPMON) TYPE SPMON
*" REFERENCE(I_WEEKN) TYPE WEEKN
*" EXPORTING
*" REFERENCE(E_WEEK) TYPE WEEKN
*" REFERENCE(E_MONDAY) LIKE SY-DATUM
*" REFERENCE(E_SUNDAY) LIKE SY-DATUM
*" EXCEPTIONS
*" PARAM_ERR
*"----------------------------------------------------------------------
*-- IMPORT 값이 존재하지 않음
* Global data declarations

  IF i_spmon = '' AND i_weekn = ''.
    RAISE param_err. EXIT.
  ENDIF.

  DATA : lv_fiday  LIKE sy-datum,
         lv_today  LIKE sy-datum,
         lv_weekn  TYPE weekn,
         lv_weeke  TYPE weekn,
         lv_seek   LIKE scal-week,
         lv_week   LIKE scal-week,
         lv_monday LIKE sy-datum,
         lv_sunday LIKE sy-datum,
         lv_eeek   LIKE scal-week,
         lv_eonday LIKE sy-datum,
         lv_eunday LIKE sy-datum.

  CONCATENATE i_spmon '01' INTO lv_fiday.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = lv_fiday
    IMPORTING
      last_day_of_month = lv_today
    EXCEPTIONS
      day_in_no_date    = 1
      OTHERS            = 2.

*-- 월의 시작일에 대한 주차를 구함.
  CLEAR : lv_monday, lv_sunday.

  CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
    EXPORTING
      date   = lv_fiday
    IMPORTING
      week   = lv_seek
      monday = lv_monday
      sunday = lv_sunday.
*-- 월의 시작일과 주 시작일이 동일한지 비교하여 동일하지 않으면
*-- 주 마지막일의 다음일(다음주차시작일)로 년의 주와 주 시작일과
*-- 주 마지막일을 다시 구한다.
  IF lv_fiday <> lv_monday.
    lv_fiday = lv_sunday + 1.

    CLEAR : lv_monday, lv_sunday.

    CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
      EXPORTING
        date   = lv_fiday
      IMPORTING
        week   = lv_week
        monday = lv_monday
        sunday = lv_sunday.
  ENDIF.

  lv_weekn = '01'.

*-- 입력받은 주별로 Export Data를 구함.
  CASE i_weekn.
    WHEN '01'.
*-- 월의 시작 주는 미리 구해진 Data를 Export함.
      e_week = '01'. e_monday = lv_monday. e_sunday = lv_sunday.
    WHEN '02'.
*-- 월의 두번째 주는 월의 시작 주 마지막일의 다음일(다음주차시작일)로
*-- 년의 주와 주 시작일과 주 마지막일을 다시 구해서 Data를 Export함.
      lv_fiday = lv_sunday + 1.
      CLEAR : lv_week.
      CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
        EXPORTING
          date   = lv_fiday
        IMPORTING
          week   = lv_week
          monday = e_monday
          sunday = e_sunday.
      e_week = '02'.
    WHEN '03'.
*-- 월의 셋번째 주는 월의 시작 주 마지막일의 7일후(다다음주차시작일)로
*-- 년의 주와 주 시작일과 주 마지막일을 다시 구해서 Data를 Export함.
      lv_fiday = lv_sunday + 8.
      CLEAR : lv_week.
      CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
        EXPORTING
          date   = lv_fiday
        IMPORTING
          week   = lv_week
          monday = e_monday
          sunday = e_sunday.
      e_week = '03'.
    WHEN '04'.
*-- 월의 넷번째 주는 월의 시작 주 마지막일의 14일후(다다다음주차시작일)로
*-- 년의 주와 주 시작일과 주 마지막일을 다시 구해서 Data를 Export함.
      lv_fiday = lv_sunday + 15.
      CLEAR : lv_week.
      CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
        EXPORTING
          date   = lv_fiday
        IMPORTING
          week   = lv_week
          monday = e_monday
          sunday = e_sunday.
      e_week = '04'.
    WHEN '05'.
*-- 월의 다섯번째 주는 월의 시작 주 마지막일의 14일후(다다다음주차시작일)로
*-- 년의 주와 주 시작일과 주 마지막일을 다시 구해서 Data를 Export함.
      lv_fiday = lv_sunday + 22.
      IF i_spmon = lv_fiday+0(6).
        CLEAR : lv_week.
        CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
          EXPORTING
            date   = lv_fiday
          IMPORTING
            week   = lv_week
            monday = e_monday
            sunday = e_sunday.
        e_week = '05'.
      ENDIF.
  ENDCASE.

ENDFUNCTION.
