FUNCTION Z_TR_CURRENT_SEC_POS2.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BUKRS) TYPE BUKRS DEFAULT '1000'
*" TABLES
*" T_DATE STRUCTURE RSPARAMS
*" T_GSART STRUCTURE RSPARAMS
*" T_PARTNER STRUCTURE RSPARAMS OPTIONAL
*" E_POS STRUCTURE ZTRS0010
*" CHANGING
*" REFERENCE(CT_MESSAGES) TYPE BAPIERR_T OPTIONAL
*" EXCEPTIONS
*" RLDEPO_ERR
*" RANL_ERR
*" RFHA_ERR
*" PARAM_ERR
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LT_ITAB   LIKE TZPAT  OCCURS 0 WITH HEADER LINE,
         BEGIN OF LT_BUT0 OCCURS 0,
           PARTNER     LIKE BUT000-PARTNER,
         END OF LT_BUT0.
  DATA : L_DATUM1      TYPE SY-DATUM,
         L_DATUM2      TYPE SY-DATUM,
         L_DAY(3)      TYPE N.
  RANGES : LR_GSART    FOR TZPA-GSART,
           LR_PARTNER  FOR BUT000-PARTNER.

  READ TABLE T_DATE INDEX 1.
  L_DATUM1  = T_DATE-LOW.
  L_DATUM2  = T_DATE-HIGH.
  IF T_DATE-HIGH IS INITIAL.
    L_DATUM2  = T_DATE-LOW.
  ENDIF.
  L_DAY   = L_DATUM2 - L_DATUM1 + 1.

* DAT_LIST 관련기준일과 평가기준(전기일, 포지션일)
  DO L_DAY TIMES.
    MOVE : L_DATUM1  TO IT_IDAT_LIST-STICHTAG,
          '1'        TO IT_IDAT_LIST-RSTICH.
    COLLECT IT_IDAT_LIST INTO ST_IWA_CCSEL_LDB-DAT_LIST.
    ADD 1 TO L_DATUM1.
    IF L_DATUM1 > L_DATUM2.
      EXIT.
    ENDIF.
  ENDDO.

* RANGE관련 각 필드의 선택값 정의
* 회사코드
  REFRESH : IT_ISEL, IT_IRANGE, E_POS.
  MOVE : 'BUKRS' TO IT_IRANGE-FIELDNAME,
         'I'     TO IT_ISEL-SIGN,
         'EQ'    TO IT_ISEL-OPTION,
         I_BUKRS TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 유가증권 구분자 CHECK MARK
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'FLG_SE' TO IT_IRANGE-FIELDNAME,
         'X'      TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'FLG_OTC' TO IT_IRANGE-FIELDNAME,
         'X'       TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 금융상품유형
  REFRESH : IT_ISEL, IT_IRANGE, LR_GSART.
  LOOP AT T_GSART.
    MOVE-CORRESPONDING T_GSART TO LR_GSART.
    COLLECT LR_GSART.
  ENDLOOP.

  SELECT * INTO TABLE LT_ITAB
    FROM TZPAT
   WHERE GSART  IN LR_GSART.
  LOOP AT LT_ITAB.
    MOVE : 'GSART'       TO IT_IRANGE-FIELDNAME,
           'I'           TO IT_ISEL-SIGN,
           'EQ'          TO IT_ISEL-OPTION,
           LT_ITAB-GSART TO IT_ISEL-LOW.
    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  ENDLOOP.

  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* BP
  IF T_PARTNER[] IS NOT INITIAL.
    LOOP AT T_PARTNER.
      MOVE-CORRESPONDING T_PARTNER TO LR_PARTNER.
      COLLECT LR_PARTNER.
    ENDLOOP.
    SELECT PARTNER INTO TABLE LT_BUT0
      FROM BUT000
     WHERE PARTNER  IN LR_PARTNER.
    LOOP AT LT_BUT0.
      MOVE : 'PARTNR'      TO IT_IRANGE-FIELDNAME,
             'I'           TO IT_ISEL-SIGN,
             'EQ'          TO IT_ISEL-OPTION,
             LT_BUT0-PARTNER TO IT_ISEL-LOW.
      COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
    ENDLOOP.
    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.
  ENDIF.

* 시장가
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'KURSA' TO IT_IRANGE-FIELDNAME.
  MOVE '01'    TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* ???
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'NUMBR' TO IT_IRANGE-FIELDNAME.
  MOVE '001'   TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 상품 그룹(001:유가증권 003:머니마켓/외환 004:파생)
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'PRODUCT_GROUP' TO IT_IRANGE-FIELDNAME.
  MOVE '001'           TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.

  MOVE '003'           TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 포지션기준일(1) 전기일(2) 기준
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'RSTICH_PB' TO IT_IRANGE-FIELDNAME.
  MOVE '1'         TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 평가영역
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'VALUATION_AREA' TO IT_IRANGE-FIELDNAME.
  MOVE '001'            TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

  PERFORM MAKE_IT_CCSF.

  CALL FUNCTION 'FTI_LDB_PB_SEL_POSITIONS'
    EXPORTING
      IM_SELECTION    = ST_IWA_CCSEL_LDB
      IM_TAB_CCSF     = ST_IT_CCSF
    IMPORTING
      EX_TAB_POSITION = ST_ET_POSITIONS
    CHANGING
      CT_MESSAGES     = CT_MESSAGES.

  DATA LS_POSITIONS  TYPE FTI_LDB_TR_POSITIONS_T WITH HEADER LINE.

  LOOP AT ST_ET_POSITIONS INTO LS_POSITIONS.
    CLEAR E_POS.
    MOVE-CORRESPONDING LS_POSITIONS TO E_POS.
    E_POS-PDATE  = LS_POSITIONS-STICHTAG.
    E_POS-SPMON  = E_POS-PDATE.
    IF LS_POSITIONS-SECURITY_ACCOUNT IS INITIAL.
      E_POS-SECURITY_ID  = LS_POSITIONS-DEAL_NUMBER.
    ENDIF.
    COLLECT E_POS.
  ENDLOOP.

  DELETE E_POS   WHERE BOOK_VAL_PC = 0
                   AND BOOK_VAL_LC = 0.

ENDFUNCTION.
