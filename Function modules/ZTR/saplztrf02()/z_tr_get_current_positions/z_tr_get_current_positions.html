FUNCTION Z_TR_GET_CURRENT_POSITIONS.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BUKRS) TYPE BUKRS
*" REFERENCE(I_PDATE) TYPE DBESTAND
*" REFERENCE(I_RFHA) TYPE TB_RFHA
*" REFERENCE(I_RLDEPO) TYPE RLDEPO
*" REFERENCE(I_RANL) TYPE VVRANLW
*" EXPORTING
*" REFERENCE(E_NOMINAL_AMT_PC) TYPE FTI_NOMINAL_AMT_PC
*" REFERENCE(E_PURCH_PC) TYPE FTI_PURCH_PC
*" REFERENCE(E_PURCH_LC) TYPE FTI_PURCH_LC
*" REFERENCE(E_BOOK_VAL_PC) TYPE FTI_BOOK_VAL_PC
*" REFERENCE(E_BOOK_VAL_LC) TYPE FTI_BOOK_VAL_LC
*" REFERENCE(E_UNITS) TYPE FTI_UNITS
*" REFERENCE(EX_TAB_POSITION) TYPE FTI_LDB_TR_POSITIONS_T
*" CHANGING
*" REFERENCE(CT_MESSAGES) TYPE BAPIERR_T
*" EXCEPTIONS
*" RLDEPO_ERR
*" RANL_ERR
*" RFHA_ERR
*" PARAM_ERR
*"----------------------------------------------------------------------
* 거래 및 유가증권 정보가 없는 경우
* Global data declarations

  IF I_RFHA = '' AND I_RANL = '' AND I_RLDEPO = ''.
    RAISE PARAM_ERR. EXIT.
  ENDIF.

* 유가증권 계좌정보는 있으나 종목이 없는 경우
  IF I_RFHA = '' AND I_RANL = '' AND I_RLDEPO <> ''.
    RAISE RANL_ERR. EXIT.
  ENDIF.

* 유가증권 종목정보는 있으나 계좌가 없는 경우
  IF I_RFHA = '' AND I_RLDEPO = '' AND I_RANL <> ''.
    RAISE RLDEPO_ERR. EXIT.
  ENDIF.

  REFRESH : IT_IDAT_LIST, ST_IT_CCSF.
  CLEAR : ST_IWA_CCSEL_LDB.

* DAT_LIST 관련기준일과 평가기준(전기일, 포지션일)
  MOVE : I_PDATE TO IT_IDAT_LIST-STICHTAG,
         '1'     TO IT_IDAT_LIST-RSTICH.

  COLLECT IT_IDAT_LIST INTO ST_IWA_CCSEL_LDB-DAT_LIST.

* RANGE관련 각 필드의 선택값 정의
* 회사코드
  REFRESH : IT_ISEL, IT_IRANGE, IT_IDAT_LIST.

  MOVE : 'BUKRS' TO IT_IRANGE-FIELDNAME,
         'I'     TO IT_ISEL-SIGN,
         'EQ'    TO IT_ISEL-OPTION,
         I_BUKRS TO IT_ISEL-LOW.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.

  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 거래번호
  IF I_RFHA <> ''.

    REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.

    MOVE : 'DEAL_NUMBER' TO IT_IRANGE-FIELDNAME,
           I_RFHA        TO IT_ISEL-LOW.

    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

    REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.

    MOVE : 'FLG_OTC' TO IT_IRANGE-FIELDNAME,
           'X'       TO IT_ISEL-LOW.

    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.

    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

  ENDIF.
*SECURITY_ID SECURITY_ACCOUNT FLG_SE FLG_OTC

* 유가증권 종목
  IF I_RANL <> ''.

    REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.

    MOVE : 'FLG_SE' TO IT_IRANGE-FIELDNAME,
           'X' TO IT_ISEL-LOW.

    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.

    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

    REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.

    MOVE : 'SECURITY_ID' TO IT_IRANGE-FIELDNAME,
           I_RANL TO IT_ISEL-LOW.

    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.

    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

  ENDIF.

* 유가증권 계좌
  IF I_RLDEPO <> ''.
    REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
    MOVE : 'SECURITY_ACCOUNT' TO IT_IRANGE-FIELDNAME,
           I_RLDEPO           TO IT_ISEL-LOW.
    COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
    APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.
  ENDIF.

* 시장가
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'KURSA' TO IT_IRANGE-FIELDNAME,
         '01'    TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* ???
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'NUMBR' TO IT_IRANGE-FIELDNAME,
         '001'   TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 상품 그룹(001:유가증권 003:머니마켓/외환 004:파생)
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE 'PRODUCT_GROUP' TO IT_IRANGE-FIELDNAME.

  SELECT SINGLE * FROM VTBFHA
         WHERE BUKRS = I_BUKRS
         AND RFHA = I_RFHA.

  IF SY-SUBRC = 0.
    IF VTBFHA-RANTYP = '2'.
      MOVE '001' TO IT_ISEL-LOW.
    ELSEIF VTBFHA-RANTYP = '4'.
      MOVE '003' TO IT_ISEL-LOW.
    ELSEIF VTBFHA-RANTYP = '5'.
      MOVE '003' TO IT_ISEL-LOW.
    ELSEIF VTBFHA-RANTYP = '6'.
      MOVE '004' TO IT_ISEL-LOW.
    ENDIF.
  ELSE.
    IF I_RANL <> ''.
      MOVE '001' TO IT_ISEL-LOW.
    ENDIF.
  ENDIF.

  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 포지션기준일(1) 전기일(2) 기준
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'RSTICH_PB' TO IT_IRANGE-FIELDNAME,
         '1'         TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

* 평가영역
  REFRESH : IT_ISEL, IT_IRANGE, IT_IRANGE-SELOPT_T.
  MOVE : 'VALUATION_AREA' TO IT_IRANGE-FIELDNAME,
         '001'            TO IT_ISEL-LOW.
  COLLECT IT_ISEL INTO IT_IRANGE-SELOPT_T.
  APPEND IT_IRANGE TO ST_IWA_CCSEL_LDB-RANGE.

  PERFORM MAKE_IT_CCSF.

  REFRESH : ST_ET_POSITIONS, IT_EPOS,
            EX_TAB_POSITION.

  CALL FUNCTION 'FTI_LDB_PB_SEL_POSITIONS'
    EXPORTING
      IM_SELECTION    = ST_IWA_CCSEL_LDB
      IM_TAB_CCSF     = ST_IT_CCSF
    IMPORTING
      EX_TAB_POSITION = ST_ET_POSITIONS
    CHANGING
      CT_MESSAGES     = CT_MESSAGES.

  MOVE ST_ET_POSITIONS[] TO :IT_EPOS[],
                             EX_TAB_POSITION[].

  CLEAR E_BOOK_VAL_PC.

  LOOP AT IT_EPOS.

    MOVE : IT_EPOS-NOMINAL_AMT_PC TO E_NOMINAL_AMT_PC,
           IT_EPOS-PURCH_PC       TO E_PURCH_PC,
           IT_EPOS-PURCH_LC       TO E_PURCH_LC,
           IT_EPOS-BOOK_VAL_PC    TO E_BOOK_VAL_PC,
           IT_EPOS-BOOK_VAL_LC    TO E_BOOK_VAL_LC,
* 2009.10.12 : Unit 추가
           IT_EPOS-UNITS          TO E_UNITS.

  ENDLOOP.

ENDFUNCTION.
