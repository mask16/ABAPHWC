  clear: gt_dfies, dref, gv_off, ldtab.

* IF gs_fieldcat-inttype CA 'CNTD'.

  call function 'DDIF_FIELDINFO_GET'
    exporting
      tabname              = i_tabname
* FIELDNAME = ' '
* LANGU = SY-LANGU

* ALL_TYPES = ' '
* GROUP_NAMES = ' '

* IMPORTING

* DDOBJTYPE =
* DFIES_WA =
* LINES_DESCR =
    tables
      dfies_tab            = gt_dfies
* FIXED_VALUES =
    exceptions
      not_found            = 1
      internal_error       = 2
      others               = 3.
  if sy-subrc <> 0.
    message id sy-msgid type sy-msgty number sy-msgno
            with sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  endif.

  try.
      create data dref type (i_tabname).
      assign dref->* to <gt_line>.
    catch cx_sy_create_data_error.
      write 'Wrong Database!'.
  endtry.

  <gt_line> = i_structure_data.
  if i_gubun is initial.
    loop at gt_dfies.
      assign component gt_dfies-fieldname of structure <gt_line>
                    to <fs>.
      if gt_dfies-inttype ca 'CNTD'.
        o_string_data+gv_off(gt_dfies-outputlen) = <fs>.
      else.
        write <fs> to o_string_data+gv_off(gt_dfies-outputlen)
                   no-grouping.
      endif.
      gv_off = gv_off + gt_dfies-outputlen.
    endloop.
  else.
    data i type i.
    loop at gt_dfies.
      assign component gt_dfies-fieldname of structure <gt_line>
                    to <fs>.
      concatenate o_string_data <fs> i_gubun into o_string_data.
    endloop.
  endif.


endfunction.
