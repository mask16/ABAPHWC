FUNCTION ZIF_RFC_TABLE_DATA_MIGRATION.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_TABLE) LIKE DD02L-TABNAME
*" TABLES
*" T_VALUE STRUCTURE ZCMS001
*" T_WHERE STRUCTURE RSDSWHERE
*"----------------------------------------------------------------------


* Global data declarations
* Global data declarations

  DATA: LDTAB       TYPE REF TO DATA.
  DATA: GT_FIELDCAT TYPE LVC_T_FCAT.

  DATA: WA_VALUE  TYPE ZCMS001-T_VALUE.

  FIELD-SYMBOLS: <ALL_TABLE> TYPE TABLE.
  FIELD-SYMBOLS: <WA>        TYPE ANY.

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      I_STRUCTURE_NAME = I_TABLE
    CHANGING
      CT_FIELDCAT      = GT_FIELDCAT.

  CALL METHOD CL_ALV_TABLE_CREATE=>CREATE_DYNAMIC_TABLE
    EXPORTING
      IT_FIELDCATALOG = GT_FIELDCAT
    IMPORTING
      EP_TABLE        = LDTAB.

  ASSIGN LDTAB->* TO <ALL_TABLE>.

  ASSIGN LOCAL COPY OF INITIAL LINE OF <ALL_TABLE> TO <WA>.

  IF T_WHERE[] IS INITIAL.
    SELECT *
      FROM (I_TABLE)
      INTO TABLE <ALL_TABLE>.
  ELSE.
    SELECT *
        FROM (I_TABLE)
        INTO TABLE <ALL_TABLE>
        WHERE (T_WHERE).
  ENDIF.

  LOOP AT <ALL_TABLE> ASSIGNING <WA>.

    call function 'ZIF_STRUCTURE_TO_STRING'
      EXPORTING
        I_TABNAME        = I_TABLE
        I_STRUCTURE_DATA = <WA>
      IMPORTING
        O_STRING_DATA    = WA_VALUE.

    APPEND WA_VALUE  TO T_VALUE.
    CLEAR: WA_VALUE.
  ENDLOOP.


ENDFUNCTION.
