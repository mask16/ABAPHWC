MODULE SET_USER_ZPMT1010 INPUT.

  DATA : LS_1010 TYPE ZPMT1010.
  CLEAR: LS_1010.

  IF ZPMT1010-ROLEID IS INITIAL.
    MESSAGE E000 WITH 'ROLE ID 값을 없습니다.'.
  ENDIF.
  IF ZPMT1010-TCODE IS INITIAL.
    MESSAGE E000 WITH 'T-CODE 값을 없습니다.'.
  ENDIF.

  ZPMT1010-AENAM = SY-UNAME.
  ZPMT1010-AEDAT = SY-DATUM.
  ZPMT1010-AEZET = SY-UZEIT.

  SELECT SINGLE * INTO LS_1010
    FROM ZPMT1010
   WHERE ROLEID = ZPMT1010-ROLEID
     AND TCODE  = ZPMT1010-TCODE.

  IF SY-SUBRC <> 0.
    ZPMT1010-ERNAM = ZPMT1010-AENAM.
    ZPMT1010-ERDAT = ZPMT1010-AEDAT.
    ZPMT1010-ERZET = ZPMT1010-AEZET.
  ENDIF.

ENDMODULE. " SET_USER_ZPMT1010 INPUT
