FUNCTION ZPM_ATTACH_FILE.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_DOCTY) TYPE ZPMT0150-DOCTY
*" REFERENCE(I_DOCNO) TYPE ZPMT0150-DOCNO
*" EXPORTING
*" REFERENCE(E_INFO) TYPE ZPMT0150
*" REFERENCE(E_RETURN) TYPE ZPMS0010
*" TABLES
*" T_DATA STRUCTURE ZPMT0150 OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  CLEAR : T_DATA, T_DATA[], E_INFO.

  DATA : LS_0150 LIKE ZPMT0150 ,
         LT_0150 LIKE ZPMT0150 OCCURS 0 WITH HEADER LINE.
  DATA : LV_DOKNR LIKE DRAW-DOKNR.
  DATA : LV_DOKVR LIKE DRAW-DOKVR.
  DATA : LV_DOKTL(3) TYPE N.
  DATA : LS_RETURN LIKE ZPMS0010.
  DATA : LT_FILE LIKE BAPI_DOC_FILES2 OCCURS 0 WITH HEADER LINE.

  CLEAR: LS_0150, LT_0150, LT_0150[], LV_DOKNR, LV_DOKVR, LV_DOKTL, LS_RETURN,
  LT_FILE, LT_FILE[].

  MOVE I_DOCTY TO LS_0150-DOCTY.
  MOVE I_DOCNO TO LS_0150-DOCNO.

  PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING LS_0150-DOCTY.
  PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING LS_0150-DOCNO.

  SELECT *
    INTO TABLE LT_0150
    FROM ZPMT0150
   WHERE DOCTY = LS_0150-DOCTY
     AND DOCNO = LS_0150-DOCNO.
* AND DOKAR = 'DRW'
  IF SY-SUBRC = 0.

    LS_0150-DOKAR = 'DRW'.           "문서유형
    SORT LT_0150 BY DOKVR DESCENDING DOKTL DESCENDING.

    READ TABLE LT_0150 INDEX 1.               " DOKVR,  DOKTL   DOKVR, DOKTL
    IF LT_0150-DOKTL = '999'.                 " 00   ,  999  -> 01   , 000
      LV_DOKVR = LT_0150-DOKVR + 1.
      PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING LV_DOKVR.
      LS_0150-DOKVR = LV_DOKVR.
      LS_0150-DOKTL = '001'.
    ELSE.                                         " DOKVR,  DOKTL   DOKVR, DOKTL
      LV_DOKTL = LT_0150-DOKTL.               " 00   ,  015  -> 00   , 016
      LV_DOKTL = LV_DOKTL + 1.
      PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING LV_DOKTL.
      LS_0150-DOKTL = LV_DOKTL.
      LS_0150-DOKVR = LT_0150-DOKVR.
    ENDIF.

  ELSE.

    LS_0150-DOKAR = 'DRW'.           "문서유형
    LS_0150-DOKVR = '00'.
    LS_0150-DOKTL = '001'.           "부분문서(시리얼 번호)

  ENDIF.

* PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING LS_0150-DOKNR.

* 실제 파일 저장하는 BAPI 함수 수행
  call function 'ZPM_CREATE_DOCUMENT'
    EXPORTING
      I_DOKAR  = LS_0150-DOKAR
      I_DOKNR  = LS_0150-DOKNR
      I_DOKVR  = LS_0150-DOKVR
      I_DOKTL  = LS_0150-DOKTL
    IMPORTING
      E_RETURN = LS_RETURN
      E_DOKNR  = LS_0150-DOKNR
    TABLES
      T_FILES  = LT_FILE.

  IF LS_RETURN-TYPE = 'E' or LS_RETURN-TYPE = 'N'.
    LS_RETURN-TYPE = 'E'.
    E_RETURN = LS_RETURN.
    EXIT.
  ENDIF.

  READ TABLE LT_FILE INDEX 1.
  IF LT_FILE-WSAPPLICATION = ''.
    EXIT.
  ENDIF.

  LS_0150-DAPPL    = LT_FILE-WSAPPLICATION.
  LS_0150-FILENAME = LT_FILE-DOCFILE.

  LS_0150-ERDAT = SY-DATUM.
  LS_0150-ERZET = SY-UZEIT.
  LS_0150-ERNAM = SY-UNAME.
  LS_0150-AEDAT = SY-DATUM.
  LS_0150-AEZET = SY-UZEIT.
  LS_0150-AENAM = SY-UNAME.

  MODIFY ZPMT0150 FROM LS_0150.

  APPEND LS_0150 TO T_DATA.
  E_INFO = LS_0150 .

ENDFUNCTION.
