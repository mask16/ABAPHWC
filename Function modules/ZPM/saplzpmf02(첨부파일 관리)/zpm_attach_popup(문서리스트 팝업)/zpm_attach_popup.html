FUNCTION ZPM_ATTACH_POPUP.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_DOCTY)
*" REFERENCE(I_DOCNO)
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_DOCTY TYPE ZPMT0150-DOCTY,
         L_DOCNO TYPE ZPMT0150-DOCNO,
         L_CNT   TYPE I.
  CLEAR: L_DOCTY, L_DOCNO, L_CNT.

  MOVE I_DOCTY TO L_DOCTY.
  MOVE I_DOCNO TO L_DOCNO.

  PERFORM CONVERSION_EXIT_ALPHA_INPUT  CHANGING L_DOCNO.

  call function 'ZPM_ATTACH_CHECK'
    EXPORTING
      I_DOCTY = L_DOCTY
      I_DOCNO = L_DOCNO
    IMPORTING
      E_COUNT = L_CNT.

  CHECK L_CNT > 0.

  CLEAR : GT_ZPMT0150, GT_ZPMT0150[].

  SELECT *
    INTO TABLE GT_ZPMT0150
    FROM ZPMT0150
   WHERE DOCTY = L_DOCTY
     AND DOCNO = L_DOCNO.

  G_SCR = '0100'.
  CASE L_DOCTY.
    WHEN 'E'.
      G_TITLE = '설비문서보기'.
    WHEN 'P'.
      G_TITLE = '예방계획문서보기'.
    WHEN 'R'.
      G_TITLE = '결과문서보기'.
  ENDCASE.

  CALL SCREEN 100  STARTING AT 20  5
                   ENDING   AT 120 23.


ENDFUNCTION.
