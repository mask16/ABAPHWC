function zpm_get_user_list.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BUKRS) TYPE BUKRS OPTIONAL
*" VALUE(I_WERKS) TYPE WERKS_D OPTIONAL
*" TABLES
*" T_DATA STRUCTURE ZPMS0002
*"----------------------------------------------------------------------
* Global data declarations

  data : lt_tab type zpms0002 occurs 0 with header line.
  ranges: r_bukrs for pa0001-bukrs,
          r_werks for pa0001-werks.

  clear: lt_tab, lt_tab[], r_bukrs, r_bukrs[], r_werks, r_werks[].


  if i_bukrs is not initial.
    r_bukrs = 'IEQ'. r_bukrs-low = i_bukrs. append r_bukrs.
  endif.

* if i_werks is not initial.
* r_werks = 'IEQ'. r_werks-low = i_werks. append r_werks.
* endif.

  select a~pernr a~sname a~orgeh b~orgtx
    into table lt_tab
    from pa0001 as a inner join
         t527x  as b
      on b~orgeh = a~orgeh
     and b~sprsl = sy-langu
     and b~endda = a~endda
   where a~bukrs in r_bukrs
     and a~werks in r_werks
     and a~endda = '99991231'.

  loop at lt_tab.
    move-corresponding lt_tab to t_data.
    concatenate t_data-sname '-' t_data-orgtx
           into t_data-snnsd.
    append t_data.
  endloop.

endfunction.
