FUNCTION ZPM_GET_MATNR_INFO.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_MATNR) TYPE MATNR OPTIONAL
*" VALUE(I_WERKS) TYPE WERKS_D OPTIONAL
*" VALUE(I_LGORT) TYPE LGORT_D OPTIONAL
*" VALUE(I_CHARG) TYPE CHARG_D OPTIONAL
*" VALUE(I_MTART) TYPE MTART DEFAULT 'ERSA'
*" VALUE(I_BUKRS) TYPE BUKRS DEFAULT '1000'
*" VALUE(I_RSNUM) TYPE RSNUM OPTIONAL
*" VALUE(I_WPLAN) TYPE ZPMT0140-WPLANN OPTIONAL
*" EXPORTING
*" VALUE(E_INFO) TYPE ZPMS0003
*" TABLES
*" T_DATA STRUCTURE ZPMS0003 OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA  : LS_DATA TYPE ZPMS0003,
          LT_DATA TYPE ZPMS0003 OCCURS 0 WITH HEADER LINE.
  DATA  : L_WAERS TYPE T001-WAERS,
          BEGIN OF LT_MATNR OCCURS 0,
            MATNR TYPE ZPMS0003-MATNR,
          END   OF LT_MATNR,
          L_RSNUM TYPE RESB-RSNUM,
          BEGIN OF LT_RESB OCCURS 0,
            MATNR LIKE RESB-MATNR,
            CHARG LIKE RESB-CHARG,
            BDMNG LIKE RESB-BDMNG,
          END   OF LT_RESB,
          L_WPLAN TYPE ZPMT0140-WPLANN.

  RANGES: R_MATNR FOR ZPMS0003-MATNR, R_WERKS FOR ZPMS0003-WERKS,
          R_LGORT FOR ZPMS0003-LGORT, R_CHARG FOR ZPMS0003-CHARG,
          R_MTART FOR MARA-MTART.

  CLEAR : LS_DATA, LT_DATA, LT_DATA[], L_WAERS, LT_MATNR, LT_MATNR[],
          R_MATNR, R_MATNR[], R_WERKS, R_WERKS[],
          R_LGORT, R_LGORT[], R_CHARG, R_CHARG[],
          R_MTART, R_MTART[].

  SELECT SINGLE WAERS
    INTO L_WAERS
    FROM T001
   WHERE BUKRS = I_BUKRS.

  L_RSNUM = I_RSNUM.
  L_WPLAN = I_WPLAN.

  IF I_MATNR IS NOT INITIAL.
    R_MATNR = 'IEQ'. R_MATNR-LOW = I_MATNR. APPEND R_MATNR.
  ENDIF.

  IF I_WERKS IS NOT INITIAL.
    R_WERKS = 'IEQ'. R_WERKS-LOW = I_WERKS. APPEND R_WERKS.
  ENDIF.

  IF I_LGORT IS NOT INITIAL.
    R_LGORT = 'IEQ'. R_LGORT-LOW = I_LGORT. APPEND R_LGORT.
  ENDIF.

  IF I_CHARG IS NOT INITIAL.
    R_CHARG = 'IEQ'. R_CHARG-LOW = I_CHARG. APPEND R_CHARG.
  ENDIF.

  IF I_MTART IS NOT INITIAL.
    R_MTART = 'IEQ'. R_MTART-LOW = I_MTART. APPEND R_MTART.
    IF I_MATNR IS NOT INITIAL.
      CLEAR : R_MTART, R_MTART[].
    ENDIF.
  ENDIF.

  SELECT A~MATNR E~MAKTX A~WERKS B~NAME1 A~CHARG A~LGORT C~LGOBE
         A~CLABS D~MEINS F~VERPR F~PEINH D~BISMT
    INTO CORRESPONDING FIELDS OF TABLE LT_DATA
    FROM MCHB AS A INNER JOIN
         T001W AS B
      ON B~WERKS = A~WERKS INNER JOIN
         T001L AS C
      ON C~LGORT = A~LGORT
     AND C~WERKS = A~WERKS INNER JOIN
         MARA AS D
      ON D~MATNR = A~MATNR INNER JOIN
         MAKT AS E
      ON E~MATNR = A~MATNR
     AND E~SPRAS = SY-LANGU INNER JOIN
         MBEW AS F
      ON F~MATNR = A~MATNR
     and f~BWKEY = a~werks
     AND F~BWTAR = A~CHARG inner join
         MARD as g
      on a~MATNR = g~matnr
     and a~WERKS = g~werks
     and a~LGORT = g~lgort
     and g~LVORM = ''
   WHERE A~MATNR IN R_MATNR
     AND A~WERKS IN R_WERKS
     AND A~LGORT IN R_LGORT
     AND A~CHARG IN R_CHARG
     AND D~MTART IN R_MTART.

  SORT LT_DATA BY WERKS MATNR CHARG LGORT.
  DELETE ADJACENT DUPLICATES FROM LT_DATA.

  IF L_WPLAN IS NOT INITIAL.
    CLEAR : LT_RESB, LT_RESB[].
    SELECT A~MATNR A~CHARG A~BDMNG
      INTO TABLE LT_RESB
      FROM RESB     AS A INNER JOIN
           ZPMT0090 AS B
        ON A~RSNUM  =  B~RSNUM
     WHERE B~WPLANN =  L_WPLAN
       AND B~RSNUM  <> ''
       AND A~MATNR    IN R_MATNR
       AND A~CHARG    IN R_CHARG.
  ENDIF.

  IF L_RSNUM IS NOT INITIAL.
    CLEAR : LT_RESB, LT_RESB[].
    SELECT MATNR CHARG BDMNG
      INTO TABLE LT_RESB
      FROM RESB
     WHERE RSNUM =  L_RSNUM
       AND MATNR IN R_MATNR
       AND CHARG IN R_CHARG.
  ENDIF.

  LOOP AT LT_DATA.
    LT_MATNR-MATNR = LT_DATA-MATNR.
    APPEND LT_MATNR.
    LT_DATA-WAERS = L_WAERS.
    MODIFY LT_DATA.
  ENDLOOP.

  CLEAR : XBDART, XBDART[].
  XBDART-SIGN   = 'I'.
  XBDART-OPTION = 'EQ'.
  XBDART-LOW    = 'AR'.
  APPEND XBDART.
  XBDART-LOW    = 'MR'.
  APPEND XBDART.

  LOOP AT LT_MATNR.

    CLEAR : XTAB1, XTAB1[].

    CALL FUNCTION 'MB_ADD_RESERVATION_QUANTITIES'
      EXPORTING
        X_MATNR = LT_MATNR-MATNR
        X_XLOEK = SPACE
        X_KZEAR = SPACE
        X_KZWSO = KZWSO          "ALRK014884 SW
      TABLES
        XTAB1   = XTAB1
        XWERKS  = MS_WERKS
        XBDART  = XBDART.
    LOOP AT XTAB1.
      LT_DATA-BDMNG = XTAB1-BDMNG.
      MODIFY LT_DATA TRANSPORTING BDMNG
                     WHERE MATNR = XTAB1-MATNR
                       AND WERKS = XTAB1-WERKS
                       AND LGORT = XTAB1-LGORT
                       AND CHARG = XTAB1-CHARG.
    ENDLOOP.
  ENDLOOP.

  LOOP AT LT_DATA.
    LT_DATA-QUANT = LT_DATA-CLABS - LT_DATA-BDMNG.
    READ TABLE LT_RESB WITH KEY MATNR = LT_DATA-MATNR
                                CHARG = LT_DATA-CHARG.
    IF SY-SUBRC = 0.
      LT_DATA-QUANT = LT_DATA-QUANT + LT_RESB-BDMNG.
    ENDIF.
    MODIFY LT_DATA.
  ENDLOOP.

  READ TABLE LT_DATA INTO LS_DATA INDEX 1.
  IF SY-SUBRC = 0.
    E_INFO = LS_DATA.
  ENDIF.

  T_DATA[] = LT_DATA[].

ENDFUNCTION.
