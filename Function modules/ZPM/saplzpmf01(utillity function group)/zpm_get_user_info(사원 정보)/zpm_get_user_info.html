function ZPM_GET_USER_INFO.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_PERNR) TYPE PA0001-PERNR
*" EXPORTING
*" REFERENCE(E_INFO) TYPE ZPMS0002
*"----------------------------------------------------------------------
* Global data declarations

  data : ls_tab type zpms0002 occurs 0 with header line.
  data : l_pernr type pa0001-pernr.

  clear: ls_tab, l_pernr, e_info.

  l_pernr = i_pernr.

  select single a~pernr a~sname a~orgeh b~orgtx
    into ls_tab
    from pa0001 as a inner join
         t527x  as b
      on b~orgeh = a~orgeh
     and b~sprsl = sy-langu
     and b~endda = a~endda
   where a~pernr = l_pernr
     and a~endda = '99991231'.

  if sy-subrc = 0.
    move-corresponding ls_tab to e_info.
    concatenate e_info-sname '-' e_info-orgtx
           into e_info-snnsd.
  endif.

endfunction.
