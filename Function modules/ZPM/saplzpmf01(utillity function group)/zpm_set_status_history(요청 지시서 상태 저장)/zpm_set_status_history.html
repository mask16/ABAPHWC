*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_QMNUM) TYPE ZEQMNUM OPTIONAL
*" VALUE(I_WPLANN) TYPE ZEWPLANN OPTIONAL
*" REFERENCE(I_STATUS) TYPE CHAR1
*" REFERENCE(I_VALUE) TYPE CHAR1 DEFAULT 'X'
*"----------------------------------------------------------------------
* Global data declarations

  data : l_qmnum  type zpmt0040-qmnum,
         l_wplann type zpmt0040-wplann,
         l_status type CHAR1,
         l_value  type char1.

  clear: l_qmnum, l_wplann.

  l_qmnum  = I_QMNUM.
  l_wplann = I_WPLANN.
  l_status = I_status.
  l_value  = I_VALUE.

  check l_qmnum is not initial or l_wplann is not initial.
  check l_status is not initial.

  perform set_status_history using 'ZPMT1040' l_qmnum  l_status l_value.
  perform set_status_history using 'ZPMT1050' l_wplann l_status l_value.

ENDFUNCTION.
