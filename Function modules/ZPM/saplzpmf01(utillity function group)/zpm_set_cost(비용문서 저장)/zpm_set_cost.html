FUNCTION ZPM_SET_COST.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_WPLANN) TYPE ZPMT0170-WPLANN
*" REFERENCE(I_WPITEM) TYPE ZPMT0170-WPITEM
*" REFERENCE(I_COSTCAT) TYPE ZPMT0170-COSTCAT
*" REFERENCE(I_DOCTY) TYPE ZPMT0170-DOCTY
*" REFERENCE(I_DOCYR) TYPE ZPMT0170-DOCYR
*" REFERENCE(I_DOCNO) TYPE ZPMT0170-DOCNO
*" VALUE(I_DOCIT) OPTIONAL
*" VALUE(I_RESULTN) TYPE ZPMT0170-RESULTN OPTIONAL
*" VALUE(I_RSNUM) OPTIONAL
*" VALUE(I_CANCEL) TYPE C OPTIONAL
*" EXPORTING
*" VALUE(ES_COST) TYPE ZPMT0170
*"----------------------------------------------------------------------
* Global data declarations

  DATA  : LS_PMT0170 TYPE ZPMT0170.
  RANGES: R_RSNUM FOR MSEG-RSNUM.

  CLEAR : LS_PMT0170, R_RSNUM, R_RSNUM[].

  IF I_RSNUM IS NOT INITIAL.
    R_RSNUM = 'IEQ'. R_RSNUM-LOW = I_RSNUM. APPEND R_RSNUM.
  ENDIF.

  MOVE :
    I_WPLANN  TO LS_PMT0170-WPLANN,
    I_WPITEM  TO LS_PMT0170-WPITEM,
    I_COSTCAT TO LS_PMT0170-COSTCAT,
    I_DOCTY   TO LS_PMT0170-DOCTY,
    I_DOCYR   TO LS_PMT0170-DOCYR,
    I_DOCNO   TO LS_PMT0170-DOCNO,
    I_DOCIT   TO LS_PMT0170-DOCIT,
    I_RESULTN TO LS_PMT0170-RESULTN.

  IF LS_PMT0170-DOCIT IS INITIAL.
    LS_PMT0170-DOCIT = 010.
  ENDIF.

  LS_PMT0170-AENAM = SY-UNAME.
  LS_PMT0170-AEDAT = SY-DATUM.
  LS_PMT0170-AEZET = SY-UZEIT.
  call function 'ZPM_GET_EMPNO'
    IMPORTING
      E_PERNR = LS_PMT0170-AEPSN.
  LS_PMT0170-ERNAM = LS_PMT0170-AENAM.
  LS_PMT0170-ERDAT = LS_PMT0170-AEDAT.
  LS_PMT0170-ERZET = LS_PMT0170-AEZET.
  LS_PMT0170-ERPSN = LS_PMT0170-AEPSN.

  SELECT SINGLE MAX( COSTSEQ )
    INTO LS_PMT0170-COSTSEQ
    FROM ZPMT0170
   WHERE WPLANN  = LS_PMT0170-WPLANN
     AND WPITEM  = LS_PMT0170-WPITEM
     AND COSTCAT = LS_PMT0170-COSTCAT.

  IF LS_PMT0170-COSTSEQ IS INITIAL.
    LS_PMT0170-COSTSEQ = '010'.
  ELSE.
    LS_PMT0170-COSTSEQ = LS_PMT0170-COSTSEQ + 10.
  ENDIF.

  CASE LS_PMT0170-DOCTY.
    WHEN 'MSEG'.

      IF LS_PMT0170-COSTCAT = 'MAEC'.
        SELECT SINGLE BUKRS WERKS WAERS SUM( DMBTR )
          INTO (LS_PMT0170-BUKRS, LS_PMT0170-WERKS, LS_PMT0170-WAERS, LS_PMT0170-PRICE)
          FROM MSEG
         WHERE MBLNR = LS_PMT0170-DOCNO
           AND MJAHR = LS_PMT0170-DOCYR
           AND RSNUM IN R_RSNUM
         GROUP BY BUKRS WERKS WAERS.
      ELSE.
        SELECT SINGLE BUKRS WERKS WAERS DMBTR
          INTO (LS_PMT0170-BUKRS, LS_PMT0170-WERKS, LS_PMT0170-WAERS, LS_PMT0170-PRICE)
          FROM MSEG
         WHERE MBLNR = LS_PMT0170-DOCNO
           AND MJAHR = LS_PMT0170-DOCYR
           AND BUZEI = LS_PMT0170-DOCIT.
      ENDIF.

      SELECT SINGLE BUDAT
        INTO (LS_PMT0170-BUDAT)
        FROM MKPF
       WHERE MBLNR = LS_PMT0170-DOCNO
         AND MJAHR = LS_PMT0170-DOCYR.

    WHEN 'RSEG'.
      SELECT SINGLE B~BUKRS B~WERKS A~WAERS A~BUDAT B~WRBTR
        INTO (LS_PMT0170-BUKRS, LS_PMT0170-WERKS, LS_PMT0170-WAERS, LS_PMT0170-BUDAT, LS_PMT0170-PRICE)
        FROM RBKP AS A INNER JOIN RSEG AS B
          ON A~BELNR = B~BELNR
         AND A~GJAHR = B~GJAHR
       WHERE A~BELNR = LS_PMT0170-DOCNO
         AND A~GJAHR = LS_PMT0170-DOCYR
         AND B~BUZEI = LS_PMT0170-DOCIT.
  ENDCASE.

  IF I_CANCEL = 'X'.
    LS_PMT0170-SHKZG = 'S'.
  ELSE.
    LS_PMT0170-SHKZG = 'H'.
  ENDIF.

  MODIFY ZPMT0170 FROM LS_PMT0170.

  ES_COST = LS_PMT0170.

ENDFUNCTION.
