    LS_FCAT-REF_TABLE = LT_DYNTAB-TABNAME.
    APPEND LS_FCAT TO LT_FCAT.
  ENDLOOP.

  CALL METHOD CL_ALV_TABLE_CREATE=>CREATE_DYNAMIC_TABLE
    EXPORTING
      IT_FIELDCATALOG = LT_FCAT
    IMPORTING
      EP_TABLE        = D_REF.
  ASSIGN D_REF->* TO <FS_TAB>.

  SELECT *
    INTO TABLE <FS_TAB>
    FROM (L_TNAME)
   WHERE (L_WHERE) .

  IF SY-SUBRC = 0.
    LOOP AT <FS_TAB> ASSIGNING <FS_WA>.
      LOOP AT LT_DYNTAB.
        IF LT_DYNTAB-FIELDNAME+0(3) = 'ST_'.
          DATA : L_NAME TYPE STRING.
          CLEAR: L_NAME.
          CONCATENATE '<fs_wa>-' LT_DYNTAB-FIELDNAME INTO L_NAME.

          ASSIGN (L_NAME) TO <FS_FNAME>.
          IF <FS_FNAME> = 'X'.
            IF L_STR IS INITIAL.
              L_STR = LT_DYNTAB-FIELDNAME+3(1).
            ELSE.
              CONCATENATE L_STR LT_DYNTAB-FIELDNAME+3(1)
                     INTO L_STR
                SEPARATED BY SPACE.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDLOOP.
      EXIT.

  ENDIF.
