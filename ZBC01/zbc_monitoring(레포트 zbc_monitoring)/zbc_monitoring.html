  SELECT A~BNAME A~KOSTL B~NAME_FIRST B~NAME_LAST C~DEPARTMENT
    INTO CORRESPONDING FIELDS OF TABLE IT_ADRP
    FROM USR21 AS A
   INNER JOIN ADRP AS B ON B~CLIENT = A~MANDT AND
                           B~PERSNUMBER = A~PERSNUMBER
   INNER JOIN ADCP AS C ON C~CLIENT = A~MANDT AND
                           C~PERSNUMBER = A~PERSNUMBER AND
                           C~ADDRNUMBER = A~ADDRNUMBER
   WHERE B~DATE_FROM <= SY-DATUM
     AND B~NATION = SPACE
     AND C~DATE_FROM <= SY-DATUM
     AND C~NATION = SPACE.

  SORT IT_ADRP.
  LOOP AT IT_ADRP.
    SELECT SINGLE TRDAT LTIME USTYP ERDAT UFLAG
      INTO (IT_ADRP-TRDAT, IT_ADRP-LTIME, IT_ADRP-USTYP, IT_ADRP-ERDAT,
IT_ADRP-UFLAG)

  IF SY-SUBRC NE 0.

FORM MAKE_FIELDCAT USING P_GUB P_FNAME P_CON.

  IF P_GUB = 'S'.
    CLEAR GS_FIELDCAT_ITEM.
  ENDIF.

  DATA LV_COL(40).
  FIELD-SYMBOLS <FS>.
  CONCATENATE 'GS_FIELDCAT_ITEM-' P_FNAME  INTO LV_COL.
  ASSIGN      (LV_COL)       TO       <FS>.
  MOVE         P_CON         TO       <FS>.

  IF P_GUB = 'E'.
    APPEND GS_FIELDCAT_ITEM TO <FCAT>.
  ENDIF.

ENDFORM. " MAKE_FIELDCAT
