REPORT zbcr0010.

DATA: gt_source TYPE TABLE OF usr41_mld,
      gs_source TYPE usr41_mld,
      gt_target TYPE TABLE OF zbct0002,
      gs_target TYPE zbct0002,
      gt_update TYPE TABLE OF usr41_mld.


START-OF-SELECTION.
  CLEAR: gt_source, gt_target, gt_update.

  SELECT *
  INTO TABLE gt_source
  FROM usr41_mld
  WHERE cal_year = sy-datum(4).

  LOOP AT gt_source INTO gs_source.
    MOVE-CORRESPONDING gs_source TO gs_target.
    gs_target-cal_day = sy-datum.

    "PEAK, COUNT CHECK & INITIALIZE
    IF gs_source-counter  GT 800.
      APPEND gs_source TO gt_update.
      gs_target-clear_date = sy-datum.
    ENDIF.

    APPEND gs_target TO gt_target.
    CLEAR gs_target.
  ENDLOOP.

  IF gt_target[] IS NOT INITIAL.
    MODIFY zbct0002 FROM TABLE gt_target.
    IF sy-subrc = 0.
      COMMIT WORK.
    ELSE.
      ROLLBACK WORK.
    ENDIF.
  ENDIF.
