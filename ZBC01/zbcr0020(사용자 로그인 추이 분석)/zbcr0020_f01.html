FORM get_select .
  CLEAR gt_data.

  SELECT *
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM usr41_mld
  WHERE cal_year = p_datum(4)
    AND bname IN s_bname.

  SORT gt_data BY counter.

  DATA: yest_cnt LIKE gt_data OCCURS 0,
        yesterday TYPE datum.

  yesterday = p_datum - 1.

  CLEAR yest_cnt.
  SELECT *
  INTO CORRESPONDING FIELDS OF TABLE yest_cnt
  FROM zbct0002
  WHERE cal_day = yesterday
    AND bname IN s_bname.

  LOOP AT yest_cnt INTO gs_data WHERE clear_date IS NOT INITIAL.
    READ TABLE gt_data WITH KEY bname = gs_data-bname
    TRANSPORTING NO FIELDS.

    IF sy-subrc <> 0.
      CLEAR: gs_data-counter, gs_data-peak.
      APPEND gs_data TO gt_data.
    ENDIF.
  ENDLOOP.


  SORT yest_cnt BY bname.
  LOOP AT gt_data ASSIGNING <fs01>.
    "전일 비교 증가
    READ TABLE yest_cnt WITH KEY bname = <fs01>-bname
      BINARY SEARCH INTO gs_data.
    IF sy-subrc = 0.
      <fs01>-add_cnt = <fs01>-counter - gs_data-counter.
      <fs01>-add_peak = <fs01>-peak - gs_data-peak.
      <fs01>-clear_date = gs_data-clear_date.
    ENDIF.

    "사용자명
    SELECT SINGLE department name_first
      INTO (<fs01>-department, <fs01>-name_first)
      FROM user_addr
      WHERE bname = <fs01>-bname.

    "COUNTER 색상 설정
    col-fname = 'COUNTER'.
    IF <fs01>-counter GE 200
      AND <fs01>-counter LT 800.
      col-color-col  = 3.
    ELSEIF <fs01>-counter GE 800.
      col-color-col  = 6.
    ELSEIF <fs01>-clear_date IS NOT INITIAL.
      col-color-col  = 5.
    ENDIF.
    APPEND col TO <fs01>-tabcol.
    CLEAR col.

    "COUNTER 색상 설정
    col-fname = 'ADD_CNT'.
    IF <fs01>-add_cnt GT 0.
      col-color-col  = 6.
      col-color-inv = 1.
    ELSEIF <fs01>-add_cnt LT 0.
      col-color-col  = 5.
    ENDIF.
    APPEND col TO <fs01>-tabcol.
    CLEAR col.
  ENDLOOP.

ENDFORM. " GET_SELECT
