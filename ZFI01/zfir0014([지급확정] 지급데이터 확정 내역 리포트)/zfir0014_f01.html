FORM set_main_data .

  CLEAR: gt_main, gt_main[].

  SELECT *
     INTO CORRESPONDING FIELDS OF gt_main
     FROM bsik
    WHERE bukrs EQ p_bukrs
      AND gjahr EQ p_gjahr
      AND budat <= p_sdate
      AND bldat IN s_bldat
      AND budat IN s_budat
      AND lifnr IN s_lifnr
      AND blart IN gr_blart
      AND bupla IN s_bupla
      AND umskz EQ space.

    SELECT SINGLE
           name1
           ktokk
      INTO (gt_main-name1,
            gt_main-ktokk)
      FROM lfa1
     WHERE lifnr = gt_main-lifnr.

    IF gt_main-ktokk IN s_ktokk.
      gt_main-iconn = '@5F@'.

*-- 계정명
      SELECT SINGLE
             txt20
        FROM skat
        INTO gt_main-txt20
       WHERE saknr = gt_main-hkont
         AND spras = sy-langu.
*-- 순만기일
      PERFORM net_due_date_get CHANGING gt_main.

      IF gt_main-rebzg IS INITIAL.
        gt_main-rebzg = gt_main-belnr.
        gt_main-rebzj = gt_main-gjahr.
        gt_main-rebzz = gt_main-buzei.
      ENDIF.
*-- 사업장명
      SELECT SINGLE
             name
        INTO gt_main-name
        FROM j_1bbranch
       WHERE bukrs  = gt_main-bukrs
         AND branch = gt_main-bupla.
*-- 지급보류키 텍스트
      SELECT SINGLE
             textl
        INTO gt_main-textl
        FROM t008t
       WHERE spras = sy-langu
         AND zahls = gt_main-zlspr.

      IF gt_main-shkzg = 'H'.
        gt_main-dmbtr = gt_main-dmbtr * -1.
      ENDIF.

      PERFORM set_field_color USING: 4 1 1 'ZLSPR', "지급보류키
                                     4 1 1 'ZLSCH'. "지급방법

      APPEND gt_main.
    ENDIF.

    CLEAR: gt_main.
  ENDSELECT.

  SORT gt_main BY lifnr name1 gjahr bldat budat.

ENDFORM. " SET_MAIN_DATA
