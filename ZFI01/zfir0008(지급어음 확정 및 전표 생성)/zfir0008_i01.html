MODULE user_command_0102 INPUT.

  DATA: ls_head2 LIKE LINE OF gt_head2.

  CASE sy-ucomm.

    WHEN 'BACK' OR 'EXIT' OR 'CANC' OR 'QUIT'.
      LEAVE TO SCREEN 0.

    WHEN 'EXEC'.
      LOOP AT gt_sqnum.
        READ TABLE gt_head2 INTO ls_head2 WITH KEY sqnum = gt_sqnum-sqnum.
        ls_head2-ndate = gs0102-ndate.
        ls_head2-zfbdt = ls_head2-ncdat + ls_head2-ndate.
        IF ls_head2-ndate < 0.
          MESSAGE i018 DISPLAY LIKE 'E'. "어음일수는 0이상의 값만 입력 가능합니다.
          gv_err = 'X'.
          EXIT.
        ELSE.
          MODIFY gt_head2 FROM ls_head2 TRANSPORTING ndate zfbdt
                         WHERE sqnum = gt_sqnum-sqnum.
        ENDIF.
        CLEAR: ls_head2.
      ENDLOOP.
      LEAVE TO SCREEN 0.

    WHEN OTHERS.

  ENDCASE.
