*----------------------------------------------------------------------*
FORM GET_DATA .

  DATA: LV_DDTXT LIKE DD07V-DDTEXT.

  CLEAR: LV_DDTXT, GT_ITAB, GS_LINE.

  SELECT
         A~SYSID
         B~IFTYPE
         A~MAN_SEQ
         A~IFNAME
         A~ERNAM
         A~ERDAT
         A~ERZET
         A~MSGTY
         A~MSGTX
         B~TABLE1
         B~TABLE2
         B~TABLE3
  INTO CORRESPONDING FIELDS OF TABLE GT_ITAB
  FROM ZFIT0026 AS A INNER JOIN ZFIT0038 AS B  ON A~BUKRS = B~BUKRS
                                              AND A~SYSID = B~SYSID
                                              AND A~IFNAME = B~IFNAME
  WHERE A~BUKRS  = P_BUKRS
    AND A~SYSID  IN S_SYSID
    AND A~IFNAME IN S_IFNAME
    AND B~IFTYPE IN S_IFTYPE
    AND A~ERDAT  IN S_ERDAT.

  IF GT_ITAB IS INITIAL.

    MESSAGE S000 DISPLAY LIKE 'E' WITH TEXT-005.
    LEAVE LIST-PROCESSING.

  ENDIF.

  SORT GT_ITAB BY MAN_SEQ DESCENDING.
  LOOP AT GT_ITAB INTO GS_LINE.

    IF GS_LINE-MSGTY = 'S'.
      GS_LINE-ICON = ICON_GREEN_LIGHT.

* BIZPLAY 요청 : 데이터 없는 경우 MESGTY = 'S' 송부 건 처리
      IF  GS_LINE-MSGTX = '0건 조회 되었습니다.' OR
          GS_LINE-MSGTX = '조회된 데이터가 없습니다.'.
        CLEAR: GS_LINE-TABLE2.
      ENDIF.
    ELSE.
      CLEAR: GS_LINE-TABLE2, GS_LINE-TABLE3.
      GS_LINE-ICON = ICON_RED_LIGHT.

      "MSGTY = 'E' 일 경우 MAN_SEQ로 테이블 점검 후 테이블 CLEAR
      CASE GS_LINE-IFNAME.
        WHEN 'ZFI_BIZ_VENDOR_D'.
          SELECT SINGLE * FROM ZFIT0035
                  WHERE MAN_SEQ = GS_LINE-MAN_SEQ
                    AND BUKRS   = P_BUKRS.

          IF SY-SUBRC = 0.
            GS_LINE-TABLE2 = 'ZFIT0035'.
          ENDIF.

        WHEN 'ZFI_BIZ_RESULT_DOC'.
          SELECT SINGLE * FROM ZFIT0033
                  WHERE MAN_SEQ = GS_LINE-MAN_SEQ
                    AND BUKRS   = P_BUKRS.

          IF SY-SUBRC = 0.
            GS_LINE-TABLE2 = 'ZFIT0033'.
          ENDIF.

        WHEN 'ZFI_BIZ_STATUS'.
          SELECT SINGLE * FROM ZFIT0029
                  WHERE MAN_SEQ = GS_LINE-MAN_SEQ
                    AND BUKRS   = P_BUKRS.

          IF SY-SUBRC = 0.
            GS_LINE-TABLE2 = 'ZFIT0029'.
          ENDIF.

      ENDCASE.

    ENDIF.

    PERFORM READ_DOMAIN_TEXT USING 'ZDSYSID' GS_LINE-SYSID CHANGING LV_DDTXT.
    GS_LINE-SYTEXT = LV_DDTXT.
    PERFORM FILL_RESTFLD.

    MODIFY GT_ITAB FROM GS_LINE.

    CLEAR: GS_LINE.

  ENDLOOP.

ENDFORM. " GET_DATA

*&      Form  ON_HOTSPOT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_E_ROW_ID  text
*      -->P_E_COLUMN_ID  text
*      -->P_ES_ROW_NO  text
*----------------------------------------------------------------------*
FORM ON_HOTSPOT USING P_E_ROW_ID TYPE LVC_S_ROW
                          P_E_COLUMN_ID
                          P_ES_ROW_NO.

  CASE SY-DYNNR.
    WHEN '0100'.
      CASE P_E_COLUMN_ID.
        WHEN 'TABLE1' OR 'TABLE2' OR 'TABLE3'.
          READ TABLE GT_ITAB INTO GS_LINE INDEX P_E_ROW_ID-INDEX.
          CHECK SY-SUBRC = 0.
          DATA: LV_TABLE_NAME(40).
          IF P_E_COLUMN_ID = 'TABLE1'.
            LV_TABLE_NAME = GS_LINE-TABLE1.
          ELSEIF P_E_COLUMN_ID = 'TABLE2'.
            LV_TABLE_NAME = GS_LINE-TABLE2.
          ELSEIF P_E_COLUMN_ID = 'TABLE3'.
            LV_TABLE_NAME = GS_LINE-TABLE3.
          ENDIF.

          CHECK  LV_TABLE_NAME IS NOT INITIAL.

          DATA: LO_TABLE TYPE REF TO DATA.
          CLEAR: LO_TABLE.
          TRY.
              CREATE DATA LO_TABLE TYPE TABLE OF (LV_TABLE_NAME).
            CATCH CX_ROOT.
              MESSAGE S000 WITH '해당 테이블이 존재하지 않습니다.' DISPLAY LIKE 'E'.
              RETURN.
          ENDTRY.

          FIELD-SYMBOLS <LF_TABLE> TYPE ANY TABLE.
          ASSIGN LO_TABLE->* TO <LF_TABLE>.

          TRY.
              SELECT *
                 INTO CORRESPONDING FIELDS OF TABLE <LF_TABLE>
                 FROM (LV_TABLE_NAME)
                WHERE   BUKRS = P_BUKRS
                  AND MAN_SEQ = GS_LINE-MAN_SEQ.
            CATCH CX_ROOT.
              MESSAGE S000 WITH '필드 리스트 입력이 잘못 되었습니다.' DISPLAY LIKE 'E'.
              RETURN.
          ENDTRY.

          IF <LF_TABLE> IS INITIAL.
            MESSAGE S000 WITH '데이터가 없습니다.' DISPLAY LIKE 'E'.

          ELSE.

            DATA : LO_SALVTAB TYPE REF TO CL_SALV_TABLE,
                   LR_FUNCTIONS TYPE REF TO CL_SALV_FUNCTIONS.

            DATA : LR_COLUMNS TYPE REF TO CL_SALV_COLUMNS_TABLE,
                   LR_COLUMN TYPE REF TO CL_SALV_COLUMN_TABLE.

            CALL METHOD CL_SALV_TABLE=>FACTORY
              IMPORTING
                R_SALV_TABLE = LO_SALVTAB
              CHANGING
                T_TABLE      = <LF_TABLE>.

            LR_FUNCTIONS = LO_SALVTAB->GET_FUNCTIONS( ).
            LR_FUNCTIONS->SET_ALL( IF_SALV_C_BOOL_SAP=>TRUE ).

            LR_COLUMNS = LO_SALVTAB->GET_COLUMNS( ).
            LR_COLUMNS->SET_OPTIMIZE( ABAP_TRUE ).

            CALL METHOD LO_SALVTAB->DISPLAY.

            FREE: <LF_TABLE>.
          ENDIF.
      ENDCASE.
  ENDCASE.

ENDFORM. " ON_HOTSPOT
