*       AND spras EQ p_spras.

*    SELECT SINGLE *

*       AND bukrs EQ p_bukrs.
*    IF sy-subrc = 0.

*      gt_main-hbkid = skb1-hbkid.
*      gt_main-hktid = skb1-hktid.
*    ENDIF.
*
*    APPEND gt_main.
*    CLEAR: skb1, gt_main.
*
*  ENDSELECT.
*  SELECT a~saknr a~waers a~xsalh a~bewgp a~mwskz
*         a~xmwno a~mitkz a~xopvw a~xkres a~altkt
*         a~zuawa a~fstag a~xintb a~xnkon a~xmitk
*         a~fdlev a~xgkon a~fipos a~hbkid a~hktid
*         b~txt20 b~txt50
*         c~ktoks c~xbilk
*    INTO CORRESPONDING FIELDS OF TABLE gt_main
*   FROM skb1 AS a INNER JOIN
*        skat AS b
*     ON b~spras =  p_spras
*    AND b~saknr =  a~saknr
*    AND b~KTOPL =  '1000' INNER JOIN
*        ska1 AS c
*     ON c~saknr =  a~saknr
*    AND c~ktopl =  p_ktopl
*  WHERE a~bukrs =  p_bukrs
*    AND a~saknr IN s_saknr.
  IF s_bukrs[] IS INITIAL.
    SELECT c~saknr
           b~txt20 b~txt50
           c~ktoks c~xbilk
      INTO CORRESPONDING FIELDS OF TABLE gt_main
     FROM ska1 AS c INNER JOIN
          skat AS b
       ON b~spras =  p_spras
      AND b~ktopl =  p_ktopl
      AND b~saknr =  c~saknr
    WHERE c~saknr IN s_saknr
      AND c~ktopl =  p_ktopl.

  ELSE.
    SELECT a~saknr a~waers a~xsalh a~bewgp a~mwskz
           a~xmwno a~mitkz a~xopvw a~xkres a~altkt
           a~zuawa a~fstag a~xintb a~xnkon a~xmitk
           a~fdlev a~xgkon a~fipos a~hbkid a~hktid
           A~ERDAT
           b~txt20 b~txt50
           c~ktoks c~xbilk
      INTO CORRESPONDING FIELDS OF TABLE gt_main
     FROM skb1 AS a INNER JOIN
          skat AS b
       ON b~spras =  p_spras
      AND b~ktopl =  p_ktopl
      AND b~saknr =  a~saknr INNER JOIN
          ska1 AS c
       ON c~saknr =  a~saknr
      AND c~ktopl =  p_ktopl
    WHERE a~bukrs IN s_bukrs
      AND a~saknr IN s_saknr.

  ENDIF.

  DATA : lt_altkt LIKE skat OCCURS 0 WITH HEADER LINE.

  IF '2000' IN s_bukrs AND gt_main[] IS NOT INITIAL.
    CLEAR: lt_altkt, lt_altkt[].
    SELECT * INTO TABLE lt_altkt
      FROM skat
       FOR ALL ENTRIES IN gt_main
     WHERE spras = 1
       AND ktopl = '2000'
       AND saknr = gt_main-altkt.
    SORT lt_altkt BY saknr.
    LOOP AT gt_main.
      READ TABLE lt_altkt WITH KEY saknr = gt_main-altkt.
      IF sy-subrc = 0.
        gt_main-altkt_tx = lt_altkt-txt20.
        MODIFY gt_main.
      ENDIF.
    ENDLOOP.
  ENDIF.

IF '3000' IN s_bukrs AND gt_main[] IS NOT INITIAL.
    CLEAR: lt_altkt, lt_altkt[].
    SELECT * INTO TABLE lt_altkt
      FROM skat
       FOR ALL ENTRIES IN gt_main
     WHERE spras = 'E'
       AND ktopl = '3000'
       AND saknr = gt_main-altkt.
    SORT lt_altkt BY saknr.
    LOOP AT gt_main.
      READ TABLE lt_altkt WITH KEY saknr = gt_main-altkt.
      IF sy-subrc = 0.
        gt_main-altkt_tx = lt_altkt-txt20.
        MODIFY gt_main.
      ENDIF.
    ENDLOOP.
  ENDIF.
