FORM SELECT_MAIN_DATA .

  DATA : BEGIN OF LT_BKPF OCCURS 0,
         BUKRS       TYPE BUKRS,
         GJAHR       TYPE GJAHR,
         BELNR       TYPE BELNR_D,
         BUDAT       TYPE BUDAT,
         BLDAT       TYPE BLDAT,
         BLART       TYPE BLART,
         XBLNR       TYPE XBLNR1,
         WAERS       TYPE WAERS,
         MONAT       TYPE MONAT,
         STBLG       TYPE STBLG.
  DATA :  END OF LT_BKPF.

  DATA : LV_TABIX TYPE SY-TABIX.

* Select 전표유형
  PERFORM GET_BLART.

* 전표헤더
  SELECT BUKRS BLDAT BLART MONAT GJAHR BELNR BUDAT WAERS STBLG XBLNR
        INTO CORRESPONDING FIELDS OF TABLE LT_BKPF
         FROM BKPF
         WHERE BUKRS  = PA_BUKRS AND
               GJAHR  = PA_GJAHR AND
               BELNR IN SO_BELNR AND
*               BLART IN SO_BLART AND
               BLART IN SO_BLARX AND
               BUDAT IN SO_BUDAT AND
               BLDAT IN SO_BLDAT AND
*               MONAT IN SO_MONAT AND
*               SHKZG IN SO_SHKZG AND
               STBLG = ''.

*         WHERE BLART IN SO_BLART AND
*               BUKRS  = PA_BUKRS AND
*               GJAHR  = PA_GJAHR AND
**               BELNR IN SO_BELNR AND
*               BLART IN SO_BLART AND
*               BLART IN SO_BLARX AND
*               BUDAT IN SO_BUDAT AND
*               BLDAT IN SO_BLDAT AND
*               MONAT IN SO_MONAT AND
*               STBLG = ''.

  IF LT_BKPF[] IS INITIAL.
    MESSAGE S000 WITH TEXT-M01.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SORT LT_BKPF BY BUKRS GJAHR BELNR.
  CLEAR : GT_MAIN[], GT_MAIN.


  SELECT BUKRS
*         BLDAT
*         BUDAT
*         BLART
          GJAHR
*         MONAT
          BELNR
          LIFNR
*         WAERS
          WRBTR
          SHKZG
          DMBTR
          BUPLA
          HKONT
          SHKZG
          BSCHL
          SGTXT
          INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
          FROM BSEG
          FOR ALL ENTRIES IN LT_BKPF
          WHERE BELNR =  LT_BKPF-BELNR AND
                GJAHR =  LT_BKPF-GJAHR AND
                BUKRS =  LT_BKPF-BUKRS.
*               SHKZG  <> 'H'.
*               HKONT <> SO_HKONT .


  LOOP AT GT_MAIN.
    LV_TABIX = SY-TABIX.

    READ TABLE LT_BKPF WITH KEY BUKRS = GT_MAIN-BUKRS
                                GJAHR = GT_MAIN-GJAHR
                                BELNR = GT_MAIN-BELNR.
    IF SY-SUBRC = 0.
      GT_MAIN-BUDAT = LT_BKPF-BUDAT.
      GT_MAIN-BLDAT = LT_BKPF-BLDAT.
      GT_MAIN-BLART = LT_BKPF-BLART.
      GT_MAIN-WAERS = LT_BKPF-WAERS.
      GT_MAIN-MONAT = LT_BKPF-MONAT.
    ENDIF.

*   공급업체명
    SELECT SINGLE TXT50
      INTO GT_MAIN-TXT50
      FROM SKAT
      WHERE SAKNR = GT_MAIN-HKONT.


*   전표유형 텍스트
    SELECT SINGLE LTEXT INTO GT_MAIN-LTEXT FROM T003T
      WHERE BLART = GT_MAIN-BLART AND
            SPRAS = SY-LANGU.

*   공급업체명
    SELECT SINGLE NAME1 KTOKK
      INTO (GT_MAIN-NAME1, GT_MAIN-KTOKK)
      FROM LFA1
      WHERE LIFNR = GT_MAIN-LIFNR.

*   계정그룹명
    SELECT SINGLE TXT30 INTO GT_MAIN-TXT30
      FROM T077Y
      WHERE KTOKK = GT_MAIN-KTOKK AND
            SPRAS = SY-LANGU.

*   사업장정보
    SELECT SINGLE NAME
           INTO GT_MAIN-NAME
           FROM J_1BBRANCH
           WHERE BRANCH = GT_MAIN-BUPLA AND
                 BUKRS  = GT_MAIN-BUKRS.

    IF GT_MAIN-SHKZG = 'S'.
      GT_MAIN-WRBTR = GT_MAIN-WRBTR * -1.
    ENDIF.

* 매입부가세
    SELECT SINGLE WRBTR DMBTR SHKZG
      INTO (GT_MAIN-WRBTX, GT_MAIN-DMBTX, GT_MAIN-SHKZX)
      FROM BSEG
      WHERE BUKRS = GT_MAIN-BUKRS AND
            GJAHR = GT_MAIN-GJAHR AND
            BELNR = GT_MAIN-BELNR AND
            KTOSL = 'VSTK'.

*   현지통화
    SELECT SINGLE WAERS INTO GT_MAIN-WAERX
      FROM T001
      WHERE BUKRS = GT_MAIN-BUKRS.

    IF GT_MAIN-SHKZX = 'H'.
      GT_MAIN-WRBTX = GT_MAIN-WRBTX * -1.
    ENDIF.

    GT_MAIN-WRBTD = GT_MAIN-WRBTR - GT_MAIN-WRBTX.
    GT_MAIN-DMBTD = GT_MAIN-DMBTR - GT_MAIN-DMBTX.

    MODIFY GT_MAIN INDEX LV_TABIX.
  ENDLOOP.

  LOOP AT GT_MAIN WHERE SHKZG NOT IN SO_SHKZG.

    DELETE GT_MAIN WHERE SHKZG NOT IN SO_SHKZG.
    MODIFY GT_MAIN INDEX LV_TABIX.
  ENDLOOP.

* LOOP AT GT_MAIN where hkont in so_hkont.
*    DELETE GT_MAIN where hkont in so_hkont.
*    MODIFY GT_MAIN INDEX LV_TABIX.
* ENDLOOP.

  SORT GT_MAIN BY BUDAT BLDAT LIFNR.

ENDFORM. " SELECT_MAIN_DATA
