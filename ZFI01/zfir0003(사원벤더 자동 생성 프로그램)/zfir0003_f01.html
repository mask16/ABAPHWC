FORM set_main_data .

  CLEAR: gt_main, gt_main[].

  SELECT a~pernr
         b~ename
         a~begda
         b~orgeh
         a~massn
    INTO (gt_main-pernr,
          gt_main-ename,
          gt_main-begda,
          gt_main-orgeh,
          gt_main-massn)
    FROM pa0000 AS a INNER JOIN pa0001 AS b
      ON a~pernr EQ b~pernr  AND
         a~subty EQ b~subty AND
         a~objps EQ b~objps AND
         a~sprps EQ b~sprps AND
         a~endda EQ b~endda AND
*         a~begda EQ b~begda AND
         a~seqnr EQ b~seqnr
   WHERE a~pernr IN s_pernr
     AND b~ename IN s_ename
     AND a~begda IN s_begda
     AND b~orgeh IN s_orgeh
     AND a~endda >= sy-datum
     AND b~begda <= sy-datum.

    SELECT SINGLE
           mntxt
      INTO gt_main-mntxt
      FROM t529t
     WHERE massn = gt_main-massn
       AND sprsl = sy-langu.

    SELECT SINGLE
            orgtx
       INTO gt_main-orgtx
       FROM t527x
      WHERE orgeh = gt_main-orgeh
        AND endda >= sy-datum
        AND begda <= sy-datum
        AND sprsl = sy-langu.

    SELECT SINGLE
           land1
           state
           ort01
           ort02
           stras
           pstlz
           num01
           land1
      INTO (gt_main-land1,
            gt_main-state,
            gt_main-ort01,
            gt_main-ort02,
            gt_main-stras,
            gt_main-pstlz,
            gt_main-num01,
            gt_main-land1)
      FROM pa0006
     WHERE pernr = gt_main-pernr
       AND subty = '3'
       AND endda >= sy-datum
       AND begda <= sy-datum.
    IF sy-subrc = 0.
      SELECT SINGLE
             bezei
        INTO gt_main-bezei
        FROM t005u
       WHERE land1 = gt_main-land1
         AND bland = gt_main-state.
    ENDIF.

    IF gt_main-land1 IS INITIAL.
      gt_main-land1 = 'KR'.
    ENDIF.

    SELECT SINGLE
           regno
      INTO gt_main-regno
      FROM pa0539
     WHERE pernr = gt_main-pernr
       AND endda >= sy-datum
       AND begda <= sy-datum.

    SELECT SINGLE
           bnksa
           banks
           bankl
           bankn
      INTO (gt_main-bnksa,
            gt_main-banks,
            gt_main-bankl,
            gt_main-bankn)
      FROM pa0009
     WHERE pernr = gt_main-pernr
       AND bnksa = '0'
       AND endda >= sy-datum
       AND begda <= sy-datum.

    APPEND gt_main.
    CLEAR: gt_main.
  ENDSELECT.

  SORT gt_main BY pernr.

ENDFORM. " SET_MAIN_DATA

FORM delete_vendor_data .

  DATA : ls_main LIKE LINE OF gt_main,
         ls_rows TYPE lvc_s_row,
         lt_rows TYPE lvc_t_row.

  CLEAR: gv_total, gv_rate, gv_err.

  CALL METHOD gcl_alv->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows[].

  IF lt_rows[] IS INITIAL.
    MESSAGE s009 DISPLAY LIKE 'W'.  "선택된 라인이 없습니다.
    gv_err = 'X'.
  ENDIF.

  CHECK gv_err IS INITIAL.

  DESCRIBE TABLE lt_rows LINES gv_total.

  LOOP AT lt_rows INTO ls_rows.
    READ TABLE gt_main INTO ls_main INDEX ls_rows-index.
    IF sy-subrc = 0.
      gv_rate = sy-tabix / gv_total * 100.
      PERFORM process_rate_disp USING ls_main-lifnr.

      IF ls_main-sicon = icon_led_red.
        ls_main-btext = '삭제 할 Vendor가 없습니다.'.
        gv_count = gv_count + 1.
      ELSE.
        PERFORM set_bdc_data USING:
           'X' 'SAPMF02K'    '0505',
           ' ' 'BDC_OKCODE'  '/00',
           ' ' 'RF02K-LIFNR' ls_main-lifnr,
           ' ' 'RF02K-BUKRS' '1000',

           'X' 'SAPMF02K'    '0520',
           ' ' 'BDC_OKCODE'  '=UPDA',
           ' ' 'LFA1-LOEVM' 'X',
           ' ' 'LFB1-LOEVM' 'X'.

        CALL TRANSACTION 'FK06' USING gt_bdcdata
                                MODE p_mode
                                UPDATE 'S'.
        IF sy-subrc = 0.
          ls_main-sicon = icon_led_red.
          ls_main-lifnr2 = ''.
          PERFORM set_message CHANGING ls_main-btext.
          gv_count = gv_count + 1.
        ELSE.
          PERFORM set_message CHANGING ls_main-btext.
        ENDIF.

        CLEAR: gt_bdcdata, gt_bdcdata[].
      ENDIF.
    ENDIF.

    MODIFY gt_main FROM ls_main INDEX ls_rows-index
                   TRANSPORTING sicon btext lifnr2.
    CLEAR: ls_main.

  ENDLOOP.

ENDFORM. " DELETE_VENDOR_DATA
