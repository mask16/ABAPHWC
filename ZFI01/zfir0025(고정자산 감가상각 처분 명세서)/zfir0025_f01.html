FORM set_main_data .

  DATA: lv_belnr LIKE bkpf-belnr,
        lv_bschl LIKE bseg-bschl.

  CLEAR: gt_main, gt_main[].

*-- 감가상각 계산 기간
  gv_peraf = p_monat.

*-- 2013 마이그레에션 데이터 읽어 오기
  IF p_gjahr = '2013'.
    PERFORM select_2013_data.
  ENDIF.

  SELECT a~bukrs
         a~anln1
         a~anlkl
         a~txt50
         a~zujhr
         a~aktiv
         c~kansw
         c~knafa
    INTO CORRESPONDING FIELDS OF TABLE gt_main
    FROM anla AS a INNER JOIN anlc AS c
      ON a~bukrs EQ c~bukrs
     AND a~anln1 EQ c~anln1
     AND a~anln2 EQ c~anln2
   WHERE a~bukrs EQ '1000'
     AND a~anln1 IN s_anln1
     AND a~anlkl IN s_anlkl
     AND c~gjahr EQ p_gjahr
     AND c~afabe EQ 1.
*-- 건설중 자산 삭제
  IF sy-subrc = 0.
    DELETE gt_main WHERE anlkl IN gr_anlkl.
  ENDIF.

  CHECK gt_main[] IS NOT INITIAL.

  PERFORM select_anep_data.

  LOOP AT gt_main.
*-- 2013 마이그레에션 데이터 읽어 오기
    IF p_gjahr = '2013'.
*-- 취득액-전기누계분(A)
      READ TABLE gt_2013 WITH KEY bukrs = gt_main-bukrs
                                  anln1 = gt_main-anln1.
      IF sy-subrc = 0.
        gt_main-anbtr = gt_2013-bfdec.  "취득액-전기누계분(A)
      ELSE.
        CLEAR: gt_anbtr, gt_anbtr[].

        LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                          AND anln1 EQ gt_main-anln1
                          AND bwasl IN gr_awasl
                          AND gjahr <  p_gjahr.
          MOVE-CORRESPONDING gt_anep TO gt_anbtr.
          COLLECT gt_anbtr.
          CLEAR:  gt_anbtr.
        ENDLOOP.

        READ TABLE gt_anbtr INDEX 1.
        gt_main-anbtr = gt_anbtr-anbtr.  "취득액-전기누계분(A)
      ENDIF.
*-- 취득액-당기분(B)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_awasl
                        AND gjahr EQ p_gjahr
                        AND monat <= p_monat.
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-bnbtr = gt_2013-nwdec + gt_anbtr-anbtr.  "취득액-당기분(B)

*-- 재평가차액(C)
      gt_main-cnbtr = gt_2013-gpdec.  "재평가차액(C)

*-- 자본적지출-당기(E)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_dwasl
                        AND gjahr EQ p_gjahr
                        AND monat <= p_monat.
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-enbtr = gt_2013-nwtdc + gt_anbtr-anbtr.  "자본적지출-당기(E)

*-- 자본적지출-전기말누계(D)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_dwasl
                        AND gjahr <  p_gjahr
                        AND gjahr NE gt_main-zujhr. "취득년도
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-dnbtr = gt_2013-bftdc + gt_anbtr-anbtr.  "자본적지출-전기말누계(D)

    ELSEIF p_gjahr <> '2013'.
*-- 취득액-전기누계분(A)
      READ TABLE gt_2013 WITH KEY bukrs = gt_main-bukrs
                                  anln1 = gt_main-anln1.

      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_awasl
                        AND gjahr <  p_gjahr.
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-anbtr = gt_2013-bfdec + gt_anbtr-anbtr + gt_anbtr-anbtr.  "취득액-전기누계분(A)

*-- 취득액-당기분(B)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_awasl
                        AND gjahr EQ p_gjahr
                        AND monat <= p_monat.
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        IF gt_anep-bwasl = '330' OR gt_anep-bwasl = '306'.
          COLLECT gt_anbtr.
        ELSE.
          IF gt_anep-gjahr = gt_main-zujhr.
            COLLECT gt_anbtr.
          ENDIF.
        ENDIF.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-bnbtr = gt_2013-nwdec + gt_anbtr-anbtr.  "취득액-당기분(B)

*-- 재평가차액(C)
      gt_main-cnbtr = gt_2013-gpdec.  "재평가차액(C)

*-- 자본적지출-당기(E)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_dwasl
                        AND gjahr EQ p_gjahr
                        AND monat <= p_monat.
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-enbtr = gt_anbtr-anbtr.  "자본적지출-당기(E)

*-- 자본적지출-전기말누계(D)
      CLEAR: gt_anbtr, gt_anbtr[].

      LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                        AND anln1 EQ gt_main-anln1
                        AND bwasl IN gr_dwasl
                        AND gjahr <  p_gjahr
                        AND gjahr NE gt_main-zujhr. "취득년도
        MOVE-CORRESPONDING gt_anep TO gt_anbtr.
        COLLECT gt_anbtr.
        CLEAR:  gt_anbtr.
      ENDLOOP.

      READ TABLE gt_anbtr INDEX 1.
      gt_main-dnbtr = gt_2013-bftdc + gt_anbtr-anbtr + gt_main-enbtr.  "자본적지출-전기말누계(D)
    ENDIF.
*-- 취득원가 계(F=A+B+C+D+E)
    gt_main-fnbtr = gt_main-anbtr + gt_main-bnbtr + gt_main-cnbtr +
                    gt_main-dnbtr + gt_main-enbtr.
*-- 차이(F-Z)
    gt_main-gap = gt_main-fnbtr - gt_main-kansw.
*-- 전기말상각누계액(G)
    gt_main-knafa = gt_main-knafa * -1.
*-- 상각전가액(H=Z-G)
    gt_main-hnbtr = gt_main-kansw - gt_main-knafa.
*-- 당기상각액(I)
    CLEAR: gt_nafaz, gt_nafaz[].

    SELECT *
      INTO CORRESPONDING FIELDS OF gt_nafaz
      FROM anlp
     WHERE bukrs EQ gt_main-bukrs
       AND gjahr EQ p_gjahr
       AND peraf <= gv_peraf
       AND anln1 EQ gt_main-anln1.

      COLLECT gt_nafaz.
      CLEAR:  gt_nafaz.
    ENDSELECT.

    IF sy-subrc = 0.
      READ TABLE gt_nafaz INDEX 1.
      gt_main-inbtr = gt_nafaz-nafaz * -1.
    ENDIF.
*-- 당기 말상각누계액(J=G+I)
    gt_main-jnbtr = gt_main-knafa + gt_main-lnbtr.
*-- 처분전 가액(K=Z-J)
    gt_main-knbtr = gt_main-kansw - gt_main-jnbtr.
*-- 당기처분액(L)
    CLEAR: gt_dmbtr, gt_dmbtr[].

    LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                      AND anln1 EQ gt_main-anln1
                      AND bwasl IN gr_mwasl
                      AND gjahr EQ p_gjahr
                      AND monat <= p_monat.

      IF gt_anep-bwasl = '200'.
        gt_dmbtr-dmbtr = 0.
      ELSEIF gt_anep-bwasl = '210'.
        gt_dmbtr-bukrs = gt_anep-bukrs.
        gt_dmbtr-anln1 = gt_anep-anln1.

        SELECT SINGLE
               belnr
          INTO lv_belnr
          FROM bkpf
         WHERE bukrs EQ gt_anep-bukrs
           AND gjahr EQ gt_anep-gjahr
           AND awkey EQ gt_anep-awkey.
        IF sy-subrc = 0.
          SELECT SINGLE
                 dmbtr
            INTO gt_dmbtr-dmbtr
            FROM bseg
           WHERE bukrs EQ gt_anep-bukrs
             AND gjahr EQ gt_anep-gjahr
             AND belnr EQ lv_belnr
             AND bschl EQ '50'
             AND hkont EQ '0091010010'.
        ENDIF.
      ENDIF.
      COLLECT gt_dmbtr.
      CLEAR:  gt_dmbtr, lv_belnr.
    ENDLOOP.
    IF sy-subrc = 0.
      READ TABLE gt_dmbtr INDEX 1.
      gt_main-lnbtr = gt_dmbtr-dmbtr.
    ENDIF.
*-- 처분손익(M)
    CLEAR: gt_dmbtr, gt_dmbtr[].

    LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                      AND anln1 EQ gt_main-anln1
                      AND bwasl IN gr_mwasl
                      AND gjahr EQ p_gjahr
                      AND monat <= p_monat.

      gt_dmbtr-bukrs = gt_anep-bukrs.
      gt_dmbtr-anln1 = gt_anep-anln1.

      SELECT SINGLE
             belnr
        INTO lv_belnr
        FROM bkpf
       WHERE bukrs EQ gt_anep-bukrs
         AND gjahr EQ gt_anep-gjahr
         AND awkey EQ gt_anep-awkey.
      IF sy-subrc = 0.
        SELECT dmbtr bschl
          INTO (gt_dmbtr-dmbtr, lv_bschl)
          FROM bseg
         WHERE bukrs EQ gt_anep-bukrs
           AND gjahr EQ gt_anep-gjahr
           AND belnr EQ lv_belnr
           AND hkont EQ '0071050010'
           AND bschl EQ '50'.

          gt_dmbtr-dmbtr = gt_dmbtr-dmbtr.
          COLLECT gt_dmbtr.
          CLEAR:  gt_dmbtr.
        ENDSELECT.

        SELECT dmbtr bschl
          INTO (gt_dmbtr-dmbtr, lv_bschl)
          FROM bseg
         WHERE bukrs EQ gt_anep-bukrs
           AND gjahr EQ gt_anep-gjahr
           AND belnr EQ lv_belnr
           AND hkont EQ '0075050010'
           AND bschl EQ '40'.

          gt_dmbtr-dmbtr = gt_dmbtr-dmbtr * -1.
          COLLECT gt_dmbtr.
          CLEAR:  gt_dmbtr.
        ENDSELECT.
      ENDIF.
      CLEAR:  gt_dmbtr, lv_belnr, lv_bschl.
    ENDLOOP.
    IF sy-subrc = 0.
      READ TABLE gt_dmbtr INDEX 1.
      gt_main-mnbtr = gt_dmbtr-dmbtr.
    ENDIF.
*-- 당기말잔액(N=K-(L-M))
    gt_main-nnbtr = gt_main-knbtr - ( gt_main-lnbtr - gt_main-mnbtr ).
*-- 자산 처분 구분
    LOOP AT gt_anep WHERE bukrs EQ gt_main-bukrs
                      AND anln1 EQ gt_main-anln1
                      AND bwasl IN gr_mwasl
                      AND gjahr <= p_gjahr.
      IF gt_anep-gjahr = p_gjahr. "같은년도
        IF gt_anep-monat <= p_monat.
          IF gt_anep-bwasl = '200' OR gt_anep-bwasl = '320'.
            gt_main-gubun = '폐기'.
          ELSEIF gt_anep-bwasl = '210'.
            gt_main-gubun = '매각'.
          ENDIF.
*-- 처분일자
          gt_main-bzdat = gt_anep-bzdat.
        ENDIF.
      ELSE.
        IF gt_anep-bwasl = '200' OR gt_anep-bwasl = '320'.
          gt_main-gubun = '폐기'.
        ELSEIF gt_anep-bwasl = '210'.
          gt_main-gubun = '매각'.
        ENDIF.
*-- 처분일자
        gt_main-bzdat = gt_anep-bzdat.
      ENDIF.
    ENDLOOP.
*-- 내용년수
    SELECT SINGLE
           ndjar
      INTO gt_main-ndjar
      FROM anlb
