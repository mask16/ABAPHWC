FORM set_ranges .

  DATA: lv_option(2) TYPE c.

*  IF p_kunnr IS NOT INITIAL.
*    PERFORM set_option USING p_kunnr
*                       CHANGING lv_option.
*    gr_kunnr-sign   = 'I'.
*    gr_kunnr-option = lv_option.
*    gr_kunnr-low    = p_kunnr.
*    APPEND gr_kunnr.
*  ENDIF.

  IF p_bupla IS NOT INITIAL.
    PERFORM set_option USING p_bupla
                       CHANGING lv_option.
    gr_bupla-sign   = 'I'.
    gr_bupla-option = lv_option.
    gr_bupla-low    = p_bupla.
    APPEND gr_bupla.
  ENDIF.

*-- 세금계산서 방향
  IF pr_dir1 = 'X'.
*-- 순방향
    gr_dir-sign   = 'I'.
    gr_dir-option = 'EQ'.
    gr_dir-low    = '순방향'.
    APPEND gr_dir.
    CLEAR: gr_dir.

  ELSEIF pr_dir2 = 'X'.
*-- 역방향
    gr_dir-sign   = 'I'.
    gr_dir-option = 'EQ'.
    gr_dir-low    = '역방향'.
    APPEND gr_dir.
    CLEAR: gr_dir.

  ELSE.
*-- 모두
    gr_dir-sign   = 'I'.
    gr_dir-option = 'EQ'.
    gr_dir-low    = '순방향'.
    APPEND gr_dir.
    CLEAR: gr_dir.

    gr_dir-sign   = 'I'.
    gr_dir-option = 'EQ'.
    gr_dir-low    = '역방향'.
    APPEND gr_dir.
    CLEAR: gr_dir.

  ENDIF.

*-- 년월에 대해 일자값으로 변경
  READ TABLE s_spmon INDEX 1.
  CONCATENATE s_spmon-low  '01' INTO gv_sday.
  IF s_spmon-high IS INITIAL.
    CONCATENATE s_spmon-low  '31' INTO gv_eday.
  ELSE.
    CONCATENATE s_spmon-high '31' INTO gv_eday.
  ENDIF.

ENDFORM. " SET_RANGES

FORM set_detail_data .

  CLEAR: gt_detail, gt_detail[].

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_detail
    FROM zfit0007
   WHERE kunnr IN s_kunnr
     AND ktokd IN s_ktokd
     AND sdate BETWEEN gv_sday AND gv_eday
     AND sdate IN s_sdate
     AND bupla IN gr_bupla
     AND vkbur IN s_vkbur
     AND vbeln IN s_vbeln
     AND zunnr IN s_zunnr
     AND dir   IN gr_dir.

  SORT gt_detail BY kunnr customerid duty_num itmseq.

  LOOP AT gt_detail.
*-- 고객명
    IF gt_detail-mwskz = 'B1'.
      SELECT SINGLE
             adrnr
        INTO gt_detail-adrnr
        FROM vbpa
       WHERE vbeln = gt_detail-vbeln
         AND parvw = 'AG'.
      IF sy-subrc = 0.
        SELECT SINGLE
               name1
          INTO gt_detail-name1
          FROM adrc
         WHERE addrnumber = gt_detail-adrnr.
      ENDIF.
    ELSE.
      SELECT SINGLE
             name1
        INTO (gt_detail-name1)
        FROM kna1
       WHERE kunnr = gt_detail-kunnr.
    ENDIF.
*-- 사무소 텍스트
    SELECT SINGLE
           bezei
      INTO gt_detail-bezei
      FROM tvkbt
     WHERE spras = sy-langu
       AND vkbur = gt_detail-vkbur.
*-- 사업장명
    SELECT SINGLE
           name
      FROM j_1bbranch
      INTO gt_detail-name
     WHERE bukrs  = '1000'
       AND branch = gt_detail-bupla.

*-- 반품 마이너스
  IF gt_detail-itmnet < '0'.
      gt_detail-itmnet  = gt_detail-itmnet * -1.
      gt_detail-itmqty  = gt_detail-itmqty * -1.
  ENDIF.

    gt_detail-sum = gt_detail-cost + gt_detail-vat.

    MODIFY gt_detail.
    CLEAR: gt_detail.
  ENDLOOP.

ENDFORM. " SET_DETAIL_DATA
