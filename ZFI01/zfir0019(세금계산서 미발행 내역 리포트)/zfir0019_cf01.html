FORM f_user_hotspot USING ps_row_id TYPE lvc_s_row
                              ps_column_id  TYPE lvc_s_col
                              ps_row_no  TYPE lvc_s_roid.

  DATA: ls_main LIKE LINE OF gt_main.

  CASE ps_column_id.
    WHEN 'BELNR'.
      READ TABLE gt_main INTO ls_main INDEX ps_row_id-index.
      IF ls_main-belnr IS NOT INITIAL.
        SET PARAMETER ID 'BLN' FIELD ls_main-belnr.
        SET PARAMETER ID 'BUK' FIELD ls_main-bukrs.
        SET PARAMETER ID 'GJR' FIELD ls_main-gjahr.
        CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
      ENDIF.

    WHEN 'VBELN'.
      READ TABLE gt_main INTO ls_main INDEX ps_row_id-index.
      IF ls_main-vbeln IS NOT INITIAL.
        SET PARAMETER ID 'VF'  FIELD ls_main-vbeln.
        CALL TRANSACTION 'VF03' AND SKIP FIRST SCREEN.
      ENDIF.

  ENDCASE.

ENDFORM. " F_USER_HOTSPOT
