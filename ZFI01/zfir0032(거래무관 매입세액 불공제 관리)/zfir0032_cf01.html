*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM c_ucomm_delete_proc .

  DATA : ls_main LIKE LINE OF gt_main,
         ls_main_d LIKE LINE OF gt_main_d,
         ls_rows TYPE lvc_s_row,
         lt_rows TYPE lvc_t_row.

  CLEAR: lt_rows[].

  CALL METHOD gcl_alv->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows[].

  CHECK NOT lt_rows[] IS INITIAL.

  LOOP AT lt_rows INTO ls_rows.
    READ TABLE gt_main INTO ls_main INDEX ls_rows-index.
    IF sy-subrc = 0.
*-- 존재하면 삭제테이블에 저장해 놓는다..
      IF ls_main-dmark IS INITIAL AND
         ls_main-i IS INITIAL.
        CLEAR: ls_main_d.
        ls_main-mdate = sy-datum.
        ls_main-mtime = sy-timlo.
        ls_main-mname = sy-uname.
        ls_main-dmark = 'X'.

        MOVE-CORRESPONDING ls_main TO ls_main_d.
        APPEND ls_main_d TO gt_main_d.

        MODIFY gt_main FROM ls_main INDEX ls_rows-index
                        TRANSPORTING dmark.
      ENDIF.
    ENDIF.
  ENDLOOP.

  DELETE gt_main WHERE NOT dmark IS INITIAL.

ENDFORM. " C_UCOMM_DELETE_PROC

FORM check_data_proc USING pr_data_changed
                      TYPE REF TO cl_alv_changed_data_protocol
                      CHANGING error_in_data.

  DATA: ls_modf_field  TYPE lvc_s_modi.

  CLEAR: gv_err.

  LOOP AT pr_data_changed->mt_good_cells INTO ls_modf_field.
    CASE ls_modf_field-fieldname.
*-- 작성일, 공급처명, 사업자등록번호, 공급가액, 세액
      WHEN 'RDATE' OR 'NAME1' OR 'STCD1' OR 'HWBAS' OR 'HWSTE'.
        PERFORM check_data_to_main USING pr_data_changed
                                         ls_modf_field
                                CHANGING error_in_data.

      WHEN OTHERS.

    ENDCASE.
  ENDLOOP.

ENDFORM. " CHECK_DATA_PROC

*       text
*----------------------------------------------------------------------*
*      -->P_PR_DATA_CHANGED  text
*      -->P_LS_MODF_FIELD  text
*      <--P_ERROR_IN_DATA  text
*----------------------------------------------------------------------*
FORM check_data_to_main
  USING p_data_changed TYPE REF TO cl_alv_changed_data_protocol
          p_modf_field TYPE lvc_s_modi
  CHANGING p_error_in_data.

  DATA: ls_main LIKE LINE OF gt_main,
        lv_field TYPE string,
        lv_hbkid TYPE hbkid.

  READ TABLE gt_main INTO ls_main INDEX p_modf_field-row_id.

*-- CELL 에서 변경된 필드값을 가지고 온다.
  CALL METHOD p_data_changed->get_cell_value
    EXPORTING
      i_row_id    = p_modf_field-row_id
      i_fieldname = p_modf_field-fieldname
    IMPORTING
      e_value     = lv_field.

  CASE p_modf_field-fieldname.
*-- 작성일
    WHEN 'RDATE'.
      ls_main-rdate = lv_field.
      IF ls_main-rdate < gv_sdate OR
         ls_main-rdate > gv_edate.
        MESSAGE s108 DISPLAY LIKE 'W'.  "입력하신 작성일이 부가세 기간을 초과합니다.
        gv_err = 'X'.
      ENDIF.
*-- 공급처 명
    WHEN 'NAME1'.
      ls_main-name1 = lv_field.
*-- 사업자등록번호
    WHEN 'STCD2'.
      ls_main-stcd2 = lv_field.
*-- 공급가
    WHEN 'HWBAS'.
      ls_main-hwbas = lv_field.
      ls_main-hwsum = ls_main-hwbas + ls_main-hwste.
*-- 세액
    WHEN 'HWSTE'.
      ls_main-hwste = lv_field.
      ls_main-hwsum = ls_main-hwbas + ls_main-hwste.
    WHEN OTHERS.

  ENDCASE.

*-- 변경한 내용을 화면 ALV 에 갱신한다.
  IF gv_err IS INITIAL.
    IF ls_main-i = ''.
      ls_main-m = 'X'.
    ENDIF.
    MODIFY gt_main FROM ls_main INDEX p_modf_field-row_id.
    PERFORM set_new_ok_code USING 'PBO'.
  ENDIF.

ENDFORM. " CHECK_DATA_TO_MAIN
