  CLEAR: gv_err.

  CASE pv_ucomm.
    WHEN 'ar_detail'.
      IF pr_dl = 'X'.
        PERFORM c_ucomm_ar_detail_proc.

        PERFORM c_ucomm_ap_detail_proc.
      ELSE.
        PERFORM c_ucomm_ap_sum_proc.
      ENDIF.

    WHEN 'ax_detail'.
      IF pr_dl = 'X'.
        PERFORM c_ucomm_ax_detail_proc.
      ELSE.
        PERFORM c_ucomm_ax_sum_proc.
      ENDIF.

    WHEN OTHERS.

  ENDCASE.

  IF gv_err IS INITIAL.

    CALL METHOD cl_gui_cfw=>set_new_ok_code

  ENDIF.
