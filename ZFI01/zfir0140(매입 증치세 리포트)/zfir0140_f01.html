FORM set_main_data .

  CLEAR: gt_main, gt_main[].
  DATA : lt_bkpf  LIKE bkpf   OCCURS 0 WITH HEADER LINE,
         lt_bseg  LIKE bseg   OCCURS 0 WITH HEADER LINE,
         lt_bset  LIKE bset   OCCURS 0 WITH HEADER LINE,
         lt_bsak  LIKE bsak   OCCURS 0 WITH HEADER LINE,
         lt_lfa1  LIKE lfa1   OCCURS 0 WITH HEADER LINE,
         lt_t007s LIKE t007s  OCCURS 0 WITH HEADER LINE.
  CLEAR: lt_bkpf, lt_bkpf[], lt_bseg, lt_bseg[], lt_bset, lt_bset[],
         lt_bsak, lt_bsak[], lt_lfa1, lt_lfa1[], lt_t007s, lt_t007s[].

  PERFORM set_range_main.

  SELECT *
    INTO TABLE lt_bkpf
    FROM bkpf
   WHERE bukrs IN s_bukrs
     AND gjahr IN s_gjahr
     AND blart IN r_blart
     AND monat IN s_monat
     AND bldat IN s_bldat
     AND budat IN s_budat
     AND stblg = ''.

  CHECK lt_bkpf[] IS NOT INITIAL.

  SELECT *
    INTO TABLE lt_bseg
    FROM bseg
     FOR ALL ENTRIES IN lt_bkpf
   WHERE bukrs = lt_bkpf-bukrs
     AND belnr = lt_bkpf-belnr
     AND gjahr = lt_bkpf-gjahr
     AND mwskz IN r_mwskz
     AND lifnr IN s_lifnr
     AND koart = 'K'.

  IF lt_bseg[] IS NOT INITIAL.
    SELECT *
      INTO TABLE lt_bset
      FROM bset
       FOR ALL ENTRIES IN lt_bseg
     WHERE bukrs = lt_bseg-bukrs
       AND belnr = lt_bseg-belnr
       AND gjahr = lt_bseg-gjahr.

    SELECT * INTO TABLE lt_lfa1
      FROM lfa1
       FOR ALL ENTRIES IN lt_bseg
     WHERE lifnr = lt_bseg-lifnr.

    DATA : l_kalsm LIKE t007s-kalsm.
    CLEAR: l_kalsm.
    CASE s_bukrs-low.
      WHEN '2000'.
        l_kalsm = 'TAXCN'.
      WHEN '3000'.
        l_kalsm = 'TAXVN'.
    ENDCASE.

    SELECT * INTO TABLE lt_t007s
      FROM t007s
       FOR ALL ENTRIES IN lt_bseg
     WHERE spras = sy-langu
       AND kalsm = l_kalsm
       AND mwskz = lt_bseg-mwskz.

    SELECT * INTO TABLE lt_bsak
      FROM bsak
       FOR ALL ENTRIES IN lt_bseg
     WHERE bukrs = lt_bseg-bukrs
       AND gjahr = lt_bseg-gjahr
       AND belnr = lt_bseg-belnr
       AND buzei = lt_bseg-buzei.

  ENDIF.

  SORT lt_bkpf  BY bukrs belnr gjahr.
  SORT lt_bseg  BY bukrs belnr gjahr buzei.
  SORT lt_bset  BY bukrs belnr gjahr buzei.
  SORT lt_bsak  BY bukrs belnr gjahr buzei.
  SORT lt_lfa1  BY lifnr.
  SORT lt_t007s BY mwskz.

  DATA : l_mul TYPE bset-hwbas.

  LOOP AT lt_bkpf.

    LOOP AT lt_bseg WHERE bukrs = lt_bkpf-bukrs
                      AND belnr = lt_bkpf-belnr
                      AND gjahr = lt_bkpf-gjahr.
      ">>CANCEL MM8M
      IF lt_bkpf-blart = 'RE' AND lt_bseg-koart = 'K'.
        READ TABLE lt_bseg WITH KEY rebzg = lt_bseg-belnr
                                    rebzj = lt_bseg-gjahr
                                    rebzz = lt_bseg-buzei.
        IF sy-subrc = 0.
          CONTINUE.
        ENDIF.

        IF lt_bseg-rebzg IS NOT INITIAL.
          CONTINUE.
        ENDIF.
      ENDIF.

      CLEAR: gt_main.
      MOVE-CORRESPONDING lt_bkpf TO gt_main.
      IF lt_bkpf-blart = 'RV'.
        gt_main-xblnr = lt_bkpf-xref2_hd.
      ENDIF.

      READ TABLE lt_bsak WITH KEY bukrs = lt_bseg-bukrs
                                  belnr = lt_bseg-belnr
                                  gjahr = lt_bseg-gjahr
                                  buzei = lt_bseg-buzei
                                  BINARY SEARCH.
      IF sy-subrc = 0.
        gt_main-augbl = lt_bsak-augbl.
        gt_main-augdt = lt_bsak-augdt.
      ENDIF.
      gt_main-lifnr = lt_bseg-lifnr.

      LOOP AT lt_bset WHERE bukrs = lt_bseg-bukrs
                         AND belnr = lt_bseg-belnr
                         AND gjahr = lt_bseg-gjahr.

        gt_main-hwbas = lt_bset-hwbas.
        gt_main-hwste = lt_bset-hwste.
        gt_main-fwbas = lt_bset-fwbas.
        gt_main-fwste = lt_bset-fwste.
        gt_main-mwskz = lt_bset-mwskz.

        CLEAR: l_mul.
        CASE lt_bset-shkzg.
          WHEN 'S'.
            l_mul = 1.
          WHEN 'H'.
            l_mul = -1.
        ENDCASE.

        gt_main-hwbas = gt_main-hwbas * l_mul.
        gt_main-hwste = gt_main-hwste * l_mul.
        gt_main-fwbas = gt_main-fwbas * l_mul.
        gt_main-fwste = gt_main-fwste * l_mul.

        gt_main-hwtot = gt_main-hwbas + gt_main-hwste.
        gt_main-fwtot = gt_main-fwbas + gt_main-fwste.

        READ TABLE lt_lfa1 WITH KEY lifnr = lt_bseg-lifnr
                                    BINARY SEARCH.
        IF sy-subrc = 0.
          gt_main-name1 = lt_lfa1-name1.
          gt_main-stcd1 = lt_lfa1-stcd1.
        ENDIF.

        READ TABLE lt_t007s WITH KEY mwskz = lt_bseg-mwskz
                                     BINARY SEARCH.
        IF sy-subrc = 0.
          gt_main-text1 = lt_t007s-text1.
        ENDIF.

        APPEND gt_main.

      ENDLOOP.

    ENDLOOP.

  ENDLOOP.

ENDFORM. " SET_MAIN_DATA
