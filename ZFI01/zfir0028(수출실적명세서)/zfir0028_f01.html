FORM SET_MAIN_DATA .

  DATA: LT_MAIN LIKE TABLE OF GT_MAIN WITH HEADER LINE,
        LV_SQN(7) TYPE C.

  CLEAR: GT_MAIN, GT_MAIN[].

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE LT_MAIN
    FROM BKPF
   WHERE BUKRS EQ '1000'
     AND GJAHR EQ P_GJAHR
     AND MONAT IN S_MONAT
     AND BUDAT IN S_BUDAT
     AND BLDAT IN S_BLDAT
     AND STBLG EQ SPACE
     AND BLART EQ 'RV'.
  IF SY-SUBRC = 0.
    LOOP AT LT_MAIN.
      MOVE-CORRESPONDING LT_MAIN TO GT_MAIN.

      SELECT *
        INTO CORRESPONDING FIELDS OF GT_MAIN
        FROM BSEG
       WHERE BUKRS EQ GT_MAIN-BUKRS
         AND BELNR EQ GT_MAIN-BELNR
         AND GJAHR EQ GT_MAIN-GJAHR
         AND KUNNR IN S_KUNNR
         AND HKONT IN GR_HKONT
         AND MWSKZ EQ 'AB'.
*       ZUONR => VBELN 으로 바꿈 20130422 AB01
        IF GT_MAIN-VBELN IS NOT INITIAL.
*        IF gt_main-zuonr IS NOT INITIAL.
          SELECT SINGLE VBELV
            INTO GT_MAIN-VBELV
            FROM VBFA
            WHERE VBELN EQ GT_MAIN-VBELN
*           WHERE vbeln EQ gt_main-zuonr
             AND VBTYP_V = 'J'.
          IF SY-SUBRC = 0.
*            SELECT SINGLE
*                   A~ZDCRNO
*              INTO GT_MAIN-ZDCRNO
*              FROM ZEX406 AS A INNER JOIN ZEX401 AS B
*                ON A~ZEXDCL EQ B~ZEXDCL
*             WHERE B~AUBEL = GT_MAIN-VBELV
*               AND B~AUTYP = 'J'.
            DATA : LV_ZSHNUM  TYPE ZEX303-ZSHNUM,
                   LV_ZINVNO  TYPE ZEX300-ZINVNO.
            SELECT SINGLE ZSHNUM INTO LV_ZSHNUM FROM ZEX303 WHERE AUBEL = GT_MAIN-VBELV.
            IF SY-SUBRC = 0.
              SELECT SINGLE ZINVNO INTO LV_ZINVNO FROM ZEX300 WHERE ZSHNUM = LV_ZSHNUM.
              IF SY-SUBRC = 0.
                SELECT SINGLE ZDCRNO INTO GT_MAIN-ZDCRNO FROM ZEX406 WHERE ZINVNO = LV_ZINVNO.
              ENDIF.
            ENDIF.


            IF GT_MAIN-ZDCRNO IS NOT INITIAL.
              GT_MAIN-ZDCHK = 'X'.

              SELECT SINGLE
                     ARFNR
                INTO GT_MAIN-ARFNR
                FROM ZFIT0017
               WHERE BUKRS EQ GT_MAIN-BUKRS
                 AND GJAHR EQ GT_MAIN-GJAHR
                 AND BELNR EQ GT_MAIN-BELNR.
              IF SY-SUBRC = 0.
                GT_MAIN-STATUS = '확정'.
              ELSE.
                GT_MAIN-STATUS = ''.
              ENDIF.

            ENDIF.

            SELECT SINGLE
                   NAME1
              INTO (GT_MAIN-NAME1)
              FROM KNA1
             WHERE KUNNR EQ GT_MAIN-KUNNR.

            GT_MAIN-TTERM = P_TTERM.

            APPEND GT_MAIN.
            CLEAR: GT_MAIN.


          ENDIF.
        ENDIF.
        CLEAR: GT_MAIN.
      ENDSELECT.
    ENDLOOP.
  ENDIF.

  SORT GT_MAIN BY BELNR ZDCRNO.

  LOOP AT GT_MAIN.
    LV_SQN = LV_SQN + 1.
    GT_MAIN-SQN = LV_SQN.
    MODIFY GT_MAIN.
    CLEAR: GT_MAIN.
  ENDLOOP.

ENDFORM. " SET_MAIN_DATA
