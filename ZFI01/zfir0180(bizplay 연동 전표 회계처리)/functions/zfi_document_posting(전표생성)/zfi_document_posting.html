FUNCTION ZFI_DOCUMENT_POSTING.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BDCMOD) LIKE CTU_PARAMS-DISMODE
*" REFERENCE(I_PARKDC) TYPE CHAR01
*" REFERENCE(I_CLEAR) TYPE CHAR01 OPTIONAL
*" REFERENCE(I_NUM) TYPE NUM02 OPTIONAL
*" REFERENCE(I_PARAMT) TYPE CHAR16 OPTIONAL
*" EXPORTING
*" REFERENCE(E_SUBRC) LIKE SY-SUBRC
*" REFERENCE(E_MSGTY) LIKE SY-MSGTY
*" REFERENCE(E_MSGID) LIKE SY-MSGID
*" REFERENCE(E_MSGNO) LIKE SY-MSGNO
*" REFERENCE(E_MSGV1) LIKE SY-MSGV1
*" REFERENCE(E_MSGV2) LIKE SY-MSGV2
*" REFERENCE(E_MSGV3) LIKE SY-MSGV3
*" REFERENCE(E_MSGV4) LIKE SY-MSGV4
*" REFERENCE(E_BELNR) LIKE BKPF-BELNR
*" TABLES
*" T_BBKPF STRUCTURE BBKPF
*" T_BBSEG STRUCTURE BBSEG
*" T_BBTAX STRUCTURE BBTAX OPTIONAL
*" T_BWITH STRUCTURE BWITH OPTIONAL
*" T_BSELK STRUCTURE BSELK OPTIONAL
*" T_BSELP STRUCTURE BSELP OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  CLEAR: W_BELNR.

  IF T_BBKPF[] IS INITIAL. " OR T_BBSEG[] IS INITIAL.
    RAISE ERROR.
  ENDIF.

  ANZ_MODE = MODE = I_BDCMOD.
  UPDATE   = P_UPDMODE.
  G_NUM    = I_NUM.
  G_CLEAR  = I_CLEAR.
  G_PARAMT = I_PARAMT.

* Table에대한 내역을 읽는다
  PERFORM SELECT_FIELD_NAME.

*.. DATA CLEAR
  PERFORM CLEAR_DATA.

*.. BGR00, BBKPF
  PERFORM FILL_BGR00_RECORD.

  READ TABLE T_BBKPF INDEX 1.

*.. 임시전표 '완료로 저장' 시 'X'
  MOVE I_PARKDC TO T_BBKPF-XPRFG.

  PERFORM FILL_BBKPF_RECORD USING T_BBKPF.

*.. BBSEG
  LOOP AT T_BBSEG.
    PERFORM FILL_BBSEG_RECORD USING T_BBSEG.
  ENDLOOP.

*.. BBTAX
  LOOP AT T_BBTAX.
    PERFORM FILL_BBTAX_RECORD USING T_BBTAX.
  ENDLOOP.

*.. BWITH
  LOOP AT T_BWITH.
    PERFORM FILL_BWITH_RECORD USING T_BWITH.
  ENDLOOP.

*.. BSELK
  LOOP AT T_BSELK.
    PERFORM FILL_BSELK_RECORD USING T_BSELK.
  ENDLOOP.

*.. BSELP
  LOOP AT T_BSELP.
    PERFORM FILL_BSELP_RECORD USING T_BSELP.
  ENDLOOP.


  PERFORM LOOP_AT_TABLE_TFILE.

* RESULT
  IF MSGTY = 'S' AND ( ( MSGID = 'F5' AND MSGNO = '312' ) OR
                       ( MSGID = 'FP' AND MSGNO = '001' ) ).
    E_SUBRC = 0.
  ELSE.
    E_SUBRC = 4.
  ENDIF.

  E_MSGID = MSGID.
  E_MSGTY = MSGTY.
  E_MSGNO = MSGNO.
  E_MSGV1 = MSGV1.
  E_MSGV2 = MSGV2.
  E_MSGV3 = MSGV3.
  E_MSGV4 = MSGV4.
  E_BELNR = W_BELNR.

  COMMIT WORK.
  CALL FUNCTION 'DEQUEUE_ALL'.

ENDFUNCTION.
