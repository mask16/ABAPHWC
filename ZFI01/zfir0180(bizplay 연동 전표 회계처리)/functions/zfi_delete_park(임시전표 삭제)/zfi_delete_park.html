Description: 임시전표 삭제
FUNCTION ZFI_DELETE_PARK .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_BUKRS) LIKE VBKPF-BUKRS
*" REFERENCE(I_GJAHR) LIKE VBKPF-GJAHR
*" REFERENCE(I_BELNR) LIKE VBKPF-BELNR
*" REFERENCE(I_BSTAT) LIKE VBKPF-BSTAT DEFAULT 'Z'
*" REFERENCE(I_MODE) LIKE RFPDO-ALLGAZMD DEFAULT 'N'
*" EXPORTING
*" REFERENCE(E_RETURN) TYPE BAPIRETURN
*"----------------------------------------------------------------------

* 임시전표인지 여부 체크
* Global data declarations

  SELECT SINGLE * FROM vbkpf WHERE bukrs = i_bukrs
                               AND gjahr = i_gjahr
                               AND belnr = i_belnr.
  IF sy-subrc NE 0.
    e_return-type    = 'E'.
    e_return-message = '해당전표는 임시전표가 아닙니다.'(001).
    EXIT.
  ENDIF.

* 'FBV0' BDC
  CLEAR : gt_bdc[], gt_bdc, gt_msg[], gt_msg, ctu_params.

* Set data
* 전표번호 입력
  PERFORM dynpro USING : 'X' 'SAPMF05V'    '0100',
                         ' ' 'BDC_OKCODE'  '=ENTR',
                         ' ' 'RF05V-BUKRS' i_bukrs,
                         ' ' 'RF05V-BELNR' i_belnr,
                         ' ' 'RF05V-GJAHR' i_gjahr.
  CASE vbkpf-tcode.
    WHEN 'FV60' OR 'FB60'.
* 구매처 송장
      PERFORM dynpro USING : 'X' 'SAPMF05A'   '1100',
                             ' ' 'BDC_OKCODE' '=9-PD'.
    WHEN 'FV70' OR 'FB70'.
* 고객 송장
      PERFORM dynpro USING : 'X' 'SAPMF05A'   '1200',
                             ' ' 'BDC_OKCODE' '=9-PD'.
    WHEN OTHERS.
* FI 전표
      PERFORM dynpro USING : 'X' 'SAPLF040'    '0700',
                             ' ' 'BDC_OKCODE'  '=BL'.
  ENDCASE.

* 저장
  PERFORM dynpro USING : 'X' 'SAPLSPO1'    '0200',
                         ' ' 'BDC_OKCODE'  '=YES'.

  ctu_params-dismode = i_mode.
  ctu_params-updmode = 'S'.
  ctu_params-defsize = 'X'.

* Call 'FBV0' (임시 저장 전표 전기/삭제)
  CALL TRANSACTION 'FBV0' USING gt_bdc
                          OPTIONS  FROM ctu_params
                          MESSAGES INTO gt_msg.

* 변경결과
  READ TABLE gt_msg WITH KEY msgid = 'FP' msgnr = '047'.

  IF sy-subrc = 0.
    CALL FUNCTION 'FORMAT_MESSAGE'
      EXPORTING
        ID        = gt_msg-msgid
        NO        = gt_msg-msgnr
        V1        = gt_msg-msgv1
        V2        = gt_msg-msgv2
        V3        = gt_msg-msgv3
        V4        = gt_msg-msgv4
      IMPORTING
        MSG       = e_return-message
      EXCEPTIONS
        NOT_FOUND = 1
        OTHERS    = 2.

    e_return-type = 'S'.
  ELSE.
    READ TABLE gt_msg WITH KEY msgtyp = 'E'.

    IF sy-subrc <> 0.
      READ TABLE gt_msg INDEX 1.
    ENDIF.

    e_return-type = 'E'.
    MESSAGE ID gt_msg-msgid TYPE gt_msg-msgtyp NUMBER gt_msg-msgnr
            WITH gt_msg-msgv1 gt_msg-msgv2 gt_msg-msgv3 gt_msg-msgv4
    INTO e_return-message.
  ENDIF.

ENDFUNCTION.
