FORM set_main_data .

  DATA: lt_main LIKE TABLE OF gt_main  WITH HEADER LINE,
        lt_type TYPE TABLE OF zfit9999 WITH HEADER LINE.

  CLEAR: gt_main, gt_main[], gt_sum, gt_sum[],
         gt_popup, gt_popup[].

  IF gt_domain[] IS INITIAL.
    CALL METHOD cl_reca_ddic_doma=>get_values
      EXPORTING
        id_name   = 'ZDAPTTP'
        id_langu  = sy-langu
      IMPORTING
        et_values = gt_domain.
  ENDIF.

*-- 신용카드, 현금영수증
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE lt_type
    FROM zfit9999
   WHERE tcode = sy-repid
     AND atype = 'ZFIT0015'
     AND btype = 'APTTP'.

  IF sy-subrc = 0.
    LOOP AT lt_type.
      gr_apttp-sign   = 'I'.
      gr_apttp-option = 'EQ'.
      gr_apttp-low    = lt_type-value.
      APPEND gr_apttp.
      CLEAR: gr_apttp.
    ENDLOOP.
  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF lt_main
    FROM zfit0015
   WHERE gjahr EQ p_gjahr
     AND tterm EQ p_tterm
     AND apttp IN gr_apttp
     AND dmark EQ space.

    SELECT SINGLE
           name1
      INTO (lt_main-name1)
      FROM lfa1
     WHERE lifnr = lt_main-lifnr.

    lt_main-count = 1.

    READ TABLE gt_domain INTO gs_domain
                     WITH KEY domvalue_l = lt_main-apttp.
    lt_main-aptxt = gs_domain-ddtext.

    APPEND lt_main.
    CLEAR: lt_main, gs_domain.
  ENDSELECT.

  LOOP AT lt_main.
    MOVE-CORRESPONDING lt_main TO gt_sum.
    COLLECT gt_sum.

    MOVE-CORRESPONDING lt_main TO gt_popup.
    gt_popup-hwsum = gt_popup-hwbas + gt_popup-hwste.
    APPEND gt_popup.

    CLEAR:  gt_sum, gt_popup.
  ENDLOOP.

  LOOP AT gt_sum.
    READ TABLE lt_main WITH KEY apttp = gt_sum-apttp
                                stcd2 = gt_sum-stcd2.
    MOVE-CORRESPONDING lt_main TO gt_main.
    MOVE-CORRESPONDING gt_sum TO gt_main.

    gt_main-iconn  = icon_led_green.

    IF gt_main-apttp = '07' AND gt_main-cardn IS INITIAL.
      gt_main-iconn  = icon_led_red.
      gt_main-remark = '카드번호 정보가 없습니다.'.
    ENDIF.

    IF gt_main-stcd2 IS INITIAL.
      gt_main-iconn  = icon_led_red.
      gt_main-remark = '사업자등록번호 정보가 없습니다.'.
    ENDIF.

    APPEND gt_main.
    CLEAR: gt_main, lt_main, gt_sum.
  ENDLOOP.

ENDFORM. " SET_MAIN_DATA
