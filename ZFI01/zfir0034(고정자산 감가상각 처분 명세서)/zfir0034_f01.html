FORM set_gv_budat .

  DATA: lv_budat TYPE budat.

  CONCATENATE p_gjahr p_monat '01' INTO lv_budat.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = lv_budat
    IMPORTING
      last_day_of_month = gv_budat
    EXCEPTIONS
      day_in_no_date    = 1
      OTHERS            = 2.

ENDFORM. " SET_GV_BUDAT

FORM select_anep_data .

  CLEAR: gt_anep, gt_anep[].

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_anep
    FROM anep
   WHERE bukrs EQ p_bukrs
     AND anln1 IN s_anln1
     AND gjahr <= p_gjahr.

  LOOP AT gt_anep.
    SELECT SINGLE
           budat
           monat
           aworg
      INTO (gt_anep-budat,
            gt_anep-monat,
            gt_anep-aworg)
      FROM anek
     WHERE bukrs EQ gt_anep-bukrs
       AND anln1 EQ gt_anep-anln1
       AND anln2 EQ gt_anep-anln2
       AND gjahr EQ gt_anep-gjahr
       AND lnran EQ gt_anep-lnran.

    CONCATENATE gt_anep-belnr gt_anep-aworg INTO gt_anep-awkey.

    MODIFY gt_anep.
    CLEAR: gt_anep.
  ENDLOOP.

*  DELETE gt_anep WHERE belnr IS INITIAL.

ENDFORM. " SELECT_ANEP_DATA
