FORM SET_INIT_DATA .

  IF SY-DATUM+4(2) < '04'.
    P_TTERM      = '1기 예정'.
    S_MONAT-LOW  = 1.
    S_MONAT-HIGH = 3.
    APPEND S_MONAT.

  ELSEIF SY-DATUM+4(2) < '07'.
    P_TTERM      = '1기 확정'.
    S_MONAT-LOW  = 4.
    S_MONAT-HIGH = 6.
    APPEND S_MONAT.

  ELSEIF SY-DATUM+4(2) < '10'.
    P_TTERM      = '2기 예정'.
    S_MONAT-LOW  = 7.
    S_MONAT-HIGH = 9.
    APPEND S_MONAT.

  ELSE.
    P_TTERM      = '2기 확정'.
    S_MONAT-LOW  = 10.
    S_MONAT-HIGH = 12.
    APPEND S_MONAT.

  ENDIF.

ENDFORM. " SET_INIT_DATA

FORM SET_LED_MAIN_DATE .

  DATA: LS_ZFIT0015 TYPE ZFIT0015,
        LS_CELLSTYLE TYPE LVC_S_STYL.

  LOOP AT GT_MAIN.
    SELECT SINGLE *
      INTO CORRESPONDING FIELDS OF LS_ZFIT0015
      FROM ZFIT0015
     WHERE BELNR = GT_MAIN-BELNR
       AND GJAHR = GT_MAIN-GJAHR
       AND DMARK = SPACE.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING LS_ZFIT0015 TO GT_MAIN.
      GT_MAIN-HWSUM = GT_MAIN-HWBAS + GT_MAIN-HWSTE.
      GT_MAIN-ICONN = ICON_LED_GREEN.

      LS_CELLSTYLE-FIELDNAME = 'APTTP' .
      LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
      INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.

      LS_CELLSTYLE-FIELDNAME = 'HWBAS' .
      LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
      INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.

      LS_CELLSTYLE-FIELDNAME = 'HWSTE' .
      LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
      INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.
    ELSE.
      GT_MAIN-ICONN = ICON_LED_YELLOW.

      IF GT_MAIN-APTTP = '05' OR GT_MAIN-APTTP = '06'.
      ELSE.
        LS_CELLSTYLE-FIELDNAME = 'HWBAS' .
        LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
        INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.

        LS_CELLSTYLE-FIELDNAME = 'HWSTE' .
        LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
        INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.
      ENDIF.

    CLEAR: GT_MAIN, LS_ZFIT0015, LS_CELLSTYLE.
