      CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
        EXPORTING

          value_org       = 'S'
        TABLES
          value_tab       = gt_bupla
          return_tab      = lt_return
        EXCEPTIONS
          parameter_error = 1
          no_values_found = 2
          OTHERS          = 3.
      IF sy-subrc = 0.
        READ TABLE lt_return INDEX 1.
        ls_item-bupla = lt_return-fieldval.
        MODIFY gt_item FROM ls_item INDEX es_row_no-row_id.
      ENDIF.

          return_tab      = lt_return

          no_values_found = 2

  USING p_data_changed TYPE REF TO cl_alv_changed_data_protocol
          p_modf_field TYPE lvc_s_modi

      e_value     = lv_field.

      READ TABLE gt_domain INTO gs_domain
                       WITH KEY ddtext = ls_item-stext.
      IF sy-subrc = 0.
        ls_item-shkzg = gs_domain-domvalue_l.
        IF ls_item-shkzg = 'H'.

        gv_err = 'X'.
      ENDIF.

    WHEN 'NEWKO'.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = lv_field
        IMPORTING
          output = lv_char10.

      READ TABLE gt_code WITH KEY saknr = lv_char10.
      IF sy-subrc = 0.
        ls_item-newko = gt_code-saknr.
        CALL METHOD p_data_changed->modify_cell
          EXPORTING
            i_row_id    = p_modf_field-row_id
            i_fieldname = 'TXT20'

          EXPORTING
            i_row_id    = p_modf_field-row_id
            i_fieldname = 'TXT20'
            i_value     = ''.
        gv_err = 'X'.
      ENDIF.
*-- 금액(KRW, JPY)

      ls_item-wrbtr = lv_field.
**-- 금액(기타)
*    WHEN 'FRBTR'.
*      ls_item-frbtr = lv_field.
*-- 세금코드

*      IF pr_buga1 = 'X'.
