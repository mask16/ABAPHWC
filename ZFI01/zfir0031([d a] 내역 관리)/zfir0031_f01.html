FORM set_bank_data .

  CLEAR: gt_bank, gt_bank[].

  SELECT DISTINCT
         saknr
         hbkid
    INTO CORRESPONDING FIELDS OF gt_bank
    FROM skb1
   WHERE saknr LIKE '001105%'
     AND bukrs EQ   p_bukrs.

    SELECT SINGLE
           btext
      INTO gt_bank-btext
      FROM zfit0018
     WHERE hbkid EQ gt_bank-hbkid
       AND bukrs EQ p_bukrs.

    IF sy-subrc = 0.
      APPEND gt_bank.
    ENDIF.
    CLEAR: gt_bank.
  ENDSELECT.

  SORT gt_bank BY saknr.

ENDFORM. " SET_RANGES

FORM set_main_data .

  DATA: lt_main LIKE TABLE OF gt_main WITH HEADER LINE,
        lv_date(10) TYPE c.

  CLEAR: gt_main, gt_main[].

  SELECT bukrs
         belnr
         gjahr
         budat
         waers
         xblnr
    INTO CORRESPONDING FIELDS OF TABLE lt_main
    FROM bkpf
   WHERE bukrs EQ p_bukrs
     AND blart EQ 'DY'
     AND stblg EQ space.

  LOOP AT lt_main.
    SELECT zfbdt
           dmbtr
           wrbtr
           zuonr
           hkont
           augbl
      INTO (lt_main-zfbdt,
            lt_main-dmbtr,
            lt_main-wrbtr,
            lt_main-zuonr,
            lt_main-hkont,
            lt_main-augbl)
      FROM bseg
     WHERE bukrs EQ lt_main-bukrs
       AND belnr EQ lt_main-belnr
       AND gjahr EQ lt_main-gjahr.

*-- 거래은행명
      READ TABLE gt_bank WITH KEY saknr = lt_main-hkont.
      IF sy-subrc = 0.
        gt_main-hbkid = gt_bank-hbkid.
        gt_main-btext = gt_bank-btext.
      ENDIF.
*-- 만기일, 금액
      IF lt_main-augbl IS NOT INITIAL.  "반제된 AR
        CLEAR: lv_date.
        lv_date = lt_main-zuonr.
        REPLACE ALL OCCURRENCES OF '.' IN lv_date WITH ''.
        gt_main-zfbdt = lv_date.
        gt_main-dmbtr = gt_main-dmbtr + lt_main-dmbtr.
        gt_main-wrbtr = gt_main-wrbtr + lt_main-wrbtr.
      ENDIF.
*-- 환가료율
      IF lt_main-hkont = '0075350010'.
        gt_main-zuonr = lt_main-zuonr.
      ENDIF.
    ENDSELECT.
*-- 헤더
    gt_main-bukrs = lt_main-bukrs.
    gt_main-belnr = lt_main-belnr.
    gt_main-gjahr = lt_main-gjahr.
    gt_main-budat = lt_main-budat.
    gt_main-waers = lt_main-waers.
    gt_main-xblnr = lt_main-xblnr.
*-- 금액 변환
    WRITE gt_main-wrbtr TO gt_main-zrbtr CURRENCY gt_main-waers.
*-- 키필드 컬러
    PERFORM set_field_color USING: 4 1 1 'BELNR',
                                   4 1 1 'HBKID',
                                   4 1 1 'BTEXT'.

    APPEND gt_main.
    CLEAR: gt_main, lt_main.
  ENDLOOP.

FORM set_sum_data .

  DATA: lt_tmp LIKE TABLE OF gt_sum WITH HEADER LINE.

  DATA: BEGIN OF lt_sum OCCURS 0,
          hbkid TYPE hbkid  ,
          btext TYPE zebtext,
          zmbtr TYPE zedmbtr,              "사용액
        END OF lt_sum.

  CLEAR: gt_main, gt_main[], gt_sum, gt_sum[].

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_main
    FROM zfit0020.

  SORT gt_main BY hbkid ASCENDING budat DESCENDING.

*-- 사용액(만기 미도래 USD 금액)
  LOOP AT gt_main WHERE zfbdt >= sy-datum.
    MOVE-CORRESPONDING gt_main TO lt_sum.
    COLLECT lt_sum.
    CLEAR:  lt_sum.
  ENDLOOP.

  SORT lt_sum BY hbkid.

  LOOP AT lt_sum.
    MOVE-CORRESPONDING lt_sum TO gt_sum.

*-- 한도금액
    SELECT SINGLE
           limit
      INTO gt_sum-limit
      FROM zfit0019
     WHERE bukrs = p_bukrs
       AND hbkid = gt_sum-hbkid
       AND dmark = space.
*-- 사용가능액(한도-사용액)
      gt_sum-znbtr = gt_sum-limit - gt_sum-zmbtr.
