         lt_rows TYPE lvc_t_row.

  CLEAR: gt_excel, gt_excel[], gv_err.

  CALL METHOD gcl_alv->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows[].

  IF lt_rows[] IS INITIAL.
    MESSAGE s009 DISPLAY LIKE 'W'.  "선택된 라인이 없습니다.
    gv_err = 'X'.

  CHECK gv_err IS INITIAL.

  CLEAR: gt_excel.

  gt_excel-kinty  = text-z03.

  gt_excel-name1  = text-z05.
  gt_excel-bnknm  = text-z06.

  gt_excel-bktxt  = text-z09.

  CLEAR: gt_excel.
  data : l_cnt type i.
  clear: l_cnt.
  LOOP AT lt_rows INTO ls_rows.
    READ TABLE gt_main INTO ls_main INDEX ls_rows-index.
    MOVE-CORRESPONDING ls_main TO gt_excel.

    move : l_cnt        to gt_excel-count,
           s_kidno-low  to gt_excel-kidno,
           '1'          to gt_excel-kinty.

    IF pr_jl = 'X'.     "지금내역
      IF ls_main-rbetr < 0.
        ls_main-rbetr = ls_main-rbetr * -1.
      ENDIF.
      WRITE ls_main-rbetr TO gt_excel-amount CURRENCY ls_main-waers.
    ELSEIF pr_dl = 'X'. "상세내역
      IF ls_main-dmbtr < 0.

      ENDIF.
      WRITE ls_main-dmbtr TO gt_excel-amount CURRENCY ls_main-waers.
    ENDIF.
