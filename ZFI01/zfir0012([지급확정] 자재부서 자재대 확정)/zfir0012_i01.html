MODULE user_command_0100 INPUT.

  CLEAR: gv_err.

  CASE sy-ucomm.

    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      LEAVE TO SCREEN 0.

    WHEN 'REFRESH'.
      PERFORM set_main_data.
      PERFORM set_new_ok_code USING 'PBO'.

*-- 지급보류키
    WHEN 'CHANGE'.
      PERFORM select_main_data.
      CHECK gv_err IS INITIAL.
      gsc0101-zlspr = 'P'.
      CALL SCREEN '0101' STARTING AT 5 1.

*-- 지급방법
    WHEN 'CHANGE1'.
      PERFORM select_main_data.
      CHECK gv_err IS INITIAL.
      gsc0102-zlsch = 'T'.
      CALL SCREEN '0102' STARTING AT 5 1.

*-- 만기일
    WHEN 'CHANGE2'.
      PERFORM select_main_data.
      CHECK gv_err IS INITIAL.
      CALL SCREEN '0103' STARTING AT 5 1.

*-- 저장
    WHEN 'SAVE'.
      PERFORM start_bdc_fb09.
      CHECK gv_err IS INITIAL.
      PERFORM set_new_ok_code USING 'PBO'.

    WHEN OTHERS.

  ENDCASE.

ENDMODULE. " USER_COMMAND_0100 INPUT
