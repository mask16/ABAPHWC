FORM SET_LED_MAIN_DATE .

  DATA: LS_ZFIT0013 TYPE ZFIT0013,

        LS_WWBKL LIKE LINE OF GT_WWBKL,

      INTO CORRESPONDING FIELDS OF LS_ZFIT0013
      FROM ZFIT0013
     WHERE PALEDGER = GT_MAIN-PALEDGER
       AND VRGAR    = GT_MAIN-VRGAR
       AND VERSI    = GT_MAIN-VERSI
       AND PERIO    = GT_MAIN-PERIO
       AND PAOBJNR  = GT_MAIN-PAOBJNR
       AND PASUBNR  = GT_MAIN-PASUBNR
       AND BELNR    = GT_MAIN-BELNR
       AND POSNR    = GT_MAIN-POSNR.
    IF SY-SUBRC = 0.
      GT_MAIN-ICONN  = ICON_LED_GREEN.
      GT_MAIN-DMBTR  = LS_ZFIT0013-DMBTR.
      GT_MAIN-SHKONT = LS_ZFIT0013-SHKONT.
      GT_MAIN-HHKONT = LS_ZFIT0013-HHKONT.
      GT_MAIN-ZELNR  = LS_ZFIT0013-ZELNR.
      GT_MAIN-BELNR = LS_ZFIT0013-BELNR.

      LS_CELLSTYLE-FIELDNAME = 'DMBTR' .
      LS_CELLSTYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
      INSERT LS_CELLSTYLE INTO TABLE GT_MAIN-CELLSTYLE.
    ELSE.
      GT_MAIN-ICONN = ICON_LED_YELLOW.
      READ TABLE GT_WWBKL INTO LS_WWBKL WITH KEY WWBKL = GT_MAIN-WWBKL.
      GT_MAIN-SHKONT = LS_WWBKL-SHKONT.
      GT_MAIN-HHKONT = LS_WWBKL-HHKONT.
    ENDIF.

*   장기재고그룹명
    PERFORM GET_TVKGGT_VTEXT USING GT_MAIN-KDKG1
                          CHANGING GT_MAIN-VTEXT.

*  할인율

    PERFORM GET_TVKGGT_VTEXT1 USING GT_MAIN-ARTNR
                          CHANGING GT_MAIN-VV110.

*-- 키필드 컬러
    PERFORM SET_FIELD_COLOR USING: 4 1 1 'ICONN',
                                   4 1 1 'PALEDGER',
                                   4 1 1 'VRGAR',
                                   4 1 1 'PERIO',
                                   4 1 1 'PAOBJNR',
                                   4 1 1 'PASUBNR',
                                   4 1 1 'BELNR',
                                   4 1 1 'POSNR'.
*-- 차변 계정
    SELECT SINGLE
         TXT20
    FROM SKAT
    INTO GT_MAIN-STXT20
    WHERE SAKNR EQ GT_MAIN-SHKONT
     AND SPRAS EQ SY-LANGU.
*-- 대변 계정
    SELECT SINGLE
           TXT20
      FROM SKAT
      INTO GT_MAIN-HTXT20
     WHERE SAKNR EQ GT_MAIN-HHKONT
       AND SPRAS EQ SY-LANGU.
*-- 사업영역
    IF GT_MAIN-GSBER IS INITIAL.
      IF GT_MAIN-SPART = '10' OR GT_MAIN-SPART = '80'.
        GT_MAIN-GSBER = '1000'.
      ELSEIF GT_MAIN-SPART = '20' OR GT_MAIN-SPART = '90'.
        GT_MAIN-GSBER = '1010'.
      ENDIF.
    ENDIF.
*-- 매출액 매출원가
    IF GT_MAIN-ABSMG IS NOT INITIAL.
       GT_MAIN-VV098 = GT_MAIN-VV001 / GT_MAIN-ABSMG.  "매출액/수량
       GT_MAIN-VV099 = GT_MAIN-VV003 / GT_MAIN-ABSMG.  "매출원가/수량

    ENDIF.

    IF  GT_MAIN-VV099 > GT_MAIN-VV098.  " 매출원가>매출액 매출원가(재고단가) 이상 판매시 : 30%,  이하  판매시 : 20% 반영
       GT_MAIN-VV100 = '매출총이익20%'.
    ELSE.
       GT_MAIN-VV100 = '매출총이익30%'.
    ENDIF.



    MODIFY GT_MAIN.
    CLEAR: GT_MAIN, LS_ZFIT0013, LS_CELLSTYLE, LS_WWBKL, LV_P.
  ENDLOOP.

ENDFORM. " SET_LED_MAIN_DATE

*  <--  p2        text
*----------------------------------------------------------------------*
FORM CREATE_DOCUMENT .

  DATA: LS_MAIN LIKE LINE OF GT_MAIN,
        LS_ROWS TYPE LVC_S_ROW,
        LT_ROWS TYPE LVC_T_ROW..

  CLEAR: GV_ERR.

  CALL METHOD GCL_ALV->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROWS[].

  IF LT_ROWS[] IS INITIAL.
    MESSAGE S009 DISPLAY LIKE 'W'.  "선택된 라인이 없습니다.
    GV_ERR = 'X'.
  ELSE.
    LOOP AT LT_ROWS INTO LS_ROWS.
      READ TABLE GT_MAIN INTO LS_MAIN INDEX LS_ROWS-INDEX.
      IF LS_MAIN-ICONN = ICON_LED_GREEN.
        MESSAGE I087 DISPLAY LIKE 'W' WITH LS_ROWS-INDEX.
        "& 번째 라인은 이미 회계처리되어 있습니다.
        GV_ERR = 'X'.
        EXIT.
      ENDIF.
    ENDLOOP.
  ENDIF.

  CHECK GV_ERR IS INITIAL.

  CLEAR: GT_BDC, GT_BDC[].

  LOOP AT LT_ROWS INTO LS_ROWS.
    READ TABLE GT_MAIN INTO LS_MAIN INDEX LS_ROWS-INDEX.
    MOVE-CORRESPONDING LS_MAIN TO GT_BDC.
    APPEND GT_BDC.
    CLEAR: GT_BDC, LS_MAIN.
  ENDLOOP.

  CALL SCREEN '0101' STARTING AT 5 1.

ENDFORM. " CREATE_DOCUMENT
