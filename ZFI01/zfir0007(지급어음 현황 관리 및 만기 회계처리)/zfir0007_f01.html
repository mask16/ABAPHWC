FORM set_init_data .

*-- 만기일
  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = sy-datum
    IMPORTING
      last_day_of_month = s_zfbdt-high
    EXCEPTIONS
      day_in_no_date    = 1
      OTHERS            = 2.

  CONCATENATE sy-datum+0(6) '01' INTO s_zfbdt-low.

  APPEND s_zfbdt.

*-- 특별 GL 지시자
  s_umskz-sign   = 'I'.
  s_umskz-option = 'EQ'.
  s_umskz-low    = 'U'.
  APPEND s_umskz.

  s_umskz-sign   = 'I'.
  s_umskz-option = 'EQ'.
  s_umskz-low    = 'V'.
  APPEND s_umskz.

ENDFORM. " SET_INIT_DATA

FORM set_main_data .

  CLEAR: gt_main, gt_main[].

*-- 미결항목(주요일자와 상관 없이 미결 지급어음 항목만)
  IF pr_open = 'X'.
    SELECT bukrs
           gjahr
           lifnr
           belnr
           buzei
           zuonr
           umskz
           bldat
           budat
           dmbtr
           zfbdt
      INTO CORRESPONDING FIELDS OF TABLE gt_main
      FROM bsik
     WHERE bukrs IN s_bukrs    "회사코드
       AND gjahr EQ p_gjahr    "회계년도
       AND budat <= p_sdate    "주요일자
       AND zfbdt IN s_zfbdt    "만기일
       AND bldat IN s_bldat    "증빙일
       AND budat IN s_budat    "전기일
       AND blart IN gr_blart   "전표유형
       AND bupla IN gr_bupla   "사업장
       AND belnr IN s_belnr    "전표번호
       AND zuonr IN s_zuonr    "어음번호
       AND umskz IN s_umskz.   "지급어음

    LOOP AT gt_main.
      SELECT SINGLE
             name1
        INTO gt_main-name1
        FROM lfa1
       WHERE lifnr = gt_main-lifnr.

      gt_main-zmbtr = gt_main-dmbtr * 100.

      SELECT SINGLE
             ltext
        INTO gt_main-ltext
        FROM t074t
       WHERE spras = sy-langu
         AND koart = 'K'
         AND shbkz = gt_main-umskz.

      PERFORM set_field_color USING: 4 1 1 'LIFNR',
                                     4 1 1 'NAME1',
                                     4 1 1 'BELNR',
                                     4 1 1 'BUZEI',
                                     4 1 1 'LTEXT'.

      MODIFY gt_main.
      CLEAR: gt_main.
    ENDLOOP.
  ENDIF.

*-- 반제완료(주요일자를 기준으로 완료된 지급어음 항목)
  IF pr_clear = 'X'.
    SELECT bukrs
           gjahr
           lifnr
           belnr
           buzei
           zuonr
           umskz
           bldat
           budat
           dmbtr
           zfbdt
           augdt  "반제일
           augbl  "반제전표
      INTO CORRESPONDING FIELDS OF TABLE gt_main
      FROM bsak
     WHERE bukrs IN s_bukrs    "회사코드
       AND gjahr EQ p_gjahr    "회계년도
*       AND budat <= p_sdate    "주요일자
       AND augdt <= p_sdate    "주요일자
       AND zfbdt IN s_zfbdt    "만기일
       AND shkzg EQ 'H'
       AND bldat IN s_bldat    "증빙일
       AND budat IN s_budat    "전기일
       AND blart IN gr_blart   "전표유형
       AND bupla IN gr_bupla   "사업장
       AND belnr IN s_belnr    "전표번호
       AND zuonr IN s_zuonr    "어음번호
       AND umskz IN s_umskz.   "지급어음

    LOOP AT gt_main.
      SELECT SINGLE
             name1
        INTO gt_main-name1
        FROM lfa1
       WHERE lifnr = gt_main-lifnr.

      gt_main-zmbtr = gt_main-dmbtr * 100.

      SELECT SINGLE
             ltext
        INTO gt_main-ltext
        FROM t074t
       WHERE spras = sy-langu
         AND koart = 'K'
         AND shbkz = gt_main-umskz.

*-- 반제전표 상대 계정
      SELECT SINGLE
             hkont
        INTO gt_main-saknr
        FROM bseg
       WHERE bukrs = gt_main-bukrs
         AND belnr = gt_main-augbl
         AND gjahr = gt_main-augdt+0(4)
         AND shkzg = 'H'.
      IF sy-subrc = 0.
        SELECT SINGLE
               txt20
          INTO gt_main-txt20
          FROM skat
         WHERE spras = sy-langu
           AND ktopl = '1000'
           AND saknr = gt_main-saknr.
      ENDIF.

      MODIFY gt_main.
      CLEAR: gt_main.
    ENDLOOP.

    SORT gt_main BY lifnr name1 belnr.

*-- 역분개 데이터 삭제
    LOOP AT gt_main.
      IF gt_main-stblg IS NOT INITIAL.
        DELETE gt_main INDEX sy-tabix.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDFORM. " SET_MAIN_DATA
