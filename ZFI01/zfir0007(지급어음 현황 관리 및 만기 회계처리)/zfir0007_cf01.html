FORM f_user_hotspot USING ps_row_id TYPE lvc_s_row
                              ps_column_id  TYPE lvc_s_col
                              ps_row_no  TYPE lvc_s_roid.

  DATA: ls_main LIKE LINE OF gt_main.

  CASE ps_column_id.
    WHEN 'BELNR'.
      READ TABLE gt_main INTO ls_main INDEX ps_row_id-index.
      IF ls_main-belnr IS NOT INITIAL.
        SET PARAMETER ID 'BLN' FIELD ls_main-belnr.
        SET PARAMETER ID 'BUK' FIELD ls_main-bukrs.
        SET PARAMETER ID 'GJR' FIELD ls_main-gjahr.
        CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
      ENDIF.

    WHEN 'AUGBL'.
      READ TABLE gt_main INTO ls_main INDEX ps_row_id-index.
      IF ls_main-augbl IS NOT INITIAL.
        SET PARAMETER ID 'BLN' FIELD ls_main-augbl.
        SET PARAMETER ID 'BUK' FIELD ls_main-bukrs.
        SET PARAMETER ID 'GJR' FIELD ls_main-augdt+0(4).
        CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
      ENDIF.

  ENDCASE.

ENDFORM. " F_USER_HOTSPOT

FORM check_data_proc USING pr_data_changed
                      TYPE REF TO cl_alv_changed_data_protocol
                      CHANGING error_in_data.

  DATA: ls_modf_field  TYPE lvc_s_modi.

  CLEAR: gv_err.

  LOOP AT pr_data_changed->mt_good_cells INTO ls_modf_field.
    CASE ls_modf_field-fieldname.
*-- 계정 사업영역 계좌유형
      WHEN 'SAKNR'.
        PERFORM check_data_to_main USING pr_data_changed

                                   CHANGING error_in_data.
      WHEN OTHERS.

    ENDCASE.
  ENDLOOP.

  CALL METHOD cl_gui_cfw=>set_new_ok_code

      new_code = 'PBO'.

ENDFORM. " CHECK_DATA_PROC
*&---------------------------------------------------------------------*
*&      Form  CHECK_DATA_TO_MAIN
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_PR_DATA_CHANGED  text
*      -->P_LS_MODF_FIELD  text
*      <--P_ERROR_IN_DATA  text
*----------------------------------------------------------------------*
FORM check_data_to_main USING p_data_changed TYPE REF TO cl_alv_changed_data_protocol
           p_modf_field TYPE lvc_s_modi
     CHANGING p_error_in_data.

  DATA: ls_main LIKE LINE OF gt_main,
        lv_field TYPE string,
        lv_saknr TYPE saknr.

  READ TABLE gt_main INTO ls_main INDEX p_modf_field-row_id.

*-- CELL 에서 변경된 필드값을 가지고 온다.
  CALL METHOD p_data_changed->get_cell_value
    EXPORTING
      i_row_id    = p_modf_field-row_id
      i_fieldname = p_modf_field-fieldname
    IMPORTING
      e_value     = lv_field.

  CASE p_modf_field-fieldname.
    WHEN 'SAKNR'.
      IF lv_field IS NOT INITIAL.
        SELECT SINGLE saknr
          INTO lv_saknr
          FROM skb1
         WHERE saknr = lv_field
           AND xloeb = space.
        IF lv_saknr IS NOT INITIAL.
          ls_main-saknr = lv_field.
        ELSE.
          MESSAGE i000 DISPLAY LIKE 'W'
            WITH '등록되지 않았거나 삭제된 계정입니다.'.
          gv_err = 'X'.
        ENDIF.
      ENDIF.

    WHEN OTHERS.

  ENDCASE.

*-- 변경한 내용을 화면 ALV 에 갱신한다.
  IF gv_err IS INITIAL.
    SELECT SINGLE txt20
      INTO ls_main-txt20
      FROM skat
     WHERE spras = sy-langu
       AND ktopl = '1000'
       AND saknr = ls_main-saknr.

    MODIFY gt_main FROM ls_main INDEX p_modf_field-row_id.
  ENDIF.

ENDFORM. " CHECK_DATA_TO_MAIN
