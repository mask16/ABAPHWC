  DATA: i_files TYPE sbdst_files.

  DATA: wa_file LIKE LINE OF i_files.
  DATA: lv_path(255) TYPE c.

      'WORKBOOKS' = books.

  CONCATENATE wa_file-directory '\' pv_excel '.xls' INTO lv_path.

  CALL METHOD OF
      books
      'OPEN' = book
    EXPORTING

  FREE: sheet1, book, books, excel.

ENDFORM. " CALL_OAER_EXCEL

FORM set_main_data .

  DATA: ls_excel LIKE LINE OF gt_excel,
        lv_lines TYPE i.

  SORT: gt_excel BY sqn ASCENDING dtype DESCENDING.

  LOOP AT gt_excel.
    MOVE-CORRESPONDING gt_excel TO gt_main.
*-- 엑셀양식에 헤더가 비워져 있다면
    IF gt_main-dtype IS INITIAL OR
       gt_main-bukrs IS INITIAL.
      LOOP AT gt_excel INTO ls_excel WHERE sqn = gt_main-sqn
                                       AND dtype IS NOT INITIAL
                                       AND bukrs IS NOT INITIAL.
        gt_main-dtype = ls_excel-dtype.
        gt_main-bukrs = ls_excel-bukrs.
        gt_main-bldat = ls_excel-bldat.
        gt_main-blart = ls_excel-blart.
        gt_main-budat = ls_excel-budat.
        gt_main-waers = ls_excel-waers.
        gt_main-kursf = ls_excel-kursf.
        gt_main-xblnr = ls_excel-xblnr.
        gt_main-bktxt = ls_excel-bktxt.
        EXIT.
      ENDLOOP.
      CLEAR: ls_excel.
    ENDIF.

    APPEND gt_main.
    CLEAR: gt_main.

    lv_lines = lv_lines + 1.

    AT END OF sqn.
      gt_result-sqn   = gt_excel-sqn.
      gt_result-lines = lv_lines.
      APPEND gt_result.
      CLEAR: gt_result, lv_lines.
    ENDAT.
  ENDLOOP.

  SORT: gt_main BY sqn.
  SORT: gt_result BY sqn.

ENDFORM. " SET_MAIN_DATA
