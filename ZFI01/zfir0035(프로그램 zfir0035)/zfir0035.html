*&---------------------------------------------------------------------*
*& Report Z_BSEG_EBELN_QUERY
*&---------------------------------------------------------------------*
*& This program queries the BSEG table based on BELNR and GJAHR
*& and displays the corresponding EBELN values.
*&
*& Program by Gemini
*&---------------------------------------------------------------------*
REPORT Z_BSEG_EBELN_QUERY.

*======================================================================*
* SELECTION SCREEN
*======================================================================*
SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-t01.
  PARAMETERS: p_belnr TYPE bseg-belnr,  " 입력: 전표 번호
              p_gjahr TYPE bseg-gjahr.  " 입력: 회계 연도
SELECTION-SCREEN END OF BLOCK b1.

*======================================================================*
* DATA DECLARATION
*======================================================================*
DATA: BEGIN OF gs_bseg,
        ebeln TYPE bseg-ebeln,
      END OF gs_bseg.

DATA: gt_bseg LIKE TABLE OF gs_bseg.

*======================================================================*
* INITIALIZATION
*======================================================================*
INITIALIZATION.
  " 선택 화면 텍스트 설정
*  TEXT-t01 = '조회 조건 입력'.

*======================================================================*
* AT SELECTION-SCREEN
*======================================================================*
AT SELECTION-SCREEN ON p_belnr.
  IF p_belnr IS INITIAL.
    MESSAGE '전표 번호를 입력하세요.' TYPE 'E'.
  ENDIF.

AT SELECTION-SCREEN ON p_gjahr.
  IF p_gjahr IS INITIAL.
    MESSAGE '회계 연도를 입력하세요.' TYPE 'E'.
  ENDIF.

*======================================================================*
* START-OF-SELECTION
*======================================================================*
START-OF-SELECTION.

  " Check if both parameters are provided
  IF p_belnr IS INITIAL OR p_gjahr IS INITIAL.
    MESSAGE '두 조건을 모두 입력해야 합니다.' TYPE 'I'.
    LEAVE TO SCREEN 0.
  ENDIF.

  " Select EBELN from BSEG table
  SELECT ebeln
    INTO CORRESPONDING FIELDS OF TABLE gt_bseg
    FROM bseg
    WHERE belnr = p_belnr
    AND   gjahr = p_gjahr.

  " Display results
  IF sy-subrc = 0.
    ULINE.
    WRITE: / '품목 번호(EBELN)'.
    ULINE.

    LOOP AT gt_bseg INTO gs_bseg.
      WRITE: / gs_bseg-ebeln.
    ENDLOOP.
  ELSE.
    WRITE: / '입력하신 조건과 일치하는 데이터가 없습니다.'.
  ENDIF.
