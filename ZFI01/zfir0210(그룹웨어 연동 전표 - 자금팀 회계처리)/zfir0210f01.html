*&      FORM  SELECT_PROCESS
*&---------------------------------------------------------------------*
FORM SELECT_PROCESS .

  DATA: LV_LINE TYPE I.
*----------------------------------------------------------------------*

  CLEAR:GT_DISPLAY, GT_DISPLAY[].
  SELECT A~SAPID
         A~GW_STATUS
         B~BUKRS
         B~GJAHR
         B~BELNR
         B~TSL AS WRBTR
         B~HSL AS DMBTR
         B~REQ_GUBUN
         B~ZFBDT
         A~REQ_PERNR
         A~REQ_NAME
         A~APPR_PERNR
         A~APPR_NAME
    INTO CORRESPONDING FIELDS OF TABLE GT_DISPLAY
    FROM ZFIT0041 AS A INNER JOIN ZFIT0042 AS B
                               ON B~SAPID EQ A~SAPID
                              AND B~BUKRS EQ A~BUKRS
                              AND B~GJAHR EQ A~GJAHR
    WHERE A~BUKRS     EQ PA_BUKRS
      AND A~AEDAT     IN SO_DATE
      AND A~GW_STATUS EQ '4'
      AND B~BLR_MSGTY EQ SPACE.

  LV_LINE = LINES( GT_DISPLAY ).
  IF LV_LINE IS INITIAL.
    "-- 조회 조건에 해당되는 데이터가 없습니다.
    MESSAGE S000 DISPLAY LIKE 'E' WITH TEXT-003.
    GV_EXIT = ABAP_TRUE.
  ELSE.
    " 상세 데이터 추출 및 가공.
    PERFORM GET_BKPF.
    "-- LV_LINE 건의 데이터를 조회하였습니다.
    MESSAGE S000 DISPLAY LIKE 'S' WITH LV_LINE TEXT-010.
  ENDIF.

ENDFORM. " SELECT_PROCESS

*&      Form  GET_BKPF
*&---------------------------------------------------------------------*
FORM GET_BKPF .

  DATA: LS_BKPF TYPE BKPF.

  FIELD-SYMBOLS <FS> LIKE LINE OF GT_DISPLAY.
*----------------------------------------------------------------------*

  LOOP AT GT_DISPLAY ASSIGNING <FS>.
    CLEAR LS_BKPF.
    SELECT SINGLE *
      FROM BKPF
      INTO CORRESPONDING FIELDS OF LS_BKPF
      WHERE BUKRS EQ <FS>-BUKRS
        AND BELNR EQ <FS>-BELNR
        AND GJAHR EQ <FS>-GJAHR.

    IF SY-SUBRC EQ 0.
      <FS>-ICON = ICON_LED_YELLOW.
      IF <FS>-ERRYN EQ 'S'.
        <FS>-ICON = ICON_LED_GREEN.
      ELSEIF <FS>-ERRYN EQ 'E'.
        <FS>-ICON = ICON_LED_RED.
      ENDIF.

      MOVE-CORRESPONDING LS_BKPF TO <FS>.

      " 전표유형명.
      SELECT SINGLE LTEXT
        FROM T003T
        INTO <FS>-LTEXT
        WHERE BLART EQ LS_BKPF-BLART
          AND SPRAS EQ SY-LANGU.

      " 결재상태명.
      CALL FUNCTION 'STF4_GET_DOMAIN_VALUE_TEXT'
        EXPORTING
          IV_DOMNAME      = 'ZDGW_STATUS'
          IV_VALUE        = <FS>-GW_STATUS
        IMPORTING
          EV_VALUE_TEXT   = <FS>-STATUSTXT
        EXCEPTIONS
          VALUE_NOT_FOUND = 1
          OTHERS          = 2.

      " 전표상태명.
      CALL FUNCTION 'STF4_GET_DOMAIN_VALUE_TEXT'
        EXPORTING
          IV_DOMNAME      = 'BSTAT'
          IV_VALUE        = LS_BKPF-BSTAT
        IMPORTING
          EV_VALUE_TEXT   = <FS>-BSTXT
        EXCEPTIONS
          VALUE_NOT_FOUND = 1
          OTHERS          = 2.

      " 작성부서.
      SELECT SINGLE B~DEPARTMENT AS NAME_TEXT
         INTO <FS>-NAME_TEXT
         FROM USR21 AS A INNER JOIN ADCP AS B
                                 ON A~PERSNUMBER EQ B~PERSNUMBER
         WHERE A~BNAME EQ LS_BKPF-USNAM.
    ENDIF.
  ENDLOOP.

ENDFORM. " GET_BKPF
