FORM set_main_data .

  CLEAR: gt_main, gt_main[].

  SELECT *
    FROM tcurr
    INTO CORRESPONDING FIELDS OF TABLE gt_main
   WHERE kurst IN s_kurst
     AND fcurr IN s_fcurr
     AND tcurr IN s_tcurr
     AND gdatu IN gr_gdatu.

  SORT gt_main BY kurst fcurr tcurr gdatu.

  data : lt_tcurf like tcurf occurs 0 with header line.
  clear: lt_tcurf, lt_tcurf[].
  select * into table lt_tcurf
    from tcurf
     for all entries in gt_main
   where kurst = gt_main-kurst
     and fcurr = gt_main-fcurr
     and tcurr = gt_main-tcurr.

  sort lt_tcurf by kurst fcurr tcurr gdatu DESCENDING.

  loop at gt_main.
    read table lt_tcurf with key kurst = gt_main-kurst
                                 fcurr = gt_main-fcurr
                                 tcurr = gt_main-tcurr
                                 BINARY SEARCH.
    if sy-subrc = 0.
      gt_main-FFACT = lt_tcurf-FFACT.
      gt_main-TFACT = lt_tcurf-TFACT.
    endif.
    modify gt_main.
  endloop.

ENDFORM. " SET_MAIN_DATA

FORM conversion_gdatu.

  DATA: lv_lines TYPE i.

  DESCRIBE TABLE s_datum[] LINES lv_lines.

  IF s_datum[] IS NOT INITIAL.
    IF s_datum-high IS INITIAL AND lv_lines = 1.
      gr_gdatu-sign   = s_datum-sign.
      gr_gdatu-option = s_datum-option.
      CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
        EXPORTING
          input  = s_datum-low
        IMPORTING
          output = gr_gdatu-low.
      APPEND gr_gdatu.
    ELSEIF s_datum[] IS NOT INITIAL AND lv_lines > 1.
      LOOP AT s_datum.
        gr_gdatu-sign   = s_datum-sign.
        gr_gdatu-option = s_datum-option.
        CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
          EXPORTING
            input  = s_datum-low
          IMPORTING
            output = gr_gdatu-low.
        APPEND gr_gdatu.
        CLEAR: gr_gdatu.
      ENDLOOP.
    ELSE.
      LOOP AT s_datum.
        gr_gdatu-sign   = s_datum-sign.
        gr_gdatu-option = s_datum-option.
        IF s_datum-low IS NOT INITIAL.
          CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
            EXPORTING
              input  = s_datum-low
            IMPORTING
              output = gr_gdatu-high.
        ENDIF.
        IF s_datum-high IS NOT INITIAL.
          CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
            EXPORTING
              input  = s_datum-high
            IMPORTING
              output = gr_gdatu-low.
        ENDIF.
        APPEND gr_gdatu.
        CLEAR: gr_gdatu.
      ENDLOOP.
    ENDIF.
  ENDIF.

ENDFORM. " CONVERSION_DATE
