FORM f_user_hotspot USING ps_row_id TYPE lvc_s_row
                              ps_column_id  TYPE lvc_s_col
                              ps_row_no  TYPE lvc_s_roid.

  CLEAR: gs_main.

  CASE ps_column_id.
    WHEN 'COUNT'.
      READ TABLE gt_main INTO gs_main INDEX ps_row_id-index.
      IF gs_main-count IS NOT INITIAL.
        CLEAR: gt_0200, gt_0200[].

        LOOP AT gt_popup WHERE zstcd = gs_main-zstcd
                            AND type1 = gs_main-type1.
          MOVE-CORRESPONDING gt_popup TO gt_0200.
          IF gt_0200-lifnr IS NOT INITIAL.
            gt_0200-kunnr = gt_0200-lifnr.
          ENDIF.
          APPEND gt_0200.
          CLEAR: gt_0200.
        ENDLOOP.

        CALL SCREEN '0200' STARTING AT 5 1.
      ENDIF.

  ENDCASE.

ENDFORM. " F_USER_HOTSPOT
