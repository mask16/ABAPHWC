*  <--  p2        text
*----------------------------------------------------------------------*
FORM set_maechul_data .

  DATA: lt_main LIKE TABLE OF gt_main WITH HEADER LINE,
        lt_type TYPE TABLE OF zfit9999 WITH HEADER LINE.

  CLEAR: gr_arttp, gr_arttp[], gt_sum, gt_sum[].

*-- 매출-데이터구분
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE lt_type
    FROM zfit9999
   WHERE tcode = sy-repid
     AND atype = 'ZFIT0012'
     AND btype = 'ARTTP'.

  IF sy-subrc = 0.
    LOOP AT lt_type.
      gr_arttp-sign   = 'I'.
      gr_arttp-option = 'EQ'.
      gr_arttp-low    = lt_type-value.
      APPEND gr_arttp.
      CLEAR: gr_arttp.
    ENDLOOP.
  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF lt_main
    FROM zfit0012
   WHERE gjahr EQ p_gjahr
     AND tterm EQ p_tterm
     AND arttp IN gr_arttp
     AND dmark EQ space.

    lt_main-type1 = '매출'.
    lt_main-count = 1.

    IF lt_main-stcd1 IS NOT INITIAL.
      lt_main-zstcd = lt_main-stcd1.
      lt_main-type2 = '주민'.
    ENDIF.

    IF lt_main-stcd2 IS NOT INITIAL.
      lt_main-zstcd = lt_main-stcd2.
      lt_main-type2 = '사업자'.
    ENDIF.

    SELECT SINGLE
           name1
           j_1kftbus
           j_1kftind
      INTO (lt_main-name1,
            lt_main-rece-uptae,
            lt_main-rece-upjong)
      FROM kna1
     WHERE kunnr = lt_main-kunnr.

    IF lt_main-kunnr < '0000100000'.
      SELECT SINGLE
             sangho
        INTO lt_main-name1
        FROM zfit0008
       WHERE clientno = lt_main-zstcd.
    ENDIF.

    lt_main-rece-saupjano = lt_main-zstcd.
    lt_main-rece-sangho   = lt_main-name1.

    APPEND lt_main.
    CLEAR: lt_main.
  ENDSELECT.

  LOOP AT lt_main.
    gt_sum-zstcd = lt_main-zstcd.
    gt_sum-hwbas = lt_main-hwbas.
    gt_sum-hwste = lt_main-hwste.
    gt_sum-count = lt_main-count.
    COLLECT gt_sum.

    MOVE-CORRESPONDING lt_main TO gt_popup.
    gt_popup-hwsum = gt_popup-hwbas + gt_popup-hwste.
    APPEND gt_popup.

    CLEAR: gt_sum, gt_popup.
  ENDLOOP.

  SORT gt_sum BY zstcd.

  LOOP AT gt_sum.
    READ TABLE lt_main WITH KEY zstcd = gt_sum-zstcd.
    MOVE-CORRESPONDING lt_main TO gt_main.
    MOVE-CORRESPONDING gt_sum TO gt_main.

    gt_main-amt = gt_main-hwbas * 100.
    gt_main-vat = gt_main-hwste * 100.

    APPEND gt_main.
    CLEAR: gt_main, lt_main, gt_sum.
  ENDLOOP.

ENDFORM. " SET_MAECHUL_DATA

*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM merge_main_data .

  DATA: lv_sqn TYPE char04,
        lv_type1 TYPE char02,
        lv_eletp TYPE zeeletp.

  SORT gt_main BY type1 eletp.

  lv_sqn = 1.                         "구분, 세금계산서 형태 별 순번

  LOOP AT gt_main.
    IF sy-tabix = 1.
      lv_type1 = gt_main-type1.
      lv_eletp = gt_main-eletp.
    ENDIF.

    IF lv_type1 <> gt_main-type1.
      lv_type1 = gt_main-type1.
      lv_sqn = 1.
    ELSE.
      IF lv_eletp <> gt_main-eletp.
        lv_eletp = gt_main-eletp.
        lv_sqn = 1.
      ENDIF.
    ENDIF.

    gt_main-iconn  = icon_led_green.
    gt_main-sqn    = lv_sqn.
    gt_main-client = 1.               "거래처수 계산을 위한 세팅

    IF gt_main-type2 IS INITIAL.
      gt_main-iconn  = icon_led_red.
      gt_main-remark = '사업자/주민등록번호 정보가 없습니다.'.
    ENDIF.

    IF gt_main-eletp IS INITIAL.
      gt_main-iconn  = icon_led_red.
      gt_main-remark = '실물/전자 정보가 없습니다.'.
    ENDIF.

    MODIFY gt_main TRANSPORTING sqn client iconn remark.
    lv_sqn = lv_sqn + 1.
  ENDLOOP.

ENDFORM. " MERGE_MAIN_DATA

*&      Form  SET_MK
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_GT_MAIN_AMT  text
*      <--P_GT_2_AMT  text
*----------------------------------------------------------------------*
FORM set_mk USING p_bas
             CHANGING p_amt.


  DATA : lv_amt(14) TYPE c,
         lv_amt_p TYPE p DECIMALS 0,
         lv_offset TYPE i.

  DATA : lv_len TYPE i,
         lv_len_r TYPE i,
         lv_len_z TYPE i,
         lv_zero_str TYPE string.

  IF p_bas < 0.

    lv_amt_p = p_bas * -1 .
    MOVE lv_amt_p TO lv_amt.
    SHIFT lv_amt LEFT DELETING LEADING space.
    lv_offset = strlen( lv_amt ) - 1.

    CASE lv_amt+lv_offset(1).
      WHEN '0'.
        lv_amt+lv_offset(1) = '}'.
      WHEN '1'.
        lv_amt+lv_offset(1) = 'J'.
      WHEN '2'.
        lv_amt+lv_offset(1) = 'K'.
      WHEN '3'.
        lv_amt+lv_offset(1) = 'L'.
      WHEN '4'.
        lv_amt+lv_offset(1) = 'M'.
      WHEN '5'.
        lv_amt+lv_offset(1) = 'N'.
      WHEN '6'.
        lv_amt+lv_offset(1) = 'O'.
      WHEN '7'.
        lv_amt+lv_offset(1) = 'P'.
      WHEN '8'.
        lv_amt+lv_offset(1) = 'Q'.
      WHEN '9'.
        lv_amt+lv_offset(1) = 'R'.
    ENDCASE.

  ELSE.

    WRITE p_bas TO lv_amt DECIMALS 0 NO-GROUPING.
    SHIFT lv_amt LEFT DELETING LEADING space.

  ENDIF.

  DESCRIBE FIELD p_amt LENGTH lv_len IN CHARACTER MODE.

  lv_len_r = strlen( lv_amt ).
  lv_len_z = lv_len - lv_len_r.

  DO lv_len_z TIMES.
    CONCATENATE lv_zero_str '0' INTO lv_zero_str.
  ENDDO.

  CONCATENATE lv_zero_str lv_amt INTO p_amt.
  CONDENSE p_amt.

ENDFORM. " SET_MK_BASE
