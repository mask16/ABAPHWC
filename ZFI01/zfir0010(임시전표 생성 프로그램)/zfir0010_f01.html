MODULE modify_gs_header-kursf INPUT.

  TABLES: tcurr.

  DATA: lv_date TYPE sy-datum,
        lv_week LIKE scal-indicator,
        lv_gdatu  LIKE tcurr-gdatu.

  DATA: BEGIN OF lt_tcurr OCCURS 0,
          datum TYPE sy-datum,
          gdatu TYPE gdatu_inv,
          ukurs TYPE ukurs_curr,
        END OF lt_tcurr.

  CLEAR: lv_date, lv_week, lv_gdatu,
         lt_tcurr, lt_tcurr[].

    gs_vendor-wrbtr = 0.

*-- 요일 가져오기

      EXPORTING

      IMPORTING
        day  = lv_week.

    CASE lv_week.
      WHEN '6'.    "토요일

*-- 가장 최근 날짜의 환율정보 가져오기

           ukurs
      INTO (lt_tcurr-gdatu,
            lt_tcurr-ukurs)
      FROM tcurr
     WHERE kurst = 'M'
       AND fcurr = gs_header-waers
       AND tcurr = 'KRW'
       AND gdatu => lv_gdatu.

*      CALL FUNCTION 'CONVERSION_EXIT_INVDT_OUTPUT'

*        IMPORTING

  ENDIF.
