MODULE user_command_0100 INPUT.

  DATA: lv_field TYPE fieldname.

  CLEAR: gv_err, lv_field.

  CASE sy-ucomm.

    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      LEAVE TO SCREEN 0.

    WHEN 'SAVE'.


      PERFORM check_data_chaged.
      PERFORM check_document_data.
      CHECK gv_err IS INITIAL.
      PERFORM posting_document.

    WHEN 'NEWR'.
      CLEAR : gs_header, gs_vendor, gt_item, gt_item[].
*      clear : BKPF.
      PERFORM set_header_data.
      PERFORM set_vendor_data.
      PERFORM set_item_data.
      PERFORM set_new_ok_code USING 'PBO'.

    WHEN 'SELECT'.
      IF gs_vendor-hkont IS INITIAL.
        MESSAGE s042 DISPLAY LIKE 'W'.
        "공급업제 정보란에 공급업체를 입력하셔야 합니다.
      ELSE.
        PERFORM set_hkont_data.
        CALL SCREEN '0200' STARTING AT 5 1.
      ENDIF.

    WHEN 'HDC'.
      GET CURSOR FIELD lv_field.

      CASE lv_field.
        WHEN 'GS_HEADER-BELNR'.
          IF gs_header-belnr IS NOT INITIAL.
            SET PARAMETER ID 'BLN' FIELD gs_header-belnr.
            SET PARAMETER ID 'BUK' FIELD gs_header-bukrs.
            SET PARAMETER ID 'GJR' FIELD gs_header-budat+0(4).
            CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN OTHERS.

      ENDCASE.

    WHEN OTHERS.

  ENDCASE.

ENDMODULE. " USER_COMMAND_0100 INPUT
