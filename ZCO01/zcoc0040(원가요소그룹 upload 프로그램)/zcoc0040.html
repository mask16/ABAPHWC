FORM file_upload .
  CLEAR: gs_data, gt_data[].

  DATA: l_result,
        l_file  TYPE string."localfile.

  DATA: lt_tab TYPE TABLE OF alsmex_tabline,
        ls_tab TYPE alsmex_tabline.

  l_file = p_path.

  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = l_file
    RECEIVING
      result               = l_result
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4.

  IF l_result IS INITIAL.
    MESSAGE i003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE p_path TO UPPER CASE.
  IF p_path NS '.XLS'.
    MESSAGE i004.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_path
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 10
      i_end_row               = 65536
    TABLES
      intern                  = lt_tab
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    MESSAGE i007.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK lt_tab[] IS NOT INITIAL.
  SORT lt_tab BY row col.

  DELETE lt_tab WHERE row IS INITIAL AND col IS INITIAL.
  CHECK lt_tab[] IS NOT INITIAL.

  FIELD-SYMBOLS: <fs>.

  LOOP AT lt_tab INTO ls_tab.
    ASSIGN COMPONENT ls_tab-col OF STRUCTURE gs_data TO <fs>.
    <fs> = ls_tab-value.

    AT END OF row.
*      gs_data-icon = '@00@'.
      PERFORM conversion_alpha_input CHANGING:  gs_data-kstar_f,
                                                gs_data-kstar_t.

      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDAT.

    CLEAR ls_tab.
  ENDLOOP.


* --> Adjust Data Layout.------------------
  DATA: l_index TYPE i,
        ls_temp LIKE gs_data.

  LOOP AT gt_data INTO gs_data.
    l_index = sy-tabix.

    IF gs_data-field1 IS INITIAL.
      MOVE: ls_temp-field1  TO gs_data-field1,
            ls_temp-desc1   TO gs_data-desc1.
    ENDIF.

    MOVE-CORRESPONDING gs_data TO ls_temp.
    MODIFY gt_data FROM gs_data INDEX l_index.

    CLEAR: gs_data, l_index.
  ENDLOOP.
ENDFORM. " FILE_UPLOAD

FORM celemgroup_upload_processing .
  DATA: lt_get_nodes  TYPE TABLE OF bapiset_hier,
        lt_nodes      TYPE TABLE OF bapiset_hier,
        ls_nodes      TYPE bapiset_hier,

        l_lines       TYPE i,
        l_desc(50).

        ls_temp LIKE gs_data.

* Selected Low...
  DATA: lt_rows  TYPE lvc_t_row,
        ls_rows  TYPE lvc_s_row.

  PERFORM get_selected_rows TABLES lt_rows
                                  USING g_grid.
  CHECK lt_rows[] IS NOT INITIAL.

  "-- Data Change --------------------------------------

    READ TABLE gt_data INTO gs_data INDEX ls_rows-index.
    APPEND gs_data TO lt_temp.

  "--------------------------------------------------

  ls_languages-langu      = sy-langu.
  ls_languages-langu_iso  = sy-langu.

  l_coa       = p_kokrs.

FORM f4_group .
  DATA: l_field TYPE tabname,
        l_kokrs TYPE kokrs,
        lt_data TYPE tihttpnvp,
        ls_data TYPE ihttpnvp.

  DATA: ret_tab LIKE ddshretval OCCURS 0 WITH HEADER LINE.

  CLEAR: gt_group[].
  SELECT  a~setname   AS name
          b~descript  AS value
    INTO TABLE gt_group
    FROM setheader AS a INNER JOIN setheadert AS b
                        ON  a~setclass = b~setclass
                        AND a~subclass = b~subclass
                        AND a~setname  = b~setname
   WHERE a~setclass = '0102'
     AND a~subclass = p_kokrs
     AND a~xuniq    = 'X'.

  CHECK gt_group[] IS NOT INITIAL.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield   = 'NAME'
      value_org  = 'S'
    TABLES
      value_tab  = gt_group
      return_tab = ret_tab.

  IF sy-subrc EQ 0.
    READ TABLE ret_tab INDEX 1.
    READ TABLE gt_group INTO ls_data WITH KEY name = ret_tab-fieldval.
    MOVE: ls_data-name TO p_group.
  ENDIF.
ENDFORM. " F4_GROUP
