FORM LOAD_STATUS .

  SELECT SINGLE * FROM ZCOT0130
    WHERE DBTABNAME = 'ZCOT0130'.

  IF SY-SUBRC = 0 AND ZCOT0130-ZLOCK = 'X'.

  ELSE.

FORM TOGGLE_STATUS .

  DATA : GT_T130     LIKE ZCOT0130 OCCURS 0 WITH HEADER LINE.

  IF PA_UNLCK = 'X'.
    UPDATE ZCOT0130
       SET ZLOCK = 'X'
       WHERE DBTABNAME = 'ZCOT0130'.
    IF SY-SUBRC <> 0.
      GT_T130-DBTABNAME = 'ZCOT0130'.
      GT_T130-ZLOCK = 'X'.
      APPEND GT_T130.
      MODIFY ZCOT0130 FROM TABLE GT_T130.
    ENDIF.
  ELSE.
    UPDATE ZCOT0130
       SET ZLOCK = ' '
       WHERE DBTABNAME = 'ZCOT0130'.
    IF SY-SUBRC <> 0.
      GT_T130-DBTABNAME = 'ZCOT0130'.
      GT_T130-ZLOCK = ' '.
      APPEND GT_T130.
      MODIFY ZCOT0130 FROM TABLE GT_T130.
    ENDIF.
  ENDIF.

ENDFORM. " TOGGLE_STATUS

FORM CHK_ZLOCK USING P_ERR.

  CLEAR P_ERR.

  SELECT SINGLE * FROM ZCOT0130
      WHERE DBTABNAME = 'ZCOT0130'.

  IF SY-SUBRC = 0 AND ZCOT0130-ZLOCK = 'X'.
    MESSAGE S000 WITH TEXT-002.
    P_ERR = 'X'.
  ENDIF.

ENDFORM. " CHK_ZLOCK
