FORM file_upload .
  CLEAR: gs_data, gt_data[].

  CLEAR g_budat.
  CONCATENATE p_perio(4) p_perio+5 '01' INTO g_budat.

  CALL FUNCTION 'SG_PS_GET_LAST_DAY_OF_MONTH'
    EXPORTING
      day_in            = g_budat
    IMPORTING
      last_day_of_month = g_budat.


  DATA: l_result,
        l_file  TYPE string."localfile.

  DATA: lt_tab TYPE TABLE OF alsmex_tabline,
        ls_tab TYPE alsmex_tabline.

  l_file = p_path.

  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = l_file
    RECEIVING
      result               = l_result
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4.

  IF l_result IS INITIAL.
    MESSAGE i003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE p_path TO UPPER CASE.
  IF p_path NS '.XLS'.
    MESSAGE i004.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_path
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 24
      i_end_row               = 65536
    TABLES
      intern                  = lt_tab
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    MESSAGE i007.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK lt_tab[] IS NOT INITIAL.
  SORT lt_tab BY row col.

  DELETE lt_tab WHERE row IS INITIAL AND col IS INITIAL.
  CHECK lt_tab[] IS NOT INITIAL.

  FIELD-SYMBOLS: <fs>.

  LOOP AT lt_tab INTO ls_tab.
    ASSIGN COMPONENT ls_tab-col OF STRUCTURE gs_data TO <fs>.
    REPLACE ALL OCCURRENCES OF ',' IN ls_tab-value WITH ''.
    <fs> = ls_tab-value.

    AT END OF row.
      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDAT.

    CLEAR ls_tab.
  ENDLOOP.


* --> Adjust Data Layout.------------------
  DATA: l_index TYPE i.

  LOOP AT gt_data INTO gs_data.
    l_index = sy-tabix.

    PERFORM conversion_alpha_input CHANGING:  gs_data-kndnr,
                                              gs_data-artnr,
                                              gs_data-kaufn,
                                              gs_data-prctr,
                                              gs_data-bzirk.

    PERFORM currency_display_to_sap USING c_waers_krw
                                  CHANGING gs_data-vv001.

    MODIFY gt_data FROM gs_data INDEX l_index.
    CLEAR: gs_data, l_index.
  ENDLOOP.
ENDFORM. " FILE_UPLOAD

FORM get_data .
  DATA: l_index TYPE i.
  CLEAR: gt_data[], gs_data, g_ledbo.

  MOVE p_ledbo TO g_ledbo.
  PERFORM conversion_ledbo_input CHANGING g_ledbo.

  SELECT  bukrs
          werks
          artnr
          vkorg
          vtweg
          spart
          kndnr
          kdgrp
          land1
          vkbur
          brsch
          vkgrp
          bzirk
          prctr
          prat1
          wwbkl
          wwpct
          wwku2
          wwku3
          inco1
          kaufn
          kdpos
          SUM( vv001 ) AS vv001
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM ce11000
   WHERE paledger = g_ledbo
     AND vrgar    = p_vrgar
     AND perio    = p_perio
     AND kokrs    = p_kokrs
   GROUP BY  bukrs werks artnr vkorg
          vtweg spart kndnr kdgrp land1 vkbur
          brsch vkgrp bzirk prctr prat1 wwbkl
          wwpct wwku2 wwku3 inco1 kaufn kdpos.

  LOOP AT gt_data INTO gs_data.
    l_index = sy-tabix.

    PERFORM currency_sap_to_display USING c_waers_krw
                                    CHANGING gs_data-vv001.
    MODIFY gt_data FROM gs_data INDEX l_index
                                TRANSPORTING vv001.
    CLEAR: gs_data, l_index.
  ENDLOOP.
ENDFORM. " GET_DATA
