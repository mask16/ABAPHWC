FORM fi_reverse_document_processing .
  DATA: lt_rows TYPE lvc_t_row,
        ls_rows TYPE lvc_s_row.

*  DATA: lt_belnr TYPE TABLE OF belnr_d,
*        ls_belnr TYPE belnr_d.

  DATA: lr_belnr TYPE RANGE OF belnr_d,
      ls_belnr LIKE LINE OF lr_belnr.

  DATA: l_lines TYPE i,
        l_page  TYPE i,
        l_num(2) TYPE n,
        l_index TYPE i,
        l_start TYPE i,
        l_end   TYPE i,
        l_mod   TYPE i,
        l_fname TYPE fieldname,
        l_tcode_f80  TYPE tcode VALUE 'F.80',
        l_msg   TYPE bapi_msg.

  DATA: ls_opt  TYPE ctu_params.
  ls_opt-dismode  = p_mode.
  ls_opt-defsize  = 'X'.

  PERFORM get_grid_selected_rows TABLES lt_rows.
  IF lt_rows[] IS INITIAL.
    MESSAGE i000 WITH text-w03.
    EXIT.
  ENDIF.

  LOOP AT lt_rows INTO ls_rows.
    READ TABLE gt_data INTO gs_data INDEX ls_rows-index.
    CHECK gs_data-xstov IS INITIAL.

    "-- Checking --------
    READ TABLE gt_bkpf WITH KEY belnr = gs_data-belnr
                        TRANSPORTING NO FIELDS.
    CHECK sy-subrc NE 0.
    "--------------------------------------------------

    _set_ranges:  lr_belnr ls_belnr 'I' 'EQ' gs_data-belnr ''.
    "ls_belnr = gs_data-belnr.
    "APPEND ls_belnr TO lt_belnr.
    CLEAR: ls_rows, gs_data, ls_belnr.
  ENDLOOP.

  IF lr_belnr[] IS INITIAL.       "lt_belnr[].
    MESSAGE i000 WITH text-w02.
    EXIT.
  ENDIF.

*  DESCRIBE TABLE lt_belnr LINES l_lines.

  "F.80 -> SAPF080.-----

  SUBMIT sapf080  WITH br_bukrs-low = p_bukrs
                  WITH br_gjahr-low = p_gjahr
                  WITH br_belnr IN lr_belnr
                  WITH br_rldnr-low = '0L'
                  WITH stogrd = '01'
                  WITH testlauf = ''
                  WITH bvor_no = 'X'
                  AND RETURN.

  "------------------------
*  IF l_lines < 9.
*    l_page = 1.
*  ELSE.
*    l_page = ( l_lines / 7 ) + 1.
*    l_mod = l_lines MOD 7.
*    IF l_mod < 2 OR l_mod > 3.
*      l_page = l_page - 1.
*    ENDIF.
*  ENDIF.
*
*  CLEAR: bdcdata, bdcdata[], msgdata, msgdata[].
*  PERFORM bdc_dynpro USING:
*    'X' 'SAPF080'       '1000',
*    ' ' 'BDC_OKCODE'    '=%003',
*    ' ' 'BR_RLDNR-LOW'  '0L',
*    ' ' 'STOGRD'        '01', "역분개 사유.
*    ' ' 'TESTLAUF'      ' ',
*    ' ' 'BVOR_NO'       'X'.
*
*  DO l_page TIMES.
*    l_index = sy-index.
*    l_start = ( l_index - 1 ) * 7 + 1.
*    l_end = l_start + 7.
*    IF l_end > l_lines.
*      l_end = l_lines.
*    ENDIF.
*
*    IF l_page EQ l_index.
*      PERFORM bdc_dynpro USING:
*        'X'  'SAPLALDB'    '3000',
*        ' ' 'BDC_OKCODE'  '=ACPT'.
*    ELSE.
*      PERFORM bdc_dynpro USING:
*        'X'  'SAPLALDB'    '3000',
*        ' ' 'BDC_OKCODE'  '/00'.
*    ENDIF.
*
*    LOOP AT lt_belnr INTO ls_belnr FROM l_start TO l_end.
*      ADD 1 TO l_num.
*
*      CONCATENATE 'RSCSEL_255-SLOW_I(' l_num ')' INTO l_fname.
*      PERFORM bdc_dynpro USING:
*        ' ' l_fname ls_belnr.
*
*      CLEAR: ls_belnr, l_fname.
*    ENDLOOP.
*
*    CLEAR: l_num.
*  ENDDO.
*
*  READ TABLE lt_belnr INTO ls_belnr INDEX 1.
*  PERFORM bdc_dynpro USING:
*    'X' 'SAPF080'         '1000',
*    ' ' 'BDC_OKCODE'      '=ONLI',
*    ' ' 'BR_BUKRS-LOW'    p_bukrs,
*    ' ' 'BR_BELNR-LOW'    ls_belnr,
*    ' ' 'BR_GJAHR-LOW'    p_gjahr,
*    ' ' 'BR_RLDNR-LOW'    '0L',
*    ' ' 'STOGRD'          '01', "역분개 사유.
*    ' ' 'BVOR_NO'         'X',
*
*    'X' 'SAPMSSY0'        '0120',
*    ' ' 'BDC_OKCODE'      '=BACK',
*
*    'X' 'SAPF080'         '1000',
*    ' ' 'BDC_OKCODE'      '/EE'.
*
** -- BDC Processing...
*  CALL TRANSACTION l_tcode_f80
*             USING bdcdata
*      OPTIONS FROM ls_opt
*     MESSAGES INTO msgdata.

  IF sy-subrc EQ 0.
    PERFORM modify_result_bdc.
  ELSE.
    PERFORM get_message_build USING sy-msgid
                                    sy-msgno
                                    sy-msgv1
                                    sy-msgv2
                                    sy-msgv3
                                    sy-msgv4
                                    l_msg.
    MESSAGE i000 WITH l_msg.
  ENDIF.
ENDFORM. " FI_REVERSE_DOCUMENT_PROCESSING
