*----------------------------------------------------------------------*
FORM SELECT_MAIN_DATA .

  DATA : LV_TABIX TYPE SY-TABIX.

  PERFORM GET_KOSTL_OBJNR.
  PERFORM GET_KSGRP_OBJNR.
  PERFORM GET_KTGRP_OBJNR.

  CLEAR : GT_MAIN[], GT_MAIN.
  SELECT KOKRS BELNR BUZEI KSTAR SGTXT GKONT OBJNR OWAER WKGBTR PERIO
         INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
         FROM COEP
         WHERE KOKRS =  PA_KOKRS AND
               GJAHR =  PA_GJAHR AND
               PERIO IN SO_PERIO AND
               KSTAR IN SO_KSTAR AND
               KSTAR IN SO_KSTAX AND
               OBJNR IN SO_OBJNR AND
               VRGNG = 'COIN' AND
               ( WRTTP = '04' OR WRTTP = '11' ).

  LOOP AT GT_MAIN.
    LV_TABIX = SY-TABIX.

    IF GT_MAIN-OBJNR(2) <> 'KS'.
      CONTINUE.
    ENDIF.

    IF NOT GT_MAIN-KSTAR IN SO_KSTAR.
      DELETE GT_MAIN. CONTINUE.
    ENDIF.

    IF NOT GT_MAIN-OBJNR IN SO_OBJNR.
      DELETE GT_MAIN. CONTINUE.
    ENDIF.


*   계정명
    SELECT SINGLE KTEXT INTO GT_MAIN-KTEXT
           FROM CSKU
           WHERE KSTAR = GT_MAIN-KSTAR AND
                 SPRAS = PA_SPRAS AND
                 KTOPL = '1000'.

*   전표번호
    SELECT SINGLE BUDAT REFBN INTO (GT_MAIN-BUDAT, GT_MAIN-REFBN)
           FROM COBK
           WHERE BELNR = GT_MAIN-BELNR AND
                 KOKRS = PA_KOKRS.

*   구매처명
    SELECT SINGLE NAME1 INTO GT_MAIN-NAME1
           FROM LFA1
           WHERE LIFNR = GT_MAIN-GKONT.
    IF SY-SUBRC <> 0.
      SELECT SINGLE TXT20 INTO GT_MAIN-TXT20 FROM SKAT
             WHERE  SPRAS = PA_SPRAS AND
                    KTOPL = '1000' AND
                    SAKNR = GT_MAIN-GKONT.
      IF SY-SUBRC = 0.
        MOVE GT_MAIN-TXT20 TO GT_MAIN-NAME1.
      ENDIF.
    ENDIF.
    GT_MAIN-KOSTL = GT_MAIN-OBJNR+6(10).

    CONCATENATE PA_GJAHR GT_MAIN-PERIO+1(2) '01' INTO GT_MAIN-DATAB.
    SELECT SINGLE PRCTR INTO GT_MAIN-PRCTR
           FROM CSKS
           WHERE KOSTL = GT_MAIN-KOSTL AND
                 KOKRS = PA_KOKRS AND
                 DATAB <= GT_MAIN-DATAB.

    CASE GT_MAIN-PRCTR+6(4).
      WHEN '1210'.
        IF GT_MAIN-KOSTL >= '0002201401' AND
           GT_MAIN-KOSTL <= '0002201403'.
          GT_MAIN-AMT01 = GT_MAIN-WKGBTR.
          GT_MAIN-AMTSM = GT_MAIN-WKGBTR.
        ENDIF.
        IF GT_MAIN-KOSTL >= '0002201401' AND
           GT_MAIN-KOSTL <= '0002201402'.
        ELSE.
          GT_MAIN-AMT08 = GT_MAIN-WKGBTR.
          GT_MAIN-AMTSM = GT_MAIN-WKGBTR.
        ENDIF.

      WHEN '1220'.
        GT_MAIN-AMT02 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1230'.
        GT_MAIN-AMT03 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1240'.
        GT_MAIN-AMT04 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1250'.
        GT_MAIN-AMT05 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1260'.
        GT_MAIN-AMT06 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1270'.
        IF GT_MAIN-KOSTL >= '0002201401' AND
           GT_MAIN-KOSTL <= '0002201403'.
          GT_MAIN-AMT01 = GT_MAIN-WKGBTR.
          GT_MAIN-AMTSM = GT_MAIN-WKGBTR.
        ENDIF.
        IF GT_MAIN-KOSTL = '0002201403'.
        ELSE.
          GT_MAIN-AMT07 = GT_MAIN-WKGBTR.
          GT_MAIN-AMTSM = GT_MAIN-WKGBTR.
        ENDIF.

      WHEN '1320'.
        GT_MAIN-AMT09 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1330'.
        GT_MAIN-AMT10 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1371'.
        GT_MAIN-AMT11 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1310'.
        GT_MAIN-AMT12 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1410'.
        GT_MAIN-AMT15 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1420'.
        IF GT_MAIN-KOSTL = '0004201301'.
          GT_MAIN-AMT13 = GT_MAIN-WKGBTR.
        ELSE.
          GT_MAIN-AMT14 = GT_MAIN-WKGBTR.
        ENDIF.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1510'.
        GT_MAIN-AMT18 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1520'.
        GT_MAIN-AMT16 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

      WHEN '1530'.
        GT_MAIN-AMT17 = GT_MAIN-WKGBTR.
        GT_MAIN-AMTSM = GT_MAIN-WKGBTR.

    ENDCASE.

    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
    COLLECT GT_RESULT.
  ENDLOOP.

ENDFORM. " SELECT_MAIN_DATA
