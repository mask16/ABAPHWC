FORM file_upload .
  CLEAR: gs_data, gt_data[].

  DATA: l_result,
        l_file  TYPE string."localfile.

  DATA: lt_tab TYPE TABLE OF alsmex_tabline,
        ls_tab TYPE alsmex_tabline.

  l_file = p_path.

  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = l_file
    RECEIVING
      result               = l_result
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4.

  IF l_result IS INITIAL.
    MESSAGE i003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE p_path TO UPPER CASE.
  IF p_path NS '.XLS'.
    MESSAGE i004.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_path
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 19
      i_end_row               = 65536
    TABLES
      intern                  = lt_tab
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    MESSAGE i007.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK lt_tab[] IS NOT INITIAL.
  SORT lt_tab BY row col.

  DELETE lt_tab WHERE row IS INITIAL AND col IS INITIAL.
  CHECK lt_tab[] IS NOT INITIAL.

  FIELD-SYMBOLS: <fs>.

  LOOP AT lt_tab INTO ls_tab.
    ASSIGN COMPONENT ls_tab-col OF STRUCTURE gs_data TO <fs>.
    REPLACE ALL OCCURRENCES OF ',' IN ls_tab-value WITH ''.
    <fs> = ls_tab-value.

    AT END OF row.
*      gs_data-icon = '@00@'.
      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDAT.

    CLEAR ls_tab.
  ENDLOOP.


* --> Adjust Data Layout.------------------
  DATA: l_index TYPE i,
        l_flag(1).

  SELECT SINGLE * FROM t001 WHERE bukrs = p_bukrs.

  LOOP AT gt_data INTO gs_data.
    l_index = sy-tabix.
    IF gs_data-bukrs NE p_bukrs.
      l_flag = 'X'.
      EXIT.
    ENDIF.

    PERFORM conversion_alpha_input USING: gs_data-prctr,
                                          gs_data-anlkl,
                                          gs_data-kostl.
    PERFORM currency_display_to_sap  USING t001-waers  "c_waers_krw
                                    CHANGING gs_data-dmbtr.
    MODIFY gt_data FROM gs_data INDEX l_index.

    CLEAR: gs_data, l_index.
  ENDLOOP.

  IF l_flag IS NOT INITIAL.

        l_fname2      TYPE fieldname,

        l_zratio2     TYPE zratio2,

  DATA: ls_opt  TYPE ctu_params.
  ls_opt-dismode  = p_mode.
  ls_opt-defsize  = 'X'.

  FIELD-SYMBOLS: <fs>.

  LOOP AT gt_data INTO gs_data.
    l_tabix = sy-tabix.

    "checking..------------------
    CLEAR: l_cnt, l_zratio2.
    DO 12 TIMES.
      l_num = sy-index.
      CONCATENATE 'ZWKG' l_num INTO l_fname.
      ASSIGN COMPONENT l_fname OF STRUCTURE gs_data TO <fs>.

      IF <fs> IS NOT INITIAL.
        ADD 1 TO l_cnt.
        IF l_zratio2 IS INITIAL.
          l_month = l_num.
        ENDIF.
      ENDIF.
      ADD <fs> TO l_zratio2.
    ENDDO.
    "-----------------------------------

    IF l_zratio2 NE '100.00'.
      CONCATENATE l_tabix ' Line : Incorrect 100%.' INTO l_msg.
      APPEND l_msg TO gt_result.
      CONTINUE.
    ENDIF.

    CONCATENATE:  p_gjahr l_month '01' INTO l_datum.

    CLEAR: bdcdata[], bdcdata, msgdata[], msgdata, l_msg, l_num, l_num2.
    PERFORM bdc_dynpro  USING:
      'X' 'SAPMKAUF'    '0100',
      ' ' 'BDC_OKCODE'  '/00',
      ' ' 'COAS-AUART'  'Z201',

      'X' 'SAPMKAUF'    '0600',
      ' ' 'BDC_OKCODE'  '=BUT2',
      ' '  'COAS-KTEXT'  gs_data-name1,  "자동
*      ' '  'COAS-BUKRS'  '1000'      "자동
      ' ' 'COAS-GSBER'  gs_data-gsber,
