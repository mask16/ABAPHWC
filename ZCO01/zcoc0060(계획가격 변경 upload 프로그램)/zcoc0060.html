FORM FILE_UPLOAD .

    EXPORTING

  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING

    MESSAGE I007.

  ENDIF.

  CHECK LT_TAB[] IS NOT INITIAL.

  DELETE LT_TAB WHERE ROW IS INITIAL AND COL IS INITIAL.

*      gs_data-icon = '@00@'.

FORM PLANPRICE_CHANGE_PROCESSING .
  DATA: LT_ROWS TYPE LVC_T_ROW,
        LS_ROWS TYPE LVC_S_ROW.

  DATA: L_TCODE       TYPE TCODE VALUE 'MM02',
        L_MODE(1)                 VALUE 'N',
        L_SUCCESS(5)  TYPE C VALUE '0',
        L_ERROR(5)    TYPE C VALUE '0',
        L_TOTAL(5)    TYPE C VALUE '0',
        L_AMOUNT      TYPE DZPLP1,
        L_STR         TYPE STRING,
        L_MESSAGE     TYPE BAPI_MSG,
        L_WERKS       TYPE WERKS_D.

  DATA: L_UCOMM TYPE SYUCOMM.

  DATA: LS_OPT  TYPE CTU_PARAMS.
  LS_OPT-DISMODE  = P_MODE.

  PERFORM GET_SELECTED_ROWS TABLES LT_ROWS USING G_GRID.
  CHECK LT_ROWS[] IS NOT INITIAL.

  LOOP AT LT_ROWS INTO LS_ROWS.
    CLEAR: BDCDATA[], BDCDATA, MSGDATA[], MSGDATA, GS_DATA.
    CLEAR: L_UCOMM.
    READ TABLE GT_DATA INTO GS_DATA INDEX LS_ROWS-INDEX.

    PERFORM GET_TAB_PAGE_FOR_COST USING GS_DATA-MATNR CHANGING L_UCOMM.

    PERFORM BDC_DYNPRO  USING:
      'X' 'SAPLMGMM'      '0060',
      ' ' 'BDC_OKCODE'    '=ENTR',
      ' ' 'RMMG1-MATNR'   GS_DATA-MATNR,

      'X' 'SAPLMGMM'              '0070',
      ' ' 'BDC_OKCODE'            '=ENTR',
      ' ' 'MSICHTAUSW-KZSEL(01)'  'X',

*----------------------------------------------------------------------
      'X' 'SAPLMGMM'      '4004',
      ' ' 'BDC_OKCODE'    L_UCOMM,

      'X' 'SAPLMGMM'      '0081',
      ' ' 'BDC_OKCODE'    '=ENTR',
      ' ' 'RMMG1-WERKS'   GS_DATA-WERKS,
*----------------------------------------------------------------------

      'X' 'SAPLMGMM'      '4000',
      ' ' 'BDC_OKCODE'    '/00'.
*      '  ' 'MAKT-MAKTX'    SM70 1.2*20 SPOOL+BOX
    IF GS_DATA-ZPLP1 IS NOT INITIAL.
      L_AMOUNT = GS_DATA-ZPLP1.
      PERFORM: CURRENCY_SAP_TO_DISPLAY USING GS_DATA-WAERS CHANGING L_AMOUNT,
               GET_MONEY_CHAR USING L_AMOUNT L_STR GS_DATA-WAERS.

      PERFORM BDC_DYNPRO  USING:
          ' ' 'MBEW-ZPLP1'  L_STR,
          ' ' 'MBEW-ZPLD1'  GS_DATA-ZPLD1.
    ENDIF.

    IF GS_DATA-ZPLP2 IS NOT INITIAL.
      L_AMOUNT = GS_DATA-ZPLP2.
      PERFORM: CURRENCY_SAP_TO_DISPLAY USING GS_DATA-WAERS CHANGING L_AMOUNT,
               GET_MONEY_CHAR USING L_AMOUNT L_STR GS_DATA-WAERS.

      PERFORM BDC_DYNPRO  USING:
          ' ' 'MBEW-ZPLP2'  L_STR,
          ' ' 'MBEW-ZPLD2'  GS_DATA-ZPLD2.
    ENDIF.

    IF GS_DATA-ZPLP3 IS NOT INITIAL.
      L_AMOUNT = GS_DATA-ZPLP3.
      PERFORM: CURRENCY_SAP_TO_DISPLAY USING GS_DATA-WAERS CHANGING L_AMOUNT,
               GET_MONEY_CHAR USING L_AMOUNT L_STR GS_DATA-WAERS.

      PERFORM BDC_DYNPRO  USING:
          ' ' 'MBEW-ZPLP3'  L_STR,
          ' ' 'MBEW-ZPLD3'  GS_DATA-ZPLD3.
    ENDIF.

    PERFORM BDC_DYNPRO  USING:
        'X' 'SAPLSPO1'      '0300',
        ' ' 'BDC_OKCODE'    '=YES'.

    CALL TRANSACTION L_TCODE
               USING BDCDATA
        OPTIONS FROM LS_OPT
*                MODE p_mode
            MESSAGES INTO MSGDATA.

    IF SY-SUBRC NE 0.
      READ TABLE MSGDATA WITH KEY MSGTYP = 'E'.
      PERFORM GET_MESSAGE_BUILD USING MSGDATA-MSGID
                                      MSGDATA-MSGNR
                                      MSGDATA-MSGV1
                                      MSGDATA-MSGV2
                                      MSGDATA-MSGV3
                                      MSGDATA-MSGV4
                                      GS_DATA-MSG.
      GS_DATA-ICON = '@02@'.
      ADD 1 TO L_ERROR.
    ELSE.
      GS_DATA-ICON = '@01@'.
      ADD 1 TO L_SUCCESS.

      IF L_WERKS NE GS_DATA-WERKS.
        L_WERKS = GS_DATA-WERKS.
        COMMIT WORK.
        WAIT UP TO 1 SECONDS.
      ENDIF.
    ENDIF.
    ADD 1 TO L_TOTAL.

    MODIFY GT_DATA FROM GS_DATA INDEX LS_ROWS-INDEX.

    CLEAR LS_ROWS.
  ENDLOOP.

  CONDENSE: L_TOTAL, L_SUCCESS, L_ERROR.

  CONCATENATE `Total ` L_TOTAL
              `, Success ` L_SUCCESS
              `, Fail ` L_ERROR INTO L_MESSAGE.
