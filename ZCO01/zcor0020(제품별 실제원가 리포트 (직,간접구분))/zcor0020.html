FORM get_base_data .
  DATA: lt_csks     TYPE TABLE OF csks,
        ls_csks     TYPE csks,
        l_objnr     TYPE j_objnr,
        l_name      TYPE fieldname.

        l_subclass  TYPE setsubcls.

  DATA: lr_setname TYPE RANGE OF setnamenew,
        ls_setname LIKE LINE OF lr_setname.

  _set_ranges:  lr_setname ls_setname 'I' 'EQ' 'P100' '',
                lr_setname ls_setname 'I' 'EQ' 'P200' '',
                lr_setname ls_setname 'I' 'EQ' 'P300' '',
                lr_setname ls_setname 'I' 'EQ' 'P400' '',
                lr_setname ls_setname 'I' 'EQ' 'P500' '',
                lr_setname ls_setname 'I' 'EQ' 'P600' '',
                lr_setname ls_setname 'I' 'EQ' 'Y400' ''.

  l_subclass = p_bukrs.

  "-- 원가요소 구하기...
  IF gt_setleaf[] IS INITIAL.
    SELECT *
      INTO TABLE gt_setleaf
      FROM setleaf
     WHERE setclass = l_setclass
       AND subclass = l_subclass
       AND setname  IN lr_setname.

    SORT gt_setleaf BY setname.
    LOOP AT lr_setname INTO ls_setname.
      CONCATENATE 'GR_KST' ls_setname-low INTO l_name.  "gr_kstp100. ...
      ASSIGN (l_name) TO <fs>.
      LOOP AT gt_setleaf INTO gs_setleaf WHERE setname EQ ls_setname-low.
        _set_ranges:  <fs>
                      gs_kstar
                      gs_setleaf-valsign
                      gs_setleaf-valoption
                      gs_setleaf-valfrom
                      gs_setleaf-valto.
        CLEAR: gs_setleaf, gs_kstar.
      ENDLOOP.

      CLEAR: ls_setname, l_name.
    ENDLOOP.

  ENDIF.

  "--- 회사코드 관련 정보...
  IF gt_t001k[] IS INITIAL.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE gt_t001k
      FROM t001 AS a INNER JOIN t001k AS b ON a~bukrs = b~bukrs

  ENDIF.

  "-- 필드명...
  IF gt_tckh[] IS INITIAL.
    SELECT  b~elehk
            b~elemt
            b~el_hv
            a~txele
      INTO CORRESPONDING FIELDS OF TABLE gt_tckh
      FROM tckh1 AS a INNER JOIN tckh3 AS b
                      ON a~elehk = b~elehk
                     AND a~elemt = b~elemt
     WHERE a~spras = sy-langu
       AND a~elehk = '10'.
  ENDIF.

  "-- Cost Center List.
  IF gr_kostl[] IS INITIAL.
    SELECT *
      INTO TABLE lt_csks
      FROM csks
     WHERE kokrs  EQ p_kokrs
       AND bukrs  EQ p_bukrs
       AND kosar  EQ 'F'.

    SORT lt_csks BY kostl.
    LOOP AT lt_csks INTO ls_csks.
      CONCATENATE 'KS' p_kokrs ls_csks-kostl INTO l_objnr.
      _set_ranges:  gr_kostl gs_kostl 'I' 'EQ' ls_csks-kostl '',
                    gr_objnr gs_objnr 'I' 'EQ' l_objnr ''.
      CASE ls_csks-kostl(4).
        WHEN '0002'. _set_ranges: gr_objnr002 gs_objnr 'I' 'EQ' l_objnr ''.
        WHEN '0003'. _set_ranges: gr_objnr003 gs_objnr 'I' 'EQ' l_objnr ''.
        WHEN '0004'. _set_ranges: gr_objnr004 gs_objnr 'I' 'EQ' l_objnr ''.
        WHEN '0005'. _set_ranges: gr_objnr005 gs_objnr 'I' 'EQ' l_objnr ''.
        WHEN OTHERS.
      ENDCASE.
      CLEAR: gs_kostl, ls_csks, l_objnr, gs_objnr.
    ENDLOOP.
*
*      CLEAR: s_poper, l_datum1, l_datum2.
*    ENDLOOP.

    SORT: gr_objnr002 BY low,
          gr_objnr003 BY low,
          gr_objnr004 BY low,
          gr_objnr005 BY low.

                                      gr_objnr003 COMPARING low,
                                      gr_objnr004 COMPARING low,
                                      gr_objnr005 COMPARING low.
*    SORT gr_kostl BY low.
*    DELETE ADJACENT DUPLICATES FROM gr_kostl COMPARING low.
  ENDIF.

  "-- COSP, COSS 테이블 구하기..
  IF gt_cosp[] IS INITIAL.
    SELECT objnr gjahr wrttp kstar
            SUM( wkg001 ) AS wkg001
            SUM( wkg002 ) AS wkg002
            SUM( wkg003 ) AS wkg003
            SUM( wkg004 ) AS wkg004
            SUM( wkg005 ) AS wkg005
            SUM( wkg006 ) AS wkg006
            SUM( wkg007 ) AS wkg007
            SUM( wkg008 ) AS wkg008
            SUM( wkg009 ) AS wkg009
            SUM( wkg010 ) AS wkg010
            SUM( wkg011 ) AS wkg011
            SUM( wkg012 ) AS wkg012
      INTO CORRESPONDING FIELDS OF TABLE gt_cosp
      FROM cosp
     WHERE objnr IN gr_objnr
       AND gjahr EQ p_bdatj
       AND wrttp EQ '04'
     GROUP BY objnr gjahr wrttp kstar .

    SELECT objnr gjahr wrttp kstar
            SUM( wkg001 ) AS wkg001
            SUM( wkg002 ) AS wkg002
            SUM( wkg003 ) AS wkg003
            SUM( wkg004 ) AS wkg004
            SUM( wkg005 ) AS wkg005
            SUM( wkg006 ) AS wkg006
            SUM( wkg007 ) AS wkg007
            SUM( wkg008 ) AS wkg008
            SUM( wkg009 ) AS wkg009
            SUM( wkg010 ) AS wkg010
            SUM( wkg011 ) AS wkg011
            SUM( wkg012 ) AS wkg012
      APPENDING CORRESPONDING FIELDS OF TABLE gt_cosp
      FROM coss
     WHERE objnr IN gr_objnr
       AND gjahr EQ p_bdatj
       AND wrttp EQ '04'
       AND beknz EQ 'S'
     GROUP BY objnr gjahr wrttp kstar  .

    SORT gt_cosp BY objnr kstar.
  ENDIF.
ENDFORM. " GET_BASE_DATA

*&---------------------------------------------------------------------*
*&      Form  process_data
*&---------------------------------------------------------------------*
FORM process_data USING p_amount p_setname p_werks p_month p_waers
                  CHANGING c_amount1 c_amount2.
  FIELD-SYMBOLS: <fs_kstar> TYPE table,
                 <fs_objnr> TYPE table,
                 <fs>.

  DATA: lt_cosp TYPE TABLE OF cosp.
  DATA: lr_kstar005 TYPE RANGE OF kstar,
        lr_kstar008 TYPE RANGE OF kstar,
        ls_kstar    LIKE LINE OF lr_kstar005.
  DATA: l_name1 TYPE fieldname,
        l_name2 TYPE fieldname,
        l_fname TYPE fieldname,
        l_tot1  TYPE wertv9,  "직접비.
        l_tot2  TYPE wertv9,  "간접비.
        l_amt   TYPE wertv8,
        l_amt1  TYPE wertv8,
        l_amt2  TYPE wertv8.

  "--추가.
  CHECK p_amount NE 0.
  CLEAR gs_total.

  IF ( p_setname EQ 'P500' OR p_setname EQ 'P600' )
    AND p_werks EQ '1040'.
    CLEAR: c_amount1, c_amount2.
    EXIT.
  ENDIF.

  MOVE p_amount TO l_amt.

  CONCATENATE:  'WKG' p_month INTO l_fname,
                'GR_KST' p_setname INTO l_name1.
  IF p_setname EQ 'P400' AND p_werks EQ '1040'.
    l_name1 = 'GR_KSTY400'.
  ENDIF.

  CASE p_werks.
    WHEN '1010'. l_name2 = 'GR_OBJNR002'.
    WHEN '1020'. l_name2 = 'GR_OBJNR003'.
    WHEN '1030'. l_name2 = 'GR_OBJNR004'.
    WHEN '1040'. l_name2 = 'GR_OBJNR005'.
  ENDCASE.

  "Check Exist or Not.
  READ TABLE gt_total INTO gs_total WITH KEY kstarnm = l_name1
                                             objnrnm = l_name2
                                             poper   = p_month.
  IF sy-subrc NE 0.

    ASSIGN: (l_name1) TO <fs_kstar>,
            (l_name2) TO <fs_objnr>.

    _set_ranges:  lr_kstar005 ls_kstar 'I' 'CP' '005*' '',
                  lr_kstar008 ls_kstar 'I' 'CP' '008*' ''.

    lt_cosp[] = gt_cosp[].
    DELETE: lt_cosp WHERE objnr NOT IN <fs_objnr>,
            lt_cosp WHERE kstar NOT IN <fs_kstar>.

    "-- 직접비..기준.
    LOOP AT lt_cosp INTO gs_cosp WHERE kstar IN lr_kstar005.
      ASSIGN COMPONENT l_fname OF STRUCTURE gs_cosp TO <fs>.
      ADD <fs> TO l_tot1.
      CLEAR gs_cosp.
    ENDLOOP.

    "-- 간접비...기준.
    LOOP AT lt_cosp INTO gs_cosp WHERE kstar IN lr_kstar008.
      ASSIGN COMPONENT l_fname OF STRUCTURE gs_cosp TO <fs>.
      ADD <fs> TO l_tot2.
      CLEAR gs_cosp.
    ENDLOOP.

    MOVE: l_name1   TO gs_total-kstarnm,
          l_name2   TO gs_total-objnrnm,
          p_month   TO gs_total-poper,
          l_tot1    TO gs_total-d_tot,
          l_tot2    TO gs_total-i_tot.
    APPEND gs_total TO gt_total.
  ELSE.
    l_tot1 = gs_total-d_tot.
    l_tot2 = gs_total-i_tot.
  ENDIF.

*  PERFORM currency_sap_to_display USING p_waers CHANGING: l_amt,
*                                                          l_tot1,
*                                                          l_tot2.
  CHECK l_tot1 IS NOT INITIAL.
  l_amt1 = l_amt * l_tot1 / ( l_tot1 + l_tot2 ).
  l_amt2 = l_amt - l_amt1.

*  PERFORM currency_display_to_sap USING p_waers CHANGING: l_amt1,
*                                                          l_amt2.
  MOVE: l_amt1 TO c_amount1,
        l_amt2 TO c_amount2.
ENDFORM. "process_data

FORM build_color .
  DATA: l_index   TYPE i,
        lt_color  TYPE lvc_t_scol,
        ls_color  TYPE lvc_s_scol,
        l_color   TYPE lvc_col.
  DATA: ls_fcat   TYPE lvc_s_fcat.

  DATA: lr_fname1 TYPE RANGE OF fieldname,
        lr_fname2 TYPE RANGE OF fieldname,
        ls_fname  LIKE LINE OF lr_fname2.
  _set_ranges:  lr_fname1 ls_fname 'I' 'EQ' 'KST011_1' '',
                lr_fname1 ls_fname 'I' 'EQ' 'KST011_2' '',
                lr_fname1 ls_fname 'I' 'EQ' 'KST017_1' '',
                lr_fname1 ls_fname 'I' 'EQ' 'KST017_2' '',
                lr_fname1 ls_fname 'I' 'EQ' 'KST021_1' '',
                lr_fname1 ls_fname 'I' 'EQ' 'KST021_2' '',

                lr_fname2 ls_fname 'I' 'EQ' 'KST015_1' '',
                lr_fname2 ls_fname 'I' 'EQ' 'KST015_2' '',
                lr_fname2 ls_fname 'I' 'EQ' 'KST019_1' '',
                lr_fname2 ls_fname 'I' 'EQ' 'KST019_2' '',
                lr_fname2 ls_fname 'I' 'EQ' 'KST023_1' '',
                lr_fname2 ls_fname 'I' 'EQ' 'KST023_2' ''.

  LOOP AT gt_data INTO gs_data.
    l_index = sy-tabix.

    LOOP AT gt_fcat INTO ls_fcat.
      CLEAR: ls_color, l_color.

      IF ls_fcat-fieldname IN lr_fname1. l_color = '3'. ENDIF.
      IF ls_fcat-fieldname IN lr_fname2. l_color = '5'. ENDIF.

      CHECK l_color IS NOT INITIAL.

      MOVE: ls_fcat-fieldname TO ls_color-fname,
            l_color           TO ls_color-color-col,
            '1'               TO ls_color-color-int,
            '0'               TO ls_color-color-inv.
      APPEND ls_color TO lt_color.
      CLEAR ls_fcat.
    ENDLOOP.

    gs_data-cellcolor[] = lt_color[].
    MODIFY gt_data FROM gs_data INDEX l_index TRANSPORTING cellcolor.
    CLEAR: gs_data, lt_color[], l_index.
  ENDLOOP.
ENDFORM. " BUILD_COLOR
