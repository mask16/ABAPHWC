FORM FILE_UPLOAD .
  CLEAR: GS_DATA, GT_DATA[].

  DATA: L_RESULT,
        L_FILE  TYPE STRING."localfile.

  DATA: LT_TAB TYPE TABLE OF ALSMEX_TABLINE,
        LS_TAB TYPE ALSMEX_TABLINE.

  L_FILE = P_PATH.

  CALL METHOD CL_GUI_FRONTEND_SERVICES=>FILE_EXIST
    EXPORTING
      FILE                 = L_FILE
    RECEIVING
      RESULT               = L_RESULT
    EXCEPTIONS
      CNTL_ERROR           = 1
      ERROR_NO_GUI         = 2
      WRONG_PARAMETER      = 3
      NOT_SUPPORTED_BY_GUI = 4.

  IF L_RESULT IS INITIAL.
    MESSAGE I003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE P_PATH TO UPPER CASE.
  IF P_PATH NS '.XLS'.
    MESSAGE I004.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = P_PATH
      I_BEGIN_COL             = 1
      I_BEGIN_ROW             = 2
      I_END_COL               = 15
      I_END_ROW               = 65536
    TABLES
      INTERN                  = LT_TAB
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      OTHERS                  = 3.

  IF SY-SUBRC NE 0.
    MESSAGE I007.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK LT_TAB[] IS NOT INITIAL.
  SORT LT_TAB BY ROW COL.

  DELETE LT_TAB WHERE ROW IS INITIAL AND COL IS INITIAL.
  CHECK LT_TAB[] IS NOT INITIAL.

  FIELD-SYMBOLS: <FS>.

  LOOP AT LT_TAB INTO LS_TAB.
    ASSIGN COMPONENT LS_TAB-COL OF STRUCTURE GS_DATA TO <FS>.
    REPLACE ALL OCCURRENCES OF ',' IN LS_TAB-VALUE WITH ''.
    <FS> = LS_TAB-VALUE.

    AT END OF ROW.
*      gs_data-icon = '@00@'.
      APPEND GS_DATA TO GT_DATA.
      CLEAR GS_DATA.
    ENDAT.

    CLEAR LS_TAB.
  ENDLOOP.


* --> Adjust Data Layout.------------------
  DATA: L_INDEX TYPE I,
        LS_TEMP LIKE GS_DATA.

  SORT GT_DATA BY KOSTL KSTAR.
  DELETE ADJACENT DUPLICATES FROM GT_DATA COMPARING KOSTL KSTAR.

  PERFORM GET_WARES USING P_KOKRS GV_WAERS.

  LOOP AT GT_DATA INTO GS_DATA.
    L_INDEX = SY-TABIX.

    PERFORM CONVERSION_ALPHA_INPUT USING: GS_DATA-KOSTL,
                                          GS_DATA-KSTAR.
    PERFORM CURRENCY_DISPLAY_TO_SAP  USING GV_WAERS "C_WAERS_KRW
                                    CHANGING: GS_DATA-WKF01,
                                              GS_DATA-WKF02,
                                              GS_DATA-WKF03,
                                              GS_DATA-WKF04,
                                              GS_DATA-WKF05,
                                              GS_DATA-WKF06,
                                              GS_DATA-WKF07,
                                              GS_DATA-WKF08,
                                              GS_DATA-WKF09,
                                              GS_DATA-WKF10,
                                              GS_DATA-WKF11,
                                              GS_DATA-WKF12.

    GS_DATA-TOTAL = GS_DATA-WKF01 + GS_DATA-WKF02 + GS_DATA-WKF03 +
                    GS_DATA-WKF04 + GS_DATA-WKF05 + GS_DATA-WKF06 +
                    GS_DATA-WKF07 + GS_DATA-WKF08 + GS_DATA-WKF09 +
                    GS_DATA-WKF10 + GS_DATA-WKF11 + GS_DATA-WKF12.
    GS_DATA-WAERS = GV_WAERS. "C_WAERS_KRW.
    MODIFY GT_DATA FROM GS_DATA INDEX L_INDEX.

    CLEAR: GS_DATA, L_INDEX.
  ENDLOOP.
ENDFORM. " FILE_UPLOAD

FORM COSTELEMENT_PLAN_PROCESSIONG .
  DATA: LT_ROWS TYPE LVC_T_ROW,
        LS_ROWS TYPE LVC_S_ROW.

  DATA: LT_DATA TYPE TABLE OF ZCOS0001,

        LT_BASE TYPE TABLE OF ZCOS0001,
        LS_BASE TYPE ZCOS0001.

  DATA: L_MESSAGE     TYPE BAPI_MSG,
        L_INDEX       TYPE I,
        L_TYPE        TYPE BAPI_MTYPE,
        L_SUCCESS(5)  TYPE C,
        L_ERROR(5)    TYPE C.

  CLEAR: GT_RESULT[], GS_RESULT.
  PERFORM GET_SELECTED_ROWS TABLES LT_ROWS USING G_GRID.
*  -- Subtotal, Total Line Delete.
  DELETE LT_ROWS WHERE ROWTYPE IS NOT INITIAL.

  LOOP AT LT_ROWS INTO LS_ROWS.
    READ TABLE GT_DATA INTO GS_DATA INDEX LS_ROWS-INDEX.
