FORM FILE_UPLOAD .
  CLEAR: GS_DATA, GT_DATA[].

  DATA: L_RESULT,
        L_FILE  TYPE STRING."localfile.

  DATA: LT_TAB TYPE TABLE OF ALSMEX_TABLINE,
        LS_TAB TYPE ALSMEX_TABLINE.

  L_FILE = P_PATH.

  CALL METHOD CL_GUI_FRONTEND_SERVICES=>FILE_EXIST
    EXPORTING
      FILE                 = L_FILE
    RECEIVING
      RESULT               = L_RESULT
    EXCEPTIONS
      CNTL_ERROR           = 1
      ERROR_NO_GUI         = 2
      WRONG_PARAMETER      = 3
      NOT_SUPPORTED_BY_GUI = 4.

  IF L_RESULT IS INITIAL.
    MESSAGE I003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE P_PATH TO UPPER CASE.
  IF P_PATH NS '.XLS'.
    MESSAGE I004.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = P_PATH
      I_BEGIN_COL             = 1
      I_BEGIN_ROW             = 2
      I_END_COL               = 10
      I_END_ROW               = 65536
    TABLES
      INTERN                  = LT_TAB
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      OTHERS                  = 3.

  IF SY-SUBRC NE 0.
    MESSAGE I007.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK LT_TAB[] IS NOT INITIAL.
  SORT LT_TAB BY ROW COL.

  DELETE LT_TAB WHERE ROW IS INITIAL AND COL IS INITIAL.
  CHECK LT_TAB[] IS NOT INITIAL.

  FIELD-SYMBOLS: <FS>.

  LOOP AT LT_TAB INTO LS_TAB.
    ASSIGN COMPONENT LS_TAB-COL OF STRUCTURE GS_DATA TO <FS>.
    <FS> = LS_TAB-VALUE.

    AT END OF ROW.
*      gs_data-icon = '@00@'.
      APPEND GS_DATA TO GT_DATA.
      CLEAR GS_DATA.
    ENDAT.

    CLEAR LS_TAB.
  ENDLOOP.


* --> Adjust Data Layout.------------------
  DATA: L_INDEX TYPE I,
        LS_TEMP LIKE GS_DATA.

  LOOP AT GT_DATA INTO GS_DATA.
    L_INDEX = SY-TABIX.

    IF GS_DATA-FIELD1 IS INITIAL.
      MOVE: LS_TEMP-FIELD1  TO GS_DATA-FIELD1,
            LS_TEMP-DESC1   TO GS_DATA-DESC1.
    ENDIF.

    IF GS_DATA-FIELD2 IS INITIAL.
      MOVE: LS_TEMP-FIELD2  TO GS_DATA-FIELD2,
            LS_TEMP-DESC2   TO GS_DATA-DESC2.
    ENDIF.


    IF GS_DATA-FIELD3 IS INITIAL.
      MOVE: LS_TEMP-FIELD3  TO GS_DATA-FIELD3,
            LS_TEMP-DESC3   TO GS_DATA-DESC3.
    ENDIF.

    IF GS_DATA-FIELD4 IS INITIAL.
      MOVE: LS_TEMP-FIELD4  TO GS_DATA-FIELD4,
            LS_TEMP-DESC4   TO GS_DATA-DESC4.
    ENDIF.

    MOVE-CORRESPONDING GS_DATA TO LS_TEMP.
    MODIFY GT_DATA FROM GS_DATA INDEX L_INDEX.

    CLEAR: GS_DATA, L_INDEX.
  ENDLOOP.
ENDFORM. " FILE_UPLOAD

FORM CCTRGROUP_UPLOAD_PROCESSING .
  DATA: LT_GET_NODES  TYPE TABLE OF BAPISET_HIER,
        LT_NODES      TYPE TABLE OF BAPISET_HIER,
        LS_NODES      TYPE BAPISET_HIER,
        LT_VALUES     TYPE TABLE OF BAPI1112_VALUES,
        LS_VALUES     TYPE BAPI1112_VALUES,
        LT_LANGUAGES  TYPE TABLE OF BAPI0015_10,
        LS_LANGUAGES  TYPE BAPI0015_10,

        L_GROUPNAME   TYPE BAPISET_GROUPNAME,
        LS_RETURN     TYPE BAPIRET2,
        L_CO_AREA     TYPE KOKRS,
        L_LINES       TYPE I.

  DATA: L_FIELD1(10),
        L_FIELD2(10),
        L_FIELD3(10),
        L_FIELD4(10),
        L_FIELD5(10).

* Selected Low...
  DATA: LT_ROWS  TYPE LVC_T_ROW,
        LS_ROWS  TYPE LVC_S_ROW.

  PERFORM GET_SELECTED_ROWS TABLES LT_ROWS
                                  USING G_GRID.
  CHECK LT_ROWS[] IS NOT INITIAL.

  "-- Data Change --------------------------------------
  LOOP AT LT_ROWS INTO LS_ROWS.
    READ TABLE GT_DATA INTO GS_DATA INDEX LS_ROWS-INDEX.

    IF L_FIELD1 IS INITIAL.
      LS_NODES-GROUPNAME  = GS_DATA-FIELD1.
      LS_NODES-HIERLEVEL  = 0.
      LS_NODES-VALCOUNT   = 0.
      LS_NODES-DESCRIPT   = GS_DATA-DESC1.
      APPEND LS_NODES TO LT_NODES.

      L_FIELD1 = GS_DATA-FIELD1.
    ENDIF.

    IF L_FIELD2 NE GS_DATA-FIELD2.
      LS_NODES-GROUPNAME  = GS_DATA-FIELD2.
      LS_NODES-DESCRIPT   = GS_DATA-DESC2.
      LS_NODES-HIERLEVEL  = 1.
      LS_NODES-VALCOUNT   = 0.
      APPEND LS_NODES TO LT_NODES.

      L_FIELD2 = GS_DATA-FIELD2.
    ENDIF.

    IF L_FIELD3 NE GS_DATA-FIELD3.
      LS_NODES-GROUPNAME  = GS_DATA-FIELD3.
      LS_NODES-DESCRIPT   = GS_DATA-DESC3.
      LS_NODES-HIERLEVEL  = 2.
      LS_NODES-VALCOUNT   = 0.
      APPEND LS_NODES TO LT_NODES.

      L_FIELD3 = GS_DATA-FIELD3.
    ENDIF.

    IF L_FIELD4 NE GS_DATA-FIELD4.
      LS_NODES-GROUPNAME  = GS_DATA-FIELD4.
      LS_NODES-DESCRIPT   = GS_DATA-DESC4.
      LS_NODES-HIERLEVEL  = 3.
      LS_NODES-VALCOUNT   = 0.
      APPEND LS_NODES TO LT_NODES.

      L_FIELD4 = GS_DATA-FIELD4.
    ENDIF.

    IF L_FIELD5 NE GS_DATA-FIELD5.
      LS_NODES-GROUPNAME  = GS_DATA-FIELD5.
      LS_NODES-DESCRIPT   = GS_DATA-DESC5.
      LS_NODES-HIERLEVEL  = 4.
      LS_NODES-VALCOUNT   = 0.
      APPEND LS_NODES TO LT_NODES.

      L_FIELD5 = GS_DATA-FIELD5.
    ENDIF.

    CLEAR: LS_ROWS, GS_DATA.
  ENDLOOP.
  "--------------------------------------------------

  LS_LANGUAGES-LANGU      = SY-LANGU.
  LS_LANGUAGES-LANGU_ISO  = SY-LANGU.

  L_CO_AREA   = P_KOKRS.
  L_GROUPNAME = L_FIELD1.

* -- ADD Exist CostCenter for Only 1000 group---------------
  PERFORM GET_COSTCENTER_LIST TABLES LT_VALUES
                              USING L_CO_AREA
                                    L_GROUPNAME
                                    L_LINES.
  IF L_LINES > 0.
    CLEAR LS_NODES.
    READ TABLE LT_NODES INTO LS_NODES WITH KEY GROUPNAME = L_GROUPNAME.
    LS_NODES-VALCOUNT = L_LINES.
    MODIFY LT_NODES FROM LS_NODES INDEX 1.
  ENDIF.
* -----------------------------------------------------------------

  CALL FUNCTION 'BAPI_COSTCENTERGROUP_CREATE'
    EXPORTING
      CONTROLLINGAREAIMP = L_CO_AREA
      LANGUAGE           = LS_LANGUAGES
    IMPORTING
      GROUPNAME          = L_GROUPNAME
      RETURN             = LS_RETURN
    TABLES
      HIERARCHYNODES     = LT_NODES
      HIERARCHYVALUES    = LT_VALUES.

  IF LS_RETURN-TYPE EQ 'E'.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    MESSAGE E000 WITH LS_RETURN-MESSAGE.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'.
    MESSAGE S009.
  ENDIF.
ENDFORM. " CCTRGROUP_UPLOAD_PROCESSING
