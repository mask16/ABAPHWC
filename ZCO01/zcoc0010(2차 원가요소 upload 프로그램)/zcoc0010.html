FORM file_upload .
  DATA: l_result,
        l_file TYPE string.

  DATA: lt_tab TYPE TABLE OF alsmex_tabline,
        ls_tab TYPE alsmex_tabline.

  l_file = p_path.

  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = l_file
    RECEIVING
      result               = l_result
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4.

  IF l_result IS INITIAL.
*    MESSAGE '파일을 찾을 수 없습니다.' TYPE 'I'.
    MESSAGE i003.
    LEAVE LIST-PROCESSING.
  ENDIF.

  TRANSLATE p_path TO UPPER CASE.

  IF p_path NS '.XLS' AND p_path NS '.XLSX'.
*    MESSAGE '엑셀파일이 아닙니다.' TYPE 'I'.
    MESSAGE i004.
    LEAVE LIST-PROCESSING.
*    EXIT.
  ENDIF.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_path
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 7
      i_end_row               = 65536
    TABLES
      intern                  = lt_tab
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  CHECK lt_tab[] IS NOT INITIAL.
  SORT lt_tab BY row col.

  DELETE lt_tab WHERE row IS INITIAL AND col IS INITIAL.
  CHECK lt_tab[] IS NOT INITIAL.

  FIELD-SYMBOLS: <fs>.

  CLEAR: gt_data[], gs_data.
  LOOP AT lt_tab INTO ls_tab.
    ASSIGN COMPONENT ls_tab-col OF STRUCTURE gs_data TO <fs>.
    REPLACE ALL OCCURRENCES OF '.' IN ls_tab-value WITH ''.
    <fs> = ls_tab-value.

    AT END OF row.
      gs_data-icon = '@00@'.
      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDAT.

    CLEAR ls_tab.
  ENDLOOP.

*  FORM set_light_led .
*  LOOP AT gt_disp.
*    IF gt_disp-simultyp EQ space.
*      gt_disp-icon = '@01@'.  "'@08@'.  "Success..

FORM bdc_processing_ka06 .
  " Secondary Cost Element Create ...
  DATA: lt_rows TYPE lvc_t_row,
        ls_rows TYPE lvc_s_row.

  DATA: l_index TYPE i,
        l_kstar(10),
        l_tcode TYPE tcode VALUE 'KA06',
        l_tcode_okks TYPE tcode VALUE 'OKKS',
        l_mode(1) VALUE 'N',
        l_group(12) VALUE 'KA06-CREATE',
        l_sucess(10) TYPE n,
        l_error(10) TYPE n,
        l_total(10) TYPE n,
        l_msg TYPE string.

  PERFORM get_grid_selected_rows TABLES lt_rows.
  CHECK lt_rows[] IS NOT INITIAL.

* OKKS -> Controlling Area Setting by BDC ---------------
  CLEAR: bdcdata[], bdcdata, msgdata[], msgdata, gs_data.
  READ TABLE: lt_rows INTO ls_rows INDEX 1,
              gt_data INTO gs_data INDEX ls_rows-index.
  PERFORM bdc_dynpro USING:
    'X'  'SAPLSPO4'         '0300',
    ' '  'BDC_OKCODE'       '=COD1',
    ' '  'SVALD-VALUE(01)'  gs_data-kokrs.

  CALL TRANSACTION l_tcode_okks
             USING bdcdata
              MODE l_mode
          MESSAGES INTO msgdata.
* ----------------------------------------------------

*  PERFORM open_group USING l_group sy-uname.
  LOOP AT lt_rows INTO ls_rows.
    CLEAR: bdcdata[], bdcdata, msgdata[], msgdata, gs_data.
    READ TABLE gt_data INTO gs_data INDEX ls_rows-index.
*    PERFORM conversion_alpha_input CHANGING gs_data-kstar.

    PERFORM bdc_dynpro USING:
      'X'  'SAPLKMA4'         '0200',
      ' '  'BDC_OKCODE'       '/00',
      ' '  'CSKBZ-KOKRS'      gs_data-kokrs,
      ' '  'CSKBZ-KSTAR'      gs_data-kstar,
      ' '  'CSKBZ-DATAB_ANFO' gs_data-datab,
      ' '  'CSKBZ-DATBI_ANFO' gs_data-datbi,

      'X'  'SAPLKMA4'     '0299',
      ' '  'BDC_OKCODE'   '=BU',
      ' '  'CSKBZ-KTEXT'  gs_data-ktext,
      ' '  'CSKBZ-LTEXT'  gs_data-ltext,
      ' '  'CSKBZ-KATYP'  gs_data-katyp.

    CALL TRANSACTION l_tcode
               USING bdcdata
                MODE l_mode
            MESSAGES INTO msgdata.

    IF sy-subrc NE 0.
      READ TABLE msgdata WITH KEY msgtyp = 'E'.
      PERFORM get_message_build USING msgdata-msgid
                                      msgdata-msgnr
                                      msgdata-msgv1
                                      msgdata-msgv2
                                      msgdata-msgv3
                                      msgdata-msgv4
                                      gs_data-msg.
      gs_data-icon = '@02@'.
      ADD 1 TO l_error.
    ELSE.
      gs_data-icon = '@01@'.
