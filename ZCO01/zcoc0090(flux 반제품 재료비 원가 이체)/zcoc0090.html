FORM get_data .
  CLEAR: gt_data[], gs_data, gt_mbew[], gs_mbew,
          gt_mara[], gs_mara.

  DATA: l_capid   TYPE capid  VALUE 'PP01',
        l_mehrs   TYPE xfeld  VALUE 'X'.

  DATA: lt_stb    TYPE TABLE OF stpox,
        ls_stb    TYPE stpox,
        lt_temp   TYPE TABLE OF stpox.

  DATA: ls_matnr LIKE LINE OF gr_matnr.
  FIELD-SYMBOLS: <fs> LIKE gs_data.

  LOOP AT s_matnr.
    LOOP AT s_werks.
      CALL FUNCTION 'CS_BOM_EXPL_MAT_V2'
        EXPORTING
          capid                 = l_capid
          datuv                 = p_bwdat
          mehrs                 = l_mehrs
          stlal                 = g_stlal
          stlan                 = g_stlan
          werks                 = s_werks-low
          mtnrv                 = s_matnr-low
        TABLES
          stb                   = lt_stb
        EXCEPTIONS
          alt_not_found         = 1
          call_invalid          = 2
          material_not_found    = 3
          missing_authorization = 4
          no_bom_found          = 5
          no_plant_data         = 6
          no_suitable_bom_found = 7
          conversion_error      = 8
          OTHERS                = 9.
      CHECK sy-subrc EQ 0.

      MOVE: p_kokrs       TO gs_data-kokrs,
            s_werks-low   TO gs_data-werks,
            s_matnr-low   TO gs_data-mtnrv,
            g_stlan       TO gs_data-stlan,
            g_stlal       TO gs_data-stlal.

      CLEAR: gs_makt, gs_stko.
      READ TABLE: gt_stko   INTO gs_stko WITH KEY matnr = gs_data-mtnrv
                                                  werks = gs_data-werks,
                  gt_makt   INTO gs_makt WITH KEY matnr = gs_data-mtnrv,
                  gt_bwprs  INTO gs_bwprs WITH KEY matnr = gs_data-mtnrv
                                                   bwkey = gs_data-werks.
      MOVE-CORRESPONDING: gs_stko TO gs_data, gs_makt TO gs_data.
      MOVE gs_bwprs-bwprs TO gs_data-bwprs.

      LOOP AT lt_stb INTO ls_stb.
        MOVE: ls_stb-datuv  TO gs_data-datuv,
              ls_stb-idnrk  TO gs_data-idnrk,
              ls_stb-menge  TO gs_data-menge,
              ls_stb-meins  TO gs_data-meins,
              tka01-waers   TO gs_data-waers.

        _set_ranges:  gr_matnr ls_matnr 'I' 'EQ' gs_data-idnrk ''.
        APPEND gs_data TO gt_data.
        CLEAR: ls_stb.
      ENDLOOP.

      CLEAR: gs_data.
    ENDLOOP.
  ENDLOOP.

  SORT gr_matnr BY low.
  DELETE ADJACENT DUPLICATES FROM gr_matnr COMPARING low.

  "MBEW.
  SELECT matnr bwkey zplp1 zpld1 peinh
    INTO CORRESPONDING FIELDS OF TABLE gt_mbew
    FROM mbew
   WHERE matnr IN gr_matnr
     AND bwkey IN s_werks.

  "MARA.
  SELECT matnr meins
    INTO CORRESPONDING FIELDS OF TABLE gt_mara
    FROM mara
   WHERE matnr IN gr_matnr.

  LOOP AT gt_data ASSIGNING <fs>.
    CLEAR: gs_mbew, gs_mara.
    READ TABLE: gt_mbew INTO gs_mbew WITH KEY matnr = <fs>-idnrk
                                              bwkey = <fs>-werks,
                gt_mara INTO gs_mara WITH KEY matnr = <fs>-idnrk.
    MOVE: gs_mbew-zplp1 TO <fs>-zplp1,
          gs_mbew-zpld1 TO <fs>-zpld1,
*          gs_mbew-bwprs TO <fs>-bwprs,
          p_bwdat       TO <fs>-bwdat,
          gs_mbew-peinh TO <fs>-peinh,
          gs_mara-meins TO <fs>-meins2.
  ENDLOOP.
ENDFORM. " GET_DATA

*&---------------------------------------------------------------------*
FORM get_base_data .
  CLEAR: gt_stko[], gs_stko,
        gt_makt[], gs_makt,
        gt_bwprs[], gs_bwprs.

  "반제품 수량, 단위.
  SELECT  a~matnr a~werks b~bmein b~bmeng
    INTO CORRESPONDING FIELDS OF TABLE gt_stko
    FROM mast AS a INNER JOIN stko AS b ON a~stlnr = b~stlnr
                                       AND a~stlal = b~stlal
   WHERE a~matnr IN s_matnr
     AND a~werks IN s_werks
     AND a~stlan EQ g_stlan
     AND a~stlal EQ g_stlal.

  "반제품 명칭.
  SELECT matnr maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_makt
    FROM makt
   WHERE matnr IN s_matnr
     AND spras EQ sy-langu.

  "반제품 변격계획가격.
  SELECT matnr bwkey bwprs
    INTO CORRESPONDING FIELDS OF TABLE gt_bwprs
    FROM mbew
   WHERE matnr IN s_matnr
     AND bwkey IN s_werks.

  SELECT SINGLE *
    FROM tka01
   WHERE kokrs = p_kokrs.
ENDFORM. " GET_BASE_DATA
