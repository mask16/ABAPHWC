REPORT ZCOC0054 MESSAGE-ID ZCOM.

TABLES : ZCOT0010.

DATA : GT_T010 LIKE ZCOT0010 OCCURS 0 WITH HEADER LINE,
       XT_T010 LIKE ZCOT0010 OCCURS 0 WITH HEADER LINE.

SELECTION-SCREEN BEGIN OF BLOCK FRAM1 WITH FRAME TITLE TEXT-S01.
PARAMETERS : P_KOKRS TYPE KOKRS OBLIGATORY DEFAULT '2000',
             P_BUKRS TYPE BUKRS OBLIGATORY DEFAULT '2000',
             P_VKORG TYPE VKORG OBLIGATORY DEFAULT '2000',
             P_WERKS TYPE WERKS_D NO-DISPLAY DEFAULT '2500',
             P_WERKX TYPE WERKS_D NO-DISPLAY,
             P_VERSF TYPE COVERSI OBLIGATORY DEFAULT '90',
             P_VERST TYPE COVERSI OBLIGATORY DEFAULT '91',
             P_GJAHR TYPE GJAHR OBLIGATORY,
             P_ZMOD  TYPE ZMOD DEFAULT 'CO' NO-DISPLAY.
SELECTION-SCREEN END OF BLOCK FRAM1.

  PERFORM GET_DATA.

END-OF-SELECTION.

  P_GJAHR = SY-DATUM(4). P_GJAHR = P_GJAHR + 1.

ENDFORM. " INIT_DATA

*&      Form  GET_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM GET_DATA .

  DATA : LV_SCNT(20),
         LV_TABIX     LIKE SY-TABIX,
         LV_CNT       TYPE I,
         L_ZINDX TYPE ZINDX.

  CASE P_KOKRS.
    WHEN '2000'.
      P_WERKS = '1050'.
      P_WERKX = '2050'.
    WHEN '3000'.
      P_WERKS = '1060'.
      P_WERKX = '3060'.
  ENDCASE.

*  CASE P_VERSF(2).
*    WHEN '09'.
*      P_ZMOD = 'CO'.
*    WHEN '01'.
*      P_ZMOD = 'SD'.
*  ENDCASE.

  CLEAR : GT_T010, GT_T010[], XT_T010, XT_T010[].

* Select...
  SELECT * INTO TABLE GT_T010 FROM ZCOT0010
    WHERE  KOKRS = '1000' AND
           BUKRS = '1000' AND
           VKORG = '1000' AND
           WERKS = P_WERKS AND
           VERSI = P_VERSF AND
           GJAHR = P_GJAHR AND
           PALEDGER = '01' AND
*           VRGAR = 'F' AND
*           SPART = '10' AND
*           VTWEG = '20' AND
           WWBKL = '3100' AND
           ZMOD  = P_ZMOD.

  SELECT MAX( ZINDX ) INTO L_ZINDX
    FROM ZCOT0010
   WHERE ZMOD  = P_ZMOD.
  ADD 1 TO L_ZINDX.

  LOOP AT GT_T010.
*    GT_T010-ZMOD  = P_ZMOD.
*    GT_T010-ZINDX = L_ZINDX.
*    GT_T010-ZSEQ  =  SY-TABIX.

    CLEAR GT_T010-ZSEQ.

    GT_T010-PALEDGER = '02'.
    GT_T010-KOKRS = P_KOKRS.
    GT_T010-BUKRS = P_BUKRS.
    GT_T010-VKORG = P_VKORG.
    GT_T010-WERKS = P_WERKX.
*    GT_T010-VKGRP = 'C90'.
*    GT_T010-VKBUR = 'C900'.
    GT_T010-WWBKL = '7920'.
    GT_T010-FRWAE = GT_T010-WAERS.
    GT_T010-ZSOUP = GT_T010-ZPUUP.
    GT_T010-VERSI = P_VERST.

    IF GT_T010-VTWEG = '10'.
      GT_T010-BZIRK = '800000'.
      GT_T010-VKGRP = '800'.
      GT_T010-VKBUR = '8000'.
      GT_T010-KNDNR = '0000100000'.
    ENDIF.
    GT_T010-VTWEG = '20'.

    CLEAR : GT_T010-WAERS,
            GT_T010-ZPUUP,
            GT_T010-ZFRUP,
            GT_T010-ZINUP,
            GT_T010-CRDAT,
            GT_T010-CRZET,
            GT_T010-CRNAM,
            GT_T010-AEDAT,
            GT_T010-AEZET,
            GT_T010-AENAM.

*   손익센터 변환 규칙
    PERFORM GET_PRCTR USING GT_T010-PRCTR.

*    GT_T010-CRDAT = SY-DATUM.
*    GT_T010-CRZET = SY-UZEIT.
*    GT_T010-CRNAM = SY-UNAME.

    SORT XT_T010 BY PALEDGER VRGAR VERSI GJAHR KOKRS BUKRS VKORG
                    WERKS SPART VTWEG VKGRP VKBUR BZIRK KNDNR PRCTR
                    WWBKL WWKU2 PAPH3 MVGR1 MVGR2 INCO1 ARTNR.

    READ TABLE XT_T010 WITH KEY  PALEDGER  = GT_T010-PALEDGER
                                 VRGAR  = GT_T010-VRGAR
                                 VERSI  = GT_T010-VERSI
                                 GJAHR  = GT_T010-GJAHR
                                 KOKRS  = GT_T010-KOKRS
                                 BUKRS  = GT_T010-BUKRS
                                 VKORG  = GT_T010-VKORG
                                 WERKS  = GT_T010-WERKS
                                 SPART  = GT_T010-SPART
                                 VTWEG  = GT_T010-VTWEG
                                 VKGRP  = GT_T010-VKGRP
                                 VKBUR  = GT_T010-VKBUR
                                 BZIRK  = GT_T010-BZIRK
                                 KNDNR  = GT_T010-KNDNR
                                 PRCTR  = GT_T010-PRCTR
                                 WWBKL  = GT_T010-WWBKL
                                 WWKU2  = GT_T010-WWKU2
                                 PAPH3  = GT_T010-PAPH3
                                 MVGR1  = GT_T010-MVGR1
                                 MVGR2  = GT_T010-MVGR2
                                 INCO1  = GT_T010-INCO1
                                 ARTNR  = GT_T010-ARTNR
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      CLEAR GT_T010-ZSOUP.
    ENDIF.

    MOVE-CORRESPONDING GT_T010 TO XT_T010.
    SORT XT_T010.
    COLLECT XT_T010.

  ENDLOOP.
  GT_T010[] = XT_T010[].

  LOOP AT GT_T010.
    GT_T010-ZMOD  = P_ZMOD.
    GT_T010-ZINDX = L_ZINDX.
    GT_T010-ZSEQ  =  SY-TABIX.

*    GT_T010-PALEDGER = '02'.
*    GT_T010-KOKRS = P_KOKRS.
*    GT_T010-BUKRS = P_BUKRS.
*    GT_T010-VKORG = P_VKORG.
*    GT_T010-WERKS = P_WERKX.
**    GT_T010-VKGRP = 'C90'.
**    GT_T010-VKBUR = 'C900'.
*    GT_T010-WWBKL = '7920'.
*    GT_T010-FRWAE = GT_T010-WAERS.
*    GT_T010-ZSOUP = GT_T010-ZPUUP.
*    GT_T010-VERSI = P_VERST.
*
*    IF GT_T010-VTWEG = '10'.
*      GT_T010-BZIRK = '800000'.
*      GT_T010-VKGRP = '800'.
*      GT_T010-VKBUR = '8000'.
*      GT_T010-KNDNR = '0000100000'.
*    ENDIF.
*    GT_T010-VTWEG = '20'.
*
*    CLEAR : GT_T010-WAERS,
*            GT_T010-ZPUUP,
*            GT_T010-ZFRUP,
*            GT_T010-ZINUP.
*
**   손익센터 변환 규칙
*    PERFORM GET_PRCTR USING GT_T010-PRCTR.

    GT_T010-CRDAT = SY-DATUM.
    GT_T010-CRZET = SY-UZEIT.
    GT_T010-CRNAM = SY-UNAME.
    MODIFY GT_T010.

  ENDLOOP.

  IF GT_T010[] IS NOT INITIAL.
*   기존 데이터 삭제후 저장
    SET UPDATE TASK LOCAL.
    DELETE FROM ZCOT0010 WHERE ZMOD  = P_ZMOD AND
                               KOKRS = P_KOKRS AND
                               BUKRS = P_BUKRS AND
                               VKORG = P_VKORG AND
                               WERKS = P_WERKX AND
                               VERSI = P_VERST AND
                               GJAHR = P_GJAHR AND
                               PALEDGER = '02' AND
*                               VRGAR = 'F' AND
*                               SPART = '10' AND
                               VTWEG = '20' AND
*                               VKGRP = 'C90' AND
*                               VKBUR = 'C900' AND
                               WWBKL = '7920'.

    INSERT ZCOT0010 FROM TABLE GT_T010.
    IF SY-SUBRC  = 0.
      COMMIT WORK.
      MESSAGE S034(ZPPM).
*     조회건수 DISPLAY
      CLEAR : LV_CNT.
      DESCRIBE TABLE GT_T010 LINES LV_CNT.
      WRITE LV_CNT TO LV_SCNT.
      MESSAGE S038(ZPPM) WITH LV_SCNT.
    ELSE.
      ROLLBACK WORK.
      MESSAGE E035(ZPPM).
    ENDIF.
  ELSE.
    MESSAGE S030(ZPPM).
  ENDIF.

ENDFORM. " GET_DATA
