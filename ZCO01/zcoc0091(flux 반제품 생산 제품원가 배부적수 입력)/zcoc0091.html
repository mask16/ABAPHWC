FORM dynp_values_read USING p_field.
  DATA: lt_dynpfields TYPE TABLE OF dynpread,
        ls_dynpfields TYPE dynpread,
        l_fieldname   TYPE fieldname.

  l_fieldname = p_field.
  MOVE l_fieldname TO ls_dynpfields-fieldname.
  APPEND ls_dynpfields TO lt_dynpfields.
  FIELD-SYMBOLS: <fs>.

  ASSIGN (l_fieldname) TO <fs>.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname     = sy-repid
      dynumb     = sy-dynnr
    TABLES
      dynpfields = lt_dynpfields.

  CHECK sy-subrc EQ 0.
  READ TABLE lt_dynpfields INTO ls_dynpfields WITH KEY fieldname = l_fieldname.
  <fs> = ls_dynpfields-fieldvalue.
ENDFORM. " DYN_VALUE_READ

FORM get_data .
  CLEAR: gt_data[], gs_data.

  DATA: lt_ckmlmv013 TYPE TABLE OF ckmlmv013,
        ls_ckmlmv013 TYPE ckmlmv013,
        lt_coep      TYPE TABLE OF coep,
        ls_coep      TYPE coep.

  DATA: lr_objnr TYPE RANGE OF j_objnr,
        ls_objnr LIKE LINE OF lr_objnr,
        lr_kstar TYPE RANGE OF kstar,
        ls_kstar LIKE LINE OF lr_kstar.

  DATA: l_objnr TYPE j_objnr,
        l_perio TYPE co_perio.
  MOVE p_perio TO l_perio.

  "CKMLMV013 ---
  SELECT pkosa prwrk pmatn
    INTO CORRESPONDING FIELDS OF TABLE lt_ckmlmv013
    FROM ckmlmv013
   WHERE pkosa    NE space
     AND flg_wbwg EQ 'X'
     AND loekz    EQ space
     AND prwrk    IN s_werks
     AND pmatn    IN s_matnr.

  CHECK lt_ckmlmv013[] IS NOT INITIAL.
  SORT lt_ckmlmv013 BY pkosa.
*  DELETE ADJACENT DUPLICATES FROM lt_ckmlmv013 COMPARING pkosa.

  LOOP AT lt_ckmlmv013 INTO ls_ckmlmv013.
    CONCATENATE 'OR' ls_ckmlmv013-pkosa INTO l_objnr.
    _set_ranges:  lr_objnr ls_objnr 'I' 'EQ' l_objnr ''.
    CLEAR: ls_ckmlmv013, l_objnr, ls_objnr.
  ENDLOOP.

  "반제품만.
  _set_ranges: lr_kstar ls_kstar 'I' 'EQ' '0057950020' ''.

  "COEP ------
  SELECT  kokrs
          objnr
          SUM( mbgbtr ) AS mbgbtr
    INTO CORRESPONDING FIELDS OF TABLE lt_coep
    FROM coep
   WHERE kokrs  EQ p_kokrs
     AND objnr  IN lr_objnr
     AND gjahr  EQ p_gjahr
     AND perio  EQ l_perio
     AND kstar  IN lr_kstar
   GROUP BY kokrs objnr.

  SORT lt_coep BY objnr.

  READ TABLE gt_stagr INTO gs_stagr WITH KEY stagr = p_stagr.

  LOOP AT lt_ckmlmv013 INTO ls_ckmlmv013.
    CLEAR: gs_data, ls_coep, gs_makt, gs_mbew.

    MOVE: p_kokrs             TO gs_data-kokrs,
          p_gjahr             TO gs_data-gjahr,
          l_perio             TO gs_data-perio,
          p_stagr             TO gs_data-stagr,
          ls_ckmlmv013-pkosa  TO gs_data-pkosa,
          ls_ckmlmv013-prwrk  TO gs_data-werks,
          ls_ckmlmv013-pmatn  TO gs_data-matnr.
    CONCATENATE 'OR' ls_ckmlmv013-pkosa INTO gs_data-objnr.

    READ TABLE: lt_coep INTO ls_coep WITH KEY objnr = gs_data-objnr.
    CHECK sy-subrc EQ 0.

    READ TABLE: gt_makt INTO gs_makt WITH KEY matnr = gs_data-matnr,
                gt_mbew INTO gs_mbew WITH KEY matnr = gs_data-matnr
                                              bwkey = gs_data-werks.

    MOVE: gs_makt-maktx TO gs_data-maktx,
          gs_makt-meins TO gs_data-meins,
          gs_mbew-zplp3 TO gs_data-zplp3,
          gs_mbew-zpld3 TO gs_data-zpld3,
          tka01-waers   TO gs_data-waers,
          gs_stagr-bezei TO gs_data-bezei.

    gs_data-mbgbtr = ls_coep-mbgbtr * ( -1 ).
    gs_data-amount = gs_data-zplp3 * gs_data-mbgbtr.
*    PERFORM round_amount_by_1000 USING  gs_data-waers
*                                  CHANGING gs_data-amount.
    APPEND gs_data TO gt_data.

    CLEAR: ls_ckmlmv013.
  ENDLOOP.
ENDFORM. " GET_DATA

FORM get_base_data .
  CLEAR: tka01, gt_mbew[], gs_mbew.

  SELECT SINGLE *
  FROM tka01
 WHERE kokrs = p_kokrs.

  SELECT matnr bwkey zplp3 zpld3
    INTO CORRESPONDING FIELDS OF TABLE gt_mbew
    FROM mbew
   WHERE matnr IN s_matnr
     AND bwkey IN s_werks.
ENDFORM. " GET_BASE_DATA

*&---------------------------------------------------------------------*
*&      Form  INIT_DATA
*&---------------------------------------------------------------------*
FORM init_data .
  IF gt_stagr[] IS INITIAL.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE gt_stagr
      FROM tka03 AS a INNER JOIN tkt03 AS b
                                       ON a~kokrs = b~kokrs
                                      AND a~stagr = b~stagr
     WHERE b~spras EQ sy-langu.
  ENDIF.

  IF gt_makt[] IS INITIAL.
    SELECT a~matnr a~meins b~maktx
      INTO CORRESPONDING FIELDS OF TABLE gt_makt
      FROM mara AS a INNER JOIN makt AS b ON a~matnr = b~matnr
     WHERE a~matnr IN s_matnr
       AND b~spras EQ sy-langu.
  ENDIF.
ENDFORM. " INIT_DATA
