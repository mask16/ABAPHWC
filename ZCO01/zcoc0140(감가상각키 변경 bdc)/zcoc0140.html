FORM BDC_AS02_PROCESSING .

  DATA: lt_rows TYPE lvc_t_row,
        ls_rows TYPE lvc_s_row.

  DATA: l_tcode_abaa  TYPE tcode VALUE 'AS02',
        l_mode(1)     TYPE c VALUE 'N',
        l_amt         TYPE string,
        l_rowcnt      TYPE i VALUE 15,
        l_total(10)   TYPE c,
        l_message     TYPE bapi_msg.

  DATA: ls_opt  TYPE ctu_params.
  ls_opt-dismode  = p_mode.
  ls_opt-defsize  = 'X'.

  PERFORM get_selected_rows TABLES lt_rows USING g_grid.
  CHECK lt_rows[] IS NOT INITIAL.

  CLEAR: gt_result[], g_tot, g_sus, g_err.

  LOOP AT lt_rows INTO ls_rows WHERE INDEX IS NOT INITIAL.
    CLEAR: bdcdata[], bdcdata, msgdata[], msgdata, gs_data.
    CLEAR: gs_result, l_amt.

    READ TABLE gt_data INTO gs_data INDEX ls_rows-index.

*    PERFORM: currency_sap_to_display USING gs_data-waers
*                                  CHANGING gs_data-zuumb,
*             get_money_char          USING gs_data-zuumb l_amt gs_data-waers.

    PERFORM bdc_dynpro USING:
      'X' 'SAPLAIST'            '0100',
      ' ' 'BDC_OKCODE'      '/00',
      ' ' 'ANLA-ANLN1'  gs_data-anln1,
      ' ' 'ANLA-ANLN2'  gs_data-anln2,
      ' ' 'ANLA-BUKRS'  gs_data-bukrs,

      'X' 'SAPLAIST'            '1000',
      ' ' 'BDC_OKCODE'      '=TAB08',

      'X' 'SAPLAIST'            '1000',
      ' ' 'BDC_OKCODE'      '=SELZ',

      'X' 'SAPLAIST'            '0195',
      ' ' 'BDC_OKCODE'      '=TDDP_OVRVW',

      'X' 'SAPLAIST'            '0195',
      ' ' 'BDC_OKCODE'      '=TDDP_CINV',

      'X' 'SAPLAIST'            '3011',
      ' ' 'BDC_OKCODE'      '=ENTE',
      ' ' 'ANLBZA-ADATU'  gs_data-adatu,

      'X' 'SAPLAIST'            '0195',
      ' ' 'BDC_OKCODE'      '=BUCH',
      ' ' 'ANLBZA-AFASL(02)'  gs_data-afasl.

    CALL TRANSACTION l_tcode_abaa
               USING bdcdata
        OPTIONS FROM ls_opt
       MESSAGES INTO msgdata.

    IF sy-subrc EQ 0.
      ADD 1 TO g_sus.
      gs_data-icon = '@01@'.
      COMMIT WORK AND WAIT.
      WAIT UP TO '0.5' SECONDS.
    ELSE.
      ADD 1 TO g_err.
      gs_data-icon = '@02@'.
      READ TABLE msgdata WITH KEY msgtyp = 'E'.
      PERFORM get_message_build USING msgdata-msgid
                                      msgdata-msgnr
                                      msgdata-msgv1
                                      msgdata-msgv2
                                      msgdata-msgv3
                                      msgdata-msgv4
                                      gs_data-msg.
    ENDIF.

    ADD 1 TO g_tot.

    MODIFY gt_data FROM gs_data INDEX ls_rows-index transporting icon msg .
