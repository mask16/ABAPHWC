*&---------------------------------------------------------------------*
*&      Form  CHANGED_CELLS
*&---------------------------------------------------------------------*
FORM changed_cells USING p_data_changed
                      TYPE REF TO cl_alv_changed_data_protocol.

  CHECK p_data_changed->mt_good_cells[] IS NOT INITIAL.
  DATA: ls_cells TYPE lvc_s_modi,
        lt_cells TYPE lvc_t_modi,
        l_value  TYPE lvc_value,
        l_kostl  TYPE kostl,
        l_fname  TYPE fieldname.

  lt_cells[] = p_data_changed->mt_good_cells[].
  LOOP AT lt_cells INTO ls_cells.
    l_value = ls_cells-value.
    CHECK l_value IS NOT INITIAL.

    CLEAR: gs_users, gs_cskt.
    CASE ls_cells-fieldname.
      WHEN 'BNAME'.
        READ TABLE gt_users INTO gs_users WITH KEY name = l_value.
        PERFORM modify_cell_value USING p_data_changed
                                        ls_cells-row_id
                                        'NAME_TEXT'
                                        gs_users-value.
      WHEN 'KOSTL1' OR 'KOSTL2'.
*        l_kostl = l_value.
*        PERFORM conversion_alpha_input CHANGING l_kostl.
        READ TABLE gt_cskt INTO gs_cskt WITH KEY kostl = l_value.
        CONCATENATE 'KTEXT' ls_cells-fieldname+5(1) INTO l_fname.
        PERFORM modify_cell_value USING p_data_changed
                                        ls_cells-row_id
                                        l_fname
                                        gs_cskt-ktext.
    ENDCASE.

    CLEAR: ls_cells, l_value, l_fname.
  ENDLOOP.

ENDFORM. " CHANGED_CELLS
