FORM get_excel_file TABLES p_table
                     USING  p_filename  LIKE rlgrap-filename.

  DATA : it_line  TYPE alsmex_tabline OCCURS 0 WITH HEADER LINE.
  DATA : l_index LIKE sy-tabix.
  DATA : l_start_col TYPE i VALUE '1',
         l_start_row TYPE i VALUE '1',
         l_end_col   TYPE i VALUE '256',
         l_end_row   TYPE i VALUE '65536'.
  FIELD-SYMBOLS : <fs>.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_filename
      i_begin_col             = l_start_col
      i_begin_row             = l_start_row
      i_end_col               = l_end_col
      i_end_row               = l_end_row
    TABLES
      intern                  = it_line
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.
  IF sy-subrc <> 0.
    MESSAGE s016(rp)   WITH 'File Upload Failed !'  .
    EXIT.
  ENDIF.

  CHECK NOT it_line[] IS INITIAL.

  SORT it_line BY row col.

  LOOP AT it_line.
    MOVE : it_line-col TO l_index.
    ASSIGN COMPONENT l_index OF STRUCTURE p_table TO <fs>.
    MOVE : it_line-value TO <fs>.
    AT END OF row.
      APPEND p_table.
      CLEAR p_table.
    ENDAT.
  ENDLOOP.

ENDFORM. " GET_EXCEL_FILE
