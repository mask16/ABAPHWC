FORM get_data .

  DATA: BEGIN OF lt_last_tab OCCURS 0,
        matnr       LIKE zcot0097-matnr,    "자재코드
        mbwbest_l   LIKE zcot0097-mbwbest,  "전월 기말재고 수량
        wbwbest_l   LIKE zcot0097-wbwbest,  "전월 기말재고 금액
        waers       LIKE zcot0097-waers,    "통화 키
        END OF lt_last_tab.

  DATA: BEGIN OF lt_this_tab OCCURS 0,
        matnr       LIKE zcot0097-matnr,    "자재코드
        mbwbest_t   LIKE zcot0097-mbwbest,  "전월 기말재고 수량
        wbwbest_t   LIKE zcot0097-wbwbest,  "전월 기말재고 금액
        waers       LIKE zcot0097-waers,    "통화 키
        END OF lt_this_tab.

  DATA: BEGIN OF lt_all_tab OCCURS 0,
        matnr       LIKE zcot0097-matnr,    "자재코드
        END OF lt_all_tab.

  DATA: BEGIN OF lt_mara_tab OCCURS 0,
        matnr     LIKE mara-matnr,
        mtart     LIKE mara-mtart,
        END OF lt_mara_tab.

  DATA: BEGIN OF lt_makt_tab OCCURS 0,
        matnr     LIKE makt-matnr,
        maktx     LIKE makt-maktx,
        END OF lt_makt_tab.

  DATA: lv_mbwbest_l   LIKE zcot0097-mbwbest,
        lv_wbwbest_l   LIKE zcot0097-wbwbest,
        lv_mbwbest_t   LIKE zcot0097-mbwbest,
        lv_wbwbest_t   LIKE zcot0097-wbwbest.

  DATA: ld_this_date   LIKE sy-datlo,
        ld_last_date   LIKE sy-datlo,
        ld_last_spmon  LIKE zcot0097-spmon.

* 전월.
  CONCATENATE p_spmon '01' INTO ld_this_date.
  ld_last_date   = ld_this_date - 1.
  CLEAR gv_last_spmon.
  gv_last_spmon  = ld_last_spmon  = ld_last_date(6).

* 전월 데이터.
  CLEAR: lt_last_tab,     lt_last_tab[].
  SELECT matnr
         mbwbest AS mbwbest_l
         wbwbest AS wbwbest_l
         waers
         APPENDING CORRESPONDING FIELDS OF TABLE lt_last_tab
         FROM zcot0097
        WHERE spmon =  ld_last_spmon
          AND matnr IN s_matnr.
  SORT lt_last_tab BY matnr.

* 당월 데이터.
  CLEAR: lt_this_tab,     lt_this_tab[].
  SELECT matnr
         mbwbest AS mbwbest_t
         wbwbest AS wbwbest_t
         waers
         APPENDING CORRESPONDING FIELDS OF TABLE lt_this_tab
         FROM zcot0097
        WHERE spmon =  p_spmon
          AND matnr IN s_matnr.
  SORT lt_this_tab BY matnr.

* 전월 + 당월 데이터
  CLEAR: lt_all_tab,     lt_all_tab[].
  LOOP AT lt_last_tab.
    MOVE: lt_last_tab-matnr  TO lt_all_tab-matnr.
    APPEND lt_all_tab.  CLEAR lt_all_tab.
  ENDLOOP.

  LOOP AT lt_this_tab.
    MOVE: lt_this_tab-matnr  TO lt_all_tab-matnr.
    APPEND lt_all_tab.  CLEAR lt_all_tab.
  ENDLOOP.

  SORT lt_all_tab BY matnr.
  DELETE ADJACENT DUPLICATES FROM lt_all_tab COMPARING matnr.

  CHECK NOT lt_all_tab[] IS INITIAL.

* 자재유형.
  CLEAR: lt_mara_tab,     lt_mara_tab[].
  SELECT matnr
         mtart
         INTO CORRESPONDING FIELDS OF TABLE lt_mara_tab
         FROM mara FOR ALL ENTRIES IN lt_all_tab
        WHERE matnr  =  lt_all_tab-matnr
          AND mtart  IN s_mtart.
  SORT lt_mara_tab BY matnr.

* 자재내역
  CLEAR: lt_makt_tab,     lt_makt_tab[].
  SELECT matnr
         maktx
         INTO CORRESPONDING FIELDS OF TABLE lt_makt_tab
         FROM makt FOR ALL ENTRIES IN lt_all_tab
        WHERE matnr  =  lt_all_tab-matnr
          AND spras  = sy-langu.
  SORT lt_makt_tab BY matnr.

* 데이터 채우기.
  CLEAR: gt_disp_tab,     gt_disp_tab[].
  LOOP AT lt_all_tab.
    CLEAR: gt_disp_tab.
    MOVE: lt_all_tab-matnr     TO gt_disp_tab-matnr.

    CLEAR: lt_mara_tab.
    READ TABLE lt_mara_tab WITH KEY matnr = gt_disp_tab-matnr
                                BINARY SEARCH.
    IF sy-subrc = 0.
      MOVE: lt_mara_tab-mtart   TO gt_disp_tab-mtart.
    ELSE.
      CONTINUE. "선택조건 및 마스터에 없을 경우.
    ENDIF.

    CLEAR: lt_makt_tab.
    READ TABLE lt_makt_tab WITH KEY matnr = gt_disp_tab-matnr
                                BINARY SEARCH.
    IF sy-subrc = 0.
      MOVE: lt_makt_tab-maktx   TO gt_disp_tab-maktx.
    ENDIF.

* 전월합계.
    CLEAR: lv_mbwbest_l,     lv_wbwbest_l.
    LOOP AT lt_last_tab WHERE matnr = gt_disp_tab-matnr.
      lv_mbwbest_l = lv_mbwbest_l + lt_last_tab-mbwbest_l.
      lv_wbwbest_l = lv_wbwbest_l + lt_last_tab-wbwbest_l.
    ENDLOOP.

    MOVE: lv_mbwbest_l     TO gt_disp_tab-mbwbest_l,
          lv_wbwbest_l     TO gt_disp_tab-wbwbest_l.
    IF lv_mbwbest_l > 0.
      gt_disp_tab-verpr_l = lv_wbwbest_l / lv_mbwbest_l.
      MOVE: lt_last_tab-waers   TO gt_disp_tab-waers.
    ENDIF.

* 당월합계.
    CLEAR: lv_mbwbest_t,     lv_wbwbest_t.
    LOOP AT lt_this_tab WHERE matnr = gt_disp_tab-matnr.
      lv_mbwbest_t = lv_mbwbest_t + lt_this_tab-mbwbest_t.
      lv_wbwbest_t = lv_wbwbest_t + lt_this_tab-wbwbest_t.
    ENDLOOP.

    MOVE: lv_mbwbest_t     TO gt_disp_tab-mbwbest_t,
          lv_wbwbest_t     TO gt_disp_tab-wbwbest_t.
    IF lv_mbwbest_t > 0.
      gt_disp_tab-verpr_t = lv_wbwbest_t / lv_mbwbest_t.
      MOVE: lt_this_tab-waers   TO gt_disp_tab-waers.
    ENDIF.

* 전월/당월 단가차이.
    gt_disp_tab-diff_verpr
           = gt_disp_tab-verpr_t - gt_disp_tab-verpr_l.

    APPEND gt_disp_tab.
    CLEAR gt_disp_tab.

  ENDLOOP.

ENDFORM. " GET_DATA

FORM set_fieldcat TABLES p_tabdata
                   USING pt_fieldcat TYPE lvc_t_fcat
                         p_tabname.

  DATA: lv_text(72).

  DATA: lv_row_pos    LIKE lvc_s_fcat-row_pos.
  DATA: lt_fieldcat   TYPE slis_t_fieldcat_alv,
        ls_fieldcat   LIKE LINE OF lt_fieldcat,
        lt_alv_cat    TYPE lvc_t_fcat,
        ls_alv_cat    LIKE LINE OF lt_alv_cat.

  PERFORM reuse_alv_fieldcat USING p_tabname
                          CHANGING lt_fieldcat.

  PERFORM complete_fieldcat  TABLES p_tabdata
                           CHANGING lt_fieldcat
                                    lt_alv_cat.

  CLEAR lv_row_pos.
  LOOP AT lt_alv_cat INTO ls_alv_cat.
    CASE ls_alv_cat-fieldname.
        lv_row_pos = lv_row_pos + 1.
*-[S] Input fields.
      WHEN 'MTART'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-key         = 'X'.
        ls_alv_cat-coltext     = text-d01.

      WHEN 'MATNR'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-key         = 'X'.
        ls_alv_cat-coltext     = text-d02.

      WHEN 'MAKTX'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d03.

      WHEN 'MBWBEST_L'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d04.
        ls_alv_cat-no_zero     = 'X'.

      WHEN 'WBWBEST_L'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d05.
        ls_alv_cat-tabname     = 'GT_DISP_TAB'.
        ls_alv_cat-cfieldname  = 'WAERS'.
        ls_alv_cat-no_zero     = 'X'.

      WHEN 'VERPR_L'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-no_zero     = 'X'.

        CLEAR lv_text.
        CONCATENATE '['
                    gv_last_spmon
                    ']'
                    text-d06 INTO lv_text.
        ls_alv_cat-coltext     = lv_text.

      WHEN 'MBWBEST_T'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d07.
        ls_alv_cat-no_zero     = 'X'.

      WHEN 'WBWBEST_T'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d08.
        ls_alv_cat-tabname     = 'GT_DISP_TAB'.
        ls_alv_cat-cfieldname  = 'WAERS'.
        ls_alv_cat-no_zero     = 'X'.

      WHEN 'VERPR_T'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-no_zero     = 'X'.

        CLEAR lv_text.
        CONCATENATE '['
                    p_spmon
                    ']'
                    text-d09 INTO lv_text.
        ls_alv_cat-coltext     = lv_text.

      WHEN 'DIFF_VERPR'.
        ls_alv_cat-row_pos     = lv_row_pos.
        ls_alv_cat-coltext     = text-d10.
        ls_alv_cat-no_zero     = 'X'.

      WHEN OTHERS.
        ls_alv_cat-no_out      = 'X'.
    ENDCASE.

    MODIFY lt_alv_cat FROM ls_alv_cat.

  ENDLOOP.

  pt_fieldcat = lt_alv_cat[].

ENDFORM. " SET_FIELDCAT
