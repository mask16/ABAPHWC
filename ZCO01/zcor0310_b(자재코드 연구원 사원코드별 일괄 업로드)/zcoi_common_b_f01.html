                      CHANGING p_output  TYPE  any.

        lv_length TYPE i ,

      CONCATENATE lv_number_out lv_one INTO lv_number_out.
    ELSE.
      p_output = 'X'.
      EXIT.
*      RAISE no_numeric.
    ENDIF.

    SHIFT lv_number.
  ENDDO.
