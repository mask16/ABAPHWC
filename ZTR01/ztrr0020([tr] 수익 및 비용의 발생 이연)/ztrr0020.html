FORM CHECK_AUTHORITY
  USING
    PI_BUKRS
    PI_RFHA
    PI_SGSART
    PI_DBLFZ
    PI_DELFZ
  CHANGING
    EX_FLAG_ALLOWED TYPE C.

  DATA:
    L_DUMMY       TYPE TRDIR-SECU,
    L_TAX_AUDITOR TYPE FLAG.

* If any authority check fails, the record is to be rejected.
* So only continue checking if the current check passes.
*
* For future changes, be aware that if all authority checks are
* processed in order, then the sy-subrc will be reset to 0
* if one check fails but the next check passes.  This results in
* the record not being rejected correctly (see Note 1101578).

  EX_FLAG_ALLOWED = XTRUE.

  AUTHORITY-CHECK OBJECT 'F_INFO_BUK'
     ID 'BUKRS' FIELD PI_BUKRS
     ID 'GSART' FIELD PI_SGSART
     ID 'SECU'  FIELD L_DUMMY.

  IF SY-SUBRC <> 0.
    EX_FLAG_ALLOWED = XFALSE.

  ELSE.
*   check company code seperately
    AUTHORITY-CHECK OBJECT 'T_DEAL_PD'
          ID 'BUKRS'   FIELD PI_BUKRS
          ID 'GSART'   DUMMY
          ID 'SFHAART' DUMMY
          ID 'TRFCT'   DUMMY
          ID 'ACTVT'   DUMMY.
    IF SY-SUBRC <> 0.
      EX_FLAG_ALLOWED = XFALSE.

    ELSE.
*     check for GDPdU
      CALL FUNCTION 'FTR_GDPDU_CHECK_OVERLAP'
        EXPORTING
          IM_APPLICATION     = 'TR-TM'
          IM_COMPANY_CODE    = PI_BUKRS
          IM_USER            = SY-UNAME
          IM_PROGRAM         = SY-CPROG
          IM_START_DATE      = PI_DBLFZ
          IM_END_DATE        = PI_DELFZ
        IMPORTING
          EX_FLG_TAX_AUDITOR = L_TAX_AUDITOR
        EXCEPTIONS
          NO_AUTHORITY       = 1
          WRONG_INPUT        = 2
          OTHERS             = 3.
      IF SY-SUBRC <> 0.
        EX_FLAG_ALLOWED = XFALSE.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM. " check_authority
