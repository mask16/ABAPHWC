FORM SELECT_RTN .

  _CLEAR : R_RANTYP, GT_VTBF, GT_MAIN, GT_BUT, GT_TZPA.

*  IF P_CB1 = 'X'.
    _RANGES : R_RANTYP  'EQ'  '5'  ''.
*  ENDIF.

*  IF P_CB2 = 'X'.
*    _RANGES : R_RANTYP  'EQ'  '2'  ''.
*  ENDIF.
*
*  IF P_CB1 = '' AND P_CB2 = ''.
*    _RANGES : R_RANTYP  'EQ'  ''  ''.
*  ENDIF.

  SELECT * INTO TABLE GT_VTBF
    FROM VTBFHA
   WHERE BUKRS     = P_BUKRS
     AND RFHA     IN S_RFHA
     AND RANTYP   IN R_RANTYP[]
     AND SGSART   IN S_SGSART
     AND KONTRH   IN S_KONTRH
     AND SFHAART  IN S_SFHAAR
     AND SAKTIV    = '0'.

  SELECT PARTNER  NAME_ORG1 BU_SORT1 INTO TABLE GT_BUT
    FROM BUT000
   WHERE PARTNER   LIKE '%'.

  SELECT GSART  LTX INTO TABLE GT_TZPA
    FROM TZPAT
   WHERE SPRAS  = SY-LANGU.

  SORT : GT_BUT   BY PARTNER,
         GT_TZPA  BY GSART.

ENDFORM. " SELECT_RTN

FORM MAKE_DATA .

  DATA : LT_HAPO   LIKE VTBFHAPO  OCCURS 0 WITH HEADER LINE.
  DATA : LS_HAPO   LIKE VTBFHAPO,
         L_DZTERM  LIKE VTBFHAPO-DZTERM,
         L_DAY     TYPE I.

  LOOP AT GT_VTBF.

    _CLEAR LT_HAPO.

    SELECT * INTO TABLE LT_HAPO
      FROM VTBFHAPO
     WHERE BUKRS    = P_BUKRS
       AND RFHA     = GT_VTBF-RFHA
       AND DZTERM  IN S_DZTERM
       AND SFHAZBA IN ('1100', '1110', '1120', '1105')
       AND SBEWEBE  < '3'.

    LOOP AT LT_HAPO.

      CLEAR : GT_MAIN, GT_BUT, GT_TZPA, L_DAY.

      IF LT_HAPO-SFHAZBA  = '1100' AND GT_VTBF-SANLF = '510'.
        GT_MAIN-GUBUN  = '가입'.
      ELSEIF LT_HAPO-SFHAZBA  = '1105' AND GT_VTBF-SANLF = '550'.
        GT_MAIN-GUBUN  = '차입'.
      ELSEIF LT_HAPO-SFHAZBA  = '1120' AND GT_VTBF-SANLF = '510'.
        GT_MAIN-GUBUN  = '회수'.
      ELSEIF LT_HAPO-SFHAZBA  = '1120' AND GT_VTBF-SANLF = '550'.
        GT_MAIN-GUBUN  = '상환'.
      ENDIF.

      GT_MAIN-RFHA    = GT_VTBF-RFHA.
      GT_MAIN-KONTRH  = GT_VTBF-KONTRH.
      GT_MAIN-SGSART  = GT_VTBF-SGSART.
      GT_MAIN-DZTERM  = LT_HAPO-DZTERM.
      GT_MAIN-WZBETR  = LT_HAPO-WZBETR.
      GT_MAIN-BZBETR  = LT_HAPO-BZBETR.
      GT_MAIN-RANTYP  = LT_HAPO-RANTYP.

      CASE GT_VTBF-RANTYP.
        WHEN '5'.       " 머니마켓
          SELECT SINGLE DBLFZ DELFZ INTO (GT_MAIN-DEBEG, GT_MAIN-DENDF)
            FROM VTBFHAZU
           WHERE BUKRS    = P_BUKRS
             AND RFHA     = GT_VTBF-RFHA.

          SELECT PKOND INTO GT_MAIN-PKOND1
            FROM VTBFINKO
           WHERE BUKRS     = P_BUKRS
             AND RFHA      = GT_VTBF-RFHA
             AND DGUEL_KP <= GT_MAIN-DZTERM
             AND SKOART   IN ('1200', '1150', '1201').
          ENDSELECT.

        WHEN '2'.       " 유가증권

      ENDCASE.

      READ TABLE GT_BUT   WITH KEY PARTNER  = GT_MAIN-KONTRH
                           BINARY SEARCH.
      GT_MAIN-NAME1   = GT_BUT-NAME1.
      IF GT_BUT-NAME1 IS INITIAL.
        GT_MAIN-NAME1  = GT_BUT-BU_SORT1.
      ENDIF.

      READ TABLE GT_TZPA  WITH KEY GSART   = GT_MAIN-SGSART
                           BINARY SEARCH.
      GT_MAIN-LTX     = GT_TZPA-LTX.

      SELECT SINGLE LOANNO INTO GT_MAIN-LOANNO
        FROM ZTRT0041
       WHERE BUKRS    = P_BUKRS
         AND RFHA     = GT_MAIN-RFHA.

      COLLECT GT_MAIN.

    ENDLOOP.

  ENDLOOP.

* 최종이율
  LOOP AT GT_MAIN.

    CLEAR : LS_HAPO, L_DZTERM, GT_MAIN-PKOND2.

    SELECT MAX( DZTERM ) INTO L_DZTERM
      FROM VTBFHAPO
     WHERE BUKRS    = P_BUKRS
       AND RFHA     = GT_MAIN-RFHA
       AND DZTERM  IN S_DZTERM
       AND SFHAZBA IN ('1200', '1201', '1150', '0870')
       AND SBEWEBE  < '3'.

    CHECK L_DZTERM IS NOT INITIAL.

    SELECT SINGLE * INTO LS_HAPO
      FROM VTBFHAPO
     WHERE BUKRS    = P_BUKRS
       AND RFHA     = GT_MAIN-RFHA
       AND DZTERM   = L_DZTERM
       AND SFHAZBA IN ('1200', '1201', '1150', '0870')
       AND SBEWEBE  < '3'.

    CHECK SY-SUBRC = 0 AND LS_HAPO-BBASIS <> 0 AND LS_HAPO-ABASTAGE <> 0.

    GT_MAIN-PKOND2  = ( LS_HAPO-BZBETR * LS_HAPO-ABASTAGE * 100 ) /
                      ( LS_HAPO-BBASIS * LS_HAPO-ATAGE ).

    MODIFY GT_MAIN.

  ENDLOOP.

ENDFORM. " MAKE_DATA

FORM DOUBLE_CLICK USING P_ROW
                            P_COLUMN.

  DATA: LT_ROWID        TYPE LVC_T_ROW,
        LS_ROWID        TYPE LINE OF LVC_T_ROW,
        L_FUNCTIONCLASS TYPE  TB_FUNCTIONCLASS.

  CHECK P_ROW+(1) IS INITIAL.

  CASE P_COLUMN.

    WHEN 'RFHA'.

      READ TABLE GT_MAIN INDEX P_ROW.

      CHECK NOT GT_MAIN-RFHA IS INITIAL.

      L_FUNCTIONCLASS = 'DISPLAY'.

      CALL FUNCTION 'FTR_DISPATCH'
        EXPORTING
          I_BUKRS                = P_BUKRS
          I_RFHA                 = GT_MAIN-RFHA
          I_FUNCTIONCLASS        = L_FUNCTIONCLASS
        EXCEPTIONS
          NOT_FOUND              = 1
          FUNCTION_NOT_SUPPORTED = 2
          OTHERS                 = 3.

      IF SY-SUBRC <> 0.

        MESSAGE ID SY-MSGID TYPE 'S' NUMBER SY-MSGNO
                WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4
                DISPLAY LIKE SY-MSGTY.

      ENDIF.

  ENDCASE.

ENDFORM. " DOUBLE_CLICK
