FORM sub_total .

         lv_lev2    TYPE char1,
         lv_lqpos   LIKE flqlpost-lqpos,
         lv_dmbtr   LIKE bsis-dmbtr.

  DATA : lt_main LIKE gt_main OCCURS 0 WITH HEADER LINE.

  DATA : BEGIN OF lt_lqpos OCCURS 0,
           lqpos LIKE ztrt0060-lqpos,
         END OF lt_lqpos.

  lt_lqpos-lqpos = 'AA9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.
  lt_lqpos-lqpos = 'AB9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.
  lt_lqpos-lqpos = 'AC9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.
  lt_lqpos-lqpos = 'BA9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.
  lt_lqpos-lqpos = 'BB9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.
  lt_lqpos-lqpos = 'BC9999999'. APPEND lt_lqpos. CLEAR : lt_lqpos.

  CLEAR : gt_main. SORT gt_main BY cod1 cod2.
*-- 합계 값 구해서 넣기
  LOOP AT gt_main.
    AT NEW cod2.
      CLEAR : lv_lev1, lv_lev2, lv_lqpos, lv_dmbtr.

      lv_lev1 = gt_main-cod1. lv_lev2 = gt_main-cod2.
      CONCATENATE lv_lev1 lv_lev2 '9999999' INTO lv_lqpos.
    ENDAT.

    lv_dmbtr = gt_main-dmbtr + lv_dmbtr.

    AT END OF cod2.
      CLEAR : lt_main.

      CASE lv_lev1.
        WHEN c_a. lt_main-lev1 = text-001. lt_main-cod1 = lv_lev1.
        WHEN c_b. lt_main-lev1 = text-002. lt_main-cod1 = lv_lev1.
      ENDCASE.

      CASE lv_lev2.
        WHEN c_a.
          lt_main-cod2 = lv_lev2.
          IF lt_main-cod1 = c_a.
            lt_main-lev2 = text-036.
          ELSEIF lt_main-cod1 = c_b.
            lt_main-lev2 = text-040.
          ENDIF.
        WHEN c_b.
          lt_main-cod2 = lv_lev2.
          IF lt_main-cod1 = c_a.
            lt_main-lev2 = text-037.
          ELSEIF lt_main-cod1 = c_b.
            lt_main-lev2 = text-041.
          ENDIF.
        WHEN c_c.
          lt_main-cod2 = lv_lev2.
          IF lt_main-cod1 = c_a.
            lt_main-lev2 = text-038.
          ELSEIF lt_main-cod1 = c_b.
            lt_main-lev2 = text-042.
          ENDIF.
      ENDCASE.

      lt_main-cod3 = '99'.
      lt_main-lqpos = lv_lqpos.
      lt_main-dmbtr = lv_dmbtr.
      APPEND lt_main. CLEAR : lt_main.
    ENDAT.
  ENDLOOP.
*-- DATA가 존재하지 않을시 합계값 넣기
  CLEAR lt_main.
  IF lt_main[] IS INITIAL.
    lt_main-lev1 = text-001. lt_main-cod1 = c_a.
    lt_main-lev2 = text-036. lt_main-cod2 = c_a. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.

    lt_main-lev1 = text-001. lt_main-cod1 = c_a.
    lt_main-lev2 = text-037. lt_main-cod2 = c_b. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.

    lt_main-lev1 = text-001. lt_main-cod1 = c_a.
    lt_main-lev2 = text-038. lt_main-cod2 = c_c. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.

    lt_main-lev1 = text-002. lt_main-cod1 = c_b.
    lt_main-lev2 = text-040. lt_main-cod2 = c_a. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.

    lt_main-lev1 = text-002. lt_main-cod1 = c_b.
    lt_main-lev2 = text-041. lt_main-cod2 = c_b. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.

    lt_main-lev1 = text-002. lt_main-cod1 = c_b.
    lt_main-lev2 = text-042. lt_main-cod2 = c_c. lt_main-cod3 = '99'.
    CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
    APPEND lt_main. CLEAR : lt_main.
  ENDIF.

*-- 합계값이 일부만 존재할시 빠진 합계 넣기
  CLEAR : lt_lqpos, lt_main. SORT lt_main BY lqpos.
  LOOP AT lt_lqpos.
    READ TABLE lt_main WITH KEY lqpos = lt_lqpos-lqpos BINARY SEARCH.
    IF sy-subrc <> 0.
      CLEAR : lt_main.
      CASE lt_lqpos-lqpos.
        WHEN 'AA9999999'.
          lt_main-lev1 = text-001. lt_main-cod1 = c_a.
          lt_main-lev2 = text-036. lt_main-cod2 = c_a. lt_main-cod3 = '99'.
          CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
          APPEND lt_main. CLEAR : lt_main. SORT lt_main BY lqpos.

        WHEN 'AB9999999'.
          lt_main-lev1 = text-001. lt_main-cod1 = c_a.
          lt_main-lev2 = text-037. lt_main-cod2 = c_b. lt_main-cod3 = '99'.
          CONCATENATE lt_main-cod1 lt_main-cod2 '9999999' INTO lt_main-lqpos.
          APPEND lt_main. CLEAR : lt_main. SORT lt_main BY lqpos.

FORM sub_sum .

         lv_lqpos   LIKE flqlpost-lqpos,
         lv_dmbtr   LIKE bsis-dmbtr.

  CLEAR : gt_main. SORT gt_main BY cod1 cod2.

  LOOP AT gt_main.
    AT NEW cod1.
      CLEAR : lv_lev1, lv_lqpos, lv_dmbtr.

      lv_lev1 = gt_main-cod1.
      CONCATENATE lv_lev1 'D9999999' INTO lv_lqpos.
    ENDAT.

    CASE gt_main-lqpos.
      WHEN 'AA9999999'.
      WHEN 'AB9999999'.
      WHEN 'AC9999999'.
      WHEN 'BA9999999'.
      WHEN 'BB9999999'.
      WHEN 'BC9999999'.
      WHEN OTHERS.
        lv_dmbtr = gt_main-dmbtr + lv_dmbtr.
    ENDCASE.

    AT END OF cod1.
      CLEAR : lt_main.

        WHEN c_b. lt_main-lev1 = text-002. lt_main-cod1 = lv_lev1.

        lt_main-lev2 = text-039.
      ELSEIF lt_main-cod1 = c_b.
        lt_main-lev2 = text-043.
      ENDIF.

      lt_main-lqpos = lv_lqpos.
      lt_main-dmbtr = lv_dmbtr.
      APPEND lt_main. CLEAR : lt_main.
    ENDAT.
  ENDLOOP.

  CLEAR lt_main.

  LOOP AT lt_main.
    MOVE-CORRESPONDING lt_main TO gt_main.
    APPEND gt_main. CLEAR gt_main.
  ENDLOOP.

FORM select_data_header .
  DATA : lt_bsik LIKE bsik OCCURS 0 WITH HEADER LINE,
         lt_bsak LIKE bsak OCCURS 0 WITH HEADER LINE,
         lt_bsid LIKE bsid OCCURS 0 WITH HEADER LINE,
         lt_bsad LIKE bsad OCCURS 0 WITH HEADER LINE.

  DATA : BEGIN OF lt_act OCCURS 0,
           saknr LIKE flqacc_act-saknr,
         END OF lt_act.

  DATA : BEGIN OF lt_bsis OCCURS 0,
           wrbtr LIKE bsis-wrbtr,
           shkzg LIKE bsis-shkzg,
           waers LIKE bsis-waers,
         END OF lt_bsis.

  DATA : lv_hmbtr LIKE bsik-dmbtr,
         lv_smbtr LIKE bsik-dmbtr.

  RANGES : lr_hkont FOR bsis-hkont.

  _clear : gt_wrbtr. CLEAR : gv_ambtr, gv_gmbtr.
*----------------------------------------------------------------------*
* 지급어음
*----------------------------------------------------------------------*
  CLEAR : lv_hmbtr, lv_smbtr, gv_gmbtr.

  SELECT * FROM bsik
    INTO CORRESPONDING FIELDS OF TABLE lt_bsik
   WHERE bukrs EQ c_bukrs AND umskz IN ('U', 'V')
     AND budat LE gv_lqday.

  SELECT * FROM bsak
    INTO CORRESPONDING FIELDS OF TABLE lt_bsak
   WHERE bukrs EQ c_bukrs AND umskz IN ('U', 'V')
     AND budat LE gv_lqday.

  CLEAR : lt_bsik, lv_hmbtr, lv_smbtr.
  SORT lt_bsik BY shkzg.

  LOOP AT lt_bsik.
    IF lt_bsik-shkzg = c_h.
      lv_hmbtr = lv_hmbtr + lt_bsik-dmbtr * -1.
    ELSEIF lt_bsik-shkzg = c_s.
      lv_smbtr = lv_smbtr + lt_bsik-dmbtr.
    ENDIF.
  ENDLOOP.

  gv_gmbtr = lv_smbtr + lv_hmbtr.

  CLEAR : lt_bsak, lv_hmbtr, lv_smbtr.
  SORT lt_bsak BY shkzg.

  LOOP AT lt_bsak.
    IF lt_bsak-shkzg = c_h.
      lv_hmbtr = lv_hmbtr + lt_bsak-dmbtr * -1.
    ELSEIF lt_bsak-shkzg = c_s.
      lv_smbtr = lv_smbtr + lt_bsak-dmbtr.
    ENDIF.
  ENDLOOP.

  gv_gmbtr = gv_gmbtr + lv_smbtr + lv_hmbtr.
*----------------------------------------------------------------------*
* 받을어음
*----------------------------------------------------------------------*
  CLEAR : lv_hmbtr, lv_smbtr, gv_ambtr.

  SELECT * FROM bsid
    INTO CORRESPONDING FIELDS OF TABLE lt_bsid
   WHERE bukrs EQ c_bukrs AND umskz EQ 'W'
     AND budat LE gv_lqday.

  CLEAR : lt_bsid, lv_hmbtr, lv_smbtr.
  SORT lt_bsid BY shkzg.

  LOOP AT lt_bsid.

      lv_hmbtr = lv_hmbtr + lt_bsid-dmbtr * -1.
    ELSEIF lt_bsid-shkzg = c_s.
      lv_smbtr = lv_smbtr + lt_bsid-dmbtr.
    ENDIF.
  ENDLOOP.

  gv_ambtr = lv_smbtr + lv_hmbtr.

  CLEAR : lt_bsad, lv_hmbtr, lv_smbtr.
  SORT lt_bsad BY shkzg.

  LOOP AT lt_bsad.
    IF lt_bsad-shkzg = c_h.
      lv_hmbtr = lv_hmbtr + lt_bsad-dmbtr * -1.
    ELSEIF lt_bsid-shkzg = c_s.
      lv_smbtr = lv_smbtr + lt_bsad-dmbtr.
    ENDIF.
  ENDLOOP.

  gv_ambtr = gv_ambtr + lv_smbtr + lv_hmbtr.
*----------------------------------------------------------------------*
* 외화잔액
*----------------------------------------------------------------------*
  SELECT saknr FROM flqacc_act
    INTO CORRESPONDING FIELDS OF TABLE lt_act
   WHERE ktopl EQ c_bukrs AND bukrs EQ c_bukrs
     AND nonact EQ ''.

  CLEAR : lt_act. SORT lt_act BY saknr.
  DELETE ADJACENT DUPLICATES FROM lt_act COMPARING saknr.

  LOOP AT lt_act. _ranges : lr_hkont 'EQ' lt_act-saknr ''. ENDLOOP.

  SELECT wrbtr waers shkzg FROM bsis
    INTO CORRESPONDING FIELDS OF TABLE lt_bsis
   WHERE bukrs EQ c_bukrs AND budat LE gv_lqday
     AND hkont IN lr_hkont AND waers NE c_krw.

  SORT lt_bsis BY shkzg waers. CLEAR : lv_hmbtr, lv_smbtr.

  LOOP AT lt_bsis.
    AT NEW waers.
      gt_wrbtr-waers = lt_bsis-waers.
