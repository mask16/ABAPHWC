FORM MAKE_DERIVERED_FLOW .

  DATA: L_OLD_DATE    LIKE AT21-DABGUELT.

  LOOP AT GT_AT21.
    IF SY-TABIX = 1.
      GT_AT21-ENDDATE = '29991231'.

FORM MAKE_MONEY_MARKET_DATA .

  LOOP AT GT_HAPO.

    CLEAR : GT_MAIN, GT_BP, GT_BD001, G_EXTBUSTRANSID, G_BUSTRANSID, G_BASE.

    GT_MAIN-BUKRS    = GT_HAPO-BUKRS.        "회사코드
    GT_MAIN-POSTDATE = GT_HAPO-DBUCHUNG.     "원천징수일(전기일)

    IF GT_HAPO-SGSART >= '51A' AND GT_HAPO-SGSART <= '52Z'.
      GT_MAIN-DESC = '예금이자'.
    ELSEIF GT_HAPO-SGSART(2) = '53'.
      GT_MAIN-DESC = '어음매입시'.
    ELSE.
      GT_MAIN-DESC = '수입이자 수령시'.
    ENDIF.

    READ TABLE GT_BP  WITH KEY PARTNER  = GT_HAPO-KONTRH
                       BINARY SEARCH.
    GT_MAIN-CUST   = GT_BP-NAME1.
    GT_MAIN-STCD2  = GT_BP-SORT2.

    READ TABLE GT_BD001 WITH KEY PARTNER = GT_HAPO-KONTRH
                         BINARY SEARCH.
    GT_MAIN-KUNNR  = BD001-KUNNR.

*   전표번호 조회
    CONCATENATE GT_HAPO-BUKRS  GT_HAPO-RFHA   GT_HAPO-RFHAZU
                GT_HAPO-DCRDAT GT_HAPO-TCRTIM GT_HAPO-RFHAZB
                INTO G_EXTBUSTRANSID.

    SELECT SINGLE BUSTRANSID INTO G_BUSTRANSID
     FROM TRDT_TRDBUSTRANS
    WHERE EXTBUSTRANSID = G_EXTBUSTRANSID.

    IF SY-SUBRC = 0.
      SELECT SINGLE BELNR INTO GT_MAIN-BELNR
        FROM TRACV_ACCITEM
       WHERE BUSTRANSID = G_BUSTRANSID.
    ENDIF.

*-- 전기일이 아닌 증빙일로 처리 --------------
    SELECT SINGLE BLDAT INTO GT_MAIN-POSTDATE
      FROM BKPF
     WHERE BUKRS = GT_MAIN-BUKRS     "회사코드
       AND BELNR = GT_MAIN-BELNR     "전표번호
       AND BUDAT = GT_MAIN-POSTDATE  "전기일
       AND BLDAT IN S_BLDAT.         "증빙일

    CHECK SY-SUBRC = 0.

    IF GT_HAPO-WBASIS <> C_KRW.  "원화가 아닌경우 원화로 환산
      G_BASE = ( GT_HAPO-BBASIS * GT_HAPO-KHWKURS ) / 100.
    ELSE.
      G_BASE = GT_HAPO-BBASIS.         "이자금액
    ENDIF.

*-- 계산기간내에 세율이 바뀐 경우에는 일할계산하여 레코드 분리
    PERFORM APPEND_ITAB_FOR_SPLIT_TAX  USING GT_HAPO-DBERVON
                                             GT_HAPO-DBERBIS
                                             G_BASE
                                             GT_HAPO-PKOND
                                             GT_HAPO-BHWBETR.

  ENDLOOP.

ENDFORM. " MAKE_MONEY_MARKET_DATA

*&      Form  APPEND_ITAB_FOR_SPLIT_TAX
*&---------------------------------------------------------------------*
FORM APPEND_ITAB_FOR_SPLIT_TAX USING P_DBERVON " 계산시작일
                                         P_DBERBIS   " 계산종료일
                                         P_BASE      " 이자금액
                                         P_PKOND     " 세율
                                         P_BHWBETR.  " 법인세

  DATA: L_SUB_DAY       TYPE I,               "전체일수
        L_SUM_DAY       TYPE I,               "누적일수
        L_TOT_DAY       TYPE I,               "구간일수
        L_SUM_BASE_AMT  LIKE VTBFHAPO-BBASIS, "총이자금액
        L_SUM_TAX_AMT   LIKE VTBFHAPO-BZBETR. "총법인세

  CLEAR: L_TOT_DAY, L_SUB_DAY, L_SUM_DAY.

  L_SUM_BASE_AMT = P_BASE.              "총이자금액
  L_SUM_TAX_AMT  = P_BHWBETR.           "총법인세

  L_TOT_DAY = P_DBERBIS - P_DBERVON + 1.  "전체일수

  LOOP AT GT_AT21  WHERE ( DABGUELT >  P_DBERVON AND DABGUELT <= P_DBERBIS ) OR
                         ( DABGUELT <= P_DBERVON AND ENDDATE  >= P_DBERVON ).

    IF GT_AT21-ENDDATE < P_DBERBIS.

      IF GT_AT21-DABGUELT > P_DBERVON.
        L_SUB_DAY = GT_AT21-ENDDATE - GT_AT21-DABGUELT + 1.
      ELSEIF GT_AT21-DABGUELT <= P_DBERVON.
        L_SUB_DAY = GT_AT21-ENDDATE - P_DBERVON + 1.
      ENDIF.

    ELSEIF GT_AT21-ENDDATE >= P_DBERBIS.

      IF GT_AT21-DABGUELT > P_DBERVON.
        L_SUB_DAY = P_DBERBIS - GT_AT21-DABGUELT + 1.
      ELSEIF GT_AT21-DABGUELT <= P_DBERVON.
        L_SUB_DAY = P_DBERBIS - P_DBERVON + 1.
      ENDIF.

    ENDIF.

    IF L_TOT_DAY = L_SUB_DAY.            "세율이 변경이 안되었으면
      IF P_PKOND IS INITIAL.
        GT_MAIN-RATE   = GT_AT21-PABSATZ.
      ELSE.
        GT_MAIN-RATE   = P_PKOND.        "세율
      ENDIF.
      GT_MAIN-TAX_AMT  = P_BHWBETR.      "법인세
      GT_MAIN-BASE_AMT = P_BASE.         "이자금액
    ELSE.
      L_SUM_DAY        = L_SUM_DAY + L_SUB_DAY.

      GT_MAIN-TAX_AMT  = P_BASE * ( GT_AT21-PABSATZ / 100 ) *  "법인세
                                  ( L_SUB_DAY / L_TOT_DAY ).
      GT_MAIN-RATE     = GT_AT21-PABSATZ.      "세율
      GT_MAIN-BASE_AMT = P_BASE * ( L_SUB_DAY / L_TOT_DAY ).   "이자금액

      IF L_SUM_DAY = L_TOT_DAY.        "마지막 레코드이면
*------ 총이자금액,총법인세에서 계산상의 소숫점 차이 보존.
        GT_MAIN-BASE_AMT  = L_SUM_BASE_AMT.    "이자금액
        GT_MAIN-TAX_AMT   = L_SUM_TAX_AMT.     "법인세
      ELSE.
        L_SUM_BASE_AMT = L_SUM_BASE_AMT - GT_MAIN-BASE_AMT.
        L_SUM_TAX_AMT  = L_SUM_TAX_AMT  - GT_MAIN-TAX_AMT.
      ENDIF.
    ENDIF.

    APPEND GT_MAIN.

  ENDLOOP.

ENDFORM. " APPEND_ITAB_FOR_SPLIT_TAX

FORM MAKE_SECURITIES_DATA .

  DATA : LS_CLAS       LIKE TRST_CLASFLO.
  DATA : L_OS_GUID     LIKE TRST_CLASPOS-OS_GUID,
         L_SECURITY    TYPE TRACV_ACCITEM-SECURITY_ID,
         L_RDPBANK     TYPE TWD01-RDPBANK,
         L_TAX         LIKE VTBFHAPO-BZBETR. "법인세

  LOOP AT GT_ITEM.

    CLEAR : GT_MAIN, L_OS_GUID, LS_CLAS, L_RDPBANK, GT_BP, GT_BD001, L_TAX, G_BASE.

    SELECT SINGLE OS_GUID INTO L_OS_GUID
      FROM TRST_CLASPOS
     WHERE COMPANY_CODE      = GT_ITEM-COMPANY_CODE
       AND SECURITY_ACCOUNT  = GT_ITEM-SECURITY_ACCOUNT
       AND SECURITY_ID       = GT_ITEM-SECURITY_ID.

    SELECT SINGLE * INTO LS_CLAS
      FROM TRST_CLASFLO
     WHERE CLASPOS_GUID    = L_OS_GUID
       AND POSITION_DATE   = GT_ITEM-ACPOSTINGDATE
       AND DIS_FLOWTYPE   IN ('SE8700','SE8701','SAM5000').

*-- 이자관련 갱신유형에 의해 DESC 구분
    IF LS_CLAS-DIS_FLOWTYPE  = 'SAM5000' OR
       GT_ITEM-PRODUCT_TYPE  = '04A'  OR GT_ITEM-PRODUCT_TYPE  = '04B'  OR
       GT_ITEM-PRODUCT_TYPE  = '04C'  OR GT_ITEM-PRODUCT_TYPE  = '04D'  OR
       GT_ITEM-PRODUCT_TYPE  = '04E'  OR GT_ITEM-PRODUCT_TYPE  = '04F'  OR
       GT_ITEM-PRODUCT_TYPE  = '04G'.
      GT_MAIN-DESC = '수입이자 수령시'.
    ELSE.
      GT_MAIN-DESC = '채권매입/매각시'.
    ENDIF.

*   유가증권 계정마스터 발췌
    SELECT SINGLE RDPBANK INTO L_RDPBANK
      FROM TWD01
     WHERE BUKRS  = GT_ITEM-COMPANY_CODE
       AND RLDEPO = GT_ITEM-SECURITY_ACCOUNT.

    READ TABLE GT_BP  WITH KEY PARTNER  = L_RDPBANK
                       BINARY SEARCH.
    GT_MAIN-CUST   = GT_BP-NAME1.
    GT_MAIN-STCD2  = GT_BP-SORT2.

    READ TABLE GT_BD001 WITH KEY PARTNER = L_RDPBANK
                         BINARY SEARCH.
    GT_MAIN-KUNNR  = BD001-KUNNR.

    GT_MAIN-BUKRS    = GT_ITEM-COMPANY_CODE.      "회사코드
    GT_MAIN-POSTDATE = GT_ITEM-ACPOSTINGDATE.     "원천징수일(전기일)
    GT_MAIN-BELNR    = GT_ITEM-BELNR.             "전표번호

*-- 전기일이 아닌 증빙일로 처리 --------------
    SELECT SINGLE BLDAT INTO GT_MAIN-POSTDATE
      FROM BKPF
     WHERE BUKRS = GT_MAIN-BUKRS     "회사코드
       AND BELNR = GT_MAIN-BELNR     "전표번호
       AND BUDAT = GT_MAIN-POSTDATE  "전기일
       AND BLDAT IN S_BLDAT.         "증빙일

    CHECK SY-SUBRC = 0.

*-- 계산기간내에 세율이 바뀐 경우에는 일할계산하여 레코드 분리
    G_BASE      = LS_CLAS-LOCAL_AMT.        "이자금액
    L_TAX       = GT_ITEM-LOCAL_AMT.        "법인세

    SELECT * FROM AT21
     WHERE SABVERF   = C_SABVERF
       AND RANTYP    = C_RANTYP
       AND DABGUELT =< LS_CLAS-DBERBIS.
      LS_CLAS-PKOND = AT21-PABSATZ.           "이자율 CLEAR
    ENDSELECT.

*-- 계산기간내에 세율이 바뀐 경우에는 일할계산하여 레코드 분리
    PERFORM APPEND_ITAB_FOR_SPLIT_TAX  USING LS_CLAS-DBERVON
                                             LS_CLAS-DBERBIS
                                             G_BASE
                                             LS_CLAS-PKOND
                                             L_TAX.

  ENDLOOP.

ENDFORM. " MAKE_SECURITIES_DATA

FORM MAKE_FI_DATA .

  DATA : LS_BSEG    LIKE BSEG OCCURS 0 WITH HEADER LINE.

  LOOP AT GT_BSIS.

    CLEAR : GT_MAIN, LS_BSEG, G_BASE, GT_KNA1, GT_LFA1.

    SELECT SINGLE * INTO LS_BSEG
     FROM BSEG
    WHERE BUKRS = BSIS-BUKRS          "회사코드
      AND BELNR = BSIS-BELNR          "전표번호
      AND GJAHR = BSIS-GJAHR          "회계연도
      AND BUZEI = BSIS-BUZEI.         "개별항목

    GT_MAIN-BUKRS       = LS_BSEG-BUKRS.          "회사코드
    GT_MAIN-DESC        = '수입이자 수령'.     "내역
    REPLACE ALL OCCURRENCES OF '-' IN LS_BSEG-XREF1 WITH ' '.
    GT_MAIN-STCD2       = LS_BSEG-XREF1.          "사업자등록번호

*   원천징수의무자 발췌
    READ TABLE GT_KNA1  WITH KEY STCD2  = GT_MAIN-STCD2
                         BINARY SEARCH.
    IF SY-SUBRC = 0.
      GT_MAIN-CUST    = GT_KNA1-NAME1.    "원천징수의무자
    ELSE.
      READ TABLE GT_LFA1  WITH KEY STCD2  = GT_MAIN-STCD2
                           BINARY SEARCH.
      GT_MAIN-CUST    = GT_KNA1-NAME1.    "원천징수의무자
    ENDIF.

    GT_MAIN-BELNR       = LS_BSEG-BELNR.          "전표번호
    GT_MAIN-POSTDATE    = GT_BSIS-BLDAT.          "원천징수일
    G_BASE              = LS_BSEG-XREF3.          "이자금액
    GT_MAIN-BASE_AMT    = G_BASE / 100.           "이자금액
    IF LS_BSEG-XREF2 = '' AND  G_BASE > 0.
      GT_MAIN-RATE = ( ( LS_BSEG-DMBTR * 100 ) / G_BASE ) * 100.
    ELSE.
      GT_MAIN-RATE      = LS_BSEG-XREF2.          "세율
    ENDIF.
    GT_MAIN-TAX_AMT     = LS_BSEG-DMBTR.          "법인세

    APPEND GT_MAIN.

  ENDLOOP.

ENDFORM. " MAKE_FI_DATA

FORM DOUBLE_CLICK USING P_ROW
                            P_COLUMN.

  DATA: LT_ROWID        TYPE LVC_T_ROW,
        LS_ROWID        TYPE LINE OF LVC_T_ROW.

  DATA : L_FUNCTIONCLASS TYPE  TB_FUNCTIONCLASS.

  CHECK P_ROW+(1) IS INITIAL.

  READ TABLE GT_MAIN INDEX P_ROW.

  CASE P_COLUMN.

    WHEN 'BELNR'.

      CHECK GT_MAIN-BELNR IS NOT INITIAL.

      SET PARAMETER ID 'BLN' FIELD GT_MAIN-BELNR.
      SET PARAMETER ID 'BUK' FIELD GT_MAIN-BUKRS.
      SET PARAMETER ID 'GJR' FIELD GT_MAIN-POSTDATE(4).

      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM. " DOUBLE_CLICK
