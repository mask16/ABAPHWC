FORM conversion_datum .

  DATA : l_datum(10).

  LOOP AT gt_t056p.

    CLEAR : gt_t056p-datum, l_datum.

    CALL FUNCTION 'CONVERSION_EXIT_INVDT_OUTPUT'
      EXPORTING
        input  = gt_t056p-datab
      IMPORTING
        output = l_datum.

    REPLACE ALL OCCURRENCES OF '.' IN l_datum WITH ''.
    gt_t056p-datum  = l_datum.

    MODIFY gt_t056p.

  ENDLOOP.

  DELETE gt_t056p   WHERE datum > p_datum.

  SORT gt_t056p  BY referenz datum DESCENDING.

ENDFORM. " CONVERSION_DATUM

FORM calculation_interest .

  CHECK gt_main-referenz IS NOT INITIAL.

  CLEAR gt_t056p.

  LOOP AT gt_t056p  WHERE referenz  = gt_main-referenz
                      AND datum    <= p_datum.

  IF gt_main-ssign = '-'.

  ELSE.
    gt_main-zsoll  = gt_t056p-zsoll + gt_main-zsoll.
  ENDIF.

ENDFORM. " CALCULATION_INTEREST
