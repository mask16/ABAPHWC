FORM MAKE_UPLOAD_FILE TABLES PT_ITAB STRUCTURE ALSMEX_TABLINE.

  DATA : BEGIN OF LT_CALID OCCURS 0,
           SCALID LIKE VTBKOND-SKALID,
         END OF LT_CALID.
  DATA : L_FIELD(20),
         L_INDEX(2)  TYPE N,
         L_NUMC(6)   TYPE N,
         L_DBLFZ     TYPE SY-DATUM,
         L_DELFZ     TYPE SY-DATUM,
         L_BZBETR    LIKE VTBFHAPO-BZBETR.

  LOOP AT PT_ITAB.

    IF PT_ITAB-COL < 5.

      L_INDEX  = PT_ITAB-COL.
      ASSIGN COMPONENT L_INDEX OF STRUCTURE GT_LOAD  TO <FS_1>.
      <FS_1> = PT_ITAB-VALUE.

    ENDIF.

    AT END OF ROW.
      APPEND GT_LOAD.  CLEAR GT_LOAD.
    ENDAT.

  ENDLOOP.

  DELETE GT_LOAD INDEX 1.

  REFRESH GT_MAIN.

  GT_MAIN-SFHAZBA  = '1105'.
  GT_MAIN-BZBETR   = G_BZBETR.
  GT_MAIN-DBLFZ    = G_DBLFZ.
  GT_MAIN-DELFZ    = G_DBLFZ.
  APPEND GT_MAIN.

  LT_CALID-SCALID  = 'KR'.
  APPEND LT_CALID.

  LOOP AT GT_LOAD.

    CLEAR : L_DBLFZ, L_DELFZ, GT_MAIN.

    REPLACE ALL OCCURRENCES OF '.' IN GT_LOAD-DBLFZ  WITH ' '.
    REPLACE ALL OCCURRENCES OF '.' IN GT_LOAD-DELFZ  WITH ' '.
    REPLACE ALL OCCURRENCES OF ',' IN GT_LOAD-BZBETR WITH ' '.
    REPLACE ALL OCCURRENCES OF ',' IN GT_LOAD-PKOND  WITH ' '.

    PERFORM CURRENCY_AMOUNT_IDOC_TO_SAP    USING GT_LOAD-BZBETR
                                                 C_KRW
                                        CHANGING GT_LOAD-BZBETR.

    PERFORM CURRENCY_AMOUNT_IDOC_TO_SAP    USING GT_LOAD-PKOND
                                                 C_KRW
                                        CHANGING GT_LOAD-PKOND.
    MODIFY GT_LOAD.

    MOVE-CORRESPONDING GT_LOAD TO GT_MAIN.

    GT_MAIN-SFHAZBA  = '1200'.
    MOVE GT_LOAD-PKOND TO GT_MAIN-BZBETR.
    APPEND GT_MAIN.

    MOVE GT_LOAD-BZBETR TO GT_MAIN-BZBETR.
    GT_MAIN-SFHAZBA  = '1110'.
    APPEND GT_MAIN.

  ENDLOOP.

  DELETE GT_MAIN   WHERE BZBETR = 0.

  L_BZBETR  = G_BZBETR.

  LOOP AT GT_MAIN.

    CALL FUNCTION 'FIMA_DAY_CONVERT_TO_WORKINGDAY'
      EXPORTING
        I_DATE  = GT_MAIN-DELFZ
        I_SWERK = P_SFWERK
      IMPORTING
        E_DATE  = GT_MAIN-DELFZ
      TABLES
        ICALID  = LT_CALID.

    GT_MAIN-WRBTR = L_BZBETR.

    MODIFY GT_MAIN.

    IF GT_MAIN-SFHAZBA = '1110'.
      L_BZBETR  = L_BZBETR - GT_MAIN-BZBETR.
    ENDIF.

  ENDLOOP.

ENDFORM. " MAKE_UPLOAD_FILE
