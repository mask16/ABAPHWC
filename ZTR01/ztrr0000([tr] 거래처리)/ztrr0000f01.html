FORM POSTING_RTN .

  DATA: L_SGSART TYPE VTBFHAZU-SGSART,
        L_KKURS1 TYPE VTBFHAZU-KKURS,
        L_KKURS2 TYPE VTBFHAZU-KKURS.

* range forward의 경우 최초 입력환율에서 조정 작업을 수행했는지
* 검증 후 변경되지 않은 것으로 판단되는 경우 WARNING 처리
  SELECT SINGLE KKURS INTO L_KKURS1
    FROM VTBFHAZU
   WHERE BUKRS  IN S_BUKRS
     AND RFHA   IN S_BUKRS
     AND RFHAZU  = '00001'.

  SELECT SINGLE SGSART KKURS INTO (L_SGSART, L_KKURS2)
    FROM VTBFHAZU
   WHERE BUKRS  IN S_BUKRS
     AND RFHA   IN S_RFHA
     AND SAKTIV  = '0'.

  IF L_KKURS1 = L_KKURS2 AND  VTBFHAZU-SGSART = C_SGSART.
*    MESSAGE I050.
  ENDIF.

  SUBMIT RFTBBB00 VIA SELECTION-SCREEN
                  WITH S_BUKRS IN S_BUKRS
                  WITH S_RFHA  IN S_RFHA
                  WITH P_BUDAT  = SY-DATUM
                  WITH P_BLDAT  = SY-DATUM
                  AND RETURN.

ENDFORM. " POSTING_RTN

FORM CHG_DISP_RTN .

  DATA : L_FUNCTIONCLASS TYPE  TB_FUNCTIONCLASS.

  IF P_CT = 'X'.

    L_FUNCTIONCLASS = 'CHANGE'.

  ELSE.

    L_FUNCTIONCLASS = 'DISPLAY'.

  ENDIF.

  CALL FUNCTION 'FTR_DISPATCH'
    EXPORTING
      I_BUKRS                = S_BUKRS-LOW
      I_RFHA                 = S_RFHA-LOW
      I_FUNCTIONCLASS        = L_FUNCTIONCLASS
    EXCEPTIONS
      NOT_FOUND              = 1
      FUNCTION_NOT_SUPPORTED = 2
      OTHERS                 = 3.

  IF SY-SUBRC <> 0.

    MESSAGE ID SY-MSGID TYPE 'S' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4
            DISPLAY LIKE SY-MSGTY.

  ENDIF.

ENDFORM. " CHG_DISP_RTN

FORM REVERSE_RTN .

  SUBMIT RTPM_SLD_ASYNC_POST VIA SELECTION-SCREEN
                             WITH PA_SEC  = 'X'
                             WITH PA_POS  = 'X'
                             WITH PA_DEA  = 'X'
                             WITH PA_DES  = 'X'
                             WITH SO_BUK IN S_BUKRS
                             WITH SO_DNR IN S_RFHA
                             WITH PA_TRL  = 'X'
                             WITH SO_OTC IN S_RFHA
                             WITH SO_DAT  = SY-DATUM
                             WITH PA_RER  = '01'
                             WITH PA_BUD  = SY-DATUM
                             WITH PA_BEL  = SY-DATUM
                             AND RETURN.

ENDFORM. " REVERSE_RTN
