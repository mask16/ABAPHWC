        L_CHECK(1).

  DESCRIBE TABLE GT_MAIN  LINES L_CNT.

  LOOP AT PR_DATA_CHANGED->MT_GOOD_CELLS INTO LS_GOOD.

    CHECK LS_GOOD-ROW_ID <= L_CNT.    "신규레코드는 제외

      WHEN 'HBKIDO'.                 "출금은행

          FROM T012
         WHERE BUKRS  = P_BUKRS
           AND HBKID  = L_HBKID.

        IF SY-SUBRC <> 0.
          CALL METHOD PR_DATA_CHANGED->ADD_PROTOCOL_ENTRY
            EXPORTING

              I_MSGNO     = '000'

              I_MSGV2     = L_HBKID

              I_ROW_ID    = LS_GOOD-ROW_ID.
        ENDIF.

      WHEN 'HKTIDO'.                 "출금계좌

        L_HKTID = LS_GOOD-VALUE.    "은행코드

        READ TABLE GT_T012T   WITH KEY BUKRS  = P_BUKRS

              I_MSGV1     = '존재하지 않는 계좌입니다.'

    PERFORM DELETE_FLAG.

    PERFORM MAKE_DATA.

  ENDIF.
