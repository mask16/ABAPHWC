*&      Module  SCREEN_DISPLAY  OUTPUT
*&---------------------------------------------------------------------*
MODULE SCREEN_DISPLAY OUTPUT.

  IMPORT G_TCODE FROM MEMORY ID 'TCODE'.
  IMPORT G_BUKRS FROM MEMORY ID '00402'.
  IMPORT G_RFHA  FROM MEMORY ID '00401'.

  FREE MEMORY ID  'TCODE'.

  CASE G_TCODE.

    WHEN 'TM_53' OR 'TM43'.

      LOOP AT SCREEN.
        IF SCREEN-NAME = 'GS_0040-LOANNO'.
          SCREEN-INPUT = 0.
          MODIFY SCREEN.
        ENDIF.
      ENDLOOP.

  ENDCASE.

  IF G_TCODE <> 'TM_51' AND G_TCODE <> 'TM41'.

    IF G_LOANNO IS NOT INITIAL.

      SELECT SINGLE * INTO CORRESPONDING FIELDS OF GS_0040
        FROM ZTRT0040
       WHERE BUKRS    = G_BUKRS
         AND LOANNO   = G_LOANNO.

    ELSE.

      SELECT SINGLE * INTO CORRESPONDING FIELDS OF GS_0040
        FROM ZTRT0040  AS A  INNER JOIN ZTRT0041  AS B
          ON A~BUKRS    = B~BUKRS
         AND A~LOANNO   = B~LOANNO
       WHERE A~BUKRS    = G_BUKRS
         AND B~RFHA     = G_RFHA.

    ENDIF.

    SELECT SINGLE BANKA INTO G_TEXT
      FROM BNKA
     WHERE BANKS   = 'KR'
       AND BANKL   = GS_0040-BANKL.

  ENDIF.

ENDMODULE. " SCREEN_DISPLAY OUTPUT
