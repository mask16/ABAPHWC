FORM MAKE_UPLOAD_FILE TABLES PT_ITAB STRUCTURE ALSMEX_TABLINE.

  DATA : L_FIELD(20),
         L_INDEX(2)  TYPE N.

  FIELD-SYMBOLS <FS_1>  TYPE ANY.

  LOOP AT PT_ITAB.

    IF PT_ITAB-COL < 11.

      L_INDEX  = PT_ITAB-COL.
      ASSIGN COMPONENT L_INDEX OF STRUCTURE GT_LOAD  TO <FS_1>.
      <FS_1> = PT_ITAB-VALUE.

    ENDIF.

    AT END OF ROW.
      APPEND GT_LOAD.  CLEAR GT_LOAD.
    ENDAT.

  ENDLOOP.

ENDFORM. " MAKE_UPLOAD_FILE

FORM TABLE_UPDATE .

  DATA : L_GDATU    LIKE TCURR-GDATU.

  REFRESH GT_TCURR.

  SELECT WAERS INTO TABLE GT_TCURC
    FROM TCURC.

  CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
    EXPORTING
      INPUT  = P_DATUM
    IMPORTING
      OUTPUT = L_GDATU.

  LOOP AT GT_LOAD.

    LOOP AT GT_TCURC.

      CLEAR GT_TCURR.

      FIND GT_TCURC-WAERS IN  GT_LOAD-WAERS   RESULTS GS_RESULT_TAB.

      CHECK SY-SUBRC = 0.

      GT_TCURR-KURST   = 'M'.
      GT_TCURR-FCURR   = GT_TCURC-WAERS.
      GT_TCURR-TCURR   = 'KRW'.
      GT_TCURR-GDATU   = L_GDATU.
      GT_TCURR-UKURS   = GT_LOAD-UKURS07.

      APPEND GT_TCURR.

    ENDLOOP.

  ENDLOOP.

ENDFORM. " TABLE_UPDATE
