FORM select_head .
  DATA : lv_tabix LIKE sy-tabix.

  SELECT a~rfha a~kontrh a~sfhaart
         b~xtext c~name_grp1 AS name1
    FROM vtbfha AS a LEFT OUTER JOIN at10t AS b
                       ON a~sgsart = b~sgsart
                      AND a~sfhaart = b~sfhaart
                      AND b~spras = sy-langu
                     LEFT OUTER JOIN but000 AS c
                       ON a~kontrh = c~partner
   INTO CORRESPONDING FIELDS OF TABLE gt_head
  WHERE a~bukrs EQ p_bukrs AND a~saktiv NE '3'
    AND a~sgsart IN ('55A', '55B', '56A', '59A').

  CLEAR : gt_head, lv_tabix.

  LOOP AT gt_head.
    lv_tabix = lv_tabix + 1.
    gt_head-cod1 = '1'.
    IF gt_head-sfhaart <> '500'.
      gt_head-cod2 = '1'.
    ELSE.
      gt_head-cod2 = '2'.
    ENDIF.
    gt_head-cod3 = gt_head-sfhaart+0(1).
    MODIFY gt_head INDEX lv_tabix.
  ENDLOOP.

  CLEAR : gt_head. SORT gt_head BY rfha.

ENDFORM. " SELECT_HEAD

FORM create_data .
  CLEAR : gt_head, gt_item.

  SORT gt_head BY rfha. SORT gt_item BY rfha.

  LOOP AT gt_head.
    READ TABLE gt_item WITH KEY rfha = gt_head-rfha BINARY SEARCH.
    IF sy-subrc = 0.
      LOOP AT gt_item FROM sy-tabix.
        IF gt_item-rfha = gt_head-rfha.
          MOVE-CORRESPONDING gt_head TO gt_main.
          gt_main-waers = c_krw.
          gt_main-bhwbetr01 = gt_item-bhwbetr01.
          gt_main-bhwbetr02 = gt_item-bhwbetr02.
          gt_main-bhwbetr03 = gt_item-bhwbetr03.
          gt_main-bhwbetr04 = gt_item-bhwbetr04.
          COLLECT gt_main. CLEAR gt_main.
        ELSE.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

  CLEAR : gt_main. SORT gt_main BY cod1 cod2 cod3.
