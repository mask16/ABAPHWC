FORM GET_DATE_RATE TABLES PT_HAPO STRUCTURE VW_VTBFHA
                    USING    PS_RFHA     TYPE VTBFHA
                             P_DAY
                             P_DEBEG
                             P_DENDF
                             P_PKOND
                             P_PERIOD.

  DATA : L_YEAR    TYPE I.

  CASE PS_RFHA-RANTYP.
    WHEN '5'.       " 머니마켓
      SELECT SINGLE DBLFZ DELFZ INTO (P_DEBEG, P_DENDF)
        FROM VTBFHAZU
       WHERE BUKRS    = P_BUKRS
         AND RFHA     = PS_RFHA-RFHA
         AND SAKTIV   = '0'.

      SELECT PKOND INTO P_PKOND
        FROM VTBFINKO
       WHERE BUKRS     = P_BUKRS
         AND RFHA      = PS_RFHA-RFHA
         AND DGUEL_KP <= GT_POS-PDATE
         AND SKOART   IN ('1200', '1150', '1201').
      ENDSELECT.

    WHEN '2'.       " 유가증권
      SELECT SINGLE A~DEBEG A~DENDF INTO (P_DEBEG, P_DENDF)
        FROM VWPANLE AS A INNER JOIN VWPANLA AS B
          ON A~RANL        = B~RANL
       WHERE A~RANL        = PS_RFHA-RANL
         AND ( B~SROLEXT   < '200' OR B~SROLEXT  >= '300' ).

      IF SY-SUBRC = 0.
        P_DAY  = P_DENDF - P_DEBEG + 1.
        IF P_DAY > '700'.
          CLEAR P_DENDF.
        ENDIF.
      ENDIF.

      REFRESH PT_HAPO.
      SELECT * INTO CORRESPONDING FIELDS OF TABLE PT_HAPO
        FROM VW_VTBFHA
       WHERE BUKRS            = P_BUKRS
         AND DZTERM          <= GT_POS-PDATE
         AND SGSART_VTBFHA   IN S_GSART
         AND KONTRH          IN S_KONTRH
         AND SFHAART_VTBFHA   = '200'
         AND RANL_VTBFHA      = PS_RFHA-RANL
         AND RLDEPO_VTBFHA    = PS_RFHA-RLDEPO
         AND SFHAZBA          = '0870'
         AND SBEWEBE          < '3'.
      IF SY-SUBRC = 0.
        SORT PT_HAPO  BY DZTERM  DESCENDING.
        READ TABLE PT_HAPO  INDEX 1.
        IF PT_HAPO-BBASIS <> 0 AND PT_HAPO-ATAGE <> 0.
          P_PKOND = 100 * PT_HAPO-BZBETR * PT_HAPO-ABASTAGE /
                  ( PT_HAPO-BBASIS * PT_HAPO-ATAGE ).
        ENDIF.
      ELSE.
        SELECT SINGLE PKOND INTO P_PKOND
          FROM VZZKOPO
         WHERE RKEY1    = PS_RFHA-RANL
           AND SKOART  IN ('0100', '0101').
      ENDIF.
  ENDCASE.

  IF P_DENDF IS INITIAL.
    P_PERIOD = 0.
  ELSE.
    CALL FUNCTION 'HR_AUPBS_MONTH_DAY'
      EXPORTING
        BEG_DA   = P_DEBEG
        END_DA   = P_DENDF
      IMPORTING
        NO_MONTH = P_PERIOD
        NO_YEAR  = L_YEAR.
  ENDIF.

  IF L_YEAR <> 0.
    P_PERIOD   = P_PERIOD + L_YEAR * 12.
  ENDIF.

  IF P_DEBEG > GT_POS-PDATE.
    P_DAY = P_DENDF - P_DEBEG + 1.
  ELSE.
    P_DAY = P_DENDF - GT_POS-PDATE + 1.
  ENDIF.
  IF P_DAY < 31.
    P_PERIOD  = 0.
  ENDIF.

  IF P_DENDF IS INITIAL.
    P_DAY = 0.
  ENDIF.

ENDFORM. " GET_DATE_RATE
