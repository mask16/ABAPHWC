*&      Form  MAKE_F4_FILE_DIR
*&---------------------------------------------------------------------*
FORM MAKE_F4_FILE_DIR USING P_FNAME.

  DATA: LINE_COUNT      TYPE   I,
        USER_ACTION     TYPE   I,
        RTN_CODE        LIKE   SY-SUBRC,
        LT_FILETABLE    TYPE   FILETABLE,
        LS_FILETABLE    TYPE   FILE_TABLE.
*        l_title         TYPE   string.

  REFRESH : LT_FILETABLE.

  CALL METHOD CL_GUI_FRONTEND_SERVICES=>FILE_OPEN_DIALOG
    EXPORTING
      WINDOW_TITLE      = 'Upload'
      MULTISELECTION    = ' '
      DEFAULT_EXTENSION = 'XLS'  "'TXT'
      DEFAULT_FILENAME  = ' '
      FILE_FILTER       = '*.XLS'  "'*.TXT'
    CHANGING
      FILE_TABLE        = LT_FILETABLE
      RC                = RTN_CODE
      USER_ACTION       = USER_ACTION.

  CALL METHOD CL_GUI_CFW=>FLUSH.

  CASE  USER_ACTION.
    WHEN  CL_GUI_FRONTEND_SERVICES=>ACTION_OK.
      READ   TABLE LT_FILETABLE INTO LS_FILETABLE INDEX 1.
      CHECK  SY-SUBRC IS INITIAL.
      MOVE:  LS_FILETABLE-FILENAME      TO      P_FNAME.
    WHEN  CL_GUI_FRONTEND_SERVICES=>ACTION_CANCEL.
  ENDCASE.

ENDFORM. " MAKE_F4_FILE_DIR

FORM EXCEL_UPLOAD .

  DATA: LT_ITAB     LIKE ALSMEX_TABLINE OCCURS 0 WITH HEADER LINE,
        L_START_COL TYPE I VALUE '1',
        L_START_ROW TYPE I VALUE '1',
        L_END_COL   TYPE I VALUE '256',
        L_END_ROW   TYPE I VALUE '65536'.

*  CALL FUNCTION 'KCD_EXCEL_OLE_TO_INT_CONVERT'
  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = P_FNAME
      I_BEGIN_COL             = L_START_COL
      I_BEGIN_ROW             = L_START_ROW
      I_END_COL               = L_END_COL
      I_END_ROW               = L_END_ROW
    TABLES
      INTERN                  = LT_ITAB
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2.

  IF SY-SUBRC NE 0.
    MESSAGE I516(T2) WITH P_FNAME.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CHECK NOT LT_ITAB[] IS INITIAL.

  SORT LT_ITAB BY ROW COL.

  PERFORM MAKE_UPLOAD_FILE  TABLES LT_ITAB.

ENDFORM. " EXCEL_UPLOAD

FORM MAKE_UPLOAD_FILE TABLES PT_ITAB STRUCTURE ALSMEX_TABLINE.

  DATA : L_FIELD(20),
         L_INDEX(2)  TYPE N.

  FIELD-SYMBOLS <FS_1>  TYPE ANY.

  _CLEAR GT_BP.

  LOOP AT PT_ITAB.

    IF PT_ITAB-COL < 22.

      L_INDEX  = PT_ITAB-COL.
      ASSIGN COMPONENT L_INDEX OF STRUCTURE GT_BP  TO <FS_1>.
      <FS_1> = PT_ITAB-VALUE.

    ENDIF.

    AT END OF ROW.
      APPEND GT_BP.  CLEAR GT_BP.
    ENDAT.

  ENDLOOP.

ENDFORM. " MAKE_UPLOAD_FILE
