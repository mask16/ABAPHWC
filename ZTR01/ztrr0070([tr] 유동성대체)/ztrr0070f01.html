FORM make_money_market .

  LOOP AT gt_loan  WHERE sgsart  IN r_sgsart1[]
                      OR sgsart  IN r_sgsart2[].

    _clear : gt_vtpo.
    CLEAR : g_amt1, g_amt2, gs_data, gt_post, gt_main1, gt_0011, gt_tzpa, gt_bp, gt_skat.

    READ TABLE gt_0011   WITH KEY rfha   = gt_loan-rfha.
    IF ( p_rb1 = 'X' AND sy-subrc = 0 ) OR ( p_rb2 = 'X' AND sy-subrc <> 0 ).
      CONTINUE.
    ENDIF.

*   거래흐름
    PERFORM select_vtbfhapo.

*   기준일 이후 잔액 계산
    CLEAR : g_amt1, g_amt2.
    LOOP AT gt_vtpo WHERE dzterm <= g_edate
                      AND dzterm >  p_budat
                      AND ( ( sfhazba >= c_flow3 AND sfhazba <= c_flow4 ) OR     "원금감소
                            ( sfhazba >= c_flow5 AND sfhazba <= c_flow6 ) OR     "최종상환
                              sfhazba = c_flow7 ).     "할부상환
      g_amt1 = g_amt1 + gt_vtpo-bzbetr.
    ENDLOOP.

*   기준일까지 잔액계산
    LOOP AT gt_vtpo WHERE dzterm <= p_budat
                      AND ( ( sfhazba >= c_flow1 AND sfhazba <= c_flow2 ) OR     "차입/원금증가
                            ( sfhazba >= c_flow3 AND sfhazba <= c_flow4 ) OR     "원금감소
                              sfhazba = c_flow7 ).     "할부상환
      g_amt2 = g_amt2 + gt_vtpo-bzbetr.
    ENDLOOP.

*   원금보다 상환금액이 많은 경우 CHECK 부분. 주로 정기적금의 경우
    IF g_amt1 > g_amt2.
      g_amt1 = g_amt2.
    ENDIF.

    CHECK g_amt1 > 0.

*   계정/대체계정
    PERFORM get_account1.

    CHECK gs_data-zielk <> ' '.

*   총원금 및 상환 후 잔액 계산
    LOOP AT gt_vtpo WHERE ( sfhazba = c_flow0 OR sfhazba = c_flow1 )
                      AND dzterm <=  p_budat.
      ADD gt_vtpo-bzbetr TO gs_data-tamount.
    ENDLOOP.

*   현재 잔고
    PERFORM get_current_position.

*-- 대여와 차입에 대한 구분을 하기 위해 대여의 경우 * -1.
*    IF gt_loan-sgsart IN r_sgsart2[].
*      g_amt1    = - g_amt1.
*      gs_data-tamount = - gs_data-tamount.
*      gs_data-lamount = - gs_data-lamount.
**      GS_DATA-BAMOUNT = - GS_DATA-BAMOUNT.
*    ENDIF.

    gs_data-lamount = gs_data-bamount - g_amt1.

*   Make internal table
    PERFORM make_main_data_mm.

    COLLECT gt_post.
    APPEND gt_main1.

  ENDLOOP.

ENDFORM. " MAKE_MONEY_MARKET

*&      Form  SELECT_VTBFHAPO

FORM select_vtbfhapo .

  CASE 'X'.

    WHEN p_chk1 OR p_chk3.   " 차입금/예적금

      SELECT * INTO TABLE gt_vtpo
        FROM vtbfhapo
       WHERE bukrs      = gt_loan-bukrs
         AND rfha       = gt_loan-rfha
         AND rfhazu     = gt_loan-rfhazu
         AND sbewebe   IN ('0', '1', '2')
         AND wzbetr    IN s_snwhr
         AND sfhazba   IN r_flow1
         AND sstornobwg = ' '.

    WHEN p_chk4.             " 사채할인발행차금

      SELECT * INTO TABLE gt_vtpo
        FROM vtbfhapo
       WHERE bukrs      = gt_loan-bukrs
         AND rfha       = gt_loan-rfha
         AND rfhazu     = gt_loan-rfhazu
         AND sbewebe   IN ('0', '1', '2')
         AND wzbetr    IN s_snwhr
         AND dberbis   >= s_kdate-low            " 기간내 존재
         AND dbervon   <= s_kdate-high
         AND sfhazba   IN r_flow2
         AND sstornobwg = ' '.

  ENDCASE.

  READ TABLE gt_vtpo INDEX 1.

  CONCATENATE gt_vtpo-bukrs gt_vtpo-rfha gt_vtpo-rfhazu gt_vtpo-dcrdat
              gt_vtpo-tcrtim gt_vtpo-rfhazb

FORM make_main_data_mm .

  DATA : l_hkont   LIKE bsis-hkont.

  CLEAR gt_0011.

  gt_post-bukrs = p_bukrs.
  gt_post-hkont = gs_data-hkont.
  gt_post-zielk = gs_data-zielk.
  gt_post-bnwhr = g_amt1.
  gt_post-snwhr = gt_vtpo-wzbetr.
  gt_post-kostl = gs_data-rportb.

  g_budat  = p_budat.
  READ TABLE gt_0011  WITH KEY rfha  = gt_loan-rfha
                               budat = g_budat
                      BINARY SEARCH.
  gt_post-belnr = gt_0011-belnr.

  g_budat  = p_budat + 1.
  READ TABLE gt_0011  WITH KEY rfha  = gt_loan-rfha
                               budat = g_budat
                      BINARY SEARCH.
  gt_post-belnr2 = gt_0011-belnr.

  MOVE-CORRESPONDING gt_post TO gt_main1.

  MOVE : gt_loan-rfha      TO gt_main1-rfha,
         g_amt1            TO gt_main1-bnwhr,
         gt_loan-sgsart    TO gt_main1-gsart,
         gt_loan-dblfz     TO gt_main1-dblfz,
         gt_loan-delfz     TO gt_main1-delfz,
         gs_data-rportb    TO gt_main1-kostl,
         gs_data-tamount   TO gt_main1-tamount,
         gs_data-bamount   TO gt_main1-bamount,
         gs_data-lamount   TO gt_main1-lamount.

* 전표
  SELECT SINGLE a~belnr a~document_date
    INTO (gt_main1-belnr, gt_main1-budat)
    FROM tracv_doc_item AS a INNER JOIN trdt_trdbustrans AS b
      ON a~bustransid    = b~bustransid
   WHERE b~extbustransid = gs_data-extbustransid.

* 금융상품명
  READ TABLE gt_tzpa   WITH KEY gsart   = gt_main1-gsart
                        BINARY SEARCH.
  gt_main1-merkm     = gt_tzpa-ltx.

* BP
  READ TABLE gt_bp     WITH KEY partner = gt_vtpo-rpzahl
                        BINARY SEARCH.
  gt_main1-bpname   = gt_bp-name1.

* 계정
  READ TABLE gt_skat  WITH KEY saknr  = gt_main1-hkont
                       BINARY SEARCH.
  gt_main1-htext    = gt_skat-txt20.

  PERFORM conversion_exit_alpha_output    USING gt_main1-hkont
                                       CHANGING l_hkont.

  CONCATENATE l_hkont gt_main1-htext INTO gt_main1-htext
              SEPARATED BY space.

ENDFORM. " MAKE_MAIN_DATA_MM

FORM make_security .

  DATA : lt_ident LIKE dift_pos_ident OCCURS 0 WITH HEADER LINE.

  LOOP AT gt_scrt.

    CLEAR : gs_data.
    _clear : lt_ident.

*   계정/대체계정
    PERFORM get_account2.

    CHECK gs_data-zielk <> ' '.

    SELECT * INTO TABLE lt_ident
      FROM dift_pos_ident
     WHERE context      = c_trl
       AND company_code = p_bukrs
       AND security_id  = gt_scrt-ranl.

    LOOP AT lt_ident.

      CLEAR : g_amt1, g_amt2, gt_post, gt_main1, gt_tzpa, gt_bp, gt_skat,
              gs_data-rfha, gs_data-tamount, gs_data-lamount, gs_data-bamount.

*     거래번호/금액
      PERFORM get_rfha USING lt_ident-security_account
                             lt_ident-security_id
                             lt_ident-identified_oid.

      CHECK NOT gs_data-rfha IS INITIAL AND gs_data-rfha IN s_rfha AND g_amt1 <> 0.

      READ TABLE gt_0011   WITH KEY rfha   = gs_data-rfha
                           BINARY SEARCH.
      IF ( p_rb1 = 'X' AND sy-subrc = 0 ) OR ( p_rb2 = 'X' AND sy-subrc <> 0 ).
        CONTINUE.
      ENDIF.

*     Make internal table
      PERFORM make_main_data_secu  USING lt_ident-security_account
                                         lt_ident-security_id.

*     현재 잔고
*      PERFORM GET_CURRENT_POSITION_SEC.

      COLLECT gt_post.
      APPEND gt_main1.

    ENDLOOP.

  ENDLOOP.

ENDFORM. " MAKE_SECURITY
