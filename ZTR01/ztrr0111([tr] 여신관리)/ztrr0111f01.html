FORM line_delete .

  DATA : lt_row_table   TYPE lvc_t_row    WITH HEADER LINE.

  CLEAR : lt_row_table, lt_row_table[], g_check.

  CALL METHOD grf_grid->get_selected_rows
    IMPORTING
      et_index_rows = lt_row_table[].

  IF lt_row_table[] IS INITIAL.
    MESSAGE e018.
  ENDIF.

  SORT lt_row_table BY index DESCENDING.

  LOOP AT lt_row_table.

    READ TABLE gt_main INDEX lt_row_table-index.
    CHECK sy-subrc = 0.

      SELECT SINGLE mandt INTO g_mandt
        FROM ztrt0041

        APPEND gt_dele.
        DELETE gt_main INDEX lt_row_table-index.
      ENDIF.

    ENDIF.

FORM save_data .

  DATA : lt_save       LIKE ztrt0040  OCCURS 0 WITH HEADER LINE.
  DATA : l_seqnr(6)    TYPE n,
         l_loanno1(10),
         l_loanno2     LIKE ztrt0040-loanno.

  PERFORM change_data_update   USING grf_grid.

  PERFORM popup_to_confirm_step    USING text-h01
                                         text-s03
                                         text-s04
                                CHANGING g_answer.

  CHECK g_answer = 'J'.

* 필수필드 입력여부 check
  PERFORM field_check.

*  SORT GT_MAIN  BY PERIO BANKL DATAB.

  CONCATENATE sy-datum(4) '%' INTO l_loanno1.

  SELECT MAX( loanno ) INTO l_loanno2
    FROM ztrt0040
   WHERE bukrs       = p_bukrs
     AND loanno   LIKE l_loanno1.

  IF sy-subrc = 0.
    l_seqnr   = l_loanno2+4(6).
  ENDIF.

  LOOP AT gt_main.

*   차입번호
    IF gt_main-loanno IS INITIAL.
      ADD 1 TO l_seqnr.
      CONCATENATE sy-datum(4)  l_seqnr INTO gt_main-loanno.
      gt_main-flag = 'X'.
    ENDIF.

    MOVE-CORRESPONDING gt_main TO lt_save.
    lt_save-upnam  = sy-uname.
    lt_save-updat  = sy-datum.

    APPEND lt_save.
    MODIFY gt_main.

  ENDLOOP.

  MODIFY ztrt0040 FROM TABLE lt_save.

  IF sy-subrc = 0.
    CLEAR g_save_ch.

    DELETE ztrt0040  FROM TABLE gt_dele.

    COMMIT WORK.

    PERFORM select_rtn.
    PERFORM make_data.

    CALL METHOD grf_grid->refresh_table_display.
    CALL METHOD cl_gui_cfw=>flush.

    MESSAGE s000 WITH '성공적으로 저장되었습니다.'.

  ELSE.
    LOOP AT gt_main  WHERE flag = 'X'.
      CLEAR gt_main-loanno.
      MODIFY gt_main.
    ENDLOOP.
    ROLLBACK WORK.
    MESSAGE e000 WITH '저장에 실패하였습니다.'.
  ENDIF.

ENDFORM. " SAVE_DATA
