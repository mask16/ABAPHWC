FORM MAKE_DATA1 .

  DATA : LT_INDE    LIKE VTBFHAPO  OCCURS 0 WITH HEADER LINE, " 투자원금증가, 감소
         LT_FLOW    LIKE VTBFHAPO  OCCURS 0 WITH HEADER LINE. " 이자자본화, 액면이자
*  DATA : L_TCURR like TCURR.
   DATA : L_bseg LIKE bseg.
  LOOP AT GT_HAPO.

    _CLEAR : LT_INDE, LT_FLOW.
    CLEAR : GT_MAIN, G_FLOW, G_EXTBUSTRANSID, G_BUSTRANSID, G_CHECK.

*   REVERSE 되지 않은 건만
    SELECT SINGLE SGSART KONTRH SANLF INTO (GT_MAIN-SGSART, GT_MAIN-PARTNER, GT_MAIN-SANLF)
      FROM VTBFHA
     WHERE BUKRS   = GT_HAPO-BUKRS
       AND RFHA    = GT_HAPO-RFHA
       AND SAKTIV <> '3'.

    CHECK SY-SUBRC = 0 AND GT_MAIN-SGSART IN R_SGSART[].

*   기본 데이터
    PERFORM MAKE_BASIC_DATA.

*   투자원금증가, 감소
    PERFORM SELECT_MAKE_DATA   TABLES LT_INDE
                                      LT_FLOW.

    CHECK G_CHECK IS INITIAL.

    LOOP AT LT_FLOW.

      MOVE : LT_FLOW-PKOND  TO GT_MAIN-PKOND,
             LT_FLOW-BBASIS TO GT_MAIN-BBASIS,
             LT_FLOW-BZBETR TO GT_MAIN-BZBETR1.

*     원금이 없는 경우 원금을 계상
      IF GT_MAIN-BBASIS IS INITIAL.
        LOOP AT LT_INDE WHERE DBUCHUNG < LT_FLOW-DBERBIS.
          IF LT_INDE-SFHAZBA = G_FLOW.
            ADD LT_INDE-BZBETR TO GT_MAIN-BBASIS.
          ELSE.
            SUBTRACT LT_INDE-BZBETR FROM GT_MAIN-BBASIS.
          ENDIF.
        ENDLOOP.
      ENDIF.

      GT_MAIN-TERMS = LT_FLOW-DBERBIS - LT_FLOW-DBERVON.
      GT_MAIN-TERMS = GT_MAIN-TERMS + 1.

      IF GT_MAIN-PKOND IS INITIAL AND LT_FLOW-SFHAZBA(2) <> '19'.
        IF LT_FLOW-SZBMETH = '3'.                         "ACT/365
          GT_MAIN-PKOND = GT_MAIN-BZBETR1 * 100 * 365 /
                      ( GT_MAIN-BBASIS * GT_MAIN-TERMS ).
        ELSE.
          GT_MAIN-PKOND = GT_MAIN-BZBETR1 * 100 * 360 /
                      ( GT_MAIN-BBASIS * GT_MAIN-TERMS ).
        ENDIF.
      ENDIF.
      GT_MAIN-BZBETR2 =  GT_HAPO-BHWBETR  .
      GT_MAIN-BZBETR3 =  GT_MAIN-BZBETR1 * GT_HAPO-KHWKURS / 100  .

      APPEND GT_MAIN.

    ENDLOOP.

  ENDLOOP.

ENDFORM. " MAKE_DATA1

FORM MAKE_BASIC_DATA .

  GT_MAIN-BUKRS     = GT_HAPO-BUKRS.
  GT_MAIN-RFHA      = GT_MAIN-RFHA1 = GT_HAPO-RFHA.
  GT_MAIN-DBUCHUNG  = GT_HAPO-DBUCHUNG.
  GT_MAIN-DBERVON   = GT_HAPO-DBERVON.
  GT_MAIN-DBERBIS   = GT_HAPO-DBERBIS.
  GT_MAIN-BZBETR    = GT_HAPO-BZBETR.
  GT_MAIN-WZBETR    = GT_HAPO-WZBETR.
  GT_MAIN-ATAGE     = GT_HAPO-ATAGE.

  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
    EXPORTING
      INPUT  = GT_MAIN-RFHA
    IMPORTING
      OUTPUT = GT_MAIN-RFHA.

* 금융상품명
  READ TABLE GT_TZPA   WITH KEY GSART   = GT_MAIN-SGSART
                        BINARY SEARCH.
  GT_MAIN-LTX       = GT_TZPA-LTX.
* BP
  READ TABLE GT_BP     WITH KEY PARTNER = GT_MAIN-PARTNER
                        BINARY SEARCH.
  GT_MAIN-NAME1     = GT_BP-NAME_GRP1.

* 시작일 만기일 추출
  SELECT SINGLE DZSTND DBLFZ DELFZ
    INTO (GT_MAIN-DZSTND, GT_MAIN-DBLFZ, GT_MAIN-DELFZ)
    FROM VTBFHAZU
   WHERE BUKRS    = GT_HAPO-BUKRS
     AND RFHA     = GT_HAPO-RFHA
     AND SAKTIV   = 0.
  IF GT_MAIN-DBLFZ = GT_MAIN-DELFZ.
    MOVE GT_MAIN-DZSTND TO GT_MAIN-DBLFZ.
  ENDIF.

* 전표번호 조회
  CONCATENATE GT_HAPO-BUKRS  GT_HAPO-RFHA   GT_HAPO-RFHAZU
              GT_HAPO-DCRDAT GT_HAPO-TCRTIM GT_HAPO-RFHAZB
              INTO G_EXTBUSTRANSID.

  SELECT SINGLE BUSTRANSID INTO G_BUSTRANSID
   FROM TRDT_TRDBUSTRANS
  WHERE EXTBUSTRANSID = G_EXTBUSTRANSID.

  IF SY-SUBRC = 0.
    SELECT SINGLE BELNR INTO GT_MAIN-BELNR
      FROM TRACV_ACCITEM
     WHERE BUSTRANSID = G_BUSTRANSID.
  ENDIF.

ENDFORM. " MAKE_BASIC_DATA

FORM SELECT_MAKE_DATA TABLES PT_INDE STRUCTURE VTBFHAPO
                                PT_FLOW STRUCTURE VTBFHAPO.

  _CLEAR : R_FLOW3, R_FLOW4.

  IF GT_MAIN-SGSART IN R_SGSART1[].      " 예적금
    _RANGES : R_FLOW3 'EQ'  '1100' '',
              R_FLOW3 'EQ'  '1110' ''.
    _RANGES : R_FLOW4 'EQ'  '1150' '',
              R_FLOW4 'EQ'  '1200' ''.
    G_FLOW  = '1100'.
  ELSEIF GT_MAIN-SGSART IN R_SGSART2[].  " 차입금
    _RANGES : R_FLOW3 'EQ'  '1105' '',
              R_FLOW3 'EQ'  '1110' ''.
    IF GT_HAPO-SFHAZBA IN R_FLOW1[].
      _RANGES : R_FLOW4 'EQ'  '1150' '',
                R_FLOW4 'EQ'  '1200' ''.
    ELSEIF GT_HAPO-SFHAZBA  = '2101' OR GT_HAPO-SFHAZBA = '2106'.
      _RANGES : R_FLOW4 'EQ'  '1911' ''.
    ELSE.
      _RANGES : R_FLOW4 'EQ'  '1912' ''.
    ENDIF.
    G_FLOW  = '1105'.
  ENDIF.

  SELECT * INTO TABLE PT_INDE
    FROM VTBFHAPO
   WHERE BUKRS    = GT_HAPO-BUKRS
     AND RFHA     = GT_HAPO-RFHA
     AND SFHAZBA IN R_FLOW3
     AND SBEWEBE  = '2'.

  IF SY-SUBRC <> 0.
    G_CHECK = 'X'.
  ENDIF.

  SELECT * INTO TABLE PT_FLOW
    FROM VTBFHAPO
   WHERE BUKRS    = GT_HAPO-BUKRS
     AND RFHA     = GT_HAPO-RFHA
     AND SFHAZBA IN R_FLOW4
     AND SBEWEBE <> '4'
     AND SSPRGRD <> '3'
     AND DBERVON <= GT_MAIN-DBUCHUNG
     AND DBERBIS >= GT_MAIN-DBUCHUNG.

ENDFORM. " SELECT_MAKE_DATA

FORM DOUBLE_CLICK USING P_ROW
                            P_COLUMN.

  DATA: LT_ROWID        TYPE LVC_T_ROW,
        LS_ROWID        TYPE LINE OF LVC_T_ROW.

  DATA : L_FUNCTIONCLASS TYPE  TB_FUNCTIONCLASS.

  CHECK P_ROW+(1) IS INITIAL.

  READ TABLE GT_MAIN INDEX P_ROW.

  CASE P_COLUMN.

    WHEN 'BELNR'.

      CHECK GT_MAIN-BELNR IS NOT INITIAL.

      SET PARAMETER ID 'BLN' FIELD GT_MAIN-BELNR.
      SET PARAMETER ID 'BUK' FIELD GT_MAIN-BUKRS.
      SET PARAMETER ID 'GJR' FIELD GT_MAIN-DBUCHUNG(4).

      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

    WHEN 'RFHA'.

      CHECK GT_MAIN-RFHA IS NOT INITIAL.

      L_FUNCTIONCLASS = 'DISPLAY'.

      CALL FUNCTION 'FTR_DISPATCH'
        EXPORTING
          I_BUKRS                = GT_MAIN-BUKRS
          I_RFHA                 = GT_MAIN-RFHA1
          I_FUNCTIONCLASS        = L_FUNCTIONCLASS
        EXCEPTIONS
          NOT_FOUND              = 1
          FUNCTION_NOT_SUPPORTED = 2
          OTHERS                 = 3.

      IF SY-SUBRC <> 0.

        MESSAGE ID SY-MSGID TYPE 'S' NUMBER SY-MSGNO
                WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4
                DISPLAY LIKE SY-MSGTY.

      ENDIF.

  ENDCASE.

ENDFORM. " DOUBLE_CLICK
