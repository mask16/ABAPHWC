FORM PARTIAL_PAYING_SCHEDULE .

  DATA: L_TCOUNT    TYPE I,
        L_COUNT     TYPE I,
        L_TPERIOD   TYPE I,
        L_DBLFZ     LIKE VTBFHA-DBLFZ,
        LS_DATE     TYPE GS_DATE,
        L_CHKCNT    TYPE I,
        L_DIRECTION TYPE SCAL-INDICATOR,
        LT_ICAL     LIKE VTBFCAL OCCURS 0 WITH HEADER LINE.

  CLEAR : G_AMOUNT, G_TAMOUNT, L_TCOUNT, L_COUNT, L_TPERIOD, LT_ICAL[].

  MOVE P_CALEND TO LT_ICAL-IDENT.
  APPEND LT_ICAL.

*& 공휴일인경우 날짜 이동방식
  CASE P_SFWERK.
    WHEN '0'.  L_DIRECTION = ''.
    WHEN '1'.  L_DIRECTION = '+'.
    WHEN '2'.  L_DIRECTION = '+'.
    WHEN '3'.  L_DIRECTION = '-'.
    WHEN '4'.  L_DIRECTION = '-'.
  ENDCASE.

  MOVE G_DBLFZ TO LS_DATE.

*& 지급 횟수를 계산하기 위한 부분
  CLEAR L_TCOUNT.
  MOVE LS_DATE   TO L_DBLFZ.

*& period = 전체일수 / (30 * 원금 납입 PERIOD)
  L_TPERIOD = ( G_DELFZ - L_DBLFZ ) / ( 30 * P_ARHYTM ).
*& 첫번째 PERIOD는 이미 입력되어 있으므로 -1을 해줌.
  L_TPERIOD = L_TPERIOD - 1.

  CLEAR L_COUNT.

  DO L_TPERIOD TIMES.
    LS_DATE-MM = LS_DATE-MM + P_ARHYTM.

    IF LS_DATE-MM > 12.
      LS_DATE-YYYY = LS_DATE-YYYY + 1.
      LS_DATE-MM   = LS_DATE-MM - 12.
    ENDIF.

    MOVE G_DBLFZ+6(2) TO LS_DATE-DD.
    PERFORM VALID_DATE_CHECK USING LS_DATE.

    L_COUNT = L_COUNT + 1.
    G_DATE  = LS_DATE.

    CALL FUNCTION 'DATE_CONVERT_TO_WORKINGDAY'
      EXPORTING
        DATE              = G_DATE
        DIRECTION         = L_DIRECTION
      IMPORTING
        WORKINGDAY        = G_DATE
      TABLES
        FACTORY_CALENDARS = LT_ICAL.

*&  만기일과 다음 지급 예정일이 26보다 작은 경우(공휴일 고려) ----
    L_CHKCNT = G_DELFZ - G_DATE - ( 30 * P_ARHYTM ).
    IF L_CHKCNT < 26.
      ADD 1 TO L_COUNT.
    ENDIF.

    IF L_TPERIOD > L_COUNT.
      PERFORM BDC_DATA_BODY.
    ELSE.
      PERFORM BDC_DATA_BODY2.
      EXIT.
    ENDIF.
  ENDDO.

ENDFORM. " PARTIAL_PAYING_SCHEDULE

FORM VALID_DATE_CHECK USING PS_DATE TYPE GS_DATE.

  DATA : L_RESTNUM   TYPE I,
         L_RESTCHK   TYPE C.

  CLEAR : L_RESTNUM, L_RESTCHK.

* 윤년 CHECK LOGIC
  L_RESTNUM = PS_DATE-YYYY MOD 4.

  IF L_RESTNUM = 0.
    L_RESTCHK = 'X'.
  ENDIF.

  IF L_RESTCHK = 'X'.
    L_RESTNUM = PS_DATE-YYYY MOD 400.
    IF L_RESTNUM = 0.
    ELSE.
      L_RESTNUM = PS_DATE-YYYY MOD 100.
      IF L_RESTNUM = 0.
        L_RESTCHK = ' '.
      ENDIF.
    ENDIF.
  ENDIF.

* 2월에 대한 처리
  IF L_RESTCHK = 'X' AND PS_DATE-MM = 02 AND PS_DATE-DD > 28.
    MOVE 29 TO PS_DATE-DD.
  ELSEIF L_RESTCHK = ' ' AND PS_DATE-MM = 02 AND PS_DATE-DD > 28.
    MOVE 28 TO PS_DATE-DD.
  ENDIF.

* 기타 30일까지만 있는 달에 대한 처리
  IF ( PS_DATE-MM = 04 OR PS_DATE-MM = 06 OR PS_DATE-MM = 09 OR
       PS_DATE-MM = 11 ) AND PS_DATE-DD > 30.
    MOVE 30 TO PS_DATE-DD.
  ENDIF.

ENDFORM. " VALID_DATE_CHECK
