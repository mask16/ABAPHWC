*----------------------------------------------------------------------*
FORM create_main .
  CLEAR : gt_head, gt_item.

  SORT gt_head BY hkont. SORT gt_item BY hkont.

  LOOP AT gt_head.
    READ TABLE gt_item WITH KEY hkont = gt_head-hkont BINARY SEARCH.
    IF sy-subrc = 0.
      LOOP AT gt_item FROM sy-tabix.
        IF gt_item-hkont = gt_head-hkont.
          MOVE-CORRESPONDING gt_head TO gt_main.
          gt_main-waers = c_krw.
          gt_main-dmbtr01 = gt_item-dmbtr01.
          gt_main-dmbtr02 = gt_item-dmbtr02.
          gt_main-dmbtr03 = gt_item-dmbtr03.
          gt_main-dmbtr04 = gt_item-dmbtr04.
          COLLECT gt_main. CLEAR gt_main.
        ELSE.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

  CLEAR : gt_main. SORT gt_main BY cod1 cod2.

ENDFORM. " CREATE_MAIN

*----------------------------------------------------------------------*
FORM select_fi .
  DATA : BEGIN OF lt_head OCCURS 0,
           rfha        LIKE vtbfha-rfha,
           cod1        TYPE char1,
           cod2        TYPE char1,
           cod3        TYPE char1,
           kontrh      LIKE vtbfha-kontrh,
           name1       LIKE but000-name_grp1,
           sfhaart     LIKE vtbfha-sfhaart,
           xtext       LIKE at10t-xtext,
         END OF lt_head.

  DATA : BEGIN OF lt_item OCCURS 0,
           rfha        LIKE vtbfhapo-rfha,
           bhwbetr01   LIKE vtbfhapo-bhwbetr,
           bhwbetr02   LIKE vtbfhapo-bhwbetr,
           bhwbetr03   LIKE vtbfhapo-bhwbetr,
           bhwbetr04   LIKE vtbfhapo-bhwbetr,
         END OF lt_item.

  DATA : BEGIN OF lt_item01 OCCURS 0,
           rfha        LIKE vtbfhapo-rfha,
           ssign       LIKE vtbfhapo-ssign,
           waers       LIKE t001-waers,
           bhwbetr     LIKE vtbfhapo-bhwbetr,
         END OF lt_item01.

  DATA : lt_item02 LIKE lt_item01 OCCURS 0 WITH HEADER LINE,
         lt_item03 LIKE lt_item01 OCCURS 0 WITH HEADER LINE,
         lt_item04 LIKE lt_item01 OCCURS 0 WITH HEADER LINE.

  DATA : lv_tabix LIKE sy-tabix.
  RANGES : lr_rfha FOR vtbfhapo-rfha.
*----------------------------------------------------------------------*
*  HEAD
*----------------------------------------------------------------------*
  SELECT a~rfha a~kontrh a~sfhaart
         b~xtext c~name_grp1 AS name1
    FROM vtbfha AS a LEFT OUTER JOIN at10t AS b
                       ON a~sgsart = b~sgsart
                      AND a~sfhaart = b~sfhaart
                      AND b~spras = sy-langu
                     LEFT OUTER JOIN but000 AS c
                       ON a~kontrh = c~partner
   INTO CORRESPONDING FIELDS OF TABLE lt_head
  WHERE a~bukrs EQ p_bukrs AND a~saktiv NE '3'
    AND a~sgsart IN ('51A', '51B', '52A', '52B').

  CLEAR : lt_head, lv_tabix.

  LOOP AT lt_head.
    lv_tabix = lv_tabix + 1.
    lt_head-cod1 = '2'.
    lt_head-cod2 = '1'.
    lt_head-cod3 = lt_head-sfhaart+0(1).
    MODIFY lt_head INDEX lv_tabix.
  ENDLOOP.
*----------------------------------------------------------------------*
*  ITEM
*----------------------------------------------------------------------*
  CLEAR : lt_head. SORT lt_head BY rfha.
  LOOP AT lt_head. _ranges : lr_rfha c_eq lt_head-rfha ''. ENDLOOP.
  DELETE ADJACENT DUPLICATES FROM lr_rfha.

*-- 입력받은 월
  SELECT rfha ssign bhwbetr
    FROM vtbfhapo
    INTO CORRESPONDING FIELDS OF TABLE lt_item01
   WHERE rfha IN lr_rfha AND sbewebe EQ '2'
     AND sberfima IN ('TTRA', 'TTAN', 'AA', 'SS', 'TTEN')
     AND dbuchung LE p_datum AND bukrs EQ p_bukrs.
*-- 입력받은 월 - 1
  SELECT rfha ssign bhwbetr
    FROM vtbfhapo
APPENDING CORRESPONDING FIELDS OF TABLE lt_item02
   WHERE rfha IN lr_rfha AND sbewebe EQ '2'
    AND sberfima IN ('TTRA', 'TTAN', 'AA', 'SS', 'TTEN')
    AND dbuchung LE gv_enday1 AND bukrs EQ p_bukrs.
*-- 입력받은 월 - 2
  SELECT rfha ssign bhwbetr
    FROM vtbfhapo
APPENDING CORRESPONDING FIELDS OF TABLE lt_item03
   WHERE rfha IN lr_rfha AND sbewebe EQ '2'
    AND sberfima IN ('TTRA', 'TTAN', 'AA', 'SS', 'TTEN')
    AND dbuchung LE gv_enday2 AND bukrs EQ p_bukrs.
*-- 입력받은 월 - 3
  SELECT rfha ssign bhwbetr
    FROM vtbfhapo
APPENDING CORRESPONDING FIELDS OF TABLE lt_item04
   WHERE rfha IN lr_rfha AND sbewebe EQ '2'
    AND sberfima IN ('TTRA', 'TTAN', 'AA', 'SS', 'TTEN')
    AND dbuchung LE gv_enday3 AND bukrs EQ p_bukrs.

  CLEAR : lt_item01, lt_item02, lt_item03, lt_item04.

  LOOP AT lt_item01.
    MOVE-CORRESPONDING lt_item01 TO lt_item.
    IF lt_item01-ssign = '+'.
      lt_item-bhwbetr01 = lt_item01-bhwbetr * -1.
    ELSE.
      lt_item-bhwbetr01 = lt_item01-bhwbetr.
    ENDIF.
    COLLECT lt_item. CLEAR lt_item.
  ENDLOOP.

  LOOP AT lt_item02.
    MOVE-CORRESPONDING lt_item02 TO lt_item.
    IF lt_item02-ssign = '+'.
      lt_item-bhwbetr02 = lt_item02-bhwbetr * -1.
    ELSE.
      lt_item-bhwbetr02 = lt_item02-bhwbetr.
    ENDIF.
    COLLECT lt_item. CLEAR lt_item.
  ENDLOOP.

  LOOP AT lt_item03.
    MOVE-CORRESPONDING lt_item03 TO lt_item.
    IF lt_item03-ssign = '+'.
      lt_item-bhwbetr03 = lt_item03-bhwbetr * -1.
    ELSE.
      lt_item-bhwbetr03 = lt_item03-bhwbetr.
    ENDIF.
    COLLECT lt_item. CLEAR lt_item.
  ENDLOOP.

  LOOP AT lt_item04.
    MOVE-CORRESPONDING lt_item04 TO lt_item.
    IF lt_item04-ssign = '+'.
      lt_item-bhwbetr04 = lt_item04-bhwbetr * -1.
    ELSE.
      lt_item-bhwbetr04 = lt_item04-bhwbetr.
    ENDIF.
    COLLECT lt_item. CLEAR lt_item.
  ENDLOOP.

  CLEAR lt_item. SORT lt_item BY rfha.
  CLEAR : lt_head. SORT lt_head BY rfha.

*----------------------------------------------------------------------*
*  합치기
*----------------------------------------------------------------------*
  CLEAR : gt_main.

  LOOP AT lt_head.
    READ TABLE lt_item WITH KEY rfha = lt_head-rfha BINARY SEARCH.
    IF sy-subrc = 0.

      CASE lt_head-sfhaart.
        WHEN '100'.
          LOOP AT lt_item FROM sy-tabix.
            IF lt_item-rfha = lt_head-rfha.
              gt_main-bcod = '998'. gt_main-banka = '별단&정기성'.
              gt_main-kcod = 'A'.   gt_main-ktext = '정기성예금'.
              gt_main-cod1 = '2'.   gt_main-cod2 = '2'.
              gt_main-waers = c_krw.
              gt_main-dmbtr01 = lt_item-bhwbetr01.
              gt_main-dmbtr02 = lt_item-bhwbetr02.
              gt_main-dmbtr03 = lt_item-bhwbetr03.
              gt_main-dmbtr04 = lt_item-bhwbetr04.
              COLLECT gt_main. CLEAR gt_main.
            ELSE.
              EXIT.
            ENDIF.
          ENDLOOP.
        WHEN '200'.
          LOOP AT lt_item FROM sy-tabix.
            IF lt_item-rfha = lt_head-rfha.
              gt_main-bcod = '998'. gt_main-banka = '별단&정기성'.
              gt_main-kcod = 'B'. gt_main-ktext = '정기성적금'.
              gt_main-cod1 = '2'. gt_main-cod2 = '3'.
              gt_main-waers = c_krw.
              gt_main-dmbtr01 = lt_item-bhwbetr01.
              gt_main-dmbtr02 = lt_item-bhwbetr02.
              gt_main-dmbtr03 = lt_item-bhwbetr03.
              gt_main-dmbtr04 = lt_item-bhwbetr04.
              COLLECT gt_main. CLEAR gt_main.
            ELSE.
              EXIT.
            ENDIF.
          ENDLOOP.
      ENDCASE.
    ENDIF.
  ENDLOOP.

ENDFORM. " SELECT_FI

      lt_main01-kcod = ''.

      MOVE-CORRESPONDING gt_main TO lt_main02.
      lt_main02-cod2 = '9'. lt_main02-banka = ''.
      lt_main02-kcod = ''.

      COLLECT lt_main02. CLEAR  lt_main02.

      lt_main03-kcod = ''.

    ENDIF.
    MOVE-CORRESPONDING gt_main TO lt_main.
    lt_main-cod1 = '9'. lt_main-bcod = '합계'.
    lt_main-cod2 = '9'. lt_main-banka = ''.
    lt_main-kcod = ''.  lt_main-ktext = ''.
    COLLECT lt_main. CLEAR  lt_main.
  ENDLOOP.

  IF lt_main01[] IS INITIAL.
    lt_main01-cod2 = '9'. lt_main01-banka = ''.
    lt_main01-kcod = ''.  lt_main01-ktext = ''.
    lt_main01-bcod = '소계(현금계정)'. lt_main01-cod1 = '1'.
    lt_main01-waers = c_krw.
    APPEND lt_main01. CLEAR  lt_main01.
  ENDIF.

  IF lt_main02[] IS INITIAL.
    lt_main02-cod2 = '9'. lt_main02-banka = ''.
    lt_main02-kcod = ''.  lt_main02-ktext = ''.
    lt_main02-bcod = '소계(별단&정기성)'. lt_main02-cod1 = '2'.
    lt_main02-waers = c_krw.
    APPEND lt_main02. CLEAR  lt_main02.
  ENDIF.

  IF lt_main03[] IS INITIAL.
    lt_main03-cod2 = '9'. lt_main03-banka = ''.
    lt_main03-kcod = ''.  lt_main03-ktext = ''.
    lt_main03-bcod = '소계(퇴직연금)'. lt_main03-cod1 = '3'.
    lt_main03-waers = c_krw.
    APPEND lt_main03. CLEAR  lt_main03.
  ENDIF.

  IF gt_main[] IS INITIAL.
    lt_main-cod1 = '9'. lt_main-bcod = '합계'.
    lt_main-cod2 = '9'. lt_main-banka = ''.
    lt_main-kcod = ''.  lt_main-ktext = ''.
    lt_main-waers = c_krw.
    APPEND lt_main. CLEAR  lt_main.
  ENDIF.

  CLEAR : lt_main01, lt_main02, lt_main03, lt_main, gt_main.
  LOOP AT lt_main01.
    MOVE-CORRESPONDING lt_main01 TO gt_main.
    APPEND gt_main. CLEAR gt_main.
  ENDLOOP.
  LOOP AT lt_main02.
    MOVE-CORRESPONDING lt_main02 TO gt_main.
    APPEND gt_main. CLEAR gt_main.
  ENDLOOP.
  LOOP AT lt_main03.
    MOVE-CORRESPONDING lt_main03 TO gt_main.
    APPEND gt_main. CLEAR gt_main.
  ENDLOOP.
  LOOP AT lt_main.
    MOVE-CORRESPONDING lt_main TO gt_main.
    APPEND gt_main. CLEAR gt_main.
  ENDLOOP.

ENDFORM. " SUB_TOTAL
