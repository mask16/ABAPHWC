FORM SELECT_BSIS_DATA .

           BANKA       LIKE BNKA-BANKA,  " 은행명
           TEXT1       LIKE T012T-TEXT1,
           WAERS       LIKE T012K-WAERS, " 통화키
           HKONT       LIKE T012K-HKONT, " G/L계정
         END OF LT_REC.

  DATA : BEGIN OF LT_HEAD OCCURS 0,
           HBKID       LIKE T012K-HBKID,
           HKTID       LIKE T012K-HKTID,
           HKONT       LIKE T012K-HKONT,
           WAERS       LIKE T012K-WAERS,
         END OF LT_HEAD.

  DATA : BEGIN OF LT_BANK OCCURS 0,
           BANKL       LIKE BNKA-BANKL,
           BANKA       LIKE BNKA-BANKA,
         END OF LT_BANK.

  DATA : BEGIN OF LT_T012T OCCURS 0,
           HBKID       LIKE T012T-HBKID,
           HKTID       LIKE T012T-HKTID,
           TEXT1       LIKE T012T-TEXT1,
         END OF LT_T012T.

  DATA : LT_0021 LIKE ZTRT0021 OCCURS 0 WITH HEADER LINE.

  DATA : BEGIN OF LT_ITEM OCCURS 0,
           HKONT LIKE BSIS-HKONT,
           SHKZG LIKE BSIS-SHKZG,
           WAERS LIKE BSIS-WAERS,
           WRBTR LIKE BSIS-WRBTR,
           DMBTR LIKE BSIS-DMBTR,
         END OF LT_ITEM.

  DATA : LT_MAIN LIKE GT_MAIN OCCURS 0 WITH HEADER LINE,
         LT_MOT LIKE GT_MAIN OCCURS 0 WITH HEADER LINE.

  RANGES : LR_HKONT FOR BSIS-HKONT.

  SELECT BANKL BANKA
    FROM BNKA
    INTO CORRESPONDING FIELDS OF TABLE LT_BANK
   WHERE BANKS EQ 'KR'.

** S) 2026.03.09 NYG - 은행 단축키 99998(기타은행) 제외
** 요청자 : 경리부 김진실P
  SELECT HBKID HKTID HKONT WAERS
    FROM T012K
    INTO CORRESPONDING FIELDS OF TABLE LT_HEAD
   WHERE BUKRS IN S_BUKRS AND HKONT NE '0091050010'
       AND HBKID NE '99998'.

  SELECT HBKID HKTID TEXT1
    FROM T012T
    INTO CORRESPONDING FIELDS OF TABLE LT_T012T
   WHERE BUKRS IN S_BUKRS AND SPRAS EQ SY-LANGU
        AND HBKID NE '99998'.
** 요청자 : 경리부 김진실P
** E) 2026.03.09 NYG - 은행 단축키 99998(기타은행) 제외

  CLEAR : LT_HEAD, LT_BANK, LT_T012T.
  SORT : LT_HEAD BY HKTID WAERS, LT_BANK BY BANKL,
         LT_T012T BY HBKID HKTID.

  LOOP AT LT_HEAD.
    MOVE-CORRESPONDING LT_HEAD TO LT_REC.
    READ TABLE LT_T012T WITH KEY HBKID = LT_REC-HBKID
                                 HKTID = LT_REC-HKTID
                                 BINARY SEARCH.
    IF SY-SUBRC = 0. LT_REC-TEXT1 = LT_T012T-TEXT1. ENDIF.

    LT_REC-KCOD = LT_HEAD-HKTID+2(1).
    LT_REC-BCOD = LT_HEAD-HBKID+0(3).

    READ TABLE LT_BANK WITH KEY BANKL = LT_REC-BCOD BINARY SEARCH.
    IF SY-SUBRC = 0. LT_REC-BANKA = LT_BANK-BANKA. ENDIF.

    CASE LT_REC-KCOD.
      WHEN 'C'. LT_REC-HTEXT = '당좌예금'.
      WHEN 'R'. LT_REC-HTEXT = '보통예금'.
      WHEN 'O'. LT_REC-HTEXT = '보통예금'.
      WHEN '0'. LT_REC-HTEXT = '현금'.
    ENDCASE.

    CASE LT_REC-HKONT.
      WHEN '11010010'. LT_REC-TEXT1 = '현금(본사)'.
      WHEN '11010020'. LT_REC-TEXT1 = '현금(포항공장)'.
      WHEN '11010030'. LT_REC-TEXT1 = '현금(이천공장)'.
      WHEN '11010040'. LT_REC-TEXT1 = '현금(고창공장)'.
    ENDCASE.

    COLLECT LT_REC. CLEAR LT_REC.
  ENDLOOP.

  CLEAR  LT_REC.
  LOOP AT LT_REC. _RANGES : LR_HKONT 'EQ' LT_REC-HKONT ''. ENDLOOP.

  DELETE ADJACENT DUPLICATES FROM LR_HKONT.
*----------------------------------------------------------------------*
* 은행원장
*----------------------------------------------------------------------*
  SELECT * FROM ZTRT0021
    INTO CORRESPONDING FIELDS OF TABLE LT_0021
   WHERE BUKRS IN S_BUKRS AND DATUM IN S_BUDAT.
*----------------------------------------------------------------------*
* 회계원장
*----------------------------------------------------------------------*
  SELECT HKONT SHKZG WAERS WRBTR DMBTR FROM BSIS
    INTO CORRESPONDING FIELDS OF TABLE LT_ITEM
   WHERE BUKRS IN S_BUKRS AND BUDAT IN S_BUDAT
     AND HKONT IN LR_HKONT.
*----------------------------------------------------------------------*
* 합치기
*----------------------------------------------------------------------*
  CLEAR : LT_REC, LT_0021, LT_ITEM.

  SORT : LT_0021 BY HBKID HKTID WAERS,
         LT_ITEM BY HKONT WAERS.

  LOOP AT LT_REC.
*-- 은행원장
    READ TABLE LT_0021 WITH KEY HBKID = LT_REC-HBKID
                                HKTID = LT_REC-HKTID
                                WAERS = LT_REC-WAERS BINARY SEARCH.
    IF SY-SUBRC = 0.
      LOOP AT LT_0021 FROM SY-TABIX.
        IF LT_0021-HBKID = LT_REC-HBKID AND LT_0021-HKTID = LT_REC-HKTID
       AND LT_0021-WAERS = LT_REC-WAERS.
          MOVE-CORRESPONDING LT_REC TO LT_MAIN.
          LT_MAIN-WAERS01 = LT_REC-WAERS. LT_MAIN-WAERS02 = C_KRW.
          LT_MAIN-DMBTR01 = LT_0021-DMBTR. LT_MAIN-DMBTR02 = LT_0021-WRBTR.
          COLLECT LT_MAIN. CLEAR LT_MAIN.
        ELSE.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDIF.
*-- 회계원장
* 2013.05.03 START
*    READ TABLE lt_item WITH KEY hkont = lt_rec-hkont
*                                waers = lt_rec-waers BINARY SEARCH.
    READ TABLE LT_ITEM WITH KEY HKONT = LT_REC-HKONT
                                                               BINARY SEARCH.
* 2013.05.03 END
    IF SY-SUBRC = 0.
      LOOP AT LT_ITEM FROM SY-TABIX.
* 2013.05.03 START
*        IF lt_item-hkont = lt_rec-hkont AND lt_item-waers = lt_rec-waers.
        IF LT_ITEM-HKONT = LT_REC-HKONT.
* 2013.05.03 END
          MOVE-CORRESPONDING LT_REC TO LT_MAIN.
          LT_MAIN-WAERS01 = LT_REC-WAERS. LT_MAIN-WAERS02 = C_KRW.

          IF LT_ITEM-SHKZG = C_S.
            LT_MAIN-DMBTR03 = LT_ITEM-DMBTR.
            LT_MAIN-DMBTR04 = LT_ITEM-WRBTR.
          ELSEIF LT_ITEM-SHKZG = C_H.
            LT_MAIN-DMBTR03 = LT_ITEM-DMBTR * -1.
            LT_MAIN-DMBTR04 = LT_ITEM-WRBTR * -1.
          ENDIF.
          COLLECT LT_MAIN. CLEAR LT_MAIN.
        ELSE.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDIF.

    CLEAR : LT_MAIN.
  ENDLOOP.

  IF LT_MAIN[] IS NOT INITIAL.
    LOOP AT LT_MAIN.
      MOVE-CORRESPONDING LT_MAIN TO LT_MOT.
      COLLECT LT_MOT. CLEAR LT_MOT.
    ENDLOOP.

    LOOP AT LT_MOT.
      MOVE-CORRESPONDING LT_MOT TO GT_MAIN.
      APPEND GT_MAIN. CLEAR GT_MAIN.
    ENDLOOP.
  ENDIF.
