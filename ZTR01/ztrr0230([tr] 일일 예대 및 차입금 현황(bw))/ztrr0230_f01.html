         ls_main02 LIKE gt_main.

  LOOP AT gt_main.
    CASE gt_main-cod1.
      WHEN '01'.
        MOVE-CORRESPONDING gt_main TO ls_main01.
      WHEN '02'.
        MOVE-CORRESPONDING gt_main TO ls_main02.
    ENDCASE.
  ENDLOOP.

  gt_main-cod1 = '03'. gt_main-cod2 = '01'.
