FORM GET_DATA .

  DATA: LV_VTWEG TYPE ZPPT0090-VTWEG,
        LT_TEMP  TYPE STANDARD TABLE OF TY_DATA.

  CLEAR: GT_DATA , GT_DATA[].

  " 1단계: ZPPT0110 데이터 조회
  SELECT PBARCODE ZPLNUM ARBPL HOGI PDA_SHIFT T_TIME LOTNO MSG WERKS
         PROD_GR_DATE SHIP_GR_DATE PROD_GR_CHG_ID SHIP_GR_CHG_ID
         PROD_WK_DATE PROD_WK_TIME SHIP_WK_DATE SHIP_WK_TIME PERNR
         MENGE AS MENGE0110
    INTO CORRESPONDING FIELDS OF TABLE GT_DATA
    FROM ZPPT0110
    WHERE WERKS           IN S_WERKS                " 플랜트
      AND PBARCODE        IN S_PBARCD               " 팔레트바코드
      AND PROD_GR_DATE    IN S_PRDATE               " 생산 입고일
      AND SHIP_GR_DATE    IN S_SHDATE               " 출하 입고일
      AND PROD_GR_CHG_ID  IN S_PRID                 " 생산 입고 데이터 변경 ID
      AND SHIP_GR_CHG_ID  IN S_SHID                 " 출하 입고 데이터 변경 ID
      AND PDA_SHIFT       IN S_SHIFT.               " PDA 작업조

  IF GT_DATA[] IS NOT INITIAL.

    LOOP AT GT_DATA INTO GS_DATA.

      " ZPPT0170: 품질LOT, 생산타입, 자재번호, 수량, 기본단위, 포장공장 조회
      SELECT SINGLE  QCHARG PRODTYPE MATNR MENGE MEINS ZWERKSP
        INTO CORRESPONDING FIELDS OF GS_DATA
        FROM ZPPT0170
        WHERE PBARCODE  = GS_DATA-PBARCODE
          AND BARCODE   = GS_DATA-LOTNO             " MFG 체크
          AND MENGE     = GS_DATA-MENGE0110         " 수량 체크
          AND PRODTYPE IN S_PRTYPE                  " 생산 타입
          AND QCHARG   IN S_QCHARG                  " 품질 Lot.
          AND MATNR    IN S_MATNR                   " 자재번호.
          AND MATNR   <> '000000000020004914'       " CUT WIRE [A-3 ] 품목 특정 자재 제외 추가
          AND ZWERKSP  IN S_ZWERKS.                 " 포장 공장
      IF SY-SUBRC <> 0.
        DELETE GT_DATA.  " 조건 불충족 시 삭제
        CONTINUE.
      ENDIF.


      " MARA: 자재 유형, 자재 그룹, 크기/치수
      SELECT SINGLE MTART MATKL GROES
       INTO (GS_DATA-MTART, GS_DATA-MATKL, GS_DATA-GROES)
       FROM MARA
       WHERE MATNR = GS_DATA-MATNR
         AND MTART IN S_MTART                     " 자재 유형
         AND MATKL IN S_MATKL                     " 자재 그룹
         AND GROES IN S_GROES.                    " 크기/치수
      IF SY-SUBRC <> 0.
        DELETE GT_DATA.  " 조건 불충족 시 삭제
        CONTINUE.
      ENDIF.

      " MARC: MRP 관리자 저장위치
      SELECT SINGLE DISPO
            INTO GS_DATA-DISPO
            FROM MARC
            WHERE MATNR = GS_DATA-MATNR
              AND WERKS = GS_DATA-WERKS
              AND DISPO IN S_DISPO.               " MRP 관리자
      IF SY-SUBRC <> 0.
        DELETE GT_DATA.  " 조건 불충족 시 삭제
        CONTINUE.
      ENDIF.

      " MAKT: 자재 내역 조회
      SELECT SINGLE MAKTX
        INTO GS_DATA-MAKTX
        FROM MAKT
        WHERE MATNR = GS_DATA-MATNR
          AND SPRAS = SY-LANGU.
      IF SY-SUBRC <> 0.
        DELETE GT_DATA.  " 조건 불충족 시 삭제
        CONTINUE.
      ENDIF.

      " ZPPT0090 저장위치 조회
      SELECT SINGLE VTWEG ZWERKS
        INTO (LV_VTWEG, GS_DATA-ZWERKS)
        FROM ZPPT0090
        WHERE ZPLNUM = GS_DATA-ZPLNUM.
      IF SY-SUBRC <> 0.
        DELETE GT_DATA.
        CONTINUE.
      ELSEIF SY-SUBRC = 0 AND GS_DATA-WERKS = '1010'.
        CASE LV_VTWEG.
          WHEN '10'.  " 내수
            GS_DATA-LGORT = '1010'.
          WHEN '20'.  " 수출
            GS_DATA-LGORT = '1011'.
        ENDCASE.
      ELSEIF SY-SUBRC = 0 AND GS_DATA-WERKS = '1020'.
        CASE LV_VTWEG.
          WHEN '10'.  " 내수
            GS_DATA-LGORT = '1020'.
          WHEN '20'.  " 수출
            GS_DATA-LGORT = '1021'.
        ENDCASE.
      ENDIF.
      " 시험품일 경우 5900 고정
      IF GS_DATA-PRODTYPE = '시험품'.
        GS_DATA-LGORT = '5900'.
      ENDIF.

      " s_lgpro 조건으로 필터링
      IF GS_DATA-LGORT NOT IN S_LGPRO.
        DELETE GT_DATA.
        CONTINUE.
      ENDIF.

      " 생산 작업자 이름
      SELECT SINGLE NAME
        INTO GS_DATA-PROD_GR_NAME
        FROM ZSDT5000
        WHERE USERID = GS_DATA-PROD_GR_CHG_ID.

      " 출하 작업자 이름
      SELECT SINGLE NAME
          INTO GS_DATA-SHIP_GR_NAME
          FROM ZSDT5000
          WHERE USERID = GS_DATA-SHIP_GR_CHG_ID.

      " 정보 저장
      MODIFY GT_DATA FROM GS_DATA.
    ENDLOOP.


*   팔레트 수량 집계 레이아웃 처리 조건 체크
    IF P_PALLET = 'X'.

      " 기존 GT_DATA 백업
      LT_TEMP[] = GT_DATA[].

      " GT_DATA 비우기 (덮어쓰기 위해)
      CLEAR GT_DATA.

      " 팔레트별 수량 집계
      LOOP AT LT_TEMP INTO GS_DATA.

        " T_TIME, LOTNO, ZPLNUM 초기화 - 팔레트별 다른 데이터 숨김
        CLEAR: GS_DATA-T_TIME,
               GS_DATA-LOTNO,
               GS_DATA-ZPLNUM.

        " 집계: 기존 GT_DATA에 COLLECT
        COLLECT GS_DATA INTO GT_DATA.

      ENDLOOP.

    ENDIF.
  ENDIF.

ENDFORM. " GET_DATA

FORM MAKE_FIELDCAT USING P_GUB P_FNAME P_CON.

  "FIELD 속성 READ
  IF   P_GUB =  'S'.
    CLEAR : GS_FIELDCAT, G_FIELDCAT_H.
    READ  TABLE G_FIELDCAT_H WITH KEY FIELDNAME  = P_FNAME.
    MOVE-CORRESPONDING G_FIELDCAT_H TO GS_FIELDCAT.
    GS_FIELDCAT-FIELDNAME  = P_FNAME.
    GS_FIELDCAT-SCRTEXT_L  = G_FIELDCAT_H-SELTEXT_L.
    GS_FIELDCAT-SCRTEXT_M  = G_FIELDCAT_H-SELTEXT_M.
    GS_FIELDCAT-SCRTEXT_S  = G_FIELDCAT_H-SELTEXT_S.
    GS_FIELDCAT-REPTEXT    = G_FIELDCAT_H-SELTEXT_L.
    GS_FIELDCAT-COLDDICTXT = G_FIELDCAT_H-DDICTXT.
    GS_FIELDCAT-REF_FIELD  = G_FIELDCAT_H-REF_FIELDNAME.
    GS_FIELDCAT-REF_TABLE  = G_FIELDCAT_H-REF_TABNAME.
    GS_FIELDCAT-ROUNDFIELD = G_FIELDCAT_H-ROUNDFIELDNAME.
    GS_FIELDCAT-DECMLFIELD = G_FIELDCAT_H-DECIMALSFIELDNAME.
    GS_FIELDCAT-DECIMALS   = G_FIELDCAT_H-DECIMALS_OUT.

    EXIT.
  ENDIF.

  "속성 MOVE
  DATA L_COL(40).
  FIELD-SYMBOLS <FS>.
  CONCATENATE 'GS_FIELDCAT-' P_FNAME  INTO L_COL.
  ASSIGN      (L_COL)         TO       <FS>.
  MOVE         P_CON          TO       <FS>.

  IF P_FNAME = 'SCRTEXT_S'.
    GS_FIELDCAT-COLDDICTXT = 'S'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_L'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.
  ELSEIF P_FNAME = 'SCRTEXT_L'.
    GS_FIELDCAT-COLDDICTXT = 'L'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.
  ELSEIF P_FNAME = 'SCRTEXT_M'.
    GS_FIELDCAT-COLDDICTXT = 'M'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_L'  INTO L_COL.
    ASSIGN      (L_COL)         TO       <FS>.
    MOVE         P_CON          TO       <FS>.
  ENDIF.

* DATA  APPEND
  CHECK  P_GUB = 'E'.

  " 컬럼 순서 번호를 부여하고 필드카탈로그에 추가
  COL_POS = COL_POS + 1.
  GS_FIELDCAT-COL_POS = COL_POS.
  IF GS_FIELDCAT-OUTPUTLEN IS INITIAL.
    GS_FIELDCAT-OUTPUTLEN = GS_FIELDCAT-INTLEN + 2.
  ENDIF.

  APPEND GS_FIELDCAT TO FIELDCAT.

ENDFORM. " MAKE_FIELDCAT
