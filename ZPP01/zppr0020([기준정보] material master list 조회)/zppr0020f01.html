FORM select_main_data .

  CLEAR : gt_main[], gt_main.

  SELECT a~matnr c~maktx a~mbrsh a~mtart a~matkl
         a~bismt a~meins a~lvorm AS lvorm_c  a~mstae a~ersda
         a~ernam a~zmaktx a~groes
         b~werks b~lvorm AS lvorm_p b~mmsta b~dismm
         b~dispo b~disls b~beskz b~bstfe b~sobsl
         b~rgekz b~lgpro b~lgfsb b~dzeit b~plifz
         b~webaz b~fhori b~eisbe b~strgr b~vrmod
         b~vint1 b~vint2 b~mtvfp b~altsl b~sbdkz
         b~sauft b~sfepr b~fevor b~sfcpf b~kzech
         a~xchpf b~prctr d~bklas b~ausss b~kausf a~extwg
        a~prdha t1~mvgr1  t2~mvgr2 t3~mvgr3
    INTO CORRESPONDING FIELDS OF TABLE gt_main
                  FROM mara AS a
            INNER JOIN marc AS b
                    ON a~matnr = b~matnr
            INNER JOIN makt AS c
                    ON a~matnr = c~matnr
                   AND c~spras = sy-langu
            INNER JOIN mvke AS k
                    ON a~matnr = k~matnr
             INNER JOIN tvm1t AS t1
                   ON t1~mvgr1 = k~mvgr1
                   AND t1~spras = sy-langu
            INNER JOIN tvm2t AS t2
                   ON t2~mvgr2 = k~mvgr2
                   AND t2~spras = sy-langu
            INNER JOIN tvm3t AS t3
                   ON t3~mvgr3 = k~mvgr3
                   AND t3~spras = sy-langu
       LEFT OUTER JOIN mbew AS d
                    ON b~matnr = d~matnr
                   AND b~werks = d~bwkey
                 WHERE a~matnr IN so_matnr
                   AND a~bismt IN so_bismt
                   AND a~mtart IN so_mtart
                   AND a~matkl IN so_matkl
                   AND a~ersda IN so_ersda
                   AND a~ernam IN so_ernam
                   AND b~werks IN so_werks
                   AND b~dispo IN so_dispo
                   AND b~fevor IN so_fevor.

  LOOP AT gt_main.

*    PH1 필드 추가
    SELECT SINGLE prodh
        INTO (gt_main-prdha1)
    FROM t179t
    WHERE prodh = gt_main-prdha(5)
      AND spras = sy-langu.

    SELECT SINGLE vtext
        INTO (gt_main-vtext1)
    FROM t179t
    WHERE prodh = gt_main-prdha(5)
      AND spras = sy-langu.

*    PH2 필드 추가
    SELECT SINGLE prodh
        INTO (gt_main-prdha2)
    FROM t179t
    WHERE prodh = gt_main-prdha(10)
      AND spras = sy-langu.

    SELECT SINGLE vtext
        INTO (gt_main-vtext2)
    FROM t179t
    WHERE prodh = gt_main-prdha(10)
      AND spras = sy-langu.

*    PH3 필드 추가
    SELECT SINGLE vtext
        INTO (gt_main-vtext3)
    FROM t179t
    WHERE prodh = gt_main-prdha
     AND spras = sy-langu.

*    규격1 필드 추가
    SELECT SINGLE bezei
        INTO (gt_main-bezei1)
    FROM tvm1t
    WHERE mvgr1 = gt_main-mvgr1
       AND spras = sy-langu.

*   규격2 필드 추가
    SELECT SINGLE bezei
        INTO (gt_main-bezei2)
    FROM tvm2t
    WHERE mvgr2 = gt_main-mvgr2
       AND spras = sy-langu.

*    포장사양 필드 추가
    SELECT SINGLE bezei
        INTO (gt_main-bezei3)
    FROM tvm3t
    WHERE mvgr3 = gt_main-mvgr3
       AND spras = sy-langu.

    MODIFY gt_main.

  ENDLOOP.

DELETE ADJACENT DUPLICATES FROM gt_main COMPARING matnr werks.

ENDFORM. " SELECT_MAIN_DATA

FORM make_data .

  DATA : lv_cnt TYPE i,
         lv_scnt(20).

  SORT gt_main BY matnr werks.
  CLEAR : gt_result[], gt_result.
  LOOP AT gt_main.

    MOVE-CORRESPONDING gt_main TO gt_result.
    APPEND gt_result. CLEAR : gt_result.
  ENDLOOP.

*조회건수 DISPLAY

  DESCRIBE TABLE gt_result LINES gv_cnt.
  WRITE gv_cnt TO lv_scnt.
  MESSAGE s021 WITH lv_scnt.

ENDFORM. " MAKE_DATA
