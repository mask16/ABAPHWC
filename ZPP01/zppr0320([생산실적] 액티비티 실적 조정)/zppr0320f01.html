*----------------------------------------------------------------------*
FORM SET_QTY_OF_IT_FROM_QTY_OF_HEAD.

  DATA : BEGIN OF LT_AUFNR OCCURS 0,
          AUFNR       TYPE AFKO-AUFNR,
          END OF LT_AUFNR.

  DATA : BEGIN OF LT_COEP OCCURS 0,
          CHK(01),    "MARK
          KOKRS       TYPE COBK-KOKRS,
          BELNR       TYPE COEP-BELNR,
          BUZEI       TYPE COEP-BUZEI,
          OBJNR       TYPE COEP-OBJNR,
          PAROB       TYPE COEP-PAROB,
          AUFNR       TYPE AFKO-AUFNR,

          ACODE(3),
          MATNR       TYPE MARA-MATNR,
          MATNR_NM    TYPE MAKT-MAKTX,
          VERID       TYPE CKMLMV013-VERID,
          VERID_NM    TYPE CRTX-KTEXT,
*          BUDAT       TYPE COBK-BUDAT,
          MEGBTR      TYPE MEGXXX,
*          MTIME       TYPE MEGXXX,
*          LTIME       TYPE MEGXXX,
*          PTIME       TYPE MEGXXX,
          MEINH       TYPE CO_MEINH,

         END OF LT_COEP.
  DATA : LT_ITEM LIKE GT_ITEM OCCURS 0 WITH HEADER LINE,
         LS_ITEM LIKE GT_ITEM.

  DATA : LV_PAROB TYPE COEP-PAROB,
         LV_MENGE      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG1      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG2      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG3      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG4      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG5      TYPE DMF_MAXLS, "MESXXX,
         LV_MENG6      TYPE DMF_MAXLS, "MESXXX,
*         LV_MEGBTR1     TYPE MEGXXX,
*         LV_MEGBTR2     TYPE MEGXXX,
         LV_RATE(15)    TYPE P DECIMALS 8,
         LV_RATE1(15)    TYPE P DECIMALS 8,
         LV_RATE2(15)    TYPE P DECIMALS 8,
         LV_RATE3(15)    TYPE P DECIMALS 8,
         LV_RATE4(15)    TYPE P DECIMALS 8,
         LV_RATE5(15)    TYPE P DECIMALS 8,
         LV_RATE6(15)    TYPE P DECIMALS 8.

  LV_PAROB(2) = 'KL'.
  LV_PAROB+2(1) = PA_WERKS(1).
  LV_PAROB+3(3) = '000'.
  LV_PAROB+6(10) = GT_HEAD-KOSTL.
  LV_PAROB+16(1) = '%'.

  READ TABLE SO_BUDAT INDEX 1.

* 기생성된 액티비티
  SELECT A~MEGBTR A~MEINH A~PAROB A~BELNR A~BUZEI A~KOKRS
         A~MATNR A~OBJNR
    INTO CORRESPONDING FIELDS OF TABLE LT_COEP
                  FROM COEP  AS A
            INNER JOIN COBK AS B
                    ON A~KOKRS = B~KOKRS AND
                       A~BELNR = B~BELNR
                 WHERE A~WERKS EQ PA_WERKS AND
                       A~VRGNG = 'RKL' AND
                       A~PAROB LIKE LV_PAROB AND
                       B~BUDAT IN SO_BUDAT.

  IF SY-SUBRC EQ 0.

    LOOP AT LT_COEP.
      CLEAR LT_ITEM.

      MOVE-CORRESPONDING LT_COEP TO LT_ITEM.
      LT_ITEM-KOSTL = LT_COEP-PAROB+6(10).
      LT_COEP-AUFNR = LT_COEP-OBJNR+2(12).

      LT_COEP-ACODE = LT_COEP-PAROB+16(3).

*      SELECT SINGLE PRWRK PMATN VERID
*             INTO (LT_ITEM-WERKS, LT_ITEM-MATNR, GT_RESULT-VERID)
*             FROM CKMLMV013
*             WHERE AUFNR = LT_COEP-AUFNR.
      SELECT SINGLE PMATN VERID
             INTO (LT_ITEM-MATNR, LT_ITEM-VERID)
             FROM CKMLMV013
             WHERE AUFNR = LT_COEP-AUFNR AND
                   PRWRK = PA_WERKS.
      IF SY-SUBRC EQ 0.

*     생산실적
        READ TABLE LT_AUFNR WITH KEY AUFNR = LT_COEP-AUFNR.
        IF SY-SUBRC <> 0.
          CLEAR LV_MENGE.
          SELECT SUM( A~MENGE ) INTO LV_MENGE
                 FROM MSEG AS A INNER JOIN MKPF AS B ON A~MBLNR = B~MBLNR
                 WHERE A~AUFNR = LT_COEP-AUFNR AND
                       A~BWART = '131' AND
                       B~BUDAT IN SO_BUDAT.
          LT_ITEM-MENGE = LV_MENGE.

          SELECT SUM( A~MENGE ) INTO LV_MENGE
                 FROM MSEG AS A INNER JOIN MKPF AS B ON A~MBLNR = B~MBLNR
                 WHERE A~AUFNR = LT_COEP-AUFNR AND
                       A~BWART = '132' AND
                       B~BUDAT IN SO_BUDAT.
          LT_ITEM-MENGE = LT_ITEM-MENGE - LV_MENGE.

          LT_AUFNR-AUFNR = LT_COEP-AUFNR.
          APPEND LT_AUFNR.
          SORT LT_AUFNR.
          DELETE ADJACENT DUPLICATES FROM LT_AUFNR COMPARING AUFNR.
        ENDIF.

*     기본단위
        SELECT SINGLE MEINS INTO LT_ITEM-MEINS FROM MARA
               WHERE MATNR = LT_ITEM-MATNR.

*     분단위 시간 변환
        IF LT_ITEM-MEINH = 'MIN'.
          LT_ITEM-MEINH = 'H'.
          LT_COEP-MEGBTR = LT_COEP-MEGBTR / 60.
        ENDIF.

        CASE LT_COEP-ACODE.
          WHEN '100'.
            LT_ITEM-ATIME1 = LT_COEP-MEGBTR.
          WHEN '200'.
            LT_ITEM-ATIME2 = LT_COEP-MEGBTR.
          WHEN '300'.
            LT_ITEM-ATIME3 = LT_COEP-MEGBTR.
          WHEN '400'.
            LT_ITEM-ATIME4 = LT_COEP-MEGBTR.
          WHEN '500'.
            LT_ITEM-BTIME1 = LT_COEP-MEGBTR.
          WHEN '600'.
            LT_ITEM-BTIME2 = LT_COEP-MEGBTR.
        ENDCASE.

*        LV_MENGE = LV_MENGE + LT_COEP-MEGBTR.

*        CLEAR : GT_RESULT-AUFNR, GT_RESULT-ACODE.

        DATA LV_DATUV TYPE PLPO-DATUV.
        PERFORM GET_DATUV USING LT_ITEM-MATNR LT_ITEM-KOSTL LV_DATUV.

        SELECT     E~ARBPL A~VORNR
                     INTO (LT_ITEM-ARBPL, LT_ITEM-VORNR)
                     FROM PLPO AS A INNER JOIN MAPL AS B ON  A~PLNNR = B~PLNNR
                                                         AND A~PLNTY = B~PLNTY
                                    INNER JOIN PLAS AS C ON  A~PLNNR = C~PLNNR
                                                         AND A~PLNTY = C~PLNTY
                                                         AND A~PLNKN = C~PLNKN
                                                         AND A~AENNR = C~AENNR
                                    INNER JOIN PLKO AS D ON  A~PLNNR = D~PLNNR
                                                         AND A~PLNTY = D~PLNTY
                                                         AND A~AENNR = D~AENNR
                                    INNER JOIN CRHD AS E ON A~ARBID = E~OBJID
                                                         AND E~OBJTY = 'A'
                                    INNER JOIN CRCO AS F ON E~OBJID = F~OBJID
                                                         AND F~OBJTY = 'A'
                     WHERE A~LOEKZ = ' ' AND
                           A~DATUV = LV_DATUV AND
                           B~MATNR = LT_ITEM-MATNR AND
                           B~WERKS = PA_WERKS AND
                           C~LOEKZ = ' ' AND
                           D~LOEKZ = ' ' AND
*                   E~ARBPL = LT_ITEM-VERID AND
                           F~KOSTL = LT_ITEM-KOSTL AND
                           F~LSTAR = '100'.

          SELECT SINGLE  B~KTEXT INTO  LT_ITEM-ARBPL_NM
             FROM CRHD AS A INNER JOIN CRTX AS B ON A~OBJTY = B~OBJTY AND
                                                    A~OBJID = B~OBJID
             WHERE A~OBJTY = 'A' AND
                   A~ARBPL = LT_ITEM-ARBPL AND
                   B~SPRAS = SY-LANGU.

          COLLECT LT_ITEM.
        ENDSELECT.

      ENDIF.
    ENDLOOP.

*   표준액티비티 가져오기/총수량구하기
    CLEAR LV_MENGE.
    LOOP AT LT_ITEM.
*     표준액티비티
      SELECT "SINGLE
             A~LAR01 A~VGE01 A~VGW01
             A~LAR02 A~VGE02 A~VGW02
             A~LAR03 A~VGE03 A~VGW03
             A~LAR04 A~VGE04 A~VGW04
             A~LAR05 A~VGE05 A~VGW05
             A~LAR06 A~VGE06 A~VGW06
             A~BMSCH A~MEINH
             INTO (LT_ITEM-LAR01, LT_ITEM-VGE01, LT_ITEM-VGW01,
                   LT_ITEM-LAR02, LT_ITEM-VGE02, LT_ITEM-VGW02,
                   LT_ITEM-LAR03, LT_ITEM-VGE03, LT_ITEM-VGW03,
                   LT_ITEM-LAR04, LT_ITEM-VGE04, LT_ITEM-VGW04,
                   LT_ITEM-LAR05, LT_ITEM-VGE05, LT_ITEM-VGW05,
                   LT_ITEM-LAR06, LT_ITEM-VGE06, LT_ITEM-VGW06,
                   LT_ITEM-BMSCH, LT_ITEM-MEINX)
             FROM PLPO AS A INNER JOIN MAPL AS B ON  A~PLNNR = B~PLNNR
                                                 AND A~PLNTY = B~PLNTY
                            INNER JOIN PLAS AS C ON  A~PLNNR = C~PLNNR
                                                 AND A~PLNTY = C~PLNTY
                                                 AND A~PLNKN = C~PLNKN
                                                 AND A~AENNR = C~AENNR
                            INNER JOIN PLKO AS D ON  A~PLNNR = D~PLNNR
                                                 AND A~PLNTY = D~PLNTY
                                                 AND A~AENNR = D~AENNR
                            INNER JOIN CRHD AS E ON A~ARBID = E~OBJID
                            INNER JOIN CRCO AS F ON E~OBJID = F~OBJID
                                                 AND F~OBJTY = 'A'
             WHERE A~LOEKZ = ' ' AND
                   A~DATUV <= SO_BUDAT-LOW AND
                   B~MATNR = LT_ITEM-MATNR AND
                   B~WERKS = PA_WERKS AND
                   C~LOEKZ = ' ' AND
                   D~LOEKZ = ' ' AND
                   E~ARBPL = LT_ITEM-ARBPL AND
                   F~KOSTL = LT_ITEM-KOSTL
                   ORDER BY A~DATUV DESCENDING.

        LT_ITEM-VGW01 = LT_ITEM-VGW01 * 10000 / LT_ITEM-BMSCH.
        LT_ITEM-VGW02 = LT_ITEM-VGW02 * 10000 / LT_ITEM-BMSCH.
        LT_ITEM-VGW03 = LT_ITEM-VGW03 * 10000 / LT_ITEM-BMSCH.
        LT_ITEM-VGW04 = LT_ITEM-VGW04 * 10000 / LT_ITEM-BMSCH.
        LT_ITEM-VGW05 = LT_ITEM-VGW05 * 10000 / LT_ITEM-BMSCH.
        LT_ITEM-VGW06 = LT_ITEM-VGW06 * 10000 / LT_ITEM-BMSCH.
        EXIT.
      ENDSELECT.

      LV_MENG1 = LV_MENG1 + LT_ITEM-MENGE * LT_ITEM-VGW01.
      LV_MENG2 = LV_MENG2 + LT_ITEM-MENGE * LT_ITEM-VGW02.
      LV_MENG3 = LV_MENG3 + LT_ITEM-MENGE * LT_ITEM-VGW03.
      LV_MENG4 = LV_MENG4 + LT_ITEM-MENGE * LT_ITEM-VGW04.
      LV_MENG5 = LV_MENG5 + LT_ITEM-MENGE * LT_ITEM-VGW05.
      LV_MENG6 = LV_MENG6 + LT_ITEM-MENGE * LT_ITEM-VGW06.

      MODIFY LT_ITEM.
    ENDLOOP.

*   배분
    IF LT_ITEM[] IS NOT INITIAL.
      LOOP AT LT_ITEM.
        MOVE-CORRESPONDING LT_ITEM TO GT_ITEM.
        GT_ITEM-ICON = ICON_CREATE.

        GT_ITEM-KOSTL_NM = GT_HEAD-KOSTL_NM.
*     작업장명
        PERFORM GET_CRTX_KTEXT(ZPPR0250) USING GT_ITEM-VERID GT_ITEM-VERID_NM.
*     자재내역
        PERFORM GET_MAKT_MAKTX(ZPPR0250) USING GT_ITEM-MATNR GT_ITEM-MATNR_NM.
**     표준액티비티
*        PERFORM GET_PLPO_TIME.
*     시간
        LV_RATE1 = ( LT_ITEM-MENGE * LT_ITEM-VGW01 ) / LV_MENG1.
        LV_RATE2 = ( LT_ITEM-MENGE * LT_ITEM-VGW02 ) / LV_MENG2.
        LV_RATE3 = ( LT_ITEM-MENGE * LT_ITEM-VGW03 ) / LV_MENG3.
        LV_RATE4 = ( LT_ITEM-MENGE * LT_ITEM-VGW04 ) / LV_MENG4.
        LV_RATE5 = ( LT_ITEM-MENGE * LT_ITEM-VGW05 ) / LV_MENG5.
        LV_RATE6 = ( LT_ITEM-MENGE * LT_ITEM-VGW06 ) / LV_MENG6.

        GT_ITEM-PTIME1 = LV_RATE1 * GT_HEAD-PTIME1.
        LS_ITEM-PTIME1 = LS_ITEM-PTIME1 + GT_ITEM-PTIME1.

        GT_ITEM-PTIME2 = LV_RATE2 * GT_HEAD-PTIME2.
        LS_ITEM-PTIME2 = LS_ITEM-PTIME2 + GT_ITEM-PTIME2.

        GT_ITEM-PTIME3 = LV_RATE3 * GT_HEAD-PTIME3.
        LS_ITEM-PTIME3 = LS_ITEM-PTIME3 + GT_ITEM-PTIME3.

        GT_ITEM-PTIME4 = LV_RATE4 * GT_HEAD-PTIME4.
        LS_ITEM-PTIME4 = LS_ITEM-PTIME4 + GT_ITEM-PTIME4.

        GT_ITEM-MTIME1 = LV_RATE5 * GT_HEAD-MTIME1.
        LS_ITEM-MTIME1 = LS_ITEM-MTIME1 + GT_ITEM-MTIME1.

        GT_ITEM-MTIME2 = LV_RATE6 * GT_HEAD-MTIME2.
        LS_ITEM-MTIME2 = LS_ITEM-MTIME2 + GT_ITEM-MTIME2.

        APPEND GT_ITEM.
      ENDLOOP.

*     단수차이 보정
      IF LS_ITEM-PTIME1 <> GT_HEAD-PTIME1 OR
         LS_ITEM-PTIME2 <> GT_HEAD-PTIME2 OR
         LS_ITEM-PTIME3 <> GT_HEAD-PTIME3 OR
         LS_ITEM-PTIME4 <> GT_HEAD-PTIME4 OR
         LS_ITEM-MTIME1 <> GT_HEAD-MTIME1 OR
         LS_ITEM-MTIME2 <> GT_HEAD-MTIME2.

        LS_ITEM-PTIME1 = GT_HEAD-PTIME1 - LS_ITEM-PTIME1.
        LS_ITEM-PTIME2 = GT_HEAD-PTIME2 - LS_ITEM-PTIME2.
        LS_ITEM-PTIME3 = GT_HEAD-PTIME3 - LS_ITEM-PTIME3.
        LS_ITEM-PTIME4 = GT_HEAD-PTIME4 - LS_ITEM-PTIME4.
        LS_ITEM-MTIME1 = GT_HEAD-MTIME1 - LS_ITEM-MTIME1.
        LS_ITEM-MTIME2 = GT_HEAD-MTIME2 - LS_ITEM-MTIME2.

        LOOP AT GT_ITEM WHERE KOSTL = GT_HEAD-KOSTL.
          GT_ITEM-PTIME1 = GT_ITEM-PTIME1 + LS_ITEM-PTIME1.
          GT_ITEM-PTIME2 = GT_ITEM-PTIME2 + LS_ITEM-PTIME2.
          GT_ITEM-PTIME3 = GT_ITEM-PTIME3 + LS_ITEM-PTIME3.
          GT_ITEM-PTIME4 = GT_ITEM-PTIME4 + LS_ITEM-PTIME4.
          GT_ITEM-MTIME1 = GT_ITEM-MTIME1 + LS_ITEM-MTIME1.
          GT_ITEM-MTIME2 = GT_ITEM-MTIME2 + LS_ITEM-MTIME2.
          MODIFY GT_ITEM.
          EXIT.
        ENDLOOP.
      ENDIF.

    ENDIF.

  ELSE.

    CLEAR GT_ITEM.

    GT_ITEM-ICON = ICON_LED_RED.
    GT_ITEM-KOSTL = GT_HEAD-KOSTL.
    GT_ITEM-KOSTL_NM = GT_HEAD-KOSTL_NM.
    GT_ITEM-MSG = TEXT-M01.
    APPEND GT_ITEM.

  ENDIF.

ENDFORM. " SET_QTY_OF_IT_FROM_QTY_OF_HEAD
