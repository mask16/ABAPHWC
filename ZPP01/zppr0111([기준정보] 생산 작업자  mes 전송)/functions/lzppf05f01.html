FORM ENQUEUE_READ CHANGING P_LOCK.

  DATA : L_NUMBER   TYPE SY-TABIX,
         L_SUBRC    TYPE SY-SUBRC,
         LT_ENQ     TYPE TABLE OF SEQG3 WITH HEADER LINE.

  P_LOCK = 'X'.
  CLEAR : L_NUMBER, L_SUBRC, LT_ENQ, LT_ENQ[].
  CALL FUNCTION 'ENQUEUE_READ'
    IMPORTING
      NUMBER                = L_NUMBER
      SUBRC                 = L_SUBRC
    TABLES
      ENQ                   = LT_ENQ
    EXCEPTIONS
      COMMUNICATION_FAILURE = 1
      SYSTEM_FAILURE        = 2
      OTHERS                = 3.
  IF SY-SUBRC <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
*         WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ELSE.
    IF LT_ENQ[] IS NOT INITIAL.
      LOOP AT LT_ENQ WHERE GNAME = 'AUFK' OR
                           GNAME = 'RKPF' OR
                           GNAME = 'RESB'.
      ENDLOOP.
      IF SY-SUBRC EQ 0.
        P_LOCK = 'X'.
      ELSE.
        CLEAR : P_LOCK.
      ENDIF.
    ELSE.
      CLEAR : P_LOCK.
    ENDIF.
  ENDIF.

ENDFORM. " ENQUEUE_READ
