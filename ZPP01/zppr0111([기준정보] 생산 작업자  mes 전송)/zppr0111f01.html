*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM SELECT_MAIN_DATA .

  CLEAR : GT_MAIN[], GT_MAIN.
  SELECT A~WERKS A~KOKRS A~KOSTL B~PERNR B~SNAME AS PERNR_NM
    INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
                  FROM ZPPT0003 AS A
            INNER JOIN PA0001   AS B
                    ON A~KOKRS = B~KOKRS
                   AND A~KOSTL = B~KOSTL
                 WHERE A~WERKS IN SO_WERKS
                   AND A~KOKRS IN SO_KOKRS
                   AND A~KOSTL IN SO_KOSTL
                   AND B~SNAME IN SO_SNAME
                   AND B~BEGDA LE SY-DATLO
                   AND B~ENDDA GE SY-DATLO.

*퇴직및 생산부서에서 타부서 이동자
  CLEAR : GT_DELETE[], GT_DELETE.
  SELECT C~WERKS A~KOKRS A~KOSTL A~PERNR A~SNAME AS PERNR_NM A~ENDDA
    INTO CORRESPONDING FIELDS OF TABLE GT_DELETE
                  FROM PA0001   AS A
            INNER JOIN ZPPT0004 AS B
                    ON A~PERNR = B~PERNR
       LEFT OUTER JOIN ZPPT0003 AS C
                    ON A~KOKRS = C~KOKRS
                   AND A~KOSTL = C~KOSTL
                 WHERE A~KOKRS IN SO_KOKRS
                   AND A~KOSTL IN SO_KOSTL
                   AND A~SNAME IN SO_SNAME
                   AND A~ENDDA < SY-DATLO
                   AND B~CHANGED_DATE IN SO_CHGDT.
  IF SY-SUBRC EQ 0.

    DELETE GT_DELETE WHERE WERKS NOT IN SO_WERKS.
    SORT GT_DELETE DESCENDING BY PERNR ENDDA.
    DELETE ADJACENT DUPLICATES FROM GT_DELETE COMPARING PERNR ENDDA.

    LOOP AT GT_DELETE.
      READ TABLE GT_MAIN WITH KEY PERNR = GT_DELETE-PERNR.
      IF SY-SUBRC NE 0.
        MOVE-CORRESPONDING GT_DELETE TO GT_MAIN.
        APPEND GT_MAIN. CLEAR : GT_MAIN.
        GT_DELETE-DEL_F = C_X.
        MODIFY GT_DELETE. CLEAR : GT_DELETE.
      ENDIF.
    ENDLOOP.

    SORT GT_DELETE BY PERNR DEL_F.

  ENDIF.

ENDFORM. " SELECT_MAIN_DATA

FORM SELECT_ETC_DATA .

*관리회계영역 정보
  CLEAR : GT_KOKRS[], GT_KOKRS.
  SELECT KOKRS BEZEI WAERS
    INTO CORRESPONDING FIELDS OF TABLE GT_KOKRS
                  FROM TKA01
                 WHERE KOKRS IN SO_KOKRS.
  IF SY-SUBRC EQ 0.
    SORT GT_KOKRS BY KOKRS.
  ENDIF.

  CLEAR : GT_CSK[],   GT_CSK,
          GT_T0004[], GT_T0004.
  IF GT_MAIN[] IS NOT INITIAL.

*코스트센터 정보
    SELECT KOKRS KOSTL KTEXT
      INTO CORRESPONDING FIELDS OF TABLE GT_CSK
                    FROM CSKT
      FOR ALL ENTRIES IN GT_MAIN
                   WHERE KOKRS = GT_MAIN-KOKRS
                     AND KOSTL = GT_MAIN-KOSTL
                     AND DATBI GE SY-DATUM.
    IF SY-SUBRC EQ 0.
      SORT GT_CSK BY KOKRS KOSTL.
    ENDIF.

* 생산 작업자 관리
    SELECT PERNR PRDCERNO UPRICE CREATED_DATE CREATED_TIME
           CREATED_BY
      INTO CORRESPONDING FIELDS OF TABLE GT_T0004
                    FROM ZPPT0004
      FOR ALL ENTRIES IN GT_MAIN
                   WHERE PERNR = GT_MAIN-PERNR.
    IF SY-SUBRC EQ 0.
      SORT GT_T0004 BY PERNR.
    ENDIF.

  ENDIF.

FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20).

  SORT GT_MAIN BY WERKS KOKRS KOSTL PERNR.
  CLEAR : GT_RESULT[], GT_RESULT.
  LOOP AT GT_MAIN.

*관리회계 영역 내역 / 통화
    READ TABLE GT_KOKRS WITH KEY KOKRS = GT_MAIN-KOKRS
                        BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GT_MAIN-KOKRS_NM = GT_KOKRS-BEZEI.
      GT_MAIN-WAERS    = GT_KOKRS-WAERS.
    ENDIF.

*코스트센터내역
    READ TABLE GT_CSK WITH KEY  KOKRS = GT_MAIN-KOKRS
                                KOSTL = GT_MAIN-KOSTL
                      BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GT_MAIN-KOSTL_NM = GT_CSK-KTEXT.
    ENDIF.

*생산번호 / 단가
    READ TABLE GT_T0004 WITH KEY PERNR = GT_MAIN-PERNR
                        BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GT_MAIN-PRDCERNO  = GT_T0004-PRDCERNO.
      GT_MAIN-UPRICE    = GT_T0004-UPRICE.
      GT_MAIN-ICON      = ICON_LED_GREEN.
    ENDIF.

*삭제대상 아이콘
    READ TABLE GT_DELETE WITH KEY PERNR = GT_MAIN-PERNR
                                  DEL_F = C_X
                         BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GT_MAIN-ICON      = ICON_LED_RED.
    ENDIF.

    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
    APPEND GT_RESULT. CLEAR : GT_RESULT.

  ENDLOOP.

  IF GT_RESULT[] IS INITIAL.
    MESSAGE S009.
    LEAVE LIST-PROCESSING.
  ENDIF.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S021 WITH LV_SCNT.

ENDFORM. " MAKE_DATA

          include structure ZPPS0030.
  DATA : END OF LT_TAB.

  DATA : LT_T041 LIKE ZPPT0041 OCCURS 0 WITH HEADER LINE.

         FROM T001W
         WHERE WERKS IN SO_WERKS.

                                          PERNR = LT_T041-PERNR AND

        LT_T041-CREATED_BY   = SY-UNAME.
        APPEND LT_T041.

        LV_TCNT = LV_TCNT + 1.
      ELSE.
        DELETE DT_T041 INDEX LV_TABIX.
      ENDIF.
    ENDLOOP.

    CALL FUNCTION 'ZPPF02_PP_PPMES_RFC_003'
      DESTINATION 'ZPP_MES'
      EXPORTING
        I_WERKS               = LT_WERKS-WERKS
      IMPORTING
        E_RET                 = LT_T041-E_RET
        E_MSG                 = LT_T041-E_MSG

        COMMUNICATION_FAILURE = 2
