FUNCTION ZPPF03_SAVE_BOM_INQUIRY_LOG.
*"----------------------------------------------------------------------
*"*"Local interface:
*" TABLES
*" IT_LOG STRUCTURE ZPPS0001
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LT_T0002 TYPE TABLE OF ZPPT0002 WITH HEADER LINE,
         LS_LOG   TYPE ZPPS0001.

  DATA : LV_LOGID    TYPE ZPPT0002-LOGID,
         LV_SEQ      TYPE ZPPT0002-SEQ,
         LV_TERMINAL TYPE ZPPT0002-TERMINAL.

  PERFORM GET_TERMINAL CHANGING LV_TERMINAL.
  GET TIME STAMP FIELD LV_LOGID.

  CLEAR : LV_SEQ, LT_T0002[], LT_T0002, LS_LOG.
  LOOP AT IT_LOG INTO LS_LOG.
    ADD 1 TO LV_SEQ.
    MOVE-CORRESPONDING LS_LOG TO LT_T0002.
    LT_T0002-LOGID         = LV_LOGID.
    LT_T0002-SEQ           = LV_SEQ.
    LT_T0002-TERMINAL      = LV_TERMINAL.
    LT_T0002-CREATED_DATE  = SY-DATUM.
    LT_T0002-CREATED_TIME  = SY-UZEIT.
    LT_T0002-CREATED_BY    = SY-UNAME.
    LT_T0002-CHANGED_DATE  = SY-DATUM.
    LT_T0002-CHANGED_TIME  = SY-UZEIT.
    LT_T0002-CHANGED_BY    = SY-UNAME.
    APPEND LT_T0002. CLEAR : LT_T0002.

  ENDLOOP.

  IF LT_T0002[] IS NOT INITIAL.
    MODIFY ZPPT0002 FROM TABLE LT_T0002.
  ENDIF.

ENDFUNCTION.
