                      ON B~MATNR = C~MATNR
                      INNER JOIN MARC AS D
                            ON A~WERKS = D~WERKS
                            AND A~MATNR = D~MATNR
                            INNER JOIN MARA AS E   "공정구분은 생산품 기준으로
                                  ON A~BAUGR = E~MATNR
           WHERE A~WERKS = P_WERKS
             AND A~MATNR IN S_MATNR
             AND A~BDTER IN S_BDTER
             AND A~KZEAR <> 'X' "해당 예약에 대한 최종출고

             AND B~MTART IN S_MTART1 "제외건
             AND C~SPRAS = SY-LANGU
             AND D~DISPO IN S_DISPO.
  ELSE.
    SELECT A~WERKS A~MATNR "B~ZZPART
           B~MTART C~MAKTX A~MEINS A~BDMNG
           A~SHKZG D~BESKZ D~SOBSL  D~EISBE D~PLIFZ D~WEBAZ A~BDTER
           A~BAUGR A~PLNUM D~DISPO
           INTO CORRESPONDING FIELDS OF TABLE LT_RESB
           FROM MDSM AS A INNER JOIN MARA AS B
                ON A~MATNR = B~MATNR
                INNER JOIN MAKT AS C
                      ON B~MATNR = C~MATNR
                      INNER JOIN MARC AS D
                            ON A~WERKS = D~WERKS
                            AND A~MATNR = D~MATNR
           WHERE A~PLSCN = P_PLSCN
             AND A~KZKUP NE 'X'
             AND A~WERKS = P_WERKS
             AND A~MATNR IN S_MATNR
             AND A~BDTER IN S_BDTER
             AND B~MTART IN S_MTART
             AND B~MTART IN S_MTART1 "제외건
             AND C~SPRAS = SY-LANGU
             AND D~DISPO IN S_DISPO.
*             AND B~ZZPART IN S_ZPART.

  CLEAR GT_MAR[].
  LOOP AT LT_RESB.
    CLEAR GT_RESB.
    MOVE-CORRESPONDING LT_RESB TO GT_RESB.

            GT_RESB-PLNUM,
            GT_RESB-AUFNR,
            GT_RESB-PLIFZ,
            GT_RESB-WEBAZ.

FORM DISPLAY_ADD .

  DATA : L_INDEX TYPE SY-TABIX.
  DATA : L_MENGE TYPE ETMEN.

  DATA : BEGIN OF LT_MARD OCCURS 0,
           MATNR TYPE MATNR,
           WERKS TYPE WERKS_D,
           LGORT TYPE LGORT_D,
           LABST TYPE LABST,
           INSME TYPE INSME,
         END OF LT_MARD.

  DATA : BEGIN OF LT_VJAUFK OCCURS 0,
           AUFNR TYPE AUFNR,
           OBJNR TYPE J_OBJNR,
           STAT  TYPE J_STATUS,
           POSNR TYPE CO_POSNR,
           WERKS TYPE WERKS_D,
           MATNR TYPE CO_MATNR,
         END OF LT_VJAUFK.

  DATA : BEGIN OF LT_MCHB OCCURS 0,
          WERKS LIKE MCHB-WERKS,
          MATNR LIKE MCHB-MATNR,
          CINSM LIKE MCHB-CINSM,
         END OF LT_MCHB.

  RANGES : S_STAT FOR JEST-STAT.

*계획오더 조건용.
  DATA : LT_RESB LIKE TABLE OF GT_RESB WITH HEADER LINE,
         LT_MCHB_T LIKE LT_MCHB OCCURS 0 WITH HEADER LINE.

**** MAKE RANGE
  DEFINE APPEND_S_STAT.
    MOVE : &1 TO S_STAT-LOW.
    APPEND S_STAT.
  END-OF-DEFINITION.
  MOVE : 'I' TO S_STAT-SIGN,
         'EQ' TO S_STAT-OPTION.
  APPEND_S_STAT 'I0076'.
  APPEND_S_STAT 'I0012'.
  APPEND_S_STAT 'I0046'.
  APPEND_S_STAT 'I0056'.
  APPEND_S_STAT 'I0045'.

  CLEAR GT_EBAN[].
  SELECT BANFN BNFPO WERKS MATNR MENGE FIXKZ
         MEINS BADAT LFDAT
         INTO CORRESPONDING FIELDS OF TABLE GT_EBAN
         FROM EBAN
         FOR ALL ENTRIES IN GT_RESB
         WHERE WERKS = GT_RESB-WERKS
           AND MATNR = GT_RESB-MATNR
           AND KNTTP = ' '
           AND EBELN = ' '
           AND LOEKZ = ' '.

  CLEAR GT_EKPO[].
  SELECT A~EBELN A~EBELP A~WERKS A~MATNR A~LOEKZ
         B~MENGE B~WEMNG A~MEINS B~EINDT A~UMREZ
         A~UMREN
         INTO CORRESPONDING FIELDS OF TABLE GT_EKPO
         FROM EKPO AS A INNER JOIN EKET AS B
              ON  A~EBELN = B~EBELN
              AND A~EBELP = B~EBELP
         FOR ALL ENTRIES IN GT_RESB
         WHERE A~WERKS = GT_RESB-WERKS
           AND A~MATNR = GT_RESB-MATNR
           AND A~KNTTP = ' '
           AND A~ELIKZ = ' '. "납품완료
  DELETE GT_EKPO WHERE LOEKZ = 'L'. "삭제표시

  SELECT MATNR WERKS LGORT LABST INSME
         INTO CORRESPONDING FIELDS OF TABLE LT_MARD
         FROM MARD
         FOR ALL ENTRIES IN GT_RESB
         WHERE WERKS = GT_RESB-WERKS
           AND MATNR = GT_RESB-MATNR.

  CLEAR GT_AFPO[].
  SELECT AUFNR POSNR PWERK AS WERKS MATNR PSMNG WEMNG
         AMEIN DGLTP
         INTO CORRESPONDING FIELDS OF TABLE GT_AFPO
         FROM AFPO
         FOR ALL ENTRIES IN GT_RESB
         WHERE PWERK = GT_RESB-WERKS
           AND MATNR = GT_RESB-MATNR
           AND DAUAT IN ('ZP01', 'ZP02', 'ZP03', 'ZP04', 'ZP05',
                         'ZP06', 'ZP07', 'ZP08', 'ZP09').
  SORT GT_AFPO BY AUFNR POSNR.
  IF GT_AFPO[] IS NOT INITIAL.
    SELECT AUFNR OBJNR STAT POSNR WERKS MATNR
           INTO CORRESPONDING FIELDS OF TABLE LT_VJAUFK
           FROM VJAUFK
           FOR ALL ENTRIES IN GT_AFPO
           WHERE AUFNR = GT_AFPO-AUFNR
             AND POSNR = GT_AFPO-POSNR
             AND STAT  IN S_STAT.
    SORT LT_VJAUFK BY AUFNR POSNR.
    CLEAR L_INDEX.
    LOOP AT GT_AFPO.
      ADD 1 TO L_INDEX.
      READ TABLE LT_VJAUFK WITH KEY AUFNR = GT_AFPO-AUFNR
                                    POSNR = GT_AFPO-POSNR.
      IF SY-SUBRC = 0.
        DELETE GT_AFPO INDEX L_INDEX.
        L_INDEX = L_INDEX - 1.
      ENDIF.
    ENDLOOP.
  ENDIF.

*계획오더
  CLEAR : LT_RESB[], LT_RESB.
  LT_RESB[] = GT_RESB[].
  SORT LT_RESB BY WERKS MATNR.
  DELETE ADJACENT DUPLICATES FROM LT_RESB COMPARING WERKS MATNR.

  CLEAR : GT_PLAF[], GT_PLAF.
  IF LT_RESB[] IS NOT INITIAL.
    SELECT PLNUM PLWRK AS WERKS MATNR GSMNG MEINS
           PSTTR PEDTR
      INTO CORRESPONDING FIELDS OF TABLE GT_PLAF
                    FROM PLAF
      FOR ALL ENTRIES IN LT_RESB
                   WHERE PLWRK = LT_RESB-WERKS
                     AND MATNR = LT_RESB-MATNR
                     AND PEDTR IN S_BDTER
                     AND PLSCN = '000'.
  ENDIF.

  SORT : GT_EBAN BY WERKS MATNR,
         GT_EKPO BY WERKS MATNR,
         LT_MARD BY WERKS MATNR,
         GT_PLAF BY WERKS MATNR.

  CLEAR GT_DISP.
  LOOP AT GT_RESB.
    CLEAR GT_DISP.
    MOVE-CORRESPONDING GT_RESB TO GT_DISP.

* 안전재고
    CLEAR GT_MAR.
    READ TABLE GT_MAR WITH KEY WERKS = GT_RESB-WERKS
                                MATNR = GT_RESB-MATNR.
    MOVE : GT_MAR-EISBE TO GT_DISP-EISBE,
           GT_MAR-PLIFZ TO GT_DISP-PLIFZ,
           GT_MAR-WEBAZ TO GT_DISP-WEBAZ.

* 소요량
    GT_DISP-BDMNG = GT_RESB-BDMNG - GT_RESB-ENMNG.

* 계획오더수량
    LOOP AT GT_PLAF WHERE WERKS = GT_DISP-WERKS
                      AND MATNR = GT_DISP-MATNR.
      GT_DISP-GSMNG = GT_DISP-GSMNG + GT_PLAF-GSMNG.
    ENDLOOP.


* 구매요청수량
    LOOP AT GT_EBAN WHERE WERKS = GT_DISP-WERKS
                      AND MATNR = GT_DISP-MATNR.
      IF GT_EBAN-FIXKZ IS INITIAL.
        GT_DISP-MENGE = GT_DISP-MENGE + GT_EBAN-MENGE.
      ELSE.
        GT_DISP-FMENGE = GT_DISP-FMENGE + GT_EBAN-MENGE.
      ENDIF.
    ENDLOOP.

* 구매오더수량
    LOOP AT GT_EKPO WHERE WERKS = GT_DISP-WERKS
                      AND MATNR = GT_DISP-MATNR.
      CLEAR L_MENGE.
      L_MENGE = GT_EKPO-MENGE - GT_EKPO-WEMNG.

* 자재기본단위와 발주단위가 다를때 환산하기 위한 값 umrez/UMREN
      L_MENGE = L_MENGE * GT_EKPO-UMREZ / GT_EKPO-UMREN.
      GT_DISP-KTMNG = GT_DISP-KTMNG + L_MENGE.
    ENDLOOP.

* MARD 수량
    LOOP AT LT_MARD WHERE WERKS = GT_DISP-WERKS
                      AND MATNR = GT_DISP-MATNR.
      GT_DISP-LABST = GT_DISP-LABST + LT_MARD-LABST.
      GT_DISP-CINSM = GT_DISP-CINSM + LT_MARD-INSME.
    ENDLOOP.
*검사수량
    READ TABLE LT_MCHB WITH KEY WERKS = GT_DISP-WERKS
                                MATNR = GT_DISP-MATNR
                                BINARY SEARCH.
    IF SY-SUBRC = 0.
      GT_DISP-CINSM = LT_MCHB-CINSM.
    ENDIF.
* 생산오더 수량
    LOOP AT GT_AFPO WHERE WERKS = GT_DISP-WERKS
                      AND MATNR = GT_DISP-MATNR.
      GT_DISP-PSMNG = GT_DISP-PSMNG + GT_AFPO-PSMNG - GT_AFPO-WEMNG.
    ENDLOOP.
*    GT_DISP-LSMNG = GT_DISP-LABST + GT_DISP-MENGE + GT_DISP-KTMNG +
*                    GT_DISP-PSMNG + GT_DISP-B_BDMNG + GT_DISP-GSMNG
*                    - GT_DISP-BDMNG - GT_DISP-EISBE.
    GT_DISP-LSMNG = GT_DISP-LABST + GT_DISP-MENGE  + GT_DISP-FMENGE + GT_DISP-KTMNG +
                    GT_DISP-PSMNG + GT_DISP-GSMNG
                    - GT_DISP-BDMNG - GT_DISP-EISBE.
    PERFORM APPEND_FCOL USING GT_DISP-FCOL.
    APPEND GT_DISP.
  ENDLOOP.

  SORT GT_DISP BY WERKS MATNR.

ENDFORM. " DISPLAY_ADD
