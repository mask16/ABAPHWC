FORM DOUBLE_CLICK USING P_ROW TYPE ANY P_COLUMN TYPE ANY.
  DATA : L_COL(40).
  FIELD-SYMBOLS <FS>.

* 자재 및 오더번호에 대해서 DOUBLE CLICK한 필드의 값을 읽어와서
* 해당 T-CODE를 CALL한다.

  CHECK SY-SUBRC = 0.

  CHECK NOT <FS> IS INITIAL.

  CASE P_COLUMN.
    WHEN 'MATNR'.
      SET PARAMETER ID 'MAT'  FIELD <FS>.
      SET PARAMETER ID 'MXX'  FIELD 'K'.
      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.
  ENDCASE.

ENDFORM. " 8000_DOUBLE_CLICK

*----------------------------------------------------------------------*

FORM BLPK_MSEG_SELECT .

* 실적은 BLPK, BLPP, MSEG를 기준으로 한다.

  DATA : L_MJAHR TYPE MJAHR.
  DATA : BEGIN OF LT_BLPK OCCURS 0,
           WERKS TYPE WERKS_D,
           MATNR TYPE MATNR,
           MAKTX TYPE MAKTX,
           MTART TYPE MTART,
           BUDAT TYPE BUDAT,
           BELNR TYPE BELNR_D,
           MJAHR TYPE MJAHR,
           ZEILE TYPE MBLPO,
           BWART TYPE BWART,
           MCOMP TYPE MATNR,
           MENGE TYPE MENGE_D,
           MEINS TYPE MEINS,
           MTART1 TYPE MTART,
           MATKL TYPE MATKL,
         END OF LT_BLPK.
  DATA : BEGIN OF LT_MAKT OCCURS 0,
          MATNR TYPE MATNR,
          MAKTX TYPE MAKTX,
         END OF LT_MAKT.


  L_MJAHR = S_SPMON-LOW+0(4).
  SELECT A~WERKS A~MATNR B~MAKTX D~MTART A~BUDAT E~BELNR
         M~MJAHR M~ZEILE M~BWART M~MATNR AS MCOMP M~MENGE M~MEINS
         M1~MTART AS MTART1
         M1~MATKL
         INTO TABLE LT_BLPK
         FROM BLPK AS A INNER JOIN MAKT AS B
              ON A~MATNR = B~MATNR
                 INNER JOIN MARA AS D
                   ON A~MATNR = D~MATNR
                   INNER JOIN BLPP AS E
                      ON A~PRTNR = E~PRTNR
                      INNER JOIN MSEG AS M
                         ON E~BELNR = M~MBLNR
                         INNER JOIN MARA AS M1
                            ON M~MATNR = M1~MATNR
                            INNER JOIN MARC AS M2
                               ON M~MATNR = M2~MATNR AND
                                  M~WERKS = M2~WERKS
         WHERE A~WERKS = P_WERKS
           AND A~BUDAT IN S_DATE
           AND B~SPRAS = SY-LANGU
           AND E~BELNR <> ''
           AND M~MJAHR = L_MJAHR
           AND M~BWART IN S_BWART
           AND M~MATNR IN S_MATNR
           AND M1~MATKL IN S_MATKL
           AND M1~MTART IN S_MTART
           AND M2~DISPO IN S_DISPO.
  CHECK LT_BLPK[] IS NOT INITIAL.


  SELECT MATNR MAKTX
         INTO TABLE LT_MAKT
         FROM MAKT
         FOR ALL ENTRIES IN LT_BLPK
         WHERE MATNR = LT_BLPK-MCOMP
           AND SPRAS = SY-LANGU.

*

  LOOP AT LT_BLPK WHERE MCOMP IN S_MATNR
                    AND MTART1 IN S_MTART
                    AND MATKL IN S_MATKL.
    IF LT_BLPK-BWART = '262' OR LT_BLPK-BWART = '532'.
      LT_BLPK-MENGE = LT_BLPK-MENGE * -1.
    ENDIF.
    CLEAR GT_BLPK.
    MOVE : LT_BLPK-WERKS  TO GT_BLPK-WERKS,
           LT_BLPK-MATNR  TO GT_BLPK-MATNR,
           LT_BLPK-MAKTX  TO GT_BLPK-MAKTX,
           LT_BLPK-MTART  TO GT_BLPK-MTART,
           LT_BLPK-BUDAT  TO GT_BLPK-BUDAT,
           LT_BLPK-MCOMP  TO GT_BLPK-MCOMP,
           LT_BLPK-MTART1 TO GT_BLPK-MTART1,
           LT_BLPK-MENGE  TO GT_BLPK-MENGE,
           LT_BLPK-MEINS  TO GT_BLPK-MEINS.
    CLEAR LT_MAKT.
    READ TABLE LT_MAKT WITH KEY MATNR = GT_BLPK-MCOMP.
    MOVE : LT_MAKT-MAKTX TO GT_BLPK-MAKTX1.
    COLLECT GT_BLPK.
  ENDLOOP.

ENDFORM. " BLPK_MSEG_SELECT

FORM SELECT_AUFM_AFPO .

  DATA : BEGIN OF LT_AUFM OCCURS 0,
           AUFNR LIKE AFPO-AUFNR,
           WERKS LIKE AFPO-PWERK,
           MATNR LIKE AFPO-MATNR,
           MAKTX LIKE MAKT-MAKTX,
           MTART LIKE MARA-MTART,
           MBLNR LIKE AUFM-MBLNR,
           MJAHR LIKE AUFM-MJAHR,
           ZEILE LIKE AUFM-ZEILE,
           BUDAT LIKE AUFM-BUDAT,
           BWART LIKE AUFM-BWART,
           MCOMP LIKE AUFM-MATNR,
           MENGE LIKE AUFM-MENGE,
           MEINS LIKE AUFM-MEINS,
           MAKTX1 LIKE MAKT-MAKTX,
           MTART1 LIKE MARA-MTART,
         END OF LT_AUFM.
  DATA : BEGIN OF LT_MARA OCCURS 0,
           MATNR TYPE MATNR,
           MAKTX TYPE MAKTX,
           MTART TYPE MTART,
         END OF LT_MARA.

  CLEAR LT_AUFM[].
  SELECT A~AUFNR A~PWERK AS WERKS A~MATNR B~MAKTX C~MTART
         F~MBLNR F~MJAHR F~ZEILE F~BUDAT F~BWART
         F~MATNR AS MCOMP F~MENGE F~MEINS
         INTO TABLE LT_AUFM
         FROM AFPO AS A INNER JOIN MAKT AS B
                   ON A~MATNR = B~MATNR
                      INNER JOIN MARA AS C
                            ON A~MATNR = C~MATNR
                         INNER JOIN MARC AS D
                            ON A~MATNR = D~MATNR AND
                               A~PWERK = D~WERKS
                               INNER JOIN AUFM AS F
                                     ON A~AUFNR = F~AUFNR
         WHERE A~PWERK = P_WERKS
           AND F~MATNR IN S_MATNR
           AND B~SPRAS = SY-LANGU
           AND C~MATKL IN S_MATKL
           AND C~MTART IN S_MTART
           AND D~DISPO IN S_DISPO
           AND F~BUDAT IN S_DATE
           AND F~BWART IN S_BWART.

  CHECK LT_AUFM[] IS NOT INITIAL.

  SELECT A~MATNR B~MAKTX A~MTART
