*       text
*----------------------------------------------------------------------*
*      -->P_SY_TCODE  text
*      -->P_SY_UNAME  text
*      -->P_RC29L_MATNR  text
*      -->P_RC29L_WERKS  text
*      -->P_LV_RET  text

FORM PP_BOM_LOG USING P_MATNR P_WERKS P_RET.

  CASE SY-TCODE.
    WHEN 'CS11' OR 'CS12' OR 'CS13' OR 'CS15'.

      DATA : LT_S0001 TYPE TABLE OF ZPPS0001,
             LS_S0001 LIKE LINE  OF LT_S0001,
             LV_STLAN TYPE MAST-STLAN,
             LV_USNAM TYPE USNAM.

      SELECT SINGLE STLAN INTO LV_STLAN FROM MAST
             WHERE MATNR = P_MATNR AND
                   WERKS = P_WERKS.

      IF LV_STLAN = 'W' OR LV_STLAN = 'X' OR LV_STLAN = 'Y' OR LV_STLAN = 'Z'.

        SELECT SINGLE USNAM INTO LV_USNAM FROM ZPPT0130
               WHERE USNAM = SY-UNAME AND
                     STLAN = LV_STLAN.
        IF SY-SUBRC <> 0.
          MESSAGE S032(ZPPM) DISPLAY LIKE 'E'.
          P_RET = 'E'.
          EXIT.
        ENDIF.

        CLEAR : LS_S0001.
        MOVE : SY-DATUM      TO LS_S0001-SDATE,
               SY-UZEIT      TO LS_S0001-STIME,
               SY-UNAME      TO LS_S0001-UNAME,
               LV_STLAN      TO LS_S0001-STLAN,
               P_WERKS   TO LS_S0001-WERKS,
               P_MATNR   TO LS_S0001-MATNR,
               SY-TCODE      TO LS_S0001-TCODE.
        APPEND LS_S0001 TO  LT_S0001. CLEAR : LS_S0001.

        call function 'ZPPF03_SAVE_BOM_INQUIRY_LOG'
          TABLES
            IT_LOG = LT_S0001.

      ENDIF.

    WHEN OTHERS.

  ENDCASE.

ENDFORM. " PP_BOM_LOG
