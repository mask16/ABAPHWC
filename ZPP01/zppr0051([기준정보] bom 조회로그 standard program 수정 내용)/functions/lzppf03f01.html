FORM GET_TERMINAL CHANGING P_TERMINAL.

  DATA : LS_USR41 TYPE USR41.

  DATA : LV_OPCODE_USR_ATTR(1) TYPE X VALUE 5,
         LV_TERMINAL LIKE USR41-TERMINAL,
         LV_TID                 LIKE USR41-TERMID,
         LV_HOST                LIKE USR41-SERVER,
         LV_HOSTADDR            LIKE MSXXLIST-HOSTADR,
         LV_ACT_SESSIONS        LIKE SM04DIC-COUNTER,
         LV_MAX_SESSIONS        LIKE SM04DIC-COUNTER,
         LV_MY_SESSION          LIKE SM04DIC-COUNTER,
         LV_MY_INTERNAL_SESSION LIKE SM04DIC-COUNTER,
         LV_MY_SERVERNAME       LIKE MSXXLIST-NAME.

  CALL 'ThUsrInfo' ID 'OPCODE'              FIELD LV_OPCODE_USR_ATTR
                   ID 'TERMINAL'            FIELD LV_TERMINAL
                   ID 'TID'                 FIELD LV_TID
                   ID 'HOST'                FIELD LV_HOST
                   ID 'HOSTADDR'            FIELD LV_HOSTADDR
                   ID 'ACT_SESSIONS'        FIELD LV_ACT_SESSIONS
                   ID 'MAX_SESSIONS'        FIELD LV_MAX_SESSIONS
                   ID 'MY_SESSION'          FIELD LV_MY_SESSION
                   ID 'MY_INTERNAL_SESSION' FIELD LV_MY_INTERNAL_SESSION.

  CALL 'C_SAPGPARAM'   ID 'NAME' FIELD 'rdisp/myname'
                       ID 'VALUE' FIELD LV_MY_SERVERNAME.

  CLEAR: LS_USR41.
  SELECT SINGLE TERMINAL SERVER
    INTO (LS_USR41-TERMINAL, LS_USR41-SERVER)
    FROM USR41
   WHERE BNAME  = SY-UNAME
     AND TERMID = LV_TID
     AND SERVER = LV_MY_SERVERNAME.
  IF SY-SUBRC EQ 0.
    P_TERMINAL = LS_USR41-TERMINAL.
  ENDIF.

ENDFORM. " GET_TERMINAL
