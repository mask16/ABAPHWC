*----------------------------------------------------------------------*
FORM select_main_data .

  DATA: tmp_data LIKE TABLE OF gt_main,
        wa_data LIKE gt_main.

  DATA: t_resb TYPE TABLE OF resb,
        s_resb TYPE resb.

  CLEAR : gt_main[], gt_main.
*  CLEAR : gt_main[], gt_main.

*----------------------------------------------------------------------*
* RESB TABLE.
* 오더번호  , BOM품목번호,  구성부품소요수량
* RESB.AUFNR, RESB.POSNR ,  RESB.BDMNG

* MARA TABLE.
* 자재       , 자재내역
* MAKT.MATNR , MAKT.MAKTX

* MSEG TABLE.
* 출고수량    ,기본단위    ,현지통화금액  ,저장위치    ,이동유형    ,전기일자        ,트랜잭션코드      ,자재전표번호
* MSEG.MENGE  ,MSEG.MEINS  ,MSEG.DMBTR    ,MSEG.LGORT  ,MSEG.BWART  ,MSEG.BUDAT_MKPF ,MSEG.TCODE2_MKPF  ,MSEG.MBLNR
*----------------------------------------------------------------------*

  SELECT a~werks a~aufnr a~matnr SUM( a~menge ) AS menge a~meins a~lgort a~bwart a~budat_mkpf a~tcode2_mkpf b~zmaktx
    INTO CORRESPONDING FIELDS OF TABLE gt_main
    FROM mseg AS a
         INNER JOIN mara AS b
            ON a~matnr = b~matnr
    WHERE a~werks IN so_werks
          AND a~aufnr IN so_bismt
          AND a~budat_mkpf IN so_ersda
    GROUP BY a~werks a~aufnr a~matnr a~meins a~lgort a~bwart a~budat_mkpf a~tcode2_mkpf b~zmaktx
    ORDER BY a~matnr.

  IF sy-subrc = 0.
    tmp_data[] = gt_main[].

    SORT tmp_data BY aufnr werks.

    DELETE ADJACENT DUPLICATES FROM tmp_data COMPARING aufnr werks.

    SELECT aufnr werks matnr bdmng
    INTO CORRESPONDING FIELDS OF TABLE t_resb
    FROM resb
    FOR ALL ENTRIES IN tmp_data
    WHERE aufnr = tmp_data-aufnr
      AND werks = tmp_data-werks.

    SORT gt_main BY aufnr werks matnr.

  ENDIF.

  LOOP AT t_resb INTO s_resb.
    READ TABLE gt_main WITH KEY aufnr = s_resb-aufnr
                                werks = s_resb-werks
                                matnr = s_resb-matnr
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_main-bdmng = s_resb-bdmng.

      MODIFY gt_main INDEX sy-tabix TRANSPORTING bdmng.

      CLEAR gt_main.

    ELSE.

      MOVE-CORRESPONDING s_resb TO gt_main.

      SELECT SINGLE meins zmaktx
        INTO (gt_main-meins, gt_main-zmaktx)
        FROM mara
        WHERE matnr = gt_main-matnr.

      APPEND gt_main.
      CLEAR gt_main.
      SORT gt_main BY aufnr werks matnr.

    ENDIF.
  ENDLOOP.
  SORT gt_main.

ENDFORM. " SELECT_MAIN_DATA
