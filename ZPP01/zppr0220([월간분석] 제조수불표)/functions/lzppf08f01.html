FORM MAKE_REM USING P_WERKS.

DATA : BEGIN OF LT_AUFM OCCURS 0,
         WERKS TYPE WERKS_D,
         AUFNR TYPE AUFNR,
         AUART TYPE AUART,
         PLNBEZ TYPE MATNR,
         MBLNR TYPE MBLNR,
         MJAHR TYPE MJAHR,
         ZEILE TYPE MBLPO,
         BUDAT TYPE BUDAT,
         BWART TYPE BWART,
         MATNR TYPE MATNR,
         MENGE TYPE MENGE_D,
         MEINS TYPE MEINS,
         LGORT TYPE LGORT_D,
       END OF LT_AUFM.

 SELECT A~WERKS A~AUFNR A~AUART B~PLNBEZ C~MBLNR C~MJAHR C~ZEILE C~BUDAT
        C~BWART C~MATNR C~MENGE C~MEINS
        INTO TABLE LT_AUFM
        FROM AUFK AS A INNER JOIN AFKO AS B
             ON A~AUFNR = B~AUFNR
             INNER JOIN AUFM AS C
                   ON A~WERKS = C~WERKS
                   AND A~AUFNR = C~AUFNR
        WHERE A~WERKS = P_WERKS
          AND A~AUART = 'ZP03'
          AND C~BUDAT IN S_BUDAT.

 CLEAR GT_AUFM[].
  LOOP AT LT_AUFM.
     CLEAR GT_AUFM.
     MOVE-CORRESPONDING LT_AUFM TO GT_AUFM.
     IF LT_AUFM-BWART = '262'.
        GT_AUFM-MENGE = GT_AUFM-MENGE * -1.
     ENDIF.
     COLLECT GT_AUFM.
  ENDLOOP.


* CHECK AUFNR

  PERFORM CHECK_AUFNR.

ENDFORM. " MAKE_REM

FORM MAKE_ORDER USING P_WERKS.


* 입력 받은 일자가 AFPO-LTRMI 또는 AUFK-IDAT2 일자보다 빠르면 재공이므로
* 데이터가 보여져야 하고, 느리면 데이터 제외 함

  CLEAR GT_AFPO[].
  SELECT A~DWERK AS WERKS A~AUFNR A~PSMNG A~WEMNG A~AMEIN
         C~MATNR D~MAKTX C~BDMNG C~MEINS F~OBJNR
         INTO TABLE GT_AFPO
         FROM AFPO AS A INNER JOIN AFKO AS B
                   ON A~AUFNR = B~AUFNR
                      INNER JOIN AUFK AS F
                            ON B~AUFNR = F~AUFNR
                      INNER JOIN RESB AS C
              ON  A~DWERK = C~WERKS
              AND A~AUFNR = C~AUFNR
                  INNER JOIN MAKT AS D
                        ON C~MATNR = D~MATNR
         WHERE A~DWERK = P_WERKS

*           AND ( A~LTRMI >= P_GSTRP OR A~LTRMI = '00000000' )

           AND C~BWART = '261'
           AND D~SPRAS = SY-LANGU.

  CHECK GT_AFPO[] IS NOT INITIAL.

* 삭제 DATA 체크

  PERFORM CHECK_DLFL.


* 일자체크

  PERFORM CHECK_DELI_DATE.


* AUFM DATA 선택

  CLEAR GT_AUFM2[].
  SELECT MBLNR MJAHR ZEILE BUDAT BWART MATNR WERKS
         AUFNR MENGE MEINS
         INTO TABLE GT_AUFM2
         FROM AUFM
         FOR ALL ENTRIES IN GT_AFPO
         WHERE AUFNR = GT_AFPO-AUFNR
           AND BUDAT IN S_BUDAT
           AND BWART IN ('261', '262').


* 재공 계산을 위한 SUB DATA MAKE

  LOOP AT GT_AFPO.

* 출고내역을 ASSIGN

    LOOP AT GT_AUFM2 WHERE AUFNR = GT_AFPO-AUFNR
                       AND WERKS = GT_AFPO-WERKS
                       AND MATNR = GT_AFPO-MATNR.
       IF GT_AUFM2-BWART = '262'.
            GT_AUFM2-MENGE = GT_AUFM2-MENGE * -1.
       ENDIF.
       GT_AFPO-MENGE = GT_AFPO-MENGE + GT_AUFM2-MENGE. "기투입량
    ENDLOOP.
    MODIFY GT_AFPO.
  ENDLOOP.


* 입고환산량, 재공량 계산
* 입고환산량 : 소요량/지시량*입고량
* 재공량 : 기투입량-입고환산량

  LOOP AT GT_AFPO.
    IF GT_AFPO-PSMNG IS NOT INITIAL.
       GT_AFPO-GR_H = ( GT_AFPO-BDMNG / GT_AFPO-PSMNG ) * GT_AFPO-WEMNG.
    ENDIF.
    GT_AFPO-WIP  = GT_AFPO-MENGE - GT_AFPO-GR_H.

* 재공량 '-'일때 '0' 처리

    IF GT_AFPO-WIP < 0.
       CLEAR GT_AFPO-WIP.
    ENDIF.
    MODIFY GT_AFPO.
  ENDLOOP.

ENDFORM. " MAKE_ORDER

FORM MAKE_DATE_RANGE USING P_SPMON
                           PS_T150 TYPE ZPPT0150.


* MAKE BUDAT RANGE.

 CLEAR S_BUDAT[].
 MOVE : 'I'  TO S_BUDAT-SIGN,
        'BT' TO S_BUDAT-OPTION.

 CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
        EXPORTING

       S_BUDAT-LOW = S_BUDAT-HIGH - 90.
       APPEND S_BUDAT.
 ENDCASE.

ENDFORM. " MAKE_DATE_RANGE

FORM CHECK_DELI_DATE .

 DATA : LT_JCDS LIKE JCDS OCCURS 0 WITH HEADER LINE.
 DATA : L_INDEX TYPE SY-TABIX.
 SELECT * INTO TABLE LT_JCDS
          FROM JCDS
          FOR ALL ENTRIES IN GT_AFPO
          WHERE OBJNR = GT_AFPO-OBJNR
            AND STAT = 'I0012'.
 SORT LT_JCDS BY OBJNR STAT ASCENDING
                 CHGNR DESCENDING.

 CLEAR L_INDEX.
 LOOP AT GT_AFPO.
   ADD 1 TO L_INDEX.
   CLEAR LT_JCDS.
   READ TABLE LT_JCDS WITH KEY OBJNR = GT_AFPO-OBJNR.
   IF SY-SUBRC <> 0.
   ELSE.
      IF LT_JCDS-INACT = '' AND LT_JCDS-UDATE <= S_BUDAT-HIGH.
         DELETE GT_AFPO INDEX L_INDEX.
         L_INDEX = L_INDEX - 1.
      ENDIF.
   ENDIF.
 ENDLOOP.

ENDFORM. " CHECK_DELI_DATE
