FORM DATA_SELECT .

  DATA : L_INDEX TYPE SY-TABIX,
         LV_SDATE TYPE SY-DATUM,
         LV_EDATE TYPE SY-DATUM.

  CLEAR : GT_PBIM[],
          GT_FIELD[].

  CONCATENATE S_SPMON-LOW '01' INTO LV_SDATE.
  IF S_SPMON-HIGH NE '000000'.
    CONCATENATE S_SPMON-HIGH '01' INTO LV_EDATE.
    PERFORM LAST_DAY_OF_MONTHS USING LV_EDATE
                               CHANGING LV_EDATE.
  ELSE.
    PERFORM LAST_DAY_OF_MONTHS USING LV_SDATE
                               CHANGING LV_EDATE.
  ENDIF.

  SELECT A~VERSB A~WERKS A~MATNR C~MAKTX A~BDZEI
         B~ENTLI B~PDATU B~PERXX B~PLNMG B~MEINS
         INTO CORRESPONDING FIELDS OF TABLE GT_PBIM
                       FROM PBIM AS A
                 INNER JOIN PBED AS B
                         ON A~BDZEI = B~BDZEI
                 INNER JOIN MARA AS D
                         ON A~MATNR = D~MATNR
                 INNER JOIN MAKT AS C
                       ON A~MATNR = C~MATNR
         WHERE A~VERSB = P_VERSB
           AND A~WERKS = P_WERKS
           AND A~LOEVR <> 'D'
           AND B~PDATU BETWEEN LV_SDATE AND LV_EDATE
           AND B~PLNMG > 0
           AND D~MATNR IN S_MATNR
           AND D~MTART IN S_MTART
           AND C~SPRAS = SY-LANGU.

  LOOP AT GT_PBIM.
    ADD 1 TO L_INDEX.
    CASE GT_PBIM-ENTLI.
      WHEN '1'. "일
        MOVE : GT_PBIM-PDATU TO GT_PBIM-SORTD.
      WHEN '3'. "월
        CONCATENATE GT_PBIM-PERXX '00' INTO GT_PBIM-SORTD.
    ENDCASE.
    MODIFY GT_PBIM INDEX L_INDEX.
    MOVE : GT_PBIM-SORTD TO GT_FIELD-FIELD.
    COLLECT GT_FIELD.
  ENDLOOP.

  SORT GT_PBIM BY MATNR SORTD.
ENDFORM. " DATA_SELECT
