FORM SELECT_MAIN_DATA .

  CLEAR : GT_MAIN[], GT_MAIN.
  SELECT B~WERKS A~MATNR A~MTART B~BESKZ B~SOBSL
         A~MEINS C~MAKTX A~BISMT B~XCHAR A~PSTAT
         B~PSTAT AS PSTATA
         A~MSTAE B~MMSTA
    INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
                  FROM MARA AS A
            INNER JOIN MARC AS B
                    ON A~MATNR = B~MATNR
            INNER JOIN MAKT AS C
                    ON A~MATNR = C~MATNR
                   AND C~SPRAS = SY-LANGU
                 WHERE A~MATNR IN SO_MATNR
                   AND A~MTART IN SO_MTART
                   AND B~WERKS IN SO_WERKS
                   AND A~ERSDA IN SO_ERSDA
                   AND B~BESKZ IN SO_BESKZ
                   AND A~LVORM <> C_X
                   AND B~LVORM <> C_X.

ENDFORM. " SELECT_MAIN_DATA

FORM SET_BOM USING P_MATNR
                     P_WERKS
                     P_STLAN
            CHANGING P_ICON.

  IF P_STLAN = 'Z'.

    DATA LV_DISPO TYPE DISPO.
    SELECT SINGLE DISPO INTO LV_DISPO FROM MARC
           WHERE MATNR = P_MATNR AND
                 WERKS = P_WERKS.
    IF LV_DISPO = '201' OR LV_DISPO = '202'  OR LV_DISPO = '203'  OR LV_DISPO = '204' .

      READ TABLE GT_MAST WITH KEY MATNR = P_MATNR
                                  WERKS = P_WERKS
                                  STLAN = 'W'
                                  BINARY SEARCH.
      IF SY-SUBRC = 0.
        P_ICON = ICON_LED_GREEN.
      ELSE.
        READ TABLE GT_MAST WITH KEY MATNR = P_MATNR
                                    WERKS = P_WERKS
                                    STLAN = 'X'
                                    BINARY SEARCH.
        IF SY-SUBRC = 0.
          P_ICON = ICON_LED_GREEN.
        ELSE.
          READ TABLE GT_MAST WITH KEY MATNR = P_MATNR
                                      WERKS = P_WERKS
                                      STLAN = 'Y'
                                      BINARY SEARCH.
          IF SY-SUBRC = 0.
            P_ICON = ICON_LED_GREEN.
          ELSE.
            READ TABLE GT_MAST WITH KEY MATNR = P_MATNR
                                        WERKS = P_WERKS
                                        STLAN = 'Z'
                                        BINARY SEARCH.
            IF SY-SUBRC = 0.
              P_ICON = ICON_LED_GREEN.
            ELSEIF SY-SUBRC <> 0 AND P_ICON = ICON_LIGHT_OUT.
              P_ICON = ICON_LED_RED.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

    ELSE.
      CLEAR P_ICON.
    ENDIF.

  ELSE.

    READ TABLE GT_MAST WITH KEY MATNR = P_MATNR
                                WERKS = P_WERKS
                                STLAN = P_STLAN
                       BINARY SEARCH.

    ENDIF.

  ENDIF.
