FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20).

  DATA : LT_STB    TYPE STPOX OCCURS 0 WITH HEADER LINE,
         LV_EMENG  TYPE BASMN VALUE 1,
         LS_CSTMAT LIKE CSTMAT.

  SORT GT_MAIN BY WERKS MATNR.
  CLEAR : GT_RESULT[], GT_RESULT.
  LOOP AT GT_MAIN.
    MOVE GT_MAIN-BMENG TO LV_EMENG.

    CLEAR : LT_STB[], LT_STB, LS_CSTMAT.
    CALL FUNCTION 'CS_BOM_EXPL_MAT_V2'
      EXPORTING
        CAPID                 = 'PP01'
        DATUV                 = SO_DATUV-LOW
        EMENG                 = LV_EMENG
        MKTLS                 = 'X'
*       MEHRS                 =
        MTNRV                 = GT_MAIN-MATNR
        STLAL                 = GT_MAIN-STLAL
        STLAN                 = GT_MAIN-STLAN
        WERKS                 = GT_MAIN-WERKS
      IMPORTING
        TOPMAT                = LS_CSTMAT
      TABLES
        STB                   = LT_STB
      EXCEPTIONS
        ALT_NOT_FOUND         = 1
        CALL_INVALID          = 2
        MATERIAL_NOT_FOUND    = 3
        MISSING_AUTHORIZATION = 4
        NO_BOM_FOUND          = 5
        NO_PLANT_DATA         = 6
        NO_SUITABLE_BOM_FOUND = 7
        CONVERSION_ERROR      = 8
        OTHERS                = 9.

    CHECK LT_STB[] IS NOT INITIAL.
    LOOP AT LT_STB.
      MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.

      MOVE : LS_CSTMAT-BMENG TO GT_RESULT-BMENG,   "기준수량
             LS_CSTMAT-BMEIN TO GT_RESULT-BMEIN,   "단위

             LT_STB-POSTP    TO GT_RESULT-POSTP,   "ICt
             LT_STB-POSNR    TO GT_RESULT-POSNR,   "항목
             LT_STB-MTART    TO GT_RESULT-MTART2,  "자재유형
             LT_STB-IDNRK    TO GT_RESULT-IDNRK,   "구성품
             LT_STB-OJTXP    TO GT_RESULT-MAKTX2,  "구성품내역
             LT_STB-MNGLG    TO GT_RESULT-MENGE,   "수량
             LT_STB-MEINS    TO GT_RESULT-MEINS,   "단위
             LT_STB-DATUV    TO GT_RESULT-DATUV2.  "효력시작일

      APPEND GT_RESULT. CLEAR : GT_RESULT.
    ENDLOOP.

  ENDLOOP.

*조회건수 DISPLAY

  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
