FORM data_select .
  CLEAR: gt_data.

  SELECT *
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zppt1000 AS a
  JOIN mara AS b ON a~matnr = b~matnr
  JOIN marc AS c ON a~matnr = c~matnr
                 AND a~werks = c~werks
  WHERE a~matnr IN s_matnr
    AND a~erdat IN s_erdat
    AND b~spart IN s_spart
    AND c~dismm IN s_dismm.

  LOOP AT gt_data INTO gs_data.



  ENDLOOP.

ENDFORM. " DATA_SELECT
