FORM SELECT_PART.


* 실적은 BLPK, BLPP TABLE의 DATA를 기준으로 읽어온다.
  PERFORM SELECT_BLPK_BLPP_MSEG.

* 바코드 입고
  PERFORM SELECT_ZPPT0170.

  CLEAR : IT_LIST, IT_LIST[].

  LOOP AT IT_MSEG.
    CLEAR IT_LIST.
    MOVE IT_MSEG-WERKS TO IT_LIST-WERKS.
    READ TABLE IT_BLPK WITH KEY  WERKS = IT_MSEG-WERKS
                                 BELNR = IT_MSEG-MBLNR
                                 BINARY SEARCH.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    IF IT_MSEG-BWART = '132' OR IT_MSEG-BWART = '532'.
      IT_MSEG-MENGE = IT_MSEG-MENGE * -1.
    ENDIF.
    MOVE IT_MSEG-BUDAT TO IT_LIST-BUDAT.
    MOVE IT_MSEG-MATNR TO IT_LIST-MATNR.
    MOVE IT_MSEG-LGORT TO IT_LIST-LGORT.
    MOVE IT_MSEG-MEINS TO IT_LIST-MEINS.
    MOVE IT_MSEG-MENGE TO IT_LIST-PRMNG.

    COLLECT IT_LIST.
  ENDLOOP.

  LOOP AT IT_T170.
    CLEAR IT_LIST.
    MOVE IT_T170-WERKS TO IT_LIST-WERKS.
    MOVE IT_T170-CREATED_DATE TO IT_LIST-BUDAT.
    MOVE IT_T170-MATNR TO IT_LIST-MATNR.
*    MOVE IT_T170-LGORT TO IT_LIST-LGORT.
    MOVE IT_T170-MEINS TO IT_LIST-MEINS.
    MOVE IT_T170-MENGE TO IT_LIST-BRMNG.

    SELECT SINGLE LGPRO INTO IT_LIST-LGORT FROM MARC
           WHERE MATNR = IT_LIST-MATNR AND
                 WERKS = IT_LIST-WERKS.

    COLLECT IT_LIST.
  ENDLOOP.

  CHECK IT_LIST[] IS NOT INITIAL.

FORM MAKE_DISPLAY .

         END OF LT_MAKT.

              ON A~MATNR = B~MATNR

  ENDLOOP.
