FORM SELECT_MAIN_DATA.

*1. 조회
  CLEAR : GT_MAIN[], GT_MAIN.
  SELECT A~PLWRK AS WERKS B~MTART A~MATNR C~MAKTX A~MEINS
         A~PEDTR A~GSMNG
    INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
                  FROM PLAF AS A
            INNER JOIN MARA AS B
                    ON A~MATNR = B~MATNR
            INNER JOIN MAKT AS C
                    ON A~MATNR = C~MATNR
                   AND C~SPRAS = SY-LANGU
                 WHERE A~PLWRK IN SO_WERKS
                   AND A~PLSCN IN SO_PLSCN
                   AND A~PEDTR IN GR_PEDTR
                   AND A~MATNR IN SO_MATNR
                   AND B~MTART IN SO_MTART
                   AND NOT ( A~BESKZ EQ 'F' AND A~SOBES EQ '0' ).

* 부산물 입고량에 대한 부분은 종속소요량 mdsm에
* 존재하기에 mdsm을 추가로 읽는다.
  SELECT A~WERKS B~MTART A~MATNR C~MAKTX A~MEINS
         A~BDTER AS PEDTR A~BDMNG AS GSMNG
    APPENDING CORRESPONDING FIELDS OF TABLE GT_MAIN
                       FROM MDSM AS A
                 INNER JOIN MARA AS B
                         ON A~MATNR = B~MATNR
                 INNER JOIN MAKT AS C
                         ON A~MATNR = C~MATNR
                        AND C~SPRAS = SY-LANGU
                 INNER JOIN MARC AS D
                         ON A~WERKS = D~WERKS
                        AND A~MATNR = D~MATNR
                      WHERE A~WERKS IN SO_WERKS
                        AND A~PLSCN IN SO_PLSCN
                        AND A~BDTER IN GR_PEDTR
                        AND A~MATNR IN SO_MATNR
                        AND B~MTART = 'BYPR'
                        AND D~BESKZ <> 'F'.


ENDFORM. " SELECT_MAIN_DATA

FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20).
  DATA : LV_TABIX LIKE SY-TABIX.
  DATA : LV_DATE(10) .

* dynamic table 생성
  PERFORM MAKE_DYNAMIC_FILDE.

  FIELD-SYMBOLS : <DYS_T>, <VALUE>.
  ASSIGN LOCAL COPY OF INITIAL LINE OF <GT_RESULT> TO <DYS_T>.

  DEFINE SET_SYMBOLS.
    UNASSIGN <VALUE>.
    ASSIGN COMPONENT &1 OF STRUCTURE <DYS_T> TO <VALUE>.
    IF <VALUE> IS ASSIGNED.
      CLEAR : <VALUE>.
      <VALUE> = &2.
    ENDIF.
  END-OF-DEFINITION.

  DEFINE SET_SM_MENG.
    UNASSIGN <VALUE>.
    ASSIGN COMPONENT &1 OF STRUCTURE <DYS_T> TO <VALUE>.
    IF <VALUE> IS ASSIGNED.
      <VALUE> = <VALUE> + &2.
    ENDIF.
  END-OF-DEFINITION.

  REFRESH <GT_RESULT>.
  SORT GT_MAIN BY WERKS MTART MATNR MEINS PEDTR.
  LOOP AT GT_MAIN.

    CASE GV_OK.
      WHEN 'DETL'.
        LV_DATE = GT_MAIN-PEDTR.
      WHEN OTHERS.
        LV_DATE = GT_MAIN-PEDTR(6).
    ENDCASE.

    CLEAR : <DYS_T>.
    READ TABLE <GT_RESULT> INTO <DYS_T>
                       WITH KEY ('WERKS') = GT_MAIN-WERKS
                                ('MTART') = GT_MAIN-MTART
                                ('MATNR') = GT_MAIN-MATNR
                                ('MEINS') = GT_MAIN-MEINS.
    IF SY-SUBRC EQ 0.
      LV_TABIX = SY-TABIX.

      SET_SM_MENG 'GSTOT'  GT_MAIN-GSMNG.
      SET_SM_MENG LV_DATE  GT_MAIN-GSMNG.

      MODIFY <GT_RESULT> FROM <DYS_T> INDEX LV_TABIX.
      CLEAR : <DYS_T>.

    ELSE.

      SET_SYMBOLS 'WERKS'  GT_MAIN-WERKS.
      SET_SYMBOLS 'MTART'  GT_MAIN-MTART.
      SET_SYMBOLS 'MATNR'  GT_MAIN-MATNR.
      SET_SYMBOLS 'MAKTX'  GT_MAIN-MAKTX.
      SET_SYMBOLS 'MEINS'  GT_MAIN-MEINS.
      SET_SM_MENG 'GSTOT'  GT_MAIN-GSMNG.
      SET_SM_MENG LV_DATE  GT_MAIN-GSMNG.

      APPEND <DYS_T> TO <GT_RESULT>.
      CLEAR : <DYS_T>.
    ENDIF.

  ENDLOOP.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE <GT_RESULT> LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S021 WITH LV_SCNT.

ENDFORM. " MAKE_DATA
