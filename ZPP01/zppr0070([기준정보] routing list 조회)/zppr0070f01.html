FORM SELECT_ETC_DATA .

  DATA : BEGIN OF LT_SPLMZ OCCURS 0,
          PLNTY  TYPE PLMZ-PLNTY,
          PLNNR  TYPE PLMZ-PLNNR,
          PLNKN  TYPE PLMZ-PLNKN,
         END OF LT_SPLMZ.

*조회조건 만들기
  CLEAR : LT_SPLMZ[], LT_SPLMZ.
  LOOP AT GT_MAIN.
    MOVE-CORRESPONDING GT_MAIN TO LT_SPLMZ.
    APPEND LT_SPLMZ.
  ENDLOOP.
  SORT LT_SPLMZ.
  DELETE ADJACENT DUPLICATES FROM LT_SPLMZ.

*조회
  CLEAR : GT_PLMZ[], GT_PLMZ.
  IF LT_SPLMZ[] IS NOT INITIAL.
    SELECT PLNTY PLNNR ZUONR ZAEHL PLNKN RGEKZ
      INTO CORRESPONDING FIELDS OF TABLE GT_PLMZ
                    FROM PLMZ
      FOR ALL ENTRIES IN LT_SPLMZ
                   WHERE PLNTY = LT_SPLMZ-PLNTY
                     AND PLNNR = LT_SPLMZ-PLNNR
                     AND PLNKN = LT_SPLMZ-PLNKN
                     AND LOEKZ = ' '.
  ENDIF.

ENDFORM. " SELECT_ETC_DATA
