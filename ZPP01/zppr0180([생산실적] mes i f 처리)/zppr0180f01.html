FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20),
         LV_TABIX LIKE SY-TABIX,
         LV_CHK.

  IF SY-BATCH = 'X'.

  ENDIF.

  CLEAR : GT_RESULT[], GT_RESULT.
  LOOP AT GT_MAIN.
    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.

*   자재코드
    IF GT_RESULT-MATNR IS INITIAL.
      SELECT SINGLE MATNR INTO GT_RESULT-MATNR FROM ZPPT0090
             WHERE WERKS  = GT_RESULT-WERKS AND
                   ZPLNUM = GT_RESULT-ZPLNUM AND
                   ARBPL  = GT_RESULT-ARBPL.
    ENDIF.
    IF GT_RESULT-MATNR IS NOT INITIAL.
*   MRP 관리자
      SELECT SINGLE A~DISPO B~DSNAM
             INTO (GT_RESULT-DISPO, GT_RESULT-DISPO_NM)
             FROM MARC AS A INNER JOIN T024D AS B ON A~WERKS = B~WERKS AND
                                                     A~DISPO = B~DISPO
             WHERE A~WERKS = GT_RESULT-WERKS AND
                   A~MATNR = GT_RESULT-MATNR.
*     자재내역
      SELECT SINGLE MAKTX INTO GT_RESULT-MAKTX FROM MAKT
          WHERE SPRAS  = SY-LANGU AND
                MATNR  = GT_RESULT-MATNR.
    ENDIF.

    IF SO_DISPO[] IS NOT INITIAL.
      IF NOT GT_RESULT-DISPO IN SO_DISPO.
        CONTINUE.
      ENDIF.
    ENDIF.

*   체크
    GT_RESULT-CHK = LV_CHK.

*   아이콘
    CASE GT_RESULT-STATUS.
      WHEN 'A'.
        GT_RESULT-ICON = C_I_NEW.
      WHEN 'E' OR 'B'.
        GT_RESULT-ICON = C_I_RED.
      WHEN 'P'.
        GT_RESULT-ICON = C_I_YEL.
      WHEN 'S'.
        GT_RESULT-ICON = C_I_GRE.
      WHEN 'X' OR 'Y'.
        GT_RESULT-ICON = C_I_DEL.
      WHEN 'Z'.
        GT_RESULT-ICON = C_I_CNC.
    ENDCASE.

**   삭제여부 확인
*    CLEAR GT_T111.
*    READ TABLE GT_T111 WITH KEY  ZPLNUM = GT_RESULT-ZPLNUM
*                                 ARBPL  = GT_RESULT-ARBPL
*                                 HOGI   = GT_RESULT-HOGI
*                                 SHIFT  = GT_RESULT-SHIFT
*                                 T_TIME = GT_RESULT-T_TIME
*                                 BINARY SEARCH.
*    IF SY-SUBRC EQ 0.
*      CONTINUE.
*    ENDIF.


    GT_RESULT-TDATE = GT_RESULT-T_TIME(8).
    GT_RESULT-TTIME = GT_RESULT-T_TIME+8(6).

    APPEND GT_RESULT. CLEAR : GT_RESULT.
  ENDLOOP.
  SORT GT_RESULT BY T_TIME.

  IF GT_RESULT[] IS INITIAL.
    MESSAGE S009.
    LEAVE LIST-PROCESSING.
  ENDIF.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
