  LOOP AT GT_RESULT.

*----------------------------------------------------------------------*
FORM GET_ACTUAL .

  DATA : BEGIN OF LT_BLPK OCCURS 0,
          WERKS    LIKE MDSM-WERKS,
          MATNR    LIKE MDSM-MATNR,
          VERID    LIKE BLPK-VERID,
          MENGE    LIKE S226-BDMNG,
          MEINH    LIKE MARA-MEINS,
          BUDAT    LIKE BLPK-BUDAT,
          BELNR    LIKE BLPP-BELNR,
          STORN    LIKE BLPK-STORN,
         END OF LT_BLPK.

  DATA : BEGIN OF LT_MSEG OCCURS 0,
          MBLNR    LIKE MSEG-MBLNR,
          MJAHR    LIKE MSEG-MJAHR,
          ZEILE    LIKE MSEG-ZEILE,
          WERKS    LIKE MDSM-WERKS,
          MATNR    LIKE MDSM-MATNR,
          LGORT    LIKE MARD-LGORT,
          MENGE    LIKE BLPK-MENGE,
          BWART    LIKE MSEG-BWART,
          MTART    LIKE MARA-MTART,
          BESKZ    LIKE MARC-BESKZ,
          CHARG    LIKE MSEG-CHARG,
          MEINS    LIKE MSEG-MEINS,
          BUDAT    LIKE MKPF-BUDAT,
          DISPO    LIKE MARC-DISPO, "MRP 관리자
          MATKL    LIKE MARA-MATKL,
         END OF LT_MSEG.

  DATA : LT_MSEG2 LIKE LT_MSEG OCCURS 0 WITH HEADER LINE.

* 실적
  SELECT A~WERKS A~MATNR A~VERID A~MENGE A~MEINH B~BELNR
         A~STORN A~BUDAT
  INTO CORRESPONDING FIELDS OF TABLE LT_BLPK
  FROM BLPK AS A JOIN BLPP AS B
    ON A~PRTNR = B~PRTNR
  WHERE A~WERKS IN SO_WERKS
    AND A~MATNR IN SO_MATNR
    AND A~BUDAT IN SO_BUDAT
    AND B~BELNR NE ''.

* 실적 정보상의 내용으로 자재문서정보를 추가로 읽어서 it_mseg상에 넣어준다.
  CHECK LT_BLPK[] IS NOT INITIAL.

* 이동유형구성
  CLEAR S_BWART[].
  M_APPEND_BWART '131'. "REM 입고
  M_APPEND_BWART '132'.
  M_APPEND_BWART '261'.
  M_APPEND_BWART '262'.
  M_APPEND_BWART '531'.
  M_APPEND_BWART '532'.

  SELECT A~MBLNR A~MJAHR B~ZEILE B~WERKS B~MATNR B~LGORT B~MENGE B~BWART
         C~MTART D~BESKZ B~CHARG B~MEINS A~BUDAT D~DISPO C~MATKL
       INTO CORRESPONDING FIELDS OF TABLE LT_MSEG
       FROM MKPF AS A JOIN MSEG AS B
            ON A~MBLNR = B~MBLNR
               INNER JOIN MARA AS C
                     ON B~MATNR = C~MATNR
                       INNER JOIN MARC AS D
                             ON  B~WERKS = D~WERKS
                             AND B~MATNR = D~MATNR
       FOR ALL ENTRIES IN LT_BLPK
       WHERE A~BUDAT IN SO_BUDAT
         AND A~MBLNR = LT_BLPK-BELNR
         AND B~WERKS = LT_BLPK-WERKS.
  SORT LT_MSEG BY MATNR WERKS MBLNR.


* 투입내역만 별도로 모은다.
  CLEAR S_BWART[].
  M_APPEND_BWART '261'.
  M_APPEND_BWART '262'.
  LOOP AT LT_MSEG INTO LT_MSEG2 WHERE BWART IN S_BWART.
    APPEND LT_MSEG2.
  ENDLOOP.

  SORT LT_BLPK BY BELNR MATNR.
  SORT LT_MSEG BY MBLNR MJAHR.
  SORT LT_MSEG2 BY MBLNR MJAHR MATNR.


* LT_MSEG의 DATA중 자재유형 FERT, 이동유형 261,262 조달유형 'E' 이고 MRP관리자 '502'가
* 아닌 DATA가 있으면 이전표에 해당하는 DATA를 모두 삭제한다.
* 261,262로 투입된 자재중 BLPK 생산품으로 정의된 DATA가 있으면 해당 전표 DATA 삭제
  LOOP AT LT_MSEG2.
    IF LT_MSEG2-MTART = 'FERT' AND LT_MSEG2-BESKZ = 'E' AND LT_MSEG2-DISPO <> '502'.
      DELETE LT_MSEG WHERE WERKS = LT_MSEG2-WERKS
                       AND MBLNR = LT_MSEG2-MBLNR.
      DELETE LT_BLPK WHERE WERKS = LT_MSEG2-WERKS
                       AND BELNR = LT_MSEG2-MBLNR.
      DELETE LT_MSEG2 WHERE WERKS = LT_MSEG2-WERKS
                        AND MBLNR = LT_MSEG2-MBLNR.
      CONTINUE.
    ENDIF.
    READ TABLE LT_BLPK WITH KEY BELNR = LT_MSEG2-MBLNR
                                MATNR = LT_MSEG2-MATNR.
    IF SY-SUBRC = 0.
      DELETE LT_MSEG WHERE WERKS = LT_MSEG2-WERKS
                       AND MBLNR = LT_MSEG2-MBLNR.
      DELETE LT_BLPK WHERE WERKS = LT_MSEG2-WERKS
                       AND BELNR = LT_MSEG2-MBLNR.
      DELETE LT_MSEG2 WHERE WERKS = LT_MSEG2-WERKS
                        AND MBLNR = LT_MSEG2-MBLNR.
    ENDIF.
  ENDLOOP.

  SORT LT_MSEG BY WERKS MBLNR.

* Collect
  SORT LT_BLPK BY WERKS BELNR.
  CLEAR S_BWART[].
  M_APPEND_BWART '131'. "REM 입고
  M_APPEND_BWART '132'.
  M_APPEND_BWART '531'.
  M_APPEND_BWART '532'.
  LOOP AT LT_MSEG WHERE BWART IN S_BWART.
    CLEAR GT_RESULT.

    READ TABLE LT_BLPK WITH KEY  WERKS = LT_MSEG-WERKS
                                 BELNR = LT_MSEG-MBLNR
                                 BINARY SEARCH.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    IF LT_MSEG-BWART = '132' OR LT_MSEG-BWART = '532'.
      LT_MSEG-MENGE = LT_MSEG-MENGE * -1.
    ENDIF.
    MOVE : LT_MSEG-MATNR TO GT_RESULT-MATNR,
           LT_MSEG-BUDAT TO GT_RESULT-BUDAT,
           LT_MSEG-WERKS TO GT_RESULT-WERKS,
           LT_MSEG-MENGE TO GT_RESULT-AENGE,
           LT_MSEG-MEINS TO GT_RESULT-MEINH.

    COLLECT GT_RESULT.
  ENDLOOP.


ENDFORM. " GET_ACTUAL

FORM GET_PLAN .

  DATA : BEGIN OF LT_PLAN OCCURS 0,
           SPMON TYPE SPMON,
           WERKS TYPE WERKS_EXT,
           MATNR TYPE MATNR,
           MENGE LIKE BLPK-MENGE,
           MEINS LIKE MSEG-MEINS,
         END OF LT_PLAN.

  DATA : BEGIN OF LT_SPMON OCCURS 0,
           SPMON TYPE SPMON,
         END OF LT_SPMON.

  DATA : LV_INT TYPE I.


  REFRESH SO_SPMON.
  READ TABLE SO_BUDAT INDEX 1.
  CHECK SY-SUBRC = 0.
  IF SO_BUDAT-HIGH IS INITIAL.
    SO_SPMON-SIGN = 'I'.
    SO_SPMON-OPTION = 'EQ'.
    SO_SPMON-LOW = SO_BUDAT-LOW.
    APPEND SO_SPMON.
  ELSE.
    SO_SPMON-SIGN = 'I'.
    SO_SPMON-OPTION = 'BT'.
    SO_SPMON-LOW = SO_BUDAT-LOW.
    SO_SPMON-HIGH = SO_BUDAT-HIGH.
    APPEND SO_SPMON.
  ENDIF.

* 1.계획 가져오기
  SELECT A~SPMON  A~WERKS A~MATNR A~TOTMNG AS MENGE A~MEINS
         FROM ZSDT0400 AS A INNER JOIN MARA AS B ON A~MATNR = B~MATNR
         INTO CORRESPONDING FIELDS OF TABLE LT_PLAN
         WHERE B~MTART <> 'HAWA'
           AND A~WERKS IN SO_WERKS
           AND A~MATNR IN SO_MATNR
           AND A~SPMON IN SO_SPMON
           AND A~EXTYP = 'E'.
  IF SY-SUBRC <> 0.
    SELECT A~SPMON  A~WERKS A~MATNR A~TOTMNG AS MENGE A~MEINS
           FROM ZSDT0400 AS A INNER JOIN MARA AS B ON A~MATNR = B~MATNR
           INTO CORRESPONDING FIELDS OF TABLE LT_PLAN
           WHERE B~MTART <> 'HAWA'
             AND A~WERKS IN SO_WERKS
             AND A~MATNR IN SO_MATNR
             AND A~SPMON IN SO_SPMON
             AND A~EXTYP = 'P'.
  ENDIF.
  CHECK LT_PLAN[] IS NOT INITIAL.
  SORT LT_PLAN BY SPMON.

  LOOP AT LT_PLAN.
    LT_SPMON-SPMON = LT_PLAN-SPMON.
    APPEND LT_SPMON.
  ENDLOOP.
  SORT LT_SPMON.
  DELETE ADJACENT DUPLICATES FROM LT_SPMON COMPARING SPMON.

* 2. 월별로 계획 처리
  LOOP AT LT_SPMON.
    LOOP AT LT_PLAN WHERE SPMON = LT_SPMON-SPMON.
*     월/플랜트 근무일수 구하기
      PERFORM GET_WORKING_DAYS USING LT_PLAN-SPMON LT_PLAN-WERKS LV_INT .
      CLEAR GT_RESULT.
      IF LV_INT <> 0.
        MOVE : LT_PLAN-MATNR TO GT_RESULT-MATNR,
*               LT_PLAN-BUDAT TO GT_RESULT-BUDAT,
               LT_PLAN-WERKS TO GT_RESULT-WERKS,
               LT_PLAN-MENGE TO GT_RESULT-PENGE,
               LT_PLAN-MEINS TO GT_RESULT-MEINH.

        GT_RESULT-PENGE = GT_RESULT-PENGE / LV_INT.
        LOOP AT GT_BUDAT.
          MOVE : GT_BUDAT-BUDAT TO GT_RESULT-BUDAT.
          COLLECT GT_RESULT.
        ENDLOOP.
      ENDIF.

    ENDLOOP.
  ENDLOOP.


ENDFORM. " GET_PLAN
