*&---------------------------------------------------------------------*

*  <--  p2        text
*----------------------------------------------------------------------*
FORM SELECT_DATA .

  CLEAR: GT_DISPLAY, R_DEV, GT_PROGRAM, GT_PRO, GT_TCODE, GT_TABLE,
         GT_OBJECT.
  REFRESH: GT_DISPLAY, R_DEV, GT_PROGRAM, GT_TABLE, GT_OBJECT,
           GT_TCODE, GT_PRO.

*  LOOP AT GT_001.
*    PERFORM RANGES_DEVCLASS USING GT_001-DEVCLASS.
*  ENDLOOP.
*
  IF S_DEV[] IS INITIAL.
    SELECT DEVCLASS OBJECT OBJ_NAME AS PROGRAM INTO TABLE GT_PRO
           FROM TADIR
           WHERE PGMID = 'R3TR'
*           AND DEVCLASS IN S_DEV.
              AND ( DEVCLASS LIKE  'Z%' OR
                    DEVCLASS LIKE  'Y%' )
              AND DELFLAG NE 'X'.
  ELSE.
    SELECT DEVCLASS OBJECT OBJ_NAME AS PROGRAM INTO TABLE GT_PRO
           FROM TADIR
           WHERE PGMID = 'R3TR'
             AND DEVCLASS IN S_DEV
             AND DELFLAG  NE 'X'.
  ENDIF.

  LOOP AT GT_PRO.
    IF GT_PRO-OBJECT = C_PROG.
      MOVE: GT_PRO-DEVCLASS TO GT_PROGRAM1-DEVCLASS,
            GT_PRO-PROGRAM TO GT_PROGRAM1-PROGRAM.

      PERFORM FIND_PROGRAM_NAME USING GT_PROGRAM1-PROGRAM
                         CHANGING GT_PROGRAM1-CTEXT.
      APPEND GT_PROGRAM1. CLEAR GT_PROGRAM1.

      SELECT SINGLE * FROM TRDIR
           WHERE NAME = GT_PRO-PROGRAM
             AND ( SUBC = '1' OR SUBC = 'M' ).
      IF SY-SUBRC = 0.
        MOVE: GT_PRO-DEVCLASS TO GT_PROGRAM-DEVCLASS,
              GT_PRO-PROGRAM  TO GT_PROGRAM-PROGRAM,
              TRDIR-UCCHECK   TO GT_PROGRAM-UCCHECK.

        PERFORM FIND_PROGRAM_NAME USING GT_PROGRAM-PROGRAM
                           CHANGING GT_PROGRAM-CTEXT.
        APPEND GT_PROGRAM. CLEAR GT_PROGRAM.
      ENDIF.
    ELSEIF GT_PRO-OBJECT = C_W3MI.
      MOVE: GT_PRO-DEVCLASS TO GT_OBJECT-DEVCLASS,
            GT_PRO-PROGRAM TO GT_OBJECT-OBJ_NAME.
      APPEND GT_OBJECT. CLEAR GT_OBJECT.
    ELSEIF GT_PRO-OBJECT = C_TRAN.
      MOVE: GT_PRO-DEVCLASS TO GT_TCODE-DEVCLASS,
            GT_PRO-PROGRAM TO GT_TCODE-TCODE.
      APPEND GT_TCODE. CLEAR GT_TCODE.
*    ELSEIF GT_PRO-OBJECT = C_TABL.
*      MOVE: GT_PRO-DEVCLASS TO GT_TABLE-DEVCLASS,
*            GT_PRO-PROGRAM TO GT_TABLE-OBJ_NAME.
*      APPEND GT_TABLE. CLEAR GT_TABLE.
    ELSE.
      DELETE GT_PRO. CLEAR GT_PRO.
    ENDIF.
  ENDLOOP.
ENDFORM. " SELECT_DATA

      SHIFT TEMP_LINE UP TO SPACE.
      PERFORM FIND_TABLES_ON_ONE_LINE TABLES TABLE_NAMES
                                      USING TEMP_LINE EOL.
