FORM GET_SQL_STATEMENT CHANGING P_SQL_STATEMENT TYPE STRING
                                 PT_SQL_STATEMENT TYPE TY_TEXTTAB.
  DATA: L_STMTLINE    TYPE TY_TEXTLINE.
  DATA: LT_STATEMENT  TYPE TY_TEXTTAB.
  DATA: L_NO_OF_LINES TYPE I.

*   EXPORTING

  DESCRIBE TABLE LT_STATEMENT LINES L_NO_OF_LINES.
  DO.
    IF L_NO_OF_LINES = 0. EXIT. ENDIF.  " empty table
    READ TABLE LT_STATEMENT INTO L_STMTLINE INDEX L_NO_OF_LINES.

    DESCRIBE TABLE LT_STATEMENT LINES L_NO_OF_LINES. " next last
  ENDDO.

  LOOP AT LT_STATEMENT INTO L_STMTLINE.
