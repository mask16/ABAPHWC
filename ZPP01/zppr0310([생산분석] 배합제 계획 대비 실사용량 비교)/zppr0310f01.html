*----------------------------------------------------------------------*

*----------------------------------------------------------------------*
FORM SELECT_MAIN_DATA .

  CLEAR : GT_MAT[], GT_MAT, GT_PQTY[], GT_PQTY, GT_AQTY[], GT_AQTY.

  DATA : LT_PQTY LIKE GT_PQTY OCCURS 0 WITH HEADER LINE.


* 전기일 세팅
  REFRESH : SO_BUDAT.
  SO_BUDAT-SIGN = 'I'.
  SO_BUDAT-OPTION = 'BT'.
  CONCATENATE PA_SPMON '01' INTO SO_BUDAT-LOW.
  CALL FUNCTION 'SG_PS_GET_LAST_DAY_OF_MONTH'
    EXPORTING
      DAY_IN            = SO_BUDAT-LOW
    IMPORTING
      LAST_DAY_OF_MONTH = SO_BUDAT-HIGH.
  APPEND SO_BUDAT.


* 자재리스트
  SELECT A~WERKS  A~MATNR B~MEINS
    INTO CORRESPONDING FIELDS OF TABLE GT_MAT
                  FROM MARC  AS A
            INNER JOIN MARA AS B
                    ON A~MATNR = B~MATNR
            INNER JOIN MAST AS C
                    ON A~MATNR = C~MATNR AND
                       A~WERKS = C~WERKS
                 WHERE A~WERKS IN SO_WERKS AND
                       A~DISPO IN ('201', '202', '203', '204', '205') AND
                       C~STLAN IN ('W', 'X', 'Y', 'Z') AND
                       B~MTART = 'HALB'.
  CHECK GT_MAT[] IS NOT INITIAL.
  SORT GT_MAT BY WERKS MATNR.


* 계획수량
  SELECT A~WERKS A~MATNR A~MENGE A~MEINS A~BWART A~MBLNR A~MJAHR A~ZEILE
        INTO CORRESPONDING FIELDS OF TABLE LT_PQTY
                  FROM MSEG  AS A INNER JOIN MKPF AS B ON
                                A~MBLNR = B~MBLNR AND
                                A~MJAHR = B~MJAHR
                 FOR ALL ENTRIES IN GT_MAT
                 WHERE A~WERKS IN SO_WERKS AND
                       A~BWART IN ('131', '132') AND
                       B~BUDAT IN SO_BUDAT AND
                       A~WERKS = GT_MAT-WERKS AND
                       A~MATNR = GT_MAT-MATNR.
  IF LT_PQTY[] IS NOT INITIAL.
    LOOP AT LT_PQTY.
      MOVE-CORRESPONDING LT_PQTY TO GT_PQTY.
      IF GT_PQTY-BWART = '132'.
        GT_PQTY-MENGE = GT_PQTY-MENGE * -1.
      ENDIF.
      CLEAR : GT_PQTY-BWART, GT_PQTY-MBLNR, GT_PQTY-MJAHR, GT_PQTY-ZEILE.

      COLLECT GT_PQTY.
    ENDLOOP.
    SORT GT_PQTY BY WERKS MATNR.
  ENDIF.

* 실적수량(201,202 ROH)
  SELECT A~WERKS A~MATNR A~MENGE A~MEINS A~BWART A~KOSTL A~MBLNR A~MJAHR A~ZEILE
        INTO CORRESPONDING FIELDS OF TABLE GT_AQTY
                  FROM MSEG  AS A INNER JOIN MKPF AS B ON
                                A~MBLNR = B~MBLNR AND
                                A~MJAHR = B~MJAHR
                                  INNER JOIN MARA AS C ON
                                A~MATNR = C~MATNR
                 WHERE A~WERKS IN SO_WERKS AND
*                       A~MATNR = P_MATNR AND
                       A~BWART IN ('201', '202') AND
                       B~BUDAT IN SO_BUDAT AND
                       C~MATKL IN ('4A11', '4A12').

* 실적수량(261,262 HALB)
  SELECT A~WERKS A~MATNR A~MENGE A~MEINS A~BWART A~KOSTL A~MBLNR A~MJAHR A~ZEILE
        APPENDING CORRESPONDING FIELDS OF TABLE GT_AQTY
                  FROM MSEG  AS A INNER JOIN MKPF AS B ON
                                A~MBLNR = B~MBLNR AND
                                A~MJAHR = B~MJAHR
                                  INNER JOIN MARC AS C ON
                                A~MATNR = C~MATNR AND
                                A~WERKS = C~WERKS
                 WHERE A~WERKS IN SO_WERKS AND
*                       A~MATNR = P_MATNR AND
                       A~BWART IN ('261', '262') AND
                       B~BUDAT IN SO_BUDAT AND
                       C~DISPO EQ '204'.
  SORT GT_AQTY BY WERKS MATNR.


ENDFORM. " SELECT_MAIN_DATA

*  <--  p2        text
*----------------------------------------------------------------------*
FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20),
         LV_TABIX LIKE SY-TABIX.

  DATA : LT_MATCAT    LIKE TABLE OF CSCMAT WITH HEADER LINE,
         LT_STPOX     LIKE TABLE OF STPOX WITH HEADER LINE.

  DATA : BEGIN OF LT_MATNR OCCURS 0,
         WERKS       TYPE WERKS_D,
         MATNR       TYPE MATNR.
  DATA : END OF LT_MATNR.

  DATA : BEGIN OF LT_BOM OCCURS 0,
         BMENG       TYPE STKO-BMENG,
         IDNRK       TYPE MATNR,
         MENGE       TYPE STPO-MENGE,
         MEINS       TYPE STPO-MEINS.
  DATA : END OF LT_BOM.

  CHECK GT_MAT[] IS NOT INITIAL.
  SORT GT_MAT BY WERKS MATNR.

  CLEAR : GT_RESULT[], GT_RESULT,
          GT_MAIN[], GT_MAIN, GT_KOSTL[].

  LOOP AT GT_MAT.
    CLEAR GT_RESULT.

*   계획수량(상위자재)
    READ TABLE GT_PQTY WITH KEY WERKS = GT_MAT-WERKS
                                MATNR = GT_MAT-MATNR
                                BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GT_MAT-MENGE = GT_PQTY-MENGE.
    ENDIF.

    IF GT_MAT-MENGE <> 0.
*   BOM 정보
      REFRESH : LT_BOM.
      SELECT B~BMENG A~IDNRK A~MENGE A~MEINS INTO TABLE LT_BOM
        FROM STPO AS A INNER JOIN STKO AS B ON A~STLTY = B~STLTY AND
                                               A~STLNR = B~STLNR
                       INNER JOIN MAST AS C ON A~STLNR = C~STLNR
        WHERE A~DATUV <= SY-DATUM AND
              A~LKENZ = '' AND
              B~DATUV <= SY-DATUM AND
              B~LKENZ = '' AND
              B~LOEKZ = '' AND
              C~MATNR = GT_MAT-MATNR AND
              C~WERKS = GT_MAT-WERKS AND
              C~STLAN <> '1'.
      LOOP AT LT_BOM.
        GT_RESULT-WERKS = GT_MAT-WERKS.
        GT_RESULT-MATNR = LT_BOM-IDNRK.
        GT_RESULT-MEINS = LT_BOM-MEINS.
*        GT_RESULT-MENGE = GT_MAT-MENGE * ( LT_BOM-MENGE / LT_BOM-BMENG ).
        GT_RESULT-MENGE = ( GT_MAT-MENGE / LT_BOM-BMENG ) * LT_BOM-MENGE.
        COLLECT GT_RESULT.
      ENDLOOP.

**   BOM 정보
*      PERFORM GET_BOM_LIST_DATA TABLES LT_STPOX
*                                USING GT_MAT-MATNR GT_MAT-WERKS GT_MAT-MENGE.
*      LOOP AT LT_STPOX.
*        GT_RESULT-WERKS = LT_STPOX-WERKS.
*        GT_RESULT-MATNR = LT_STPOX-IDNRK.
*        GT_RESULT-MEINS = LT_STPOX-MEINS.
*        GT_RESULT-MENGE = LT_STPOX-MNGLG.
*        COLLECT GT_RESULT.
*      ENDLOOP.

    ENDIF.
*    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
*    APPEND GT_RESULT. CLEAR : GT_RESULT.
  ENDLOOP.
  SORT GT_RESULT BY WERKS MATNR.

* 생산출고수량 가져오기
  LOOP AT GT_RESULT.
    CLEAR GT_MAIN.
    MOVE-CORRESPONDING GT_RESULT TO GT_MAIN.

*   자재내역
    SELECT SINGLE MAKTX INTO GT_MAIN-MATNR_NM FROM MAKT
           WHERE MATNR = GT_MAIN-MATNR AND
                 SPRAS = SY-LANGU.

*   생산출고
    PERFORM GET_GI_QTY USING GT_MAIN-MATNR.

*   합계수량
    GT_MAIN-PGISUM = GT_MAIN-PGIMNG +
                     GT_MAIN-PGIMNG1 + GT_MAIN-PGIMNG2 + GT_MAIN-PGIMNG3 +
                     GT_MAIN-PGIMNG4 + GT_MAIN-PGIMNG5 + GT_MAIN-PGIMNG6 +
                     GT_MAIN-PGIMNG7 + GT_MAIN-PGIMNG8 + GT_MAIN-PGIMNG9.
    GT_MAIN-DMNG =   GT_MAIN-MENGE - GT_MAIN-PGISUM.

    IF GT_MAIN-PGISUM IS INITIAL AND GT_MAIN-MENGE IS INITIAL.
*      DELETE GT_RESULT INDEX LV_TABIX.
    ELSE.
      APPEND GT_MAIN.
    ENDIF.

  ENDLOOP.

* 계획에 없는 출고
  IF GT_AQTY[] IS NOT INITIAL.

    LOOP AT GT_AQTY.
      MOVE : GT_AQTY-MATNR TO LT_MATNR-MATNR,
             GT_AQTY-WERKS TO LT_MATNR-WERKS.
      APPEND LT_MATNR.
    ENDLOOP.
    SORT LT_MATNR.
    DELETE ADJACENT DUPLICATES FROM LT_MATNR COMPARING MATNR.


    LOOP AT LT_MATNR.
      CLEAR GT_MAIN.
      MOVE-CORRESPONDING LT_MATNR TO GT_MAIN.

*   자재내역
      SELECT SINGLE MAKTX INTO GT_MAIN-MATNR_NM FROM MAKT
             WHERE MATNR = GT_MAIN-MATNR AND
                   SPRAS = SY-LANGU.

*   기본단위
      SELECT SINGLE MEINS INTO GT_MAIN-MEINS FROM MARA
             WHERE MATNR = GT_MAIN-MATNR.

*   생산출고
      PERFORM GET_GI_QTY USING GT_MAIN-MATNR.

*   합계수량
      GT_MAIN-PGISUM = GT_MAIN-PGIMNG +
                       GT_MAIN-PGIMNG1 + GT_MAIN-PGIMNG2 + GT_MAIN-PGIMNG3 +
                       GT_MAIN-PGIMNG4 + GT_MAIN-PGIMNG5 + GT_MAIN-PGIMNG6 +
                       GT_MAIN-PGIMNG7 + GT_MAIN-PGIMNG8 + GT_MAIN-PGIMNG9.
      GT_MAIN-DMNG =   GT_MAIN-MENGE - GT_MAIN-PGISUM.

      IF GT_MAIN-PGISUM IS INITIAL AND GT_MAIN-MENGE IS INITIAL.
*        DELETE GT_RESULT INDEX LV_TABIX.
      ELSE.
        APPEND GT_MAIN.
      ENDIF.

    ENDLOOP.

  ENDIF.

* 조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE GT_MAIN LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S021 WITH LV_SCNT.

ENDFORM. " MAKE_DATA

FORM SET_LAYOUT CHANGING PS_LAYOUT TYPE LVC_S_LAYO.

  CLEAR : PS_LAYOUT.
  PS_LAYOUT-SEL_MODE      = C_A.
  PS_LAYOUT-BOX_FNAME     = 'CHK'().
  PS_LAYOUT-CWIDTH_OPT    = C_A.

FORM GET_GI_QTY USING P_MATNR.

*  DATA : LT_PQTY LIKE GT_PQTY OCCURS 0 WITH HEADER LINE.
*
*  SELECT A~WERKS A~MATNR A~MENGE A~MEINS A~BWART A~KOSTL A~MBLNR A~MJAHR A~ZEILE
*        INTO CORRESPONDING FIELDS OF TABLE LT_PQTY

*                                A~MBLNR = B~MBLNR AND
*                                A~MJAHR = B~MJAHR
*                 WHERE A~WERKS IN SO_WERKS AND
*                       A~MATNR = P_MATNR AND
*                       A~BWART IN ('261', '262', '201', '202') AND

  CHECK SY-SUBRC EQ 0.

    IF GT_AQTY-BWART = '201' OR
       GT_AQTY-BWART = '202'.
