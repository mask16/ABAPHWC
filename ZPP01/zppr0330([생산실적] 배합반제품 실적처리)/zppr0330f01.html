FORM READ_DATA .

  SELECT * INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
         FROM ZPPT0113
         WHERE WERKS  IN SO_WERKS AND
               DISPO  IN SO_DISPO AND
               BUDAT  IN SO_BUDAT AND
               STATUS IN SO_STS.

  LOOP AT IT_MAIN.

    IF IT_MAIN-MAKTX IS INITIAL.
      SELECT SINGLE MAKTX INTO IT_MAIN-MAKTX FROM MAKT
        WHERE MATNR = IT_MAIN-MATNR AND
              SPRAS = SY-LANGU.
    ENDIF.

*   MRP 관리자명
    SELECT SINGLE DSNAM
           INTO  IT_MAIN-DISPO_NM
           FROM T024D
           WHERE WERKS = IT_MAIN-WERKS AND
                 DISPO = IT_MAIN-DISPO.

*   ICON
    CASE IT_MAIN-STATUS.
      WHEN 'A'.
        IT_MAIN-ICON = ICON_LED_INACTIVE.
      WHEN 'P'.
        IT_MAIN-ICON = ICON_LED_YELLOW.
      WHEN 'S'.
        IT_MAIN-ICON = ICON_LED_GREEN.
      WHEN 'E'.
        IT_MAIN-ICON = ICON_LED_RED.
    ENDCASE.

    MODIFY IT_MAIN.
  ENDLOOP.

  IF IT_MAIN[] IS INITIAL.
    MESSAGE S009.
    LEAVE LIST-PROCESSING.
  ENDIF.

  CALL SCREEN 200.

ENDFORM. " READ_DATA

FORM MOVE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20).

  DATA : LV_RET.

  CHECK GT_UPF[] IS NOT INITIAL.
*  SORT GT_UPF BY WERKS MATNR.
  CLEAR : IT_MAIN[], IT_MAIN.
  LOOP AT GT_UPF INTO GS_UPF.
    CLEAR : IT_MAIN.
    MOVE-CORRESPONDING GS_UPF TO IT_MAIN.

    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        INPUT  = IT_MAIN-MATNR
      IMPORTING
        OUTPUT = IT_MAIN-MATNR.

*    IF IT_MAIN-MAKTX IS INITIAL.
    SELECT SINGLE MAKTX INTO IT_MAIN-MAKTX FROM MAKT
      WHERE MATNR = IT_MAIN-MATNR AND
            SPRAS = SY-LANGU.
*    ENDIF.

    SELECT SINGLE MEINS INTO IT_MAIN-MEINS FROM MARA
      WHERE MATNR = IT_MAIN-MATNR .

*   MRP 관리자
    SELECT SINGLE A~DISPO B~DSNAM
           INTO (IT_MAIN-DISPO, IT_MAIN-DISPO_NM)
           FROM MARC AS A INNER JOIN T024D AS B ON A~WERKS = B~WERKS AND
                                                   A~DISPO = B~DISPO
           WHERE A~WERKS = IT_MAIN-WERKS AND
                 A~MATNR = IT_MAIN-MATNR.

    IT_MAIN-ICON = ICON_LED_INACTIVE.
    APPEND IT_MAIN.

  ENDLOOP.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE IT_MAIN LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S038 WITH LV_SCNT.

ENDFORM. " MOVE_DATA

FORM UC_0200-SAVE .

  DATA : L_ANSWER  TYPE  C,
         L_ERROR   TYPE  C.

  DATA : LV_CNT  TYPE I,
         LV_SCNT TYPE I,
         LV_MSG(100),
         LV_CNTC(10) TYPE C,
         LV_SCNTC(10) TYPE C,
         LV_TABIX TYPE SY-TABIX.

  DATA : LT_T113 LIKE ZPPT0113 OCCURS 0 WITH HEADER LINE.

  IT_MAIN-MARK = ''.
  MODIFY IT_MAIN TRANSPORTING MARK WHERE MARK = 'X'.


  CLEAR : IT_ROWS[], IT_ROWS.
  CALL METHOD G_GRID1->GET_SELECTED_ROWS
    IMPORTING
      ET_ROW_NO = IT_ROWS[].

*- No selected rows
  IF IT_ROWS[] IS INITIAL.
    MESSAGE E004.
*    EXIT.
  ENDIF.

*- Internal Table, Select whether to save
  LOOP AT IT_ROWS.
    CLEAR: IT_MAIN.
    READ TABLE IT_MAIN INDEX IT_ROWS-ROW_ID.
    IF SY-SUBRC = 0.
      IT_MAIN-MARK = 'X'.
      MODIFY IT_MAIN INDEX IT_ROWS-ROW_ID.
    ENDIF.
  ENDLOOP.

*- Confirm
  PERFORM POPUP_TO_CONFIRM  USING     'Run'
                                      'Save the selected information ?'
                                      ''
                            CHANGING  L_ANSWER.

  CHECK  L_ANSWER  =  C_ANSWER-YES.

  LOOP AT IT_MAIN WHERE MARK = 'X' AND
                        ICON = ICON_LED_INACTIVE.
    CLEAR :LT_T113, LT_T113[].

    MOVE-CORRESPONDING IT_MAIN TO LT_T113.
    LV_CNT = LV_CNT + 1.
    MOVE LV_CNT TO LV_CNTC.

    LT_T113-STATUS = 'A'.

*     생성로그
    LT_T113-CREATED_DATE = SY-DATUM.
    LT_T113-CREATED_TIME = SY-UZEIT.
    LT_T113-CREATED_BY   = SY-UNAME.

    CONCATENATE LT_T113-CREATED_DATE LT_T113-CREATED_TIME LV_CNTC
    INTO LT_T113-UTIME.
    APPEND LT_T113.

    MODIFY ZPPT0113 FROM TABLE LT_T113.

    IF SY-SUBRC = 0.
      IT_MAIN-ICON = ICON_LED_GREEN.
      LV_SCNT = LV_SCNT + 1.
    ELSE.
      IT_MAIN-ICON = ICON_LED_RED.
      IT_MAIN-MSG = TEXT-012.
    ENDIF.
    MODIFY IT_MAIN.

  ENDLOOP.

* 메시지 처리
  MOVE : LV_CNT TO  LV_CNTC,
         LV_SCNT TO LV_SCNTC.
  CONDENSE : LV_CNTC, LV_SCNTC.

  IF LV_CNT IS INITIAL.
    MESSAGE S030 .
  ELSE.
    CONCATENATE LV_CNTC TEXT-013 LV_SCNTC TEXT-015 TEXT-014
    INTO LV_MSG SEPARATED BY SPACE.
    MESSAGE S000 WITH LV_MSG.
  ENDIF.

  PERFORM ALV_REFRESH_TABLE_DISPLAY USING G_GRID1.

ENDFORM. " UC_0200-SAVE
