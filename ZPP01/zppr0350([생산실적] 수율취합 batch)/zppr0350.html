          meins TYPE mseg-meins,

         END OF lt_mseg.

  DATA : lv_tabix TYPE sy-tabix.

  PERFORM get_bwart.

*2. 생산오더 데이터
  REFRESH : gt_t260, lt_mseg.
  SELECT a~mblnr a~mjahr a~zeile a~werks a~aufnr a~matnr a~menge a~meins a~bwart
    INTO TABLE lt_mseg
    FROM mseg AS a INNER JOIN mkpf AS b ON a~mblnr = b~mblnr AND
                                           a~mjahr = b~mjahr
    WHERE a~werks IN so_werks AND
          b~budat IN so_budat AND
          a~bwart IN so_bwart AND
          a~meins = 'KG'
         %_HINTS ORACLE 'LEADING(T_01) USE_NL(T_00) INDEX(T_00 SAPSR3.MSEG~0)'. " 쿼리 수정 2023.09.25_1

        nt_t260-aufnr =                 lt_t260-aufnr = lt_mseg-aufnr.
        gt_t250-matnr = lt_t260-matnr = lt_mseg-matnr.

        gt_t250-ipmng = mt_t260-idqty = xt_t260-idqty = lt_mseg-menge.
        nt_t260-meins = gt_t250-meins = xt_t260-meins = xt_t260-meins = lt_mseg-meins.

        nt_t260-spmon = gt_t250-spmon = mt_t260-spmon = xt_t260-spmon = pa_spmon.
        nt_t260-aufnr =                 mt_t260-aufnr = xt_t260-aufnr = lt_mseg-aufnr.
        gt_t250-matnr = mt_t260-idnrk =                 lt_mseg-matnr.
        gt_t250-ipmng = mt_t260-idqty = xt_t260-idqty = lt_mseg-menge * -1.
        nt_t260-meins = gt_t250-meins = xt_t260-meins = xt_t260-meins = lt_mseg-meins.
        COLLECT : mt_t260, xt_t260, nt_t260.

    COLLECT : gt_t250.

  DELETE ADJACENT DUPLICATES FROM lt_t260 COMPARING aufnr.
  SORT nt_t260 BY aufnr.

  SORT mt_t260 BY aufnr.
  SORT xt_t260 BY aufnr.

  LOOP AT nt_t260.

    ENDIF.

      nt_t260-itqty = xt_t260-idqty.
    ELSE.
      CLEAR nt_t260-itqty.
    ENDIF.

        APPEND gt_t260.

      APPEND gt_t260.
    ENDIF.

  ENDLOOP.
