*----------------------------------------------------------------------*
FORM SELECT_MAIN_DATA .

  CLEAR : GT_MAIN[], GT_MAIN.
  SELECT A~WERKS  B~NAME1 AS WERKS_NM
         A~ARBPL  C~KTEXT AS ARBPL_NM
         A~VERWE  D~KTEXT AS VERWE_NM
         A~STAND  E~KTEXT AS STAND_NM
         A~VERAN  F~KTEXT AS VERAN_NM
         A~PLANV  A~VGWTS    A~VGE01       A~VGE02  A~VGE03
         A~VGE04  A~VGE05             A~VGE06       A~STEUS  A~FORT2
         A~BEGDA  A~ENDDA             A~AENAM_GRND  A~OBJTY  A~OBJID
    INTO CORRESPONDING FIELDS OF TABLE GT_MAIN
                  FROM CRHD  AS A
            INNER JOIN T001W AS B
                    ON A~WERKS = B~WERKS
            INNER JOIN CRTX AS C
                    ON A~OBJTY = C~OBJTY
                   AND A~OBJID = C~OBJID
                   AND C~SPRAS = SY-LANGU
            INNER JOIN TC30T AS D
                    ON A~VERWE = D~VERWE
                   AND D~SPRAS = SY-LANGU
            LEFT OUTER JOIN T499S AS E
                    ON A~WERKS = E~WERKS
                   AND A~STAND = E~STAND
            LEFT OUTER JOIN TC24 AS F
                    ON A~WERKS = F~WERKS
                   AND A~VERAN = F~VERAN
                 WHERE A~WERKS IN SO_WERKS
                   AND A~ARBPL IN SO_ARBPL
                   AND A~AENAM_GRND IN SO_AGRND
                   AND A~PLANV IN SO_PLANV
                   AND A~VERWE IN SO_VERWE
                   AND A~STAND IN SO_STAND
                   AND A~VERAN IN SO_VERAN
                   AND A~ENDDA >= SY-DATUM.

ENDFORM. " SELECT_MAIN_DATA

*&      Form  MAKE_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20).

  DATA : LV_LSTAR(20),
         LV_LEINH(20),
         LV_FORML(20),
         LV_NUM TYPE NUM02.

  FIELD-SYMBOLS: <VALUE1>, <VALUE2>, <VALUE3>.

  CLEAR : GT_RESULT[], GT_RESULT.
  LOOP AT GT_MAIN.
    LOOP AT GT_CRCO WHERE OBJTY = GT_MAIN-OBJTY
                      AND OBJID = GT_MAIN-OBJID .

      IF GT_MAIN-KOSTL IS INITIAL.
        GT_MAIN-KOKRS       = GT_CRCO-KOKRS   . "관리회계영역
        GT_MAIN-KOSTL       = GT_CRCO-KOSTL   . "코스트센터
        GT_MAIN-KOSTL_NM    = GT_CRCO-KOSTL_NM. "코스트센터 내역
      ENDIF.

      CLEAR : LV_LSTAR, LV_LEINH, LV_FORML, LV_NUM.
      LV_NUM = GT_CRCO-LANUM+2(2).
      CONCATENATE 'LSTAR_'  LV_NUM INTO LV_LSTAR.
      CONCATENATE 'LEINH_'  LV_NUM INTO LV_LEINH.
      CONCATENATE 'FORML_'  LV_NUM INTO LV_FORML.

      UNASSIGN : <VALUE1>, <VALUE2>, <VALUE3>.
      ASSIGN COMPONENT LV_LSTAR OF STRUCTURE GT_MAIN TO <VALUE1>.
      ASSIGN COMPONENT LV_LEINH OF STRUCTURE GT_MAIN TO <VALUE2>.
      ASSIGN COMPONENT LV_FORML OF STRUCTURE GT_MAIN TO <VALUE3>.

      CHECK <VALUE1> IS ASSIGNED AND
            <VALUE2> IS ASSIGNED AND
            <VALUE2> IS ASSIGNED.

      <VALUE1> = GT_CRCO-LSTAR.
      <VALUE2> = GT_CRCO-LEINH.
      <VALUE3> = GT_CRCO-FORML.
    ENDLOOP.

    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
    APPEND GT_RESULT. CLEAR : GT_RESULT.
  ENDLOOP.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S021 WITH LV_SCNT.

ENDFORM. " MAKE_DATA
