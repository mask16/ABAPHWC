FORM MOVE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20),
         LV_XCHAR,
         LV_CHARG(10).

  DATA : LV_BARXX    TYPE ZEBARCODE,

         LV_MXBARCD  TYPE ZEBARCODE,
         LV_MXLOTNM  TYPE ZPPT0170-PCHARG,
         LV_BARNC(4) TYPE N,
         LV_LOTNC(4) TYPE N,
         LV_RET.

  DATA : LT_BARCD LIKE TABLE OF GT_BARCD WITH HEADER LINE.

  CHECK GT_UPF[] IS NOT INITIAL.
*  SORT GT_UPF BY WERKS MATNR.
  CLEAR : IT_MAIN[], IT_MAIN, GT_BARCD, GT_BARCD[].
  LOOP AT GT_UPF INTO GS_UPF.
    CLEAR : IT_MAIN.
    MOVE-CORRESPONDING GS_UPF TO IT_MAIN.

    IT_MAIN-WERKS = PA_WERKS.
    IT_MAIN-ITQTY = IT_MAIN-REMNG = IT_MAIN-MENGE = IT_MAIN-ISCNT = 1.
    IT_MAIN-PCHARG = IT_MAIN-CHARG.
    IT_MAIN-ICODE1 = IT_MAIN-CHARG+0(1).
    IT_MAIN-ICODE2 = IT_MAIN-CHARG+1(1).
    IT_MAIN-LGUBUN = 'M'.
    IT_MAIN-MEINS = 'EA'.

    CLEAR LV_XCHAR.
    SELECT SINGLE XCHAR INTO LV_XCHAR FROM MARC
           WHERE WERKS = IT_MAIN-WERKS AND
                 MATNR = IT_MAIN-MATNR.
    IF LV_XCHAR IS INITIAL.
      IT_MAIN-CHARG = ''.
    ENDIF.

    SELECT SINGLE A~ZMAKTX A~GROES B~MAKTX
           INTO (IT_MAIN-ZMAKTX, IT_MAIN-GROES, IT_MAIN-MAKTX)
           FROM MARA AS A INNER JOIN MAKT AS B ON A~MATNR = B~MATNR
           WHERE A~MATNR = IT_MAIN-MATNR AND
                 B~SPRAS = SY-LANGU.
    IF SY-SUBRC <> 0.
      CONCATENATE TEXT-F01 TEXT-010 INTO IT_MAIN-MSG SEPARATED BY SPACE.
      IT_MAIN-ICON = ICON_LED_RED.
      APPEND IT_MAIN. CONTINUE.
    ENDIF.

    SELECT SINGLE IDESC INTO IT_MAIN-IDESC1 FROM ZPPT0070
           WHERE HGUBUN = '1' AND IGUBUN = IT_MAIN-ICODE1.
    IF SY-SUBRC <> 0.
      CONCATENATE TEXT-F02 TEXT-010 INTO IT_MAIN-MSG SEPARATED BY SPACE.
      IT_MAIN-ICON = ICON_LED_RED.
      APPEND IT_MAIN. CONTINUE.
    ENDIF.

    SELECT SINGLE IDESC INTO IT_MAIN-IDESC2 FROM ZPPT0070
           WHERE HGUBUN = '2' AND IGUBUN = IT_MAIN-ICODE2.
    IF SY-SUBRC <> 0.
      CONCATENATE TEXT-F03 TEXT-010 INTO IT_MAIN-MSG SEPARATED BY SPACE.
      IT_MAIN-ICON = ICON_LED_RED.
      APPEND IT_MAIN. CONTINUE.
    ENDIF.

    SELECT SINGLE PCHARG INTO LV_CHARG FROM ZPPT0170
           WHERE PCHARG = IT_MAIN-PCHARG.
    IF SY-SUBRC EQ 0.
      CONCATENATE TEXT-F12 TEXT-011 TEXT-010 INTO IT_MAIN-MSG SEPARATED BY SPACE.
      IT_MAIN-ICON = ICON_LED_RED.
      APPEND IT_MAIN. CONTINUE.
    ENDIF.

*   Max 바코드/Max 생산 LOT 번호
    READ TABLE GT_BARCD WITH KEY WERKS = IT_MAIN-WERKS
                                 MATNR = IT_MAIN-MATNR
                                 ICODE1 = IT_MAIN-ICODE1
                                 ICODE2 = IT_MAIN-ICODE2
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      CLEAR : LT_BARCD, LT_BARCD[].
      LOOP AT GT_BARCD.
        MOVE-CORRESPONDING GT_BARCD TO LT_BARCD.
        APPEND LT_BARCD.
      ENDLOOP.
      SORT LT_BARCD DESCENDING BY BARCD.
      READ TABLE LT_BARCD INDEX 1.
      LV_BARCD = LT_BARCD-BARCD.
    ELSE.
      LV_BARXX+0(4) = IT_MAIN-WERKS.
      LV_BARXX+4(8) = IT_MAIN-MATNR+10(8).
      LV_BARXX+12(1) = '0'.
      LV_BARXX+13(1) = IT_MAIN-ICODE1.
      LV_BARXX+14(1) = IT_MAIN-ICODE2.
      LV_BARXX+15(1) = '%'.

      SELECT SINGLE MAX( BARCODE ) INTO LV_BARCD
             FROM ZPPT0170
             WHERE WERKS = PA_WERKS AND
                   BARCODE LIKE LV_BARXX.

      LV_BARCD+0(15) = LV_BARXX(15).

    ENDIF.


    LV_BARNC = LV_BARCD+15(4).
    LV_BARNC = LV_BARNC + 1.
    LV_BARCD+15(4) = LV_BARNC.
    IT_MAIN-BARCODE = LV_BARCD.

    IT_MAIN-ICON = ICON_LED_INACTIVE.
    APPEND IT_MAIN.

    GT_BARCD-WERKS   = IT_MAIN-WERKS.
    GT_BARCD-MATNR   = IT_MAIN-MATNR.
    GT_BARCD-ICODE1  = IT_MAIN-ICODE1.
    GT_BARCD-ICODE2  = IT_MAIN-ICODE2.
    GT_BARCD-BARCD   = IT_MAIN-BARCODE.
    APPEND GT_BARCD.
*    SORT GT_BARCD DESCENDING BY BARCD.
    SORT GT_BARCD BY WERKS MATNR ICODE1 ICODE2.

  ENDLOOP.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE IT_MAIN LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S038 WITH LV_SCNT.

ENDFORM. " MOVE_DATA
