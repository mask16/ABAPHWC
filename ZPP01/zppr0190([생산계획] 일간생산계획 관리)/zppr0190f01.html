FORM SET_INITIAL_DATA .

* 실행일 주차

      EXPORTING

        WEEK         = PA_SPYWK
      EXCEPTIONS
        DATE_INVALID = 1
        OTHERS       = 2.
    IF SY-SUBRC <> 0.

    ENDIF.
  ENDIF.

  CHECK GV_SPYWK <> PA_SPYWK.
  CLEAR : SO_PLNDT, SO_PLNDT[].
  CALL FUNCTION 'WEEK_GET_FIRST_DAY'
    EXPORTING
      WEEK         = PA_SPYWK

  IF SY-SUBRC <> 0.

  ENDIF.

  APPEND SO_PLNDT.

FORM MAKE_DATA .

           WHERE WERKS = GT_RESULT-WERKS AND DISPO = GT_RESULT-DISPO.

    IF PA_G0003 = 'X'.

      SELECT SINGLE A~VTWEG A~KUNNR A~VDATU A~TODAT A~BSTKD_E
             INTO (GT_RESULT-VTWEG, GT_RESULT-KUNNR, GT_RESULT-VDATU,
                   GT_RESULT-TODAT, GT_RESULT-BSTKD_E)
             FROM ZSDT0301 AS A INNER JOIN ZPPT0080 AS B ON A~SPNUM = B~SPNUM
                                                        AND A~SPDAT = B~SPDAT
                                                        AND A~SPYWK = B~SPYWK
             WHERE A~SPNUM = GT_RESULT-SPNUM AND
                   A~SPYWK = PA_SPYWK.

      ENDIF.

             FROM KNA1
             WHERE KUNNR = GT_RESULT-KUNNR.
    ENDIF.

*   계획수량

*   구분코드
    PERFORM GUBUN_CODE USING 'P'.

*     실적수량

  DO GV_DAYS TIMES.

    CONCATENATE 'GT_RESULT-MNG' GV_SEQ1 INTO GV_FIELD1.
    ASSIGN (GV_FIELD1) TO <FS1>.
    CLEAR <FS1>.

    IF P_GB = 'P'.
      READ TABLE GT_T140 WITH KEY ZPLNUM = GT_RESULT-ZPLNUM
                                  SEQNS  = GT_RESULT-SEQNS

                                  BINARY SEARCH.

    ELSEIF P_GB = 'A'.
      LOOP AT GT_T110 WHERE ZPLNUM = GT_RESULT-ZPLNUM.

          <FS1> = <FS1> + GT_T110-MENGE.
        ENDIF.
      ENDLOOP.
    ENDIF.

    GT_RESULT-MNG00 = GT_RESULT-MNG00 + <FS1>.
    GV_DATE1 = GV_DATE1 + 1.
    GV_SEQ1 = GV_SEQ1 + 1.

  ELSEIF P_GB = 'A'.
