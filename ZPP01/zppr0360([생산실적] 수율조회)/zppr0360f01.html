FORM MAKE_DATA .

  DATA : LV_CNT TYPE I,
         LV_SCNT(20),
         LV_MENG1    TYPE ZPPT0270-DGMNG.

  CLEAR : GT_RESULT[], GT_RESULT.
  LOOP AT GT_MAIN.
    SELECT SINGLE MAKTX INTO GT_MAIN-MAKTX FROM MAKT
      WHERE MATNR = GT_MAIN-MATNR AND
            SPRAS = SY-LANGU.

*   0 값 제품 제외
    IF PA_EXCLD = 'X' AND
       GT_MAIN-MAQTY IS INITIAL AND
       GT_MAIN-WBMNG IS INITIAL AND
       GT_MAIN-WEMNG IS INITIAL AND
       GT_MAIN-WRMNG IS INITIAL AND
       GT_MAIN-FBMNG IS INITIAL AND
       GT_MAIN-FEMNG IS INITIAL AND
       GT_MAIN-FRMNG IS INITIAL AND
       GT_MAIN-BNMNG IS INITIAL AND
       GT_MAIN-DGMNG IS INITIAL.
      CONTINUE.
    ENDIF.
    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
    APPEND GT_RESULT.

    GT_RESULT-MATNR = 'ALL'.
    GT_RESULT-DISPF = 'ALL'.
    CLEAR : GT_RESULT-STUFE, GT_RESULT-MATNX, GT_RESULT-MAKTX,
            GT_RESULT-IDNRK, GT_RESULT-VERID, GT_RESULT-ZGROP,
            GT_RESULT-MATKL, GT_RESULT-DISPO,
            GT_RESULT-ZMAKTX, GT_RESULT-OTMNG, GT_RESULT-IDQTY,
            GT_RESULT-IPMNG, GT_RESULT-FRATE, GT_RESULT-YRATE, GT_RESULT-ARATE,
            GT_RESULT-BSMNG, GT_RESULT-ENMNG, GT_RESULT-ACTCK,
            GT_RESULT-CREATED_DATE, GT_RESULT-CREATED_TIME,
            GT_RESULT-CREATED_BY, GT_RESULT-CHANGED_DATE,
            GT_RESULT-CHANGED_TIME, GT_RESULT-CHANGED_BY.
    COLLECT GT_RESULT.
    CLEAR : GT_RESULT.
  ENDLOOP.

  LOOP AT GT_RESULT WHERE MATNR = 'ALL'.
*     수율
      LV_MENG1 = ( ( GT_RESULT-WBMNG + GT_RESULT-WRMNG - GT_RESULT-WEMNG ) +
                   ( GT_RESULT-FBMNG + GT_RESULT-FRMNG - GT_RESULT-FEMNG ) +
                     GT_RESULT-BNMNG ).
      IF LV_MENG1 IS NOT INITIAL.
        GT_RESULT-FRATE = GT_RESULT-MAQTY / LV_MENG1.
      ENDIF.
      GT_RESULT-FRATE = GT_RESULT-FRATE * 100.
*     수율2
      LV_MENG1 = ( ( GT_RESULT-WBMNG + GT_RESULT-WRMNG - GT_RESULT-WEMNG ) +
                   ( GT_RESULT-FBMNG + GT_RESULT-FRMNG - GT_RESULT-FEMNG ) ).
      IF LV_MENG1 IS NOT INITIAL.
        GT_RESULT-ARATE = GT_RESULT-MAQTY / LV_MENG1.
      ENDIF.
      GT_RESULT-ARATE = GT_RESULT-ARATE * 100.
    MODIFY GT_RESULT.
  ENDLOOP.

*조회건수 DISPLAY
  CLEAR : GV_CNT.
  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
  WRITE GV_CNT TO LV_SCNT.
  MESSAGE S021 WITH LV_SCNT.

ENDFORM. " MAKE_DATA
