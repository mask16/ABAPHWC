FORM MAKE_DATA .

  CLEAR : GT_RESULT[], GT_RESULT.

           FROM MARC AS A INNER JOIN T024D AS B ON A~WERKS = B~WERKS AND
                                                   A~DISPO = B~DISPO
           WHERE A~WERKS = GT_RESULT-WERKS AND
                 A~MATNR = GT_RESULT-MATNR.
*   자재내역
    SELECT SINGLE MAKTX INTO GT_RESULT-MAKTX FROM MAKT
        WHERE SPRAS  = SY-LANGU AND
              MATNR  = GT_RESULT-MATNR.

    MOVE-CORRESPONDING GT_MAIN TO GT_RESULT.
    APPEND GT_RESULT. CLEAR : GT_RESULT.

  ENDLOOP.

  DESCRIBE TABLE GT_RESULT LINES GV_CNT.
