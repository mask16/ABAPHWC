*&---------------------------------------------------------------------*
*&      Form  SELECT_KNA1
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_kna1 .

  DATA: lt_data_0100 LIKE gt_data_0100,
        ls_data_0100 LIKE LINE OF gt_data_0100.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.
* 생성
  CLEAR: lt_data_0100.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM kna1
   WHERE kunnr IN so_kunnr
     AND ktokd IN ('A100', 'A200')
     AND EXISTS ( SELECT objectid
                    FROM cdhdr
                   WHERE objectid   EQ kna1~kunnr
                     AND udate      IN so_udate
                     AND objectclas EQ 'DEBI'
                     AND tcode      IN ('XD01', 'VD01', 'FD01') ).
  SORT lt_data_0100 BY kunnr.
  LOOP AT lt_data_0100 INTO ls_data_0100.
    ls_data_0100-flag = 'C'.
    APPEND ls_data_0100 TO gt_data_0100.
  ENDLOOP.
* 변경
  CLEAR: lt_data_0100.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM kna1
   WHERE kunnr IN so_kunnr
     AND ktokd IN ('A100', 'A200')
     AND EXISTS ( SELECT objectid
                    FROM cdhdr
                   WHERE objectid   EQ kna1~kunnr
                     AND udate      IN so_udate
                     AND objectclas EQ 'DEBI'
                     AND tcode      IN ('XD02','VD02', 'FD02') ).
  SORT lt_data_0100 BY kunnr.
  LOOP AT lt_data_0100 INTO ls_data_0100.
    ls_data_0100-flag = 'U'.
    APPEND ls_data_0100 TO gt_data_0100.
  ENDLOOP.
  LOOP AT gt_data_0100 INTO ls_data_0100.
*   변경일자
    ls_data_0100-udate = sy-datum.
*   국가명
    PERFORM get_land1_text USING ls_data_0100-land1
                        CHANGING ls_data_0100-landx.
*   주소
    CONCATENATE ls_data_0100-ort01 ' ' ls_data_0100-stras
           INTO ls_data_0100-address RESPECTING BLANKS.
*   사무소
    PERFORM get_knvv_vkbur USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-vkbur.
*   사무소명
    PERFORM get_vkbur_text USING ls_data_0100-vkbur
                        CHANGING ls_data_0100-bezei.
*   영업사원
    PERFORM get_knvp_kunn2 USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-spcode.
*   영업사원명
    PERFORM get_kunnr_text USING ls_data_0100-spcode
                        CHANGING ls_data_0100-spname.
    ls_data_0100-ernam = sy-uname.
    MODIFY gt_data_0100 FROM ls_data_0100.
  ENDLOOP.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_KNA1

FORM send_data_0100 USING p_werks.

  DATA: lt_data_0100 LIKE gt_data_0100,
        lt_zsdt9510  TYPE STANDARD TABLE OF zsdt9510,
        lt_zsds9510  TYPE STANDARD TABLE OF zsds9510,
        ls_data_0100 LIKE LINE OF lt_data_0100,
        ls_zsdt9510  LIKE LINE OF lt_zsdt9510,
        ls_zsds9510  LIKE LINE OF lt_zsds9510,
        ls_zsds9500  TYPE zsdt9500,
        lv_datum     TYPE sy-datum,
        lv_uzeit     TYPE sy-uzeit,
        lv_udate     TYPE char01,
        lv_all       TYPE char01,
        lv_ret       TYPE bapi_mtype,
        lv_msg       TYPE char100.

  PERFORM sapgui_progress_indicator USING 1 '데이터 전송중..'.

  lv_uzeit = sy-uzeit.

  ls_zsds9500-trsdt = lv_datum.
* 전송시각
  ls_zsds9500-trstm = lv_uzeit.
* RFC Function 명
  ls_zsds9500-fname = 'ZSD_IF_MES_CUSTOMER'.
  INSERT zsdt9500 FROM ls_zsds9500.

  IF lines( gt_data_0100 ) EQ 0.
*   전송결과
    ls_zsds9500-trsrt = 'W'.
    ls_zsds9500-trsms = '전송할 정보가 없습니다.'.
    ls_zsds9500-ernam = sy-uname.
