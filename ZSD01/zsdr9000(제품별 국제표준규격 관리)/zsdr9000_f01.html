*&      Form  SELECT_DATA_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM SELECT_DATA_0100 .

  DATA: LT_DATA_0100   LIKE GT_DATA_0100,
        LR_BELNR       TYPE RANGE OF ZSDT0130-BELNR,
        LS_BELNR       LIKE LINE OF LR_BELNR,
        LS_DATA_0100   LIKE LINE OF GT_DATA_0100,
        LV_DATUM       TYPE DATUM,
        LV_ROWCOUNT    TYPE CHAR10.

  CLEAR: GT_DATA_0100.

  PERFORM SAPGUI_PROGRESS_INDICATOR USING 1 '조회중...'.

  SELECT A~MATNR   A~AWS_CODE A~JIS_CODE  A~KS_CODE  A~MEMO
         A~ERNAM   A~ERDAT    A~ERZET     A~AENAM    A~AEDAT
         A~AEZET
         B~ZMAKTX  B~GROES    B~MATKL     B~PRDHA
         C~MAKTX
    INTO CORRESPONDING FIELDS OF TABLE LT_DATA_0100
    FROM ZSDT9000 AS A INNER JOIN
         MARA     AS B
      ON A~MATNR = B~MATNR INNER JOIN
         MAKT     AS C
      ON B~MATNR = C~MATNR
     AND C~SPRAS = SY-LANGU
   WHERE B~MATKL IN SO_MATKL
     AND B~PRDHA IN SO_PRDHA
     AND A~MATNR IN SO_MATNR
     AND A~AWS_CODE IN SO_AWSCD
     AND A~JIS_CODE IN SO_JISCD
     AND A~KS_CODE  IN SO_KS_CD
     AND A~LOEVM <> 'X'.

  DATA: BEGIN OF LT_T023T OCCURS 0,
    MATKL TYPE T023T-MATKL,
    WGBEZ TYPE T023T-WGBEZ,
  END OF LT_T023T,
  BEGIN OF LT_T179T OCCURS 0,
    PRDHA TYPE T179T-PRODH,
    VTEXT TYPE T179T-VTEXT,
  END OF LT_T179T.

  IF LT_DATA_0100[] IS NOT INITIAL.

    CLEAR: LT_T023T.
    SELECT MATKL WGBEZ
      INTO TABLE LT_T023T
      FROM T023T
       FOR ALL ENTRIES IN LT_DATA_0100
     WHERE MATKL = LT_DATA_0100-MATKL
       AND SPRAS = SY-LANGU.
    SORT LT_T023T.
    DELETE ADJACENT DUPLICATES FROM LT_T023T.

    CLEAR: LT_T179T.
    SELECT PRODH AS PRDHA VTEXT
      INTO TABLE LT_T179T
      FROM T179T
       FOR ALL ENTRIES IN LT_DATA_0100
     WHERE PRODH = LT_DATA_0100-PRDHA
       AND SPRAS = SY-LANGU.
    SORT LT_T179T.
    DELETE ADJACENT DUPLICATES FROM LT_T179T.

  ENDIF.
  LOOP AT LT_DATA_0100 INTO LS_DATA_0100.

    READ TABLE LT_T023T WITH KEY MATKL = LS_DATA_0100-MATKL
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      LS_DATA_0100-WGBEZ = LT_T023T-WGBEZ.
    ENDIF.

    READ TABLE LT_T179T WITH KEY PRDHA = LS_DATA_0100-PRDHA
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      LS_DATA_0100-VTEXT = LT_T179T-VTEXT.
    ENDIF.

    APPEND LS_DATA_0100 TO GT_DATA_0100.
