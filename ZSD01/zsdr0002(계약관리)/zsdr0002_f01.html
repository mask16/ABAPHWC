*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM make_fieldcat USING p_gub p_fname p_con.

  IF p_gub = 'S'.
    CLEAR gs_fcat.
  ENDIF.

  DATA l_col(40).
  FIELD-SYMBOLS <fs>.
  CONCATENATE 'GS_FCAT-' p_fname  INTO l_col.
  ASSIGN      (l_col)         TO       <fs>.
  MOVE         p_con          TO       <fs>.

  IF p_gub = 'E'.
    APPEND gs_fcat TO <fcat>.
  ENDIF.

ENDFORM. " MAKE_FIELDCAT

*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: lt_data_0100   LIKE gt_data_0100,
        lr_gbsta       TYPE RANGE OF vbup-gbsta,
        ls_data_0100   LIKE LINE OF gt_data_0100,
        ls_gbsta       LIKE LINE OF lr_gbsta,
        lv_rowcount    TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

* 모든오더 미결판매오더 조건 생성.
  CASE 'X'.
    WHEN pa_r1.
*   모든오더
      ls_gbsta-sign = 'I'.
      ls_gbsta-option = 'EQ'.
      ls_gbsta-low = 'C'.
      APPEND ls_gbsta TO lr_gbsta.
    WHEN pa_r2.
*   미결판매오더

      ls_gbsta-option = 'NE'.
      ls_gbsta-low = 'C'.
      APPEND ls_gbsta TO lr_gbsta.
  ENDCASE.

  SELECT a~auart
         a~vbeln
         a~vtweg
         a~erdat
         a~vdatu
         a~angdt
         a~bnddt
         a~kunnr
         a~spart
         a~augru
         a~bstnk
         a~bstdk
         a~vkbur
         a~vkgrp
         a~kkber
         a~guebg
         a~submi
         a~gueen
         b~posnr
         b~matnr
         b~kdmat
         b~kwmeng
         b~kbmeng
         b~meins
         b~netpr    AS onetpr
         b~kzwi2
         b~kzwi3
         b~kzwi4
         b~netwr
         b~mwsbp
         b~waerk
         b~abgru
         b~prodh
         b~mvgr1
         b~mvgr2
         b~mvgr3
         b~mvgr4
         b~mvgr5
         b~werks
         b~lgort
         b~vgbel
         c~prsdt
         c~zterm
         c~inco1
         c~inco2
         c~bzirk
         c~kdgrp
         c~ktgrd
         d~mtart
         d~zmaktx
         d~groes
         e~prat1
         f~gbsta
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM vbak    AS a
   INNER JOIN
         vbap    AS b
      ON a~vbeln EQ b~vbeln
   INNER JOIN
         vbkd    AS c
      ON a~vbeln EQ c~vbeln
     AND c~posnr EQ space
   INNER JOIN
         mara    AS d
      ON b~matnr EQ d~matnr
   INNER JOIN
         mvke    AS e
      ON b~matnr EQ e~matnr
     AND e~vkorg EQ '1000'
     AND e~vtweg EQ '10'
   INNER JOIN
         vbup    AS f
      ON b~vbeln EQ f~vbeln
     AND b~posnr EQ f~posnr
   WHERE a~vbeln IN so_vbeln
     AND a~erdat IN so_erdat
     AND a~ernam IN so_ernam
     AND a~auart IN so_auart
     AND a~augru IN so_augru
     AND a~vkorg IN so_vkorg
     AND a~vtweg IN so_vtweg
     AND a~spart IN so_spart
     AND a~vkgrp IN so_vkgrp
     AND a~vkbur IN so_vkbur
     AND a~vdatu IN so_vdatu
     AND a~bstnk IN so_bstnk
     AND a~kunnr IN so_kunnr
     AND a~trvog EQ '4'
     AND b~matnr IN so_matnr
     AND b~abgru IN so_abgru
     AND b~prodh IN so_prodh
     AND b~kdmat IN so_kdmat
     AND c~bzirk IN so_bzirk
     AND d~mtart IN so_mtart
     AND e~mvgr1 IN so_mvgr1
     AND e~mvgr2 IN so_mvgr2
     AND e~mvgr3 IN so_mvgr3
     AND e~mvgr4 IN so_mvgr4
     AND e~mvgr5 IN so_mvgr5
     AND f~gbsta IN lr_gbsta.

* 추가정보 조회
  LOOP AT lt_data_0100 INTO ls_data_0100.
*   영업사원(본사)
    PERFORM get_vbpa_kunnr USING ls_data_0100-vbeln
                                 'S1'
                        CHANGING ls_data_0100-yunnr1.
*   영업사원(현지)
    PERFORM get_vbpa_kunnr USING ls_data_0100-vbeln
                                 'S2'
                        CHANGING ls_data_0100-yunnr2.
    MODIFY lt_data_0100 FROM ls_data_0100.
  ENDLOOP.

* 추가 조건, TEXT정보 조회
  LOOP AT lt_data_0100 INTO ls_data_0100
                      WHERE ( yunnr1 IN so_yunnr
                         OR   yunnr2 IN so_yunnr ).

      WHEN 'C'.

        ls_data_0100-stats = icon_led_green.
      WHEN OTHERS.
        ls_data_0100-stats = icon_led_yellow.
    ENDCASE.
*   납품처
    PERFORM get_vbpa_kunnr USING ls_data_0100-vbeln
                                 'WE'
                        CHANGING ls_data_0100-sunnr.
*   실수요자(국내)
    PERFORM get_vbpa_kunnr USING ls_data_0100-vbeln
                                 'ZA'
                        CHANGING ls_data_0100-punnr.
    IF ls_data_0100-punnr IS INITIAL.
*     실수요자(현지)
      PERFORM get_vbpa_kunnr USING ls_data_0100-vbeln
                                   'ZB'
                          CHANGING ls_data_0100-punnr.
    ENDIF.
*   매출액(확정판가 + 부가세)
    ls_data_0100-sales = ls_data_0100-netwr + ls_data_0100-mwsbp.
*   세부사항
    PERFORM get_vbkd_bstkd_e USING ls_data_0100-vbeln
                                   ls_data_0100-posnr
                          CHANGING ls_data_0100-bstkd_e.
*   대금청구문서 조회
    PERFORM get_vbrk CHANGING ls_data_0100.
*   환율(2차)
    IF ls_data_0100-kurrf EQ space.
      PERFORM get_vbkd_kursk USING ls_data_0100-vbeln
                                   space
                          CHANGING ls_data_0100-kurrf.
    ENDIF.
*   환산 매출액(KRW)
    CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
      EXPORTING
        date             = sy-datum
        foreign_amount   = ls_data_0100-sales
        foreign_currency = ls_data_0100-waerk
        local_currency   = 'KRW'
        rate             = ls_data_0100-kurrf
      IMPORTING
        local_amount     = ls_data_0100-esales.

*   환산화폐
    ls_data_0100-waekr = 'KRW'.
*   판매처명
    PERFORM get_kunnr_text USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-name1.

    PERFORM get_kunnr_text USING ls_data_0100-sunnr
                        CHANGING ls_data_0100-name2.

                        CHANGING ls_data_0100-maktx.

    PERFORM get_augru_text USING ls_data_0100-augru

    ls_data_0100-prodh1 = ls_data_0100-prodh(5).
    PERFORM get_prodh_text USING ls_data_0100-prodh1
                        CHANGING ls_data_0100-vtext1.

*&      Form  EXCEL_0100
*&---------------------------------------------------------------------*
*       계약서
*----------------------------------------------------------------------*
FORM excel_0100 .

  DATA: lt_rows         TYPE lvc_t_row,
        lt_head_0100    LIKE gt_data_0100,
        lt_excel_0100   TYPE STANDARD TABLE OF typ_excel_0100,
        lt_excel_0101   TYPE STANDARD TABLE OF typ_excel_0101,
        ls_rows         LIKE LINE OF lt_rows,
        ls_head_0100    LIKE LINE OF lt_head_0100,
        ls_excel_0100   LIKE LINE OF lt_excel_0100,
        ls_excel_0100_1 LIKE LINE OF lt_excel_0100,
        ls_excel_0101   LIKE LINE OF lt_excel_0101,
        ls_data_0100    LIKE LINE OF gt_data_0100,
        lv_title        TYPE char20,
        lv_answer       TYPE char01.

  CALL METHOD gv_grid_0100->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows.

* 선택행 체크
  IF lines( lt_rows ) LE 0.
    MESSAGE s006.
    EXIT.
  ENDIF.

* 선택한 행의 데이터로 테이블생성
  LOOP AT lt_rows INTO ls_rows.
    CLEAR: ls_data_0100.
    READ TABLE gt_data_0100 INDEX ls_rows-index
                             INTO ls_data_0100.
    APPEND ls_data_0100 TO lt_head_0100.
  ENDLOOP.

* 계약문서 기준으로 헤더생성
  SORT lt_head_0100 BY vbeln.
  DELETE ADJACENT DUPLICATES
    FROM lt_head_0100 COMPARING vbeln.

  LOOP AT lt_head_0100 INTO ls_head_0100.
    CLEAR: lt_excel_0100, lt_excel_0101.
    CASE ls_head_0100-vtweg.
      WHEN '10'.
        LOOP AT gt_data_0100 INTO ls_data_0100
                            WHERE vbeln EQ ls_head_0100-vbeln.
          CLEAR: ls_excel_0100, ls_excel_0100_1.
          MOVE-CORRESPONDING ls_data_0100 TO ls_excel_0100.
          WRITE ls_data_0100-kwmeng TO ls_excel_0100-kwmeng UNIT ls_data_0100-meins.
          WRITE ls_data_0100-onetpr TO ls_excel_0100-onetpr CURRENCY ls_data_0100-waerk.
          WRITE ls_data_0100-netwr TO ls_excel_0100-netwr CURRENCY ls_data_0100-waerk.
*         자재별 sum
          READ TABLE lt_excel_0100 WITH KEY vbeln = ls_excel_0100-vbeln
                                            matnr = ls_excel_0100-matnr
                                       INTO ls_excel_0100_1.
          IF sy-subrc EQ 0.
            ADD ls_excel_0100-kwmeng TO ls_excel_0100_1-kwmeng.
            ADD ls_excel_0100-netwr TO ls_excel_0100_1-netwr.

            MODIFY lt_excel_0100 FROM ls_excel_0100_1
              TRANSPORTING kwmeng netwr
             WHERE vbeln = ls_excel_0100-vbeln
               AND matnr = ls_excel_0100-matnr.
            CONTINUE.
          ENDIF.
*         사무소 정보
          PERFORM get_adrc USING ls_excel_0100-vkbur
                        CHANGING ls_excel_0100-city1
                                 ls_excel_0100-tel_number
                                 ls_excel_0100-fax_number.
*         결재조건 내역
          PERFORM get_t052u_text1 USING ls_excel_0100-zterm
                               CHANGING ls_excel_0100-text1.
*         담당자 정보
          PERFORM get_kunnr_info USING ls_excel_0100-yunnr1
                              CHANGING ls_excel_0100-name2
                                       ls_excel_0100-telf1
                                       ls_excel_0100-telf2
                                       ls_excel_0100-smtp_addr.

          APPEND ls_excel_0100 TO lt_excel_0100.
        ENDLOOP.
