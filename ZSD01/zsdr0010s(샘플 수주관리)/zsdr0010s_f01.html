FORM set_default_values .

* 생성일자
  CLEAR: so_erdat, so_erdat[].
  so_erdat-sign = 'I'.
  so_erdat-option = 'BT'.
  so_erdat-low = sy-datum - 14.
  so_erdat-high = sy-datum.
  APPEND so_erdat.


  DATA : l_vkorg TYPE likp-vkorg.
  CLEAR: l_vkorg.
  SELECT SINGLE parva INTO l_vkorg
    FROM usr05
   WHERE bname = sy-uname
     AND parid = 'VKO'.

  IF l_vkorg IS NOT INITIAL.
    CLEAR: so_vkorg, so_vkorg[].
    so_vkorg-sign   = 'I'.
    so_vkorg-option = 'EQ'.
    so_vkorg-low    = l_vkorg.
    APPEND so_vkorg.
  ELSE.
    CLEAR: so_vkorg, so_vkorg[].
    so_vkorg-sign   = 'I'.
    so_vkorg-option = 'EQ'.
    so_vkorg-low    = '1000'.
    APPEND so_vkorg.
  ENDIF.


  CLEAR : gv_vkbur.
  IF sy-uname(3) = 'HBC'.
    gv_vkbur = '*'.
  ELSE.
    SELECT SINGLE vkbur
        INTO gv_vkbur
        FROM zsdt9010
        WHERE uname = sy-uname.

    IF gv_vkbur NE '*' .
      CLEAR : so_vkbur, so_vkbur[].
      so_vkbur-sign = 'I'.
      so_vkbur-option = 'EQ'.
      so_vkbur-low  = gv_vkbur.
      APPEND so_vkbur.
    ENDIF.
  ENDIF.

ENDFORM. " SET_DEFAULT_VALUES
