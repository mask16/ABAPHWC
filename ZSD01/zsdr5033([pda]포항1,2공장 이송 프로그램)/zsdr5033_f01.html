FORM ALV_DATA_CHANGED USING PR_DATA_CHANGED TYPE REF TO
                             CL_ALV_CHANGED_DATA_PROTOCOL
                             PR_SENDER      TYPE REF TO CL_GUI_ALV_GRID.

  DATA : LS_MOD_CELLS  TYPE LVC_S_MODI.
  DATA : LS_DEL_ROWS   TYPE LVC_S_MOCE.

  DATA : LV_MAKTX LIKE MAKT-MAKTX,
             LV_FNAME TYPE LVC_FNAME,
             LV_NAME2 LIKE T001W-NAME2,
             LV_MEINS LIKE MARA-MEINS,
             LV_MATKL LIKE MARA-MATKL.

  CLEAR : LV_FNAME, LV_MAKTX, LV_NAME2, LV_MEINS, LV_MATKL.

  LOOP AT PR_DATA_CHANGED->MT_GOOD_CELLS INTO LS_MOD_CELLS.
    CASE LS_MOD_CELLS-FIELDNAME.

      WHEN 'MATNR'.
        LV_FNAME = 'MAKTX'.
        SELECT SINGLE MAKTX INTO LV_MAKTX
           FROM MAKT
         WHERE MATNR EQ LS_MOD_CELLS-VALUE
             AND SPRAS EQ '3'.

        IF LV_MAKTX IS NOT INITIAL.
          CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = LV_FNAME
              I_VALUE     = LV_MAKTX.
        ENDIF.

        CLEAR LV_FNAME.
        LV_FNAME = 'MEINS'.
        SELECT SINGLE MEINS INTO LV_MEINS
           FROM MARA
        WHERE MATNR EQ LS_MOD_CELLS-VALUE.

        IF LV_MEINS IS NOT INITIAL.
          CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = LV_FNAME
              I_VALUE     = LV_MEINS.
        ENDIF.

        CLEAR LV_FNAME.
        LV_FNAME = 'MATKL'.
        SELECT SINGLE MATKL INTO LV_MATKL
           FROM MARA
         WHERE MATNR EQ LS_MOD_CELLS-VALUE.

        IF LV_MATKL IS NOT INITIAL.
          CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = LV_FNAME
              I_VALUE     = LV_MATKL.
        ENDIF.
      WHEN 'WERKS_F'.
        LV_FNAME = 'WERKS_F_T'.
        SELECT SINGLE NAME2 INTO LV_NAME2
          FROM T001W
        WHERE WERKS EQ LS_MOD_CELLS-VALUE.

        IF LV_NAME2 IS NOT INITIAL.
          CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = LV_FNAME
              I_VALUE     = LV_NAME2.
        ENDIF.

      WHEN 'WERKS_T'.
        LV_FNAME = 'WERKS_T_T'.
        SELECT SINGLE NAME2 INTO LV_NAME2
          FROM T001W
        WHERE WERKS EQ LS_MOD_CELLS-VALUE.

        IF LV_NAME2 IS NOT INITIAL.
          CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = LV_FNAME
              I_VALUE     = LV_NAME2.
        ENDIF.
      WHEN OTHERS.
    ENDCASE.
  ENDLOOP.

ENDFORM. " ALV_DATA_CHANGED
