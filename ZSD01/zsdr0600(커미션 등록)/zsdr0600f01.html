FORM get_raw .

  CLEAR: gt_raw, gt_raw[].
  SELECT a~fkdat a~vbeln b~posnr a~kunag b~matnr b~fkimg b~vrkme b~netwr a~waerk
         b~aubel b~aupos c~auart
         b~werks a~vkorg a~vtweg a~spart b~vkbur a~bzirk b~gsber
    INTO CORRESPONDING FIELDS OF TABLE gt_raw
    FROM vbrk AS a INNER JOIN
         vbrp AS b
      ON a~vbeln = b~vbeln INNER JOIN
         vbak AS c
      ON c~vbeln = b~aubel
     FOR ALL ENTRIES IN gt_cmt
   WHERE c~auart IN ('ZOR', 'ZRE', 'ZTH', 'ZRH')
     AND a~kunag = gt_cmt-kunag
     AND a~vbeln IN so_vbeln
     AND a~vkorg IN so_vkorg
     AND a~fkdat IN so_fkdat
     AND a~fksto <> 'X'
     AND a~sfakn = ''
     AND b~matnr = gt_cmt-matnr.

  DATA : lt_0230 LIKE zsdt0230 OCCURS 0 WITH HEADER LINE,
         lt_vbpa LIKE vbpa     OCCURS 0 WITH HEADER LINE,
         lt_marc LIKE marc     OCCURS 0 WITH HEADER LINE.

  CLEAR: lt_0230, lt_0230[], lt_vbpa, lt_vbpa[], lt_marc, lt_marc[].

  IF gt_raw[] IS NOT INITIAL.

    SELECT * INTO TABLE lt_0230
      FROM zsdt0230
       FOR ALL ENTRIES IN gt_raw
     WHERE vkorg = gt_raw-vkorg
       AND bzirk = gt_raw-bzirk
       AND loevm = ''.
    SORT lt_0230 BY vkorg bzirk.

    SELECT * INTO TABLE lt_vbpa
      FROM vbpa
       FOR ALL ENTRIES IN gt_raw
     WHERE vbeln = gt_raw-vbeln
       AND posnr = ''
       AND parvw IN ('S1', 'S2').
    SORT lt_vbpa BY vbeln parvw.

    SELECT * INTO TABLE lt_marc
      FROM marc
       FOR ALL ENTRIES IN gt_raw
     WHERE matnr = gt_raw-matnr
       AND werks = gt_raw-werks.
    SORT lt_marc BY matnr werks.

    DATA : lt_0610 LIKE zsdt0610 OCCURS 0 WITH HEADER LINE.
    CLEAR: lt_0610, lt_0610[].
    SELECT * INTO TABLE lt_0610
      FROM zsdt0610
       FOR ALL ENTRIES IN gt_raw
     WHERE vbeln = gt_raw-vbeln
       AND posnr = gt_raw-posnr.
    SORT lt_0610 BY vbeln posnr bukrs belnr DESCENDING gjahr DESCENDING.

  ENDIF.

  LOOP AT gt_raw.
    CLEAR : i_aramt, i_bramt.
    READ TABLE gt_cmt WITH KEY kunag = gt_raw-kunag
                               matnr = gt_raw-matnr.
    IF sy-subrc = 0.
      gt_raw-kunnr = gt_cmt-kunnr.
      gt_raw-artyp = gt_cmt-artyp.
      gt_raw-aramt = gt_cmt-aramt.
      i_bramt = gt_raw-aramt.
      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = gt_raw-waerk
          amount_external      = i_bramt
          max_number_of_digits = 11
        IMPORTING
          amount_internal      = i_aramt.

      CASE gt_raw-artyp.
        WHEN 'A'.
          gt_raw-inamt = gt_raw-fkimg * i_aramt.
        WHEN 'R'.
          gt_raw-inamt = gt_raw-netwr * i_aramt.
      ENDCASE.

    ENDIF.

    IF gt_raw-auart = 'ZRE' OR gt_raw-auart = 'ZRH'.
      gt_raw-fkimg = gt_raw-fkimg * -1.
      gt_raw-netwr = gt_raw-netwr * -1.
      gt_raw-inamt = gt_raw-inamt * -1.
    ENDIF.

    READ TABLE lt_0230 WITH KEY vkorg = gt_raw-vkorg
                                bzirk = gt_raw-bzirk
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_raw-kostl = lt_0230-kostl.
    ENDIF.

    READ TABLE lt_vbpa WITH KEY vbeln = gt_raw-vbeln
                                parvw = 'S1'
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_raw-kuns1 = lt_vbpa-kunnr.
    ENDIF.

    READ TABLE lt_vbpa WITH KEY vbeln = gt_raw-vbeln
                                parvw = 'S2'
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_raw-kuns2 = lt_vbpa-kunnr.
    ENDIF.

    READ TABLE lt_marc WITH KEY matnr = gt_raw-matnr
                                werks = gt_raw-werks
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_raw-prctr = lt_marc-prctr.
    ENDIF.

    READ TABLE lt_0610 WITH KEY vbeln = gt_raw-vbeln
                                posnr = gt_raw-posnr
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gt_raw-bukrs = lt_0610-bukrs.
      gt_raw-belnr = lt_0610-belnr.
      gt_raw-gjahr = lt_0610-gjahr.
    ENDIF.

    MODIFY gt_raw.

  ENDLOOP.

ENDFORM. " GET_RAW
