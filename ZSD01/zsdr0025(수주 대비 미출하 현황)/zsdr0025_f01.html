FORM variant_init USING p_layout.

* 레이아웃
  DATA: gs_vari TYPE disvariant.


  CLEAR gs_vari.
  gs_vari-report   = sy-repid.
  gs_vari-username = sy-uname.

  IF p_layout <> ''.
    gs_vari-variant = p_layout.
  ENDIF.

  CALL FUNCTION 'LVC_VARIANT_DEFAULT_GET'
    EXPORTING
      i_save        = 'A'
    CHANGING
      cs_variant    = gs_vari
    EXCEPTIONS
      wrong_input   = 1
      not_found     = 2
      program_error = 3
      OTHERS        = 4.

  CASE sy-subrc.
    WHEN 0.
      p_layout = gs_vari-variant.
    WHEN 2.
      CLEAR p_layout.
  ENDCASE.

FORM prepare_select .

  DATA: l_stdat TYPE sy-datum,
        l_endat TYPE sy-datum.

  CLEAR: l_stdat, l_endat.


  "기준월 날짜
  CLEAR: r_datum, r_datum[].

  r_datum = 'IBT'.
  CONCATENATE p_spmon '01' INTO l_stdat.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = l_stdat
    IMPORTING
      last_day_of_month = l_endat
    EXCEPTIONS
      OTHERS            = 1.

  r_datum-low = l_stdat.
  r_datum-high = l_endat.
  APPEND r_datum.
  CLEAR r_datum.


  "주차 계산
  DATA: l_day TYPE scal-indicator,
        l_diff TYPE t5a4a-dlydy,
        l_caldat TYPE sy-datum,
        l_numc TYPE n.

  DATA: wa_week LIKE r_1week.

  CLEAR: r_0week[], r_1week[], r_2week[],
         r_3week[], r_4week[], r_5week[], r_6week[].

  DO.
    "선출하 일자 만들기
    IF sy-index = 1.
      r_0week = 'ILT'.
      r_0week-low = l_stdat.
      APPEND r_0week.
      CLEAR r_0week.
    ENDIF.

    "이월출하 일자 만들기
    IF l_stdat >= l_endat.
      r_6week = 'IGT'.
      r_6week-low = l_endat.
      APPEND r_6week.
      CLEAR r_6week.
      EXIT.
    ENDIF.

    CLEAR: l_day, l_diff, l_caldat, l_numc.

    l_numc = sy-index.

    "요일 구하기
    CALL FUNCTION 'DATE_COMPUTE_DAY'
      EXPORTING
        date = l_stdat
      IMPORTING
        day  = l_day.

    IF l_day = 7.
      l_diff = 6.
    ELSE.
      l_diff = 6 - l_day.
    ENDIF.

    "주차 마지막 날
    CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
      EXPORTING
        date      = l_stdat
        days      = l_diff
        months    = 0
        years     = 0
        signum    = '+'
      IMPORTING
        calc_date = l_caldat.


    IF l_caldat > l_endat.
      l_caldat = l_endat.
    ENDIF.

    CLEAR v_fieldnm.

    CONCATENATE 'R_' l_numc 'WEEK[]' INTO v_fieldnm.
    ASSIGN (v_fieldnm) TO <fs01>.

    IF sy-subrc = 0.
      CLEAR wa_week.
      wa_week = 'IBT'.
      wa_week-low = l_stdat.
      wa_week-high = l_caldat.
      APPEND wa_week TO <fs01>.

    ELSE.
      EXIT.
    ENDIF.

    CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
      EXPORTING
        date      = l_caldat
        days      = 1
        months    = 0
        years     = 0
        signum    = '+'
      IMPORTING
        calc_date = l_stdat.
  ENDDO.


  "내수/수출 조회조건
  CASE 'X'.
    WHEN p_all.
      CLEAR: r_bsart, r_bsart[], r_vtweg, r_vtweg[].
    WHEN p_in.
      r_bsart = 'IEQ'.
      r_bsart-low = 'UB'.
      APPEND r_bsart.
      CLEAR r_bsart.

      r_vtweg = 'IEQ'.
      r_vtweg-low = '10'.
      APPEND r_vtweg.
      CLEAR r_vtweg.

      r_vtweg = 'IEQ'.
      r_vtweg-low = '30'.
      APPEND r_vtweg.
      CLEAR r_vtweg.

    WHEN p_out.
      r_bsart = 'IEQ'.
      r_bsart-low = 'ZUB'.
      APPEND r_bsart.
      CLEAR r_bsart.

      r_vtweg = 'IEQ'.
      r_vtweg-low = '20'.
      APPEND r_vtweg.
      CLEAR r_vtweg.

  ENDCASE.


ENDFORM. " PREPARE_SELECT
