FORM data_select .

  FIELD-SYMBOLS: <wa> TYPE ty_data.

  CLEAR: gt_data.

  "마일리지 적립 기준정보
  SELECT bukrs mtgrp kunnr
         MAX( spmon ) AS spmon
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zsdt1010
  WHERE  bukrs IN s_bukrs
    AND mtgrp IN s_mtgrp
    AND spmon LE p_spmon
    AND kunnr IN s_kunnr
  GROUP BY bukrs mtgrp kunnr.

  IF sy-subrc = 0.
    "마일리지 적립 로그
    CLEAR t_log.
    SELECT *
    INTO TABLE t_log
    FROM zsdt1040
    WHERE bukrs IN s_bukrs
      AND mtgrp IN s_mtgrp
      AND spmon EQ p_spmon
      AND kunnr IN s_kunnr.

    SORT t_log BY bukrs kunnr.
  ENDIF.

  LOOP AT gt_data ASSIGNING <wa>.

    "기준수량/고객명
    SELECT SINGLE a~menge b~name1 c~butxt
    INTO (<wa>-menge, <wa>-name1, <wa>-butxt)
    FROM zsdt1010 AS a
    LEFT JOIN kna1 AS b ON a~kunnr = b~kunnr
    LEFT JOIN t001 AS c ON a~bukrs = c~bukrs
    WHERE a~bukrs EQ <wa>-bukrs
      AND a~mtgrp EQ <wa>-mtgrp
      AND a~spmon EQ <wa>-spmon
      AND a~kunnr EQ <wa>-kunnr.

    "당월 누적수량
    SELECT SINGLE accme
    INTO (<wa>-accme)
    FROM zsdt1020
    WHERE bukrs EQ <wa>-bukrs
      AND mtgrp EQ <wa>-mtgrp
      AND spmon EQ p_spmon
      AND kunnr EQ <wa>-kunnr.

    <wa>-meins = 'KG'.

    "마일리지
    SELECT SINGLE pntsm waers
    INTO (<wa>-pntsm, <wa>-waers)
    FROM zsdt1030
    WHERE bukrs EQ <wa>-bukrs
*      AND mtgrp EQ <wa>-mtgrp
      AND kunnr EQ <wa>-kunnr.


    READ TABLE t_log WITH KEY bukrs = <wa>-bukrs
*                              mtgrp = <wa>-mtgrp
                              kunnr = <wa>-kunnr
      BINARY SEARCH TRANSPORTING NO FIELDS.

    IF sy-subrc = 0.
      LOOP AT t_log INTO s_log FROM sy-tabix.
        IF s_log-bukrs <> <wa>-bukrs
*          OR s_log-mtgrp <> <wa>-mtgrp
          OR s_log-kunnr <> <wa>-kunnr.
          EXIT.
        ENDIF.

        "당월 적립/차감 마일리지
        CASE s_log-pntty.
          WHEN 'A' OR 'C'.
            ADD s_log-pntsm TO <wa>-addsm.
          WHEN 'U' OR 'D'.
            ADD s_log-pntsm TO <wa>-minsm.
        ENDCASE.
      ENDLOOP.
    ENDIF.

  ENDLOOP.


ENDFORM. " DATA_SELECT

FORM p1020_move_transaction USING p_fieldname.
*         addsm TYPE zsdt1030-pntsm,
*         minsm TYPE zsdt1030-pntsm,

  CASE p_fieldname.
    WHEN 'ADDSM' OR 'MINSM'.
      CLEAR gt_log.

      READ TABLE t_log WITH KEY bukrs = gs_data-bukrs
*                                mtgrp = gs_data-mtgrp
                                kunnr = gs_data-kunnr
        BINARY SEARCH TRANSPORTING NO FIELDS.

      IF sy-subrc = 0.
        LOOP AT t_log INTO s_log FROM sy-tabix.
          IF s_log-bukrs <> gs_data-bukrs OR
*             s_log-mtgrp <> gs_data-mtgrp OR
*             s_log-spmon <> gs_data-spmon OR
             s_log-kunnr <> gs_data-kunnr.
            EXIT.
          ENDIF.


          "당월 적립/차감 마일리지
          CASE p_fieldname.
            WHEN 'ADDSM'.
              IF NOT ( s_log-pntty = 'A' OR s_log-pntty = 'A' ).
                CONTINUE.
              ENDIF.

            WHEN 'MINSM'.
              IF NOT ( s_log-pntty = 'U' OR s_log-pntty = 'D' ).
                CONTINUE.
              ENDIF.
          ENDCASE.

          MOVE-CORRESPONDING s_log TO gs_log.

          SELECT SINGLE maktx

          FROM makt

           AND spras = sy-langu.

                                    CHANGING gs_log-dmtxt.

          APPEND gs_log TO gt_log.
          CLEAR s_log.
        ENDLOOP.

        IF gt_log IS NOT INITIAL.
          CALL SCREEN 0200 STARTING AT 15 5.
        ENDIF.
      ENDIF.

FORM read_domain_value USING p_domain
                                 p_value
                        CHANGING p_text.
  DATA : l_domname   LIKE dd07v-domname,
         l_domvalue  LIKE dd07v-domvalue_l,
         l_ddtext    LIKE dd07v-ddtext.

  CHECK p_value NE ' '.
  MOVE : p_domain  TO l_domname,
         p_value   TO l_domvalue.

  CALL FUNCTION 'DOMAIN_VALUE_GET'
    EXPORTING
      i_domname  = l_domname
      i_domvalue = l_domvalue
    IMPORTING
      e_ddtext   = l_ddtext.

  MOVE l_ddtext TO p_text.
ENDFORM. " read_domain_value
