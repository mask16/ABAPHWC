FUNCTION ZIM_CHARGE_DOCUMENT_CANCEL.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(P_ZCSDNO) LIKE ZIM610-ZCSDNO
*" REFERENCE(REASONREVERSAL) LIKE BAPI_INCINV_FLD-REASON_REV
*" REFERENCE(POSTINGDATE) LIKE BAPI_INCINV_FLD-PSTNG_DATE
*" REFERENCE(PAYEECN) TYPE CHAR1 OPTIONAL
*" TABLES
*" RETURN STRUCTURE BAPIRET2 OPTIONAL
*" EXCEPTIONS
*" POST_ERROR
*"----------------------------------------------------------------------

* Global data declarations

  REFRESH : BDCDATA.
  REFRESH : RETURN.
  CLEAR   : V_SUBRC.
  CLEAR   : INVOICEDOCNUMBER,
            FISCALYEAR,
            INVOICEDOCNUMBER_2,
            FISCALYEAR_2.

* CHECK
  CLEAR  ZIM610.
  SELECT SINGLE *
    FROM ZIM610
   WHERE ZCSDNO = P_ZCSDNO.
  IF SY-SUBRC NE 0.
    IF SY-LANGU EQ '3'.
      MESSAGE E001 WITH '비용문서 관리번호를 확인하시오.'
              RAISING POST_ERROR.
    ELSE.
      MESSAGE E001 WITH 'Invalid data : expense document no.'
              RAISING POST_ERROR.
    ENDIF.

  ENDIF.

* 포스팅 제외건
  IF ZIM610-ZEXCEP = 'X'.
    UPDATE ZIM610 SET ZAMTTY = '2'
                      ZCOMPL = ' '
                      ZAENAM = SY-UNAME
                      ZAEDAT = SY-DATUM
                      ZAEZET = SY-UZEIT
     WHERE ZCSDNO = P_ZCSDNO.

    IF SY-SUBRC NE 0.
      ROLLBACK WORK.
      IF SY-LANGU EQ '3'.
        MESSAGE E011 WITH '전기후 저장' RAISING POST_ERROR.
      ELSE.
        MESSAGE E011 WITH 'save after posting' RAISING POST_ERROR.
      ENDIF.
    ENDIF.

    COMMIT WORK.
    EXIT.
  ENDIF.

* --
  IF ZIM610-ZCOMPL IS INITIAL.
    IF SY-LANGU EQ '3'.
      MESSAGE E001 WITH '전기미완료 상태입니다.'
                  RAISING POST_ERROR.
    ELSE.
      MESSAGE E001 WITH 'Not yet posted data.'
                  RAISING POST_ERROR.
    ENDIF.
  ENDIF.

* 관세의 경우 입고전기가 되어 있으면 취소불가
* SELECT SINGLE *
* FROM ZIM611
* WHERE ZCSDNO EQ P_ZCSDNO
* AND ZCHRCD = 'C01'.
* IF SY-SUBRC = 0.
* SELECT *
* FROM ZIM551
* WHERE ZICCDN = ZIM610-ZREDOC.
* SELECT SINGLE *
* FROM ZIM550
* WHERE ZMIGON = ZIM551-ZMIGON
* AND ZCOMPL = 'X'.
* IF SY-SUBRC = 0.
* MESSAGE E001 WITH '등록된 입고가 전기처리상태입니다.'
* RAISING POST_ERROR.
* ENDIF.
* ENDSELECT.
* ENDIF.

  CLEAR ZIM006.
  SELECT SINGLE *
    FROM ZIM006
   WHERE BUKRS = ZIM610-BUKRS.
  MOVE 'N' TO CTU_PARAMS-DISMODE.
  IF SY-UNAME = ZIM006-ZSYMGR.
    IF ZIM006-DISMODE IS INITIAL.
      MOVE 'N' TO CTU_PARAMS-DISMODE.
    ELSE.
      MOVE ZIM006-DISMODE TO CTU_PARAMS-DISMODE.
    ENDIF.
  ENDIF.
  CTU_PARAMS-UPDMODE  = 'S'.
  CTU_PARAMS-CATTMODE = ' '.
  CTU_PARAMS-DEFSIZE  = ' '.
  CTU_PARAMS-RACOMMIT = 'X'.
  CTU_PARAMS-NOBINPT  = 'X'.
  CTU_PARAMS-NOBIEND  = 'X'.

* 전기
  SELECT MAX( ZISEQN ) INTO V_ZISEQN
    FROM ZIM612
   WHERE ZCSDNO EQ P_ZCSDNO.

  CLEAR  ZIM612.
  SELECT SINGLE *
    FROM ZIM612
   WHERE ZCSDNO EQ P_ZCSDNO
     AND ZISEQN EQ V_ZISEQN.

* PAYEE 취소
  IF ZIM610-ZPOSPR = 'A'.
    IF NOT ZIM610-ZBELNR1 IS INITIAL.
      PERFORM CLEAR_TO_PAYEE_CANCEL TABLES RETURN
                                     USING REASONREVERSAL
                                           POSTINGDATE
                                           INVOICEDOCNUMBER
                                           FISCALYEAR.

* 처리여부
      SELECT SINGLE * FROM BSAK WHERE GJAHR = ZIM612-GJAHR
                                  AND BELNR = ZIM612-BELNR.
      IF SY-SUBRC = 0.
        IF SY-LANGU EQ '3'.
          MESSAGE E011 WITH '취소전기 후 저장' RAISING POST_ERROR.
        ELSE.
          MESSAGE E011 WITH 'save after posting cancel' RAISING POST_ERROR.
        ENDIF.
      ELSE.
        UPDATE ZIM610 SET ZBELNR1 = ''
                          ZGJAHR1 = ''
                    WHERE ZCSDNO  = ZIM610-ZCSDNO.
        COMMIT WORK.
      ENDIF.
    ENDIF.
  ENDIF.

  IF PAYEECN = 'X'.
    EXIT.
  ENDIF.
*
  SELECT SINGLE *
    FROM RBKP
   WHERE BELNR EQ ZIM612-BELNR
     AND GJAHR EQ ZIM612-GJAHR.

  IF SY-SUBRC NE 0.
    SELECT *
      FROM EKBE UP TO 1 ROWS
     WHERE BELNR EQ ZIM612-BELNR
       AND GJAHR EQ ZIM612-GJAHR.
    ENDSELECT.
  ENDIF.

  IF SY-SUBRC EQ 0.
    PERFORM LIV_CHARGE_DOCUMENT_CANCEL TABLES RETURN
                                        USING REASONREVERSAL
                                              POSTINGDATE
                                              INVOICEDOCNUMBER
                                              FISCALYEAR.
  ELSE.
    PERFORM FI_CHARGE_DOCUMENT_CANCEL TABLES RETURN
                                       USING REASONREVERSAL
                                             POSTINGDATE
                                             INVOICEDOCNUMBER
                                             FISCALYEAR.
  ENDIF.

* CHECK
  IF INVOICEDOCNUMBER IS INITIAL.
    IF SY-LANGU EQ '3'.
      MESSAGE E011 WITH '전기' RAISING POST_ERROR.
    ELSE.
      MESSAGE E011 WITH 'posting' RAISING POST_ERROR.
    ENDIF.
  ENDIF.

* 비용문서 헤더 UPDATE
  UPDATE ZIM610 SET ZBELNR1 = ' '
                    ZGJAHR1 = ' '
                    ZCOMPL  = ' '
                    ZAENAM  = SY-UNAME
                    ZAEDAT  = SY-DATUM
                    ZAEZET  = SY-UZEIT
   WHERE ZCSDNO = P_ZCSDNO.

  IF SY-SUBRC NE 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    ROLLBACK WORK.
    IF SY-LANGU EQ '3'.
      MESSAGE E011 WITH '전기후 저장' RAISING POST_ERROR.
    ELSE.
      MESSAGE E011 WITH 'save after posting' RAISING POST_ERROR.
    ENDIF.
  ENDIF.

* 전기이력 UPDATE
  MOVE : INVOICEDOCNUMBER TO ZIM612-ZBELNR,
         FISCALYEAR       TO ZIM612-ZGJAHR,
         POSTINGDATE      TO ZIM612-ZBUDAT,
         SY-UNAME         TO ZIM612-ZAENAM,
         SY-DATUM         TO ZIM612-ZAEDAT,
         SY-UZEIT         TO ZIM612-ZAEZET.

  MODIFY ZIM612 FROM ZIM612.
  IF SY-SUBRC NE 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    ROLLBACK WORK.
    IF SY-LANGU EQ '3'.
      MESSAGE E011 WITH '전기후 저장' RAISING POST_ERROR.
    ELSE.
      MESSAGE E011 WITH 'save after posting' RAISING POST_ERROR.
    ENDIF.
  ENDIF.

* END
  CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'.
  COMMIT WORK.

ENDFUNCTION.
