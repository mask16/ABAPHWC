MODULE USER_COMMAND_0196 INPUT.

  SAVE_OK = OK_CODE.

  CASE SAVE_OK.
    WHEN 'FILE'.

      LOOP AT IT_030_OTH WHERE CHECK = 'X'.
        MOVE SY-TABIX  TO V_TABIX.
      ENDLOOP.

      CALL FUNCTION 'WS_FILENAME_GET'
        EXPORTING
          DEF_PATH         = 'C:'
          MODE             = 'O'
        IMPORTING

        EXCEPTIONS
          INV_WINSYS       = 1
          NO_BATCH         = 2
          SELECTION_CANCEL = 3
          SELECTION_ERROR  = 4.

        IF NOT V_TABIX IS INITIAL.
          MODIFY IT_030_OTH INDEX V_TABIX.
        ELSE.
          APPEND IT_030_OTH.
        ENDIF.

    WHEN 'CONT' OR 'EXCEL'.
      LOOP AT IT_030 WHERE CHECK = 'X'.
        MOVE IT_030-ZFLFRM TO FLFRM.
      ENDLOOP.
      ANSWER = 'Y'.
      LEAVE TO SCREEN 0.
    WHEN 'CANC'.
      LOOP AT IT_030 WHERE CHECK = 'X'.
        CLEAR IT_030-CHECK.
        MODIFY IT_030.
      ENDLOOP.
      LOOP AT IT_092 WHERE CHECK = 'X'.
        CLEAR IT_092-CHECK.
        MODIFY IT_092.
      ENDLOOP.

      ANSWER = 'N'.
      LEAVE TO SCREEN 0.
    WHEN 'DOCU'.
      CALL TRANSACTION 'ZEXDAY'.
  ENDCASE.

ENDMODULE. " USER_COMMAND_0196 INPUT
