FUNCTION ZIM_CHARGE_DOCUMENT_POST .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(P_BLDAT) LIKE ZIM610-BLDAT
*" REFERENCE(P_BUDAT) LIKE ZIM610-BUDAT
*" REFERENCE(P_ZFBDT) LIKE ZIM610-ZFBDT
*" REFERENCE(P_BUPLA) LIKE ZIM610-BUPLA
*" REFERENCE(P_XBLNR) LIKE ZIM610-XBLNR
*" REFERENCE(P_BKTXT) LIKE ZIM610-BKTXT
*" TABLES
*" RETURN STRUCTURE BAPIRET2
*" IT_610_I STRUCTURE ZIM610
*" EXCEPTIONS
*" POST_ERROR
*"----------------------------------------------------------------------
* Global data declarations

  DATA : V_EBELN LIKE EKKO-EBELN.

  REFRESH : RETURN, MESSTAB.
  CLEAR   : INVOICEDOCNUMBER,
            FISCALYEAR,
            INVOICEDOCNUMBER_2,
            FISCALYEAR_2.

  IT_610[] = IT_610_I[].

  CLEAR ZIM610.
  READ TABLE IT_610 INDEX 1.
  MOVE-CORRESPONDING IT_610 TO ZIM610.

  LOOP AT IT_610.
    IF ZIM610-ZTERM   NE IT_610-ZTERM   OR
       ZIM610-BUKRS   NE IT_610-BUKRS   OR
       ZIM610-LIFNR   NE IT_610-LIFNR   OR
       ZIM610-ZLOCRP  NE IT_610-ZLOCRP  OR
       ZIM610-ZMIROP  NE IT_610-ZMIROP  OR
       ZIM610-WAERS   NE IT_610-WAERS   OR
       ZIM610-HWAER   NE IT_610-HWAER   OR
       ZIM610-KURSF   NE IT_610-KURSF   OR
       ZIM610-XMWST   NE IT_610-XMWST   OR
       ZIM610-ZLSPR   NE IT_610-ZLSPR   OR
       ZIM610-VORGANG NE IT_610-VORGANG OR
       ZIM610-BLART   NE IT_610-BLART   OR
       ZIM610-ZPOSPR  NE IT_610-ZPOSPR  OR
       ZIM610-ZPAYEE  NE IT_610-ZPAYEE  OR
       ZIM610-ZBELNR1 NE IT_610-ZBELNR1 OR
       ZIM610-ZGJAHR1 NE IT_610-ZGJAHR1 OR
       ZIM610-ZCOMPL  NE IT_610-ZCOMPL.
      IF SY-LANGU EQ '3'.
        MESSAGE E001 WITH '동시처리가 불가능한 비용문서들입니다.'
               RAISING POST_ERROR.
      ELSE.
        MESSAGE E001 WITH 'Cannot post at the same time.'
               RAISING POST_ERROR.
      ENDIF.
    ENDIF.
  ENDLOOP.

  CLEAR ZIM006.
  SELECT SINGLE *
    FROM ZIM006
   WHERE BUKRS EQ ZIM610-BUKRS.
  MOVE 'N' TO CTU_PARAMS-DISMODE.
  IF SY-UNAME = ZIM006-ZSYMGR.
    IF ZIM006-DISMODE IS INITIAL.
      MOVE 'N' TO CTU_PARAMS-DISMODE.
    ELSE.
      MOVE ZIM006-DISMODE TO CTU_PARAMS-DISMODE.
    ENDIF.
  ENDIF.
  CTU_PARAMS-UPDMODE  = 'S'.
  CTU_PARAMS-CATTMODE = ' '.
  CTU_PARAMS-DEFSIZE  = ' '.
  CTU_PARAMS-RACOMMIT = 'X'.
  CTU_PARAMS-NOBINPT  = 'X'.
  CTU_PARAMS-NOBIEND  = 'X'.

* CLEAR
  CLEAR : DEL_COST, V_SUBRC, V_DMBTR, V_WRBTR, V_WMWST.

  REFRESH : IT_611, IT_613.
  LOOP AT IT_610.
* 비용내용
    SELECT *
      FROM ZIM611
     WHERE ZCSDNO EQ IT_610-ZCSDNO.
      MOVE-CORRESPONDING ZIM611 TO IT_611.
      APPEND IT_611.
    ENDSELECT.

* 배부내역
    SELECT *
      FROM ZIM613
     WHERE ZCSDNO EQ IT_610-ZCSDNO.
      CLEAR IT_613.
      MOVE-CORRESPONDING ZIM613 TO IT_613.
* --
      IT_613-V_EBELN = IT_613-EBELN.
      IT_613-V_EBELP = IT_613-EBELP.
* --
      MOVE P_BUPLA              TO IT_613-BUPLA.
      APPEND IT_613.
      IF NOT IT_613-EBELN IS INITIAL.
        V_EBELN = IT_613-EBELN.
      ENDIF.
    ENDSELECT.

    SELECT SINGLE *
      FROM ZIM613
     WHERE ZCSDNO EQ IT_610-ZCSDNO
       AND ZDCSTX EQ 'X'.
    IF SY-SUBRC EQ 0.
      MOVE 'YES' TO DEL_COST.
    ENDIF.
    ADD : IT_610-DMBTR TO V_DMBTR,
          IT_610-WRBTR TO V_WRBTR,
          IT_610-WMWST TO V_WRBTR,
          IT_610-WMWST TO V_WMWST.
  ENDLOOP.

* 임가공은 FI로 직접 POSTING되도록 처리
  IF ZIM610-ZFIPST = 'X'.
    CLEAR DEL_COST.
  ENDIF.


*==================================================================
* ※ 지광섭 수정( 대체 수취인로직 사용안함 ) ※
*==================================================================
* PAYEE ONLY : 전기는 됐는데 반제만 안된경우 : 이부분 수행하고 끝.
  IF ZIM610-ZCOMPL = 'X'.
    IF ZIM610-ZPOSPR = 'A' AND ZIM610-ZBELNR1 IS INITIAL.
      SELECT MAX( ZISEQN ) INTO V_ZISEQN
        FROM ZIM612
       WHERE ZCSDNO EQ ZIM610-ZCSDNO.
      CLEAR ZIM612.
      SELECT SINGLE *
        FROM ZIM612
       WHERE ZCSDNO EQ ZIM610-ZCSDNO
         AND ZISEQN EQ V_ZISEQN.

      PERFORM CLEAR_TO_PAYEE_POST TABLES RETURN
                                   USING P_BLDAT
                                         P_BUDAT
                                         P_ZFBDT.
      UPDATE ZIM610 SET ZBELNR1 = INVOICEDOCNUMBER_2
                        ZGJAHR1 = FISCALYEAR_2
                  WHERE ZCSDNO  = ZIM610-ZCSDNO.
      COMMIT WORK.
      EXIT.
    ELSE.
      EXIT.
    ENDIF.
  ENDIF.
*==================================================================

* --
  IF DEL_COST = 'YES'.
    PERFORM CHARGE_POSTING_LIV TABLES RETURN
                                USING P_BLDAT
                                      P_BUDAT
                                      P_ZFBDT
                                      P_BUPLA
                                      P_XBLNR
                                      P_BKTXT.
  ELSE.
* PERFORM NET_AMOUNT_POSTING.
    PERFORM CHARGE_POSTING_FI  TABLES RETURN
                                USING P_BLDAT
                                      P_BUDAT
                                      P_ZFBDT
                                      P_BUPLA
                                      P_XBLNR
                                      P_BKTXT.
  ENDIF.

  IF INVOICEDOCNUMBER IS INITIAL.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    RAISE POST_ERROR.
  ENDIF.

* UPDATE
  LOOP AT IT_610.
    UPDATE ZIM610 SET BLDAT   = P_BLDAT
                      BUDAT   = P_BUDAT
                      ZFBDT   = P_ZFBDT
                      BUPLA   = P_BUPLA
                      XBLNR   = P_XBLNR
                      BKTXT   = P_BKTXT
                      ZAENAM  = SY-UNAME
                      ZAEDAT  = SY-DATUM
                      ZAEZET  = SY-UZEIT
                      ZCOMPL  = 'X'
                WHERE ZCSDNO  = IT_610-ZCSDNO.

    IF SY-SUBRC NE 0.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      ROLLBACK WORK.
      IF SY-LANGU EQ '3'.
        MESSAGE E011 WITH '전기후 저장' RAISING POST_ERROR.
      ELSE.
        MESSAGE E011 WITH 'save after posting' RAISING POST_ERROR.
      ENDIF.
    ENDIF.

    SELECT MAX( ZISEQN ) INTO V_ZISEQN
      FROM ZIM612
     WHERE ZCSDNO = IT_610-ZCSDNO.
    CLEAR ZIM612.
    ADD    10               TO V_ZISEQN.
    MOVE : IT_610-ZCSDNO    TO ZIM612-ZCSDNO,
           V_ZISEQN         TO ZIM612-ZISEQN,
           INVOICEDOCNUMBER TO ZIM612-BELNR,
           FISCALYEAR       TO ZIM612-GJAHR,
           P_BUDAT          TO ZIM612-BUDAT,
           P_BLDAT          TO ZIM612-BLDAT,
           V_ZPMMFI         TO ZIM612-ZPMMFI,
           SY-UNAME         TO ZIM612-ZERNAM,
           SY-DATUM         TO ZIM612-ZERDAT,
           SY-UZEIT         TO ZIM612-ZERZET,
           SY-UNAME         TO ZIM612-ZAENAM,
           SY-DATUM         TO ZIM612-ZAEDAT,
           SY-UZEIT         TO ZIM612-ZAEZET.
    INSERT ZIM612.
    IF SY-SUBRC NE 0.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      ROLLBACK WORK.
      IF SY-LANGU EQ '3'.
        MESSAGE E011 WITH '전기후 저장' RAISING POST_ERROR.
      ELSE.
        MESSAGE E011 WITH 'save after posting' RAISING POST_ERROR.
      ENDIF.
    ENDIF.
  ENDLOOP.

* COMMIT
  CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          WAIT = 'X'.
  COMMIT WORK.

*==================================================================
* ※ 지광섭 수정( 대체 수취인로직 사용안함 ) ※
*==================================================================
* PAYEE( 지급처로 반제 하는 로직 변경 )
  IF ZIM610-ZPOSPR = 'A'.
    PERFORM CLEAR_TO_PAYEE_POST TABLES RETURN
                                 USING P_BLDAT
                                       P_BUDAT
                                       P_ZFBDT.
    UPDATE ZIM610 SET ZBELNR1 = INVOICEDOCNUMBER_2
                      ZGJAHR1 = FISCALYEAR_2
                WHERE ZCSDNO  = ZIM610-ZCSDNO.
    COMMIT WORK.
  ENDIF.


ENDFUNCTION.
