FORM GET_FROM_MEMORY_ZREDOC USING P_ZREDOC.

  REFRESH IT_ZREDOC.
  CLEAR   IT_ZREDOC.

  IMPORT IT_ZREDOC FROM MEMORY ID 'IT_ZREDOC'.
  FREE MEMORY ID 'IT_ZREDOC'.
  IF IT_ZREDOC[] IS INITIAL.
    MOVE  : P_ZREDOC TO IT_ZREDOC-ZREDOC.
    APPEND  IT_ZREDOC.
  ENDIF.

ENDFORM. " GET_FROM_MEMORY_ZREDOC

MODULE FIELD_ATTRIBUTE OUTPUT.

  LOOP AT SCREEN.
    SCREEN-INPUT = 0.
    IF SCREEN-GROUP1 = 'IN'.
      SCREEN-INPUT = 1.
    ENDIF.
    IF TR_MODE EQ 'D'.
      SCREEN-INPUT = 0.
    ENDIF.
    IF SCREEN-GROUP1 = 'TAB'.
      SCREEN-INPUT = 1.
    ENDIF.
    MODIFY SCREEN.
  ENDLOOP.

ENDMODULE. " FIELD_ATTRIBUTE OUTPUT

FORM MESSAGE_POPUP_STEP USING TITLE
                              TEXTLINE1
                              TEXTLINE2
                              CANCEL_DISPLAY
                              SAVE_OK.
  DATA : TEXTLINE(400).

  CLEAR TEXTLINE.

  ELSE.

      EXPORTING
        TITLEBAR              = TITLE

        TEXT_BUTTON_1         = 'Yes'
        TEXT_BUTTON_2         = 'No'

        WHEN 'BACK'.

            EXIT.
          ENDIF.
*         IF sy-calld = 'X'.
*           LEAVE PROGRAM.
*         ENDIF.
          LEAVE TO SCREEN 0.
        WHEN 'POSTCN'.
          PERFORM POSTCN IN PROGRAM (SY-REPID).
          IF NOT ERR_CHK IS INITIAL.
            EXIT.
          ENDIF.
          LEAVE TO SCREEN 0.
        WHEN 'JAA'.
          PERFORM JAA IN PROGRAM (SY-REPID).
          IF NOT ERR_CHK IS INITIAL.
            EXIT.
          ENDIF.
        WHEN 'FN1V'.

          LEAVE TO SCREEN 0.

          PERFORM FN1V IN PROGRAM (SY-REPID).
          LEAVE TO SCREEN 0.
        WHEN 'FN5V'.
          PERFORM FN5V IN PROGRAM (SY-REPID).
          LEAVE TO SCREEN 0.
        WHEN 'FN4V'.
          PERFORM FN4V IN PROGRAM (SY-REPID).
          LEAVE TO SCREEN 0.
        WHEN 'FNM3'.
          PERFORM FNM3 IN PROGRAM (SY-REPID).
          LEAVE TO SCREEN 0.
        WHEN 'CNCL'.
          PERFORM CNCL IN PROGRAM (SY-REPID).
          LEAVE TO SCREEN 0.
      ENDCASE.

    WHEN '2'.
      IF SY-CALLD = 'X'.
*       LEAVE PROGRAM.
      ENDIF.
      CASE SAVE_OK.
        WHEN 'BACK'.
          LEAVE TO SCREEN 0.
        WHEN 'EXIT'.
          LEAVE PROGRAM.
        WHEN 'SAVE' OR 'DELE' OR 'POST' OR 'POSTCN' OR
             'FN1V' OR 'FN4V' OR 'FN5V' OR 'FNM3'   OR

      ENDCASE.

        EXIT.
      ENDIF.

      IF SY-CALLD = 'X'.
        LEAVE PROGRAM.
      ENDIF.
      IF SAVE_OK = 'BACK' OR
         SAVE_OK = 'EXIT'.
        LEAVE SCREEN.
      ENDIF.
  ENDCASE.
ENDFORM. " POPUP_STEP
