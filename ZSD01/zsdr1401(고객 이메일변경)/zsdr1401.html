Code listing for: ZSDR1401

Description: 고객 이메일변경
*&---------------------------------------------------------------------*
*& Report ZCHANGE_CUSTOMER_EMAIL
*&---------------------------------------------------------------------*
*& 이메일 변경을 위한 ABAP 프로그램
*팝업나옴
*&---------------------------------------------------------------------*
REPORT ZCHANGE_CUSTOMER_EMAIL.

TABLES: KNA1, ADR6.

PARAMETERS: P_KUNNR TYPE KUNNR OBLIGATORY. " 고객코드
PARAMETERS: P_EMAIL TYPE AD_SMTPADR. " 변경할 이메일

DATA: LV_ADDRNUM TYPE ADRC-ADDRNUMBER,
      LV_NAME1 TYPE KNA1-NAME1,
      LV_SMTP TYPE AD_SMTPADR,
      LV_RESPONSE. " 팝업 응답 저장 변수

DATA: LT_TEXT TYPE TABLE OF STEXT,
      LS_TEXT TYPE STEXT.

START-OF-SELECTION.

  " 1. 고객 주소번호(Address Number) 및 이름 확인
  SELECT SINGLE ADRNR NAME1
    INTO (LV_ADDRNUM, LV_NAME1)
    FROM KNA1
    WHERE KUNNR = P_KUNNR.

  IF SY-SUBRC <> 0.
    WRITE: / '해당 고객번호를 찾을 수 없습니다.'.
    EXIT.
  ENDIF.

  " 2. 현재 이메일 주소 조회
  SELECT SINGLE SMTP_ADDR
    INTO LV_SMTP
    FROM ADR6
    WHERE ADDRNUMBER = LV_ADDRNUM.

  " 3. 새로운 이메일 주소 입력 시 팝업 호출
  IF P_EMAIL IS NOT INITIAL.
    " 팝업창에 표시될 텍스트를 테이블에 담기
    LS_TEXT = '고객코드: ' && P_KUNNR.
    APPEND LS_TEXT TO LT_TEXT.

    LS_TEXT = '고객이름: ' && LV_NAME1.
    APPEND LS_TEXT TO LT_TEXT.

    LS_TEXT = '현재 이메일: ' && LV_SMTP.
    APPEND LS_TEXT TO LT_TEXT.

    LS_TEXT = '변경할 이메일: ' && P_EMAIL.
    APPEND LS_TEXT TO LT_TEXT.

    " 여러 줄 메시지를 지원하는 POPUP_WITH_TABLE_DISPLAY 함수 호출
    CALL FUNCTION 'POPUP_WITH_TABLE_DISPLAY'
      EXPORTING
        ENDPOS_COL = 50
        ENDPOS_ROW = 10
        STARTPOS_COL = 10
        STARTPOS_ROW = 5
        TITLETEXT     = '이메일 변경 확인'
*      IMPORTING
*        CHOISE = LV_RESPONSE
      TABLES
        VALUETAB      = LT_TEXT.
*      EXCEPTIONS
*        OTHERS        = 1.

    " 4. 이메일 주소 업데이트
    UPDATE ADR6
      SET SMTP_ADDR = P_EMAIL
    WHERE ADDRNUMBER = LV_ADDRNUM.

    IF SY-SUBRC = 0.
      COMMIT WORK.
      WRITE: / '고객님의 요청에 따라 이메일이 성공적으로 업데이트되었습니다.'.
      SKIP.
      WRITE: / '고객코드       :', P_KUNNR.
      WRITE: / '고객이름       :', LV_NAME1.
      WRITE: / '변경 전 이메일:', LV_SMTP.
      WRITE: / '변경 후 이메일:', P_EMAIL.
    ELSE.
      ROLLBACK WORK.
      WRITE: / '이메일 변경 중 오류가 발생했습니다.'.
      WRITE: / 'SAP SYSTEM-SUBRC:', SY-SUBRC.
    ENDIF.
  ENDIF.


*Selection texts
*----------------------------------------------------------
* P_EMAIL         변경할이메일
* P_KUNNR         고객번호
