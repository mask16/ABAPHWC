*&---------------------------------------------------------------------*

*----------------------------------------------------------------------*

*  <--  p2        text
*----------------------------------------------------------------------*
FORM get_data .

  CLEAR : gt_data , gt_data[] , lt_zsdt0700, lt_zsdt0700[] .


  SELECT a~matnr  mtart prodh
                        mvgr2
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
    INNER JOIN  mvke AS b
    ON a~matnr = b~matnr
    WHERE a~matnr IN so_matnr
      AND mtart IN so_mtart
      AND vkorg = '1000'
      AND vtweg  = '10'.


  IF sy-subrc = 0.
    LOOP AT gt_data.

      CASE gt_data-prodh.
        WHEN 'A100001000' OR 'A100002000' OR 'A100003000' OR
                'A100004000'  OR  'A100005000' OR 'A100006000' OR
                'A100007000' OR  'A100009000'  OR 'A100010000' OR
                'A100011000'.


          gt_data-cate1 = '01'.
        WHEN 'A100008000'.

          gt_data-cate1 = '02'.
        WHEN  'D100001000' OR 'D100002000' OR 'D100003000' OR
                  'D100004000' OR  'D100005000' OR 'D100006000'.

          gt_data-cate1 = '03'.

        WHEN 'B100001000' OR 'B100003000'  OR 'B100004000' OR
             'B100005000' OR 'B100006000'.

          IF gt_data-mvgr2  < '116'.
            gt_data-cate1 = '04'.
          ELSE.
            gt_data-cate1 = '05'.
