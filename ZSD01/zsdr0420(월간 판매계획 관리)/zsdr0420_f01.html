*&---------------------------------------------------------------------*
*&      Form  SELECT_DATA_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: ls_data_0100   LIKE LINE OF gt_data_0100,
        lv_rowcount    TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT a~spdat
         a~spmon
         a~vkbur
         a~conflg
         b~bezei   AS vkbtx
    INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
    FROM zsdt0401  AS a
   INNER JOIN
         tvkbt     AS b
      ON a~vkbur   EQ b~vkbur
     AND b~spras   EQ sy-langu
   WHERE a~spdat   EQ gv_spdat
     AND a~spmon   EQ pa_spmon.

  LOOP AT gt_data_0100 INTO ls_data_0100.
    IF ls_data_0100-conflg EQ 'X'.
      ls_data_0100-contx = '마감'.
    ELSE.
      ls_data_0100-contx = '수정가능'.
    ENDIF.

    MODIFY gt_data_0100 FROM ls_data_0100.
  ENDLOOP.

* 타이틀 및 조회건수
  lv_rowcount = lines( gt_data_0100 ).
  CONCATENATE '월간판매계획 관리' lv_rowcount
         INTO gv_title_0100.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

*&---------------------------------------------------------------------*
*&      Form  GET_WEEK_INFO
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM get_week_info USING p_spdat
                 CHANGING ps_date TYPE typ_date.

  DATA: lv_firstday TYPE sy-datum,
        lv_lastday  TYPE sy-datum.

  CLEAR: ps_date.

* 입력받은 기준일자의 월요일, 일요일
  CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
    EXPORTING
      date   = p_spdat
    IMPORTING
      monday = ps_date-frdat
      sunday = ps_date-todat.

* 기준일의 월의 시작일
  CONCATENATE p_spdat(6) '01'
         INTO lv_firstday.

* 기준일의 월의 마지막일
  CALL FUNCTION 'DATE_GET_MONTH_LASTDAY'
    EXPORTING
      i_date = p_spdat
    IMPORTING
      e_date = lv_lastday.

* 기준일의 월요일이 이전달로 넘어갈 경우 월시작일로 지정.
  IF ps_date-frdat LT lv_firstday.
    ps_date-frdat = lv_firstday.
  ENDIF.

* 기준일의 일요일이 지난달로 넘어갈 경우 월마지막로 지정.
  IF ps_date-todat GT lv_lastday.
    ps_date-todat = lv_lastday.
  ENDIF.

* 시작일의 년주차
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date = ps_date-frdat
    IMPORTING
      week = ps_date-spywk.

* 시작일의 년월
  ps_date-spmon = ps_date-frdat(6).

ENDFORM. " GET_WEEK_INFO

*&      Form  OPEN_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM open_0100 .

  DATA: lt_rows        TYPE lvc_t_row,
        lt_data_0100   LIKE gt_data_0100,
        lr_vkbur       TYPE RANGE OF zsdt0401-vkbur,
        ls_rows        LIKE LINE OF lt_rows,
        ls_data_0100   LIKE LINE OF lt_data_0100,
        ls_vkbur       LIKE LINE OF lr_vkbur,
        lv_answer      TYPE char01.

  CALL METHOD gv_grid_0100->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows.
* 선택행 체크
  IF lines( lt_rows ) LE 0.
    MESSAGE s006.
    EXIT.
  ENDIF.
* 선택한 행의 데이터로 테이블생성
  LOOP AT lt_rows INTO ls_rows WHERE rowtype EQ space.
    CLEAR: ls_vkbur.
    READ TABLE gt_data_0100 INDEX ls_rows-index
                             INTO ls_data_0100.
    IF ls_data_0100-conflg EQ 'X'.
      ls_vkbur-sign = 'I'.
      ls_vkbur-option = 'EQ'.
      ls_vkbur-low = ls_data_0100-vkbur.
      APPEND ls_vkbur TO lr_vkbur.
    ENDIF.
  ENDLOOP.

  IF lr_vkbur IS INITIAL.
    MESSAGE s006.
    EXIT.
  ENDIF.
* 생성여부
  PERFORM popup_to_confirm USING '해제 여부'
                                 '마감을 해제하시겠습니까?'
                                 ''
                                 ''
                        CHANGING lv_answer.
  CHECK lv_answer EQ '1'.
