FORM get_konp_kbetr USING p_knumh
                 CHANGING p_kbetr
                          p_konwa
                          p_kpein
                          p_kmein.

  DATA: lv_kopos TYPE konp-kopos.

  CLEAR: p_kbetr, p_konwa, p_kpein, p_kmein.
  CHECK p_knumh IS NOT INITIAL.

  SELECT MAX( kopos )
    INTO lv_kopos
    FROM konp
   WHERE knumh EQ p_knumh.

  SELECT SINGLE
         kbetr
         konwa
         kpein
         kmein
    INTO (p_kbetr, p_konwa, p_kpein, p_kmein)
    FROM konp
   WHERE knumh EQ p_knumh
     AND kopos EQ lv_kopos.

ENDFORM. " GET_KONP_KBETR
