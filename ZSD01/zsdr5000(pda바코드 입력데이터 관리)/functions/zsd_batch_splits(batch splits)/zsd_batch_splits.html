FUNCTION zsd_batch_splits.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_VTWEG) TYPE VTWEG
*" VALUE(I_WADAT_IST) TYPE WADAT_IST
*" TABLES
*" ITEM_DATA STRUCTURE ZSDS0210
*" RETURN STRUCTURE BAPIRET2 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_data_0100        TYPE zsds0210,
        ls_header_data      LIKE bapiobdlvhdrchg,
        ls_header_control   LIKE bapiobdlvhdrctrlchg,
        ls_techn_control    LIKE bapidlvcontrol,
        lt_header_deadlines TYPE STANDARD TABLE OF bapidlvdeadln,
        ls_header_deadlines LIKE LINE OF lt_header_deadlines,
        lt_item_data        TYPE STANDARD TABLE OF bapiobdlvitemchg,
        ls_item_data        LIKE LINE OF lt_item_data,
        lt_item_control     TYPE STANDARD TABLE OF bapiobdlvitemctrlchg,
        ls_item_control     LIKE LINE OF lt_item_control,
        lt_item_data_spl    TYPE STANDARD TABLE OF /spe/bapiobdlvitemchg,
        ls_item_data_spl    LIKE LINE OF lt_item_data_spl,
        lv_timestamp_utc    TYPE char15,
        lv_lgort            TYPE lgort_d.

* 1. 납품문서 위치 변경

* 유통경로에 따른 저장위치
* CASE i_vtweg.
* WHEN '10' OR '30'.
* lv_lgort = '9000'.
* WHEN '20'.
* lv_lgort = 'CY00'.
* ENDCASE.

  READ TABLE item_data INDEX 1 INTO ls_data_0100.
* 헤더생성.
  ls_header_data-deliv_numb = ls_data_0100-vbeln.
  ls_header_control-deliv_numb = ls_data_0100-vbeln.
  ls_techn_control-upd_ind = 'U'.

  ls_header_deadlines-deliv_numb = ls_data_0100-vbeln.
  ls_header_deadlines-timetype = 'WSHDRLFDAT'.
  CONCATENATE i_wadat_ist '000000'
         INTO lv_timestamp_utc.
  ls_header_deadlines-timestamp_utc = lv_timestamp_utc.
  ls_header_deadlines-timezone = 'UTC+9'.
  APPEND ls_header_deadlines TO lt_header_deadlines.

  LOOP AT item_data INTO ls_data_0100.
    CLEAR: ls_item_data, ls_item_control, ls_item_data_spl.
    ls_item_data-deliv_numb = ls_data_0100-vbeln.
    ls_item_data-deliv_item = ls_data_0100-posnr.
    ls_item_data-material = ls_data_0100-matnr.
    ls_item_data-fact_unit_nom = '1'.
    ls_item_data-fact_unit_denom = '1'.
    ls_item_data-base_uom = ls_data_0100-meins.
    ls_item_data-sales_unit = ls_data_0100-meins.
    APPEND ls_item_data TO lt_item_data.

    ls_item_control-deliv_numb = ls_data_0100-vbeln.
    ls_item_control-deliv_item = ls_data_0100-posnr.
    APPEND ls_item_control TO lt_item_control.

    ls_item_data_spl-deliv_numb = ls_data_0100-vbeln.
    ls_item_data_spl-deliv_item = ls_data_0100-posnr.
    ls_item_data_spl-pick_denial = 'X'.
    ls_item_data_spl-stge_loc = ls_data_0100-lgort.
    APPEND ls_item_data_spl TO lt_item_data_spl.

  ENDLOOP.

  CALL FUNCTION 'BAPI_OUTB_DELIVERY_CHANGE'
    EXPORTING
      header_data      = ls_header_data
      header_control   = ls_header_control
      delivery         = ls_data_0100-vbeln
      techn_control    = ls_techn_control
    TABLES
      header_deadlines = lt_header_deadlines
      item_data        = lt_item_data
      item_control     = lt_item_control
      return           = return
      item_data_spl    = lt_item_data_spl.

  READ TABLE return WITH KEY type = 'E'
    TRANSPORTING NO FIELDS.

  IF sy-subrc EQ 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    EXIT.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = 'X'.
  ENDIF.

* 2. 배치생성
  CLEAR: ls_header_data, ls_header_control, ls_techn_control,
         lt_item_data, lt_item_control, lt_item_data_spl.

* 헤더생성.
  ls_header_data-deliv_numb = ls_data_0100-vbeln.
  ls_header_control-deliv_numb = ls_data_0100-vbeln.
  ls_techn_control-upd_ind = 'U'.

  LOOP AT item_data INTO ls_data_0100.
    CLEAR: ls_item_data, ls_item_control, ls_item_data_spl.
    ls_item_data-deliv_numb = ls_data_0100-vbeln.
    ls_item_data-material = ls_data_0100-matnr.

    ls_item_data-batch = ls_data_0100-charg.
    ls_item_data-dlv_qty = ls_data_0100-dlv_qty.
    ls_item_data-dlv_qty_imunit = ls_data_0100-dlv_qty_imunit.
    ls_item_data-hieraritem = ls_data_0100-hieraritem.

    ls_item_data-usehieritm = 1.
    ls_item_data-fact_unit_nom = 1.
    ls_item_data-fact_unit_denom = 1.
    ls_item_data-base_uom = ls_data_0100-meins.
    ls_item_data-sales_unit = ls_data_0100-meins.
    APPEND ls_item_data TO lt_item_data.

    ls_item_control-deliv_numb = ls_data_0100-vbeln.
    ls_item_control-chg_delqty = 'X'.
    APPEND ls_item_control TO lt_item_control.

    ls_item_data_spl-deliv_numb = ls_data_0100-vbeln.
    ls_item_data_spl-pick_denial = 'X'.
    ls_item_data_spl-stge_loc = ls_data_0100-lgort.
    APPEND ls_item_data_spl TO lt_item_data_spl.

  ENDLOOP.

  CALL FUNCTION 'BAPI_OUTB_DELIVERY_CHANGE'
    EXPORTING
      header_data    = ls_header_data
      header_control = ls_header_control
      delivery       = ls_data_0100-vbeln
      techn_control  = ls_techn_control
    TABLES
      item_data      = lt_item_data
      item_control   = lt_item_control
      return         = return
      item_data_spl  = lt_item_data_spl.

  READ TABLE return WITH KEY type = 'E'
    TRANSPORTING NO FIELDS.

  IF sy-subrc EQ 0.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    EXIT.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = 'X'.
  ENDIF.

ENDFUNCTION.
