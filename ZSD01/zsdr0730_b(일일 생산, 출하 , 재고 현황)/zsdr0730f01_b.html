FORM get_product .

  DATA gt_product_tmp LIKE gt_product OCCURS 0 WITH HEADER LINE.

  LOOP AT gs_lgort.
    SELECT
        budat
        werks
        lgort
        a~matnr
        bwart
        shkzg
        menge
        meins

    INTO CORRESPONDING FIELDS OF TABLE gt_product_tmp
     FROM mseg AS a

      INNER JOIN mkpf AS b
      ON a~mblnr = b~mblnr
      AND b~mjahr = b~mjahr

     WHERE budat IN ra_budat
      AND  werks  = gs_lgort-werks.
*       AND ( bwart = '131' OR  bwart = '132' )
*      AND ( lgort = gs_lgort-lgort_l OR  lgort = gs_lgort-lgort_e ).


    DELETE gt_product_tmp  WHERE  bwart  NE '131' AND  bwart NE'132' .
    DELETE gt_product_tmp  WHERE  lgort NE gs_lgort-lgort_l
                           AND  lgort NE gs_lgort-lgort_e .


    APPEND LINES OF gt_product_tmp TO gt_product.
  ENDLOOP.

  LOOP AT gt_product.

                                   BINARY SEARCH.

      gt_product-cate1 = gt_zsdt0700-cate1.
    ENDIF.

    MODIFY gt_product.
    CLEAR gt_product.
  ENDLOOP.

  SORT gt_product BY cate1 werks budat.



  DATA lv_budat LIKE mkpf-budat.
  DATA lv_tabix LIKE sy-tabix.

  SORT gt_zsdt0700t BY cate1.

    CLEAR lv_budat.
    LOOP AT gs_lgort.

      LOOP AT gt_product  WHERE cate1 = gt_zsdt0700t-cate1

        MOVE-CORRESPONDING gt_product TO gs_product.

            WHEN  gs_lgort-lgort_e.
