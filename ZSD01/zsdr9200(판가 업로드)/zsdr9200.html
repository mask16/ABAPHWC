FORM generate_internal_table USING p_layout.

  DATA: ld_ref1     TYPE REF TO data,
        lt_alv_cat  LIKE lvc_s_fcat OCCURS 0 WITH HEADER LINE.


  CLEAR: it_dd03l[].

  SELECT * INTO TABLE it_dd03l
    FROM dd03l
    WHERE tabname = gs_t681-kotab
      AND keyflag = 'X'.

  SORT it_dd03l BY position.

  LOOP AT it_dd03l.
    IF it_dd03l-fieldname = 'MANDT' OR
       it_dd03l-fieldname = 'KAPPL' OR
       it_dd03l-fieldname = 'KSCHL' OR
       it_dd03l-fieldname = 'KFRST' OR
       it_dd03l-fieldname = 'DATBI'.
      DELETE it_dd03l.
      CONTINUE.
    ENDIF.
  ENDLOOP.

  CLEAR: lt_alv_cat[].

  LOOP AT it_dd03l.
    CLEAR lt_alv_cat.
    IF p_layout = 'X'.
      lt_alv_cat-datatype = 'CHAR'.
      lt_alv_cat-intlen = '30'.
    ELSE.
      lt_alv_cat-ref_table = it_dd03l-tabname.
      lt_alv_cat-ref_field = it_dd03l-fieldname.
    ENDIF.
    lt_alv_cat-fieldname = it_dd03l-fieldname.
    APPEND lt_alv_cat.
  ENDLOOP.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'CHAR'.
  lt_alv_cat-intlen = '15'.
  lt_alv_cat-fieldname = 'VALUE'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'CUKY'.
  lt_alv_cat-intlen = '8'.
  lt_alv_cat-fieldname = 'KONWA'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'CHAR'.
  lt_alv_cat-intlen = '15'.
  lt_alv_cat-fieldname = 'KPEIN'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'UNIT'.
  lt_alv_cat-intlen = '8'.
  lt_alv_cat-fieldname = 'KMEIN'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  IF p_layout = 'X'.
    lt_alv_cat-datatype = 'CHAR'.
    lt_alv_cat-intlen = '30'.
  ELSE.
    lt_alv_cat-ref_table = 'A304'.
    lt_alv_cat-ref_field = 'DATBI'.
  ENDIF.
  lt_alv_cat-fieldname = 'DATAB'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  IF p_layout = 'X'.
    lt_alv_cat-datatype = 'CHAR'.
    lt_alv_cat-intlen = '30'.
  ELSE.
    lt_alv_cat-ref_table = 'A304'.
    lt_alv_cat-ref_field = 'DATBI'.
  ENDIF.
  lt_alv_cat-fieldname = 'DATBI'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'CHAR'.
  lt_alv_cat-intlen = '30'.
  lt_alv_cat-fieldname = 'MESSAGE'.
  APPEND lt_alv_cat.

  CLEAR lt_alv_cat.
  lt_alv_cat-datatype = 'CHAR'.
  lt_alv_cat-intlen = '01'.
  lt_alv_cat-fieldname = 'SEL'.
  APPEND lt_alv_cat.

  CALL METHOD cl_alv_table_create=>create_dynamic_table
    EXPORTING
      it_fieldcatalog = lt_alv_cat[]
    IMPORTING
      ep_table        = ld_ref1.

  ASSIGN ld_ref1->* TO <it_tab>.

ENDFORM. " generate_internal_table
