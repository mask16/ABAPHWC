  "FIELD ## READ
  IF   p_gub =  'S'.

          WITH KEY fieldname  = p_fname.

  "## MOVE
  DATA l_col(40).
  FIELD-SYMBOLS <fs>.
  CONCATENATE 'GS_FIELDCAT-' p_fname  INTO l_col.
  ASSIGN      (l_col)         TO       <fs>.
  MOVE         p_con          TO       <fs>.

  IF p_fname = 'SCRTEXT_S'.
    gs_fieldcat-colddictxt = 'S'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_L'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.
  ELSEIF p_fname = 'SCRTEXT_L'.
    gs_fieldcat-colddictxt = 'L'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.
  ELSEIF p_fname = 'SCRTEXT_M'.
    gs_fieldcat-colddictxt = 'M'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_L'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.
  ENDIF.

  "DATA  APPEND
