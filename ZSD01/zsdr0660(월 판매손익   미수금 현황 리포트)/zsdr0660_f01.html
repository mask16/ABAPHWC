FORM variant_init USING p_layout.

  DATA: gs_vari TYPE disvariant.

  CLEAR gs_vari.
  gs_vari-report   = sy-repid.
  gs_vari-username = sy-uname.

  IF p_layout <> ''.
    gs_vari-variant = p_layout.
  ENDIF.

  CALL FUNCTION 'LVC_VARIANT_DEFAULT_GET'
    EXPORTING
      i_save        = 'A'
    CHANGING
      cs_variant    = gs_vari
    EXCEPTIONS
      wrong_input   = 1
      not_found     = 2
      program_error = 3
      OTHERS        = 4.

  CASE sy-subrc.
    WHEN 0.
      p_layout = gs_vari-variant.
    WHEN 2.
      CLEAR p_layout.
  ENDCASE.

ENDFORM. " VARIANT_INIT

    "영업사원(본사)
    READ TABLE gt_vbpa WITH KEY vbeln = gs_detail-vbeln
                                parvw = 'S1'
                                BINARY SEARCH.
    IF sy-subrc = 0.

    ENDIF.


    IF gs_detail-yunnr1 NOT IN s_kunnr.
      CONTINUE.
    ENDIF.
    "영업사원(현지)
*    READ TABLE gt_vbpa WITH KEY vbeln = gs_detail-vbeln
*                                parvw = 'S2'
*                                BINARY SEARCH.
*    IF sy-subrc = 0.
*      gs_detail-yunnr2  = gt_vbpa-kunnr.
*    ENDIF.

    READ TABLE gt_vbkd WITH KEY vbeln = gs_detail-aubel

      gs_detail-inco2 = gt_vbkd-inco2.
    ENDIF.

      gs_detail-fkimg = gs_detail-fkimg * 0.
    ENDIF.

      IF gs_detail-fktyp EQ 'A'.
        gs_detail-fkimg = gs_detail-fkimg * 0.
      ELSE.
        gs_detail-fkimg = gs_detail-fkimg * -1.
      ENDIF.
    ENDIF.

      IF gs_detail-fktyp EQ 'A'.
        gs_detail-fkimg = gs_detail-fkimg * 0.
      ENDIF.
    ENDIF.

    "금액관련 마이너스
    IF gs_detail-fkart EQ 'G2' OR gs_detail-fkart EQ 'RE'.
      gs_detail-netwr = gs_detail-netwr * -1.
    ENDIF.

    IF gs_detail-fkart EQ 'S1'.
      gs_detail-netwr = gs_detail-netwr * -1.
    ENDIF.


    "회계연도, 회계전표번호
    READ TABLE gt_bkpf WITH KEY bukrs = gs_detail-bukrs
                                awtyp = 'VBRK'
                                awkey = gs_detail-vbeln
                                BINARY SEARCH.
    IF sy-subrc = 0.
      gs_detail-waers = gt_bkpf-waers.
      gs_detail-kursf = gt_bkpf-kursf.
      gs_detail-hwaer = gt_bkpf-hwaer.
      gs_detail-gjahr = gt_bkpf-gjahr.
      gs_detail-belnr = gt_bkpf-belnr.
*      gs_detail-xref2_hd = gt_bkpf-xref2_hd.
    ENDIF.

    "상품코드 제품
    CLEAR l_werks.
    l_werks = gs_detail-werks.
    CASE gs_detail-werks.
      WHEN '1050'.
        l_werks = '2050'.
      WHEN '1060'.
        l_werks = '3060'.
    ENDCASE.

    "자재원가정보
    READ TABLE gt_coprc WITH KEY werks = l_werks
                                 matnr = gs_detail-matnr
                                 BINARY SEARCH INTO gs_coprc.
    IF sy-subrc = 0.
      MOVE-CORRESPONDING gs_coprc TO gs_detail.

      IF gs_detail-losgr <> 0.
        "실제단가=원가합/원가계산로트
        gs_detail-pvprs =
            ( gs_detail-kst001 + gs_detail-kst003 + gs_detail-kst005 + gs_detail-kst007

            / gs_detail-losgr.
      ENDIF.

      IF gs_detail-lcwrs <> 'KRW'.
        CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
          EXPORTING
            date             = v_lastd
            foreign_amount   = gs_detail-pvprs
            foreign_currency = gs_detail-lcwrs

      gs_detail-costp = gs_detail-pvprs * gs_detail-fkimg.

      "상품원가=구매단가
      IF gs_detail-matnr+10(1) = '3'.
        SELECT SINGLE a~ebeln b~ebelp
        INTO (gs_detail-ebeln, gs_detail-ebelp)
        FROM ekko AS a
        JOIN ekpo AS b ON a~ebeln = b~ebeln
        JOIN ekkn AS c ON a~ebeln = c~ebeln
                       AND b~ebelp = c~ebelp
                       AND c~zekkn = '1'
        WHERE c~vbeln = gs_detail-aubel
          AND c~vbelp = gs_detail-aupos.

        IF sy-subrc = 0.
          "3자거래 비용 조회
          CLEAR: gt_rbkp, gt_rbkp[].

          SELECT a~gjahr a~belnr a~budat a~waers a~kursf
                 b~buzei b~wrbtr b~menge b~bstme b~shkzg
          INTO CORRESPONDING FIELDS OF TABLE gt_rbkp
          FROM rbkp AS a
          JOIN rseg AS b ON a~gjahr = b~gjahr
                         AND a~belnr = b~belnr
          WHERE b~ebeln = gs_detail-ebeln
            AND b~ebelp = gs_detail-ebelp.

          IF sy-subrc = 0.
            LOOP AT gt_rbkp.
              IF gt_rbkp-shkzg = 'H'. "대변
                gt_rbkp-wrbtr = gt_rbkp-wrbtr * -1.
              ENDIF.

              IF gt_rbkp-waers <> 'KRW'.
                CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
                  EXPORTING
                    date             = gt_rbkp-budat
                    foreign_amount   = gt_rbkp-wrbtr
                    foreign_currency = gt_rbkp-waers
                    local_currency   = 'KRW'
                  IMPORTING
                    local_amount     = gt_rbkp-wrbtr.
              ENDIF.

              gs_detail-pvprs = gs_detail-pvprs + ( gt_rbkp-wrbtr / gt_rbkp-menge ).
            ENDLOOP.

            "매출원가
            gs_detail-costp = gs_detail-pvprs * gs_detail-fkimg.
            IF gs_detail-pvprs < 0 AND  gs_detail-fkimg < 0.
              gs_detail-costp = gs_detail-costp * -1.
            ENDIF.
            gs_detail-prsfg = '3자거래PO'.
          ENDIF.

        ELSE.
          "매출-구매 연결 없는 경우 이동평균단가
          PERFORM get_verpr_mbew USING gs_detail-matnr
                                       gs_detail-werks
                                       p_spmon
                                       v_lastd
                      CHANGING gs_detail-pvprs.

          gs_detail-costp = gs_detail-pvprs * gs_detail-fkimg.
          gs_detail-prsfg = '기간이동평균'.
        ENDIF.

      ELSE.
        "제품 원가 정보 누락 색깔 표시
      ENDIF.

    ENDIF.


    CLEAR: l_fcurr.

    IF gs_detail-belnr IS NOT INITIAL.
      l_fcurr = gs_detail-waers.
    ELSE.
      l_fcurr = gs_detail-waerk.
    ENDIF.

    IF l_fcurr <> 'KRW'.
      CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
        EXPORTING
          date             = gs_detail-fkdat
          foreign_amount   = gs_detail-netwr
          foreign_currency = l_fcurr
          local_currency   = 'KRW'
        IMPORTING
          local_amount     = gs_detail-netkrw.
    ELSE.
      gs_detail-netkrw = gs_detail-netwr.
    ENDIF.

    "판매이익
    gs_detail-prfit = gs_detail-netkrw - gs_detail-costp.


    MOVE-CORRESPONDING gs_detail TO gs_data.

    gs_data-hwaer = 'KRW'.
    gs_data-vkorg = gs_detail-bukrs.
    "내수/수출 구분은 사업장으로 결정

      WHEN OTHERS.

    ENDCASE.

    CLEAR gs_data-zterm.

    COLLECT gs_data INTO gt_data.
    CLEAR gs_data.

    MODIFY gt_detail FROM gs_detail.

  ENDLOOP.

  "매출실적이 없는 데이터 삭제
  DELETE gt_data WHERE fkimg = 0 AND netkrw = 0.




  SORT gt_data BY vkorg kunag vkbur yunnr1.

  LOOP AT gt_ar INTO gs_ar.

    CASE gs_ar-vkbur(1).

FORM get_verpr_mbew USING p_matnr p_werks p_spmon p_lastd
                     CHANGING p_pvprs.

  DATA: t_mbewh TYPE TABLE OF mbewh,
        s_mbewh TYPE mbewh.

  DATA: l_spmon TYPE spmon,
        l_tabix TYPE sy-tabix,
        l_waers TYPE t001-waers.

  CLEAR: t_mbewh, s_mbewh.

  SELECT matnr bwkey bwtar verpr
  INTO CORRESPONDING FIELDS OF TABLE t_mbewh
  FROM mbew
  WHERE matnr = p_matnr
    AND bwkey = p_werks.

  LOOP AT t_mbewh INTO s_mbewh.
    s_mbewh-lfgja = sy-datum(4).
    s_mbewh-lfmon = sy-datum+4(2).
    MODIFY t_mbewh FROM s_mbewh.
  ENDLOOP.

  SELECT matnr bwkey bwtar verpr
         lfgja lfmon
  APPENDING CORRESPONDING FIELDS OF TABLE t_mbewh
  FROM mbewh
  WHERE matnr = p_matnr
    AND bwkey = p_werks.

  SORT t_mbewh DESCENDING BY lfgja lfmon.

  LOOP AT t_mbewh INTO s_mbewh.
    l_tabix = sy-tabix.

    CLEAR l_spmon.
    CONCATENATE s_mbewh-lfgja s_mbewh-lfmon
      INTO l_spmon.

    "전월 기준 최근일자인 경우
    IF l_spmon EQ p_spmon.
      p_pvprs = s_mbewh-verpr.
      EXIT.

    ELSEIF l_spmon LT p_spmon.
      IF l_tabix = 1.
        p_pvprs = s_mbewh-verpr.
        EXIT.
      ELSE.
        l_tabix = l_tabix - 1.
        CLEAR s_mbewh.
        READ TABLE t_mbewh INDEX l_tabix INTO s_mbewh.
        p_pvprs = s_mbewh-verpr.
        EXIT.
      ENDIF.
    ENDIF.
  ENDLOOP.

  "KRW 환산
  CLEAR l_waers.
  SELECT SINGLE a~waers
  INTO (l_waers)
  FROM t001 AS a
  JOIN t001w AS b ON a~bukrs = b~vkorg
  WHERE b~werks = p_werks.

  IF l_waers <> 'KRW'.
    CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
      EXPORTING
        date             = p_lastd
        foreign_amount   = p_pvprs
        foreign_currency = l_waers
        local_currency   = 'KRW'
      IMPORTING
        local_amount     = p_pvprs.
  ENDIF.


ENDFORM. " GET_VERPR_MBEW
