*----------------------------------------------------------------------*
FORM double_click USING p_e_row TYPE lvc_s_row
                        p_e_column
                        p_es_row_no.

  DATA: ls_data_0100 LIKE LINE OF gt_data_0100.

  CHECK p_e_row-index GT 0.
  READ TABLE gt_data_0100 INDEX p_e_row-index INTO ls_data_0100.
  CHECK sy-subrc EQ 0.

  CASE p_e_column.
    WHEN 'VBELN'.
*   판매문서
      SET PARAMETER ID 'AUN' FIELD ls_data_0100-vbeln.
      CALL TRANSACTION 'VA03' AND SKIP FIRST SCREEN.
   WHEN 'KUNNR' OR 'NAME1'.
*   판매처, 판매처명
      SET PARAMETER ID 'KUN' FIELD ls_data_0100-kunnr.
      CALL TRANSACTION 'XD03' AND SKIP FIRST SCREEN.
   WHEN 'MATNR' OR 'MAKTX'.
*     자재번호, 자재명
      SET PARAMETER ID 'MAT' FIELD ls_data_0100-matnr.
      SET PARAMETER ID 'MXX' FIELD 'KV'.
      SET PARAMETER ID 'VTW' FIELD '10'.

  DATA: lt_rows      TYPE lvc_t_row,
        lt_head_0100 LIKE gt_data_0100,
        lt_item_0100 LIKE gt_data_0100,
        lt_return    TYPE bapiret2_t,
