FORM SELECT_DATA .

  PERFORM SET_DATA. "SELECT 조건 set

* 1) 납품문서내역 불러오기
  SELECT A~VBELN B~POSNR A~VKBUR A~WADAT A~KUNNR B~MATNR B~LFIMG
       INTO CORRESPONDING FIELDS OF TABLE GT_LIKP
     FROM LIKP AS A INNER JOIN LIPS AS B
         ON A~VBELN EQ B~VBELN INNER JOIN VBUK AS C
         ON A~VBELN EQ C~VBELN
   WHERE A~KUNNR  IN S_KUNNR
        AND A~WADAT IN S_WADAT
        AND A~VSTEL IN S_WERKS
        AND A~LFART NOT IN ('LR', 'RL') "반품
        AND B~LFIMG GT 0
        AND B~LGORT IN GR_LGORT
        AND C~WBSTK EQ 'A'.

  IF GT_LIKP IS NOT INITIAL.
    SELECT B~VBELN B~POSNR A~PBARCODE A~WERKS AS PWERKS A~LGORTN AS PLGORTN
                 B~MOVE_TYPE B~MATNR B~LFIMG B~MEINS B~WERKS B~LGORT
                 B~LGORT_TO B~ZSTATS B~MBLNR B~EBELN
         INTO CORRESPONDING FIELDS OF TABLE GT_DATA
        FROM ZSDT5031 AS A INNER JOIN ZSDT5030 AS B
            ON A~VBELN EQ B~VBELN
          AND A~POSNR EQ B~POSNR
          FOR ALL ENTRIES IN GT_LIKP
      WHERE B~ZSTATS        NE '10' "임시저장 외 전부
          AND B~MOVE_TYPE EQ 'GI' "공장 출하
          AND B~VBELN          EQ GT_LIKP-VBELN
          AND B~POSNR         EQ GT_LIKP-POSNR.
  ENDIF.

  SORT GT_DATA BY VBELN POSNR.
  SORT GT_LIKP  BY VBELN POSNR.

  IF GT_DATA IS NOT INITIAL.
    PERFORM GET_DATA. " TXT 및 납품 수량 가져오기
  ENDIF.

ENDFORM. " SELECT_DATA
