*----------------------------------------------------------------------*
FORM SELECT_DATA_0100 .

  DATA: LS_DATA_0100   LIKE LINE OF GT_DATA_0100,
        LV_ROWCOUNT    TYPE CHAR10.

  CLEAR: GT_DATA_0100.

  PERFORM SAPGUI_PROGRESS_INDICATOR USING 1 '조회중...'.

  SELECT  A~SPMON
         A~VKBUR
         A~CONFLG
         B~BEZEI   AS VKBTX
    INTO CORRESPONDING FIELDS OF TABLE GT_DATA_0100
    FROM zsdt1401  AS A
   INNER JOIN
         TVKBT     AS B
      ON A~VKBUR   EQ B~VKBUR
     AND B~SPRAS   EQ SY-LANGU
   WHERE A~SPMON   EQ PA_SPMON.

  LOOP AT GT_DATA_0100 INTO LS_DATA_0100.
    IF LS_DATA_0100-CONFLG EQ 'X'.
      LS_DATA_0100-CONTX = '마감'.
    ELSE.
      LS_DATA_0100-CONTX = '수정가능'.
    ENDIF.

    MODIFY GT_DATA_0100 FROM LS_DATA_0100.
  ENDLOOP.

* 타이틀 및 조회건수
  LV_ROWCOUNT = LINES( GT_DATA_0100 ).
  CONCATENATE '용접봉 시장조사 마감 관리' LV_ROWCOUNT
         INTO GV_TITLE_0100.

  PERFORM SAPGUI_PROGRESS_INDICATOR USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100
