*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: lt_data_0100  LIKE gt_data_0100,
        ls_data_0100  LIKE LINE OF gt_data_0100,
        lv_rowcount   TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT a~kunnr
         a~kkber
         a~klimk
         a~skfor
         a~ssobl
         b~vkbur
         b~vkorg  "추가_20171207
         b~waers
         b~kdgrp
         c~kunn2
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM knkk AS a
   INNER JOIN
         knvv AS b
      ON a~kunnr  EQ b~kunnr
   INNER JOIN
         knvp AS c
      ON a~kunnr  EQ c~kunnr
*     AND c~vkorg  EQ '1000'
     AND c~vtweg  EQ '10'
     AND c~spart  EQ '10'
     AND c~parvw  EQ 'S1'
   WHERE a~kunnr  IN so_kunnr
     AND b~vkbur  IN so_vkbur.

  SORT lt_data_0100 BY vkbur kunnr.
** 추가정보조회
  LOOP AT lt_data_0100 INTO ls_data_0100.
    "판매조직명
    READ TABLE gt_tvkot WITH KEY vkorg = ls_data_0100-vkorg
                                 spras = sy-langu BINARY SEARCH.
    IF sy-subrc = 0.
      ls_data_0100-vtext = gt_tvkot-vtext.
    ENDIF.

    "한국법인은 여신관리 통화를 KRW로 통일_20171207
    IF ls_data_0100-vkorg = '1000'.
      ls_data_0100-waers = 'KRW'.
    ENDIF.

*   여신관리영역명
    PERFORM get_t014t_text USING ls_data_0100-kkber
                        CHANGING ls_data_0100-kkbtx.
*   고객명
    PERFORM get_kna1_text USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-name.
*   담당자명
    PERFORM get_kna1_text USING ls_data_0100-kunn2
                        CHANGING ls_data_0100-name2.
*   고객그룹명
    PERFORM get_t151t_text USING ls_data_0100-kdgrp
                        CHANGING ls_data_0100-ktext.
*   사업자명
    PERFORM get_tvkbt_text USING ls_data_0100-vkbur
                        CHANGING ls_data_0100-bezei.
*   Open주문(여신)
    PERFORM get_oeikw      USING ls_data_0100-kunnr
                                 ls_data_0100-kkber
                        CHANGING ls_data_0100-oeikw.

*   Open납품(여신)
    PERFORM get_olikw      USING ls_data_0100-kunnr
                                 ls_data_0100-kkber
                        CHANGING ls_data_0100-olikw.

*   매출환산가치(Open주문(여신) + Open납품(여신))
    ls_data_0100-zsavw = ls_data_0100-oeikw + ls_data_0100-olikw.

*   여신사용액
    ls_data_0100-zusec = ls_data_0100-zsavw + ls_data_0100-skfor + ls_data_0100-ssobl.

*   여신사용율
    IF ls_data_0100-klimk IS NOT INITIAL.
      ls_data_0100-zrate = ( ls_data_0100-zusec / ls_data_0100-klimk ) * 100.
    ENDIF.

    IF ls_data_0100-zrate  >= 1000.
      ls_data_0100-zrate = ( 1 / 100 ) * 99999.
    ENDIF.

    IF ls_data_0100-zrate  < 1.
      ls_data_0100-zrate = 0.
    ENDIF.

*   가용여신액
    IF ls_data_0100-klimk = 1 AND ls_data_0100-zusec < 0.
      ls_data_0100-zbanc = ls_data_0100-zusec.
    ELSE.
      ls_data_0100-zbanc = ls_data_0100-klimk - ls_data_0100-zusec.
    ENDIF.

    APPEND ls_data_0100 TO gt_data_0100.
  ENDLOOP.



* 타이틀 및 조회건수
  lv_rowcount = lines( gt_data_0100 ).
  CONCATENATE '여신 현황 관리-' lv_rowcount
         INTO gv_title_0100.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100
