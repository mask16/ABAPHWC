FORM select_data_0100 .
  DATA: ls_data TYPE ihttpnvp.

  CLEAR: gt_data_100[], gs_data_100.

*  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  FIELD-SYMBOLS: <fs> LIKE gs_data_100.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_data_100
    FROM zsdt0150 AS a INNER JOIN zsdt0151 AS b
                                          ON a~intno = b~intno
   WHERE a~vkbur  IN so_vkbur
     AND a~spcode IN so_spcod
     AND a~kunnr  IN so_kunnr
     AND a~intno  IN so_intno
     AND a~apnum  IN so_apnum
     AND a~crdat  IN so_crdat
     AND a~status IN so_statu
     AND a~qmsts  IN so_qmsts
     AND b~fecod  IN so_fecod
     AND a~loevm  EQ space.

  LOOP AT gt_data_100 ASSIGNING <fs>.
    READ TABLE: gt_status INTO gs_status WITH KEY name  = <fs>-status,
                gt_qmsts  INTO gs_qmsts WITH KEY name   = <fs>-qmsts,
                gt_vkbur  INTO gs_vkbur WITH KEY name   = <fs>-vkbur,
                gt_fecod  INTO gs_fecod WITH KEY name   = <fs>-fecod,
                gt_mara   INTO gs_mara  WITH KEY matnr  = <fs>-matnr,
                gt_maktx  INTO gs_maktx WITH KEY name   = <fs>-matnr.

    MOVE: gs_status-value TO <fs>-status_txt,
          gs_qmsts-value  TO <fs>-qmsts_txt,
          gs_vkbur-value  TO <fs>-bezei,
          gs_fecod-value  TO <fs>-fecod_txt,
          gs_mara-groes   TO <fs>-groes,
          gs_mara-zmaktx  TO <fs>-zmaktx,
          gs_maktx-value  TO <fs>-maktx.

    IF NOT ( <fs>-spcode CN '0123456789 ' ).
      PERFORM conversion_alpha_input CHANGING <fs>-spcode.
    ENDIF.
    READ TABLE gt_kna1 INTO gs_kna1 WITH KEY kunnr = <fs>-spcode.
    <fs>-spcode_txt = gs_kna1-name1.

    CLEAR gs_kna1.

    IF NOT ( <fs>-kunnr CN '0123456789 ' ).
      PERFORM conversion_alpha_input CHANGING <fs>-kunnr.
    ENDIF.
    READ TABLE gt_kna1 INTO gs_kna1 WITH KEY kunnr = <fs>-kunnr.
    <fs>-kunnr_txt = gs_kna1-name1.

    IF <fs>-remark1 IS NOT INITIAL. <fs>-re1 = 'X'. ENDIF.
    IF <fs>-remark2 IS NOT INITIAL. <fs>-re2 = 'X'. ENDIF.
    IF <fs>-remark3 IS NOT INITIAL. <fs>-re3 = 'X'. ENDIF.

    CLEAR: gs_status, gs_qmsts, gs_vkbur, gs_mara, gs_maktx, gs_kna1.
  ENDLOOP.
ENDFORM. " SELECT_DATA_0100

                          USING ps_fieldcat TYPE lvc_s_fcat
