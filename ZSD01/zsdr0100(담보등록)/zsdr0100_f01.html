FORM start_of_selection .

  SELECT SINGLE *
    FROM kna1
   WHERE kunnr EQ pa_kunnr.

  IF sy-subrc NE 0.
    MESSAGE s003 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

* 담보구분에 따른 등록화면 분기.
  CASE pa_gubun.
    WHEN 'A'.
*     부동산
      PERFORM call_screen_0100.
    WHEN 'B'.
*     질권
      PERFORM call_screen_0200.
    WHEN 'C'.
*     지급보증서
      PERFORM call_screen_0300.
    WHEN 'D'.
*     수출보험한도
      PERFORM call_screen_0400.
    WHEN OTHERS.
      MESSAGE s004 DISPLAY LIKE 'E'.
  ENDCASE.

ENDFORM. " START_OF_SELECTION

FORM save_zsdt0100 USING ps_data TYPE typ_data_0100.

  DATA: ls_zsdt0100  TYPE zsdt0100.

  MOVE-CORRESPONDING ps_data TO ls_zsdt0100.
* Max 순번
  PERFORM get_max_intnr CHANGING ls_zsdt0100-intnr.

  ADD 1 TO ls_zsdt0100-intnr.
* 생성일, 변경일
  ls_zsdt0100-erdat = sy-datum.
  ls_zsdt0100-aedat = ls_zsdt0100-erdat.
* 생성시각, 변경시각
  ls_zsdt0100-erzet = sy-uzeit.
  ls_zsdt0100-aezet = ls_zsdt0100-erzet.
* 생성자, 변경자.
  ls_zsdt0100-ernam = sy-uname.
  ls_zsdt0100-aenam = ls_zsdt0100-ernam.

* 테이블에 추가.
  INSERT zsdt0100 FROM ls_zsdt0100.

  IF sy-subrc EQ 0.
    COMMIT WORK.
    MESSAGE s001.
    PERFORM exit_process_0100 USING space.
  ELSE.
    ROLLBACK WORK.
    MESSAGE s002 DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM. " SAVE_ZSDT0100

FORM f4_fpath CHANGING p_fpath.

  DATA : lv_filename  TYPE file_table-filename.

* 업로드할 파일명 가져오기.
  PERFORM get_filename CHANGING lv_filename.
  IF lv_filename IS INITIAL .
    MESSAGE s000 WITH '파일경로가 올바르지 않습니다.'.
    EXIT .
  ENDIF .

  IF strlen( lv_filename ) GT 255 .
    MESSAGE s000 WITH '파일경로는 255자 이내이여야 합니다.'.
    EXIT .
  ENDIF .

  p_fpath = lv_filename.

ENDFORM. " F4_FPATH

FORM get_filename CHANGING p_filepath.

  DATA: wa_rc      TYPE i VALUE 1,
        wa_title   TYPE string VALUE 'File open',
        lt_files   TYPE filetable,
        ls_files   TYPE file_table.

  CLEAR: p_filepath.
  CALL METHOD cl_gui_frontend_services=>file_open_dialog
    EXPORTING
      window_title      = wa_title
      default_filename  = space
    CHANGING
      file_table        = lt_files
      rc                = wa_rc.

  CHECK lt_files IS NOT INITIAL.
  READ TABLE lt_files INDEX 1 INTO ls_files.
  CHECK ls_files-filename IS NOT INITIAL.
  MOVE ls_files-filename TO p_filepath.

ENDFORM. " GET_FILENAME

FORM ws_execute USING p_fpath.

  DATA: lt_dynpread TYPE icl_dynpread_t,
        ls_dynpread LIKE LINE OF lt_dynpread.

  ls_dynpread-fieldname = p_fpath.

    EXPORTING
