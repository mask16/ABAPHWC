*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_check_data .

  DATA: BEGIN OF lt_date OCCURS 0,
        frdat   LIKE zsdt0301-frdat,
        todat   LIKE zsdt0301-todat,
        spmon   LIKE zsdt0301-spmon,
        spywk   LIKE zsdt0301-spywk,
        END OF lt_date.
*
  DATA: ls_date     LIKE LINE OF gt_date,
        ls_date2    LIKE LINE OF gt_date.
*
  CLEAR: gt_date.

*[S] 계획기준일자 가져오기
  CLEAR: lt_date,      lt_date[].
  SELECT frdat
         todat
         spmon
         spywk
         INTO CORRESPONDING FIELDS OF TABLE lt_date
         FROM zsdt0302
        WHERE spdat =  pa_spdat
          AND vkbur IN gr_vkbur.

FORM get_week_info USING p_spdat
                 CHANGING ps_date TYPE typ_date.

  DATA: lv_firstday TYPE sy-datum,
        lv_lastday  TYPE sy-datum.

  CLEAR: ps_date.

  CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'

      monday = ps_date-frdat

  CONCATENATE p_spdat(6) '01'

    EXPORTING
      i_date = p_spdat

      e_date = lv_lastday.

  IF ps_date-todat GT lv_lastday.
    ps_date-todat = lv_lastday.
  ENDIF.

* 시작일의 년주차
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date = ps_date-frdat
    IMPORTING
      week = ps_date-spywk.

* 시작일의 년월
  ps_date-spmon = ps_date-frdat(6).

ENDFORM. " GET_WEEK_INFO

         INTO lv_text.

  CONCATENATE p_text lv_text
         INTO lv_text RESPECTING BLANKS.

  PERFORM sapgui_progress_indicator USING lv_barsize lv_text.
