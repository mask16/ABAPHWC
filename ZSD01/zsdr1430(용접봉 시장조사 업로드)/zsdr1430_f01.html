FORM GET_EXCEL .
  CLEAR : GT_UPLOAD, GT_UPLOAD[], GT_DATA, GT_DATA[].

  DATA: LF_RETRN TYPE CHAR1.

  DATA : LT_EXCEL_TAB  TYPE ALSMEX_TABLINE OCCURS 0 WITH HEADER LINE.
  DATA : LV_INDEX    LIKE SY-TABIX.

  DATA : BEGIN OF LT_ANLA OCCURS 0,
           BUKRS LIKE ANLA-BUKRS,
           ANLN1 LIKE ANLA-ANLN1,
           SERNR LIKE ANLA-SERNR,
         END OF LT_ANLA.

  FIELD-SYMBOLS : <FS> TYPE ANY.


  CALL FUNCTION 'WS_QUERY' " 파일 존재여부 체크
    EXPORTING
      FILENAME = FILENAME"P_FILE
      QUERY    = 'FE'
    IMPORTING
      RETURN   = LF_RETRN.      " '0' = 없다,  '1' = 존재한다

  IF LF_RETRN = '0'.

      EXPORTING

        I_BEGIN_COL             = GC_BEGIN_COL
        I_BEGIN_ROW             = GC_BEGIN_ROW
        I_END_COL               = GC_END_COL
        I_END_ROW               = GC_END_ROW
      TABLES
        INTERN                  = LT_EXCEL_TAB
      EXCEPTIONS
        INCONSISTENT_PARAMETERS = 1
        UPLOAD_OLE              = 2
        OTHERS                  = 3.
  ENDIF.


  LOOP AT LT_EXCEL_TAB WHERE ROW GE 2.
    MOVE : LT_EXCEL_TAB-COL TO LV_INDEX.
    ASSIGN COMPONENT LV_INDEX OF STRUCTURE GT_UPLOAD TO <FS>.
    MOVE : LT_EXCEL_TAB-VALUE TO <FS>.

      PERFORM CONVERT_INPUT_MENGE  USING GT_UPLOAD-USGFCW.

      APPEND GT_UPLOAD.
      CLEAR  GT_UPLOAD.
    ENDAT.
  ENDLOOP.

  LOOP AT GT_UPLOAD.
    MOVE-CORRESPONDING GT_UPLOAD TO GT_DATA.

    APPEND GT_DATA.

  ENDLOOP.
