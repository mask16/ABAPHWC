FORM fieldcat_build_proc.
  DATA: l_table TYPE tabname,
        l_field TYPE name_feld.
  DATA: ls_fieldcat TYPE lvc_s_fcat.

  LOOP AT it_field.
    IF NOT it_field-fieldname CA '-'.
      WRITE:/ ' Data format error(Table-Field)'.
      g_check = 'X'.    EXIT.
    ENDIF.

    SPLIT it_field-fieldname AT '-' INTO l_table l_field.
    SELECT SINGLE * FROM dd03l WHERE tabname   = l_table
                                 AND fieldname = l_field
                                 AND as4local  = 'A'.
    IF sy-subrc <> 0.
      WRITE:/ ' Not exist in DD02L', l_table, l_field.
      g_check = 'X'.     CONTINUE.
    ENDIF.

    ls_fieldcat-fieldname = it_field-dynfnam.
    ls_fieldcat-ref_field = l_field.
    ls_fieldcat-ref_table = l_table.

    ls_fieldcat-domname   = dd03l-domname.
    ls_fieldcat-rollname  = dd03l-rollname.
    ls_fieldcat-datatype  = dd03l-datatype.
    ls_fieldcat-inttype   = dd03l-inttype.
    ls_fieldcat-intlen    = dd03l-intlen.
    ls_fieldcat-decimals  = dd03l-decimals.
    APPEND ls_fieldcat TO gt_fieldcat.
  ENDLOOP.

ENDFORM. "fieldcat_build_proc

FORM call_transaction_proc.
  FIELD-SYMBOLS: <ls_table>.

  LOOP AT it_tcode.
    g_count = sy-tabix.
    READ TABLE it_tcode INDEX g_count.
    PERFORM bdc_get_transaction USING it_tcode-bdc_index.

    ASSIGN LOCAL COPY OF INITIAL LINE OF <gt_table> TO <ls_table>.

    LOOP AT <gt_table> INTO <ls_table> .
      CLEAR: bdc_data, bdc_data[].
      bdc_data[] = bdc_table[].

      LOOP AT it_field.
        ASSIGN COMPONENT it_field-dynfnam OF STRUCTURE <ls_table>
                                          TO <g_field>.
        CLEAR gt_fieldcat.
        READ TABLE gt_fieldcat WITH KEY fieldname = it_field-dynfnam.
        IF gt_fieldcat-datatype = 'CURR' AND gt_fieldcat-decimals > 0.
          g_value1 = trunc( <g_field> ) .
          g_value2 = <g_field> - g_value1.
          IF g_value2 = 0.
            WRITE <g_field> TO bdc_data-fval LEFT-JUSTIFIED DECIMALS 0.
          ELSE.
            WRITE <g_field> TO bdc_data-fval LEFT-JUSTIFIED .
          ENDIF.
        ELSEIF gt_fieldcat-datatype = 'DATS' AND <g_field> IS INITIAL.
          CLEAR bdc_data-fval.
        ELSE.
          WRITE <g_field> TO bdc_data-fval LEFT-JUSTIFIED .
        ENDIF.
        MODIFY bdc_data TRANSPORTING fval
                          WHERE fval = it_field-dynfnam
                            AND fnam <> 'BDC_CURSOR'.
      ENDLOOP.

      PERFORM func_call_transaction_proc  USING it_tcode-tcode .
      PERFORM write_result_proc.
      IF sy-batch IS INITIAL.
        PERFORM progress_indicator_disp.
      ENDIF.
    ENDLOOP.
  ENDLOOP.

ENDFORM. "call_transaction_proc

FORM func_call_transaction_proc USING p_tcode.
  REFRESH messtab.

  CALL TRANSACTION it_tcode-tcode USING bdc_data
                                         OPTIONS FROM ctu_params
                                         MESSAGES INTO messtab.
  IF sy-subrc = 0.
    ADD 1 TO g_count_ok.
  ELSE.
    ADD 1 TO g_count_error.
  ENDIF.

ENDFORM. " func_call_transaction_proc

FORM write_result_proc.

  WRITE:/01(10) it_tcode-tcode,  12(08) sy-subrc LEFT-JUSTIFIED .

  DESCRIBE TABLE messtab LINES sy-tfill.
  IF sy-tfill <= 0.
    WRITE: text-009.
  ELSE.
    LOOP AT messtab.
      SELECT SINGLE * FROM t100 WHERE sprsl = messtab-msgspra
                                AND   arbgb = messtab-msgid
                                AND   msgnr = messtab-msgnr.
      IF sy-subrc = 0.
        g_mstring2 = t100-text.
        IF g_mstring2 CS '&1'.
          REPLACE '&1' WITH messtab-msgv1 INTO g_mstring2.
          REPLACE '&2' WITH messtab-msgv2 INTO g_mstring2.
          REPLACE '&3' WITH messtab-msgv3 INTO g_mstring2.
          REPLACE '&4' WITH messtab-msgv4 INTO g_mstring2.
        ELSE.
          REPLACE '&' WITH messtab-msgv1 INTO g_mstring2.
          REPLACE '&' WITH messtab-msgv2 INTO g_mstring2.
          REPLACE '&' WITH messtab-msgv3 INTO g_mstring2.
          REPLACE '&' WITH messtab-msgv4 INTO g_mstring2.
        ENDIF.
        CONDENSE g_mstring2.
        CONCATENATE messtab-msgtyp messtab-msgid messtab-msgnr
                    INTO g_mstring1 SEPARATED BY space.
        WRITE: 21(07) g_mstring1, 30(100) g_mstring2.
      ELSE.
        WRITE: 30(120) messtab.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDFORM. " write_result_proc

FORM progress_indicator_disp.

  g_loop_cnt  = g_loop_cnt + 1.
  g_ratio = g_loop_cnt * 100 / g_load_cnt.

  CHECK g_ratio > g_counter.
  g_msg+00(03) = g_ratio.
  g_msg+04(01) = '%'.
  g_msg+17(06) = ' BDC 실행중입니다 !!!'.

  CALL FUNCTION 'SAPGUI_PROGRESS_INDICATOR'
    EXPORTING
      percentage = g_ratio
      text       = g_msg
    EXCEPTIONS
      OTHERS     = 1.

  g_counter = g_counter + 1.

ENDFORM. " progress_indicator_disp
