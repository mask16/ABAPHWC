*&---------------------------------------------------------------------*
*&      Form  SELECT_DATA_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: lt_data_0100   LIKE gt_data_0100,
        lr_belnr       TYPE RANGE OF zsdt0130-belnr,
        ls_belnr       LIKE LINE OF lr_belnr,
        ls_data_0100   LIKE LINE OF gt_data_0100,
        lv_rowcount    TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.
* 상태조건 생성(처리, 미처리)
  CASE 'X'.
    WHEN pa_r2.
*     처리
      ls_belnr-sign = 'I'.
      ls_belnr-option = 'NE'.
      ls_belnr-low = space.
      APPEND ls_belnr TO lr_belnr.
    WHEN pa_r3.
*     미처리
      ls_belnr-sign = 'I'.
      ls_belnr-option = 'EQ'.
      ls_belnr-low = space.
      APPEND ls_belnr TO lr_belnr.
  ENDCASE.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM zsdt0130
   WHERE admon IN so_admon
     AND dlpnt IN so_dlpnt
     AND lifnr IN so_lifnr
     AND kostl IN so_kostl.

  LOOP AT lt_data_0100 INTO ls_data_0100.
*   회계전표 역분개여부
    PERFORM get_bkpf_stblg USING ls_data_0100-bukrs
                                 ls_data_0100-belnr
                                 ls_data_0100-gjahr
                        CHANGING ls_data_0100-stblg.

    IF ls_data_0100-stblg IS NOT INITIAL.
      CLEAR: ls_data_0100-bukrs,
             ls_data_0100-belnr,
             ls_data_0100-gjahr.
    ENDIF.
    MODIFY lt_data_0100 FROM ls_data_0100.
  ENDLOOP.

* 상태조회조건 적용
  LOOP AT lt_data_0100 INTO ls_data_0100
                      WHERE belnr IN lr_belnr.
*   상태
    IF ls_data_0100-belnr IS INITIAL.
      ls_data_0100-stats = icon_led_yellow.
    ELSE.
      ls_data_0100-stats = icon_led_green.
    ENDIF.
*   출하처명
    PERFORM get_vstel_text USING ls_data_0100-dlpnt
                        CHANGING ls_data_0100-dlptx.
*   운송업체명
    PERFORM get_lifnr_text USING ls_data_0100-lifnr
                        CHANGING ls_data_0100-liftx.
*   코스트센터명
    PERFORM get_kostl_text USING ls_data_0100-kostl
                        CHANGING ls_data_0100-ktext.
*   입력자명
    PERFORM get_adrp_name_text USING ls_data_0100-ernam
                            CHANGING ls_data_0100-ertxt.

    APPEND ls_data_0100 TO gt_data_0100.
  ENDLOOP.

* 타이틀 및 조회건수
  lv_rowcount = lines( gt_data_0100 ).
  CONCATENATE '내수 운반비 정산 -' lv_rowcount
         INTO gv_title_0100.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100

*&---------------------------------------------------------------------*
*&      Form  CREATE_0100
*&---------------------------------------------------------------------*
*       임시전표 생성
*----------------------------------------------------------------------*
FORM create_0100 .

  DATA: lt_rows        TYPE lvc_t_row,
        lt_data_0100   LIKE gt_data_0100,
        lt_bapiret2    TYPE bapiret2_t,
        ls_rows        LIKE LINE OF lt_rows,
        ls_data_0100   LIKE LINE OF lt_data_0100,
        ls_bapiret2    LIKE LINE OF lt_bapiret2,
        lv_valid       TYPE char01,
        lv_answer      TYPE char01,
        lv_index       TYPE sy-tabix.

  CALL METHOD gv_grid_0100->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows.
* 선택행 체크
  IF lines( lt_rows ) LE 0.
    MESSAGE s006.
    EXIT.
  ENDIF.
* 선택한 행의 데이터로 테이블생성
  LOOP AT lt_rows INTO ls_rows WHERE rowtype EQ space.
    CLEAR: ls_data_0100.
    READ TABLE gt_data_0100 INDEX ls_rows-index
                             INTO ls_data_0100.
    APPEND ls_data_0100 TO lt_data_0100.
  ENDLOOP.
* 유효성 점검
  PERFORM validation_0100 USING lt_data_0100
                       CHANGING lv_valid.
  CHECK lv_valid EQ 'X'.
* 생성여부
  PERFORM popup_to_confirm USING '임시전표 생성 여부'
                                 '임시전표를 생성하시겠습니까?'
                                 ''
                                 ''
                        CHANGING lv_answer.
  CHECK lv_answer EQ '1'.
* 임시전표 생성
  PERFORM call_transaction_fb01 USING lt_data_0100
                             CHANGING lt_bapiret2.
* 성공시 상태 업데이트
  READ TABLE lt_bapiret2 WITH KEY type = 'E'
                             INTO ls_bapiret2.
  IF sy-subrc NE 0.
    WAIT UP TO '0.5' SECONDS.
    READ TABLE lt_data_0100 INTO ls_data_0100 INDEX 1.
    PERFORM get_bkpf_belnr USING ls_data_0100-admon
                                 ls_data_0100-fcnum
                        CHANGING ls_data_0100-budat
                                 ls_data_0100-belnr.
*   상태업데이트
    PERFORM save_zsdt0130 USING lt_data_0100
                                ls_data_0100-budat
                                ls_data_0100-belnr.
*   재조회
    PERFORM select_data_0100.
    MESSAGE s000 WITH '임시전표 생성 완료.'.
  ELSE.
*   메세지 설정
    LOOP AT lt_data_0100 INTO ls_data_0100.
      ls_data_0100-msgtb = lt_bapiret2.
      READ TABLE lt_bapiret2 WITH KEY type ='E'
                                 INTO ls_bapiret2.
      ls_data_0100-fbmsg = ls_bapiret2-message.
      MODIFY gt_data_0100 FROM ls_data_0100
        TRANSPORTING fbmsg msgtb WHERE fcnum = ls_data_0100-fcnum.
    ENDLOOP.
    MESSAGE s000 WITH '임시전표 생성중 오류발생.'.
  ENDIF.

  PERFORM refresh_grid USING gv_grid_0100
                             gs_layout.

ENDFORM. " CREATE_0100

*&---------------------------------------------------------------------*
*&      Form  DELETE_0100
*&---------------------------------------------------------------------*
*       임시전표 삭제
*----------------------------------------------------------------------*
FORM delete_0100 .

  DATA: lt_rows      TYPE lvc_t_row,
        lt_data_0100 LIKE gt_data_0100,
        lt_emsg_0100 LIKE gt_data_0100,
        ls_rows      LIKE LINE OF lt_rows,
        ls_data_0100 LIKE LINE OF lt_data_0100,
        ls_bapiret2  TYPE bapiret2,
        lv_answer    TYPE char01,
        lv_index     TYPE sy-tabix.

  CALL METHOD gv_grid_0100->get_selected_rows
    IMPORTING
      et_index_rows = lt_rows.

  IF lines( lt_rows ) LE 0.
    MESSAGE s006.
    EXIT.
  ENDIF.
* 삭제여부
  PERFORM popup_to_confirm USING '임시전표 삭제 여부'
                                 '임시전표를 삭제하시겠습니까?'
                                 ''
                                 ''
                        CHANGING lv_answer.
  CHECK lv_answer EQ '1'.

* 선택한 행의 데이터로 테이블생성
  LOOP AT lt_rows INTO ls_rows WHERE rowtype EQ space.
    CLEAR: ls_data_0100, ls_bapiret2.
    READ TABLE gt_data_0100 INDEX ls_rows-index
                             INTO ls_data_0100.
    APPEND ls_data_0100 TO lt_data_0100.
  ENDLOOP.
* 임시전표 그룹핑
  SORT lt_data_0100 BY belnr.
  DELETE ADJACENT DUPLICATES
    FROM lt_data_0100 COMPARING belnr.

  LOOP AT lt_data_0100 INTO ls_data_0100.
*   임시전표 삭제
    PERFORM call_transaction_fbv0 CHANGING ls_data_0100.
*   성공시 상태 업데이트
    READ TABLE ls_data_0100-msgtb WITH KEY type   = 'S'
                                           id     = 'FP'
                                           number = '047'
                                       INTO ls_bapiret2.
    IF sy-subrc EQ 0.
      UPDATE zsdt0130
         SET bukrs = space
             belnr = space
             gjahr = space
             budat = space
             aenam = sy-uname
             aedat = sy-datum
             aezet = sy-uzeit
       WHERE belnr EQ ls_data_0100-belnr.
      COMMIT WORK.
    ELSE.
      READ TABLE ls_data_0100-msgtb INDEX 1
                                     INTO ls_bapiret2.
      ls_data_0100-fbmsg = ls_bapiret2-message.
      APPEND ls_data_0100 TO lt_emsg_0100.
    ENDIF.
  ENDLOOP.

* 재조회
  PERFORM select_data_0100.
* 메세지 설정
  LOOP AT lt_emsg_0100 INTO ls_data_0100.
    MODIFY gt_data_0100 FROM ls_data_0100
      TRANSPORTING fbmsg msgtb WHERE belnr = ls_data_0100-belnr.
  ENDLOOP.

  PERFORM refresh_grid USING gv_grid_0100
                             gs_layout.

  IF lt_emsg_0100 IS NOT INITIAL.
    MESSAGE s000 WITH '임시전표 삭제 중 오류 발생.'.
  ENDIF.

ENDFORM. " DELETE_0100
