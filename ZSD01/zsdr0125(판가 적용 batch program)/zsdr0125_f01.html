FORM select_data_0100 .

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT a~intno
         b~item
    INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
    FROM zsdt0120  AS a
   INNER JOIN
         zsdt0121  AS b
      ON a~intno   EQ b~intno
   WHERE a~cfdat   IN so_cfdat
     AND a~status  IN ('C', 'D')
     AND b~xoflg   EQ space.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100

*&      Form  ZSD_SET_PRICE_PER_UNIT
*&---------------------------------------------------------------------*
*       판매단가 적용
*----------------------------------------------------------------------*
FORM zsd_set_price_per_unit .

  DATA: ls_zsdt0121  TYPE zsdt0121,
        ls_data_0100 LIKE LINE OF gt_data_0100,
        ls_return    TYPE bapiret2,
        lv_tcnt      TYPE numc10,
        lv_scnt      TYPE numc10.

  PERFORM sapgui_progress_indicator USING 1 '판매단가 적용중...'.

  LOOP AT gt_data_0100 INTO ls_data_0100.
    call function 'ZSD_SET_PRICE_PER_UNIT'
      EXPORTING
        i_intno   = ls_data_0100-intno
        i_item    = ls_data_0100-item
        i_dismode = pa_mode
      TABLES
        t_return  = ls_data_0100-msgtb.

    READ TABLE ls_data_0100-msgtb WITH KEY type = 'S'
                                           id   = 'VK'
                                      INTO ls_return.
    IF sy-subrc EQ 0.
      UPDATE zsdt0121 SET xodat = sy-datum
                          xoflg = 'X'
                          aenam = sy-uname
                          aedat = sy-datum
                          aezet = sy-uzeit
      WHERE intno EQ ls_data_0100-intno
        AND item  EQ ls_data_0100-item.
      IF sy-subrc EQ 0.
        ADD 1 TO lv_scnt.
        COMMIT WORK.
      ELSE.
        ROLLBACK WORK.
      ENDIF.
    ENDIF.

  ENDLOOP.

  PERFORM sapgui_progress_indicator USING 100 '판매단가 적용완료'.

* 결과처리
  lv_tcnt = lines( gt_data_0100 ).
  WRITE: '총 ', lv_tcnt, '건 중 ', lv_scnt, '건 성공.'.

ENDFORM. " ZSD_SET_PRICE_PER_UNIT
