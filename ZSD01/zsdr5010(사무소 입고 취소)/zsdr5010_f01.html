*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM select_data_0100 .

  CLEAR : gs_data_0100, gt_data_0100, gt_data_0100[].

  TYPES : BEGIN OF typ_werks,
            werks    TYPE t001w-werks,
            name1    TYPE t001w-name1,
          END OF typ_werks.

  TYPES : BEGIN OF typ_lgort,
            lgort    TYPE t001l-lgort,
            lgobe    TYPE t001l-lgobe,
          END OF typ_lgort.

  DATA : lt_werks TYPE STANDARD TABLE OF typ_werks,
         ls_werks TYPE typ_werks,
         lt_lgort TYPE STANDARD TABLE OF typ_lgort,
         ls_lgort TYPE typ_lgort.

  SELECT * FROM zsdt5010
  INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
  WHERE mtnum IN s_mtnum AND barcode IN s_bar
  AND pbarcode IN s_pbar AND lotmat IN s_lot
  AND matnr IN s_matnr AND charg IN s_charg
  AND bzgbn IN s_bzgbn AND recod IN s_recod
  AND erdat IN s_erdat AND refno IN s_ref.

  SELECT DISTINCT werks name1 FROM t001w
    INTO CORRESPONDING FIELDS OF TABLE lt_werks.

  SELECT DISTINCT lgort lgobe FROM t001l
    INTO CORRESPONDING FIELDS OF TABLE lt_lgort.

  SORT lt_werks BY werks. SORT lt_lgort BY lgort.

  LOOP AT gt_data_0100 INTO gs_data_0100.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = gs_data_0100-matnr
      IMPORTING
        output = gs_data_0100-matnr.

    READ TABLE lt_werks INTO ls_werks WITH KEY werks = gs_data_0100-werks BINARY SEARCH.
    IF sy-subrc = 0. gs_data_0100-name1 = ls_werks-name1. ENDIF.
    READ TABLE lt_lgort INTO ls_lgort WITH KEY lgort = gs_data_0100-frsloc BINARY SEARCH.
    IF sy-subrc = 0. gs_data_0100-fgobe = ls_lgort-lgobe. ENDIF.
    READ TABLE lt_lgort INTO ls_lgort WITH KEY lgort = gs_data_0100-tosloc BINARY SEARCH.
    IF sy-subrc = 0. gs_data_0100-tgobe = ls_lgort-lgobe. ENDIF.

    MODIFY gt_data_0100 FROM gs_data_0100.
  ENDLOOP.

ENDFORM. " SELECT_DATA_0100

         lv_flag TYPE char01,
         lt_row_table TYPE lvc_t_row WITH HEADER LINE.

  DATA : lt_deta TYPE STANDARD TABLE OF typ_data_mbst,
         ls_deta TYPE typ_data_mbst.

*-- 재처리여부

                                   '입고취소처리하시겠습니까?'
                                   ''
                                   ''
                          CHANGING lv_answer.
  CHECK lv_answer EQ '1'.

  CLEAR : gt_mbst, gs_mbst.

  CALL METHOD gv_grid_0100->get_selected_rows
    IMPORTING
      et_index_rows = lt_row_table[].

  IF lt_row_table[] IS INITIAL.
    MESSAGE s000 DISPLAY LIKE 'E' WITH text-003.
  ELSE.

    LOOP AT lt_row_table.
      READ TABLE gt_data_0100 INTO gs_data_0100 INDEX lt_row_table-index.
      MOVE-CORRESPONDING gs_data_0100 TO gs_mbst.
      COLLECT gs_mbst INTO gt_mbst. CLEAR gs_mbst.
    ENDLOOP.

    SORT gt_mbst BY mblnr. CLEAR lv_flag.
*-- 재처리는 결과코드의 Value가 'E'인 것만 대상이다.
    LOOP AT gt_mbst INTO gs_mbst WHERE recod <> 'S'.
      lv_flag = 'X'. EXIT.
    ENDLOOP.

    IF lv_flag = 'X'.
      MESSAGE s000 DISPLAY LIKE 'E' WITH text-004.
    ELSE.
      PERFORM mbst_data.
    ENDIF.
  ENDIF.

ENDFORM. " REXE_DATA
