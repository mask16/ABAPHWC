  SELECT * INTO TABLE LT_TEMP
    FROM ZSDT5053
   WHERE MATNR  IN SO_MATNR
     AND QCHARG IN SO_CHARG
     AND LIFNR  IN SO_LIFNR

    " =====================================================================

    CASE GS_DATA-ZSTATS.

FORM SAVE_DATA.
  DATA: LT_INS TYPE TABLE OF ZSDT5053,
        LT_UPD TYPE TABLE OF ZSDT5053,
        LS_DB  TYPE ZSDT5053,
        LV_ERR TYPE C.

  LOOP AT GT_DATA INTO GS_DATA WHERE MODE = 'I' OR MODE = 'U'.
    " 필수값 체크
    IF GS_DATA-MATNR IS INITIAL OR GS_DATA-QCHARG IS INITIAL.
      MESSAGE S000(00) WITH '자재번호와 제조번호는 필수입력입니다.' DISPLAY LIKE 'E'.
      LV_ERR = 'X'. EXIT.
    ENDIF.

    MOVE-CORRESPONDING GS_DATA TO LS_DB.

    IF GS_DATA-MODE = 'I'.
      " 중복 체크
      SELECT SINGLE MANDT INTO SY-MANDT FROM ZSDT5053 WHERE MATNR = LS_DB-MATNR AND QCHARG = LS_DB-QCHARG.
      IF SY-SUBRC = 0.
        MESSAGE S000(00) WITH LS_DB-MATNR LS_DB-QCHARG '는 이미 존재하는 제조번호입니다.' DISPLAY LIKE 'E'.
        LV_ERR = 'X'. EXIT.
      ENDIF.
      LS_DB-ERDAT = SY-DATUM. LS_DB-ERZET = SY-UZEIT. LS_DB-ERNAM = SY-UNAME.
      APPEND LS_DB TO LT_INS.
    ELSEIF GS_DATA-MODE = 'U'.
      LS_DB-UPDAT = SY-DATUM. LS_DB-UPTIM = SY-UZEIT. LS_DB-UPNAM = SY-UNAME.
      APPEND LS_DB TO LT_UPD.
    ENDIF.
  ENDLOOP.

  CHECK LV_ERR IS INITIAL.

  IF LT_INS[] IS NOT INITIAL. INSERT ZSDT5053 FROM TABLE LT_INS. ENDIF.
  IF LT_UPD[] IS NOT INITIAL. UPDATE ZSDT5053 FROM TABLE LT_UPD. ENDIF.

  COMMIT WORK AND WAIT.
  MESSAGE S000(00) WITH '저장 되었습니다.'.
  PERFORM SELECT_DATA. " 재조회
  PERFORM REFRESH_GRID.
ENDFORM. "SAVE_DATA

FORM EXCEL_UPLOAD.
  DATA: LT_FILE TYPE FILETABLE,
        LS_FILE TYPE FILE_TABLE,
        LV_RC   TYPE I.

  " 엑셀 파일 선택 팝업창
  CALL METHOD CL_GUI_FRONTEND_SERVICES=>FILE_OPEN_DIALOG
    EXPORTING
      FILE_FILTER = 'Excel Files (*.XLSX;*.XLS)|*.XLSX;*.XLS'
    CHANGING
      FILE_TABLE  = LT_FILE
      RC          = LV_RC.
  CHECK LV_RC = 1.

  DATA: LV_FILE TYPE RLGRAP-FILENAME.
  READ TABLE LT_FILE INTO LS_FILE INDEX 1.
  IF SY-SUBRC = 0.
    LV_FILE = LS_FILE-FILENAME.
  ENDIF.

  " ALSM 펑션을 위한 데이터 선언
  DATA: LT_ALSMEX TYPE STANDARD TABLE OF ALSMEX_TABLINE,
        LS_ALSMEX TYPE ALSMEX_TABLINE.

  " 가장 안정적인 ALSM 펑션으로 엑셀 읽기 (R/3 TR 창 안뜸, 자동 종료됨)
  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = LV_FILE
      I_BEGIN_COL             = 1
      I_BEGIN_ROW             = 2       " 2번째 줄부터 읽음 (첫 줄 헤더 무시)
      I_END_COL               = 6       " 엑셀의 총 열(Column) 개수
      I_END_ROW               = 65000   " 최대로 읽어올 행(Row) 개수
    TABLES
      INTERN                  = LT_ALSMEX
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      OTHERS                  = 3.

  IF SY-SUBRC <> 0.
    MESSAGE S000(00) WITH '엑셀 업로드에 실패했습니다. (열려있는 엑셀을 닫아주세요)' DISPLAY LIKE 'E'. EXIT.
  ENDIF.

  " 읽어온 클립보드 데이터를 내부 인터널 테이블 형태로 매핑
  TYPES: BEGIN OF TY_EXCEL,
           LIFNR  TYPE ZSDT5053-LIFNR,
           PDATE  TYPE CHAR8,
           MATNR  TYPE ZSDT5053-MATNR,
           QCHARG TYPE ZSDT5053-QCHARG,
           ZSTATS TYPE ZSDT5053-ZSTATS,
           REMARK TYPE ZSDT5053-REMARK,
         END OF TY_EXCEL.

  DATA: LT_EXCEL TYPE STANDARD TABLE OF TY_EXCEL,
        LS_EXCEL TYPE TY_EXCEL.

  " ALSM 데이터는 Row/Col 단위로 들어오므로 정렬 후 조립합니다.
  SORT LT_ALSMEX BY ROW COL.

  LOOP AT LT_ALSMEX INTO LS_ALSMEX.
    " 새로운 행(Row)이 시작될 때 구조체를 초기화
    AT NEW ROW.
      CLEAR LS_EXCEL.
    ENDAT.

    " 열(Col) 번호에 맞게 데이터 할당
    CASE LS_ALSMEX-COL.
      WHEN 1. LS_EXCEL-LIFNR  = LS_ALSMEX-VALUE.
      WHEN 2. LS_EXCEL-PDATE  = LS_ALSMEX-VALUE.
      WHEN 3. LS_EXCEL-MATNR  = LS_ALSMEX-VALUE.
      WHEN 4. LS_EXCEL-QCHARG = LS_ALSMEX-VALUE.
      WHEN 5. LS_EXCEL-ZSTATS = LS_ALSMEX-VALUE.
      WHEN 6. LS_EXCEL-REMARK = LS_ALSMEX-VALUE.
    ENDCASE.

    " 하나의 행(Row) 조립이 끝나면 인터널 테이블에 추가
    AT END OF ROW.
      APPEND LS_EXCEL TO LT_EXCEL.
    ENDAT.
  ENDLOOP.

  " 기존에 작성하신 데이터 가공 및 DB 조회 로직
  LOOP AT LT_EXCEL INTO LS_EXCEL.

    " 보이지 않는 특수문자 제거 및 공백 압축 (이전 단계에서 적용한 내용 유지)
    REPLACE ALL OCCURRENCES OF CL_ABAP_CHAR_UTILITIES=>CR_LF IN LS_EXCEL-LIFNR WITH ''.
    REPLACE ALL OCCURRENCES OF CL_ABAP_CHAR_UTILITIES=>HORIZONTAL_TAB IN LS_EXCEL-LIFNR WITH ''.
    CONDENSE LS_EXCEL-LIFNR NO-GAPS.

    REPLACE ALL OCCURRENCES OF CL_ABAP_CHAR_UTILITIES=>CR_LF IN LS_EXCEL-MATNR WITH ''.
    REPLACE ALL OCCURRENCES OF CL_ABAP_CHAR_UTILITIES=>HORIZONTAL_TAB IN LS_EXCEL-MATNR WITH ''.
    CONDENSE LS_EXCEL-MATNR NO-GAPS.

    " 자재번호 컨버전 루틴 추가 (0채우기)
    CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
      EXPORTING
        INPUT  = LS_EXCEL-MATNR
      IMPORTING
        OUTPUT = LS_EXCEL-MATNR.

    " 공급업체 컨버전 (LIFNR 10자리 DB에 맞춰 0 채우기)
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        INPUT  = LS_EXCEL-LIFNR
      IMPORTING
        OUTPUT = LS_EXCEL-LIFNR.

    CLEAR GS_DATA.
    GS_DATA-LIFNR  = LS_EXCEL-LIFNR.
    GS_DATA-PDATE  = LS_EXCEL-PDATE.
    GS_DATA-MATNR  = LS_EXCEL-MATNR.
    GS_DATA-QCHARG = LS_EXCEL-QCHARG.
    GS_DATA-ZSTATS = LS_EXCEL-ZSTATS.
    GS_DATA-REMARK = LS_EXCEL-REMARK.
    GS_DATA-MODE   = 'I'.

    SELECT SINGLE MAKTX INTO GS_DATA-MAKTX FROM MAKT WHERE MATNR = GS_DATA-MATNR AND SPRAS = SY-LANGU.
    SELECT SINGLE NAME1 INTO GS_DATA-LIFNR_NM FROM LFA1 WHERE LIFNR = GS_DATA-LIFNR.

    PERFORM SET_CELL_STYLE USING 'MATNR'  CL_GUI_ALV_GRID=>MC_STYLE_ENABLED CHANGING GS_DATA-CELLSTYLES.
    PERFORM SET_CELL_STYLE USING 'QCHARG' CL_GUI_ALV_GRID=>MC_STYLE_ENABLED CHANGING GS_DATA-CELLSTYLES.

    APPEND GS_DATA TO GT_DATA.
  ENDLOOP.

  " 화면 갱신 및 완료 메시지
  PERFORM REFRESH_GRID.
  MESSAGE S000(00) WITH '엑셀 데이터가 화면에 로드되었습니다. 확인 후 [저장]을 눌러주세요.'.
ENDFORM. "EXCEL_UPLOAD
