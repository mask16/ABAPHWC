CLASS LCL_EVENT_RECEIVER IMPLEMENTATION.
  METHOD HANDLE_DATA_CHANGED.
    DATA: LS_MOD_CELLS TYPE LVC_S_MODI,
          LV_MAKTX     TYPE MAKT-MAKTX,
          LV_NAME1     TYPE LFA1-NAME1.

    LOOP AT ER_DATA_CHANGED->MT_MOD_CELLS INTO LS_MOD_CELLS.
      READ TABLE GT_DATA INTO GS_DATA INDEX LS_MOD_CELLS-ROW_ID.
      CHECK SY-SUBRC = 0.

      CASE LS_MOD_CELLS-FIELDNAME.
          " 자재번호 입력 시 자재명 가져오기
        WHEN 'MATNR'.
          SELECT SINGLE MAKTX INTO LV_MAKTX FROM MAKT
           WHERE MATNR = LS_MOD_CELLS-VALUE
             AND SPRAS = SY-LANGU.
          CALL METHOD ER_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = 'MAKTX'
              I_VALUE     = LV_MAKTX.

          " 공급업체 입력 시 업체명 가져오기
        WHEN 'LIFNR'.
          DATA: LV_LIFNR TYPE LIFNR.

          " 입력된 값을 SAP 내부 포맷(앞에 0 채움)으로 변환
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
            EXPORTING
              INPUT  = LS_MOD_CELLS-VALUE
            IMPORTING
              OUTPUT = LV_LIFNR.

          " 변환된 값(LV_LIFNR)으로 LFA1 조회
          SELECT SINGLE NAME1 INTO LV_NAME1 FROM LFA1
           WHERE LIFNR = LV_LIFNR.

          CALL METHOD ER_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = 'LIFNR_NM'
              I_VALUE     = LV_NAME1.

          " 상태 입력 시 상태명 가져오기
        WHEN 'ZSTATS'.
          DATA: LV_STAT TYPE CHAR20. " 변수 명시적 선언

          " COND 대신 CASE 문 사용
          CASE LS_MOD_CELLS-VALUE.
            WHEN '10'.
              LV_STAT = '입고'.
            WHEN '20'.
              LV_STAT = '출고'.
            WHEN '99'.
              LV_STAT = '취소'.
            WHEN OTHERS.
              CLEAR LV_STAT.
          ENDCASE.

          CALL METHOD ER_DATA_CHANGED->MODIFY_CELL
            EXPORTING
              I_ROW_ID    = LS_MOD_CELLS-ROW_ID
              I_FIELDNAME = 'STATS_NM'
              I_VALUE     = LV_STAT.
      ENDCASE.

      " 수정된 행은 모드를 'U'로 변경 (단, 'I' 신규상태가 아닐때만)
      IF GS_DATA-MODE <> 'I'.
        CALL METHOD ER_DATA_CHANGED->MODIFY_CELL
          EXPORTING
            I_ROW_ID    = LS_MOD_CELLS-ROW_ID
            I_FIELDNAME = 'MODE'
            I_VALUE     = 'U'.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.                    "HANDLE_DATA_CHANGED
ENDCLASS. "LCL_EVENT_RECEIVER IMPLEMENTATION
