          lv_objid        LIKE  wwwdata-objid,
          lv_fname        LIKE  rlgrap-filename,

                          CHANGING v_answer.

      lv_objid  = 'ZSDR0035'.

          def_filename     = 'ZSDR0035_UPD.xls'

          filename         = lv_dfname
        EXCEPTIONS
          inv_winsys       = 1
          no_batch         = 2
          selection_cancel = 3
          selection_error  = 4
          OTHERS           = 5.

      IF  lv_dfname  IS NOT INITIAL.
        "Web 템플레이트(Excel Form) 다운로드

          EXPORTING
            key         = ls_wwwdatatab
