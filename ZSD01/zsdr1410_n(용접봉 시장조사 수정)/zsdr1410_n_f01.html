*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM SELECT_DATA_0100 .

  DATA: LT_DATA_0100   LIKE GT_DATA_0100,
*        LT_DATA_0100-1 LIKE GT_DATA_0100,
        LS_DATA_0100   LIKE LINE OF GT_DATA_0100,
*        LS_DATA_0100-1 LIKE LINE OF GT_DATA_0100,
*        LV_DATUM       TYPE DATUM,
        LV_ROWCOUNT    TYPE CHAR10.

  CLEAR: GT_DATA_0100.

  PERFORM SAPGUI_PROGRESS_INDICATOR USING 1 '조회중...'.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE LT_DATA_0100
    FROM ZSDT1400
   WHERE SPMON  EQ PA_SPMON
     AND PERSON IN SO_PERSN
     AND VKBUR  IN SO_VKBUR
     AND LANDX IN SO_LANDX
     AND BRTXT IN SO_BRTXT
     AND CUSGRP IN SO_CG
     AND USGTXT IN SO_UTXT
     AND TRTYP IN SO_TRTYP
     AND KTEXT_ZD IN SO_KZD.


  LOOP AT LT_DATA_0100 INTO LS_DATA_0100.
*   실행버전채크
    IF PA_EXTYP IS NOT INITIAL.
      IF LS_DATA_0100-EXTYP NE 'E'.
        CONTINUE.
      ENDIF.
    ENDIF.
    APPEND LS_DATA_0100 TO GT_DATA_0100.
  ENDLOOP.



* 타이틀 및 조회건수
  LV_ROWCOUNT = LINES( GT_DATA_0100 ).
  CONCATENATE '용접봉 시장조사 수정 -' LV_ROWCOUNT
         INTO GV_TITLE_0100.

  PERFORM SAPGUI_PROGRESS_INDICATOR USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100

FORM VALIDATION_0100 CHANGING P_VALID.

  DATA: LT_ZSDT1401 TYPE STANDARD TABLE OF ZSDT1401,
        LS_ZSDT1401 LIKE LINE OF LT_ZSDT1401.

  P_VALID = 'X'.

* 사무소별 마감 체크
  SELECT *
    INTO TABLE LT_ZSDT1401
    FROM ZSDT1401
   WHERE SPDAT EQ GV_SPDAT
     AND SPMON EQ PA_SPMON.

  LOOP AT LT_ZSDT1401 INTO LS_ZSDT1401 WHERE CONFLG EQ 'X'.
    READ TABLE GT_DATA_0100 WITH KEY VKBUR = LS_ZSDT1401-VKBUR
      TRANSPORTING NO FIELDS.

      CLEAR: P_VALID.
      MESSAGE S000 WITH '해당 기간에 대해  마감 되었습니다.' DISPLAY LIKE 'E'.
      EXIT.
    ENDIF.
  ENDLOOP.

FORM SAVE_DATA_0100 .

  DATA: LT_ZSDT1400  TYPE STANDARD TABLE OF ZSDT1400,
        LS_ZSDT1400  LIKE LINE OF LT_ZSDT1400,
        LS_DATA_0100 LIKE LINE OF GT_DATA_0100,
        LV_VALID     TYPE CHAR01,
        LV_REFRESH   TYPE CHAR01,
        LV_ANSWER    TYPE CHAR01,
        LV_SEQNR     TYPE ZSDT1400-SEQNR,
        LV_AEDAT     TYPE SY-DATUM,
        LV_AEZET     TYPE SY-UZEIT.

  CALL METHOD GV_GRID_0100->CHECK_CHANGED_DATA
    IMPORTING
      E_VALID   = LV_VALID
    CHANGING
      C_REFRESH = LV_REFRESH.
  CHECK LV_VALID EQ 'X'.

* 유효성 점검
  PERFORM VALIDATION_0100 CHANGING LV_VALID.
  CHECK LV_VALID EQ 'X'.

* 저장여부
  PERFORM  POPUP_TO_CONFIRM USING  '저장여부'
                                   '용접봉 시장조사를 저장하시겠습니까?'
                                   ''
                                   ''
                          CHANGING LV_ANSWER.
  CHECK LV_ANSWER EQ '1'.
* 저장일자
  LV_AEDAT = SY-DATUM.
* 저장시각
  LV_AEZET = SY-UZEIT.

  LOOP AT GT_DATA_0100 INTO LS_DATA_0100.
    CLEAR: LS_ZSDT1400.
    MOVE-CORRESPONDING LS_DATA_0100 TO LS_ZSDT1400.
*   수정인
    LS_ZSDT1400-AENAM = SY-UNAME.
*   수정일
    LS_ZSDT1400-AEDAT = LV_AEDAT.
*   수정시각
    LS_ZSDT1400-AEZET = LV_AEZET.
    APPEND LS_ZSDT1400 TO LT_ZSDT1400.
  ENDLOOP.


  MODIFY ZSDT1400 FROM TABLE LT_ZSDT1400.
  IF SY-SUBRC EQ 0.
    COMMIT WORK.
    PERFORM SELECT_DATA_0100.
    PERFORM REFRESH_GRID USING GV_GRID_0100
                               GS_LAYOUT.
    MESSAGE S001.
  ELSE.
    ROLLBACK WORK.
    MESSAGE S002 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

ENDFORM. " SAVE_DATA_0100
