FORM get_base_data .
  IF gt_t001l[] IS INITIAL.
    SELECT  werks lgort lgobe
      INTO CORRESPONDING FIELDS OF TABLE gt_t001l
      FROM t001l
     WHERE werks  IN s_werks
       AND lgort  IN s_lgort.
  ENDIF.

  IF gt_makt[] IS INITIAL.
    SELECT  matnr AS name
            maktx AS value
      INTO TABLE gt_makt
      FROM makt
     WHERE matnr  IN s_matnr
       AND spras  EQ sy-langu.
  ENDIF.

  IF gt_mara[] IS INITIAL.
    SELECT  matnr groes zmaktx
      INTO CORRESPONDING FIELDS OF TABLE gt_mara
      FROM mara
     WHERE matnr  IN s_matnr.
  ENDIF.

  IF gt_t023t[] IS INITIAL.
    SELECT  matkl AS name
            wgbez AS value
      INTO TABLE gt_t023t
      FROM t023t
     WHERE spras = sy-langu.
  ENDIF.

  IF gt_mard[] IS INITIAL.
    SELECT labst insme speme
      INTO CORRESPONDING FIELDS OF TABLE gt_mard
      FROM mard
     WHERE matnr  IN s_matnr
       AND werks  IN s_werks
       AND lgort  IN s_lgort.
  ENDIF.

  IF gt_t001w[] IS INITIAL.
    SELECT werks AS name
           name2 AS value
      INTO TABLE gt_t001w
      FROM t001w.
  ENDIF.
ENDFORM. " GET_BASE_DATA

FORM get_data .
  DATA: lt_zsdt0900 TYPE TABLE OF zsdt0900,
        ls_zsdt0900 TYPE zsdt0900,
        lt_temp     TYPE TABLE OF zsdt0900,
        ls_temp     TYPE zsdt0900.

  DATA: l_total TYPE zesdqty,
        l_fname TYPE fieldname.

  SELECT *
    INTO TABLE lt_zsdt0900
    FROM zsdt0900
   WHERE matkl  IN s_matkl
     AND matnr  IN s_matnr
     AND werks  IN s_werks
     AND lgort  IN s_lgort.

  CHECK lt_zsdt0900[] IS NOT INITIAL.
  SORT lt_zsdt0900 BY matkl matnr werks lgort period.
  lt_temp[] = lt_zsdt0900[].
  DELETE ADJACENT DUPLICATES FROM lt_temp
                          COMPARING matkl matnr werks lgort.

  LOOP AT lt_temp INTO ls_temp.
    PERFORM set_data_to_dyn_wa USING: '<DYN_WA>' 'MATNR' ls_temp-matnr,
                                      '<DYN_WA>' 'MATKL' ls_temp-matkl,
                                      '<DYN_WA>' 'LGORT' ls_temp-lgort,
                                      '<DYN_WA>' 'WERKS' ls_temp-werks,
                                      '<DYN_WA>' 'MEINS' ls_temp-meins.

    READ TABLE: gt_t001l INTO gs_t001l WITH KEY werks = ls_temp-werks
                                                lgort = ls_temp-lgort,
                gt_makt  INTO gs_makt  WITH KEY name = ls_temp-matnr,
                gt_mara  INTO gs_mara  WITH KEY matnr = ls_temp-matnr,
                gt_t023t INTO gs_t023t WITH KEY name = ls_temp-matkl,
                gt_mard  INTO gs_mard  WITH KEY matnr = ls_temp-matnr
                                                werks = ls_temp-werks
                                                lgort = ls_temp-lgort.

    PERFORM set_data_to_dyn_wa USING: '<DYN_WA>' 'LGOBE' gs_t001l-lgobe,
                                      '<DYN_WA>' 'MAKTX' gs_makt-value,
                                      '<DYN_WA>' 'ZMAKTX' gs_mara-zmaktx,
                                      '<DYN_WA>' 'GROES' gs_mara-groes,
                                      '<DYN_WA>' 'WGBEZ' gs_t023t-value,
                                      '<DYN_WA>' 'LABST' gs_mard-labst,
                                      '<DYN_WA>' 'INSME' gs_mard-insme,
                                      '<DYN_WA>' 'SPEME' gs_mard-speme.

    LOOP AT lt_zsdt0900 INTO ls_zsdt0900 WHERE matkl = ls_temp-matkl
                                           AND matnr = ls_temp-matnr
                                           AND werks = ls_temp-werks
                                           AND lgort = ls_temp-lgort.

      l_fname = ls_zsdt0900-period.
      CONDENSE l_fname.

      ASSIGN COMPONENT l_fname OF STRUCTURE <dyn_wa> TO <fm>.
      IF <fm> IS ASSIGNED.
        <fm> = ls_zsdt0900-qty.
      ELSE.
        ASSIGN COMPONENT 'PREV' OF STRUCTURE <dyn_wa> TO <fm>.
        ADD ls_zsdt0900-qty TO <fm>.
      ENDIF.

      ADD ls_zsdt0900-qty TO l_total.

      CLEAR: ls_zsdt0900, l_fname.
      UNASSIGN <fm>.
    ENDLOOP.

    ASSIGN COMPONENT 'TOTAL' OF STRUCTURE <dyn_wa> TO <fm>.
    <fm> = l_total.
    ASSIGN COMPONENT 'DIFF' OF STRUCTURE <dyn_wa> TO <fm>.
    <fm> = l_total - gs_mard-labst + gs_mard-insme + gs_mard-speme.

    APPEND <dyn_wa> TO <dyn_table>.
    CLEAR: ls_temp, l_total.
  ENDLOOP.
ENDFORM. " GET_DATA

FORM build_fieldcatalog TABLES pt_fieldcat TYPE lvc_t_fcat
                          USING ps_fieldcat TYPE lvc_s_fcat
                                p_gub
                                p_fname
                                p_value.
  DATA: l_name(40).
  FIELD-SYMBOLS: <fs> TYPE any.

  IF p_gub EQ 'S'.
    CLEAR ps_fieldcat.
  ENDIF.

  CONCATENATE 'PS_FIELDCAT-' p_fname INTO l_name.
  ASSIGN (l_name) TO <fs>.
  <fs> = p_value.

  IF p_gub EQ 'E'.
    APPEND ps_fieldcat TO pt_fieldcat.
  ENDIF.
ENDFORM. "build_fieldcatalog

FORM get_dyn_table.
  FREE: dyn_table, dyn_wa.
  UNASSIGN: <dyn_table>, <dyn_wa>.

  PERFORM get_dyn_fieldcat TABLES gt_fcat1.
  CHECK gt_fcat1[] IS NOT INITIAL.

  CALL METHOD cl_alv_table_create=>create_dynamic_table
    EXPORTING
      it_fieldcatalog = gt_fcat1
    IMPORTING
      ep_table        = dyn_table.

  ASSIGN dyn_table->* TO <dyn_table>.
  CREATE DATA dyn_wa LIKE LINE OF <dyn_table>.
  ASSIGN dyn_wa->* TO <dyn_wa>.
