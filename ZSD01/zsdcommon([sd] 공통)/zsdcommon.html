FORM read_long_text USING p_ebeln
                              p_ebelp
                              p_ztxtid
                     CHANGING p_text.

  DATA : v_text_object LIKE ttxob-tdobject.
  DATA : v_docno LIKE thead-tdname.
  DATA : it_tline TYPE tline OCCURS 0 WITH HEADER LINE.

  CHECK p_ebeln IS NOT INITIAL AND
        p_ebelp IS NOT INITIAL.

  v_text_object = 'EKPO'.

  CONCATENATE p_ebeln p_ebelp INTO v_docno.
  CONDENSE v_docno.

  CLEAR   : it_tline.
  REFRESH : it_tline.

  CALL FUNCTION 'READ_TEXT'
    EXPORTING
      client                  = sy-mandt
      id                      = p_ztxtid
      language                = sy-langu
      name                    = v_docno
      object                  = v_text_object
    TABLES
      lines                   = it_tline
    EXCEPTIONS
      id                      = 1
      language                = 2
      name                    = 3
      not_found               = 4
      object                  = 5
      reference_check         = 6
      wrong_access_to_archive = 7
      OTHERS                  = 8.

  IF sy-subrc = 0.
    READ TABLE it_tline INDEX 1.
    CLEAR   : p_text.
    p_text = it_tline-tdline.
  ENDIF.

ENDFORM. " GET_PALLET_WEIGHT

*&---------------------------------------------------------------------*
*       취소된 송장이 있는지 검사
*----------------------------------------------------------------------*
FORM check_iv_cancel_document USING p_belnr_iv
                                        p_gjahr
                               CHANGING p_belnr_iv_can.
  TYPE-POOLS : mmcr,
               mrm.

  DATA : rbkpv TYPE  mrm_rbkpv.
** IV에 대한 전표를 가져온다
  CALL FUNCTION 'MRM_INVOICE_READ'
    EXPORTING
      i_belnr         = p_belnr_iv
      i_gjahr         = p_gjahr
    IMPORTING
      e_rbkpv         = rbkpv
    EXCEPTIONS
      entry_not_found = 1
      lock_error      = 2
      OTHERS          = 3.

** 취소된 송장이 있는지 검사
  CALL FUNCTION 'MRM_REVERSAL_CHECK'
    EXPORTING
      i_rbkpv                    = rbkpv
    EXCEPTIONS
      invoice_not_posted         = 1
      invoice_cancelled_yet      = 2
      transfer_price_not_allowed = 3
      nota_fiscal_error          = 4
      OTHERS                     = 5.

  IF sy-subrc = 0.
    p_belnr_iv_can   = ''.
  ELSEIF sy-subrc = 2.
    p_belnr_iv_can  = 'X'.
  ENDIF.

ENDFORM. " CHECK_IV_CANCEL_DOCUMENT
