FORM select_data_0100 .

  DATA: lt_zsdt0900    TYPE STANDARD TABLE OF zsdt0900,
        ls_zsdt0900    LIKE LINE OF lt_zsdt0900,
        ls_data_0100   LIKE LINE OF gt_data_0100,
        lv_datum       TYPE sy-datum,
        lv_uzeit       TYPE sy-uzeit.

  lv_datum = sy-datum.
  lv_uzeit = sy-uzeit.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT matkl
         matnr
         werks
         lgort
         period
         qty
         meins
    INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
    FROM zsdt0910
   WHERE matkl  IN so_matkl
     AND matnr  IN so_matnr
     AND werks  IN so_werks
     AND lgort  IN so_lgort
     AND period IN so_perio.

  LOOP AT gt_data_0100 INTO ls_data_0100.
    CLEAR: ls_zsdt0900.
    MOVE-CORRESPONDING ls_data_0100 TO ls_zsdt0900.
*   사용자로그
    ls_zsdt0900-ernam = sy-uname.
    ls_zsdt0900-erdat = lv_datum.
    ls_zsdt0900-erzet = lv_uzeit.
    ls_zsdt0900-aenam = sy-uname.
    ls_zsdt0900-aedat = lv_datum.
    ls_zsdt0900-aezet = lv_uzeit.

    COLLECT ls_zsdt0900 INTO lt_zsdt0900.
  ENDLOOP.

  IF lt_zsdt0900 IS INITIAL.
    WRITE '데이터가 없습니다.'.
    EXIT.
  ENDIF.

  DELETE
    FROM zsdt0900
   WHERE matkl  IN so_matkl
     AND matnr  IN so_matnr
     AND werks  IN so_werks
     AND lgort  IN so_lgort
     AND period IN so_perio.

  IF sy-subrc NE 0.
    ROLLBACK WORK.
    WRITE '삭제 중 오류 발생'.
    EXIT.
  ENDIF.
  MODIFY zsdt0900 FROM TABLE lt_zsdt0900.
  IF sy-subrc EQ 0.
    COMMIT WORK.
    WRITE '정상 처리되었습니다.'.
  ELSE.
    ROLLBACK WORK.
    WRITE '저장 중 오류 발생'.
  ENDIF.

  PERFORM sapgui_progress_indicator USING 100 '저장 완료'.

ENDFORM. " SELECT_DATA_0100
