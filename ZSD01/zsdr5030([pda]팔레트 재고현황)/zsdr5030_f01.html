FORM SET_DATA .

  DATA : LV_KEY_FROM TYPE CHAR5,
             LV_KEY_TO   TYPE CHAR5.

  IF S_MAKTX[] IS NOT INITIAL.
    SELECT B~MATNR MAKTX INTO TABLE GT_MAKT
       FROM MARA AS A INNER JOIN MAKT AS B
          ON A~MATNR EQ B~MATNR
    WHERE MAKTX IN S_MAKTX
        AND SPRAS EQ'3'.
  ENDIF.

  " ① FROM만 입력된 경우
  IF S_DATE-LOW IS NOT INITIAL AND S_DATE-HIGH IS INITIAL.

    PERFORM CONVERT_DATE_TO_PBAR_KEY USING S_DATE-LOW
                                  CHANGING LV_KEY_FROM.

    CLEAR GR_PBAR.
    GR_PBAR-SIGN   = 'I'.
    GR_PBAR-OPTION = 'CP'.           " LIKE 패턴
    GR_PBAR-LOW    = LV_KEY_FROM && '*'.
    APPEND GR_PBAR TO GR_PBAR.

    " ② TO만 입력된 경우
  ELSEIF S_DATE-LOW IS INITIAL AND S_DATE-HIGH IS NOT INITIAL.

    PERFORM CONVERT_DATE_TO_PBAR_KEY USING S_DATE-HIGH
                                  CHANGING LV_KEY_TO.

    CLEAR GR_PBAR.
    GR_PBAR-SIGN   = 'I'.
    GR_PBAR-OPTION = 'CP'.
    GR_PBAR-LOW    = LV_KEY_TO && '*'.
    APPEND GR_PBAR TO GR_PBAR.

    " ③ FROM ~ TO 둘 다 입력된 경우
  ELSEIF S_DATE-LOW IS NOT INITIAL AND S_DATE-HIGH IS NOT INITIAL.

    PERFORM CONVERT_DATE_TO_PBAR_KEY USING S_DATE-LOW
                                  CHANGING LV_KEY_FROM.
    PERFORM CONVERT_DATE_TO_PBAR_KEY USING S_DATE-HIGH
                                  CHANGING LV_KEY_TO.

    CLEAR GR_PBAR.
    GR_PBAR-SIGN   = 'I'.
    GR_PBAR-OPTION = 'BT'.           " BETWEEN

    GR_PBAR-HIGH   = LV_KEY_TO   && '*'.
    APPEND GR_PBAR TO GR_PBAR.

*      <--P_LV_KEY_FROM  text
*----------------------------------------------------------------------*
FORM CONVERT_DATE_TO_PBAR_KEY
  USING    IV_DATE TYPE D
  CHANGING CV_KEY  TYPE CHAR5.   " YYMMDD → YYαDD (5자리)

  DATA: LV_YEAR  TYPE C LENGTH 2,
        LV_MONTH TYPE I,
        LV_DAY   TYPE C LENGTH 2,
        LV_ALPHA TYPE C LENGTH 1,
        LV_MONTH_MAP TYPE C LENGTH 13 VALUE 'ABCDEFGHIJKLM',
        LV_OFFSET TYPE I.

  LV_YEAR  = IV_DATE+2(2).   " YYYY 중 뒤 2자리
  LV_MONTH = IV_DATE+4(2).
  LV_DAY   = IV_DATE+6(2).

  LV_OFFSET = LV_MONTH - 1.

  LV_ALPHA = LV_MONTH_MAP+LV_OFFSET(1).

  CONCATENATE LV_YEAR LV_ALPHA LV_DAY INTO CV_KEY.

ENDFORM. " CONVERT_DATE_TO_PBAR_KEY
