*&      Form  DATA_CHANGED
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM data_changed USING
                p_data_changed TYPE REF TO cl_alv_changed_data_protocol
                p_onf4         TYPE char01.

  DATA: ls_s_modi      TYPE lvc_s_modi,
        ls_s_moce      TYPE lvc_s_moce,
        ls_data_0100   LIKE LINE OF gt_data_0100.

* 라인추가
  LOOP AT p_data_changed->mt_inserted_rows INTO ls_s_moce.
*   기본값 설정
    PERFORM set_inserted_value_0100 USING p_data_changed
                                          ls_s_moce
                                          ls_data_0100.
  ENDLOOP.

* 변경
  LOOP AT p_data_changed->mt_mod_cells INTO ls_s_modi.
    IF ls_s_modi-error EQ 'X'.
      CONTINUE.
    ENDIF.

    CASE ls_s_modi-fieldname.
      WHEN 'LIFNR'.
*       운송업체
        CALL METHOD p_data_changed->get_cell_value
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'LIFNR'
          IMPORTING
            e_value     = ls_data_0100-lifnr.
*       운송업체명
        PERFORM get_lifnr_text USING ls_data_0100-lifnr
                            CHANGING ls_data_0100-liftx.
        CALL METHOD p_data_changed->modify_cell
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'LIFTX'
            i_value     = ls_data_0100-liftx.
      WHEN 'KOSTL'.
*       코스트센터
        CALL METHOD p_data_changed->get_cell_value
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'KOSTL'
          IMPORTING
            e_value     = ls_data_0100-kostl.
*       코스트센터명
        PERFORM get_kostl_text USING ls_data_0100-kostl
                            CHANGING ls_data_0100-ktext.
        CALL METHOD p_data_changed->modify_cell
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'KTEXT'
            i_value     = ls_data_0100-ktext.
      WHEN 'AMOUNT'.
*       운송비
        CALL METHOD p_data_changed->get_cell_value
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'AMOUNT'
          IMPORTING
            e_value     = ls_data_0100-amount.
*       부가세
        ls_data_0100-vat = ls_data_0100-amount / 10.
        CALL METHOD p_data_changed->modify_cell
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'VAT'
            i_value     = ls_data_0100-vat.
*       총액
        ls_data_0100-total = ls_data_0100-amount + ls_data_0100-vat.
        CALL METHOD p_data_changed->modify_cell
          EXPORTING
            i_row_id    = ls_s_modi-row_id
            i_fieldname = 'TOTAL'
            i_value     = ls_data_0100-total.
    ENDCASE.
  ENDLOOP.

ENDFORM. " DATA_CHANGED

*&---------------------------------------------------------------------*
*&      Form  SELECT_DATA_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: ls_data_0100   LIKE LINE OF gt_data_0100.

  CLEAR: gt_data_0100, gs_data_0100.
* 정산년월
  gs_data_0100-admon = pa_admon.
* 출하처
  gs_data_0100-dlpnt = pa_dlpnt.
* 출하처명
  PERFORM get_vstel_text USING gs_data_0100-dlpnt
                      CHANGING gs_data_0100-dlptx.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
    FROM zsdt0130
   WHERE admon EQ gs_data_0100-admon
     AND dlpnt EQ gs_data_0100-dlpnt.

  LOOP AT gt_data_0100 INTO ls_data_0100.
*   운송업체명
    PERFORM get_lifnr_text USING ls_data_0100-lifnr
                        CHANGING ls_data_0100-liftx.
*   코스트센터명
    PERFORM get_kostl_text USING ls_data_0100-kostl
                        CHANGING ls_data_0100-ktext.
*   입력자명
    PERFORM get_adrp_name_text USING ls_data_0100-ernam
                            CHANGING ls_data_0100-ertxt.
    MODIFY gt_data_0100 FROM ls_data_0100.
  ENDLOOP.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100

FORM fill_celltab_0100 USING pv_mode
                    CHANGING pt_styl TYPE lvc_t_styl.

  DATA: ls_styl  TYPE lvc_s_styl.

  CLEAR pt_styl.
  REFRESH pt_styl.
  LOOP AT gt_fcat INTO gs_fcat.
*   카다로그의 지정 필드명 취득
    ls_styl-fieldname = gs_fcat-fieldname.
    CASE pv_mode.
      WHEN space.
        ls_styl-style = cl_gui_alv_grid=>mc_style_disabled.
      WHEN 'X'.
        CASE ls_styl-fieldname.
          WHEN 'LIFNR' OR 'KOSTL' OR 'AMOUNT' OR 'REMARK'.
            ls_styl-style = cl_gui_alv_grid=>mc_style_enabled.
          WHEN OTHERS.

        ENDCASE.
    ENDCASE.
    INSERT ls_styl INTO TABLE pt_styl.
  ENDLOOP.
* 삭제버튼 적용되지 않게 처리
  CLEAR ls_styl-fieldname.
  ls_styl-style = cl_gui_alv_grid=>mc_style_no_delete_row.
  INSERT ls_styl INTO TABLE pt_styl.
