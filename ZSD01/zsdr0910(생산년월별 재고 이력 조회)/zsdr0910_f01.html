*&---------------------------------------------------------------------*
*&      Form  DOUBLE_CLICK
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_E_ROW  text
*      -->P_E_COLUMN  text
*      -->P_ES_ROW_NO  text
*----------------------------------------------------------------------*
FORM double_click USING p_e_row TYPE lvc_s_row
                        p_e_column
                        p_es_row_no.

  DATA: ls_data_0100 LIKE LINE OF gt_data_0100.

  CHECK p_e_row-index GT 0.
  READ TABLE gt_data_0100 INDEX p_e_row-index INTO ls_data_0100.
  CHECK sy-subrc EQ 0.

  CASE p_e_column.
    WHEN 'EBELN'.
      SET PARAMETER ID 'BES' FIELD ls_data_0100-ebeln.
      CALL TRANSACTION 'ME23N' AND SKIP FIRST SCREEN.
    WHEN 'VBELN'.
      SET PARAMETER ID 'VL'  FIELD ls_data_0100-vbeln.
      CALL TRANSACTION 'VL03N' AND SKIP FIRST SCREEN.
    WHEN 'MBLNR' OR 'MJAHR'.
      SET PARAMETER ID 'MBN' FIELD ls_data_0100-mblnr.
      SET PARAMETER ID 'MJA' FIELD ls_data_0100-mjahr.
      CALL TRANSACTION 'MB03' AND SKIP FIRST SCREEN.
    WHEN 'MATNR' OR 'MAKTX'.
*     자재번호, 자재명
      SET PARAMETER ID 'MAT' FIELD ls_data_0100-matnr.
      SET PARAMETER ID 'MXX' FIELD 'KV'.
      SET PARAMETER ID 'VKO' FIELD '1000'.
      SET PARAMETER ID 'VTW' FIELD '10'.
      SET PARAMETER ID 'SPA' FIELD '10'.
      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.
  ENDCASE.

*&---------------------------------------------------------------------*
*&      Form  SELECT_DATA_0100
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_data_0100 .

  DATA: ls_data_0100   LIKE LINE OF gt_data_0100,
        lv_rowcount    TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_data_0100
    FROM zsdt0910
   WHERE erdat  IN so_erdat
     AND matkl  IN so_matkl
     AND matnr  IN so_matnr
     AND werks  IN so_werks
     AND lgort  IN so_lgort
     AND period IN so_perio
     AND pcharg IN so_pchar
     AND qcharg IN so_qchar
     AND mtype  IN so_mtype
     AND vbeln  IN so_vbeln
     AND zplnum IN so_zplnu
     AND ebeln  IN so_ebeln.

  LOOP AT gt_data_0100 INTO ls_data_0100.
*   자재명
    PERFORM get_matnr_text USING ls_data_0100-matnr
                        CHANGING ls_data_0100-maktx.

*   모델명, 규격명, 제품군, 자재그룹
    PERFORM get_mara_info USING ls_data_0100-matnr
                       CHANGING ls_data_0100-zmaktx
                                ls_data_0100-groes.

    MODIFY gt_data_0100 FROM ls_data_0100.
  ENDLOOP.

* 타이틀 및 조회건수
  lv_rowcount = lines( gt_data_0100 ).
  CONCATENATE '생산년월별 재고 이력 -' lv_rowcount
         INTO gv_title_0100.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100
