REPORT zrsadradrv_mass MESSAGE-ID f2.
*----------------------------------------------------------------------*
* This report checks for all KNA1-entries and LFA1-entries
* whether a correct ADRV-entry exists and whether the address pointed
* to really exists.
* If there is only one ADRV entry pointing to a different application
* table or if the address number referenced to does not exist then a
* new address for KNA1 respectively LFA1 will be created with the
* address data contained in KNA1 respectively LFA1 itself.
* If KNA1 respectively LFA1 points to an address having no ADRV entry
* then this pointer will be created.
*----------------------------------------------------------------------*
* Parameters:
*   TESTMODE: run report in test mode without changing database

FORM address_save USING p_kna1 LIKE kna1.

* initialize CAM-Memory
  CALL FUNCTION 'ADDR_MEMORY_CLEAR'
    EXPORTING
      force              = 'X'
    EXCEPTIONS
      unsaved_data_exist = 1
      internal_error     = 2
      OTHERS             = 3.

  MOVE p_kna1 TO kna1.
  CLEAR kna1-adrnr.

  read_address-master_tab   = 'KNA1'.
  read_address-master_field = 'ADRNR'.
  read_address-key_content  = kna1-kunnr.
  read_address-number       = kna1-adrnr.
  CLEAR read_person.

  PERFORM zav_lesen USING  read_address     read_person
                           c_type_1         c_init
                           c_conversion     c_maintain
                           c_no_export      c_no_reference.

  PERFORM zav_verbuchung USING c_not_in_update_task.
  IF zav_error <> 0.
    MESSAGE e804(f2) WITH kna1-kunnr.
  ENDIF.

  UPDATE kna1 SET adrnr = kna1-adrnr WHERE kunnr = p_kna1-kunnr.
  IF sy-subrc <> 0.
    MESSAGE e803(f2) WITH 'KNA1' p_kna1-kunnr.
  ENDIF.

  p_kna1-adrnr = kna1-adrnr.
  PERFORM switch_contacts USING p_kna1.

  COMMIT WORK.

  WRITE: / 'Customer', kna1-kunnr, '"repaired"...'.
ENDFORM. " ADDRESS_SAVE

FORM switch_contacts USING p_kna1 LIKE kna1.
  DATA: iknvk   LIKE knvk OCCURS 0 WITH HEADER LINE,
        old_ref LIKE pers_ref,
        new_ref LIKE pers_ref.

  SELECT * FROM knvk INTO TABLE iknvk
                     WHERE kunnr = p_kna1-kunnr.

  CHECK sy-subrc = 0.
  LOOP AT iknvk.
    SELECT SINGLE * FROM adcp
           WHERE persnumber = iknvk-prsnr
           AND   comp_pers  = 'C'.
    IF sy-subrc = 0.
      CALL FUNCTION 'ADDR_PERS_COMP_RELATION_SWITCH'
        EXPORTING
          address_number        = adcp-addrnumber
          person_number         = adcp-persnumber
          new_address_number    = p_kna1-adrnr
        IMPORTING
          returncode            = returncode
        TABLES
          error_table           = error_table
        EXCEPTIONS
          address_not_exist     = 1
          new_address_not_exist = 2
          parameter_error       = 3
          person_not_exist      = 4
          relation_not_exist    = 5
          new_relation_exists   = 6
          internal_error        = 7
          OTHERS                = 8.
      IF sy-subrc <> 0 OR returncode = 'E'.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

      old_ref-appl_table     = 'KNVK'.
      old_ref-appl_field     = 'PRSNR'.
      CONCATENATE sy-mandt iknvk-parnr INTO old_ref-appl_key.
      old_ref-pers_group     = c_address_group.
      old_ref-addrnumber     = adcp-addrnumber.
      CLEAR old_ref-pers_addr.
      old_ref-owner          = 'X'.

      new_ref = old_ref.
      new_ref-addrnumber     = p_kna1-adrnr.

      CALL FUNCTION 'ADDR_PERSON_REFERENCE_SWITCH'
        EXPORTING
          person_number           = adcp-persnumber
          old_person_reference    = old_ref
          new_person_reference    = new_ref
        IMPORTING
          returncode              = returncode
        TABLES
          error_table             = error_table
        EXCEPTIONS
          person_not_exist        = 1
          internal_error          = 2
          parameter_error         = 3
          old_reference_not_exist = 4
          new_reference_exists    = 5
          numberrange_error       = 6
          OTHERS                  = 7.
      IF sy-subrc <> 0 OR returncode = 'E'.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.
      CALL FUNCTION 'ADDR_MEMORY_SAVE'
        EXPORTING
          execute_in_update_task = 'X'
        EXCEPTIONS
          address_number_missing = 1
          person_number_missing  = 2
          internal_error         = 3
          database_error         = 4
          reference_missing      = 5
          OTHERS                 = 6.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

    ELSE.
      WRITE: / 'Contact', iknvk-parnr, 'not switchable, please check.'.
    ENDIF.
  ENDLOOP.
ENDFORM. " SWITCH_CONTACTS

ENDFORM. " ADDRESS_SAVE

FORM switch_contacts_2 USING p_lfa1 LIKE lfa1.
  DATA: iknvk   LIKE knvk OCCURS 0 WITH HEADER LINE,
        old_ref LIKE pers_ref,
        new_ref LIKE pers_ref.

  SELECT * FROM knvk INTO TABLE iknvk
                     WHERE lifnr = p_lfa1-lifnr.

  CHECK sy-subrc = 0.
  LOOP AT iknvk.
    SELECT SINGLE * FROM adcp
           WHERE persnumber = iknvk-prsnr
           AND   comp_pers  = 'C'.
    IF sy-subrc = 0.
      CALL FUNCTION 'ADDR_PERS_COMP_RELATION_SWITCH'
        EXPORTING
          address_number        = adcp-addrnumber
          person_number         = adcp-persnumber
          new_address_number    = p_lfa1-adrnr
        IMPORTING
          returncode            = returncode
        TABLES
          error_table           = error_table
        EXCEPTIONS
          address_not_exist     = 1
          new_address_not_exist = 2
          parameter_error       = 3
          person_not_exist      = 4

          new_relation_exists   = 6
          internal_error        = 7
          OTHERS                = 8.
      IF sy-subrc <> 0 OR returncode = 'E'.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

      old_ref-appl_table     = 'KNVK'.
      old_ref-appl_field     = 'PRSNR'.
      CONCATENATE sy-mandt iknvk-parnr INTO old_ref-appl_key.
      old_ref-pers_group     = c_address_group.
      old_ref-addrnumber     = adcp-addrnumber.
      CLEAR old_ref-pers_addr.
      old_ref-owner          = 'X'.

      new_ref = old_ref.
      new_ref-addrnumber     = p_lfa1-adrnr.

      CALL FUNCTION 'ADDR_PERSON_REFERENCE_SWITCH'
        EXPORTING
          person_number           = adcp-persnumber
          old_person_reference    = old_ref
          new_person_reference    = new_ref
        IMPORTING
          returncode              = returncode
        TABLES
          error_table             = error_table
        EXCEPTIONS
          person_not_exist        = 1
          internal_error          = 2
          parameter_error         = 3
          old_reference_not_exist = 4
          new_reference_exists    = 5
          numberrange_error       = 6
          OTHERS                  = 7.

        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.
      CALL FUNCTION 'ADDR_MEMORY_SAVE'
        EXPORTING
          execute_in_update_task = 'X'
        EXCEPTIONS
          address_number_missing = 1
          person_number_missing  = 2
          internal_error         = 3
          database_error         = 4
          reference_missing      = 5
          OTHERS                 = 6.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

    ELSE.
      WRITE: / 'Contact', iknvk-parnr, 'not switchable, please check.'.
    ENDIF.
  ENDLOOP.
ENDFORM. " SWITCH_CONTACTS
