  PERFORM call_screen_0500.

*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM get_alv_variant .

  CLEAR: gs_variant.
  gs_variant-report = sy-repid.

  CALL FUNCTION 'REUSE_ALV_VARIANT_DEFAULT_GET'
    EXPORTING
      i_save     = 'A'
    CHANGING
      cs_variant = gs_variant
    EXCEPTIONS
      not_found  = 2.

  IF sy-subrc EQ 0.
    pa_vari = gs_variant-variant.
  ENDIF.

ENDFORM. " GET_ALV_VARIANT

                                  'ZDSDGUBUN'
                         CHANGING ls_data_0500-gubuntx.

*   부동산구분명

      WHEN 'B'.

      WHEN 'C'.

      WHEN 'D'.
        ls_data_0500-gsamt = ls_data_0500-dlamt.
    ENDCASE.

*   설정금액-만기일이 현재일 보다 작으면 0.
    IF ls_data_0500-exprdt LT sy-datum.
      ls_data_0500-gsamt = 0.
    ENDIF.

*   여신한도, 총채권
    PERFORM get_knkk USING ls_data_0500-kunnr
                  CHANGING ls_data_0500-klimk
                           ls_data_0500-skfor.
