FORM make_fieldcat USING p_gub p_fname p_con.

  IF p_gub = 'S'.
    CLEAR gs_fcat.
  ENDIF.

  DATA l_col(40).
  FIELD-SYMBOLS <fs>.
  CONCATENATE 'GS_FCAT-' p_fname  INTO l_col.
  ASSIGN      (l_col)         TO       <fs>.
  MOVE         p_con          TO       <fs>.

  IF p_gub = 'E'.
    APPEND gs_fcat TO <fcat>.
  ENDIF.

ENDFORM. " MAKE_FIELDCAT

*&      Form  GET_WEEK_INFO
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM get_week_info USING p_spdat
                 CHANGING ps_date TYPE typ_date.

  DATA: lv_firstday TYPE sy-datum,
        lv_lastday  TYPE sy-datum.

  CLEAR: ps_date.

* 입력받은 기준일자의 월요일, 일요일
  CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
    EXPORTING
      date   = p_spdat
    IMPORTING
      monday = ps_date-frdat
      sunday = ps_date-todat.

* 기준일의 월의 시작일
  CONCATENATE p_spdat(6) '01'
         INTO lv_firstday.

* 기준일의 월의 마지막일
  CALL FUNCTION 'DATE_GET_MONTH_LASTDAY'
    EXPORTING
      i_date = p_spdat
    IMPORTING
      e_date = lv_lastday.

* 기준일의 월요일이 이전달로 넘어갈 경우 월시작일로 지정.
  IF ps_date-frdat LT lv_firstday.
    ps_date-frdat = lv_firstday.
  ENDIF.

* 기준일의 일요일이 지난달로 넘어갈 경우 월마지막로 지정.
  IF ps_date-todat GT lv_lastday.
    ps_date-todat = lv_lastday.
  ENDIF.

* 시작일의 년주차
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date = ps_date-frdat
    IMPORTING
      week = ps_date-spywk.

* 시작일의 년월
  ps_date-spmon = ps_date-frdat(6).

ENDFORM. " GET_WEEK_INFO

        lt_sel_list_s TYPE TABLE OF edpline,

*   일자
      APPEND 'A~VDATU' TO lt_sel_list_s.
      APPEND 'A~FRDAT AS VDATU' TO lt_sel_list_p.
      APPEND 'A~SPYWK' TO lt_sel_list.

  ENDCASE.

  CASE 'X'.
    WHEN pa_r4.

      APPEND 'A~VKBUR' TO lt_sel_list_s.

      APPEND 'C~VKBUR' TO lt_sel_list_p.
      APPEND 'C~BZIRK' TO lt_sel_list_p.
  ENDCASE.

* 자재레벨
  CASE 'X'.
    WHEN pa_r7.
*   자재
      APPEND 'A~MATNR' TO lt_sel_list.
      APPEND 'A~MAKTX' TO lt_sel_list.
      APPEND 'B~MATKL' TO lt_sel_list.
      APPEND 'B~PRDHA' TO lt_sel_list.
      APPEND 'B~ZMAKTX' TO lt_sel_list.
      APPEND 'B~GROES' TO lt_sel_list.

    WHEN pa_r9.
*   제품계층구조
      APPEND 'B~PRDHA' TO lt_sel_list.
  ENDCASE.

  APPEND 'A~VTWEG'  TO lt_sel_list.
  APPEND 'A~WERKS'  TO lt_sel_list.
  APPEND 'A~KWMENG' TO lt_sel_list.
  APPEND 'A~SPMNG'  TO lt_sel_list.
  APPEND 'A~MEINS'  TO lt_sel_list.

*[S] 주간판매계획 팔레트중량 및 세부사항 필드 by HSC01
  APPEND 'A~MVGR5'   TO lt_sel_list.
  APPEND 'A~BSTKD_E' TO lt_sel_list.
*[E] 주간판매계획 팔레트중량 및 세부사항 필드 by HSC01

  APPEND LINES OF lt_sel_list TO lt_sel_list_s.
* 판매계획 확정 조회
  SELECT (lt_sel_list_s)
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM zsdt0301 AS a
   INNER JOIN
         mara     AS b
      ON a~matnr  EQ b~matnr
   WHERE a~spdat  EQ gv_spdat
     AND a~sptyp  EQ 'P'
     AND a~vdatu  IN so_vdatu
     AND a~spywk  IN so_spywk
     AND a~spmon  IN so_spmon
     AND a~vtweg  IN gr_vtweg
     AND a~werks  IN ('1010',
                      '1020',
                      '1030')
     AND a~vkbur  IN so_vkbur
     AND a~bzirk  IN so_bzirk
     AND a~person IN so_persn
     AND a~kunnr  IN so_kunnr
     AND a~werks  IN so_werks.

  LOOP AT lt_data_0100 INTO ls_data_0100.
    COLLECT ls_data_0100 INTO pt_data_0100.

  ENDIF.
  IF pa_r4 = 'X'."거래처

  ENDIF.

  LOOP AT pt_data_0100 INTO ls_data_0100.

    PERFORM get_kunnr_text USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-kuntx.
*   사무소명
    PERFORM get_vkbur_text USING ls_data_0100-vkbur
                        CHANGING ls_data_0100-vkbtx.
*   총판매계획
    ls_data_0100-tlmng = ls_data_0100-spmng + ls_data_0100-kwmeng.

*   판매지역명
    PERFORM get_bzirk_text USING ls_data_0100-bzirk
                        CHANGING ls_data_0100-bztxt.

    MODIFY pt_data_0100 FROM ls_data_0100.
  ENDLOOP.

  SORT pt_data_0100 BY spmon spywk vdatu vkbur bzirk kunnr.
