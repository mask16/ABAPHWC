FORM DATA_SELECT .
  DATA: LT_CELLTAB TYPE LVC_T_STYL,
        L_INDEX TYPE I.

  CLEAR: GT_DATA.

  SELECT A~WERKS A~MATNR A~FDATE
    A~NETPR A~WAERS A~MENGE
    A~ERDAT A~ERZET A~ERNAM
    A~AEDAT A~AEZET A~AENAM
    B~MEINS
    C~MAKTX
  INTO CORRESPONDING FIELDS OF TABLE GT_DATA
  FROM ZSDT0570 AS A
  JOIN MARA AS B ON A~MATNR = B~MATNR
  LEFT JOIN MAKT AS C ON A~MATNR = C~MATNR
                      AND C~SPRAS = SY-LANGU
  WHERE A~WERKS IN S_WERKS
    AND A~MATNR IN S_MATNR.


  LOOP AT GT_DATA INTO GS_DATA.
    "Cell Control
    PERFORM FILL_CELLTAB USING 'RO' 'WERKS'
                       CHANGING LT_CELLTAB.
    PERFORM FILL_CELLTAB USING 'RO' 'MATNR'
                       CHANGING LT_CELLTAB.
    PERFORM FILL_CELLTAB USING 'RO' 'FDATE'
                       CHANGING LT_CELLTAB.
    INSERT LINES OF LT_CELLTAB INTO TABLE GS_DATA-CELLTAB.

    MODIFY GT_DATA FROM GS_DATA.
  ENDLOOP.

  SORT GT_DATA BY WERKS MATNR.

ENDFORM. " DATA_SELECT

FORM SAVE_PROCESS .
  DATA: LS_ZSDT0570   LIKE ZSDT0570,
        LT_ZSDT0570   LIKE ZSDT0570 OCCURS 0 WITH HEADER LINE.
  DATA: LV_TABIX     LIKE SY-TABIX,
        LV_LINES     TYPE N,
        LV_LINES1    TYPE N.

  DATA: LT_CELLTAB TYPE LVC_T_STYL,
        L_INDEX TYPE I.

  CLEAR: GS_DATA.

  GET TIME.
  LOOP AT GT_MANDELT INTO GS_DATA.
    DELETE FROM  ZSDT0570 WHERE WERKS =  GS_DATA-WERKS
                           AND MATNR =  GS_DATA-MATNR
                           AND FDATE =  GS_DATA-FDATE.
  ENDLOOP.

  DESCRIBE TABLE GT_DATA LINES LV_LINES.

*   사용자가 수정 또는 추가한 데이터 대입
  LOOP AT GT_DATA  INTO GS_DATA.
    LV_TABIX  =  SY-TABIX.

    IF  GS_DATA-WERKS IS INITIAL.
      MESSAGE E000 WITH '플랜트를 입력하세요.'.
    ELSEIF  GS_DATA-MATNR IS INITIAL.
      MESSAGE E000 WITH '자재코드를 입력하세요.'.
    ELSEIF  GS_DATA-FDATE IS INITIAL.
      MESSAGE E000 WITH '기준일자를 입력하세요'.
    ENDIF.

    CHECK NOT ( GS_DATA-WERKS IS INITIAL AND
                GS_DATA-MATNR IS INITIAL  AND
                GS_DATA-FDATE IS INITIAL ).

*--- 통화 공백 제거 2024.07.23 ---

        CONDENSE GS_DATA-WAERS NO-GAPS.

*---------------



    CLEAR LS_ZSDT0570.
    SELECT SINGLE *
      INTO LS_ZSDT0570
      FROM ZSDT0570
      WHERE WERKS = GS_DATA-WERKS
        AND MATNR = GS_DATA-MATNR
        AND FDATE = GS_DATA-FDATE.

    IF SY-SUBRC = 0.

      IF  LS_ZSDT0570-NETPR NE GS_DATA-NETPR OR
        LS_ZSDT0570-WAERS NE GS_DATA-WAERS OR
        LS_ZSDT0570-MENGE NE GS_DATA-MENGE.
        GET TIME.
        GS_DATA-AEDAT = SY-DATUM.
        GS_DATA-AEZET = SY-UZEIT.
        GS_DATA-AENAM  =  SY-UNAME.


        UPDATE ZSDT0570  SET NETPR = GS_DATA-NETPR
                               WAERS = GS_DATA-WAERS
                               MENGE = GS_DATA-MENGE
                               AEDAT = GS_DATA-AEDAT
                               AEZET = GS_DATA-AEZET
                               AENAM = GS_DATA-AENAM
               WHERE WERKS = GS_DATA-WERKS
                 AND MATNR = GS_DATA-MATNR
                 AND FDATE = GS_DATA-FDATE.
      ENDIF.

    ELSE.
      GS_DATA-ERDAT = SY-DATUM.
      GS_DATA-ERZET = SY-UZEIT.
      GS_DATA-ERNAM  =  SY-UNAME.
      MOVE-CORRESPONDING GS_DATA  TO LS_ZSDT0570.

      INSERT ZSDT0570  FROM LS_ZSDT0570.

    ENDIF.

    IF SY-SUBRC = 0.
      COMMIT WORK.
    ELSE.
      ROLLBACK WORK.
      MESSAGE E001  WITH '저장시 오류 발생'.
    ENDIF.


    CLEAR: GS_DATA-CELLTAB, LT_CELLTAB.
    PERFORM FILL_CELLTAB USING 'RO' 'WERKS'
                        CHANGING LT_CELLTAB.
    PERFORM FILL_CELLTAB USING 'RO' 'MATNR'
                       CHANGING LT_CELLTAB.
    PERFORM FILL_CELLTAB USING 'RO' 'FDATE'
                       CHANGING LT_CELLTAB.
    INSERT LINES OF LT_CELLTAB INTO TABLE GS_DATA-CELLTAB.


    MODIFY GT_DATA  FROM GS_DATA  INDEX LV_TABIX.
  ENDLOOP.


*   메세지 출력
  MESSAGE S001 WITH 'Data saved.'.
  REFRESH GT_MANDELT.


ENDFORM. " SAVE_PROCESS

    B~KPEIN B~POSNR B~MATNR
