*&---------------------------------------------------------------------*
*&      Form  SELECT_CHECK_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM select_check_data .

  DATA: ls_date     LIKE LINE OF gt_date,
        ls_date2    LIKE LINE OF gt_date,
        lv_spdat    TYPE sy-datum.

  CLEAR: gt_date.

  lv_spdat = pa_spdat.
* 12 주의 날짜 정보
  DO 12 TIMES.
    PERFORM get_week_info USING lv_spdat
                       CHANGING ls_date.
*   순번
    ls_date-seqnr = sy-index.
*   년주차 순번
    IF ls_date2-spywk NE ls_date-spywk.
      ls_date2-spywk = ls_date-spywk.
      ADD 1 TO ls_date2-seqwk.
    ENDIF.
    ls_date-seqwk = ls_date2-seqwk.
    APPEND ls_date TO gt_date.
    lv_spdat = ls_date-todat + 1.
  ENDDO.

* 계획기준일자
  READ TABLE gt_date INDEX 1 INTO ls_date.
  gv_spdat = ls_date-frdat.

*&      Form  SELECT_DATA_0100

FORM select_data_0100 .

  DATA: lt_data_0100   LIKE gt_data_0100,
        lr_belnr       TYPE RANGE OF zsdt0130-belnr,
        ls_belnr       LIKE LINE OF lr_belnr,
        ls_data_0100   LIKE LINE OF gt_data_0100,
        lv_datum       TYPE datum,
        lv_rowcount    TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  IF gv_conflg EQ 'X'.
*   확정
    SELECT spdat
           spmon
           vkorg
           vtweg
           vkbur
           person
           kunnr
           werks
           matnr
           SUM( kwmeng ) AS kwmeng
           SUM( spmng )  AS spmng
      INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
      FROM zsdt0301
     WHERE spdat  EQ gv_spdat
       AND spmon  GE pa_spmon
     GROUP BY spdat spmon vkorg vtweg vkbur person kunnr werks matnr.
  ELSE.
*   미확정
    SELECT a~spdat
           a~spmon
           a~vkorg
           a~vtweg
           b~vkbur
           a~person
           a~kunnr
           a~werks
           a~matnr
           SUM( a~kwmeng ) AS kwmeng
           SUM( a~spmng )  AS spmng
      INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
      FROM zsdt0302 AS a
     INNER JOIN
           knvv     AS b
        ON a~kunnr  EQ b~kunnr
       AND b~vkorg  EQ '1000'
       AND b~vtweg  EQ '10'
       AND b~spart  EQ '10'
     WHERE a~spdat  EQ gv_spdat
       AND a~spmon  GE pa_spmon
     GROUP BY a~spdat a~spmon a~vkorg a~vtweg b~vkbur a~person a~kunnr a~werks a~matnr.
  ENDIF.

  CONCATENATE pa_spmon '01'
         INTO lv_datum.
  SORT lt_data_0100 BY vkorg vtweg vkbur person kunnr werks matnr spmon.
  LOOP AT lt_data_0100 INTO ls_data_0100.
*   실행
    IF ls_data_0100-spmon EQ pa_spmon.
      ls_data_0100-extyp = 'E'.
    ELSE.
      ls_data_0100-extyp = 'P'.
    ENDIF.
*   총판매계획
    ls_data_0100-totmng = ls_data_0100-kwmeng + ls_data_0100-spmng.
*   화폐(KRW)
    ls_data_0100-krw = 'KRW'.
*   버전
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = '10'
      IMPORTING
        output = ls_data_0100-versi.
*   연도
    ls_data_0100-gjahr = ls_data_0100-spmon(4).
*   관리회계영역
    ls_data_0100-kokrs = ls_data_0100-vkorg.
*   회사코드
    ls_data_0100-bukrs = ls_data_0100-vkorg.
*   판매처명
    PERFORM get_kunnr_text USING ls_data_0100-kunnr
                        CHANGING ls_data_0100-kuntx.
*   사업군
    PERFORM get_kunnr_brsch USING ls_data_0100-kunnr
                         CHANGING ls_data_0100-brsch.
*   담당자명
    PERFORM get_kunnr_text USING ls_data_0100-person
                        CHANGING ls_data_0100-pertx.
*   영업그룹, 고객 그룹, 판매구역, 무역조건1
    PERFORM get_knvv_info USING ls_data_0100-kunnr
                       CHANGING ls_data_0100-vkgrp
                                ls_data_0100-kdgrp
                                ls_data_0100-bzirk
                                ls_data_0100-inco1.
*   모델명, 규격명, 제품군, 자재그룹
    PERFORM get_mara_info USING ls_data_0100-matnr
                       CHANGING ls_data_0100-zmaktx
                                ls_data_0100-groes
                                ls_data_0100-spart
                                ls_data_0100-matkl
                                ls_data_0100-meins.
*   자재그룹명
    PERFORM get_matkl_text USING ls_data_0100-matkl
                        CHANGING ls_data_0100-matxt.
*   제품계층구조, 자재그룹1, 자재그룹2
    PERFORM get_mvke_info USING ls_data_0100-matnr
                       CHANGING ls_data_0100-prodh
                                ls_data_0100-mvgr1
                                ls_data_0100-mvgr2.
*   손익센터
    PERFORM get_marc_prctr USING ls_data_0100-matnr
                                 ls_data_0100-werks
                        CHANGING ls_data_0100-prctr.
*   평가클래스
    PERFORM get_mbew_bklas USING ls_data_0100-matnr
                                 ls_data_0100-werks
                                 space
                        CHANGING ls_data_0100-bklas.
*   사무소명
    PERFORM get_vkbur_text USING ls_data_0100-vkbur
                        CHANGING ls_data_0100-vkbtx.
*   판매구역명
    PERFORM get_bzirk_text USING ls_data_0100-bzirk
                        CHANGING ls_data_0100-bztxt.
*   품명
    PERFORM get_matnr_text USING ls_data_0100-matnr
                        CHANGING ls_data_0100-maktx.
*   단가, 화폐
    call function 'ZSD_GET_KONP_KBETR'
      EXPORTING
        i_kunnr = ls_data_0100-kunnr
        i_matnr = ls_data_0100-matnr
        i_datum = lv_datum
        i_vtweg = ls_data_0100-vtweg
        i_prodh = ls_data_0100-prodh
        i_mvgr1 = ls_data_0100-mvgr1
        i_vkorg = '1000'
      IMPORTING
        e_kbetr = ls_data_0100-kbetr
        e_konwa = ls_data_0100-konwa.
*   화폐(KRW)
    ls_data_0100-krw = 'KRW'.
*   계획금액 = 총수량 * 단가
    ls_data_0100-spamt = ls_data_0100-totmng * ls_data_0100-kbetr.

    PERFORM get_tcurr_ukurs USING ls_data_0100-konwa

        foreign_currency = ls_data_0100-konwa

      IMPORTING

  ENDLOOP.

  lv_rowcount = lines( gt_data_0100 ).

*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM get_week_info USING p_spdat
                 CHANGING ps_date TYPE typ_date.

  DATA: lv_firstday TYPE sy-datum,
        lv_lastday  TYPE sy-datum.

  CLEAR: ps_date.

* 입력받은 기준일자의 월요일, 일요일
  CALL FUNCTION 'GET_WEEK_INFO_BASED_ON_DATE'
    EXPORTING
      date   = p_spdat
    IMPORTING
      monday = ps_date-frdat
      sunday = ps_date-todat.

* 기준일의 월의 시작일
  CONCATENATE p_spdat(6) '01'
         INTO lv_firstday.

* 기준일의 월의 마지막일
  CALL FUNCTION 'DATE_GET_MONTH_LASTDAY'
    EXPORTING
      i_date = p_spdat
    IMPORTING
      e_date = lv_lastday.

* 기준일의 월요일이 이전달로 넘어갈 경우 월시작일로 지정.
  IF ps_date-frdat LT lv_firstday.
    ps_date-frdat = lv_firstday.
  ENDIF.

* 기준일의 일요일이 지난달로 넘어갈 경우 월마지막로 지정.
  IF ps_date-todat GT lv_lastday.
    ps_date-todat = lv_lastday.
  ENDIF.

* 시작일의 년주차
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date = ps_date-frdat
    IMPORTING
      week = ps_date-spywk.

* 시작일의 년월
  ps_date-spmon = ps_date-frdat(6).

ENDFORM. " GET_WEEK_INFO

*&---------------------------------------------------------------------*
*&      Form  SELECT_TCURR
*&---------------------------------------------------------------------*
*       환율조회
*----------------------------------------------------------------------*
FORM select_tcurr .

  DATA: lt_tcurr LIKE gt_tcurr,
        lv_input TYPE datum,
        lv_datum TYPE tcurr-gdatu,
        lv_gdatu TYPE tcurr-gdatu.

  CLEAR: gt_tcurr.

  CONCATENATE pa_spmon '01'
         INTO lv_input.

  CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'
    EXPORTING
      input  = lv_input
    IMPORTING
      output = lv_datum.

  SELECT MIN( gdatu ) AS gdatu
         fcurr
    INTO CORRESPONDING FIELDS OF TABLE lt_tcurr
    FROM tcurr
   WHERE kurst EQ 'M'
     AND tcurr EQ 'KRW'
     AND gdatu GE lv_datum
   GROUP BY fcurr.

  SELECT *
    INTO TABLE gt_tcurr
    FROM tcurr
     FOR ALL ENTRIES IN lt_tcurr
