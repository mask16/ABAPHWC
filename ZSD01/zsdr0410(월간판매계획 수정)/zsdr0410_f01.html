FORM MAKE_FIELDCAT USING P_GUB P_FNAME P_CON.

  IF P_GUB = 'S'.
    CLEAR GS_FCAT.
  ENDIF.

  DATA L_COL(40).
  FIELD-SYMBOLS <FS>.
  CONCATENATE 'GS_FCAT-' P_FNAME  INTO L_COL.
  ASSIGN      (L_COL)         TO       <FS>.
  MOVE         P_CON          TO       <FS>.

  IF P_GUB = 'E'.
    APPEND GS_FCAT TO <FCAT>.
  ENDIF.

ENDFORM. " MAKE_FIELDCAT

*&      Form  DATA_CHANGED
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
FORM DATA_CHANGED USING
                P_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL
                P_ONF4         TYPE CHAR01.

  DATA: LS_S_MODI      TYPE LVC_S_MODI,
        LS_DATA_0100   LIKE LINE OF GT_DATA_0100,
        LV_DATUM       TYPE DATUM.

  CONCATENATE PA_SPMON '01'
         INTO LV_DATUM.

  LOOP AT P_DATA_CHANGED->MT_MOD_CELLS INTO LS_S_MODI.
    IF LS_S_MODI-ERROR EQ 'X'.
      CONTINUE.
    ENDIF.

    CASE LS_S_MODI-FIELDNAME.
      WHEN 'WERKS'.
        CALL METHOD P_DATA_CHANGED->GET_CELL_VALUE
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'WERKS'
          IMPORTING
            E_VALUE     = LS_DATA_0100-WERKS.

        CALL METHOD P_DATA_CHANGED->MODIFY_CELL
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'WERKS'
            I_VALUE     = LS_DATA_0100-WERKS.

      WHEN 'SPMNG'.
*     계획수량 변경시
        REPLACE ALL OCCURENCES OF ',' IN LS_S_MODI-VALUE WITH SPACE.
*       수주수량
        CALL METHOD P_DATA_CHANGED->GET_CELL_VALUE
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'KWMENG'
          IMPORTING
            E_VALUE     = LS_DATA_0100-KWMENG.

*       총판매계획 변경
        LS_DATA_0100-TOTMNG = LS_S_MODI-VALUE + LS_DATA_0100-KWMENG.
        CALL METHOD P_DATA_CHANGED->MODIFY_CELL
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'TOTMNG'
            I_VALUE     = LS_DATA_0100-TOTMNG.

*       단가
        CALL METHOD P_DATA_CHANGED->GET_CELL_VALUE
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'KBETR'
          IMPORTING
            E_VALUE     = LS_DATA_0100-KBETR.

*       계획금액 = 총수량 * 단가
        LS_DATA_0100-SPAMT = LS_DATA_0100-TOTMNG * LS_DATA_0100-KBETR.
        CALL METHOD P_DATA_CHANGED->MODIFY_CELL
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'SPAMT'
            I_VALUE     = LS_DATA_0100-SPAMT.

*       화폐
        CALL METHOD P_DATA_CHANGED->GET_CELL_VALUE
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'KONWA'
          IMPORTING
            E_VALUE     = LS_DATA_0100-KONWA.

*       계획환율
        CALL METHOD P_DATA_CHANGED->GET_CELL_VALUE
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'UKURS'
          IMPORTING
            E_VALUE     = LS_DATA_0100-UKURS.

*       계획금액(KRW)
        CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
          EXPORTING
            DATE             = LV_DATUM
            FOREIGN_AMOUNT   = LS_DATA_0100-SPAMT
            FOREIGN_CURRENCY = LS_DATA_0100-KONWA
            LOCAL_CURRENCY   = 'KRW'
            RATE             = LS_DATA_0100-UKURS
          IMPORTING
            LOCAL_AMOUNT     = LS_DATA_0100-KRAMT.

        CALL METHOD P_DATA_CHANGED->MODIFY_CELL
          EXPORTING
            I_ROW_ID    = LS_S_MODI-ROW_ID
            I_FIELDNAME = 'KRAMT'
            I_VALUE     = LS_DATA_0100-KRAMT.
    ENDCASE.
  ENDLOOP.

ENDFORM. " DATA_CHANGED
