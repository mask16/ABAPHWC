*&---------------------------------------------------------------------*
FORM set_zexport .

  CLEAR: gr_vtweg,     gr_vtweg[],
         gv_zexport.

  CASE 'X'.
    WHEN pa_in.
      MOVE: '1'         TO gv_zexport.

      MOVE: 'I'         TO gr_vtweg-sign,
            'EQ'        TO gr_vtweg-option,
            '10'        TO gr_vtweg-low.
      APPEND gr_vtweg.

      MOVE: '30'        TO gr_vtweg-low.
      APPEND gr_vtweg.

    WHEN pa_out.
      MOVE: '2'         TO gv_zexport.

      MOVE: 'I'         TO gr_vtweg-sign,
            'EQ'        TO gr_vtweg-option,
            '20'        TO gr_vtweg-low.
      APPEND gr_vtweg.
  ENDCASE.

ENDFORM. " SET_ZEXPORT
