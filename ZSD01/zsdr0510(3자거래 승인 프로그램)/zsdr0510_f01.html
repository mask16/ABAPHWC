FORM variant_init USING p_layout.

  DATA: gs_vari TYPE disvariant.

  CLEAR gs_vari.
  gs_vari-report   = sy-repid.
  gs_vari-username = sy-uname.

  IF p_layout <> ''.
    gs_vari-variant = p_layout.
  ENDIF.

  CALL FUNCTION 'LVC_VARIANT_DEFAULT_GET'
    EXPORTING
      i_save        = 'A'
    CHANGING
      cs_variant    = gs_vari
    EXCEPTIONS
      wrong_input   = 1
      not_found     = 2
      program_error = 3
      OTHERS        = 4.

  CASE sy-subrc.
    WHEN 0.
      p_layout = gs_vari-variant.
    WHEN 2.
      CLEAR p_layout.
  ENDCASE.

ENDFORM. " VARIANT_INIT
