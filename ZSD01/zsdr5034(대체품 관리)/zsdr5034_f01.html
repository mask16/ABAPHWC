FORM SELECT_DATA_0100.
  CLEAR GT_DATA_0100.

  SELECT * INTO TABLE GT_DATA_0100
    FROM ZSDT5050
   WHERE VTWEG  IN SO_VTWEG
     AND WADAT  IN SO_WADAT
     AND VKBUR  IN SO_VKBUR
     AND MATNR  IN SO_MATNR
     AND WERKS  IN SO_WERKS
     AND LGORT IN SO_LGORT.

  " 명칭(납품처명, 자재명)
  LOOP AT GT_DATA_0100 INTO GS_DATA_0100.

    " 영업소명
    CLEAR GS_DATA_0100-NAME1.
    SELECT SINGLE BEZEI INTO GS_DATA_0100-NAME1
      FROM TVKBT
     WHERE SPRAS = SY-LANGU               " 로그인한 언어 (한국어: '3' 또는 'KO')
       AND VKBUR = GS_DATA_0100-VKBUR.

    " 자재명 (자재 내역 테이블 MAKT)
    SELECT SINGLE MAKTX INTO GS_DATA_0100-MAKTX
      FROM MAKT
     WHERE MATNR = GS_DATA_0100-MATNR
       AND SPRAS = SY-LANGU. " 로그인한 언어 기준

    " 변경된 명칭을 인터널 테이블에 반영
    MODIFY GT_DATA_0100 FROM GS_DATA_0100.
  ENDLOOP.

ENDFORM. "SELECT_DATA_0100

FORM SET_FCAT_GRID.
  DATA: LS_FCAT     TYPE LVC_S_FCAT,
        LS_FCAT_NEW TYPE LVC_S_FCAT,
        LV_TABIX    TYPE SY-TABIX.

  " 헤더 기본 카탈로그 생성

    EXPORTING
      I_STRUCTURE_NAME = 'ZSDT5050'
    CHANGING

  " 기존 컬럼명(텍스트) 강제 변경
  LOOP AT GT_FCAT_TOP INTO LS_FCAT.
    CASE LS_FCAT-FIELDNAME.
      WHEN 'LGORT'.
        LS_FCAT-COLTEXT = '저장위치'. LS_FCAT-SCRTEXT_L = '저장위치'.
        LS_FCAT-SCRTEXT_M = '저장위치'. LS_FCAT-SCRTEXT_S = '저장위치'.

        LS_FCAT-SCRTEXT_M = '사유'. LS_FCAT-SCRTEXT_S = '사유'.

        LS_FCAT-SCRTEXT_M = '비고'. LS_FCAT-SCRTEXT_S = '비고'.
    ENDCASE.
    MODIFY GT_FCAT_TOP FROM LS_FCAT.
  ENDLOOP.

  " 영업소명(NAME1)을 납품처(VKBUR) 바로 뒤에 끼워넣기
  READ TABLE GT_FCAT_TOP INTO LS_FCAT WITH KEY FIELDNAME = 'VKBUR'.

    LS_FCAT_NEW-FIELDNAME = 'NAME1'.

    LS_FCAT_NEW-OUTPUTLEN = 15.
    INSERT LS_FCAT_NEW INTO GT_FCAT_TOP INDEX LV_TABIX.
  ENDIF.

  "  자재명(MAKTX)을 자재(MATNR) 바로 뒤에 끼워넣기
  READ TABLE GT_FCAT_TOP INTO LS_FCAT WITH KEY FIELDNAME = 'MATNR'.
  IF SY-SUBRC = 0.
    LV_TABIX = SY-TABIX + 1. " 자재 다음 인덱스

    LS_FCAT_NEW-OUTPUTLEN = 25.

  ENDIF.

  ENDLOOP.

  " 하단 디테일 카탈로그 생성
  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'

ENDFORM. "SET_FCAT_GRID
