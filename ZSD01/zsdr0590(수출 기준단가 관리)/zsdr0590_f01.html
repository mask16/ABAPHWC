FORM save_process .
  DATA: ls_zsdt0590   LIKE zsdt0590,
        lt_zsdt0590   LIKE zsdt0590 OCCURS 0 WITH HEADER LINE.
  DATA: lv_tabix     LIKE sy-tabix,
        lv_lines     TYPE n,
        lv_lines1    TYPE n.

  DATA: lt_celltab TYPE lvc_t_styl,
        l_index TYPE i.

  CLEAR: gs_data.

  GET TIME.
  LOOP AT gt_mandelt INTO gs_data.
    DELETE FROM  zsdt0590 WHERE werks =  gs_data-werks
                           AND kunnr =  gs_data-kunnr
                           AND matnr =  gs_data-matnr
                           AND fdate =  gs_data-fdate.
  ENDLOOP.

  DESCRIBE TABLE gt_data LINES lv_lines.

*   사용자가 수정 또는 추가한 데이터 대입
  LOOP AT gt_data  INTO gs_data.
    lv_tabix  =  sy-tabix.

    IF  gs_data-werks IS INITIAL.
      MESSAGE e000 WITH '플랜트를 입력하세요.'.
    ELSEIF  gs_data-kunnr IS INITIAL.
      MESSAGE e000 WITH '거래처를 입력하세요.'.
    ELSEIF  gs_data-matnr IS INITIAL.
      MESSAGE e000 WITH '자재코드를 입력하세요.'.
    ELSEIF  gs_data-inco1 IS INITIAL.
      MESSAGE e000 WITH '거래조건을 입력하세요'.
    ELSEIF  gs_data-fdate IS INITIAL.
      MESSAGE e000 WITH '기준일자를 입력하세요'.
    ENDIF.

    CHECK NOT ( gs_data-werks IS INITIAL AND
                gs_data-kunnr IS INITIAL  AND
                gs_data-matnr IS INITIAL  AND
                gs_data-fdate IS INITIAL ).

    CLEAR ls_zsdt0590.
    SELECT SINGLE *
      INTO ls_zsdt0590
      FROM zsdt0590
      WHERE werks = gs_data-werks
        AND kunnr = gs_data-kunnr
        AND matnr = gs_data-matnr
        AND fdate = gs_data-fdate.

    IF sy-subrc = 0.

      IF  ls_zsdt0590-inco1 NE gs_data-inco1 OR
        ls_zsdt0590-netpr NE gs_data-netpr OR
        ls_zsdt0590-waers NE gs_data-waers OR
        ls_zsdt0590-menge NE gs_data-menge.
        GET TIME.
        gs_data-aedat = sy-datum.
        gs_data-aezet = sy-uzeit.
        gs_data-aenam  =  sy-uname.

        UPDATE zsdt0590  SET inco1 = gs_data-inco1
                               netpr = gs_data-netpr
                               waers = gs_data-waers
                               menge = gs_data-menge
                               app_yn = gs_data-app_yn
                               aedat = gs_data-aedat
                               aezet = gs_data-aezet
                               aenam = gs_data-aenam
               WHERE werks = gs_data-werks
                 AND kunnr = gs_data-kunnr
                 AND matnr = gs_data-matnr
                 AND fdate = gs_data-fdate.
      ENDIF.

    ELSE.
      gs_data-erdat = sy-datum.
      gs_data-erzet = sy-uzeit.
      gs_data-ernam  =  sy-uname.
      MOVE-CORRESPONDING gs_data  TO ls_zsdt0590.

      INSERT zsdt0590  FROM ls_zsdt0590.

    ENDIF.

    IF sy-subrc = 0.
      COMMIT WORK.
    ELSE.
      ROLLBACK WORK.
      MESSAGE e001  WITH '저장시 오류 발생'.
    ENDIF.


    CLEAR: gs_data-celltab, lt_celltab.
    PERFORM fill_celltab USING 'RO' 'WERKS'
                        CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'MATNR'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'KUNNR'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'FDATE'
                       CHANGING lt_celltab.
    INSERT LINES OF lt_celltab INTO TABLE gs_data-celltab.


    MODIFY gt_data  FROM gs_data  INDEX lv_tabix.
  ENDLOOP.


*   메세지 출력
  MESSAGE s001 WITH 'Data saved.'.
  REFRESH gt_mandelt.


ENDFORM. " SAVE_PROCESS
