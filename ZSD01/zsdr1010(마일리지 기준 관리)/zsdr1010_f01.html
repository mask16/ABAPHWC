FORM save_process .
  DATA: ls_zsdt1010   LIKE zsdt1010,
        lt_zsdt1010   LIKE zsdt1010 OCCURS 0 WITH HEADER LINE.
  DATA: lv_tabix     LIKE sy-tabix,
        lv_lines     TYPE n,
        lv_lines1    TYPE n.

  DATA: lt_celltab TYPE lvc_t_styl,
        l_index TYPE i.

  CLEAR: gs_data.

  GET TIME.
  LOOP AT gt_mandelt INTO gs_data.
    DELETE FROM  zsdt1010 WHERE bukrs = gs_data-bukrs
                            AND mtgrp = gs_data-mtgrp
                            AND spmon = gs_data-spmon
                            AND kunnr =  gs_data-kunnr.
  ENDLOOP.

  DESCRIBE TABLE gt_data LINES lv_lines.

*   사용자가 수정 또는 추가한 데이터 대입
  LOOP AT gt_data  INTO gs_data.
    lv_tabix  =  sy-tabix.

    IF  gs_data-bukrs IS INITIAL
      OR gs_data-mtgrp IS INITIAL
      OR gs_data-spmon IS INITIAL
      OR gs_data-kunnr IS INITIAL.
      MESSAGE e001 WITH 'Please Insert all Required Values.'.
    ENDIF.

    CHECK NOT ( gs_data-bukrs IS INITIAL AND
                gs_data-mtgrp IS INITIAL AND
                gs_data-spmon IS INITIAL AND
                gs_data-kunnr IS INITIAL  ).

    CLEAR ls_zsdt1010.
    SELECT SINGLE *
      INTO ls_zsdt1010
      FROM zsdt1010
      WHERE bukrs = gs_data-bukrs
       AND mtgrp = gs_data-mtgrp
       AND spmon = gs_data-spmon
       AND kunnr =  gs_data-kunnr.

    IF sy-subrc = 0.
      IF  ls_zsdt1010-menge NE gs_data-menge.
        gs_data-aedat = sy-datum.
        gs_data-aezet = sy-uzeit.
        gs_data-aenam = sy-uname.

        UPDATE zsdt1010  SET menge = gs_data-menge
                               aedat = gs_data-aedat
                               aezet = gs_data-aezet
                               aenam = gs_data-aenam
               WHERE bukrs = gs_data-bukrs
                 AND mtgrp = gs_data-mtgrp
                 AND spmon = gs_data-spmon
                 AND kunnr =  gs_data-kunnr.
      ENDIF.

    ELSE.
      gs_data-erdat = sy-datum.
      gs_data-erzet = sy-uzeit.
      gs_data-ernam  =  sy-uname.
      MOVE-CORRESPONDING gs_data  TO ls_zsdt1010.

      INSERT zsdt1010  FROM ls_zsdt1010.

    ENDIF.

    IF sy-subrc = 0.
      COMMIT WORK.
    ELSE.
      ROLLBACK WORK.
      MESSAGE e001  WITH '저장시 오류 발생'.
    ENDIF.


    CLEAR: gs_data-celltab, lt_celltab.
    "Cell Control
    PERFORM fill_celltab USING 'RO' 'BUKRS'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'MTGRP'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'SPMON'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'KUNNR'
                       CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RO' 'MENGE'
                       CHANGING lt_celltab.

    INSERT LINES OF lt_celltab INTO TABLE gs_data-celltab.


    MODIFY gt_data  FROM gs_data  INDEX lv_tabix.
  ENDLOOP.


*   메세지 출력
