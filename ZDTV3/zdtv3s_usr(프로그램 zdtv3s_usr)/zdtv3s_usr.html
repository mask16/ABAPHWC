FORM CHANGE_DATA USING P_SUM_GB P_GB
                           P_FIELDNAME
                           PS_DATA
                  CHANGING P_OUT.

  FIELD-SYMBOLS: <STCD2> , <BUKRS> , <ZETNO>.

  CASE P_SUM_GB.
    WHEN '1'.


    WHEN '2'.

        WHEN 'BUPLA'.
          ASSIGN COMPONENT 'BUKRS' OF STRUCTURE PS_DATA TO <BUKRS>.
          CASE P_GB.
            WHEN 'AR'.
              ASSIGN COMPONENT 'SUPNO' OF STRUCTURE PS_DATA TO <STCD2>.
            WHEN 'AP'.
              ASSIGN COMPONENT 'BUYNO' OF STRUCTURE PS_DATA TO <STCD2>.
          ENDCASE.
          IF SY-SUBRC EQ 0.
            SELECT SINGLE BRANCH INTO P_OUT
              FROM J_1BBRANCH
             WHERE BUKRS EQ <BUKRS>
               AND STCD2 EQ <STCD2>.
          ENDIF.
*        WHEN 'INVKEY'.
*          IF P_GB EQ 'AR'.
*            ASSIGN COMPONENT 'BUKRS' OF STRUCTURE PS_DATA TO <BUKRS>.
*            ASSIGN COMPONENT 'ZETNO' OF STRUCTURE PS_DATA TO <ZETNO>.
