FORM FILL_IT_DISP .
  DATA : BEGIN OF LT_TMP OCCURS 0,
           BUKRS       LIKE ZDTV3T_ETC-BUKRS,
           BUPLA       LIKE ZDTV3T_ETC-BUPLA,
           ASP_CODE    LIKE ZDTV3T_ETC-ASP_CODE,
           IP_TYPTXT(20),
           NTS_SEND_YN,
           TYPE_CODE   LIKE ZDTV3T_ETC-TYPE_CODE,
           IP_ID       LIKE ZDTV3T_ETC-IP_ID,
           IP_CNT      TYPE I.
  DATA : END OF LT_TMP.
  DATA : OLD_TMP  LIKE TABLE OF LT_TMP WITH HEADER LINE.
  CLEAR: IT_DISP, IT_DISP[].

  SORT: IT_ETC,
        IT_ASP_CODE,
        IT_DD07V.


  LOOP AT IT_ETC.
    MOVE-CORRESPONDING IT_ETC TO IT_DISP.

*.. ASP 코드 TEXT
    READ TABLE IT_ASP_CODE WITH KEY ASP_CODE = IT_DISP-ASP_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_DISP-ASP_CODE_TEXT = IT_ASP_CODE-ASP_CODE_TEXT.
    ENDIF.

*.. 구분
    CASE IT_ETC-IP_TYPECODE.
      WHEN '01'.
        IT_DISP-IP_TYPTXT = '사업자번호 발행분'.
      WHEN '02' OR '03'.
        IT_DISP-IP_TYPTXT = '주민등록번호 발행분'.
    ENDCASE.

*.. 세금계산서 종류
    READ TABLE IT_DD07V WITH KEY DOMVALUE_L = IT_ETC-TYPE_CODE
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_DISP-DDTEXT = IT_DD07V-DDTEXT.
    ENDIF.

*.. 국세청 신고/미신고
    IF IT_ETC-NTS_SEND_FLAG = '31'.
      IT_DISP-NTS_SEND_YN = 'Y'.
    ELSE.
      IT_DISP-NTS_SEND_YN = 'N'.
    ENDIF.

*~  매출처
    READ TABLE OLD_TMP WITH KEY BUKRS       = IT_DISP-BUKRS
                                BUPLA       = IT_DISP-BUPLA
                                ASP_CODE    = IT_DISP-ASP_CODE
                                IP_TYPTXT   = IT_DISP-IP_TYPTXT
                                NTS_SEND_YN = IT_DISP-NTS_SEND_YN
                                TYPE_CODE   = IT_DISP-TYPE_CODE
                                IP_ID       = IT_ETC-IP_ID.
    IF SY-SUBRC <> 0.
      CLEAR LT_TMP.
      MOVE-CORRESPONDING IT_ETC TO LT_TMP.
      IF IT_ETC-NTS_SEND_FLAG = '31'.
        LT_TMP-NTS_SEND_YN = 'Y'.
      ELSE.
        LT_TMP-NTS_SEND_YN = 'N'.
      ENDIF.
      LT_TMP-IP_TYPTXT = IT_DISP-IP_TYPTXT.
      LT_TMP-IP_CNT    = 1.
      APPEND LT_TMP TO OLD_TMP.
      LT_TMP-IP_ID     = ''.
      COLLECT LT_TMP.
    ENDIF.

*.. 발급건수
    IT_DISP-ISSUE_CNT = 1.
*
    COLLECT IT_DISP.
    CLEAR   IT_DISP.
  ENDLOOP.

*.. 매출처수
  SORT LT_TMP.
  LOOP AT IT_DISP.
    READ TABLE LT_TMP WITH KEY BUKRS       = IT_DISP-BUKRS
                               BUPLA       = IT_DISP-BUPLA
                               ASP_CODE    = IT_DISP-ASP_CODE
                               IP_TYPTXT   = IT_DISP-IP_TYPTXT
                               NTS_SEND_YN = IT_DISP-NTS_SEND_YN
                               TYPE_CODE   = IT_DISP-TYPE_CODE
                               BINARY SEARCH.
    CHECK SY-SUBRC = 0.
    IT_DISP-IP_CNT = LT_TMP-IP_CNT.
    MODIFY IT_DISP TRANSPORTING IP_CNT.
  ENDLOOP.

ENDFORM. " FILL_IT_DISP
