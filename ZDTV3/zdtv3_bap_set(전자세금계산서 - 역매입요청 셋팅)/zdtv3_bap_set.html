  CASE SY-DYNNR.

        SELECT SINGLE REPTEXT INTO IT_SER-FIELDTEXT
          FROM DD03L AS A INNER JOIN DD04T AS B
            ON A~ROLLNAME = B~ROLLNAME
         WHERE A~TABNAME   EQ IT_SER-TABNAME
           AND A~FIELDNAME EQ IT_SER-FIELDNAME.
      ENDIF.
    WHEN '0103'.

*&---------------------------------------------------------------------*
*&      Form  MAKE_SELECT
*&---------------------------------------------------------------------*
FORM MAKE_SELECT .
  DATA : LT_SER    LIKE IT_SER OCCURS 0 WITH HEADER LINE.
  DATA : L_DCNT(7) TYPE N,
         CONDITION(5), NAME(8).
  DATA : L_HEADER  LIKE THEAD,
         L_LINES   LIKE TLINE OCCURS 0 WITH HEADER LINE.
  CLEAR: IT_SUR, IT_SUR[], IT_DATA, IT_DATA[].


*// INTERNAL TABLE 생성
  LOOP AT IT_SER_M.
    AT NEW TABNAME.
*   DATA : BEGIN OF.....
      D_ITAB 'S' IT_SER_M-TABNAME '' ''.
      CONCATENATE '*' IT_SUR INTO IS_DATA SEPARATED BY SPACE.
      APPEND IS_DATA TO IT_DATA.
    ENDAT.
*   FIELD LIKE....
    D_ITAB 'B' '' IT_SER_M-FIELDNAME IT_SER_M-TABNAME.
    CONCATENATE '*' IT_SUR INTO IS_DATA SEPARATED BY SPACE.
    APPEND IS_DATA TO IT_DATA.
*
    AT END OF TABNAME.
*    END OF.....
      D_ITAB 'E' IT_SER_M-TABNAME '' ''.
      CONCATENATE '*' IT_SUR INTO IS_DATA SEPARATED BY SPACE.
      APPEND IS_DATA TO IT_DATA.
      APPEND '*' TO IT_DATA.
    ENDAT.
    CLEAR IS_DATA.
  ENDLOOP.

*// SELECT문 생성
  LT_SER[] = IT_SER[].
  SORT LT_SER BY TABNAME SSEQ.

  LOOP AT LT_SER.
    AT NEW TABNAME.
      CONCATENATE 'IT_' LT_SER-TABNAME INTO I_ITAB.
      CONCATENATE  I_ITAB '[]' INTO IT_DATA.
      CONCATENATE '* CLEAR :' I_ITAB ',' IT_DATA '.'
             INTO IT_DATA SEPARATED BY SPACE.
      APPEND IT_DATA.
      MOVE '* SELECT * ' TO IT_DATA.
      APPEND IT_DATA.
      CONCATENATE '* INTO CORRESPONDING FIELDS OF TABLE' I_ITAB
             INTO IT_DATA SEPARATED BY SPACE.
      APPEND IT_DATA.
      CONCATENATE '* FROM' LT_SER-TABNAME
             INTO  IT_DATA SEPARATED BY SPACE.
      APPEND IT_DATA.
      MOVE 'WHERE' TO CONDITION.
    ENDAT.

*   WHERE 조건
    ADD 1 TO L_DCNT.
    CONCATENATE '&' L_DCNT INTO NAME.

    IF CONDITION IS INITIAL.
      MOVE 'AND' TO CONDITION.
    ENDIF.

    CASE LT_SER-KIND.
      WHEN 'P'.
        CONCATENATE '*' CONDITION LT_SER-FIELDNAME 'EQ' NAME
             INTO  IT_DATA SEPARATED BY SPACE.
        APPEND IT_DATA.
      WHEN 'S'.
        CONCATENATE '*' CONDITION LT_SER-FIELDNAME 'IN' NAME
             INTO  IT_DATA SEPARATED BY SPACE.
        APPEND IT_DATA.
    ENDCASE.
    CLEAR CONDITION.

    AT END OF TABNAME.
      APPEND '*' TO IT_DATA.
    ENDAT.
  ENDLOOP.

*// 텍스트저장
  LOOP AT IT_DATA.
    L_LINES-TDFORMAT = '/'.
