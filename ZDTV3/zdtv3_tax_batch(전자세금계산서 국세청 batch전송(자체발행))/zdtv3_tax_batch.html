FORM MODIFY_SCREEN .

  LOOP AT SCREEN.
    IF SCREEN-GROUP1 EQ 'Z1'.

      MODIFY SCREEN.
    ENDIF.

    ENDIF.

FORM SELECT_DATA .
  CLEAR : R_STATUS[], R_TYPE_CODE[].

* 2017.06.19 검색조건 추가 - 발행일자
DATA : L_EXDDTM-LOW TYPE ZDTV3T_AR_HEAD-ISSUE_DT.
DATA : L_EXDDTM-HIGH TYPE ZDTV3T_AR_HEAD-ISSUE_DT.

* 전송상태 리스트
  SELECT STATUS
    INTO R_STATUS-LOW
    FROM ZDTV3T_TAX_SEND
  WHERE BUKRS EQ P_BUKRS.
    R_STATUS-SIGN   = 'I'.
    R_STATUS-OPTION = 'EQ'.
    APPEND R_STATUS.
  ENDSELECT.

* 전송구분자 리스트
  SELECT TYPE_CODE
    INTO R_TYPE_CODE-LOW
    FROM ZDTV3T_TAX_TYPE
  WHERE BUKRS EQ P_BUKRS.
    R_TYPE_CODE-SIGN   = 'I'.
    R_TYPE_CODE-OPTION = 'EQ'.
    APPEND R_TYPE_CODE.
  ENDSELECT.

  IF  R_STATUS-LOW EQ SPACE.
    MESSAGE E000 WITH '국세청전송상태 누락!'.
  ENDIF.

  CLEAR : GT_NTS_SEND , GT_NTS_SEND[].

  CONCATENATE S_EXDDT-LOW  '000000' INTO L_EXDDTM-LOW.
  CONCATENATE S_EXDDT-HIGH '235959' INTO L_EXDDTM-HIGH.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_NTS_SEND
    FROM ZDTV3T_AR_HEAD
   WHERE BUKRS         EQ P_BUKRS
*     AND ISSUE_DATE    IN S_ISSUE
     AND PROC_STEP     NE SPACE
     AND STATUS        IN R_STATUS
     AND ASP_CODE      EQ P_ASP
     AND NTS_SEND_FLAG IN S_FLAG
*     AND TYPE_CODE IN R_TYPE_CODE
*     AND ERDAT         LE P_DATUM
     AND ISSUE_DT      GE L_EXDDTM-LOW
     AND ISSUE_DT      LE L_EXDDTM-HIGH.
*
*  IF NOT GT_NTS_SEND[] IS INITIAL .
**  READ TABLE S_ISSUE INDEX 1.
**  IF SY-SUBRC EQ 0.
**수정세금계산서 일 경우.
*    SELECT *
*       APPENDING CORRESPONDING FIELDS OF TABLE GT_NTS_SEND
*      FROM ZDTV3T_AR_HEAD
*     WHERE BUKRS         EQ P_BUKRS
**       AND ISSUE_DATE    LE S_ISSUE-LOW
*       AND PROC_STEP     NE SPACE
*       AND STATUS        IN R_STATUS
*       AND ASP_CODE      EQ P_ASP
*       AND NTS_SEND_FLAG IN S_FLAG
**       AND TYPE_CODE     IN R_TYPE_CODE
**       AND ERDAT         LE P_DATUM
*       AND AMEND_CODE NE ''
*       AND ISSUE_DT      GE L_EXDDTM-LOW
*       AND ISSUE_DT      LE L_EXDDTM-HIGH.
*  ENDIF.

  include ZDTV3I_U806 IF  FOUND.

ENDFORM. " SELECT_DATA

FORM WRITE_DATA .
  FORMAT COLOR 2 INTENSIFIED ON.
  WRITE :/ '회사코드:' , P_BUKRS .
  FORMAT COLOR 2 INTENSIFIED OFF.
*  LOOP AT S_BUPLA.
*    WRITE :/ '사 업 장:' , S_BUPLA-LOW .
*  ENDLOOP.
  FORMAT COLOR 2 INTENSIFIED ON.
*  WRITE :/ '승인일자:' , P_DATUM .
  FORMAT COLOR 2 INTENSIFIED OFF.
*  WRITE :/ '대상기간:' , S_ISSUE-LOW ,'~' , S_ISSUE-HIGH .
  FORMAT COLOR 2 INTENSIFIED ON.
  ULINE .
  WRITE :/(12) SPACE , 13 '전체건수' , 25 '성공건수' , 37 '실패건수'.
  ULINE .
  FORMAT COLOR 2 INTENSIFIED OFF.
  WRITE :/ '전송건수:' , TOT_CNT , OK_CNT , ERR_CNT.
  ULINE .
  FORMAT COLOR 2 INTENSIFIED ON.

ENDFORM. " WRITE_DATA

FORM SET_ISSUE_DATUM .
*
** 마감 관리 기준  일자 확인
*  DATA: LS_CDATE TYPE ZDTV3T_CLOSED-TODAT.
*  CLEAR LS_CDATE.
*
*  CLEAR : S_ISSUE[].
*  CLEAR : GT_CLOSED, GT_CLOSED[].
*
*
*  S_ISSUE-SIGN   = 'I'.
*  S_ISSUE-OPTION = 'BT'.
*
**  IF P_DATUM+6(2) <= '10'.
*  CONCATENATE P_DATUM+0(6) '01' INTO S_ISSUE-LOW.
*  S_ISSUE-LOW = S_ISSUE-LOW - 1.
*  CONCATENATE S_ISSUE-LOW+0(6) '01' INTO S_ISSUE-LOW.
*  S_ISSUE-HIGH = P_DATUM.
**  ELSE.
**    CONCATENATE P_DATUM+0(6) '01' INTO S_ISSUE-LOW.
**    PERFORM LAST_DAY USING P_DATUM S_ISSUE-HIGH.
**  ENDIF.
**
*  INCLUDE ZDTV3I_U807 IF  FOUND.
ENDFORM. " SET_ISSUE_DATUM
