    EXIT.
  ENDIF.

* 점검및 전송

*   대상자료읽기.
  PERFORM GET_DATA.

  CASE 'X'.
    WHEN P_AP.
      IF GT_AP_HD[] IS INITIAL.
        MESSAGE S000 DISPLAY LIKE 'E' WITH '전송대상을 찾을 수 없습니다.'.
        EXIT.
      ENDIF.
    WHEN P_AR.
      IF GT_AR_HD[] IS INITIAL.
        MESSAGE S000 DISPLAY LIKE 'E' WITH '전송대상을 찾을 수 없습니다.'.
        EXIT.
      ENDIF.
  ENDCASE.

*   발행.
  PERFORM SEND_DATA.
*   오류 LOG...출력...
  SORT GT_RETURN BY INV_SEQ.

                              GS_RETURN-CODE GS_RETURN-MESSAGE.
  ENDLOOP.

FORM GET_DATA .

  CLEAR : GS_RETURN.

* 입력자료점검(입력한 INV_SEQ가 HEAD에 있는지)
  CASE 'X'.
    WHEN P_AP.
      SELECT SINGLE MANDT INTO SY-MANDT
        FROM ZDTV3T_AP_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ
        AND STATUS NOT IN ('2797', '2888', '2999').
      IF SY-SUBRC NE 0.
        GS_RETURN-TYPE = 'E'.
        GS_RETURN-INV_SEQ = P_INVSEQ.
        GS_RETURN-CODE = '17008'.
        EXIT.
      ENDIF.

      "HEAD
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AP_HD
        FROM ZDTV3T_AP_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "ITEM
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AP_IM
        FROM ZDTV3T_AP_ITEM
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "PAYM
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AP_PM
        FROM ZDTV3T_AP_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "EXT
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AP_EX
        FROM ZDTV3T_AP_EXT
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "EXTD
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AP_EX_D
        FROM ZDTV3T_AP_EXT_D
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.

      SORT GT_AP_HD BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AP_IM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AP_PM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AP_EX BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AP_EX_D BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

    WHEN P_AR.
      SELECT SINGLE MANDT INTO SY-MANDT
        FROM ZDTV3T_AR_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ
        AND STATUS NOT IN ('1797', '1888', '1999').
      IF SY-SUBRC NE 0.
        GS_RETURN-TYPE = 'E'.
        GS_RETURN-INV_SEQ = P_INVSEQ.
        GS_RETURN-CODE = '17008'.
        EXIT.
      ENDIF.

      "HEAD
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AR_HD
        FROM ZDTV3T_AR_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "ITEM
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AR_IM
        FROM ZDTV3T_AR_ITEM
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "PAYM
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AR_PM
        FROM ZDTV3T_AR_HEAD
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "EXT
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AR_EX
        FROM ZDTV3T_AR_EXT
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.
      "EXTD
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_AR_EX
        FROM ZDTV3T_AR_EXT_D
        WHERE BUKRS EQ P_BUKRS
        AND INV_SEQ EQ P_INVSEQ.

      SORT GT_AR_HD BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AR_IM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AR_PM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AR_EX BY BUKRS ISSUE_DATE BUPLA INV_SEQ.
      SORT GT_AR_EX_D BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

  ENDCASE.

ENDFORM. " GET_DATA
