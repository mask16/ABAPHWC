      FROM DD07T
     WHERE DOMNAME    EQ 'ZDTV3D_AMEND_CODE'
       AND DDLANGUAGE EQ '3'.

    LOOP AT IT_NTS.

*     1-1. 세금계산서 종류 텍스트
      READ TABLE IT_TYPE WITH KEY DOMVALUE_L = IT_NTS-TYPE_CODE.

      ENDIF.

      IF SY-SUBRC EQ 0.

                         BINARY SEARCH.

        CONTINUE.

          IF IT_NTS-ISSUE_PD <= LV_ENDDATE.

          ENDIF.

          IF SY-SUBRC EQ 0.
            LV_ENDDATE = LV_ENDDATE + 1.
          ENDIF.

                FACTORY_CALENDAR = 'KR'
                DATE_FROM        = LV_ENDDATE

              TABLES

          ENDIF.

      ENDCASE.
