FORM SELECT_DATA .
*
  CASE 'X'.
    WHEN P_AP.
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
        FROM ZDTV3T_AP_HEAD
       WHERE BUKRS EQ P_BUKRS
         AND BUPLA EQ P_BUPLA
         AND ISSUE_DATE IN S_DATE
         AND STATUS EQ '2666'.

    WHEN P_AR.

      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
        FROM ZDTV3T_AR_HEAD
       WHERE BUKRS EQ P_BUKRS
         AND BUPLA EQ P_BUPLA
         AND ISSUE_DATE IN S_DATE
         AND STATUS EQ '1666'.

  ENDCASE.

ENDFORM. " SELECT_DATA
