FUNCTION ZDTV3SAP_SEL_CUST_IPTYPE.
*"----------------------------------------------------------------------
*"*"Local interface:
*" EXPORTING
*" VALUE(IP_TYPECODE) TYPE ZDTV3E_IP_TYPECODE
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*" CHANGING
*" VALUE(IP_ID) TYPE ZDTV3E_IP_ID
*" VALUE(DESC_TEXT1) TYPE ZDTV3E_DESC_TEXT1 OPTIONAL
*"----------------------------------------------------------------------
*사업자등록번호 형식 : nnnnnnnnnn (-불포함, 10byte)
*개인 주민번호 형식 : nnnnnnnnnnnnn (-불포함, 13byte)
*외국인 형식 : 9999999999999(고정값) (비고에 여권번호 및 외국인등록번호
*등을 기재)
************************************************************************
* Global data declarations

  DATA : L_LEN(2) TYPE N.
* 값에 "-" 포함된경우 제거함..
  call function 'ZDTV3SAP_CHK_FVALUE_HYPHEN'
    CHANGING
      VALUE = IP_ID.
*
  CONDENSE IP_ID NO-GAPS.
* 길이가 10 자리인경우 사업자번호
  CASE STRLEN( IP_ID ).
    WHEN '10'.
      IP_TYPECODE = '01'.
    WHEN '13'.
      CASE IP_ID+6(1).
        WHEN '1' OR '2' OR '3' OR '4' OR '9' OR '0'.
          IP_TYPECODE = '02'.
        WHEN '5' OR '6' OR '7' OR'8'.
* BC 에서 처리함....
* CONCATENATE DESC_TEXT1 '/' IP_ID INTO DESC_TEXT1.
          IP_TYPECODE = '03'.
* IP_ID = '9999999999999'.
        WHEN OTHERS.
          DS_RETURN RETURN 'E' '12006' '' IP_ID '' '' ''.
          EXIT.
      ENDCASE.

    WHEN OTHERS.
      DS_RETURN RETURN 'E' '12005' '' IP_ID '' '' ''.
      EXIT.
  ENDCASE.


ENDFUNCTION.
