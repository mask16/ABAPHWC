REPORT ZDTV3_USER20 .

DATA LT_KNA1 LIKE KNA1 OCCURS 0 WITH HEADER LINE.
DATA LT_HEAD LIKE ZDTV3T_AR_HEAD OCCURS 0 WITH HEADER LINE.

CLEAR : LT_KNA1, LT_KNA1[], LT_HEAD, LT_HEAD[].

SELECT *
  INTO CORRESPONDING FIELDS OF TABLE LT_KNA1
  FROM KNA1 AS A INNER JOIN KNB1 AS B
    ON A~KUNNR EQ B~KUNNR
 WHERE B~BUKRS EQ '1000'
   AND B~LOEVM EQ ' '.

SELECT *
  INTO CORRESPONDING FIELDS OF TABLE LT_HEAD
  FROM ZDTV3T_AR_HEAD AS A INNER JOIN ZDTV3T_AR_EXT AS B
    ON A~INV_SEQ EQ B~INV_SEQ
 WHERE STATUS  EQ '1797'
   AND USE_DOC EQ SPACE
   AND KUNNR   EQ SPACE.

LOOP AT LT_HEAD.
  READ TABLE LT_KNA1 WITH KEY STCD2 = LT_HEAD-IP_ID.
  IF SY-SUBRC EQ 0.
    UPDATE ZDTV3T_AR_EXT
       SET KUNNR = LT_KNA1-KUNNR
     WHERE INV_SEQ EQ LT_HEAD-INV_SEQ.
  ELSE.
    READ TABLE LT_KNA1 WITH KEY STCD1 = LT_HEAD-IP_ID.
    IF SY-SUBRC EQ 0.
      UPDATE ZDTV3T_AR_EXT
         SET KUNNR = LT_KNA1-KUNNR
       WHERE INV_SEQ EQ LT_HEAD-INV_SEQ.
    ENDIF.
  ENDIF.
ENDLOOP.
