FORM PASS_VALUE USING P_SEQ
                          PS_UPLOAD
                          PS_MAIN.

  DATA : L_FIELD TYPE FIELDNAME.
  DATA : L_IP_ID TYPE STRING.
  FIELD-SYMBOLS : <FS_SOURCE>, <FS_TARGET>.

  CASE P_TAX.
    WHEN 'X'.
      CASE 'X'.
        WHEN P_AR."매출
          READ TABLE IT_SEQ INTO IS_SEQ WITH KEY SEQ_AR = P_SEQ
                                                 BINARY SEARCH.
        WHEN P_AP."매입
          READ TABLE IT_SEQ INTO IS_SEQ WITH KEY SEQ_AP = P_SEQ
                                                 BINARY SEARCH.
      ENDCASE.
    WHEN ' '.
      CASE 'X'.
        WHEN P_AR."매출
          READ TABLE IT_SEQ INTO IS_SEQ WITH KEY SEQ_AR2 = P_SEQ
                                                 BINARY SEARCH.
        WHEN P_AP."매입
          READ TABLE IT_SEQ INTO IS_SEQ WITH KEY SEQ_AP2 = P_SEQ
                                                 BINARY SEARCH.
      ENDCASE.
  ENDCASE.


  CHECK SY-SUBRC EQ 0.

  CONCATENATE 'PS_UPLOAD-FIELD' P_SEQ+3(2) INTO L_FIELD.
  ASSIGN (L_FIELD) TO <FS_SOURCE>.
  ASSIGN COMPONENT IS_SEQ-FIELDNAME OF STRUCTURE PS_MAIN TO <FS_TARGET>.
  MOVE <FS_SOURCE> TO <FS_TARGET>.


  CASE IS_SEQ-FIELDNAME.
    WHEN 'IP_ID'.
      L_IP_ID = <FS_SOURCE> .
      CASE STRLEN( L_IP_ID ).
        WHEN '10'.
          IT_MAIN-IP_TYPECODE = '01'.
        WHEN '13'.
          IF L_IP_ID EQ '9999999999999'.
            IT_MAIN-IP_TYPECODE = '03'.
          ELSE.
            IT_MAIN-IP_TYPECODE = '02'.
          ENDIF.
      ENDCASE.

    WHEN 'PURP_CODE'.
      CASE <FS_SOURCE>.
        WHEN '영수'.
          IT_MAIN-PURP_CODE = '01'.
        WHEN '청구'.
          IT_MAIN-PURP_CODE = '02'.
      ENDCASE.
  ENDCASE.

ENDFORM. " PASS_VALUE

FORM PASS_VALUE_ETC USING PS_UPLOAD
                              PS_MAIN.
  DATA : LT_TEXT TYPE TABLE OF CHAR255,
         LS_TEXT TYPE CHAR255.
  DATA : TYPE    TYPE C LENGTH 1,
         COMP    TYPE I.
  DATA : FIELD   TYPE FIELDNAME,
         SEQ     TYPE N LENGTH 2.
  FIELD-SYMBOLS : <FS_VAL>.


  DESCRIBE FIELD: PS_UPLOAD TYPE TYPE COMPONENTS COMP.
  DO COMP TIMES.
    MOVE SY-INDEX TO SEQ.
    CONCATENATE 'PS_UPLOAD-FIELD' SEQ INTO FIELD.
    ASSIGN (FIELD) TO <FS_VAL>.
    CLEAR LS_TEXT.
    MOVE <FS_VAL> TO LS_TEXT.
    APPEND LS_TEXT TO LT_TEXT.
  ENDDO.


  SEARCH LT_TEXT FOR '세금계산서' AND MARK.
  IF SY-SUBRC EQ 0.
    SEARCH LT_TEXT FOR '일반' AND MARK.
    IF SY-SUBRC EQ 0.
