FORM COLLECT_DATA .

  DATA : LS_MAIN  LIKE IT_MAIN.
  DATA : LS_NTS02 LIKE IT_NTS02.
  CLEAR : IT_MAIN.
  LOOP AT IT_NTS02.
    MOVE-CORRESPONDING IT_NTS02 TO IT_MAIN.
    IF IT_NTS02-INVCNT IS INITIAL.
      IT_NTS02-INVCNT = 1 .
    ENDIF.
    IT_MAIN-INVCNT2      = IT_NTS02-INVCNT.
    IT_MAIN-CHARGETOTAL2 = IT_NTS02-CHARGETOTAL.
    IT_MAIN-TAXTOTAL2    = IT_NTS02-TAXTOTAL.
    IT_MAIN-GRANDTOTAL2  = IT_NTS02-GRANDTOTAL.
    COLLECT IT_MAIN. CLEAR IT_MAIN.
  ENDLOOP.

  CLEAR : IT_MAIN.
  SORT IT_NTS02 BY INVKEY.
  LOOP AT IT_NTS01.
    MOVE-CORRESPONDING IT_NTS01 TO IT_MAIN.
    READ TABLE IT_NTS02 WITH KEY INVKEY = IT_NTS01-INVKEY
                            BINARY SEARCH
                            INTO LS_NTS02.
    IF SY-SUBRC EQ 0.
      IT_MAIN-ISSUE_ID = LS_NTS02-ISSUE_ID.
      IF IT_MAIN-BUPLA IS INITIAL.
        IT_MAIN-BUPLA = LS_NTS02-BUPLA.
      ENDIF.
    ENDIF.
*
    IF IT_NTS01-INVCNT IS INITIAL.
      IT_NTS01-INVCNT = 1 .
    ENDIF.
    IT_MAIN-INVCNT1      = IT_NTS01-INVCNT.
    IT_MAIN-CHARGETOTAL1 = IT_NTS01-CHARGETOTAL.
    IT_MAIN-TAXTOTAL1    = IT_NTS01-TAXTOTAL.
    IT_MAIN-GRANDTOTAL1  = IT_NTS01-GRANDTOTAL.
    COLLECT IT_MAIN. CLEAR IT_MAIN.
  ENDLOOP.

  CLEAR : IT_MAIN.
  SORT IT_NTS02 BY ISSUE_ID.
  LOOP AT IT_NTS03.
    MOVE-CORRESPONDING IT_NTS03 TO IT_MAIN.
    READ TABLE IT_NTS02 WITH KEY ISSUE_ID = IT_NTS03-ISSUE_ID
                            BINARY SEARCH
                            INTO LS_NTS02.
    IF SY-SUBRC EQ 0.
      IT_MAIN-INVKEY = LS_NTS02-INVKEY.
      IF IT_MAIN-BUPLA IS INITIAL.
        IT_MAIN-BUPLA = LS_NTS02-BUPLA.
      ENDIF.
    ENDIF.

    IF IT_NTS03-INVCNT IS INITIAL.
      IT_NTS03-INVCNT = 1 .
    ENDIF.
    IT_MAIN-INVCNT3      = IT_NTS03-INVCNT.
    IT_MAIN-CHARGETOTAL3 = IT_NTS03-CHARGETOTAL.
    IT_MAIN-TAXTOTAL3    = IT_NTS03-TAXTOTAL.
    IT_MAIN-GRANDTOTAL3  = IT_NTS03-GRANDTOTAL.
    COLLECT IT_MAIN. CLEAR IT_MAIN.
  ENDLOOP.

ENDFORM. " COLLECT_DATA

FORM GET_FILEDCATALOG_ALV .
  DATA: LS_FIELDCAT     TYPE SLIS_FIELDCAT_ALV.

  PERFORM GET_FILEDCATALOG TABLES G_FIELDCAT_T
                           USING  'IT_MAIN'.

  LOOP AT G_FIELDCAT_T INTO LS_FIELDCAT.
    CASE LS_FIELDCAT-FIELDNAME.
      WHEN 'NAME'.
        LS_FIELDCAT-SELTEXT_S = '사업장명'.
        LS_FIELDCAT-DDICTXT  = 'S'.
      WHEN 'NAME1'.
        LS_FIELDCAT-SELTEXT_S = '고객/구매처명'.
        LS_FIELDCAT-DDICTXT  = 'S'.
*      WHEN 'ISSUE_ID'.
*        LS_FIELDCAT-FIX_COLUMN = 'X'.
      WHEN 'INVCNT1' OR 'CHARGETOTAL1' OR 'TAXTOTAL1' OR 'GRANDTOTAL1'.
        CONCATENATE '고객-' LS_FIELDCAT-SELTEXT_S
               INTO LS_FIELDCAT-SELTEXT_S.
        LS_FIELDCAT-DDICTXT  = 'S'.
        LS_FIELDCAT-CURRENCY = 'KRW'.
        LS_FIELDCAT-DO_SUM   = 'X'.
        IF LS_FIELDCAT-FIELDNAME EQ 'INVCNT1'.
          LS_FIELDCAT-HOTSPOT  = 'X'.
        ENDIF.
      WHEN 'INVCNT2' OR 'CHARGETOTAL2' OR 'TAXTOTAL2' OR 'GRANDTOTAL2'.
        CONCATENATE 'ASP-' LS_FIELDCAT-SELTEXT_S
               INTO LS_FIELDCAT-SELTEXT_S.
        LS_FIELDCAT-DDICTXT  = 'S'.
        LS_FIELDCAT-CURRENCY = 'KRW'.
        LS_FIELDCAT-DO_SUM   = 'X'.
        IF LS_FIELDCAT-FIELDNAME EQ 'INVCNT2'.
          LS_FIELDCAT-HOTSPOT  = 'X'.
        ENDIF.
      WHEN 'INVCNT3' OR 'CHARGETOTAL3' OR 'TAXTOTAL3' OR 'GRANDTOTAL3'.
        CONCATENATE 'NTS-' LS_FIELDCAT-SELTEXT_S
               INTO LS_FIELDCAT-SELTEXT_S.
        LS_FIELDCAT-DDICTXT  = 'S'.
        LS_FIELDCAT-CURRENCY = 'KRW'.
        LS_FIELDCAT-DO_SUM   = 'X'.
        IF LS_FIELDCAT-FIELDNAME EQ 'INVCNT3'.
          LS_FIELDCAT-HOTSPOT  = 'X'.
        ENDIF.
    ENDCASE.
    MODIFY G_FIELDCAT_T FROM LS_FIELDCAT.
  ENDLOOP.

ENDFORM. " GET_FILEDCATALOG_ALV
