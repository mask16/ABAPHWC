FORM FILL_IT_ALV .
  DATA : BEGIN OF LT_TMP OCCURS 0,
           BUKRS       LIKE ZDTV3T_AR_HEAD-BUKRS,
           BUPLA       LIKE ZDTV3T_AR_HEAD-BUPLA,
           ASP_CODE    LIKE ZDTV3T_AR_HEAD-ASP_CODE,
           IP_TYPTXT(20),
           TYPE_CODE   LIKE ZDTV3T_AR_HEAD-TYPE_CODE,
           STATUS      LIKE ZDTV3T_AR_HEAD-STATUS,
           SU_ID       LIKE ZDTV3T_AR_HEAD-SU_ID,
           IP_CNT      TYPE I.
  DATA : END OF LT_TMP.
  DATA : OLD_TMP  LIKE TABLE OF LT_TMP WITH HEADER LINE.
  CLEAR: IT_ALV, IT_ALV[].

  SORT: IT_AP_HEAD,
        IT_ASP_CODE,
        IT_STATUS,
        IT_DD07V.


  LOOP AT IT_AP_HEAD.
    MOVE-CORRESPONDING IT_AP_HEAD TO IT_ALV.

*.. ASP 코드 TEXT
    READ TABLE IT_ASP_CODE WITH KEY ASP_CODE = IT_ALV-ASP_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-ASP_CODE_TEXT = IT_ASP_CODE-ASP_CODE_TEXT.
    ENDIF.

*.. 상태 TEXT
    READ TABLE IT_STATUS WITH KEY STATUS = IT_ALV-STATUS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-STATUS_TEXT = IT_STATUS-STATUS_TEXT.
    ENDIF.

**.. 구분
*    CASE IT_AP_HEAD-IP_TYPECODE.
*      WHEN '01'.
*        IT_ALV-IP_TYPTXT = '사업자번호 발행분'.
*      WHEN '02' OR '03'.
*        IT_ALV-IP_TYPTXT = '주민등록번호 발행분'.
*    ENDCASE.

*.. 세금계산서 종류
    READ TABLE IT_DD07V WITH KEY DOMVALUE_L = IT_AP_HEAD-TYPE_CODE
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-DDTEXT = IT_DD07V-DDTEXT.
    ENDIF.

*~  매출처
    READ TABLE OLD_TMP WITH KEY BUKRS     = IT_ALV-BUKRS
                                BUPLA     = IT_ALV-BUPLA
                                ASP_CODE  = IT_ALV-ASP_CODE
                                IP_TYPTXT = IT_ALV-IP_TYPTXT
                                TYPE_CODE = IT_ALV-TYPE_CODE
                                STATUS    = IT_ALV-STATUS
                                SU_ID     = IT_AP_HEAD-SU_ID.
    IF SY-SUBRC <> 0.
      CLEAR LT_TMP.
      MOVE-CORRESPONDING IT_AP_HEAD TO LT_TMP.
      LT_TMP-IP_TYPTXT = IT_ALV-IP_TYPTXT.
      LT_TMP-IP_CNT    = 1.
      APPEND LT_TMP TO OLD_TMP.
      LT_TMP-SU_ID     = ''.
      COLLECT LT_TMP.
    ENDIF.

*.. 발급건수
    IT_ALV-ISSUE_CNT = 1.
*
    COLLECT IT_ALV.
    CLEAR   IT_ALV.
  ENDLOOP.

*.. 매출처수
  SORT LT_TMP.
  LOOP AT IT_ALV.
    READ TABLE LT_TMP WITH KEY BUKRS     = IT_ALV-BUKRS
                               BUPLA     = IT_ALV-BUPLA
                               ASP_CODE  = IT_ALV-ASP_CODE
                               IP_TYPTXT = IT_ALV-IP_TYPTXT
                               TYPE_CODE = IT_ALV-TYPE_CODE
                               STATUS    = IT_ALV-STATUS
                               BINARY SEARCH.
    CHECK SY-SUBRC = 0.
    IT_ALV-IP_CNT = LT_TMP-IP_CNT.
    MODIFY IT_ALV TRANSPORTING IP_CNT.
  ENDLOOP.

ENDFORM. " FILL_IT_ALV

FORM CHECK_DYNPRO_VALS .
  DATA: LV_CNT(1) TYPE N,
        LV_PARAM(10),
        LV_CHK.
  DATA: LV_STATUS  LIKE ZDTV3T_FUNCTION-STATUS.
  FIELD-SYMBOLS: <FS> TYPE ANY.

  CLEAR: R_STATUS, R_STATUS[], LV_CNT, LV_CHK.


* 일자 CHECK
  IF S_ISSDAT[] IS INITIAL AND S_ISSDT[] IS INITIAL.
    MESSAGE I000 WITH TEXT-M01.
    GV_PROGRESS = 'N'.
    EXIT.
  ENDIF.

* 문서상태 CHECK
  IF NOT IT_SOURCE[] IS INITIAL.
    DO 9 TIMES.
      ADD 1 TO LV_CNT.
      CONCATENATE 'P_STAT' LV_CNT INTO LV_PARAM.
      ASSIGN (LV_PARAM) TO <FS>.
      IF <FS> IS ASSIGNED AND <FS> = 'X'.
        LV_CHK = 'X'.
        EXIT.
      ENDIF.
    ENDDO.

    IF LV_CHK <> 'X'.
      MESSAGE I000 WITH TEXT-M02.
      GV_PROGRESS = 'N'.
      EXIT.
    ENDIF.
  ENDIF.

*// 문서상태
  IF NOT IT_SOURCE[] IS INITIAL.
    CLEAR LV_CNT.
    DO 9 TIMES.
      ADD 1 TO LV_CNT.
      CONCATENATE 'P_STAT' LV_CNT INTO LV_PARAM.
      ASSIGN (LV_PARAM) TO <FS>.
      IF <FS> IS ASSIGNED AND <FS> = 'X'.
        SELECT *
          FROM ZDTV3T_GROUP_FT
         WHERE ZPROGRAM = 'ZDTV3_AP'
           AND DOC_STAT = LV_CNT.

          SELECT STATUS  INTO LV_STATUS
                         FROM ZDTV3T_FUNCTION
                        WHERE FUNCT = ZDTV3T_GROUP_FT-FUNCT.
            R_STATUS = 'IEQ'.
            R_STATUS-LOW = LV_STATUS.
            APPEND R_STATUS.
          ENDSELECT.
        ENDSELECT.
      ENDIF.
    ENDDO.
  ENDIF.

ENDFORM. " CHECK_DYNPRO_VALS

FORM INIT_DATA .
  DATA : L_CH_DATE LIKE SY-DATUM.

  SELECT SINGLE UDAT INTO L_CH_DATE
    FROM REPOSRC
   WHERE PROGNAME = 'ZDTV3_AP_DOCSTAT_CBOX'
     AND R3STATE  = 'A'.

  SELECT SINGLE *
    FROM ZDTV3T_DOC_STAT
   WHERE ZPROGRAM  = 'ZDTV3_AP'
     AND ERDAT    >= L_CH_DATE.

  IF SY-SUBRC = 0 OR TEXT-900 = ''.
    PERFORM SET_SUB_SUBSCREEN_CBOX(ZDTV3_FORM_COMM)
            USING G_PROGRAM 'AP'.
    GENERATE REPORT G_PROGRAM.
  ENDIF.

ENDFORM. " INIT_DATA
