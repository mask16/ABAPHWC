  G_PROGRAM     = 'ZDTV3_TRAN'.

  PERFORM MAKE_SAVE_DATA TABLES INV_HD

                                INV_EX
                                INV_EX_D
                         USING  SEND ARAP.
* 세금계산서 저장.
  IF NOT GT_TRAN_HD[] IS INITIAL.
    PERFORM AR_INV_SAVE_DATA.
    READ TABLE GT_TRAN_HD INDEX 1.
  ELSE.
    RETURN[] = GT_RETURN[].
    EXIT.
  ENDIF.

* 전송
  IF SEND EQ 'X'.
    CLEAR : GT_RETURN , GT_RETURN[].
    G_UPDMODE = 'S'.
** 동기/비동기 결정.
* PERFORM GET_SYNC.
* BC전송.
    PERFORM SEND_DATA USING ARAP G_UPDMODE.
    READ TABLE GT_RETURN INDEX 1.
    IF SY-SUBRC NE 0.
      DT_RETURN GT_RETURN 'S' '10022' '' '' '' '' ''.
* 오류를 RETURN 하구 생성된 자료는 삭제함.
* PERFORM DELETE_DATA USING G_DEL_TRAN_SEQ.
      RETURN[] = GT_RETURN[].

      RETURN[] = GT_RETURN[].
    ENDIF.
  ELSE.
* MESSAGE RETURN.
    G_TEXT = LINES( GT_TRAN_HD[] ).

    CONCATENATE G_TEXT '건을 생성 하였습니다.' INTO G_TEXT.
    DT_RETURN GT_RETURN 'S' '00000' '' G_TEXT '' '' ''.
    RETURN[] = GT_RETURN[].
  ENDIF.
