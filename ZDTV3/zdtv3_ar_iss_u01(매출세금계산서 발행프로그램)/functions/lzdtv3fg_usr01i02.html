          IF G_ANSWER EQ 'J'.
            PERFORM CHECK_DATA_9100.
            CHECK G_MSG_GB NE 'E'.
            PERFORM SAVE_DATA_9100.
            LEAVE TO SCREEN 0 .
          ELSE.
            G_MSG_GB = '3'.
          ENDIF.
        ELSE.

        ENDIF.
        PERFORM FREE_RTN.
