*----------------------------------------------------------------------*
FORM DYNPRO USING P_DYNBEGIN P_NAME P_VALUE.

  IF P_DYNBEGIN = 'X'.

    CLEAR    G_BDC_T.
    MOVE:    P_NAME    TO G_BDC_T-PROGRAM,
             P_VALUE   TO G_BDC_T-DYNPRO,
            'X'        TO G_BDC_T-DYNBEGIN.
    APPEND   G_BDC_T.

  ELSE.

    CLEAR    G_BDC_T.
    MOVE:    P_NAME    TO G_BDC_T-FNAM,
             P_VALUE   TO G_BDC_T-FVAL.
    APPEND   G_BDC_T.

  ENDIF.

ENDFORM. "DYNPRO
