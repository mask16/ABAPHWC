FORM GET_ZDTV3T_AR_HEAD .

FORM GET_FILEDCAT_ALV USING GT_FIELDCAT TYPE LVC_T_FCAT.

  DATA : LS_FIELDCAT TYPE LVC_S_FCAT.
  DATA : LT_FIELDCAT TYPE SLIS_T_FIELDCAT_ALV,
         L_FIELDCAT  TYPE SLIS_FIELDCAT_ALV.

  CALL FUNCTION 'REUSE_ALV_FIELDCATALOG_MERGE'
    EXPORTING
      I_PROGRAM_NAME     = G_REPID_C
      I_INTERNAL_TABNAME = 'IT_ALV'

      I_INCLNAME         = G_REPID_C
    CHANGING
      CT_FIELDCAT        = LT_FIELDCAT[].

  LOOP AT LT_FIELDCAT INTO L_FIELDCAT.
    CASE L_FIELDCAT-FIELDNAME.
      WHEN 'NTS_SEND_TEXT' OR 'TYPE_CODE_TEXT' OR 'ERROR_CODE_TEXT'.
        L_FIELDCAT-SELTEXT_S = 'Text'.
      WHEN 'ASP_CODE'.

      WHEN 'IP_CNT'.
        L_FIELDCAT-SELTEXT_S = '매출처수'.
        L_FIELDCAT-EMPHASIZE = 'C700'.
      WHEN 'ISSUE_CNT'.
        L_FIELDCAT-SELTEXT_S = '발급건수'.
        L_FIELDCAT-EMPHASIZE = 'C700'.
        L_FIELDCAT-HOTSPOT = 'X'.
    ENDCASE.
*
    MOVE-CORRESPONDING L_FIELDCAT TO LS_FIELDCAT.
    LS_FIELDCAT-REPTEXT   = L_FIELDCAT-SELTEXT_S.
    LS_FIELDCAT-REF_TABLE = L_FIELDCAT-REF_TABNAME.

*   통화단위.

FORM FILL_IT_ALV .
  DATA : BEGIN OF LT_TMP OCCURS 0,
           BUKRS         LIKE ZDTV3T_AR_HEAD-BUKRS,
           BUPLA         LIKE ZDTV3T_AR_HEAD-BUPLA,
           NTS_SEND_FLAG LIKE ZDTV3T_AR_HEAD-NTS_SEND_FLAG,
           STATUS        LIKE ZDTV3T_AR_HEAD-STATUS,
           IP_TYPTXT(20),
           TYPE_CODE     LIKE ZDTV3T_AR_HEAD-TYPE_CODE,
           IP_ID         LIKE ZDTV3T_AR_HEAD-IP_ID,
           IP_CNT        TYPE I.
  DATA : END OF LT_TMP.
*  2012.02.22 튜닝
*  DATA : OLD_TMP  LIKE TABLE OF LT_TMP WITH HEADER LINE. <-삭제
  DATA : OLD_TMP  LIKE SORTED TABLE OF LT_TMP
                  WITH NON-UNIQUE KEY BUKRS BUPLA NTS_SEND_FLAG
                                      STATUS IP_TYPTXT TYPE_CODE
                                      IP_ID
                  WITH HEADER LINE.                      "<-추가
  CLEAR: IT_ALV, IT_ALV[].

  SORT: IT_AR_HEAD,
        IT_NTSSND_T,
        IT_STATUS,
        IT_TYPECD_T,
        IT_TAX_STAT BY BUKRS ISSUE_DATE BUPLA INV_SEQ ASCENDING
                       STAT_SEQ DESCENDING.


  LOOP AT IT_AR_HEAD.
    MOVE-CORRESPONDING IT_AR_HEAD TO IT_ALV.

*.. 구분
    CASE IT_AR_HEAD-IP_TYPECODE.
      WHEN '01'.
        IT_ALV-IP_TYPTXT = '사업자번호 발행분'.
      WHEN '02' OR '03'.
        IT_ALV-IP_TYPTXT = '주민등록번호 발행분'.
    ENDCASE.

*.. 국세청 전송상태 TEXT
    READ TABLE IT_NTSSND_T WITH KEY
                  DOMVALUE_L = IT_AR_HEAD-NTS_SEND_FLAG
                               BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-NTS_SEND_TEXT = IT_NTSSND_T-DDTEXT.
    ENDIF.

*.. 문서상태 TEXT
    READ TABLE IT_STATUS WITH KEY STATUS = IT_AR_HEAD-STATUS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-STATUS_TEXT = IT_STATUS-STATUS_TEXT.
    ENDIF.

*.. 세금계산서 종류 TEXT
    READ TABLE IT_TYPECD_T WITH KEY
                  DOMVALUE_L = IT_AR_HEAD-TYPE_CODE
                               BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-TYPE_CODE_TEXT = IT_TYPECD_T-DDTEXT.
    ENDIF.

*~  매출처
    READ TABLE OLD_TMP WITH KEY BUKRS         = IT_ALV-BUKRS
                                BUPLA         = IT_ALV-BUPLA
                                NTS_SEND_FLAG = IT_ALV-NTS_SEND_FLAG
                                STATUS        = IT_ALV-STATUS
                                IP_TYPTXT     = IT_ALV-IP_TYPTXT
                                TYPE_CODE     = IT_ALV-TYPE_CODE
                                IP_ID         = IT_AR_HEAD-IP_ID.
    IF SY-SUBRC <> 0.
      CLEAR LT_TMP.
      MOVE-CORRESPONDING IT_AR_HEAD TO LT_TMP.
      LT_TMP-IP_TYPTXT = IT_ALV-IP_TYPTXT.
      LT_TMP-IP_CNT    = 1.
*      2012.02.22 튜닝
*      APPEND LT_TMP TO OLD_TMP. <-삭제
      MOVE-CORRESPONDING LT_TMP TO OLD_TMP.
      INSERT TABLE OLD_TMP.
      CLEAR : OLD_TMP.           "<-추가

      LT_TMP-IP_ID     = ''.
      COLLECT LT_TMP.
    ENDIF.

*.. 발급건수
    IT_ALV-ISSUE_CNT = 1.
*
    COLLECT IT_ALV.
    CLEAR   IT_ALV.
  ENDLOOP.

*.. 매출처수
  SORT LT_TMP.
  LOOP AT IT_ALV.
    READ TABLE LT_TMP WITH KEY BUKRS         = IT_ALV-BUKRS
                               BUPLA         = IT_ALV-BUPLA
                               NTS_SEND_FLAG = IT_ALV-NTS_SEND_FLAG
                               STATUS        = IT_ALV-STATUS
                               IP_TYPTXT     = IT_ALV-IP_TYPTXT
                               TYPE_CODE     = IT_ALV-TYPE_CODE
                               BINARY SEARCH.
    CHECK SY-SUBRC = 0.
    IT_ALV-IP_CNT = LT_TMP-IP_CNT.
    MODIFY IT_ALV TRANSPORTING IP_CNT.
  ENDLOOP.

ENDFORM. " FILL_IT_ALV

FORM GET_DOC_STAT USING P_STAT.
  DATA: VALUE(30), L_CNT TYPE I.

  LOOP AT IT_GROUP_FT WHERE DOC_STAT = P_STAT.

      CONCATENATE  `'` IT_FUNCTION-PROC_STEP `'` INTO VALUE.
      CONCATENATE 'AND I~PROC_STEP EQ ' VALUE ')'
            INTO IT_WHERE SEPARATED BY SPACE.
      APPEND IT_WHERE.
    ENDLOOP.
