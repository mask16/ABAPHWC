FORM SELECT_DATA .
  DATA L_TABIX LIKE SY-TABIX.
  DATA L_BUPLA_ID TYPE STCD2.
  RANGES : R_STCD2 FOR KNA1-STCD2.

  CLEAR : IT_NTS01,IT_NTS01[],IT_NTS,IT_NTS[].

* 사업장명.
  CLEAR : IT_NAME[],IT_NAME, IT_NAME, IT_NAME[].
  SELECT *
    FROM J_1BBRANCH
   WHERE BUKRS EQ P_BUKRS
     AND BRANCH IN S_BUPLA.

    R_STCD2-LOW    = J_1BBRANCH-STCD2.
    R_STCD2-SIGN   = 'I'.
    R_STCD2-OPTION = 'EQ'.
    APPEND R_STCD2.  CLEAR R_STCD2.

    MOVE : J_1BBRANCH-STCD2 TO IT_NAME-STCD2,
           J_1BBRANCH-NAME TO IT_NAME-NAME,
           J_1BBRANCH-BRANCH TO IT_NAME-BUPLA.

    APPEND IT_NAME.  CLEAR IT_NAME.
  ENDSELECT.
*
  CASE I_ARAP.
    WHEN 'AR'.
      G_PROGRAM = 'ZDTV3_AR'.

      SELECT * INTO TABLE IT_NTS01
        FROM ZDTV3T_NTS
       WHERE ISSUE_DATE IN S_DATUM
         AND ISSUE_PD   IN S_ISSUE
         AND TRANS_DATE IN S_TRANS
         AND TYPE_CODE  IN S_TCODE
         AND SU_ID      IN R_STCD2
         AND IP_ID      IN S_ID.

    WHEN 'AP'.
      G_PROGRAM = 'ZDTV3_AP'.

      SELECT * INTO TABLE IT_NTS01
        FROM ZDTV3T_NTS
       WHERE ISSUE_DATE IN S_DATUM
         AND ISSUE_PD   IN S_ISSUE
         AND TRANS_DATE IN S_TRANS
         AND TYPE_CODE  IN S_TCODE
         AND SU_ID      IN S_ID
         AND IP_ID      IN R_STCD2.

  ENDCASE.

  LOOP AT IT_NTS01 .

    MOVE-CORRESPONDING : IT_NTS01 TO IT_NTS.
    MOVE P_BUKRS                  TO IT_NTS-BUKRS.

    CLEAR L_BUPLA_ID.
    CASE I_ARAP.
      WHEN 'AR'.
        MOVE: IT_NTS01-IP_ID TO IT_NTS-ID,
              IT_NTS01-SU_ID TO L_BUPLA_ID.
      WHEN 'AP'.
        MOVE: IT_NTS01-SU_ID TO IT_NTS-ID,
              IT_NTS01-IP_ID TO L_BUPLA_ID.
    ENDCASE.

    READ TABLE IT_NAME WITH KEY STCD2 = L_BUPLA_ID.
    IF SY-SUBRC EQ 0.
      MOVE IT_NAME-BUPLA TO IT_NTS-BUPLA.
    ENDIF.

    APPEND IT_NTS. CLEAR IT_NTS.

  ENDLOOP.



ENDFORM. " SELECT_DATA

FORM MODIFY_DATA .
  DATA : L_INV_SEQ TYPE ZDTV3E_SSUE_ID.
  DATA : L_LIGHTS ,  L_TABIX LIKE SY-TABIX,
         L_NTS_TYPE  TYPE    ZDTV3E_NTS_TYPE.


  SORT : IT_NTS   BY BUKRS BUPLA ID,
         IT_NTS01 BY ISSUE_ID.


  LOOP AT IT_NTS.

*   전자/일반
    call function 'ZDTV3SAP_NTS_TYPE'
      EXPORTING
        ISSUE_ID = IT_NTS-ISSUE_ID
      CHANGING
        NTS_GB   = IT_NTS-GB
        NTS_TYPE = L_NTS_TYPE.

    IF IT_NTS-GB = 'X'.
*     전자
    ELSE.
*     일반
      IT_NTS-GB = 'O'.
    ENDIF.

    READ TABLE IT_NTS01 WITH KEY ISSUE_ID = IT_NTS-ISSUE_ID
                                            BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      L_TABIX = SY-TABIX.
      IT_NTS01-ERNAM = IT_NTS-GB.
      MODIFY IT_NTS01 INDEX L_TABIX TRANSPORTING ERNAM.
    ENDIF.

*   사업자/주민번호
    call function 'ZDTV3SAP_SEL_CUST_IPTYPE'
      IMPORTING
        IP_TYPECODE = IT_NTS-TYPE_CD
      CHANGING
        IP_ID       = IT_NTS-ID.


    IF IT_NTS-TYPE_CD = '01'.            "외국인인 주민번호등록분으로..
*      IT_MAIN-TYPE_TXT = TEXT-T96.
    ELSE.
      IT_NTS-TYPE_CD = '02'.
*      IT_MAIN-TYPE_TXT = TEXT-T97.
    ENDIF.

    CONCATENATE IT_NTS-BUKRS IT_NTS-BUPLA IT_NTS-GB IT_NTS-TYPE_CD
                IT_NTS-ID
           INTO IT_NTS-KEY.

    MODIFY IT_NTS TRANSPORTING GB TYPE_CD KEY . CLEAR IT_NTS.
  ENDLOOP.

*
  SORT IT_NTS BY BUKRS BUPLA GB DESCENDING TYPE_CD ID.

  LOOP AT IT_NTS.

    MOVE-CORRESPONDING IT_NTS TO IT_MAIN.

    AT NEW KEY.
      IT_MAIN-ID_CNT = 1.
    ENDAT.
    IT_MAIN-INVCNT = 1.


    IF IT_NTS-GB = 'X'.
      IT_MAIN-GB_TXT = TEXT-T94.
    ELSE.
      IT_MAIN-GB_TXT = TEXT-T95.
    ENDIF.
    IF IT_MAIN-TYPE_CD = '01'.            "외국인인 주민번호등록분으로..
      IT_MAIN-TYPE_TXT = TEXT-T96.
    ELSE.
      IT_MAIN-TYPE_CD = '02'.
      IT_MAIN-TYPE_TXT = TEXT-T97.
    ENDIF.

    CASE I_ARAP.
      WHEN 'AR'.
        READ TABLE IT_NAME1 WITH KEY ID = IT_MAIN-ID.
        IF SY-SUBRC EQ 0.
          IT_MAIN-NAME1 = IT_NAME1-NAME1.
        ELSE.
          SELECT SINGLE NAME1 INTO IT_MAIN-NAME1
            FROM KNA1
           WHERE STCD2 EQ IT_MAIN-ID.
        ENDIF.
      WHEN 'AP'.
        READ TABLE IT_NAME1 WITH KEY ID = IT_MAIN-ID.
        IF SY-SUBRC EQ 0.
          IT_MAIN-NAME1 = IT_NAME1-NAME1.
        ELSE.
          SELECT SINGLE NAME1 INTO IT_MAIN-NAME1
            FROM LFA1
           WHERE STCD2 EQ IT_MAIN-ID.
        ENDIF.
    ENDCASE.

    COLLECT IT_MAIN. CLEAR IT_MAIN.
  ENDLOOP.

ENDFORM. " SET_TEXT
