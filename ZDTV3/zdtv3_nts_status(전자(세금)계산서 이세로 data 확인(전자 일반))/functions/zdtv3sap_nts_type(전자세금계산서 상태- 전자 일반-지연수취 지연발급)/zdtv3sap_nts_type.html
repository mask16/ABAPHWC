Code listing for function ZDTV3SAP_NTS_TYPE

Description: 전자세금계산서 상태- 전자 일반-지연수취 지연발급
FUNCTION ZDTV3SAP_NTS_TYPE.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(ISSUE_ID) TYPE ZDTV3E_SSUE_ID
*" EXPORTING
*" VALUE(RETURN) LIKE ZDTV3S_RETURN STRUCTURE ZDTV3S_RETURN
*" CHANGING
*" VALUE(NTS_ARAP) TYPE ZDTV3E_ARAP OPTIONAL
*" VALUE(NTS_GB) TYPE ZDTV3E_NTS_GB OPTIONAL
*" VALUE(NTS_TYPE) TYPE ZDTV3E_NTS_TYPE OPTIONAL
*"----------------------------------------------------------------------
*{ INSERT MCDK900646 1
* 전자세금계산서 전송기한을 발급일 다음날로 변경.
* 부가가치세법 시행령 개정(안)은 2012.7.1

*' ' 정상
*1 미발급
*2 지연발급
*3 지연수취
*4 미전송
*5 지연전송
* Global data declarations

  DATA  : DELAY_DT    TYPE    DATUM.

  CLEAR : G_CANCLE , GS_ZDTV3_AMEND,
          NTS_ARAP , NTS_GB,  NTS_TYPE.
  G_ISSUE_ID = ISSUE_ID.
*
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF GS_NTS
    FROM ZDTV3T_NTS
   WHERE ISSUE_ID EQ G_ISSUE_ID.
  IF SY-SUBRC NE 0.
    DS_RETURN RETURN 'E' '14005' '' G_ISSUE_ID '' '' ''.
    EXIT.
  ENDIF.

* AR/AP Return
  SELECT SINGLE MANDT INTO SY-MANDT
    FROM J_1BBRANCH
   WHERE STCD2 EQ GS_NTS-SU_ID.
  IF SY-SUBRC EQ 0.
    NTS_ARAP = 'AR'.
  ELSE.
    NTS_ARAP = 'AP'.
  ENDIF.

* SELECT SINGLE MANDT INTO SY-MANDT
* FROM ZDTV3T_AR_HEAD
* WHERE ISSUE_ID EQ G_ISSUE_ID.
* IF SY-SUBRC EQ 0.
* NTS_ARAP = 'AR'.
* ELSE.
* SELECT SINGLE MANDT INTO SY-MANDT
* FROM ZDTV3T_AP_HEAD
* WHERE ISSUE_ID EQ G_ISSUE_ID.
* IF SY-SUBRC EQ 0.
* NTS_ARAP = 'AR'.
* ELSE.
* DS_RETURN RETURN 'E' '14005' '' G_ISSUE_ID '' '' ''.
* EXIT.
* ENDIF.
* ENDIF.


* 마감일 체크
  DATA : CLOSE_DATE TYPE DATUM.
  CLEAR: CLOSE_DATE.
  SELECT SINGLE DATUM INTO CLOSE_DATE
    FROM ZDTV3T_NTS_CLOSE
   WHERE GJAHR EQ GS_NTS-ISSUE_DATE(4)
     AND MONAT EQ GS_NTS-ISSUE_DATE+4(2).
  IF SY-SUBRC NE 0.
      DS_RETURN RETURN 'E' '14005' '' G_ISSUE_ID '' '' ''.
      EXIT.
  ENDIF.

*
  PERFORM GET_DELAY_DT USING CLOSE_DATE.

* 전자/종이 CHECK GS_NTS-TRANS_DATE ISSUE_DATE
  IF GS_NTS-TRANS_DATE > CLOSE_DATE.
    NTS_GB = '' .  "종이(일반)
  ELSE.
    NTS_GB = 'X'.  "전자
  ENDIF.

  DELAY_DT = GS_NTS-ISSUE_PD + 1    .
  PERFORM GET_DELAY_DT USING DELAY_DT.

* 지연전송/수취 CHECK - 발행일+1 까지

  IF GS_NTS-TRANS_DATE > DELAY_DT.

    CASE NTS_ARAP.
      WHEN 'AR'.
        NTS_TYPE = '5' .  "지연진송
      WHEN 'AP'.
        NTS_TYPE = '3' .  "지연수취
    ENDCASE.

  ELSE.
    NTS_TYPE = ' '.  "정상
  ENDIF.




*} INSERT
ENDFUNCTION.
