Description: Include ZDTV3_TRAN_SEL

SELECT *
INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
FROM ZDTV3T_TRAN_HD AS A INNER JOIN ZDTV3T_TRAN_EXT AS B
ON A~BUKRS = B~BUKRS
AND A~ISSUE_DATE = B~ISSUE_DATE
AND A~BUPLA = B~BUPLA
AND A~TRAN_SEQ = B~TRAN_SEQ
INNER JOIN ZDTV3T_TRAN_EXTD AS C
ON A~BUKRS = C~BUKRS
AND A~ISSUE_DATE = C~ISSUE_DATE
AND A~BUPLA = C~BUPLA
AND A~TRAN_SEQ = C~TRAN_SEQ
WHERE A~BUKRS EQ &0000001
AND A~BUPLA EQ &0000002
AND A~ISSUE_DATE IN &0000003
*AND A~TRAN_SEQ IN &0000004
AND A~TYPE_CODE IN &0000005
AND B~KUNNR IN &0000006
AND A~AMEND_CODE IN &0000007
AND C~BUDAT IN &0000008
AND C~BLDAT IN &0000009
AND C~BLART IN &0000010
AND C~MWSKZ IN &0000011
AND A~IP_ID IN &0000012
AND A~IP_NAME IN &0000013
AND A~IP_PERSNAME1 IN &0000014
