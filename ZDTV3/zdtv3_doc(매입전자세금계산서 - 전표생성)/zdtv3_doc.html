FORM CHECK_INVSEQ .

  DATA : L_USE_DOC LIKE ZDTV3T_AP_EXT-USE_DOC.

  SELECT SINGLE USE_DOC INTO L_USE_DOC
    FROM ZDTV3T_AP_HEAD AS A INNER JOIN ZDTV3T_AP_EXT AS B
      ON A~BUKRS      = B~BUKRS
     AND A~ISSUE_DATE = B~ISSUE_DATE
     AND A~BUPLA      = B~BUPLA
     AND A~INV_SEQ    = B~INV_SEQ
   WHERE A~BUKRS   EQ P_BUKRS
     AND A~INV_SEQ EQ P_INVSEQ.
  IF SY-SUBRC NE 0.
    MESSAGE E000 WITH '회사' P_BUKRS P_INVSEQ '가 존재하지 않습니다.'.
  ENDIF.
*
  IF L_USE_DOC EQ 'X'.
    MESSAGE E000 WITH '회사' P_BUKRS P_INVSEQ '는 전표가 존재합니다.'.
  ENDIF.

ENDFORM. " CHECK_INVSEQ
