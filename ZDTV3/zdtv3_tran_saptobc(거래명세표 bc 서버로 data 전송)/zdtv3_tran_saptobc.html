                              GS_RETURN-CODE GS_RETURN-MESSAGE.
    EXIT.
  ENDIF.

* 회사코드확인.

  IF GS_RETURN-TYPE EQ 'E'.
    PERFORM SET_MESSAGE USING GS_RETURN-TYPE GS_RETURN-INV_SEQ
                              GS_RETURN-CODE GS_RETURN-MESSAGE.
    EXIT.
  ENDIF.

* 처리대상자료를 일자별로 LOCK.
* PERFORM MASS_LOCK.
  LOOP AT GT_RETURN INTO GS_RETURN

                              GS_RETURN-CODE GS_RETURN-MESSAGE.
  ENDLOOP.

* 점검및 전송
  PERFORM SET_MESSAGE USING 'S' '' '' '점검 및 전송을 시작합니다.'.
  CLEAR : I_TOT_CNT, I_ER_CNT , I_OK_CNT , I_CNT .

    PERFORM SEND_DATA.

    SORT GT_RETURN BY INV_SEQ.

      PERFORM SET_MESSAGE USING GS_RETURN-TYPE GS_RETURN-INV_SEQ
                                GS_RETURN-CODE GS_RETURN-MESSAGE.

  ENDDO.

FORM GET_DATA .
*
  SELECT *
         INTO CORRESPONDING FIELDS OF TABLE GT_HD
         FROM ZDTV3T_TRAN_HD UP TO P_CNT ROWS
        WHERE BUKRS      EQ P_BUKRS
          AND STATUS     EQ '3000'
          AND PROC_STEP  IN ('','00','11','12')
          AND ISSUE_DATE EQ  P_DATUM
*          AND ISSUE_DATE IN R_ERDAT_DEL
          AND TRAN_SEQ   IN S_KEY.

  SORT GT_HD BY BUKRS ISSUE_DATE BUPLA TRAN_SEQ.

  SELECT *
       INTO CORRESPONDING FIELDS OF TABLE GT_IM
       FROM ZDTV3T_TRAN_IM
      WHERE BUKRS      EQ P_BUKRS
        AND ISSUE_DATE EQ P_DATUM
        AND TRAN_SEQ   IN S_KEY.

  SORT GT_IM BY BUKRS ISSUE_DATE BUPLA TRAN_SEQ.

  LOOP AT GT_IM.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_IM-BUKRS
                              ISSUE_DATE = GT_IM-ISSUE_DATE
                              BUPLA      = GT_IM-BUPLA
                              TRAN_SEQ   = GT_IM-TRAN_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC NE 0.
      DELETE GT_IM.
    ENDIF.
  ENDLOOP.

  SELECT *
       INTO CORRESPONDING FIELDS OF TABLE GT_EX
       FROM ZDTV3T_TRAN_EXT
      WHERE BUKRS      EQ P_BUKRS
        AND ISSUE_DATE EQ P_DATUM
        AND TRAN_SEQ   IN S_KEY.

  SORT GT_EX BY BUKRS ISSUE_DATE BUPLA TRAN_SEQ.

  LOOP AT GT_EX.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_EX-BUKRS
                              ISSUE_DATE = GT_EX-ISSUE_DATE
                              BUPLA      = GT_EX-BUPLA
                              TRAN_SEQ   = GT_EX-TRAN_SEQ.
    IF SY-SUBRC NE 0.
      DELETE GT_EX.
    ENDIF.
  ENDLOOP.

  SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_EX_D
        FROM ZDTV3T_TRAN_EXTD
       WHERE BUKRS      EQ GT_HD-BUKRS
         AND ISSUE_DATE EQ P_DATUM
         AND TRAN_SEQ   IN S_KEY.

  SORT GT_EX_D BY BUKRS ISSUE_DATE BUPLA TRAN_SEQ.

  LOOP AT GT_EX_D.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_EX_D-BUKRS
                              ISSUE_DATE = GT_EX_D-ISSUE_DATE
                              BUPLA      = GT_EX_D-BUPLA
                              TRAN_SEQ   = GT_EX_D-TRAN_SEQ.
    IF SY-SUBRC NE 0.
      DELETE GT_EX_D.
    ENDIF.
  ENDLOOP.
*
ENDFORM. " GET_DATA
