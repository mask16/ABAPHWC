FORM FBRA_DATA USING P_BUKRS

                         P_BUDAT
                         P_TEST
                         P_BELNR
                         LT_MSG TYPE BAPIRETTAB.

  IF NOT P_BUDAT IS INITIAL.

  ENDIF.
*--- Reset payment doc
  CALL FUNCTION 'CALL_FBRA'
    EXPORTING
      I_BUKRS      = P_BUKRS

    EXCEPTIONS
      NOT_POSSIBLE = 1
      OTHERS       = 2.
*--- Reverse Payment doc.
  CALL FUNCTION 'CALL_FB08'
    EXPORTING
      I_BUKRS      = P_BUKRS
      I_BELNR      = P_AUGBL
      I_GJAHR      = P_AUGGJ
      I_STGRD      = P_STGRD
      I_BUDAT      = L_BUDAT

      I_MODE       = 'N'

      NOT_POSSIBLE = 1
      OTHERS       = 2.
*--- Reverse failed
  IF SY-SUBRC <> 0.
    CLEAR MSG.REFRESH MSG.
    IMPORT MSG FROM DATABASE RFDT(RA) ID 'FBRA'.
    LOOP AT MSG.
      LS_MSG-TYPE       = MSG-MSGTYP.
      LS_MSG-ID         = MSG-MSGID.
      LS_MSG-NUMBER     = MSG-MSGNR.
      LS_MSG-MESSAGE    = ''.
      LS_MSG-MESSAGE_V1 = MSG-MSGV1.
      LS_MSG-MESSAGE_V2 = MSG-MSGV2.
      LS_MSG-MESSAGE_V3 = MSG-MSGV3.
      LS_MSG-MESSAGE_V4 = MSG-MSGV4.
      APPEND LS_MSG TO LT_MSG.
    ENDLOOP.
    DELETE FROM DATABASE RFDT(RA) ID 'FBRA'.
    EXIT.
  ELSE.
