FUNCTION ZDTV3SAP_SND_TRAN .
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(DDEMODE) TYPE DDEMODE OPTIONAL
*" VALUE(STATUS) TYPE ZDTV3E_STATUS OPTIONAL
*" VALUE(UPDMODE) TYPE CTU_UPDATE OPTIONAL
*" TABLES
*" INV_HD STRUCTURE ZDTV3T_TRAN_HD
*" INV_IM STRUCTURE ZDTV3T_TRAN_IM
*" INV_EX STRUCTURE ZDTV3T_TRAN_EXT OPTIONAL

*변수초기화.
* Global data declarations

  PERFORM INIT_DATA.
* G_STATUS = STATUS.
  G_UPDMODE = UPDMODE.
  G_PROGRAM = 'ZDTV3_TRAN'.
* 건수.
  G_CNT = LINES( INV_HD[] ).
* 1.실행모드 확인
  CASE DDEMODE.
    WHEN 'S'."개별처리
      IF G_CNT NE 1.
        DT_RETURN RETURN 'E' '10008' '' '' '' '' ''.
        EXIT.
      ENDIF.
    WHEN 'M'."대량처리
      IF G_CNT EQ 0 .
        DT_RETURN RETURN 'E' '10009' '' '' '' '' ''.
        EXIT.
      ENDIF.
    WHEN OTHERS.
      DT_RETURN RETURN 'E' '10010' DDEMODE '' '' '' ''.
      EXIT.
  ENDCASE.
  G_DDEMODE = DDEMODE.
* 2.하나의 회사코드만 처리가능.
  PERFORM CHECK_BUKRS TABLES INV_HD
                      USING G_BUKRS
                            G_ERROR.
  IF G_ERROR EQ 'X'.
    RETURN[] = GT_RETURN[].
    EXIT.
  ENDIF.
* 3.BC 서버연결확인.
  PERFORM CHECK_CONNECT USING G_BUKRS.
  IF GS_RETURN-TYPE EQ 'E'.
    APPEND GS_RETURN TO RETURN.
    EXIT.
  ENDIF.
* 4.기본값
  include ZDTV3_AR_INITIAL_DATA_GET.
* FIELD CHECK 용 자료읽기.
  PERFORM GET_FIELD_CHECK_DATA.
* 5.BC SERVER 전송할 자료를 만듬.
  PERFORM MAKE_SNED_DATA TABLES INV_HD
                                INV_IM
                                INV_EX
                                INV_EX_D
                         USING  STATUS.
* 6.상태 UPDATE / TABLE UPDATE.
  PERFORM UPDATE_DATA .
* 7.BC SERVER 전송
  CLEAR : GT_RETURN[].
  PERFORM SEND_BC_DATA USING STATUS.
* 8.RETURN 받은 결과기록
  PERFORM RESULT_UPDATE USING 'CRD' .
  RETURN[] = GT_RETURN[].
*
ENDFUNCTION.
