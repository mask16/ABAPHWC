FORM GET_FILEDCATALOG_ALV .
  DATA: LS_FIELDCAT     TYPE SLIS_FIELDCAT_ALV.

  PERFORM GET_FILEDCATALOG TABLES G_FIELDCAT_T
                           USING  'IT_MAIN'.

  LOOP AT G_FIELDCAT_T INTO LS_FIELDCAT.
    CASE LS_FIELDCAT-FIELDNAME.
      WHEN 'NAME'.
        LS_FIELDCAT-SELTEXT_S = '사업장명'.
        LS_FIELDCAT-DDICTXT   = 'S'.
      WHEN 'NAME1'.
        LS_FIELDCAT-SELTEXT_S = '고객/구매처명'.
        LS_FIELDCAT-DDICTXT  = 'S'.
*      WHEN 'ISSUE_ID'.
*        LS_FIELDCAT-FIX_COLUMN = 'X'.
      WHEN 'INVCNT1' OR 'CHARGETOTAL1' OR 'TAXTOTAL1' OR 'GRANDTOTAL1'.
        CONCATENATE ' 전표-' LS_FIELDCAT-SELTEXT_S
               INTO LS_FIELDCAT-SELTEXT_S.
        LS_FIELDCAT-DDICTXT  = 'S'.
        LS_FIELDCAT-CURRENCY = 'KRW'.
        LS_FIELDCAT-DO_SUM   = 'X'.
        IF LS_FIELDCAT-FIELDNAME EQ 'INVCNT1'.
          LS_FIELDCAT-HOTSPOT  = 'X'.
        ENDIF.
      WHEN 'INVCNT2' OR 'CHARGETOTAL2' OR 'TAXTOTAL2' OR 'GRANDTOTAL2'.
