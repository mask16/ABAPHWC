MODULE USER_COMMAND_0100 INPUT.
*
  DATA : LT_BSEG LIKE BBSEG OCCURS 0 WITH HEADER LINE.
*
  CLEAR : GT_BKPF[] , LT_BSEG[].
  APPEND BBKPF TO GT_BKPF.
* 일회성정보
  IF GS_HEAD-XCPDK EQ 'X'.
    MOVE-CORRESPONDING BSEC TO INT_BSEG.
  ENDIF.
* 화면의 변경사항반영.
  INT_BSEG-MWSKZ = GS_HEAD-MWSKZ.
  INT_BSEG-ZTERM = GS_HEAD-ZTERM.
  INT_BSEG-ZUONR = GS_HEAD-ZUONR.
  INT_BSEG-SGTXT = GS_HEAD-SGTXT.
*
  APPEND INT_BSEG TO LT_BSEG.
  APPEND LINES OF GT_BSEG[] TO LT_BSEG[].
*
  CASE OK_CODE.
    WHEN 'SAVE'.
      call function 'ZDTV3SAP_CRD_INT_POST'
        EXPORTING
          I_BDC_MODE = G_MOD
        IMPORTING
          E_BUKRS    = GS_DOCU-BUKRS
          E_BELNR    = GS_DOCU-BELNR
          E_GJAHR    = GS_DOCU-GJAHR
        TABLES
          T_BBKPF    = GT_BKPF
          T_BBSEG    = LT_BSEG
          T_MESSTAB  = GT_FIMSG.

      IF NOT GS_DOCU IS INITIAL.
        LEAVE TO SCREEN 0.
      ELSE.

        CALL FUNCTION 'MESSAGES_INITIALIZE'.

        DATA : L_CNT TYPE I.

        LOOP AT GT_FIMSG.
          ADD 1 TO L_CNT.
          CALL FUNCTION 'MESSAGE_STORE'
            EXPORTING
              ARBGB = GT_FIMSG-MSGID
              MSGTY = GT_FIMSG-MSGTY
              MSGV1 = GT_FIMSG-MSGV1
              MSGV2 = GT_FIMSG-MSGV2
              MSGV3 = GT_FIMSG-MSGV3
              MSGV4 = GT_FIMSG-MSGV4
              TXTNR = GT_FIMSG-MSGNO
              ZEILE = L_CNT.
        ENDLOOP.

        IF NOT GT_FIMSG[] IS INITIAL.
          CALL FUNCTION 'MESSAGES_SHOW'
            EXPORTING
              SHOW_LINNO = SPACE.
        ENDIF.


      ENDIF.
  ENDCASE.
*
ENDMODULE. " USER_COMMAND_0100 INPUT

FORM INIT_BBKPF USING P_PROG.

  DATA : FIELD(30).
  FIELD-SYMBOLS <FS> TYPE ANY.
*
  CLEAR : BBKPF.
* 회사코드
  BBKPF-BUKRS = ZDTV3T_DOC_CONF-BUKRS.
* 전기키
  IF GS_HEAD-GRANDTOTAL > 0 .
    GS_HEAD-NEWBS = ZDTV3T_DOC_CONF-BSCH1.
  ELSE.
    GS_HEAD-NEWBS = ZDTV3T_DOC_CONF-BSCH1_N.
  ENDIF.
* 전기일자 처리기준
  CASE ZDTV3T_DOC_CONF-BUDAT_FLAG.
    WHEN 'A'. "작성일자
      BBKPF-BUDAT = GS_HEAD-ISSUE_DATE.
    WHEN 'B'. "서명일자
      BBKPF-BUDAT = GS_HEAD-ISSUE_DT+0(8).
    WHEN 'C'. "현재일자
      BBKPF-BUDAT = SY-DATUM.
    WHEN OTHERS.
      BBKPF-BUDAT = SY-DATUM.
  ENDCASE.
* 증빙일자 처리기준
  CASE ZDTV3T_DOC_CONF-BLDAT_FLAG.
    WHEN 'A'. "작성일자
      BBKPF-BLDAT = GS_HEAD-ISSUE_DATE.
    WHEN 'B'. "서명일자
      BBKPF-BLDAT = GS_HEAD-ISSUE_DT+0(8).
    WHEN 'C'. "현재일자
      BBKPF-BLDAT = SY-DATUM.
    WHEN OTHERS.
      BBKPF-BLDAT = SY-DATUM.
  ENDCASE.
* 세금코드 결정
  CASE GS_HEAD-TYPE_CODE.
    WHEN '0101' OR '0103' OR '0201' OR '0203'.
      GS_HEAD-MWSKZ = ZDTV3T_DOC_CONF-MWSKZ01.
    WHEN '0102' OR '0105' OR '0202' OR '0205'.
      GS_HEAD-MWSKZ = ZDTV3T_DOC_CONF-MWSKZ02.
    WHEN '0301' OR '0303' OR '0401' OR '0403'.
      GS_HEAD-MWSKZ = ZDTV3T_DOC_CONF-MWSKZ03.
  ENDCASE.
* 전표헤더의 기본값.
  SELECT *
    FROM ZDTV3T_DOC_BKPF
   WHERE ZPROGRAM   EQ P_PROG
     AND BUKRS      EQ GS_HEAD-BUKRS
     AND FIELDNAME  NE SPACE
     AND FIELDVALUE NE SPACE.

    CONCATENATE 'BBKPF-' ZDTV3T_DOC_BKPF-FIELDNAME INTO FIELD.
    ASSIGN (FIELD) TO <FS> .
    IF <FS> IS ASSIGNED.
      <FS> = ZDTV3T_DOC_BKPF-FIELDVALUE .
    ENDIF.
    UNASSIGN <FS> .
  ENDSELECT.
*
ENDFORM. " INIT_BBKPF

FORM INIT_BBSEG USING P_PROG.
*

  WRITE L_AMT  TO INT_BSEG-WRBTR CURRENCY 'KRW'.

  INT_BSEG-BUPLA = GS_HEAD-BUPLA.

  INT_BSEG-ZLSPR = GS_HEAD-ZLSPR.

* 상대계정정보.
* 전표헤더의 기본값.

    FROM ZDTV3T_DOC_BSEG

FORM INIT_BSEC USING P_INV_SEQ.
  CLEAR : BSEC.
  CHECK GS_HEAD-XCPDK EQ 'X'.

  SELECT SINGLE *
    FROM ZDTV3T_AP_HEAD
   WHERE INV_SEQ EQ P_INV_SEQ.
  IF SY-SUBRC EQ 0.
    BSEC-NAME1      = ZDTV3T_AP_HEAD-SU_NAME.
    IF STRLEN( ZDTV3T_AP_HEAD-SU_ID ) EQ '13'.
      BSEC-STCD1      = ZDTV3T_AP_HEAD-SU_ID.
    ELSE.
      BSEC-STCD2      = ZDTV3T_AP_HEAD-SU_ID.
    ENDIF.
    BSEC-J_1KFREPRE = ZDTV3T_AP_HEAD-SU_REPRES.
    BSEC-J_1KFTBUS  = ZDTV3T_AP_HEAD-SU_BUSTYPE.
    BSEC-J_1KFTIND  = ZDTV3T_AP_HEAD-SU_INDTYPE.
    BSEC-ORT01      = ZDTV3T_AP_HEAD-SU_ADDR.
    BSEC-STRAS      = '.'.
  ENDIF.

ENDFORM. " INIT_BSEC
