FORM GET_DATA .

  CASE 'X'.

    WHEN P_AR.
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
        FROM ZDTV3T_AR_HEAD
       WHERE BUKRS      IN S_BUKRS
         AND ISSUE_DATE IN S_DATUM
         AND IP_ID      IN R_ID.

      LOOP AT IT_MAIN.
*        SELECT SINGLE MANDT INTO SY-MANDT
*          FROM ZDTV3T_SCODE
*         WHERE FROM_DATE   <= IT_MAIN-ISSUE_DATE
*           AND END_DATE    >= IT_MAIN-ISSUE_DATE
*           AND COMPANY_NUM EQ IT_MAIN-IP_ID.
*        IF SY-SUBRC NE 0.
*            DELETE IT_MAIN. CLEAR IT_MAIN.
*            CONTINUE.
*        ENDIF.

        SELECT SINGLE GOOD_NAME
          INTO CORRESPONDING FIELDS OF IT_MAIN
          FROM ZDTV3T_AR_ITEM
         WHERE INV_SEQ EQ IT_MAIN-INV_SEQ.
        MODIFY IT_MAIN.

        SELECT SINGLE ZGJAH BELNR
          INTO CORRESPONDING FIELDS OF IT_MAIN
          FROM ZDTV3T_AR_EXT_D
         WHERE INV_SEQ EQ IT_MAIN-INV_SEQ.
        MODIFY IT_MAIN.
        IF IT_MAIN-BELNR IS INITIAL.
          SELECT SINGLE GJAHR BELNR
            FROM BKPF
            INTO (IT_MAIN-ZGJAH, IT_MAIN-BELNR).
*          고객사 배포 후, 주석해제
*           WHERE ZZETAXNO EQ it_main-ISSUE_ID.
          MODIFY IT_MAIN.
        ENDIF.

        IF NOT IT_MAIN-BELNR IS INITIAL.
          SELECT SINGLE HKONT
            INTO CORRESPONDING FIELDS OF IT_MAIN
            FROM BSEG
           WHERE BUKRS IN S_BUKRS
             AND GJAHR EQ IT_MAIN-ZGJAH
             AND BELNR EQ IT_MAIN-BELNR
             AND KOART EQ 'S'
             AND KTOSL NOT IN ('MWS', 'VST').
          MODIFY IT_MAIN.

          SELECT SINGLE TXT20
            INTO CORRESPONDING FIELDS OF IT_MAIN
            FROM SKAT
           WHERE SAKNR EQ IT_MAIN-HKONT.
          MODIFY IT_MAIN.
        ENDIF.
      ENDLOOP.


    WHEN P_AP.
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
        FROM ZDTV3T_AP_HEAD
       WHERE BUKRS      IN S_BUKRS
         AND ISSUE_DATE IN S_DATUM
         AND SU_ID      IN R_ID.

      LOOP AT IT_MAIN.
*        SELECT SINGLE MANDT INTO SY-MANDT
*          FROM ZDTV3T_SCODE
*         WHERE FROM_DATE   <= IT_MAIN-ISSUE_DATE
*           AND END_DATE    >= IT_MAIN-ISSUE_DATE
*           AND COMPANY_NUM EQ IT_MAIN-SU_ID.
*        IF SY-SUBRC NE 0.
*            DELETE IT_MAIN. CLEAR IT_MAIN.
*            CONTINUE.
*        ENDIF.

        SELECT SINGLE ZGJAH BELNR
        INTO CORRESPONDING FIELDS OF IT_MAIN
        FROM ZDTV3T_AP_EXT_D
        WHERE INV_SEQ EQ IT_MAIN-INV_SEQ.
        IF IT_MAIN-BELNR IS INITIAL.
          SELECT SINGLE GJAHR BELNR
            FROM BKPF
            INTO (IT_MAIN-ZGJAH, IT_MAIN-BELNR).
*          고객사 배포 후, 주석해제
*           WHERE ZZETAXNO EQ it_main-ISSUE_ID.
          MODIFY IT_MAIN.
        ENDIF.

        IF NOT IT_MAIN-BELNR IS INITIAL.
          SELECT SINGLE HKONT
          INTO CORRESPONDING FIELDS OF IT_MAIN
          FROM BSEG
          WHERE BUKRS IN S_BUKRS
            AND GJAHR EQ IT_MAIN-ZGJAH
            AND BELNR EQ IT_MAIN-BELNR
            AND KOART EQ 'S'
            AND KTOSL NOT IN ('MWS', 'VST').
         MODIFY IT_MAIN.

          SELECT SINGLE TXT20
          INTO CORRESPONDING FIELDS OF IT_MAIN
          FROM SKAT
          WHERE SAKNR EQ IT_MAIN-HKONT.
          MODIFY IT_MAIN.
          CLEAR IT_MAIN.
       ENDIF.
      ENDLOOP.

  ENDCASE.

ENDFORM. " GET_DATA

FORM GET_FILEDCAT USING gt_fieldcat TYPE lvc_t_fcat.

  DATA : LS_FIELDCAT TYPE SLIS_FIELDCAT_ALV ,
         LS_FIELD_LVC TYPE LVC_S_FCAT.

  PERFORM GET_FILEDCATALOG TABLES G_FIELDCAT_T
                           USING  'IT_MAIN'.

  LOOP AT G_FIELDCAT_T INTO LS_FIELDCAT.

    IF LS_FIELDCAT-DATATYPE EQ 'CURR'.
      LS_FIELDCAT-CURRENCY = 'KRW'.
      LS_FIELDCAT-DO_SUM = 'X'.
    ENDIF.

    CASE LS_FIELDCAT-FIELDNAME.
      WHEN 'BELNR'.
      LS_FIELDCAT-hotspot = 'X'.
    ENDCASE.

    MODIFY G_FIELDCAT_T FROM LS_FIELDCAT.

  ENDLOOP.

ENDFORM. " GET_FILEDCAT_HALV

FORM LOCAL_USERCOMMAND USING L_UCOMM LIKE SY-UCOMM
                        LS_SELFIELD TYPE SLIS_SELFIELD.
*
  DATA : L_TYPE(2).
  DATA : LS_RETURN LIKE ZDTV3S_RETURN ,
         LT_RETURN LIKE ZDTV3S_RETURN OCCURS 0 WITH HEADER LINE.
*
  CASE 'X'.
    WHEN P_AR.
      L_TYPE = 'AR'.
    WHEN P_AP.
      L_TYPE = 'AP'.
  ENDCASE.
*
  CASE L_UCOMM.
*   Double Click..
    WHEN '&IC1'.
*
      CASE LS_SELFIELD-FIELDNAME.
        WHEN 'INV_SEQ' OR 'INV_SEQ_C'. "일련번호
          READ TABLE IT_MAIN INDEX LS_SELFIELD-TABINDEX.
*
          CHECK SY-SUBRC EQ 0.
*
          CLEAR : LS_RETURN , LT_RETURN[].
          call function 'ZDTV3SAP_DSP_INVOICE'
            EXPORTING
              INV_SEQ  = IT_MAIN-INV_SEQ
              INV_TYPE = L_TYPE
            IMPORTING
              RETURN   = LS_RETURN
            EXCEPTIONS
              OTHERS   = 4.
          IF LS_RETURN-TYPE EQ 'E'.
*     처리결과 MESSAGE.
            APPEND LS_RETURN TO LT_RETURN.
            call function 'ZDTV3SAP_DSP_MESSAGE'
              TABLES
                INPUT = LT_RETURN.
          ENDIF.
      WHEN 'BELNR'.
        READ TABLE it_main INDEX LS_SELFIELD-TABINDEX.
        IF NOT it_main-belnr IS INITIAL.
          SET PARAMETER ID 'BUK' FIELD it_main-bukrs.
          SET PARAMETER ID 'BLN' FIELD it_main-belnr.
          SET PARAMETER ID 'GJR' FIELD it_main-ZGJAH.
          CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
        ELSE.
          MESSAGE E000 WITH '등록된 전표가 없습니다.'.
        ENDIF.
      ENDCASE.
*
  ENDCASE.
*
ENDFORM. " LOCAL_USERCOMMAND
