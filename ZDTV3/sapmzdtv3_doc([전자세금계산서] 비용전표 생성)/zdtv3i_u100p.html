IF IS_INVOICE-GRANDTOTAL > 0 .
  ZDTV3S_DOC_LIFNR-BSCHL    = '31'.
ELSE.
  ZDTV3S_DOC_LIFNR-BSCHL    = '21'.
ENDIF.

IF IS_INVOICE-TAXTOTAL IS INITIAL.
  ZDTV3S_DOC_LIFNR-MWSKZ    = 'V1'.
ELSE.
  ZDTV3S_DOC_LIFNR-MWSKZ    = 'V4'.
ENDIF.

ZDTV3S_DOC_LIFNR-BUPLA = IS_INVOICE-BUPLA.
* 마이너스 세금계산서 처리 위한 절대값 처리
ZDTV3S_DOC_LIFNR-DMBTR = ABS( IS_INVOICE-GRANDTOTAL ).
*ZDTV3S_DOC_LIFNR-MWSTS    = IS_INVOICE-TAXTOTAL.
ZDTV3S_DOC_LIFNR-LIFNR = IS_INVOICE-LIFNR.
ZDTV3S_DOC_LIFNR-NAME1 = IS_INVOICE-SU_NAME.
ZDTV3S_DOC_LIFNR-STCD2 = IS_INVOICE-SU_ID.

IF NOT ZDTV3S_DOC_LIFNR-LIFNR IS INITIAL.

* 구매처명...
  IF NOT ZDTV3S_DOC_LIFNR-LIFNR IS INITIAL.
    SELECT SINGLE AKONT ZTERM
      INTO (ZDTV3S_DOC_LIFNR-HKONT,ZDTV3S_DOC_LIFNR-ZTERM)
      FROM LFB1
     WHERE LIFNR EQ ZDTV3S_DOC_LIFNR-LIFNR
       AND BUKRS EQ ZDTV3S_DOC_HEAD-BUKRS.
    IF SY-SUBRC NE 0.
      MESSAGE E000 WITH '존재하지 않는 거래처입니다.'.
    ENDIF.
*
    SELECT SINGLE TXT50 INTO ZDTV3S_DOC_LIFNR-TXT50
      FROM SKAT
     WHERE SPRAS EQ SY-LANGU
       AND KTOPL EQ T001-KTOPL
       AND SAKNR EQ ZDTV3S_DOC_LIFNR-HKONT.
*
    SELECT SINGLE NAME1 INTO ZDTV3S_DOC_LIFNR-NAME1
      FROM LFA1
     WHERE LIFNR EQ ZDTV3S_DOC_LIFNR-LIFNR.
  ENDIF.
ENDIF.
*
SELECT SINGLE NAME INTO ZDTV3S_DOC_LIFNR-NAME
  FROM J_1BBRANCH
 WHERE BUKRS  EQ ZDTV3S_DOC_HEAD-BUKRS
   AND BRANCH EQ ZDTV3S_DOC_LIFNR-BUPLA.
IF SY-SUBRC NE 0.
  CLEAR : ZDTV3S_DOC_LIFNR-NAME.
ENDIF.
*
CLEAR : GS_SKLIN.
CALL FUNCTION 'FI_FIND_PAYMENT_CONDITIONS'
  EXPORTING
    I_ZTERM            = ZDTV3S_DOC_LIFNR-ZTERM
    I_BLDAT            = ZDTV3S_DOC_HEAD-BLDAT
    I_BUDAT            = ZDTV3S_DOC_HEAD-BUDAT
    I_ZFBDT            = ZDTV3S_DOC_LIFNR-ZFBDT
    I_BUKRS            = ZDTV3S_DOC_HEAD-BUKRS
  IMPORTING
    E_ZFBDT            = ZDTV3S_DOC_LIFNR-ZFBDT
    E_SKLIN            = GS_SKLIN
  EXCEPTIONS
    TERMS_INCORRECT    = 1
    TERMS_NOT_FOUND    = 2
    NO_DATE_ENTERED    = 3
    NO_DAY_LIMIT_FOUND = 4
    OTHERS             = 5.

** 전표 항목...
IF ZDTV3S_DOC_HEAD-BUKRS EQ 'KR01'.
  G_HKONT = '0013020300'.
ELSE.
  G_HKONT = '0001112101'.
ENDIF.
CLEAR : GT_ITEM[].
GT_ITEM-HKONT = G_HKONT.
SELECT SINGLE TXT50 INTO GT_ITEM-TXT50
  FROM SKAT
 WHERE SPRAS EQ SY-LANGU
   AND KTOPL EQ T001-KTOPL
   AND SAKNR EQ GT_ITEM-HKONT.

* 총금액에 따라 전기키 수정 및 금액 수정
IF IS_INVOICE-GRANDTOTAL > 0 .
  GT_ITEM-BSCHL    = '40'.
ELSE.
  GT_ITEM-BSCHL    = '50'.
  IS_INVOICE-CHARGETOTAL = IS_INVOICE-CHARGETOTAL * ( -1 ).
ENDIF.
GT_ITEM-DMBTR = ABS( IS_INVOICE-TAXTOTAL ).
WRITE IS_INVOICE-CHARGETOTAL TO GT_ITEM-FWBAS CURRENCY T001-WAERS.
