FORM GET_ZDTV3T_AR_HEAD .
  CLEAR: IT_WHERE, IT_WHERE[], GV_FIRST,
         IT_AR_HEAD,  IT_AR_HEAD[].

* 문서상태
  PERFORM GET_STATUS_VALUES.

  SELECT I~BUKRS  I~ISSUE_DATE  I~BUPLA  I~INV_SEQ
         I~ASP_CODE  I~IP_ID  I~IP_NAME  I~IP_TYPECODE I~TYPE_CODE
         I~CHARGETOTAL  I~TAXTOTAL  I~GRANDTOTAL
         I~ISSUE_ID  I~STATUS I~ISSUE_DT  I~NTS_SEND_FLAG

FORM GET_FILEDCAT_ALV USING GT_FIELDCAT TYPE LVC_T_FCAT.

  DATA : LS_FIELDCAT TYPE LVC_S_FCAT.
  DATA : LT_FIELDCAT TYPE SLIS_T_FIELDCAT_ALV,
         L_FIELDCAT  TYPE SLIS_FIELDCAT_ALV.
  CLEAR: GT_FIELDCAT, GT_FIELDCAT[].

  CALL FUNCTION 'REUSE_ALV_FIELDCATALOG_MERGE'
    EXPORTING
      I_PROGRAM_NAME     = G_REPID_C
      I_INTERNAL_TABNAME = 'IT_ALV'
      I_BYPASSING_BUFFER = 'X'
      I_INCLNAME         = G_REPID_C
    CHANGING
      CT_FIELDCAT        = LT_FIELDCAT[].

  LOOP AT LT_FIELDCAT INTO L_FIELDCAT.
    CASE L_FIELDCAT-FIELDNAME.
      WHEN 'INVDOC'.
        L_FIELDCAT-EMPHASIZE = 'C100'.
      WHEN 'IP_TYPTXT'.
        L_FIELDCAT-SELTEXT_S = '구분'.
      WHEN 'ASP_CODE' OR 'ASP_CODE_TEXT'.
        L_FIELDCAT-EMPHASIZE = 'C400'.
      WHEN 'ISSUE_DT'.
        L_FIELDCAT-JUST = 'L'.
      WHEN 'NTS_SEND_FLAG'.
        L_FIELDCAT-EMPHASIZE = 'C700'.
      WHEN 'NTS_SEND_TEXT'.
        L_FIELDCAT-SELTEXT_S = 'Text'.
        L_FIELDCAT-EMPHASIZE = 'C700'.
      WHEN 'ERROR_CODE' OR 'ERROR_CODE_TEXT'.
        L_FIELDCAT-EMPHASIZE = 'C700'.
    ENDCASE.
*
    MOVE-CORRESPONDING L_FIELDCAT TO LS_FIELDCAT.
    LS_FIELDCAT-REPTEXT   = L_FIELDCAT-SELTEXT_S.
    LS_FIELDCAT-REF_TABLE = L_FIELDCAT-REF_TABNAME.

*   통화단위.
    IF LS_FIELDCAT-DATATYPE EQ 'CURR'.
      LS_FIELDCAT-CURRENCY  = T001-WAERS.
      LS_FIELDCAT-DO_SUM    = 'X'.

                                           LS_FIELDCAT-HOTSPOT
                                           ''
                                           'O'
                                           'AR'.

    CLEAR  LS_FIELDCAT.
  ENDLOOP.

ENDFORM. " GET_FILEDCAT_ALV

FORM FILL_IT_ALV .
  CLEAR: IT_ALV, IT_ALV[].

  SORT: IT_AR_HEAD,
        IT_AR_EXT,
        IT_ASP_CODE,
        IT_STATUS,
        IT_DD07V,
        IT_TAX_STAT BY BUKRS ISSUE_DATE BUPLA INV_SEQ ASCENDING
                       STAT_SEQ DESCENDING.


  LOOP AT IT_AR_HEAD.
    MOVE-CORRESPONDING IT_AR_HEAD TO IT_ALV.
*.. 발행일자
    CONCATENATE IT_AR_HEAD-ISSUE_DT+0(4)  '.' IT_AR_HEAD-ISSUE_DT+4(2)
    '.' IT_AR_HEAD-ISSUE_DT+6(2) INTO IT_ALV-ISSUE_DT.

*.. 고객코드
    READ TABLE IT_AR_EXT WITH KEY BUKRS      = IT_AR_HEAD-BUKRS
                                  ISSUE_DATE = IT_AR_HEAD-ISSUE_DATE
                                  BUPLA      = IT_AR_HEAD-BUPLA
                                  INV_SEQ    = IT_AR_HEAD-INV_SEQ
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-INVDOC = IT_AR_EXT-INVDOC.
      IT_ALV-KUNNR  = IT_AR_EXT-KUNNR.
    ELSE.
      CONTINUE.
    ENDIF.

*.. 사업자/개인구분
    CASE IT_AR_HEAD-IP_TYPECODE.
      WHEN '01'.
        IT_ALV-IP_TYPTXT = '사업자 발행분'.
      WHEN '02' OR '03'.
        IT_ALV-IP_TYPTXT = '개인 발행분(외국인포함)'.
    ENDCASE.

*.. ASP 코드 TEXT
    READ TABLE IT_ASP_CODE WITH KEY ASP_CODE = IT_AR_HEAD-ASP_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-ASP_CODE_TEXT = IT_ASP_CODE-ASP_CODE_TEXT.
    ENDIF.

*.. 문서상태 TEXT
    READ TABLE IT_STATUS WITH KEY STATUS = IT_AR_HEAD-STATUS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-STATUS_TEXT = IT_STATUS-STATUS_TEXT.
    ENDIF.

*.. 국세청 전송상태 TEXT
    READ TABLE IT_DD07V WITH KEY DOMVALUE_L = IT_AR_HEAD-NTS_SEND_FLAG
                                 BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-NTS_SEND_TEXT = IT_DD07V-DDTEXT.
    ENDIF.

*.. 국세청 전송오류
    READ TABLE IT_TAX_STAT WITH KEY BUKRS      = IT_AR_HEAD-BUKRS
                                    ISSUE_DATE = IT_AR_HEAD-ISSUE_DATE
                                    BUPLA      = IT_AR_HEAD-BUPLA
                                    INV_SEQ    = IT_AR_HEAD-INV_SEQ
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-ERROR_CODE      = IT_TAX_STAT-ERROR_CODE.
      IT_ALV-ERROR_CODE_TEXT = IT_TAX_STAT-ERROR_CODE_TEXT.
    ENDIF.
*
    APPEND IT_ALV.
    CLEAR  IT_ALV.
  ENDLOOP.

ENDFORM. " FILL_IT_ALV

FORM GET_DOC_STAT USING P_STAT.
  DATA: VALUE(30), L_CNT TYPE I.

  LOOP AT IT_GROUP_FT WHERE DOC_STAT = P_STAT.

      CONCATENATE ' ( I~STATUS EQ ' VALUE

        CONCATENATE 'OR' IT_WHERE

      ENDIF.
      APPEND IT_WHERE.
      CONCATENATE  `'` IT_FUNCTION-PROC_STEP `'` INTO VALUE.
      CONCATENATE 'AND I~PROC_STEP EQ ' VALUE ')'
            INTO IT_WHERE SEPARATED BY SPACE.
      APPEND IT_WHERE.
    ENDLOOP.
