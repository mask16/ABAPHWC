FUNCTION ZDTV3SAP_CHK_AP_CLOSED.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(I_BUKRS) TYPE BUKRS OPTIONAL
*" EXPORTING
*" VALUE(O_CLOSED) TYPE FLAG
*" TABLES
*" T_DATUM STRUCTURE RANGE_DATE OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LS_CLOSED_AP LIKE ZDTV3T_CLOSED_AP.
  DATA : L_DATUM TYPE DATUM.
*
  SELECT SINGLE *
    INTO LS_CLOSED_AP
    FROM ZDTV3T_CLOSED_AP
   WHERE BUKRS EQ I_BUKRS.
*
  CHECK SY-SUBRC EQ 0.
*
  IF LS_CLOSED_AP-FRDAT > LS_CLOSED_AP-TODAT.
    EXIT.
  ENDIF.
*
  L_DATUM = LS_CLOSED_AP-FRDAT.
  DO 1000000 TIMES.
    IF L_DATUM IN T_DATUM.
      O_CLOSED = 'X'.
      EXIT.
    ENDIF.
    IF L_DATUM EQ LS_CLOSED_AP-TODAT.
      EXIT.
    ENDIF.
    ADD 1 TO L_DATUM.
  ENDDO.
*
ENDFUNCTION.
