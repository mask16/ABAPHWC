*

  IXML = CL_IXML=>CREATE( ).

* create a stream factory
  STREAMFACTORY = IXML->CREATE_STREAM_FACTORY( ).

* create a input stream
  ISTREAM  = STREAMFACTORY->CREATE_ISTREAM_STRING( STRING = XMLDATA ).

* create a ixml document
  DOCUMENT = IXML->CREATE_DOCUMENT( ).

* create a xml parser
  PARSER  = IXML->CREATE_PARSER( DOCUMENT       = DOCUMENT
                                 STREAM_FACTORY = STREAMFACTORY
                                 ISTREAM        = ISTREAM ).

  IF PARSER->PARSE( ) <> 0.
    EXIT.
  ENDIF.

* print DOM tree
  NODE ?= DOCUMENT.

        NODEMAP   TYPE REF TO IF_IXML_NAMED_NODE_MAP,

        PARENT1   TYPE STRING,
        PARENT2   TYPE STRING,
        PREFIX    TYPE STRING,
        VALUE     TYPE STRING,
        INDENT    TYPE I,
        COUNT     TYPE I,
        INDEX     TYPE I.

  CHECK NOT NODE IS INITIAL.

* create a node iterator
  ITERATOR  = NODE->CREATE_ITERATOR( ).
* get current node
  NODE = ITERATOR->GET_NEXT( ).
* loop over all nodes
  WHILE NOT NODE IS INITIAL.
    ATTR2    = NODE->GET_PARENT( ) .
    IF NOT ATTR2 IS INITIAL.
      PARENT2    = ATTR2->GET_NAME( ).
      ATTR1    = ATTR2->GET_PARENT( ) .
      IF NOT ATTR1 IS INITIAL.
        PARENT1    = ATTR1->GET_NAME( ).
      ENDIF.
    ENDIF.
*
    NAME    = NODE->GET_NAME( ).
    VALUE   = NODE->GET_VALUE( ).
*
*    IF     PARENT1 EQ 'TaxInvoice'
*       AND PARENT2 EQ 'ExchangedDocument'

*    ELSEIF PARENT_NAME EQ '' AND NAME EQ ''.

    ELSEIF PARENT1 EQ 'TaxInvoice'
