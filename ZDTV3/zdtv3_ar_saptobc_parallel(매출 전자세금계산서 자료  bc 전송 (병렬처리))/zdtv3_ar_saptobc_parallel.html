FORM GET_DATA .

  SELECT *
         INTO CORRESPONDING FIELDS OF TABLE GT_HD
         FROM ZDTV3T_AR_HEAD UP TO P_CNT ROWS
        WHERE BUKRS      EQ P_BUKRS
          AND STATUS     IN ('0000','1110')
          AND PROC_STEP  IN ('','00','11','12')
*          AND ISSUE_DATE EQ  P_DATUM
          AND ISSUE_DATE IN S_DATUM
*          AND ISSUE_DATE IN R_ERDAT_DEL
          AND INV_SEQ    IN S_INVSEQ.

  SORT GT_HD BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

  SELECT *
       INTO CORRESPONDING FIELDS OF TABLE GT_IM
*        FROM ZDTV3T_AR_ITEM FOR ALL ENTRIES IN GT_HD
       FROM ZDTV3T_AR_ITEM
      WHERE BUKRS      EQ P_BUKRS
*        AND ISSUE_DATE EQ P_DATUM
        AND ISSUE_DATE IN S_DATUM
*         AND BUPLA      EQ GT_HD-BUPLA
        AND INV_SEQ    IN S_INVSEQ.

  SORT GT_IM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

  LOOP AT GT_IM.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_IM-BUKRS
                              ISSUE_DATE = GT_IM-ISSUE_DATE
                              BUPLA      = GT_IM-BUPLA
                              INV_SEQ    = GT_IM-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC NE 0.
      DELETE GT_IM.
    ENDIF.
  ENDLOOP.

  SELECT *
       INTO CORRESPONDING FIELDS OF TABLE GT_PM
*        FROM ZDTV3T_AR_PAYM FOR ALL ENTRIES IN GT_HD
       FROM ZDTV3T_AR_PAYM
      WHERE BUKRS      EQ P_BUKRS
*        AND ISSUE_DATE EQ P_DATUM
        AND ISSUE_DATE IN S_DATUM
*         AND BUPLA      EQ GT_HD-BUPLA
        AND INV_SEQ    IN S_INVSEQ.

  SORT GT_PM BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

  LOOP AT GT_PM.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_PM-BUKRS
                              ISSUE_DATE = GT_PM-ISSUE_DATE
                              BUPLA      = GT_PM-BUPLA
                              INV_SEQ    = GT_PM-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC NE 0.
      DELETE GT_PM.
    ENDIF.
  ENDLOOP.

  SELECT *
       INTO CORRESPONDING FIELDS OF TABLE GT_EX
*        FROM ZDTV3T_AR_EXT FOR ALL ENTRIES IN GT_HD
       FROM ZDTV3T_AR_EXT
      WHERE BUKRS      EQ P_BUKRS
*        AND ISSUE_DATE EQ P_DATUM
        AND ISSUE_DATE IN S_DATUM
*         AND BUPLA      EQ GT_HD-BUPLA
        AND INV_SEQ    IN S_INVSEQ.

  SORT GT_EX BY BUKRS ISSUE_DATE BUPLA INV_SEQ.

  LOOP AT GT_EX.
    READ TABLE GT_HD WITH KEY BUKRS      = GT_EX-BUKRS
                              ISSUE_DATE = GT_EX-ISSUE_DATE
                              BUPLA      = GT_EX-BUPLA
                              INV_SEQ    = GT_EX-INV_SEQ.
    IF SY-SUBRC NE 0.
      DELETE GT_EX.
    ENDIF.
  ENDLOOP.

        INTO CORRESPONDING FIELDS OF TABLE GT_EX_D
*        FROM ZDTV3T_AR_EXT_D FOR ALL ENTRIES IN GT_HD
        FROM ZDTV3T_AR_EXT_D
       WHERE BUKRS      EQ GT_HD-BUKRS
*         AND ISSUE_DATE EQ P_DATUM
         AND ISSUE_DATE IN S_DATUM
*         AND BUPLA      EQ GT_HD-BUPLA
         AND INV_SEQ    IN S_INVSEQ.

FORM SEND_DATA_PARALLEL .

  DATA : LT_HD   LIKE ZDTV3T_AR_HEAD  OCCURS 0 WITH HEADER LINE ,
         LT_IM   LIKE ZDTV3T_AR_ITEM  OCCURS 0 WITH HEADER LINE ,
         LT_PM   LIKE ZDTV3T_AR_PAYM  OCCURS 0 WITH HEADER LINE ,
         LT_ST   LIKE ZDTV3T_AR_STAT  OCCURS 0 WITH HEADER LINE ,
         LT_EX   LIKE ZDTV3T_AR_EXT   OCCURS 0 WITH HEADER LINE ,
         LT_EX_D LIKE ZDTV3T_AR_EXT_D OCCURS 0 WITH HEADER LINE ,
         LT_RETURN LIKE ZDTV3S_RETURN OCCURS 0 WITH HEADER LINE.
  DATA : LV_CNT  TYPE I.

  SORT GT_IM BY INV_SEQ.
  SORT GT_PM BY INV_SEQ.
  SORT GT_EX BY INV_SEQ.
  SORT GT_EX_D BY INV_SEQ.

  LOOP AT GT_HD.
    LV_CNT = LV_CNT + 1.

    MOVE-CORRESPONDING GT_HD TO LT_HD.
    APPEND LT_HD. CLEAR LT_HD.

    READ TABLE GT_IM WITH KEY INV_SEQ = GT_HD-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      LOOP AT GT_IM FROM SY-TABIX.
        IF GT_IM-INV_SEQ NE GT_HD-INV_SEQ.
          EXIT.
        ENDIF.
        MOVE-CORRESPONDING GT_IM TO LT_IM.
        APPEND LT_IM. CLEAR LT_IM.
      ENDLOOP.
    ENDIF.

    READ TABLE GT_PM WITH KEY INV_SEQ = GT_HD-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      LOOP AT GT_PM FROM SY-TABIX.
        IF GT_PM-INV_SEQ NE GT_HD-INV_SEQ.
          EXIT.
        ENDIF.
        MOVE-CORRESPONDING GT_PM TO LT_PM.
        APPEND LT_PM. CLEAR LT_PM.
      ENDLOOP.
    ENDIF.

    READ TABLE GT_EX WITH KEY INV_SEQ = GT_HD-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      LOOP AT GT_EX FROM SY-TABIX.
        IF GT_EX-INV_SEQ NE GT_HD-INV_SEQ.
          EXIT.
        ENDIF.
        MOVE-CORRESPONDING GT_EX TO LT_EX.
        APPEND LT_EX. CLEAR LT_EX.
      ENDLOOP.
    ENDIF.

    READ TABLE GT_EX_D WITH KEY INV_SEQ = GT_HD-INV_SEQ
                              BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      LOOP AT GT_EX_D FROM SY-TABIX.
        IF GT_EX_D-INV_SEQ NE GT_HD-INV_SEQ.
          EXIT.
        ENDIF.
        MOVE-CORRESPONDING GT_EX_D TO LT_EX_D.
        APPEND LT_EX_D. CLEAR LT_EX_D.
      ENDLOOP.
    ENDIF.

    IF LV_CNT >= P_CNTP.
      CLEAR LV_CNT.
      PERFORM PARALLEL_PROCESSING
       TABLES LT_HD LT_IM LT_PM LT_EX LT_EX_D LT_RETURN.
    ENDIF.

  ENDLOOP.

  IF LV_CNT > 0.
    PERFORM PARALLEL_PROCESSING
       TABLES LT_HD LT_IM LT_PM LT_EX LT_EX_D LT_RETURN.
  ENDIF.

  WAIT UNTIL G_RECVJOBS >= G_SENDJOBS.

ENDFORM. " SEND_DATA_PARALLEL

FORM PARALLEL_PROCESSING TABLES PT_HD STRUCTURE ZDTV3T_AR_HEAD
                                    PT_IM STRUCTURE ZDTV3T_AR_ITEM
                                    PT_PM STRUCTURE ZDTV3T_AR_PAYM

      STARTING NEW TASK G_TASKNAME
      DESTINATION IN GROUP DEFAULT
      PERFORMING RETURN_RFC_INFO ON END OF TASK
      EXPORTING
        STATUS                = '1110'
        DDEMODE               = 'M'     "S:개별처리,M:대량처리

      EXCEPTIONS

        SYSTEM_FAILURE        = 2
        RESOURCE_FAILURE      = 3
        OTHERS                = 4.

        G_SENDJOBS = G_SENDJOBS + 1.
        G_RUNNING_ARFC_WPS = G_RUNNING_ARFC_WPS + 1.

        ENDIF.
    ENDCASE.
  ENDDO.
