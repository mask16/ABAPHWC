FORM save_data .

  DATA : l_cnt TYPE i.
  DATA : lt_main LIKE it_main OCCURS 0 WITH HEADER LINE.

  lt_main[] = it_main[].

  DELETE lt_main WHERE lights NE '3'.

  l_cnt = LINES( lt_main[] ).
  IF l_cnt > 1 .
    MODIFY zdtv3t_nts  FROM TABLE it_main.
    IF sy-subrc EQ 0.
      MESSAGE s304 WITH l_cnt.
    ELSE.
      MESSAGE e305.
    ENDIF.
  ELSE.
    MESSAGE s303 .
  ENDIF.
ENDFORM. " SAVE_DATA
