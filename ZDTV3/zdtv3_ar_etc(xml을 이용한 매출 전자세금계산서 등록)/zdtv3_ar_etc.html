FORM GET_ZDTV3T_ETC .

  SELECT BUKRS  BUPLA  ISSUE_ID  ISSUE_DATE
         ASP_CODE  TYPE_CODE  PURP_CODE  AMEND_CODE

FORM MODIFY_IT_DATA .

  SORT: IT_DATA,
        IT_ASP_CODE,
        IT_TYPECD_T,
        IT_AMNDCD_T.

  LOOP AT IT_DATA.
*.. ASP 코드 TEXT
    READ TABLE IT_ASP_CODE WITH KEY ASP_CODE = IT_DATA-ASP_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_DATA-ASP_CODE_TEXT = IT_ASP_CODE-ASP_CODE_TEXT.
    ENDIF.

*.. 세금계산서 종류 TEXT
    READ TABLE IT_TYPECD_T WITH KEY DOMVALUE_L = IT_DATA-TYPE_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_DATA-TYPE_CODE_TEXT = IT_TYPECD_T-DDTEXT.
    ENDIF.

*.. 영수/청구
    IF IT_DATA-PURP_CODE = '01'.
      IT_DATA-PURP_CODE_TEXT = '영수'.
    ELSE.
      IT_DATA-PURP_CODE_TEXT = '청구'.
    ENDIF.

*.. 수정사유 TEXT
    READ TABLE IT_AMNDCD_T WITH KEY DOMVALUE_L = IT_DATA-AMEND_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_DATA-AMEND_CODE_TEXT = IT_AMNDCD_T-DDTEXT.
    ENDIF.

*.. 국세청 신고여부
    IF IT_DATA-NTS_SEND_FLAG = '31'.
      IT_DATA-NTS_SEND_YN = 'Y'.
    ELSE.
      IT_DATA-NTS_SEND_YN = 'N'.
    ENDIF.
*
    MODIFY IT_DATA.
    CLEAR  IT_DATA.
  ENDLOOP.

ENDFORM. " MODIFY_IT_DATA

MODULE USER_COMMAND_0100 INPUT.

  CHECK OK_CODE_C NE ''.

  RANGES : R_ISSUE_ID FOR ZDTV3T_AR_HEAD-ISSUE_ID.
  DATA : L_BUTTON TYPE ZDTV3E_BUTTON.
  DATA : GS_RETURN LIKE ZDTV3S_RETURN.
  DATA : GT_RETURN LIKE ZDTV3S_RETURN OCCURS 0 WITH HEADER LINE.

  DATA : LT_ROWS TYPE LVC_T_ROW ,
         LS_ROWS TYPE LVC_S_ROW .
  DATA : LV_ANSWER.

  CALL METHOD G_GRID->CHECK_CHANGED_DATA.
  CALL METHOD G_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROWS.

  IF LT_ROWS[] IS INITIAL.
    MESSAGE I015.
    EXIT.
  ENDIF.


  CASE OK_CODE_C.
    WHEN 'TVIEW'.                                " 세금계산서 조회
      DESCRIBE TABLE LT_ROWS LINES SY-TFILL.
      IF SY-TFILL > 1.
        MESSAGE I020.
        EXIT.
      ENDIF.
      LOOP AT LT_ROWS INTO LS_ROWS.
        READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
        IF SY-SUBRC EQ 0.
          PERFORM CALL_EXECUTE(ZDTV3_UPLOAD_XML) USING IS_DATA-DATA.
        ENDIF.
      ENDLOOP.

    WHEN 'NTSFLG'.                               " 국세청신고 등록
      PERFORM POPUP_TO_CONFIRM  CHANGING LV_ANSWER.
      IF LV_ANSWER = '1'.

        LOOP AT LT_ROWS INTO LS_ROWS.
          READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
          IF SY-SUBRC = 0.
            IS_DATA-NTS_SEND_FLAG = '31'.
            IS_DATA-NTS_SEND_YN   = 'Y'.
            MODIFY IT_DATA FROM IS_DATA INDEX LS_ROWS-INDEX.
*
            UPDATE ZDTV3T_ETC   SET NTS_SEND_FLAG = '31'
                              WHERE ISSUE_ID = IS_DATA-ISSUE_ID.
            COMMIT WORK.
          ENDIF.
        ENDLOOP.
        MESSAGE S000 WITH TEXT-M01.

      ELSE.
        MESSAGE S006.
      ENDIF.

    WHEN 'DELFLG'.                               " 삭제
      PERFORM POPUP_TO_MESSAGE_C  USING    'N'
                                           TEXT-T10
                                           TEXT-T11
                                           TEXT-T12
                                           ''
                                  CHANGING LV_ANSWER.
      IF LV_ANSWER = 'J'.

        LOOP AT LT_ROWS INTO LS_ROWS.
          READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
          IF SY-SUBRC = 0.
            UPDATE ZDTV3T_ETC   SET DATA_DEL = 'X'
                                    DEL_ERDAT = SY-DATUM
                                    DEL_ERZET = SY-UZEIT
                                    DEL_ERNAM = SY-UNAME
                              WHERE ISSUE_ID = IS_DATA-ISSUE_ID.
            COMMIT WORK.
*
            DELETE IT_DATA INDEX LS_ROWS-INDEX.
          ENDIF.
        ENDLOOP.
        MESSAGE S000 WITH TEXT-M02.

      ELSE.
        MESSAGE S006.
      ENDIF.

    WHEN 'P_DOCUMAP'.                            "전표등록

      L_BUTTON = OK_CODE_C.

      PERFORM AUTHORITY_CHECK(ZDTV3_AR) USING L_BUTTON G_ERROR_C .
      CHECK G_ERROR_C EQ '0'.

      DESCRIBE TABLE LT_ROWS LINES SY-TFILL.
      IF SY-TFILL > 1.
        MESSAGE I020.
        EXIT.
      ENDIF.

      LOOP AT LT_ROWS INTO LS_ROWS.
        READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
        IF SY-SUBRC EQ 0.
          CLEAR : R_ISSUE_ID[].
          R_ISSUE_ID-SIGN   = 'I'.
          R_ISSUE_ID-OPTION = 'EQ'.
          R_ISSUE_ID-LOW    = IS_DATA-ISSUE_ID.
          APPEND R_ISSUE_ID.
        ENDIF.
      ENDLOOP.

      call function 'ZDTV3SAP_DSP_POPUP'
        EXPORTING
          BUTTON         = L_BUTTON
          CPROG          = 'ZDTV3_AR'
          ETC_FLAG       = 'X'
        IMPORTING
          RETURN         = GS_RETURN
        TABLES
          POPUP_ISSUE_ID = R_ISSUE_ID.

    WHEN 'P_DOCUCRD'.                            "전표생성
      L_BUTTON = OK_CODE_C.

*      버튼별 권한 확인.
      PERFORM AUTHORITY_CHECK(ZDTV3_AR) USING L_BUTTON G_ERROR_C .
      CHECK G_ERROR_C EQ '0'.

      DESCRIBE TABLE LT_ROWS LINES SY-TFILL.
      IF SY-TFILL > 1.
        MESSAGE I020.
        EXIT.
      ENDIF.
