REPORT ZDTV3_AP02 MESSAGE-ID ZDTV3M.

include ZDTV3_ALV.
TABLES : ZDTV3T_USER_VEND .


CONSTANTS : C_TEXT(10) VALUE '*LIFNR' .

*******************************************************************
* START-OF-SELECTION.
*******************************************************************
START-OF-SELECTION.

* 검색조건읽기.
  PERFORM GET_SELECTION_INFO TABLES GT_RSELINFO
                             USING 'ZDTV3_AP'.
*
  LOOP AT GT_RSELINFO.
    IF GT_RSELINFO-DBFIELD CP C_TEXT.
      EXIT.
    ENDIF.
  ENDLOOP.
  IF SY-SUBRC NE 0.
   MESSAGE E000 WITH '매출관리프로그램에 고객코드가 존재하지 않습니다.'.
  ENDIF.

  CLEAR : GT_RAPARAM , GT_RAPARAM[].
  SELECT *
    FROM ZDTV3T_USER_VEND
   WHERE BNAME = SY-UNAME.
    GT_RAPARAM-SELNAME = GT_RSELINFO-NAME.
    GT_RAPARAM-KIND    = GT_RSELINFO-KIND.
    GT_RAPARAM-SIGN    = 'I'.
    GT_RAPARAM-OPTION  = 'EQ'.
    GT_RAPARAM-LOW     = ZDTV3T_USER_VEND-LIFNR.
    GT_RAPARAM-HIGH    = ''.
    APPEND GT_RAPARAM.
  ENDSELECT.
  IF SY-SUBRC NE 0.
    MESSAGE E403 WITH SY-UNAME.
  ENDIF.

* 관리프로그램 호출.
  SUBMIT ZDTV3_AP VIA SELECTION-SCREEN
                  WITH SELECTION-TABLE GT_RAPARAM.
