  CLEAR : IT_MAIN , IT_MAIN[].

FORM MAKE_SELECT .
  DATA L_CNT(4) TYPE N.
  CLEAR : IT_DATA[].
*  IF IT_DATA[] IS INITIAL.
  IS_DATA = TEXT-T90.
  APPEND IS_DATA TO IT_DATA.
  IS_DATA = TEXT-T91.
  APPEND IS_DATA TO IT_DATA.
  CLEAR : L_CNT.
  SELECT *
    FROM ZDTV3T_F4_SER
   WHERE ZPROGRAM   EQ IS_KEY-ZPROGRAM
     AND BUTTON     EQ IS_KEY-BUTTON
     AND FIELDNAME  EQ IS_KEY-FIELDNAME
     AND XNODI      EQ SPACE
   ORDER BY DYNNR F4_TABLE SSEQ.
    ADD 1 TO L_CNT.
    CONCATENATE ZDTV3T_F4_SER-F4_FIELD '=&'
      ZDTV3T_F4_SER-DYNNR L_CNT INTO IS_DATA.
    CASE ZDTV3T_F4_SER-KIND.
      WHEN 'P'.
        CONCATENATE '* P=>' ZDTV3T_F4_SER-F4_TABLE '-' IS_DATA
               INTO IS_DATA.
      WHEN 'S'.
        CONCATENATE '* S=>' ZDTV3T_F4_SER-F4_TABLE '-' IS_DATA
               INTO IS_DATA.
    ENDCASE.
    APPEND IS_DATA TO IT_DATA.
  ENDSELECT.
  IS_DATA = TEXT-T90.
  APPEND IS_DATA TO IT_DATA.
*  ENDIF.
ENDFORM. " MAKE_SELECT

FORM DEFINE_TABLE .

                                            SEPARATED BY SPACE.

  ENDSELECT.

FORM SEL_SCR .
  DATA : L_TCNT(3) TYPE N , L_SCNT(3) TYPE N ,
          L_DCNT(4) TYPE N .
  DATA : NAME(8).
  DATA : LS_SER_M LIKE IS_SER_M.
  DATA : L_TMP LIKE DD03L.
*
  CLEAR : IT_SER_M , IT_SER_M[].
* 검색조건 만들기.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_SER_M
    FROM ZDTV3T_F4_SER
   WHERE F4_TABLE   NE SPACE
     AND F4_FIELD   NE SPACE
     AND XNODI      EQ SPACE.

* TEXT ELEMENT 읽기.
  READ TEXTPOOL C_PROG INTO IT_TEXT LANGUAGE SY-LANGU.
* 정렬
  SORT IT_SER_M BY DYNNR F4_TABLE SSEQ.

  CLEAR : L_DCNT , L_TCNT , L_SCNT.
  LOOP AT IT_SER_M INTO IS_SER_M.
**
    MOVE-CORRESPONDING IS_SER_M TO LS_SER_M.
*
    AT NEW DYNNR.
      CLEAR : L_DCNT.
*     SELECTION-SCREEN 생성.
      CONCATENATE 'SELECTION-SCREEN BEGIN OF SCREEN' LS_SER_M-DYNNR
      'AS SUBSCREEN.' INTO IT_SUR-LINE SEPARATED BY SPACE.
      APPEND IT_SUR.
    ENDAT.

    AT NEW F4_TABLE.
      ADD 1 TO L_TCNT.
      CONCATENATE 'SELECTION-SCREEN BEGIN OF BLOCK BT' L_TCNT
      ' WITH FRAME TITLE TEXT-' L_TCNT '.'
       INTO IT_SUR-LINE.
      APPEND IT_SUR.
*     ELECTION-SCREEN BLOCK 텍스트...
      SELECT SINGLE * FROM DD02T
       WHERE TABNAME    EQ LS_SER_M-F4_TABLE
         AND DDLANGUAGE EQ SY-LANGU.
      D_TEXT 'I' L_TCNT DD02T-DDTEXT.
    ENDAT.

*   PARAM/SELECT 생성.
    ADD 1 TO L_DCNT.
    CONCATENATE '&' LS_SER_M-DYNNR L_DCNT INTO NAME.
    D_SER IS_SER_M-KIND IS_SER_M-F4_FIELD IS_SER_M-F4_TABLE
                     IS_SER_M-XNODI IS_SER_M-XOBLG IS_SER_M-XOPTN NAME
                     IS_SER_M-CR_VALUE.
*
    DATA : L_TEXT(38).
    IF NOT LS_SER_M-F4_FIELDTEXT IS INITIAL.
      L_TEXT+8(30) = LS_SER_M-F4_FIELDTEXT.
    ELSE.
      SELECT SINGLE B~DDTEXT INTO L_TEXT
        FROM DD03L AS A INNER JOIN DD04T AS B
          ON A~ROLLNAME = B~ROLLNAME
         AND B~DDLANGUAGE = SY-LANGU
       WHERE A~TABNAME    EQ IS_SER_M-F4_TABLE
         AND B~DDLANGUAGE EQ SY-LANGU
         AND A~FIELDNAME  EQ IS_SER_M-F4_FIELD.
      CALL FUNCTION 'CUT_2BYTES_STRINGS'
        EXPORTING
          I_STR = L_TEXT
          I_LEN = '30'
        IMPORTING
          O_STR = L_TEXT+8(30).
    ENDIF.
*   ELEMENT TEXT 생성..
    D_TEXT 'S' NAME L_TEXT.

*   SELECTION-SCREEN END...
    AT END OF F4_TABLE.
      CONCATENATE 'SELECTION-SCREEN END   OF BLOCK BT' L_TCNT '.'
      INTO IT_SUR-LINE.
      APPEND IT_SUR.
    ENDAT.

*   SELECTION-SCREEN END...
    AT END OF DYNNR.
      CONCATENATE 'SELECTION-SCREEN END OF SCREEN' IS_SER_M-DYNNR '.'
      INTO IT_SUR-LINE SEPARATED BY SPACE.
      APPEND IT_SUR.
    ENDAT.
  ENDLOOP.


ENDFORM. " SEL_SCR
