FORM GET_ZDTV3T_AR_DATA .
  DATA : L_LINE(60),
         LT_CONDITION  LIKE TABLE OF L_LINE WITH HEADER LINE.
  CLEAR: IT_AR_DATA, IT_AR_DATA[], IT_AR_DATA2, IT_AR_DATA2[].


  CASE 'X'.
    WHEN P_DISP1.                          " 원본 세금계산서 기준
*//   HEADER
      SELECT I~BUKRS  I~ISSUE_DATE  I~BUPLA  I~INV_SEQ
             I~TYPE_CODE  I~AMEND_CODE
             I~CHARGETOTAL  I~TAXTOTAL  I~GRANDTOTAL
             I~ISSUE_ID  I~STATUS  I~NTS_SEND_FLAG
             H~INVDOC H~INV_SEQ_C
        INTO TABLE IT_AR_DATA
        FROM ZDTV3T_AR_HEAD AS I INNER JOIN ZDTV3T_AR_EXT AS H
                                    ON I~BUKRS      = H~BUKRS AND
                                       I~ISSUE_DATE = H~ISSUE_DATE AND
                                       I~BUPLA      = H~BUPLA      AND
                                       I~INV_SEQ    = H~INV_SEQ
       WHERE I~BUKRS      EQ P_BUKRS
         AND I~ISSUE_DATE IN S_ISSDAT
         AND I~BUPLA      IN S_BUPLA
         AND I~INV_SEQ IN ( SELECT INV_SEQ_C FROM ZDTV3T_AR_EXT
                             WHERE BUKRS      EQ P_BUKRS
*                              AND ISSUE_DATE IN S_ISSDAT
                               AND BUPLA      IN S_BUPLA
                               AND INV_SEQ_C  NE SPACE ).
      CHECK SY-SUBRC = 0.

*//   ITEM
      LOOP AT IT_AR_DATA.
        SELECT I~BUKRS  I~ISSUE_DATE  I~BUPLA  I~INV_SEQ
               I~TYPE_CODE  I~AMEND_CODE
               I~CHARGETOTAL  I~TAXTOTAL  I~GRANDTOTAL
               I~ISSUE_ID  I~STATUS  I~NTS_SEND_FLAG
               H~INVDOC H~INV_SEQ_C
          APPENDING TABLE IT_AR_DATA2
          FROM ZDTV3T_AR_HEAD AS I INNER JOIN ZDTV3T_AR_EXT AS H
                                      ON I~BUKRS      = H~BUKRS AND
                                         I~ISSUE_DATE = H~ISSUE_DATE AND
                                         I~BUPLA      = H~BUPLA      AND
                                         I~INV_SEQ    = H~INV_SEQ
         WHERE I~BUKRS   EQ P_BUKRS
           AND I~BUPLA   IN S_BUPLA
           AND I~INV_SEQ IN ( SELECT INV_SEQ FROM ZDTV3T_AR_EXT
                               WHERE BUKRS     = IT_AR_DATA-BUKRS
                                 AND BUPLA     = IT_AR_DATA-BUPLA
                                 AND INV_SEQ_C = IT_AR_DATA-INV_SEQ )
           AND I~AMEND_CODE IN R_AMDCOD.
      ENDLOOP.

*-----
    WHEN P_DISP2.                          " 수정 세금계산서 기준
      SELECT I~BUKRS  I~ISSUE_DATE  I~BUPLA  I~INV_SEQ
             I~TYPE_CODE  I~AMEND_CODE
             I~CHARGETOTAL  I~TAXTOTAL  I~GRANDTOTAL
             I~ISSUE_ID  I~STATUS  I~NTS_SEND_FLAG
             H~INVDOC H~INV_SEQ_C
        INTO TABLE IT_AR_DATA
        FROM ZDTV3T_AR_HEAD AS I INNER JOIN ZDTV3T_AR_EXT AS H
                                    ON I~BUKRS      = H~BUKRS AND
                                       I~ISSUE_DATE = H~ISSUE_DATE AND
                                       I~BUPLA      = H~BUPLA      AND
                                       I~INV_SEQ    = H~INV_SEQ
       WHERE I~BUKRS      EQ P_BUKRS
         AND I~ISSUE_DATE IN S_ISSDAT
         AND I~BUPLA      IN S_BUPLA
         AND I~TYPE_CODE  LIKE '02%'
         AND I~AMEND_CODE IN R_AMDCOD.
  ENDCASE.

ENDFORM. " GET_ZDTV3T_AR_DATA

FORM FILL_IT_ALV .
  CLEAR: IT_ALV, IT_ALV[].

  SORT: IT_AR_DATA,
        IT_STATUS,
        IT_AMNDCD_T,
        IT_NTSSND_T.

  LOOP AT IT_AR_DATA.
    MOVE-CORRESPONDING IT_AR_DATA TO IT_ALV.

*.. 수정사유 TEXT
    READ TABLE IT_AMNDCD_T WITH KEY
                  DOMVALUE_L = IT_AR_DATA-AMEND_CODE
                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-AMEND_CODE_TEXT = IT_AMNDCD_T-DDTEXT.
    ENDIF.

*.. 상태 TEXT
    READ TABLE IT_STATUS WITH KEY STATUS = IT_AR_DATA-STATUS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-STATUS_TEXT = IT_STATUS-STATUS_TEXT.
    ENDIF.

*.. 국세청 전송상태 TEXT
    READ TABLE IT_NTSSND_T WITH KEY
                  DOMVALUE_L = IT_AR_DATA-NTS_SEND_FLAG
                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-NTS_SEND_TEXT = IT_NTSSND_T-DDTEXT.
    ENDIF.
*
    APPEND IT_ALV.
    CLEAR  IT_ALV.
  ENDLOOP.

  IF P_DISP1 = 'X'.                          " 원본 세금계산서 기준
    PERFORM FILL_IT_ALV_ITM.
  ENDIF.

FORM GET_FILEDCATALOG_ALV .

  PERFORM GET_FILEDCATALOG TABLES G_FIELDCAT_T
                           USING  'IT_ALV'.

  LOOP AT G_FIELDCAT_T INTO G_FIELDCAT_S.
    CASE G_FIELDCAT_S-FIELDNAME.
      WHEN 'ISSUE_ID' OR 'INVDOC'.
        G_FIELDCAT_S-EMPHASIZE = 'C400'.
      WHEN 'INV_SEQ_C'.
        G_FIELDCAT_S-EMPHASIZE = 'C100'.
      WHEN 'AMEND_CODE_TEXT'.
        G_FIELDCAT_S-SELTEXT_S = 'Text'.
        G_FIELDCAT_S-OUTPUTLEN = '30'.
      WHEN 'STATUS' OR 'STATUS_TEXT'.
        G_FIELDCAT_S-EMPHASIZE = 'C500'.
      WHEN 'NTS_SEND_FLAG'.
        G_FIELDCAT_S-EMPHASIZE = 'C700'.
      WHEN 'NTS_SEND_TEXT'.
        G_FIELDCAT_S-SELTEXT_S = 'Text'.
        G_FIELDCAT_S-EMPHASIZE = 'C700'.
    ENDCASE.

*   통화단위.
    IF G_FIELDCAT_S-DATATYPE EQ 'CURR'.
      G_FIELDCAT_S-CURRENCY  = T001-WAERS.
      G_FIELDCAT_S-DO_SUM    = 'X'.
      G_FIELDCAT_S-EMPHASIZE = 'C300'.
    ENDIF.

*   HOTSPOT 정의
    PERFORM HOTSPOT(ZDTV3_FORM_COMM) USING G_FIELDCAT_S-FIELDNAME
                                           G_FIELDCAT_S-HOTSPOT
                                           ''
                                           'O'
                                           'AR'.
