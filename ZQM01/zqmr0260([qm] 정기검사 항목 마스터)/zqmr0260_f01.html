FORM find_group_master .


  DATA: BEGIN OF lt_code OCCURS 0,
          zmgrp    LIKE zqms0400-zmgrp,
          zmgrp_t  LIKE zqms0400-zmgrp_t,
          zmjcl    LIKE zqms0400-zmjcl  ,
          zmjcl_t  LIKE zqms0400-zmjcl_t,
          zmicl    LIKE zqms0400-zmicl  ,
          zmicl_t  LIKE zqms0400-zmicl_t,
        END OF lt_code.

  DATA: lt_return TYPE TABLE OF ddshretval WITH HEADER LINE.

  LOOP AT gt_zqmt0400.
    MOVE-CORRESPONDING gt_zqmt0400 TO lt_code.

    READ TABLE gt_zmjcl WITH KEY domvalue_l = lt_code-zmjcl.
    IF sy-subrc EQ 0.
      lt_code-zmjcl_t = gt_zmjcl-ddtext.
    ENDIF.

    READ TABLE gt_zmicl WITH KEY domvalue_l = lt_code-zmicl.
    IF sy-subrc EQ 0.
      lt_code-zmicl_t = gt_zmicl-ddtext.
    ENDIF.

    APPEND lt_code.
    CLEAR lt_code.
  ENDLOOP.

  SORT lt_code BY zmgrp.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'ZMGRP'
      value_org       = 'S'
    TABLES
      value_tab       = lt_code
      return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc = 0.
    READ TABLE lt_return INDEX 1.
    CLEAR gv_err.
    PERFORM check_zmgrp USING lt_return-fieldval
                        CHANGING gv_err.

    CHECK gv_err IS INITIAL.

    zqmt0460-zmgrp = lt_return-fieldval.
    READ TABLE lt_code WITH KEY zmgrp = zqmt0460-zmgrp.
    IF sy-subrc EQ 0.
      gv_zmgrp_t = lt_code-zmgrp_t.
    ENDIF.
  ENDIF.



ENDFORM. " FIND_GROUP_MASTER
