FORM report_auto_issue USING pcl_grid TYPE REF TO lcl_def_grid.

  DATA: lt_zqms0200 TYPE TABLE OF zqms0200 WITH HEADER LINE.
  DATA: lv_answer,
            lv_ret TYPE zereturn,
            lv_msg TYPE zemessage.


  CALL METHOD pcl_grid->check_changed_data.

  CLEAR lv_answer.
  PERFORM message_popup_step USING text-p01
                                                                  text-p02
                                                                  text-p03
                                                                  text-p04
                                                                  text-p05
                                                      CHANGING lv_answer.

  IF lv_answer = c_1.

    LOOP AT gt_list_regi.
      CLEAR: lt_zqms0200, lt_zqms0200[].
      MOVE-CORRESPONDING gt_list_regi TO lt_zqms0200.
      APPEND lt_zqms0200.

      CLEAR: lv_ret, lv_msg.
      call function 'ZQM_REPORT_AUTO_ISSUE'
        IMPORTING
          e_ret      = lv_ret
          e_msg      = lv_msg
        TABLES
          t_zqms0200 = lt_zqms0200.
      IF sy-subrc NE c_0.
        lv_ret  = c_e.

      ENDIF.
