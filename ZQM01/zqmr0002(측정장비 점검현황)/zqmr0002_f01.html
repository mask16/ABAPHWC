FORM move_list_data .

  CLEAR: gt_list, gt_list[].
  gt_list[] = gt_data[].

  CLEAR: gt_zqmt0110c, gt_zqmt0110c[].

  SELECT *
    FROM zqmt0110
    INTO TABLE gt_zqmt0110c
  FOR ALL ENTRIES IN gt_list
    WHERE pnumber EQ gt_list-pnumber
        AND check_y EQ p_check.

  SORT: gt_list BY pnumber,
             gt_zqmt0110c BY pnumber check_y.

  DELETE ADJACENT DUPLICATES FROM: gt_list COMPARING pnumber,
                                                                gt_zqmt0110c COMPARING pnumber check_y.


  LOOP AT gt_list.
    READ TABLE gt_zqmt0110c WITH KEY pnumber = gt_list-pnumber
                                                            check_y = p_check
                                                            BINARY SEARCH.
    IF sy-subrc EQ 0.
      MOVE: gt_zqmt0110c-m01 TO gt_list-m01,
                gt_zqmt0110c-m02 TO gt_list-m02,
                gt_zqmt0110c-m03 TO gt_list-m03,
                gt_zqmt0110c-m04 TO gt_list-m04,
                gt_zqmt0110c-m05 TO gt_list-m05,
                gt_zqmt0110c-m06 TO gt_list-m06,
                gt_zqmt0110c-m07 TO gt_list-m07,
                gt_zqmt0110c-m08 TO gt_list-m08,
                gt_zqmt0110c-m09 TO gt_list-m09,
                gt_zqmt0110c-m10 TO gt_list-m10,
                gt_zqmt0110c-m11 TO gt_list-m11,
                gt_zqmt0110c-m12 TO gt_list-m12,
                gt_zqmt0110c-check_y TO gt_list-check_y.
    ELSE.
      MOVE p_check TO gt_list-check_y.
    ENDIF.

    MODIFY gt_list.
  ENDLOOP.

ENDFORM. " MOVE_LIST_DATA
