MODULE status_0100 OUTPUT.

  CASE sy-dynnr.
    WHEN '0100'.
      IF p_r1 = 'X'.
        PERFORM including_pf USING 'GUAR'.
        PERFORM including_pf USING 'SAVE'.
        PERFORM excluding_pf USING 'CHAN'.
        PERFORM excluding_pf USING 'DELE'.
        PERFORM excluding_pf USING 'SAVE2'.
      ELSEIF p_r2 = 'X'.
        PERFORM including_pf USING 'CHAN'.
        PERFORM including_pf USING 'DELE'.
        PERFORM including_pf USING 'SAVE2'.
        PERFORM excluding_pf USING 'GUAR'.
        PERFORM excluding_pf USING 'SAVE'.
      ENDIF.

    WHEN '0200'.
      PERFORM excluding_pf USING 'CHAN'.
      PERFORM excluding_pf USING 'DELE'.
      PERFORM excluding_pf USING 'SAVE2'.
      PERFORM excluding_pf USING 'GUAR'.
      PERFORM excluding_pf USING 'SAVE'.
  ENDCASE.

  SET PF-STATUS sy-dynnr EXCLUDING pf.
  SET TITLEBAR sy-dynnr.

ENDMODULE. " STATUS_0100 OUTPUT
