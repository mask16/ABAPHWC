FORM GET_DATA .

  INIT GT_TAB.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_TAB

         AND ZMJCL   IN S_ZMJCL
         AND ZMICL   IN S_ZMICL
         AND ZDELETE NE 'X'.
