FORM SET_FIELDCATALOG_FIELD TABLES PT_FIELDCATALOG STRUCTURE
                                                      GT_FIELDCATALOG
                             USING    VALUE(P_GUBUN)
                                      VALUE(P_FIELDNAME)
                                      VALUE(P_VALUE).

  CONSTANTS: C_REPTEXT TYPE FIELDNAME VALUE 'REPTEXT'.

  FIELD-SYMBOLS: <LV_FIELD> TYPE ANY.

* clear header
  IF P_GUBUN EQ 'S'.
    CLEAR: PT_FIELDCATALOG.
  ENDIF.

* get component's attribute when inputed fieldname
  IF P_FIELDNAME EQ 'FIELDNAME'.

*   get component's attribute
    READ TABLE GT_COMPONENT WITH KEY NAME = P_FIELDNAME
                                     BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      MOVE: GT_COMPONENT-NAME      TO PT_FIELDCATALOG-FIELDNAME,
            GT_COMPONENT-TYPE_KIND TO PT_FIELDCATALOG-INTTYPE,
            GT_COMPONENT-LENGTH    TO PT_FIELDCATALOG-INTLEN,
            GT_COMPONENT-DECIMALS  TO PT_FIELDCATALOG-DECIMALS.

*     set lowcase when character
      IF GT_COMPONENT-TYPE_KIND EQ 'C'.
        MOVE 'X' TO PT_FIELDCATALOG-LOWERCASE.
      ENDIF.
    ENDIF.
  ENDIF.

* set value
  ASSIGN COMPONENT P_FIELDNAME OF STRUCTURE PT_FIELDCATALOG TO <LV_FIELD>.
  MOVE P_VALUE TO <LV_FIELD>.

* set text (short = medium = long)
  IF P_FIELDNAME EQ C_REPTEXT.
    ASSIGN COMPONENT 'SCRTEXT_S' OF STRUCTURE PT_FIELDCATALOG TO <LV_FIELD>.
    MOVE P_VALUE TO <LV_FIELD>.

    ASSIGN COMPONENT 'SCRTEXT_M' OF STRUCTURE PT_FIELDCATALOG TO <LV_FIELD>.
    MOVE P_VALUE TO <LV_FIELD>.

    ASSIGN COMPONENT 'SCRTEXT_L' OF STRUCTURE PT_FIELDCATALOG TO <LV_FIELD>.
    MOVE P_VALUE TO <LV_FIELD>.

    ASSIGN COMPONENT 'COLTEXT' OF STRUCTURE PT_FIELDCATALOG TO <LV_FIELD>.
    MOVE P_VALUE TO <LV_FIELD>.
  ENDIF.

* modify data
  IF P_GUBUN EQ 'E'.
    APPEND PT_FIELDCATALOG.
  ENDIF.

ENDFORM. " SET_FIELDCATALOG_FIELD
