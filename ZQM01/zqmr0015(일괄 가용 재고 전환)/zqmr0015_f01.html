FORM get_list_data .

  CLEAR: gt_data, gt_data[].
  SELECT a~prueflos a~werkvorg AS werk a~lifnr a~ebeln a~matnr a~ktextmat a~mblnr a~losmenge
               a~budat a~mengeneinh a~lmengezub b~name1 c~matkl d~ekgrp
    FROM qals AS a INNER JOIN
         lfa1 AS b
      ON b~lifnr EQ a~lifnr INNER JOIN
         mara AS c
      ON c~matnr = a~matnr INNER JOIN
         marc AS d
      ON d~matnr = a~matnr
     AND d~werks = a~werkvorg

    INTO CORRESPONDING FIELDS OF TABLE gt_data
    WHERE a~prueflos IN s_pruef
        AND a~werkvorg IN s_werk
        AND a~lifnr IN s_lifnr
        AND a~ebeln IN s_ebeln
        AND a~matnr IN s_matnr
        AND a~budat IN s_budat
        AND a~lmengezub > 0
        AND c~matkl IN s_matkl
        AND d~ekgrp IN s_ekgrp.

ENDFORM. " GET_LIST_DATA

FORM move_list_data .

  SELECT *
    FROM zqmt0040
    INTO TABLE gt_zqmt0040
    FOR ALL ENTRIES IN gt_data
    WHERE prueflos EQ gt_data-prueflos.
  SORT gt_zqmt0040 BY prueflos.

  SELECT *
    FROM qave
    INTO TABLE gt_qave
    FOR ALL ENTRIES IN gt_data
    WHERE prueflos EQ gt_data-prueflos
         AND vcode EQ c_o.

  IF gt_qave[] IS NOT INITIAL.
    SORT gt_qave BY prueflos kzart zaehler.
    DELETE ADJACENT DUPLICATES FROM gt_qave COMPARING prueflos.

    SORT gt_data BY prueflos.
    LOOP AT gt_data.
      READ TABLE gt_qave WITH KEY prueflos = gt_data-prueflos
                                       BINARY SEARCH.
      IF sy-subrc EQ 0.
        gt_data-vcode = gt_qave-vcode.

        SELECT SINGLE kurztext
          FROM qpct
          INTO gt_data-kurztext
          WHERE codegruppe = gt_qave-vcodegrp
               AND code            = gt_qave-vcode.
      ENDIF.

      READ TABLE gt_zqmt0040 WITH KEY prueflos = gt_data-prueflos
                                          BINARY SEARCH.
      IF sy-subrc EQ 0.
        gt_data-origin             = gt_zqmt0040-origin.
        gt_data-refno             = gt_zqmt0040-refno.
        gt_data-grquan          = gt_zqmt0040-grquan.
        gt_data-grdate           = gt_zqmt0040-grdate.
        gt_data-qaquan         = gt_zqmt0040-qaquan.
        gt_data-qadate          = gt_zqmt0040-qadate.
        gt_data-qaname        = gt_zqmt0040-qaname.
        gt_data-fecodtxt        = gt_zqmt0040-fecodtxt.
        gt_data-activity          = gt_zqmt0040-activity.
        gt_data-lfno               = gt_zqmt0040-lfno.
      ENDIF.

      MODIFY gt_data TRANSPORTING vcode kurztext origin refno grquan grdate qaquan qadate
      qaname fecodtxt activity lfno.
    ENDLOOP.

    IF gt_data[] IS NOT INITIAL.
      DELETE gt_data WHERE vcode <> c_o.
    ENDIF.

    CLEAR: gt_list, gt_list[].
    LOOP AT gt_data.
      MOVE-CORRESPONDING gt_data TO gt_list.
      APPEND gt_list.
      CLEAR: gt_list, gt_data.
    ENDLOOP.

  ELSE.
    MESSAGE s010 WITH text-m02 DISPLAY LIKE c_e.
    EXIT.
  ENDIF.

ENDFORM. " MOVE_LIST_DATA
