FORM GET_LIST_DATA .

  CLEAR: GT_DATA, GT_DATA[].
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_DATA
    FROM ZQMT0430 AS A
        INNER JOIN ZQMT0400 AS B
                 ON A~BUKRS EQ B~BUKRS
                AND A~ZMGRP EQ B~ZMGRP
      WHERE A~BUKRS   EQ P_BUKRS
        AND A~MATNR   IN S_MATNR
        AND A~ZMGRP   IN S_ZMGRP
        AND B~ZMGRP_T IN S_ZMGRPT.

  INIT GT_MVKE.
  INIT GT_TVM1T.
  INIT GT_MAKT.

  IF GT_DATA[] IS NOT INITIAL.
    SELECT *
      INTO TABLE GT_MAKT
      FROM MAKT
      FOR ALL ENTRIES IN GT_DATA
        WHERE MATNR EQ GT_DATA-MATNR
          AND SPRAS EQ SY-LANGU.

    SELECT *
       INTO TABLE GT_MVKE
       FROM MVKE
      FOR ALL ENTRIES IN GT_DATA
        WHERE MATNR EQ GT_DATA-MATNR
          AND VKORG EQ P_BUKRS.
    IF SY-SUBRC EQ 0.
      SELECT *
       INTO TABLE GT_TVM1T
       FROM TVM1T
        FOR ALL ENTRIES IN GT_MVKE
        WHERE MVGR1 EQ GT_MVKE-MVGR1
          AND SPRAS EQ SY-LANGU.
    ENDIF.
    SORT GT_MVKE  BY MATNR.
    SORT GT_TVM1T BY MVGR1.
    SORT GT_MAKT  BY MATNR.

  ENDIF.


* unpublish
*  LCL_GOS->UNPUBLISH( ).

ENDFORM. " GET_LIST_DATA
