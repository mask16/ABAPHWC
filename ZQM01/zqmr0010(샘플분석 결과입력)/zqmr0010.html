FORM get_data .
  CLEAR: gt_disp[], g_ucomm.

  DATA: ls_plant    TYPE ihttpnvp,
        l_index     TYPE i.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_disp
    FROM zqmt0120
   WHERE reqno    IN s_reqno
     AND plant    IN s_plant
     AND reqdate  IN s_reqdat
     AND matnr    IN s_matnr
     AND matkl     IN s_matkl
     AND gubun    IN s_gubun
     AND reqobjt  IN s_reqobj.

  CHECK gt_disp[] IS NOT INITIAL.
  IF p_option IS NOT INITIAL.
    DELETE gt_disp WHERE samtest IS NOT INITIAL.
  ENDIF.
  CHECK gt_disp[] IS NOT INITIAL.

  LOOP AT gt_disp INTO gs_disp.
    l_index = sy-tabix.

    READ TABLE: gt_plant  INTO ls_plant WITH KEY name = gs_disp-plant.
    MOVE: ls_plant-value  TO gs_disp-plant_t.

    IF gs_disp-matnr IS NOT INITIAL.
      SELECT SINGLE matkl
        FROM mara
        INTO gs_disp-matkl
        WHERE matnr = gs_disp-matnr.

      IF gs_disp-matkl IS NOT INITIAL.
        SELECT SINGLE wgbez
          FROM t023t
          INTO gs_disp-wgbez
          WHERE spras = sy-langu
               AND matkl = gs_disp-matkl.
      ENDIF.
    ENDIF.

    MODIFY gt_disp FROM gs_disp INDEX l_index TRANSPORTING plant_t matkl wgbez." gubun_t reqobjt_t.
    CLEAR: gs_disp, l_index, ls_plant.
  ENDLOOP.
ENDFORM. " GET_DATA
