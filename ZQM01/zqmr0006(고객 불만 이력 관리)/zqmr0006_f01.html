FORM get_list_data .

  CLEAR: gt_data, gt_data[].
  SELECT a~intno a~crdat a~vkbur a~spcode a~kunnr a~status a~totqty a~nusqty
               a~kg a~palqty a~pal a~qmsts a~qmdat b~item b~fecod b~werks b~matnr
               b~lotno b~mfgno b~lfdat b~clmqty b~vbeln b~posnr b~smpdt1 b~smpdt2
               b~smpqty b~meins b~memo
    FROM zsdt0150 AS a INNER JOIN zsdt0151 AS b
                                        ON a~intno EQ b~intno
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    WHERE a~intno IN s_intno
        AND b~werks IN s_werks
        AND b~matnr IN s_matnr
        AND a~kunnr IN s_kunnr
        AND a~crdat IN s_crdat
        AND a~qmdat IN s_qmdat.

ENDFORM. " GET_LIST_DATA
