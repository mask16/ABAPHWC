FORM select_data_0100 .

  DATA: lt_data_0100  LIKE gt_data_0100,
        ls_data_0100  LIKE LINE OF gt_data_0100,
        lv_rowcount   TYPE char10.

  CLEAR: gt_data_0100.

  PERFORM sapgui_progress_indicator USING 1 '조회중...'.

  SELECT a~prueflos
         a~werkvorg
         a~lifnr as qlifnr
         a~ebeln
         a~matnr
         a~ktextmat
         a~mblnr
         b~origin
         b~refno
         b~grquan
         b~grdate
         b~qaquan
         b~qadate
         b~qaname
         b~fecodtxt
         b~activity
         b~lfno
         b~comp_1
         b~comp_2
         b~comp_3
         b~comp_4
         b~comp_5
         b~comp_6
         b~comp_7
         b~comp_8
         b~comp_9
         b~comp_10
         b~comp_11
         b~comp_12
         b~mesh_1
         b~mesh_2
         b~mesh_3
         b~mesh_4
         b~mesh_5
         b~mesh_6
         b~mesh_7
         b~mesh_8
         b~mesh_9
         b~mesh_10
         b~mesh_11
         b~mesh_12
         c~name1
         d~vcode
         d~vcodegrp
    INTO CORRESPONDING FIELDS OF TABLE lt_data_0100
    FROM qals as a
   INNER JOIN
         zqmt0040 as b
      ON a~prueflos  EQ b~prueflos
   INNER JOIN
         lfa1 as c
      ON a~lifnr     EQ c~lifnr
   INNER JOIN
         qave as d
      ON a~prueflos  EQ d~prueflos
   WHERE a~matnr   eq pa_matnr
     AND a~werk    IN so_werk
     AND a~lifnr   IN so_lifnr
     AND a~ebeln   IN so_ebeln
     AND a~budat   IN so_budat
     AND b~qadate  IN so_qadat.

  SORT lt_data_0100 BY prueflos.
** 추가정보조회
  LOOP AT lt_data_0100 INTO ls_data_0100.
*   공급자코드
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
       EXPORTING
         input         = ls_data_0100-qlifnr
       IMPORTING
         OUTPUT        = ls_data_0100-qlifnr.

*   자재코드
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
       EXPORTING
         input         = ls_data_0100-matnr
       IMPORTING
         OUTPUT        = ls_data_0100-matnr.

*   판정이름
    PERFORM get_vcode_text USING ls_data_0100-vcode
                                 ls_data_0100-vcodegrp
                        CHANGING ls_data_0100-kurztext.

    APPEND ls_data_0100 TO gt_data_0100.
  ENDLOOP.

* 타이틀 및 조회건수
  lv_rowcount = lines( gt_data_0100 ).
  CONCATENATE '자재 검사 결과 이력 조회 -' lv_rowcount
         INTO gv_title_0100.

  PERFORM sapgui_progress_indicator USING 100 '조회완료'.

ENDFORM. " SELECT_DATA_0100
