MODULE user_command_0100 INPUT.

  CLEAR gv_save_code.
  gv_save_code = ok_code.
  CLEAR ok_code.

  CASE gv_save_code.
    WHEN 'CHAN'.
      PERFORM change_data USING lcl_grid1
                          CHANGING gv_data_cnt.

      CHECK gt_listc[] IS NOT INITIAL.
      PERFORM set_grid1 TABLES      gt_listc             "INTERNAL TABLE DATA
                                    gt_lvc_fcat2         "FIELD CATALOG : LVC_T_FCAT
                                    gt_sort2
                         USING      'GT_LISTC'         "INTERNAL TABLE NAME
                                    'OBJ2'
                         CHANGING   lcl_grid2            "ALV GRID : CL_GUI_ALV_GRID
                                    gs_layo2.          "LAYOUT : LVC_S_LAYO

    WHEN 'SAVE'.
      PERFORM change_check_data USING lcl_grid1
                                CHANGING gv_data_cnt.

      CHECK gt_listc[] IS NOT INITIAL
            AND gt_listc_tmp[] IS NOT INITIAL.

      PERFORM save_data_create USING lcl_grid1.

    WHEN 'GUAR'.
      PERFORM guarantee_spec_confirm USING lcl_grid1
                                     CHANGING gv_data_cnt.

    WHEN 'SAVE2'.
      PERFORM save_data_change USING lcl_grid1 lcl_grid2.

    WHEN 'DELE'.
      PERFORM delete_data USING lcl_grid1
                          CHANGING gv_data_cnt.

    WHEN 'CALC'.
      PERFORM calculate_data USING lcl_grid1
                                              CHANGING gv_data_cnt.

    WHEN 'F_SEAR'.
      PERFORM guarantee_spec_search USING lcl_grid1
                                    CHANGING gv_data_cnt.

    WHEN 'PLUS'.
      PERFORM insert_row USING 'OBJ3'.

    WHEN 'MINU'.
      PERFORM delete_row USING lcl_grid1.

  ENDCASE.

ENDMODULE. " USER_COMMAND_0100 INPUT
