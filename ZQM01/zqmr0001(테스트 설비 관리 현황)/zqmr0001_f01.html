FORM GET_LIST_DATA .

        AND PLANT IN S_PLANT

        AND CENTER IN S_CENTER.

FORM MOVE_LIST_DATA .

  DATA: LV_COND1 TYPE STRING,
            LV_COND2 TYPE STRING.
  DATA: LV_IDX1(2) TYPE N,
            LV_IDX2(2) TYPE N.
  DATA LV_DATE TYPE SY-DATUM.

  FIELD-SYMBOLS: <FS1>, <FS2>.

  CLEAR GV_DATA_CNT.
  GV_DATA_CNT = LINES( GT_DATA ).

  CLEAR LV_DATE.
  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = SY-DATUM
      DAYS      = 00
      MONTHS    = 00
      SIGNUM    = '-'
      YEARS     = 04
    IMPORTING
      CALC_DATE = LV_DATE.

  CLEAR: GT_DISP, GT_DISP[].
  LOOP AT GT_DATA.

    CLEAR: LV_IDX1, LV_IDX2.

    LV_IDX1 = LV_DATE+2(2).
    LV_IDX2 = 01.

    DO 5 TIMES.
      CONCATENATE 'gt_data-y' LV_IDX1 INTO LV_COND1.
      ASSIGN (LV_COND1) TO <FS1>.

      CONCATENATE 'gt_disp-year' LV_IDX2 INTO LV_COND2.
      ASSIGN (LV_COND2) TO <FS2>.

      MOVE-CORRESPONDING GT_DATA TO GT_DISP.
      MOVE <FS1> TO <FS2>.

      ADD 1 TO LV_IDX1.
      ADD 1 TO LV_IDX2..
    ENDDO.

    APPEND GT_DISP.
    CLEAR: GT_DATA, GT_DISP.
  ENDLOOP.

ENDFORM. " MOVE_LIST_DATA
