FORM download_excel_file USING p_file.

  DATA : lv_dfname       LIKE  rlgrap-filename,
            lv_objid        LIKE  wwwdata-objid,
            lv_fname        LIKE  rlgrap-filename,
            lv_tail         TYPE  string,
            ls_wwwdatatab   LIKE  wwwdatatab,
            lv_subrc        TYPE  i.

  CLEAR : lv_fname, ls_wwwdatatab, lv_subrc.

  lv_dfname = p_file. "file name.


  lv_objid  = 'ZMMR0380_UPD'.

  SELECT SINGLE  *
    INTO CORRESPONDING FIELDS OF  ls_wwwdatatab
    FROM wwwdata     " WWW 오브젝트 저장에 대한 INDX-유형 테이블
   WHERE objid = lv_objid.  " SAP WWW 게이트웨이 오브젝트 이름

  IF sy-subrc NE 0.
    MESSAGE s015 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.


  PERFORM  p_filepath USING 'S' '' 'FILENAME'
                                  lv_dfname ."filename.

  IF lv_dfname IS NOT  INITIAL.
* Web 템플레이트(Excel Form) 다운로드
    CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
      EXPORTING
        key         = ls_wwwdatatab
        destination = lv_dfname.

    CLEAR : lv_fname, lv_tail.
    SPLIT  lv_dfname  AT  ' '  INTO  lv_fname  lv_tail.
    IF  lv_dfname <> lv_fname.
      MESSAGE  s016  WITH  lv_dfname.
    ELSE.
*   Download 파일명에 공백이 없는 경우 Excel 실행
      CALL FUNCTION 'WS_EXECUTE'
        EXPORTING
          commandline = lv_dfname
          program     = 'EXCEL.EXE'.
      MESSAGE  s016  WITH  lv_dfname.
    ENDIF.
  ENDIF.

ENDFORM. " DOWNLOAD_EXCEL_FILE

FORM data_processing .

  CLEAR gt_data.

  LOOP AT gt_excel INTO gs_excel.
    MOVE-CORRESPONDING gs_excel TO gs_data.

    "필수값 체크
    IF gs_excel-zldat IS INITIAL.
      CONCATENATE gs_data-msg '실사일자'
        INTO gs_data-msg SEPARATED BY space.
    ENDIF.

    "플랜트 점검
    IF gs_excel-werks IS INITIAL.
      CONCATENATE gs_data-msg '플랜트'
        INTO gs_data-msg SEPARATED BY space.

    ELSE.
      READ TABLE gt_werks WITH KEY werks = gs_data-werks
        BINARY SEARCH INTO gs_werks TRANSPORTING name2.
      IF sy-subrc = 0.
        gs_data-name2 = gs_werks-name2.
      ENDIF.

    ENDIF.

    "저장위치 점검
    IF gs_excel-lgort IS INITIAL.
      CONCATENATE gs_data-msg '저장위치'
        INTO gs_data-msg SEPARATED BY space.

    ELSE.
      READ TABLE gt_lgort WITH KEY werks = gs_data-werks
                                   lgort = gs_data-lgort
        BINARY SEARCH INTO gs_lgort TRANSPORTING lgobe.
      IF sy-subrc = 0.
        gs_data-lgobe = gs_lgort-lgobe.
      ENDIF.
    ENDIF.

    IF gs_excel-matnr IS INITIAL.
      CONCATENATE gs_data-msg '자재코드'
        INTO gs_data-msg SEPARATED BY space.
    ENDIF.

    IF gs_data-msg IS NOT INITIAL.
      CONCATENATE text-t00 gs_data-msg
        INTO gs_data-msg SEPARATED BY space.
    ENDIF.

    "자재코드 점검
    IF gs_data-matnr IS NOT INITIAL.
      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input        = gs_data-matnr
        IMPORTING
          output       = gs_data-matnr
        EXCEPTIONS
          length_error = 1
          OTHERS       = 2.

      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input        = gs_excel-matnr
        IMPORTING
          output       = gs_excel-matnr
        EXCEPTIONS
          length_error = 1
          OTHERS       = 2.

      "자재 내역
      SELECT SINGLE a~mtart
            a~meins
            b~maktx
      INTO (gs_data-mtart, gs_data-meins, gs_data-maktx)
      FROM mara AS a
      LEFT JOIN makt AS b ON a~matnr = b~matnr
                         AND spras = sy-langu
      WHERE a~matnr = gs_data-matnr.

      IF sy-subrc = 0.
        gs_excel-meins = gs_data-meins.

        "자재 유형
        SELECT SINGLE mtbez
        INTO gs_data-mtbez
        FROM t134t
        WHERE mtart = gs_data-mtart
        AND spras = sy-langu.

      ELSE.
        CLEAR gs_data-msg.
        MESSAGE s305(m3) INTO gs_data-msg.
      ENDIF.

    ENDIF.

    CASE gs_excel-bstar.
      WHEN '2'. "품질
        gs_data-insme = gs_excel-erfmg.
      WHEN '4'. "보류
        gs_data-speme = gs_excel-erfmg.
      WHEN OTHERS. "가용
        gs_data-labst = gs_excel-erfmg.
    ENDCASE.
    "총 재고
    gs_data-stock = gs_excel-erfmg.

    IF gs_data-msg IS NOT INITIAL.
      gs_data-icon = icon_led_red.
    ELSE.
      gs_data-icon = icon_dummy.
    ENDIF.

    COLLECT gs_data INTO gt_data.
    CLEAR gs_data.

    MODIFY gt_excel FROM gs_excel.
  ENDLOOP.

ENDFORM. " DATA_PROCESSING

FORM execute_bdc .
  DATA: l_count TYPE i,
        l_subrc TYPE sy-subrc.

  DATA: t_data TYPE TABLE OF ty_excel.

  "선택행 정보
  CLEAR: et_index_rows, es_index_rows.

  CALL METHOD grid1->get_selected_rows
    IMPORTING
      et_index_rows = et_index_rows.

  CLEAR l_count.
  DESCRIBE TABLE et_index_rows LINES l_count.

  IF l_count = 0 OR l_count IS INITIAL.
    MESSAGE s003 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_excel BY zldat werks lgort matnr.

  CLEAR: t_data, l_subrc.

  LOOP AT et_index_rows INTO es_index_rows.
    READ TABLE gt_data INDEX es_index_rows-index INTO gs_data.

    "에러항목 포함 시
    IF gs_data-icon <> icon_dummy.
      CLEAR t_data.
      l_subrc = 3.
      EXIT.
    ENDIF.

    READ TABLE gt_excel WITH KEY zldat = gs_data-zldat
                                            werks = gs_data-werks
                                            lgort = gs_data-lgort
                                            matnr = gs_data-matnr
                                            BINARY SEARCH TRANSPORTING NO FIELDS.
    IF sy-subrc = 0.
      LOOP AT gt_excel INTO gs_excel FROM sy-tabix.
        IF gs_excel-zldat <> gs_data-zldat
           OR gs_excel-werks <> gs_data-werks
           OR gs_excel-lgort <> gs_data-lgort
           OR gs_excel-matnr <> gs_data-matnr.
          EXIT.
        ENDIF.

        APPEND gs_excel TO t_data.

      ENDLOOP.
    ENDIF.
  ENDLOOP.

  CASE l_subrc.
    WHEN 0.
      IF t_data IS NOT INITIAL.
        PERFORM run_bdcdata TABLES t_data.
      ENDIF.

    WHEN 3. "오류 상태 포함
      MESSAGE s001 WITH text-t01 DISPLAY LIKE 'E'.
      EXIT.
  ENDCASE.

ENDFORM. " EXECUTE_BDC

*----------------------------------------------------------------------*
FORM run_bdcdata TABLES t_data STRUCTURE gs_excel.

  DATA: l_index TYPE n,
        l_matnr TYPE c LENGTH 20,
        l_erfmg TYPE c LENGTH 20,
        l_bstar TYPE c LENGTH 20,
        l_xnull TYPE c LENGTH 20.

  "수량단위 환산
  DATA: l_quan TYPE c LENGTH 20.

  SORT t_data BY zldat werks lgort.


  CLEAR: i_options.

  i_options-dismode  = p_mode.
  i_options-updmode  = 'A'.
  i_options-defsize  = 'X'.
  i_options-nobinpt  = 'X'.

  CLEAR: it_err_list, it_err_list[].

  LOOP AT t_data.
    MOVE-CORRESPONDING t_data TO gs_excel.

    "플랜트, 저장위치가 바뀔 때 마다 저장 후 신규 문서 생성
    AT NEW lgort.
      CLEAR: bdcdata, bdcdata[].

      PERFORM p_dynpro USING   'SAPMM07I'             '0700'.
      PERFORM p_field  USING : 'BDC_OKCODE'           '/00'  ,
                               'RM07I-ZLDAT'          gs_excel-zldat,
                               'RM07I-BLDAT'          gs_excel-zldat,
                               'IKPF-WERKS'           gs_excel-werks,
                               'IKPF-LGORT'           gs_excel-lgort.

    ENDAT.

    ADD 1 TO l_index.

    "대량 입력 시 마다 엔터
    IF l_index = 1.
      PERFORM p_dynpro USING   'SAPMM07I'             '0731'.
      PERFORM p_field  USING : 'BDC_OKCODE'           '/00' .
    ENDIF.

    CLEAR: l_matnr, l_erfmg, l_bstar, l_xnull.

    CONCATENATE 'ISEG-MATNR(0' l_index ')' INTO l_matnr.
    CONCATENATE 'ISEG-ERFMG(0' l_index ')' INTO l_erfmg.
    CONCATENATE 'ISEG-BSTAR(0' l_index ')' INTO l_bstar.
    CONCATENATE 'ISEG-XNULL(0' l_index ')' INTO l_xnull.

    "재고유형: 기본값 = 가용재고
    IF gs_excel-bstar IS INITIAL.
      gs_excel-bstar = '1'.
    ENDIF.

    "재고가 없는 경우 제로 카운트 체크
    IF gs_excel-erfmg = 0 OR gs_excel-erfmg IS INITIAL.
      gs_excel-xnull = 'X'.
    ELSE.
      CLEAR gs_excel-xnull.
    ENDIF.

    "수량단위 환산
    CLEAR l_quan.
    WRITE gs_excel-erfmg TO l_quan UNIT gs_excel-meins.
    CONDENSE l_quan.

    PERFORM p_field  USING :
                               l_matnr                gs_excel-matnr,
                               l_erfmg                l_quan,
                               l_bstar                gs_excel-bstar,
                               l_xnull                gs_excel-xnull.

    "최대 배치 입력 창 6개
    IF l_index = 6.
      CLEAR l_index.
    ENDIF.

    "플랜트, 저장위치가 바뀔 때 마다 저장 후 신규 문서 생성
    AT END OF lgort.
      "기타실사 문서를 위해 문서 전기
      PERFORM p_dynpro USING   'SAPMM07I'             '0731'.
      PERFORM p_field  USING : 'BDC_OKCODE'           '=BU' .

      CLEAR l_index.

      CLEAR : messtab, messtab[].
      CALL TRANSACTION 'MI09'  USING bdcdata
                              OPTIONS FROM  i_options
                              MESSAGES INTO messtab.

      IF messtab[] IS INITIAL.
        PERFORM sy_message_make     TABLES it_err_list.
      ELSE.
        PERFORM return_message_make TABLES it_err_list.
      ENDIF.

    ENDAT.

  ENDLOOP.

  CLEAR v_count.
  DESCRIBE  TABLE it_err_list   LINES  v_count.
  IF v_count GT 0.
    include = 'POPU'.
    CALL SCREEN 9999 STARTING AT  05   3
                     ENDING   AT  115 12.
    CLEAR : include.
  ENDIF.

ENDFORM. " RUN_BDCDATA
