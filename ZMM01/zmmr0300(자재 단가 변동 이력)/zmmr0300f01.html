FORM GET_DATA .

  CLEAR : GT_MARC, GT_MARC[], GT_EKBE, GT_EKBE[], R_GRDAT, R_GRDAT[].

  SELECT B~WERKS

    FROM MARA AS A INNER JOIN
         MARC AS B
      ON A~MATNR = B~MATNR
   WHERE B~WERKS IN SO_WERKS
     AND A~MATNR IN SO_MATNR
     AND A~MTART =  PA_MTART
     AND A~MATKL IN SO_MATKL.

  R_GRDAT = 'IBT'.
  CONCATENATE SO_SPMON-LOW  '01' INTO R_GRDAT-LOW.
  CONCATENATE SO_SPMON-high '01' INTO R_GRDAT-high.

  DATA L_LASTDAY(2).

  CALL FUNCTION 'END_OF_MONTH_DETERMINE'
    EXPORTING
      DATUM  = R_GRDAT-high
    IMPORTING
      TT     = L_LASTDAY
    EXCEPTIONS
      OTHERS = 1.
  CONCATENATE SO_SPMON-high L_LASTDAY INTO R_GRDAT-high.
  APPEND R_GRDAT.

  CASE 'X'.
    WHEN PA_HIST.
      SELECT A~WERKS
             A~MATNR
             B~LIFNR
        A~WRBTR
             A~BPMNG
             A~WAERS
             A~BUDAT
             A~MENGE
             B~EBELN
             A~BELNR

        FROM EKBE AS A INNER JOIN

             A~MENGE
             B~EBELN
             A~BELNR
        INTO CORRESPONDING FIELDS OF TABLE GT_EKBE
        FROM EKBE AS A INNER JOIN
             EKKO AS B
          ON A~EBELN = B~EBELN
         FOR ALL ENTRIES IN GT_MARC
       WHERE A~MATNR = GT_MARC-MATNR
         AND A~WERKS = GT_MARC-WERKS
         AND B~EKGRP IN SO_EKGRP
         AND B~LIFNR IN SO_LIFNR

*        DESCENDING BELNR DESCENDING BUZEI DESCENDING
  ENDCASE.

  SORT GT_MARC BY WERKS MATNR.
  SORT GT_EKBE BY WERKS MATNR.
