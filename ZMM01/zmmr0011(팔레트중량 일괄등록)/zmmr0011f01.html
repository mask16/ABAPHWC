FORM get_excel_file TABLES p_table
                     USING  p_filename  LIKE rlgrap-filename.

  DATA : it_line  TYPE alsmex_tabline OCCURS 0 WITH HEADER LINE.
  DATA : l_index LIKE sy-tabix.
  DATA : l_start_col TYPE i VALUE '1',
         l_start_row TYPE i VALUE '1',
         l_end_col   TYPE i VALUE '256',
         l_end_row   TYPE i VALUE '65536'.
  FIELD-SYMBOLS : <fs>.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_filename
      i_begin_col             = l_start_col
      i_begin_row             = l_start_row
      i_end_col               = l_end_col
      i_end_row               = l_end_row
    TABLES
      intern                  = it_line
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.
  IF sy-subrc <> 0.
    MESSAGE s016(rp)   WITH 'File Upload Failed !'  .
    EXIT.
  ENDIF.

  CHECK NOT it_line[] IS INITIAL.

  SORT it_line BY row col.

  LOOP AT it_line.
    MOVE : it_line-col TO l_index.
    ASSIGN COMPONENT l_index OF STRUCTURE p_table TO <fs>.
    MOVE : it_line-value TO <fs>.
    AT END OF row.
      APPEND p_table.
      CLEAR p_table.
    ENDAT.
  ENDLOOP.

ENDFORM. " GET_EXCEL_FILE

FORM exec_excel_file .

  DATA: lv_msgno LIKE sy-msgno,
        lv_msgv1 LIKE sy-msgv1.

  LOOP AT gt_success.

    CLEAR: gt_bdc_tab,     gt_bdc_tab[],

    PERFORM  dynpro USING: 'X' 'SAPLMGMM'               '0080',

                           ' ' 'BDC_OKCODE'             '=BU',
                           ' ' 'MVKE-MVGR5'             gt_success-mvgr5.

                             MODE  p_mode

                         MESSAGES INTO gt_messtab.

    READ TABLE gt_messtab WITH KEY msgtyp = 'E'.

    ELSE.
      gv_fail = gv_fail + 1.
      MOVE-CORRESPONDING gt_success TO gt_error.
      gt_error-text = 'BDC Processing Error !'.

      lv_msgv1 = gt_messtab-msgv1(50).

        EXPORTING
          iv_message_id     = gt_messtab-msgid
          iv_message_type   = gt_messtab-msgtyp

FORM dynpro USING dynbegin name value.

  CLEAR  gt_bdc_tab.
  IF  dynbegin = 'X'.
    MOVE:  name    TO  gt_bdc_tab-program,
           value   TO  gt_bdc_tab-dynpro,
           'X'     TO  gt_bdc_tab-dynbegin.
    APPEND gt_bdc_tab.
  ELSE.
    MOVE:  name    TO  gt_bdc_tab-fnam,
           value   TO  gt_bdc_tab-fval.
    APPEND gt_bdc_tab.
  ENDIF.

ENDFORM. " DYNPRO

FORM vald_excel_file .

  DATA: BEGIN OF lt_mvke OCCURS 0,
        matnr  LIKE mvke-matnr,
        vkorg  LIKE mvke-vkorg,
        vtweg  LIKE mvke-vtweg,
        END OF lt_mvke.

  DESCRIBE TABLE gt_upload   LINES sy-tfill.
  IF sy-tfill EQ 0.
    MESSAGE s016(rp) WITH 'Upload된 data가 없습니다.'.
    STOP.
  ELSE.
    LOOP AT gt_upload.
      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input        = gt_upload-matnr
        IMPORTING
          output       = gt_upload-matnr
        EXCEPTIONS
          length_error = 1
          OTHERS       = 2.

      MODIFY gt_upload.
    ENDLOOP.
  ENDIF.

* 헤더 텍스트 삭제
  DELETE gt_upload INDEX 1.

  CLEAR: lt_mvke,     lt_mvke[].
  SELECT matnr
         vkorg
         vtweg
         INTO CORRESPONDING FIELDS OF TABLE lt_mvke
         FROM mvke FOR ALL ENTRIES IN gt_upload
        WHERE matnr = gt_upload-matnr
          AND vkorg = gt_upload-vkorg
          AND vtweg = gt_upload-vtweg.
  SORT lt_mvke BY matnr vkorg vtweg.

  CLEAR: gt_error,     gt_error[],
         gt_success,   gt_success[],
         gv_total,     gv_fail.
  LOOP AT gt_upload.
    IF gt_upload IS INITIAL.
      DELETE gt_upload.
    ENDIF.

    READ TABLE lt_mvke WITH KEY matnr = gt_upload-matnr
                                vkorg = gt_upload-vkorg
                                vtweg = gt_upload-vtweg
                            BINARY SEARCH.
    IF sy-subrc = 0.
      gv_total = gv_total + 1.
      MOVE-CORRESPONDING gt_upload TO gt_success.
      APPEND gt_success.    CLEAR gt_success.
    ELSE.
      MOVE-CORRESPONDING gt_upload TO gt_error.
      gt_error-text = '데이터가 유효성 체크하세요'.
      gv_fail = gv_fail + 1.
      APPEND gt_error.    CLEAR gt_error.
    ENDIF.
  ENDLOOP.

ENDFORM. " VALD_EXCEL_FILE
