FORM data_select .

  TYPES: BEGIN OF t_matnr,
            matnr TYPE mara-matnr,
          END OF t_matnr.

  DATA: t_matnr TYPE TABLE OF t_matnr.

  "삭제 지시자
  CASE p_del.
    WHEN 'X'.
      CLEAR s_lvorm.

    WHEN space.
      s_lvorm = 'INE'.
      s_lvorm-low = 'X'.
      APPEND s_lvorm.

  ENDCASE.


  CLEAR: gt_data.

  "자재내역으로 코드 조회
  IF s_maktx IS NOT INITIAL.
    CLEAR: t_matnr.
    SELECT matnr
    INTO TABLE t_matnr
    FROM makt
    WHERE matnr IN s_matnr
    AND maktx IN s_maktx
    AND spras = sy-langu.

    IF sy-subrc <> 0.
      EXIT.
    ENDIF.

  ENDIF.

  CLEAR gt_data.
  IF t_matnr IS INITIAL.
    SELECT a~matnr a~groes a~lvorm a~matkl a~mtart
           a~ersda  a~ernam a~laeda a~aenam a~liqdt
           a~bismt a~meins a~bstme a~vabme a~spart
           a~zmaktx a~prdha
           b~maktx
           c~ekgrp c~insmk c~xchpf c~werks c~disgr c~dismm
           c~dispo c~maabc c~beskz c~lgfsb c~plifz c~webaz
           c~perkz c~mtvfp c~prctr c~losgr
           d~bwkey d~bklas d~mlmaa d~mlast d~stprs d~verpr
           d~peinh d~vprsv d~ekalr d~hkmat
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
    LEFT JOIN makt AS b ON a~matnr = b~matnr
                        AND b~spras = sy-langu
    LEFT JOIN marc AS c ON a~matnr = c~matnr
                        AND c~werks = p_werks
    LEFT JOIN mbew AS d ON a~matnr = d~matnr
                        AND d~bwkey = p_werks
    WHERE a~matnr IN s_matnr
    AND a~groes IN s_groes
    AND a~zmaktx IN s_zmaktx
    AND a~matkl IN s_matkl
    AND a~mtart IN s_mtart
    AND a~lvorm IN s_lvorm
    AND a~ersda IN s_ersda.

  ELSE.
    SELECT a~matnr a~groes a~lvorm a~matkl a~mtart
           a~ersda  a~ernam a~laeda a~aenam a~liqdt
           a~bismt a~meins a~bstme a~vabme a~spart
           a~zmaktx a~prdha
           b~maktx
           c~ekgrp c~insmk c~xchpf c~werks c~disgr c~dismm
           c~dispo c~maabc c~beskz c~lgfsb c~plifz c~webaz
           c~perkz c~mtvfp c~prctr c~losgr
           d~bklas d~mlmaa d~mlast d~stprs d~verpr
           d~peinh d~vprsv d~ekalr d~hkmat
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
    LEFT JOIN  makt AS b ON a~matnr = b~matnr
                         AND b~spras = sy-langu
    LEFT JOIN marc AS c ON a~matnr = c~matnr
                        AND c~werks = p_werks
    LEFT JOIN mbew AS d ON a~matnr = d~matnr
                        AND d~bwkey = p_werks
    FOR ALL ENTRIES IN t_matnr
    WHERE a~matnr = t_matnr-matnr
    AND a~matnr IN s_matnr
    AND a~groes IN s_groes
    AND a~zmaktx IN s_zmaktx
    AND a~matkl IN s_matkl
    AND a~mtart IN s_mtart
    AND a~lvorm IN s_lvorm
    AND a~ersda IN s_ersda.
  ENDIF.


ENDFORM. " DATA_SELECT

FORM print_material_label .

  DATA: l_count TYPE i.

  DATA: t_data TYPE TABLE OF zmms0120,
        s_data TYPE zmms0120.

  "선택행 정보
  CLEAR: et_index_rows, es_index_rows.

  CALL METHOD grid1->get_selected_rows
    IMPORTING
      et_index_rows = et_index_rows.

  CLEAR l_count.
  DESCRIBE TABLE et_index_rows LINES l_count.

  IF l_count = 0 OR l_count IS INITIAL.
    MESSAGE s003 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  CALL FUNCTION 'SSF_FUNCTION_MODULE_NAME'
    EXPORTING
      formname = 'ZMMSF0010'
    IMPORTING
      fm_name  = ssf_function.

  CLEAR: t_data.
  LOOP AT et_index_rows INTO es_index_rows.
    READ TABLE gt_data INDEX es_index_rows-index
    INTO gs_data.

    IF sy-subrc = 0.
      MOVE-CORRESPONDING gs_data TO s_data.
      APPEND s_data TO t_data.
      CLEAR s_data.
    ENDIF.
  ENDLOOP.

  IF t_data IS NOT INITIAL.
    CALL FUNCTION ssf_function
      EXPORTING
        control_parameters = gv_control_parameters
        output_options     = gv_output_options
      TABLES
        t_data             = t_data
      EXCEPTIONS
        formatting_error   = 1
        internal_error     = 2
        send_error         = 3
        user_canceled      = 4
        OTHERS             = 5.

* 4- 연속출력 가능 폼 close
    CALL FUNCTION 'SSF_CLOSE'
      EXCEPTIONS
        formatting_error = 1
        internal_error   = 2
        send_error       = 3
        OTHERS           = 4.
  ENDIF.

ENDFORM. " PRINT_MATERIAL_LABEL
