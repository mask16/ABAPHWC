  ENDIF.

FORM DATA_PROCESSING .
  DEFINE _ASSIGN.
    CLEAR: FIELDNAME.
    CONCATENATE 'GS_TMP-' &1 GS_WORK-BUDAT+4(2) INTO FIELDNAME.
    ASSIGN (FIELDNAME) TO &2.
  END-OF-DEFINITION.

  DEFINE _ASSIGN2.
    CLEAR: FIELDNAME.
    CONCATENATE 'GS_DATA-' &1 L_INDEX INTO FIELDNAME.
    ASSIGN (FIELDNAME) TO &2.
  END-OF-DEFINITION.

  DATA: FIELDNAME TYPE STRING,
        L_FIELDNM TYPE STRING,
        L_INDEX TYPE N LENGTH 2.


  DATA : LV_MENGE LIKE PEKPO-WEMNG.
  DATA : LV_INDEX TYPE I.

  DATA: LT_CELLCOL TYPE LVC_T_SCOL.

  PERFORM PROGRESS_BAR  USING 75  TEXT-P04.

  CLEAR: GT_WORK2, GT_WORK2[].

  LOOP AT GT_WORK INTO GS_WORK.
    PERFORM GET_EKBE CHANGING GS_WORK.
    MODIFY GT_WORK FROM GS_WORK.
  ENDLOOP.

  APPEND LINES OF GT_WORK2 TO GT_WORK.

  SORT GT_WORK BY MATNR LIFNR BUDAT.

  CLEAR: GT_TMP, GS_TMP.

  "자재별 거래처별 취합
  LOOP AT GT_WORK INTO GS_WORK.
    MOVE-CORRESPONDING GS_WORK TO GS_TMP.

    IF GS_WORK-MENGE_IV <> 0.

      CASE GS_WORK-BUDAT(4).
        WHEN V_PRE_YEAR.  "전년도 평균 단가
          ADD GS_WORK-DMBTR_IV TO GS_TMP-TOT_NETWR.
          ADD GS_WORK-MENGE_GR TO GS_TMP-TOT_MENGE.

        WHEN P_GJAHR. "당해 월별 입/출고
          _ASSIGN: 'NETWR' <NETWR>, "월별 총액
                   'MENGE' <MENGE>. "월별 총수량

          IF SY-SUBRC = 0.
            ADD GS_WORK-DMBTR_IV TO <NETWR>.
            ADD GS_WORK-MENGE_GR TO <MENGE>.

          ENDIF.

      ENDCASE.

      COLLECT GS_TMP INTO GT_TMP.
      CLEAR GS_TMP.
    ENDIF.
  ENDLOOP.

  CLEAR: GT_DATA.
  LOOP AT GT_TMP INTO GS_TMP.
    MOVE-CORRESPONDING GS_TMP TO GS_DATA.

    "전년도 평균단가
    IF GS_DATA-TOT_MENGE <> 0.
      GS_DATA-AVRGE = GS_DATA-TOT_NETWR / GS_DATA-TOT_MENGE.

    ENDIF.

    "월별 단가 & 차이
    DO 12 TIMES.
      L_INDEX = SY-INDEX.

      _ASSIGN2: 'NETWR' <NETWR>,
                 'MENGE' <MENGE>, "월별 총수량
                 'NETPR' <NETPR>, "월 단가
                 'DIFFR' <DIFFR>. "차이총액

      IF <MENGE> <> 0.
        <NETPR> = <NETWR> / <MENGE>.
      ENDIF.

      IF GS_DATA-AVRGE <> 0.
        <DIFFR> = ( ( GS_DATA-AVRGE * <MENGE> ) - <NETWR> ) * -1.
      ENDIF.

      "Cell Control
      CLEAR L_FIELDNM.
      CONCATENATE 'DIFFR' L_INDEX INTO L_FIELDNM.
      IF <NETWR> <> 0.
        IF <DIFFR> < 0.
          PERFORM FILL_CELLCOL USING '400' L_FIELDNM
                             CHANGING LT_CELLCOL.
        ELSE.
          PERFORM FILL_CELLCOL USING '600' L_FIELDNM
                             CHANGING LT_CELLCOL.
        ENDIF.
      ENDIF.

      ADD <DIFFR> TO GS_DATA-TOT_DIFFR.
    ENDDO.

    IF GS_DATA-TOT_DIFFR <> 0.
      IF GS_DATA-TOT_DIFFR < 0.
        PERFORM FILL_CELLCOL USING '400' 'TOT_DIFFR'
                           CHANGING LT_CELLCOL.
      ELSE.
        PERFORM FILL_CELLCOL USING '600' 'TOT_DIFFR'
                           CHANGING LT_CELLCOL.
      ENDIF.
    ENDIF.

    INSERT LINES OF LT_CELLCOL INTO TABLE GS_DATA-CELLCOL.
    CLEAR LT_CELLCOL.

    SELECT SINGLE NAME1
    INTO (GS_DATA-NAME1)

    WHERE LIFNR = GS_DATA-LIFNR.

    APPEND GS_DATA TO GT_DATA.
    CLEAR GS_DATA.

  ENDLOOP.

ENDFORM. " DATA_PROCESSING

FORM GET_EKBE CHANGING PS_DATA LIKE GS_WORK.

  DATA LV_INDEX TYPE I.

  LOOP AT GT_EKBE WHERE EBELN = PS_DATA-EBELN
                    AND EBELP = PS_DATA-EBELP
                    AND LFBNR = PS_DATA-BELNR_GR.

    IF LV_INDEX = 0.
      PS_DATA-BELNR_IV = GT_EKBE-BELNR.
      PS_DATA-BUZEI_IV = GT_EKBE-BUZEI.
      PS_DATA-MENGE_IV = GT_EKBE-MENGE.
      PS_DATA-GJAHR_IV = GT_EKBE-GJAHR.
      PS_DATA-DMBTR_IV = GT_EKBE-DMBTR.
      PS_DATA-WAERS_IV = GT_EKBE-WAERS.

      IF GT_EKBE-SHKZG  = 'H'.
        PS_DATA-DMBTR_IV = PS_DATA-DMBTR_IV * -1.
        PS_DATA-MENGE_IV = PS_DATA-MENGE_IV * -1.
      ENDIF.
    ELSE.
      MOVE-CORRESPONDING PS_DATA TO GT_WORK2.

      GT_WORK2-BELNR_IV = GT_EKBE-BELNR.
      GT_WORK2-BUZEI_IV = GT_EKBE-BUZEI.
      GT_WORK2-MENGE_IV = GT_EKBE-MENGE.
      GT_WORK2-GJAHR_IV = GT_EKBE-GJAHR.
      GT_WORK2-DMBTR_IV = GT_EKBE-DMBTR.
      GT_WORK2-WAERS_IV = GT_EKBE-WAERS.

      IF GT_EKBE-SHKZG  = 'H'.
        GT_WORK2-DMBTR_IV = GT_WORK2-DMBTR_IV * -1.
        GT_WORK2-MENGE_IV = GT_WORK2-MENGE_IV * -1.
      ENDIF.

      APPEND GT_WORK2.
      CLEAR GT_WORK2.
    ENDIF.

    LV_INDEX = LV_INDEX + 1.
  ENDLOOP.


ENDFORM. " GET_EKBE
