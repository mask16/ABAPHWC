    IF LV_EXIT = ''.
      P_LAYOUT = GS_VARI-VARIANT.

  ENDIF.

FORM VARIANT_INIT USING P_LAYOUT.

  CLEAR GS_VARI.
  GS_VARI-REPORT   = SY-REPID.
  GS_VARI-USERNAME = SY-UNAME.

  IF P_LAYOUT <> ''.
    GS_VARI-VARIANT = P_LAYOUT.
  ENDIF.

  CALL FUNCTION 'LVC_VARIANT_DEFAULT_GET'
    EXPORTING
      I_SAVE        = 'A'
    CHANGING
      CS_VARIANT    = GS_VARI
    EXCEPTIONS
      WRONG_INPUT   = 1
      NOT_FOUND     = 2
      PROGRAM_ERROR = 3
      OTHERS        = 4.

  CASE SY-SUBRC.
    WHEN 0.
      P_LAYOUT = GS_VARI-VARIANT.
    WHEN 2.
      CLEAR P_LAYOUT.
  ENDCASE.

ENDFORM. " VARIANT_INIT

FORM COUNT_DATA TABLES PT_TABLE TYPE TABLE.

  DATA: L_CNT TYPE I.
  CLEAR L_CNT.

  IF PT_TABLE[] IS NOT INITIAL AND LINES( PT_TABLE ) GT 0.
    L_CNT = LINES( PT_TABLE ).
    MESSAGE S050 WITH L_CNT.
  ELSE.
    MESSAGE S009 DISPLAY LIKE 'W'.
    LEAVE LIST-PROCESSING.
  ENDIF.
ENDFORM. " COUNT_DATA
