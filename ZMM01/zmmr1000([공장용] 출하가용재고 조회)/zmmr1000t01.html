*----------------------------------------------------------------------*
* SELECTION-SCREEN 이벤트
*----------------------------------------------------------------------*
AT SELECTION-SCREEN ON S_WERKS.
  LOOP AT S_WERKS.
    IF S_WERKS-LOW <> '1010' AND
       S_WERKS-LOW <> '1020' AND
       S_WERKS-LOW <> '1030' AND
       S_WERKS-LOW <> '2050' AND
       S_WERKS-LOW <> '3060' .
      MESSAGE '플랜트 1010, 1020, 1030, 2050, 3060만 조회 가능합니다.' TYPE 'E'.
    ENDIF.
    IF S_WERKS-HIGH IS NOT INITIAL AND
       ( S_WERKS-HIGH <> '1010' AND
         S_WERKS-HIGH <> '1020' AND
         S_WERKS-HIGH <> '1030' AND
         S_WERKS-HIGH <> '2050'AND
         S_WERKS-HIGH <> '3060').
      MESSAGE '플랜트 1010, 1020, 1030, 2050, 3060만 조회 가능합니다.' TYPE 'E'.
    ENDIF.
  ENDLOOP.

AT SELECTION-SCREEN ON S_LGORT.
  DATA: LV_LGORT TYPE ZMMT1000-LGORT.

  " 저장위치 입력 값 검증
  LOOP AT S_LGORT.
    " LOW 값 검증
    IF S_LGORT-LOW IS NOT INITIAL.
      SELECT SINGLE LGORT  INTO LV_LGORT  FROM ZMMT1000
       WHERE LGORT = S_LGORT-LOW.
      IF SY-SUBRC <> 0.
        MESSAGE '공장 저장위치만 조회 가능합니다.' TYPE 'E'.
      ENDIF.
    ENDIF.

    " HIGH 값 검증
    IF S_LGORT-HIGH IS NOT INITIAL.
      SELECT SINGLE LGORT  INTO LV_LGORT  FROM ZMMT1000
       WHERE LGORT = S_LGORT-HIGH.
      IF SY-SUBRC <> 0.
        MESSAGE '공장 저장위치만 조회 가능합니다.' TYPE 'E'.
      ENDIF.
    ENDIF.
  ENDLOOP.


AT SELECTION-SCREEN.
