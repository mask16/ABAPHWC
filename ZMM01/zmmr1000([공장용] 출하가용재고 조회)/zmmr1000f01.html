FORM SELECT_DATA.

  PERFORM SET_WERKS_LGORT CHANGING LT_WERKS LT_LGORT.

  PERFORM CALCULATE_ZK1FD_OPEN_QTY.
  PERFORM CALCULATE_PO_OPEN_QTY.
  PERFORM SUMMARIZE_OPEN_QTY.

*&      FORM GET_SO_PO_DATA
*&---------------------------------------------------------------------*
*       SO, PO 데이터 조회
*----------------------------------------------------------------------*
FORM GET_SO_PO_DATA.
  CLEAR: LT_ZSDT1150, LT_SO, LT_PO_UB.
  LV_DATE_FROM = SY-DATUM - 365.

  SELECT VBELN POSNR ERDAT STEPS
      INTO TABLE LT_ZSDT1150
      FROM ZSDT1150
      WHERE ERDAT BETWEEN LV_DATE_FROM AND SY-DATUM
        AND STEPS = 'F'.

  " SO, PO 분리
  LOOP AT LT_ZSDT1150 INTO LS_ZSDT1150.
    IF LS_ZSDT1150-VBELN CP '4*'.
      APPEND LS_ZSDT1150 TO LT_PO_UB.
    ELSEIF LS_ZSDT1150-VBELN CP '0*'.
      APPEND LS_ZSDT1150 TO LT_SO.
    ENDIF.
  ENDLOOP.

  " SO 데이터 조회
  CLEAR: LT_SO_ALL, LT_SO_ZK1FD, LT_SO_NON_ZK1FD.
  IF LT_SO IS NOT INITIAL.
    SELECT A~VBELN B~AUART A~POSNR A~MATNR A~WERKS A~LGORT A~KWMENG A~MEINS M~MTART M~MATKL
      INTO TABLE LT_SO_ALL
      FROM VBAP AS A
      INNER JOIN VBAK AS B ON A~VBELN = B~VBELN
      INNER JOIN MARA AS M ON A~MATNR = M~MATNR
      FOR ALL ENTRIES IN LT_SO
      WHERE A~VBELN = LT_SO-VBELN
        AND A~POSNR = LT_SO-POSNR
        AND A~MATNR IN S_MATNR
        AND M~MTART IN S_MTART
        AND M~MATKL IN S_MATKL
        AND A~WERKS IN LT_WERKS
        AND A~LGORT IN LT_LGORT
        AND B~AUART NOT IN ('ZRE', 'ZK2').  "반품 제외
    " ZK1FD, ZFD 분리
    LOOP AT LT_SO_ALL INTO LS_VBAK.
      IF LS_VBAK-AUART = 'ZK1' OR LS_VBAK-AUART = 'ZFD' .
        APPEND LS_VBAK TO LT_SO_ZK1FD.
      ELSE.
        APPEND LS_VBAK TO LT_SO_NON_ZK1FD.
      ENDIF.
    ENDLOOP.
  ENDIF.


  CLEAR: LT_PO_UB_TEMP, LT_PO_EKPO.
  IF LT_PO_UB IS NOT INITIAL.
    " POSNR을 NUMC 5로 변환
    LOOP AT LT_PO_UB INTO LS_PO_UB.
      CLEAR LS_PO_UB_TEMP.
      MOVE-CORRESPONDING LS_PO_UB TO LS_PO_UB_TEMP.
      LS_PO_UB_TEMP-POSNR_NUMC = LS_PO_UB-POSNR. " CHAR 6 -> NUMC 5 변환
      APPEND LS_PO_UB_TEMP TO LT_PO_UB_TEMP.
    ENDLOOP.

    " PO 데이터 조회
    SELECT K~BSART E~EBELN E~EBELP E~MATNR E~WERKS E~RESLO E~MENGE E~MEINS M~MTART M~MATKL
    INTO TABLE LT_PO_EKPO
    FROM EKPO AS E
    INNER JOIN EKKO AS K ON E~EBELN = K~EBELN
    INNER JOIN MARA AS M ON E~MATNR = M~MATNR
    FOR ALL ENTRIES IN LT_PO_UB_TEMP
    WHERE E~EBELN = LT_PO_UB_TEMP-VBELN
      AND E~EBELP = LT_PO_UB_TEMP-POSNR_NUMC
      AND E~MATNR IN S_MATNR
      AND M~MTART IN S_MTART
      AND M~MATKL IN S_MATKL
      AND E~WERKS IN LT_WERKS
      AND E~RESLO IN LT_LGORT
      AND E~LOEKZ NOT IN ('S', 'L').       "품목 보류, 삭제인 경우 제외
  ENDIF.


ENDFORM. "GET_SO_PO_DATA


*&---------------------------------------------------------------------*
*&      FORM CALCULATE_SO_OPEN_QTY
*&---------------------------------------------------------------------*
*       기타 SO(ZK1, ZFD제외)  납품 문서 조회 및 잔량 계산
*----------------------------------------------------------------------*
FORM CALCULATE_SO_OPEN_QTY.

  CLEAR LT_LIPS.
  IF LT_SO_NON_ZK1FD IS NOT INITIAL.
    SELECT L~VBELN L~POSNR L~MATNR L~WERKS L~LGORT L~LFIMG L~MEINS
           V~VBELV V~POSNV V~RFMNG V~MEINS AS VBFA_MEINS
      INTO TABLE LT_LIPS
      FROM VBFA AS V
      INNER JOIN LIPS AS L ON L~VBELN = V~VBELN AND L~POSNR = V~POSNN
      FOR ALL ENTRIES IN LT_SO_NON_ZK1FD
      WHERE V~VBELV = LT_SO_NON_ZK1FD-VBELN
        AND V~POSNV = LT_SO_NON_ZK1FD-POSNR
        AND V~VBTYP_N = 'J'.
  ENDIF.

  CLEAR LT_OPEN_QTY.
  LOOP AT LT_SO_NON_ZK1FD INTO LS_SO_NON_ZK1FD.
    CLEAR LS_OPEN_QTY.
    LS_OPEN_QTY-VBELN = LS_SO_NON_ZK1FD-VBELN.
    LS_OPEN_QTY-POSNR = LS_SO_NON_ZK1FD-POSNR.
    LS_OPEN_QTY-MATNR = LS_SO_NON_ZK1FD-MATNR.
    LS_OPEN_QTY-WERKS = LS_SO_NON_ZK1FD-WERKS.
    LS_OPEN_QTY-LGORT = LS_SO_NON_ZK1FD-LGORT.
    LS_OPEN_QTY-KWMENG = LS_SO_NON_ZK1FD-KWMENG.
    LS_OPEN_QTY-MEINS = LS_SO_NON_ZK1FD-MEINS.
    LS_OPEN_QTY-AUART = LS_SO_NON_ZK1FD-AUART.
    LS_OPEN_QTY-MTART = LS_SO_NON_ZK1FD-MTART.
    LS_OPEN_QTY-MATKL = LS_SO_NON_ZK1FD-MATKL.

    CLEAR: LS_OPEN_QTY-RFMNG, LS_OPEN_QTY-LFIMG, LV_P_LGORT, LV_LFIMG_TOTAL.
    LS_OPEN_QTY-OPEN_QTY = LS_SO_NON_ZK1FD-KWMENG.

* LT_LIPS에서 납품 데이터 처리
    LOOP AT LT_LIPS INTO LS_LIPS WHERE VBELV = LS_SO_NON_ZK1FD-VBELN
                                   AND POSNV = LS_SO_NON_ZK1FD-POSNR.
* RFMNG 누적 (VBFA의 납품 수량)
      LS_OPEN_QTY-RFMNG = LS_OPEN_QTY-RFMNG + LS_LIPS-RFMNG.

* LIPS 처리
      IF LS_LIPS-LGORT IN LT_LGORT.
        LV_P_LGORT = 'X'.
      ELSE.
        LS_OPEN_QTY-LFIMG = LS_OPEN_QTY-LFIMG + LS_LIPS-LFIMG.
        LV_LFIMG_TOTAL = LV_LFIMG_TOTAL + LS_LIPS-LFIMG.
        LS_OPEN_QTY-OPEN_QTY = LS_OPEN_QTY-OPEN_QTY - LS_LIPS-LFIMG.
      ENDIF.
    ENDLOOP.

    IF LS_OPEN_QTY-OPEN_QTY > 0.
      APPEND LS_OPEN_QTY TO LT_OPEN_QTY.
    ENDIF.
  ENDLOOP.

ENDFORM. "CALCULATE_SO_OPEN_QTY

*&---------------------------------------------------------------------*
*&      FORM CALCULATE_ZK1FD_OPEN_QTY
*&---------------------------------------------------------------------*
*       ZK1, ZFD 자재이동 문서 조회 및 잔량 계산
*----------------------------------------------------------------------*
FORM CALCULATE_ZK1FD_OPEN_QTY.
  CLEAR LT_VBFA_ZK1FD.
  IF LT_SO_ZK1FD IS NOT INITIAL.
    SELECT V~VBELV V~POSNV V~VBTYP_N V~VBELN V~RFMNG V~MEINS
      INTO TABLE LT_VBFA_ZK1FD
      FROM VBFA AS V
      FOR ALL ENTRIES IN LT_SO_ZK1FD
      WHERE V~VBELV = LT_SO_ZK1FD-VBELN
        AND V~POSNV = LT_SO_ZK1FD-POSNR
        AND V~VBTYP_N = 'R'.
  ENDIF.

  CLEAR LT_ZK1FD_OPEN_QTY.
  IF LT_SO_ZK1FD IS NOT INITIAL.
    LOOP AT LT_SO_ZK1FD INTO LS_SO_ZK1FD.
      CLEAR LS_ZK1FD_OPEN_QTY.
      MOVE-CORRESPONDING LS_SO_ZK1FD TO LS_ZK1FD_OPEN_QTY.
      LS_ZK1FD_OPEN_QTY-OPEN_QTY = LS_SO_ZK1FD-KWMENG.

      DATA: LV_RFMNG_TOTAL TYPE VBFA-RFMNG.
      CLEAR LV_RFMNG_TOTAL.
      LOOP AT LT_VBFA_ZK1FD INTO LS_VBFA_ZK1FD
        WHERE VBELV = LS_SO_ZK1FD-VBELN
          AND POSNV = LS_SO_ZK1FD-POSNR
          AND VBTYP_N = 'R'.
        LV_RFMNG_TOTAL = LV_RFMNG_TOTAL + LS_VBFA_ZK1FD-RFMNG.
      ENDLOOP.

      IF LV_RFMNG_TOTAL IS INITIAL.
        LS_ZK1FD_OPEN_QTY-OPEN_QTY = LS_SO_ZK1FD-KWMENG. " R이 없으면 KWMENG 유지
      ELSE.
        LS_ZK1FD_OPEN_QTY-OPEN_QTY = LS_SO_ZK1FD-KWMENG - LV_RFMNG_TOTAL. " R 있으면 차감
      ENDIF.


      IF LS_ZK1FD_OPEN_QTY-OPEN_QTY  > 0.
        APPEND LS_ZK1FD_OPEN_QTY TO LT_ZK1FD_OPEN_QTY.
      ENDIF.

    ENDLOOP.
  ENDIF.


ENDFORM. "CALCULATE_ZK1FD_OPEN_QTY


*&---------------------------------------------------------------------*
*&      FORM CALCULATE_PO_OPEN_QTY
*&---------------------------------------------------------------------*
*       재고전송오더 납품문서 조회 및 잔량 계산
*----------------------------------------------------------------------*
FORM CALCULATE_PO_OPEN_QTY.
  CLEAR: LT_PO_EKBE, LT_PO_OPEN_QTY.
  IF LT_PO_EKPO IS NOT INITIAL.
    " EKBE에서 납품문서 조회
    SELECT E~EBELN E~EBELP E~VGABE E~BELNR E~MENGE P~RESLO
      INTO TABLE LT_PO_EKBE
      FROM EKBE AS E
      INNER JOIN EKPO AS P ON E~EBELN = P~EBELN AND E~EBELP = P~EBELP
      FOR ALL ENTRIES IN LT_PO_EKPO
      WHERE E~EBELN = LT_PO_EKPO-EBELN
        AND E~EBELP = LT_PO_EKPO-EBELP
        AND E~VGABE IN ('6', '8')
        AND P~MATNR IN S_MATNR
        AND P~WERKS IN LT_WERKS
        AND P~RESLO IN LT_LGORT.

    " PO 잔여수량 계산
    LOOP AT LT_PO_EKPO INTO LS_PO_EKPO.
      CLEAR LS_PO_OPEN_QTY.
      LS_PO_OPEN_QTY-AUART = LS_PO_EKPO-BSART.
      LS_PO_OPEN_QTY-VBELN = LS_PO_EKPO-EBELN.
      LS_PO_OPEN_QTY-POSNR = LS_PO_EKPO-EBELP.
      LS_PO_OPEN_QTY-MATNR = LS_PO_EKPO-MATNR.
      LS_PO_OPEN_QTY-WERKS = LS_PO_EKPO-WERKS.
      LS_PO_OPEN_QTY-LGORT = LS_PO_EKPO-RESLO.
      LS_PO_OPEN_QTY-KWMENG = LS_PO_EKPO-MENGE.
      LS_PO_OPEN_QTY-OPEN_QTY = LS_PO_EKPO-MENGE.
      LS_PO_OPEN_QTY-MEINS = LS_PO_EKPO-MEINS.
      LS_PO_OPEN_QTY-MTART = LS_PO_EKPO-MTART.
      LS_PO_OPEN_QTY-MATKL = LS_PO_EKPO-MATKL.
*    MOVE-CORRESPONDING LS_PO_EKPO TO LS_PO_OPEN_QTY.

      " EKBE 데이터 확인
      READ TABLE LT_PO_EKBE TRANSPORTING NO FIELDS
        WITH KEY EBELN = LS_PO_EKPO-EBELN
                 EBELP = LS_PO_EKPO-EBELP.
      IF SY-SUBRC = 0. " EKBE 데이터가 있는 경우
        " VGABE = '8'인 납품 수량 합산
        CLEAR LV_OPEN_MENGE_TOTAL_8.
        LOOP AT LT_PO_EKBE INTO LS_PO_EKBE
          WHERE EBELN = LS_PO_EKPO-EBELN
            AND EBELP = LS_PO_EKPO-EBELP
            AND VGABE = '8'.
          LV_OPEN_MENGE_TOTAL_8 = LV_OPEN_MENGE_TOTAL_8 + LS_PO_EKBE-MENGE.
        ENDLOOP.

        " VGABE = '6'인 납품 수량 합산
        CLEAR LV_OPEN_MENGE_TOTAL_6.
        LOOP AT LT_PO_EKBE INTO LS_PO_EKBE
          WHERE EBELN = LS_PO_EKPO-EBELN
            AND EBELP = LS_PO_EKPO-EBELP
            AND VGABE = '6'.
          LV_OPEN_MENGE_TOTAL_6 = LV_OPEN_MENGE_TOTAL_6 + LS_PO_EKBE-MENGE.
        ENDLOOP.

        " 잔여수량 계산
        IF LV_OPEN_MENGE_TOTAL_8 IS INITIAL AND LV_OPEN_MENGE_TOTAL_6 IS INITIAL.
          LS_PO_OPEN_QTY-OPEN_QTY = LS_PO_EKPO-MENGE.
        ELSEIF LV_OPEN_MENGE_TOTAL_6 IS INITIAL.
          LS_PO_OPEN_QTY-OPEN_QTY = LS_PO_EKPO-MENGE.
        ELSE.
          LS_PO_OPEN_QTY-OPEN_QTY = LS_PO_EKPO-MENGE - LV_OPEN_MENGE_TOTAL_6.
        ENDIF.
      ENDIF.


      IF LS_PO_OPEN_QTY-OPEN_QTY > 0.
        APPEND LS_PO_OPEN_QTY TO LT_PO_OPEN_QTY.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDFORM. "CALCULATE_PO_OPEN_QTY

*&---------------------------------------------------------------------*
*&      FORM SUMMARIZE_OPEN_QTY
*&---------------------------------------------------------------------*
*       SO 및 PO 수주 잔량 합산 (자재, 플랜트, 저장위치, 단위별)
*---------------------------------------------------------------------*
FORM SUMMARIZE_OPEN_QTY.

  CLEAR LT_OPEN_QTY_SUM.

  " 비 ZK1, ZFD SO 잔량 합산
  LOOP AT LT_OPEN_QTY INTO LS_OPEN_QTY.
    CLEAR LS_OPEN_QTY_SUM.
    LS_OPEN_QTY_SUM-MATNR = LS_OPEN_QTY-MATNR.
    LS_OPEN_QTY_SUM-WERKS = LS_OPEN_QTY-WERKS.
    LS_OPEN_QTY_SUM-LGORT = LS_OPEN_QTY-LGORT.
    LS_OPEN_QTY_SUM-MEINS = LS_OPEN_QTY-MEINS.
    LS_OPEN_QTY_SUM-OPEN_QTY = LS_OPEN_QTY-OPEN_QTY.
    COLLECT LS_OPEN_QTY_SUM INTO LT_OPEN_QTY_SUM.
  ENDLOOP.

  " ZK1, ZFD SO 잔량 합산
  LOOP AT LT_ZK1FD_OPEN_QTY INTO LS_ZK1FD_OPEN_QTY.
    CLEAR LS_OPEN_QTY_SUM.
    LS_OPEN_QTY_SUM-MATNR = LS_ZK1FD_OPEN_QTY-MATNR.
    LS_OPEN_QTY_SUM-WERKS = LS_ZK1FD_OPEN_QTY-WERKS.
    LS_OPEN_QTY_SUM-LGORT = LS_ZK1FD_OPEN_QTY-LGORT.
    LS_OPEN_QTY_SUM-MEINS = LS_ZK1FD_OPEN_QTY-MEINS.
    LS_OPEN_QTY_SUM-OPEN_QTY = LS_ZK1FD_OPEN_QTY-OPEN_QTY.
    COLLECT LS_OPEN_QTY_SUM INTO LT_OPEN_QTY_SUM.
  ENDLOOP.

  " PO 잔량 합산
  LOOP AT LT_PO_OPEN_QTY INTO LS_PO_OPEN_QTY.
    CLEAR LS_OPEN_QTY_SUM.
    LS_OPEN_QTY_SUM-MATNR = LS_PO_OPEN_QTY-MATNR.
    LS_OPEN_QTY_SUM-WERKS = LS_PO_OPEN_QTY-WERKS.
    LS_OPEN_QTY_SUM-LGORT = LS_PO_OPEN_QTY-LGORT.
    LS_OPEN_QTY_SUM-MEINS = LS_PO_OPEN_QTY-MEINS.
    LS_OPEN_QTY_SUM-OPEN_QTY = LS_PO_OPEN_QTY-OPEN_QTY.
    COLLECT LS_OPEN_QTY_SUM INTO LT_OPEN_QTY_SUM.
  ENDLOOP.


ENDFORM. "SUMMARIZE_OPEN_QTY

*&      FORM GET_STOCK_DATA
*&---------------------------------------------------------------------*
*       재고 데이터 조회
*---------------------------------------------------------------------*
FORM GET_STOCK_DATA.

  CLEAR GT_DATA.
  SELECT A~MATNR A~MTART A~MATKL A~MEINS B~MAKTX C~WERKS D~NAME1
         C~LGORT E~LGOBE C~LABST C~SPEME
    INTO CORRESPONDING FIELDS OF TABLE GT_DATA
    FROM MARA AS A
    INNER JOIN MAKT AS B ON A~MATNR = B~MATNR AND B~SPRAS = SY-LANGU
    INNER JOIN MARD AS C ON A~MATNR = C~MATNR
    INNER JOIN T001W AS D ON C~WERKS = D~WERKS
    LEFT JOIN T001L AS E ON C~WERKS = E~WERKS AND C~LGORT = E~LGORT
    WHERE A~MATNR IN S_MATNR
      AND A~MTART IN S_MTART
      AND A~MATKL IN S_MATKL
      AND C~WERKS IN LT_WERKS
      AND C~LGORT IN LT_LGORT.

  IF GT_DATA IS INITIAL.
    MESSAGE E052(ZMMM).
    EXIT.
  ENDIF.

ENDFORM. "GET_STOCK_DATA

*&---------------------------------------------------------------------*
*&      FORM FINALIZE_DATA.
*&---------------------------------------------------------------------*
*       재고 데이터 , 잔량 조회
*---------------------------------------------------------------------*
FORM FINALIZE_DATA.

  DATA: LS_DATA TYPE TY_DATA,
        LT_FILTERED_DATA TYPE TABLE OF TY_DATA.

  CLEAR LT_FILTERED_DATA.
  LOOP AT GT_DATA INTO LS_DATA.

    READ TABLE LT_OPEN_QTY_SUM INTO LS_OPEN_QTY_SUM
      WITH KEY MATNR = LS_DATA-MATNR
               WERKS = LS_DATA-WERKS
               LGORT = LS_DATA-LGORT.

    IF SY-SUBRC = 0.
      LS_DATA-OPEN_QTY = LS_OPEN_QTY_SUM-OPEN_QTY.
    ELSE.
      LS_DATA-OPEN_QTY = 0.
    ENDIF.


    IF LS_DATA-LABST <> 0 OR LS_DATA-OPEN_QTY <> 0.
      " 출하 가용 재고 계산
      LS_DATA-OUTBOUND_LABST = LS_DATA-LABST - LS_DATA-OPEN_QTY.
      APPEND LS_DATA TO LT_FILTERED_DATA.
    ENDIF.
  ENDLOOP.

  CLEAR GT_DATA.
  GT_DATA = LT_FILTERED_DATA.

  IF GT_DATA IS INITIAL.
    MESSAGE E052(ZMMM).
    EXIT.
  ENDIF.

  SORT GT_DATA BY MATNR WERKS LGORT.


ENDFORM. "FINALIZE_DATA



*&---------------------------------------------------------------------*
*&      FORM CHECK_ORDER
*&---------------------------------------------------------------------*
*       수주잔량 오더 조회
*---------------------------------------------------------------------*
FORM CHECK_ORDER.


  DATA: LV_USNAM TYPE ZMMT1001-USNAM.

  " 본사, 공장 출하팀만 조회 가능
  SELECT SINGLE USNAM
    INTO LV_USNAM
    FROM ZMMT1001
    WHERE USNAM = SY-UNAME.
  IF SY-SUBRC <> 0.
    MESSAGE '조회 권한이 없습니다. (출하팀만 조회 가능)' TYPE 'E'.
    EXIT.
  ENDIF.

  REFRESH GT_CHECK_ORDER.

  " LT_OPEN_QTY 데이터 통합

    APPEND GS_CHECK_ORDER TO GT_CHECK_ORDER.

  ENDIF.

      FOR ALL ENTRIES IN GT_CHECK_ORDER

  ENDLOOP.


  SORT GT_CHECK_ORDER  BY ERDAT VBELN POSNR.


  CALL SCREEN 0200.

ENDFORM. "CHECK_ORDER
