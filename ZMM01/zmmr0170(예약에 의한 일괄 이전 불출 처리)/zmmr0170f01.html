*  <--  p2        text
*----------------------------------------------------------------------*
FORM get_data .
  CLEAR : gt_data , gt_data[].
  DATA lv_where TYPE string.

  CASE 'X'.
    WHEN pa_311.
      CONCATENATE 'A~BWART = '
                  ''''
                 '311'
                 ''''
               INTO lv_where.
    WHEN pa_301.
      CONCATENATE 'A~BWART = '
            ''''
           '301'
           ''''
         INTO lv_where.
    WHEN pa_201.
      CONCATENATE 'A~BWART = '
            ''''
           '201'
           ''''
         INTO lv_where.
  ENDCASE.

  SELECT  b~rsnum
          b~rspos
          b~werks
          b~umwrk
          b~lgort
          b~umlgo
          b~matnr
          b~charg
          b~bdmng
          b~meins
          b~bwart
          a~kostl
          b~banfn
          b~bnfpo
          b~bdter
          a~wempf
          a~rsdat
          a~usnam
          b~enmng
          b~kzear
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM resb AS b
    INNER JOIN rkpf AS a
      ON a~rsnum = b~rsnum
    INNER JOIN mara AS m
      ON b~matnr = m~matnr
    WHERE b~xloek NE 'X'
      AND b~rsnum IN so_rsnum
      AND (lv_where)
*      and a~bwart = pa_bwart
      AND b~wempf <> 'PM'
      AND a~wempf IN so_wempf
      AND b~bdter IN so_bdter
      AND a~usnam IN so_usnam
      AND a~rsdat IN so_rsdat
      AND b~werks IN so_werks
      AND b~lgort IN so_lgort
      AND b~umlgo IN so_umlgo
      AND b~matnr IN so_matnr
      AND m~mtart IN so_mtart
      AND m~matkl IN so_matkl.


  SORT gt_data BY rsnum rspos.

  LOOP AT gt_data.

    SELECT SINGLE maktx
      INTO gt_data-maktx
      FROM makt
     WHERE matnr = gt_data-matnr
       AND spras = sy-langu.

    SELECT SINGLE labst
       INTO gt_data-labst
       FROM mard
       WHERE matnr = gt_data-matnr
         AND werks = gt_data-werks
         AND lgort = gt_data-lgort.
    gt_data-menge_re  = gt_data-bdmng - gt_data-enmng.

    IF gt_data-menge_re < 0.
      gt_data-menge_re = 0.
    ENDIF.

    IF gt_data-kzear = '' AND gt_data-enmng = 0.
      gt_data-icon2 = gc_red.
    ENDIF.
    IF gt_data-kzear = '' AND gt_data-enmng <> 0.
      gt_data-icon2 = gc_yellow.
    ENDIF.
    IF gt_data-kzear = 'X' .
      gt_data-icon2 = gc_green.
    ENDIF.

    gt_data-inpmg = gt_data-menge_re.
    gt_data-inpdt = sy-datum.

    MODIFY gt_data.
    CLEAR gt_data.
  ENDLOOP.

  IF pa_not = 'X'.
    DELETE gt_data WHERE kzear NE ''.
  ENDIF.
  IF pa_com = 'X'.
    DELETE gt_data WHERE kzear NE 'X'.
  ENDIF.

ENDFORM. " GET_DATA
