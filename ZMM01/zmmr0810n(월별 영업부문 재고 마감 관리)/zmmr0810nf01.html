FORM get_data .

  CLEAR: gt_data.

  SELECT a~bzirk b~spmon b~confm c~bztxt
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM t171 AS a
  LEFT JOIN zmmt0341 AS b ON a~bzirk = b~bzirk
                         AND b~spmon = p_spmon
  LEFT JOIN t171t AS c ON a~bzirk = c~bzirk
                      AND c~spras = sy-langu.

  LOOP AT gt_data INTO gs_data.

    IF gs_data-confm EQ 'X'.
      gs_data-contx = '마감'.
    ELSE.
      SELECT SINGLE spmon
      INTO gs_data-spmon
      FROM zmmt0331
      WHERE spmon = p_spmon
        AND bzirk = gs_data-bzirk.

      IF sy-subrc = 0.
        gs_data-contx = '수정가능'.
      ELSE.
        gs_data-contx = '미등록'.
      ENDIF.
    ENDIF.

    gs_data-spmon = p_spmon.

    MODIFY gt_data FROM gs_data.
  ENDLOOP.

  IF p_noout = 'X'.
    DELETE gt_data WHERE contx = '미등록'.
  ENDIF.

ENDFORM. " GET_DATA
