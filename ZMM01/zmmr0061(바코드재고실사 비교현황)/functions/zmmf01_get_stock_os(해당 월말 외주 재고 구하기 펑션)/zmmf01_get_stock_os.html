FUNCTION ZMMF01_GET_STOCK_OS.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_MATNR) TYPE MATNR
*" REFERENCE(I_WERKS) TYPE WERKS_D
*" REFERENCE(I_LIFNR) TYPE LIFNR
*" REFERENCE(I_SDATE) TYPE SPMON
*" REFERENCE(I_MTART) TYPE MTART OPTIONAL
*" EXPORTING
*" REFERENCE(O_LABST) TYPE LABST
*" REFERENCE(O_VPRSV) TYPE VPRSV
*" REFERENCE(O_VERPR) TYPE VERPR
*" REFERENCE(O_STPRS) TYPE STPRS
*" REFERENCE(O_DMBTR) TYPE DMBTR
*" REFERENCE(O_SALK3) TYPE DMBTR
*" REFERENCE(O_LBKUM) TYPE LBKUM
*" TABLES
*" T_MSLB STRUCTURE MSLBH OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA TMP_SPMON TYPE SPMON.
  DATA : P_MATNR TYPE MATNR,
         P_WERKS TYPE WERKS_D,
         P_LIFNR TYPE LIFNR,
         P_SDATE TYPE SPMON.
  DATA : V_SDATE  TYPE SY-DATUM.

  DATA V_ICNT0 TYPE I.

  P_MATNR   = I_MATNR.
  P_WERKS   = I_WERKS.
  P_LIFNR   = I_LIFNR.
  P_SDATE   = I_SDATE.

  V_SDATE =  P_SDATE.
  V_SDATE+6(2) = '01'.

* 구하는 월 >= 현재월 일때는 MSLB 에서만 읽음 된다....
  IF I_SDATE >= SY-DATUM+0(6).
    SELECT SINGLE LBLAB INTO O_LABST
        FROM      MSLB
       WHERE MATNR = P_MATNR
        AND WERKS = P_WERKS
        AND LIFNR = P_LIFNR.
  ELSE.
    SELECT LFGJA LFMON LBLAB
     APPENDING CORRESPONDING FIELDS OF TABLE T_MSLB
        FROM      MSLB
         WHERE MATNR = P_MATNR
           AND WERKS = P_WERKS
           AND LIFNR = P_LIFNR.
    SELECT LFGJA LFMON LBLAB
      APPENDING CORRESPONDING FIELDS OF TABLE T_MSLB
       FROM      MSLBH
         WHERE MATNR = P_MATNR
            AND WERKS = P_WERKS
            AND LIFNR = P_LIFNR.

    SORT T_MSLB BY LFGJA DESCENDING  LFMON DESCENDING.


* 구하는 월이 존재할 경우 해당 Record 를 읽는다

    READ TABLE T_MSLB WITH KEY LFGJA = P_SDATE+0(4)
                               LFMON = P_SDATE+4(2).

    IF SY-SUBRC = 0.
      O_LABST = T_MSLB-LBLAB.
    ELSE.

** 구하는 월이 MARD 날짜 최신 날짜 보다 크면 그 Record 를 가져온다
      READ TABLE T_MSLB INDEX 1.
      CONCATENATE T_MSLB-LFGJA   T_MSLB-LFMON INTO TMP_SPMON .

      IF I_SDATE > TMP_SPMON .
        O_LABST = T_MSLB-LBLAB.
        EXIT.
      ENDIF.
* 구하는 월에 해당하는 Record 가 없으면 + 1개월씩하면서 구한다
      DO 24 TIMES.
        CALL FUNCTION 'MONTH_PLUS_DETERMINE'
          EXPORTING
            MONTHS  = 1
            OLDDATE = V_SDATE
          IMPORTING
            NEWDATE = V_SDATE.


        IF V_SDATE+0(6) = SY-DATUM+0(6).
          READ TABLE T_MSLB INDEX 1.
          O_LABST = T_MSLB-LBLAB.

          EXIT.
        ENDIF.

        READ TABLE T_MSLB WITH KEY LFGJA = V_SDATE+0(4)
                                   LFMON = V_SDATE+4(2)  .

        IF SY-SUBRC = 0.

          O_LABST = T_MSLB-LBLAB.
          EXIT.
        ENDIF.

      ENDDO.

    ENDIF.

  ENDIF.

ENDFUNCTION.
