*       text

FORM matching_format USING p_fieldname
                               p_value.
  CASE p_fieldname.
    WHEN 'MTYPE' OR 'STYPE'.
      "대문자로 변환
      TRANSLATE p_value TO UPPER CASE.

    WHEN 'MENGE' OR 'STPRS'.
      "콤마 제거
      REPLACE ALL OCCURRENCES OF ',' IN p_value
      WITH space.

      CONDENSE p_value NO-GAPS.

    WHEN 'GRMON'.
      "점 제거
      REPLACE ALL OCCURRENCES OF '.' IN p_value
      WITH space.

      CONDENSE p_value NO-GAPS.
  ENDCASE.

ENDFORM. " MATCHING_FORMAT

        s_upload TYPE zmmt0331.

  DATA: t_makt TYPE TABLE OF zmmt0321,
        s_makt TYPE zmmt0321.

  DATA: l_cnt TYPE i.

  READ TABLE gt_data WITH KEY icon1 = icon_led_red
  TRANSPORTING NO FIELDS.

  IF sy-subrc = 0.
    MESSAGE s037 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.


  GET TIME.

  CLEAR: t_upload, t_makt.
  LOOP AT gt_data INTO gs_data.
    MOVE-CORRESPONDING gs_data TO s_upload.

    "중복 데이터 처리
    IF gs_data-icon1 = icon_led_yellow.
      CASE 'X'.
        WHEN p_wr01.  "덮어쓰기
          s_upload-aedat = sy-datum.
          s_upload-aezet = sy-uzeit.
          s_upload-aenam = sy-uname.

        WHEN p_wr02.  "건너뛰기
          CONTINUE.
      ENDCASE.
    ELSE.
      s_upload-erdat = sy-datum.
      s_upload-erzet = sy-uzeit.
      s_upload-ernam = sy-uname.
    ENDIF.

    IF gs_data-mark = 'X'.
      MOVE-CORRESPONDING gs_data TO s_makt.
      APPEND s_makt TO t_makt.
      CLEAR: s_makt.
    ENDIF.

    APPEND s_upload TO t_upload.
    CLEAR s_upload.

  ENDLOOP.

*  IF t_makt IS NOT INITIAL.
*    MODIFY zmmt0321 FROM TABLE t_makt.
*  ENDIF.

  IF t_upload IS NOT INITIAL.
    MODIFY zmmt0331 FROM TABLE t_upload.

    COMMIT WORK.

    "상태 아이콘 업데이트
    gs_data-icon1 = icon_led_green.

*----------------------------------------------------------------------*
FORM get_data .
  CLEAR: gt_data.

  SELECT a~spmon a~bzirk a~lgort a~matnr a~werks
         a~mtype a~grmon a~stype a~gubun a~menge
         a~sname a~waers a~stprs a~erdat
         a~erzet a~ernam a~aedat a~aezet a~aenam
         a~ltext1 a~ltext2 a~ltext3
         b~bztxt
         c~lgobe
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zmmt0331 AS a
  LEFT JOIN t171t AS b ON a~bzirk = b~bzirk
                      AND b~spras = sy-langu
  LEFT JOIN zmmt0311 AS c ON a~bzirk = c~bzirk
                         AND a~lgort = c~lgort
  WHERE a~spmon = p_spmon
    AND a~bzirk IN s_bzirk
    AND a~lgort IN s_lgort
