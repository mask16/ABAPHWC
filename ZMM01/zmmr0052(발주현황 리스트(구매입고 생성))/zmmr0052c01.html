FORM create_object .
  IF g_docking_container1 IS INITIAL.

    CREATE OBJECT g_docking_container1
      EXPORTING
        style     = cl_gui_control=>ws_child
        repid     = sy-cprog
        dynnr     = sy-dynnr
        side      = g_docking_container1->dock_at_left
        lifetime  = cl_gui_control=>lifetime_imode
        extension = 3000
      EXCEPTIONS
        others    = 1.

    CREATE OBJECT grid
      EXPORTING
        i_parent = g_docking_container1.

    PERFORM  alv_init_0100.
    PERFORM  append_header_0100.
    PERFORM  set_cell_attribute.

    CALL METHOD grid->set_table_for_first_display
      EXPORTING
        is_layout            = gs_layout1
        is_variant           = g_variant
        it_toolbar_excluding = gt_exclude
        i_save               = 'A' "x_save
        i_default            = 'X'
      CHANGING
        it_fieldcatalog      = fieldcat1[]
        it_outtab            = gt_data[]
        it_sort              = gt_sort1[].

  ELSE.
    PERFORM  append_header_0100.
    PERFORM  set_cell_attribute.
    PERFORM  refresh_grid USING 'X'.

  ENDIF.

ENDFORM. "CREATE_OBJECT
