FORM alv_data_changed USING er_data_changed
      TYPE REF TO cl_alv_changed_data_protocol
      e_onf4_after TYPE c.

  DATA : ls_good TYPE lvc_s_modi.

  DATA: deleted_row  TYPE lvc_s_moce,
        inserted_row TYPE lvc_s_moce.

* Check deleted data
  LOOP AT er_data_changed->mt_deleted_rows INTO deleted_row.
    CLEAR: gs_data.
    READ TABLE gt_data INDEX deleted_row-row_id INTO gs_data.

    IF sy-subrc = 0.
      APPEND gs_data TO gt_mandelt.
    ENDIF.
  ENDLOOP.


  DATA: lt_celltab TYPE lvc_t_styl,
        l_index TYPE i.
* Check inserted data
  LOOP AT er_data_changed->mt_inserted_rows INTO inserted_row.
    CLEAR: gs_data, lt_celltab.

    PERFORM fill_celltab USING 'RW' 'WERKS'
                         CHANGING lt_celltab.
    PERFORM fill_celltab USING 'RW' 'MATNR'
                         CHANGING lt_celltab.

    INSERT LINES OF lt_celltab INTO TABLE gs_data-celltab.

    INSERT gs_data INTO gt_data INDEX inserted_row-row_id.
  ENDLOOP.

* Check changed data
  IF er_data_changed->mt_inserted_rows[] IS INITIAL.
    LOOP AT er_data_changed->mt_good_cells INTO ls_good.
      CLEAR: gs_data, fieldname.
      READ TABLE gt_data INDEX ls_good-row_id INTO gs_data.

      IF sy-subrc = 0.
*     Get input data.
        CONCATENATE 'GS_DATA-' ls_good-fieldname INTO fieldname.
        ASSIGN (fieldname) TO <fs01>.

        <fs01> = ls_good-value.

        CASE ls_good-fieldname.
          WHEN 'WERKS' OR 'MATNR'.
            IF NOT ( gs_data-werks IS INITIAL AND
                     gs_data-matnr IS INITIAL ).
              PERFORM get_material_info USING gs_data.

              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'NAME1'
                  i_value     = gs_data-name1.

              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'MAKTX'
                  i_value     = gs_data-maktx.

              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'MTART'
                  i_value     = gs_data-mtart.


              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'MTBEZ'
                  i_value     = gs_data-mtbez.

              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'MATKL'
                  i_value     = gs_data-matkl.

              CALL METHOD er_data_changed->modify_cell
                EXPORTING
                  i_row_id    = ls_good-row_id
                  i_tabix     = ls_good-tabix
                  i_fieldname = 'WGBEZ'
                  i_value     = gs_data-wgbez.

            ENDIF.
        ENDCASE.

        UNASSIGN <fs01>.
        MODIFY gt_data FROM gs_data INDEX ls_good-row_id.
        AT END OF row_id.
          CLEAR: lt_celltab.
        ENDAT.
      ENDIF.

    ENDLOOP.
