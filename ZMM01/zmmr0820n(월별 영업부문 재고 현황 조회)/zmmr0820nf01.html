FORM get_default_value .
  DATA lv_pdatum LIKE p0001-begda.

  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      date      = sy-datum
      days      = '00'
      months    = '01'
      signum    = '-'
      years     = '00'
    IMPORTING
      calc_date = lv_pdatum.

  s_spmon = 'IEQ'.
  s_spmon-low = lv_pdatum(6).
  CLEAR: s_spmon-high.
  APPEND s_spmon.

ENDFORM. " GET_DEFAULT_VALUE

FORM get_data .

  CLEAR: gt_data.

  SELECT a~spmon a~bzirk a~lgort a~matnr a~werks
         a~mtype a~grmon a~stype a~menge a~sname
         a~waers a~stprs a~erdat a~erzet
         a~ernam a~aedat a~aezet a~aenam a~gubun
         a~ltext1 a~ltext2 a~ltext3
         b~bztxt
         c~lgobe
         d~confm
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zmmt0331 AS a
  LEFT JOIN t171t AS b ON a~bzirk = b~bzirk
                      AND b~spras = sy-langu
  LEFT JOIN zmmt0311 AS c ON a~bzirk = c~bzirk
                         AND a~lgort = c~lgort
  LEFT JOIN zmmt0341 AS d ON d~spmon = a~spmon
                         AND d~bzirk = a~bzirk
  WHERE a~spmon IN s_spmon
    AND a~bzirk IN s_bzirk
    AND a~lgort IN s_lgort
    AND a~matnr IN s_matnr
    AND a~werks IN s_werks
    AND a~mtype IN s_mtype
    AND a~grmon IN s_grmon
    AND a~stype IN s_stype.

  IF sy-subrc = 0.
    CLEAR: gt_t179.
    SELECT prodh vtext
    INTO CORRESPONDING FIELDS OF TABLE gt_t179
    FROM t179t
    WHERE spras EQ sy-langu.

    SORT gt_t179 BY prodh.
  ENDIF.


ENDFORM. " GET_DATA

FORM data_processing .

  DATA: v_date TYPE sy-datum,
        v_werks TYPE t001w-werks.

  DATA: v_from TYPE string,
        v_where TYPE string.

  TYPES: BEGIN OF ty_stprs,
           matnr TYPE mbewh-matnr,
           bwkey TYPE mbewh-bwkey,
           lfgja TYPE mbewh-lfgja,
           lfmon TYPE mbewh-lfmon,
           value TYPE mbewh-stprs,
         END OF ty_stprs.

  DATA: t_stprs TYPE TABLE OF ty_stprs,
        s_stprs TYPE ty_stprs.


  "재고 없는 라인 삭제
  IF p_zero = 'X'.
    DELETE gt_data WHERE menge = 0.
  ENDIF.

  LOOP AT gt_data INTO gs_data.
    "기준년월
*    gs_data-spmon = p_spmon.

    "자재내역
    SELECT SINGLE a~meins a~groes a~zmaktx a~prdha
           b~maktx
           e~mvgr1 e~mvgr2 e~mvgr3 e~mvgr4 e~mvgr5
    INTO (gs_data-meins, gs_data-groes,
          gs_data-zmaktx, gs_data-prodh,
          gs_data-maktx,
          gs_data-mvgr1, gs_data-mvgr2,
          gs_data-mvgr3, gs_data-mvgr4,
          gs_data-mvgr5)
    FROM mara AS a
    LEFT JOIN makt AS b ON a~matnr = b~matnr
                        AND b~spras = sy-langu
    LEFT JOIN mvke AS e ON e~matnr = a~matnr
                   AND e~vkorg = '1000' "내수기준
                   AND e~vtweg = '10' "내자
    WHERE a~matnr = gs_data-matnr.

    IF sy-subrc <> 0.
      SELECT SINGLE meins maktx
                    mvgr1 mvgr2 mvgr3 mvgr4 mvgr5
      INTO (gs_data-meins, gs_data-maktx
            , gs_data-mvgr1, gs_data-mvgr2, gs_data-mvgr3
            , gs_data-mvgr4, gs_data-mvgr5)
      FROM zmmt0321
      WHERE bzirk = gs_data-bzirk
        AND matnr = gs_data-matnr.
    ENDIF.

    "생산공장
    IF gs_data-werks IS NOT INITIAL.
      PERFORM read_domain_value
        USING 'ZWERKS_D' gs_data-werks
        CHANGING gs_data-name1.

      "생산플랜트 표준원가
      CLEAR: v_werks, t_stprs.

      CASE gs_data-werks.
        WHEN '8000' OR '9000'.  "OEM or 상품
          v_werks = '1010'.
          SELECT matnr bwkey bwtar lfgja lfmon
                 verpr AS value
          INTO CORRESPONDING FIELDS OF TABLE t_stprs
          FROM mbewh
          WHERE matnr = gs_data-matnr
            AND bwkey = v_werks
            AND ( ( lfgja EQ gs_data-spmon(4)
                AND lfmon LE gs_data-spmon+4(2) )
                 OR lfgja LT gs_data-spmon(4) ).

        WHEN '1050' OR '1060'.  "중국 베트남 상품
          SELECT matnr bwkey bwtar lfgja lfmon
                 verpr AS value
          INTO CORRESPONDING FIELDS OF TABLE t_stprs
          FROM mbewh
          WHERE matnr = gs_data-matnr
            AND bwkey = gs_data-werks
            AND ( ( lfgja EQ gs_data-spmon(4)
                AND lfmon LE gs_data-spmon+4(2) )
                 OR lfgja LT gs_data-spmon(4) ).

        WHEN OTHERS.  "국내 제품
          SELECT matnr bwkey bwtar lfgja lfmon
                 stprs AS value
          INTO CORRESPONDING FIELDS OF TABLE t_stprs
          FROM mbewh
          WHERE matnr = gs_data-matnr
            AND bwkey = gs_data-werks
            AND ( ( lfgja EQ gs_data-spmon(4)
                AND lfmon LE gs_data-spmon+4(2) )
                 OR lfgja LT gs_data-spmon(4) ).
      ENDCASE.

      "기준월 최근 표준(변동)단가
      IF t_stprs IS NOT INITIAL.
        SORT t_stprs DESCENDING BY lfgja lfmon.
        READ TABLE t_stprs INDEX 1 INTO s_stprs.

        gs_data-spprs = s_stprs-value.
      ENDIF.
    ENDIF.

    "재고구분
    PERFORM read_domain_value
      USING 'ZMTYPE' gs_data-mtype
      CHANGING gs_data-mtext.

    "판매유형
    IF gs_data-stype IS NOT INITIAL.
      PERFORM read_domain_value
        USING 'ZSTYPE' gs_data-stype
        CHANGING gs_data-stext.
    ENDIF.


    "계층구조
    IF gs_data-prodh IS NOT INITIAL.
      "PH3
      READ TABLE gt_t179 WITH KEY prodh = gs_data-prodh
      BINARY SEARCH INTO gs_t179.

      IF sy-subrc = 0.
        gs_data-vtext = gs_t179-vtext.
      ENDIF.

      "PH1
      gs_data-prodh1 = gs_data-prodh(5).
      READ TABLE gt_t179 WITH KEY prodh = gs_data-prodh1
      BINARY SEARCH INTO gs_t179.

      IF sy-subrc = 0.
        gs_data-vtext1 = gs_t179-vtext.
      ENDIF.

      "PH2
      gs_data-prodh2 = gs_data-prodh(10).
      READ TABLE gt_t179 WITH KEY prodh = gs_data-prodh2
      BINARY SEARCH INTO gs_t179.

      IF sy-subrc = 0.
        gs_data-vtext2 = gs_t179-vtext.
      ENDIF.
    ENDIF.

    "그룹명
    DO 5 TIMES.
      v_numc = sy-index.

      CLEAR: v_fieldname.
      CONCATENATE 'GS_DATA-MVGR' v_numc INTO v_fieldname.
      ASSIGN (v_fieldname) TO <fs01>.

      IF sy-subrc = 0 AND <fs01> IS NOT INITIAL.
        CLEAR: v_fieldname, v_from, v_where.
        CONCATENATE 'GS_DATA-MVTX' v_numc INTO v_fieldname.
        ASSIGN (v_fieldname) TO <fs02>.

        CONCATENATE 'TVM' v_numc 'T' INTO v_from.
        CONCATENATE 'MVGR' v_numc INTO v_where.
        CONCATENATE v_where '= <FS01> AND SPRAS = SY-LANGU'
        INTO v_where SEPARATED BY space.

        SELECT SINGLE bezei
        INTO <fs02>
        FROM (v_from)
        WHERE (v_where).
      ENDIF.
    ENDDO.

    IF gs_data-waers IS NOT INITIAL.

      IF gs_data-waers <> 'KRW'. "외화원가 원화 환산
        gs_data-lwaer = 'KRW'.

        CLEAR: v_date.
        CONCATENATE gs_data-spmon '01' INTO v_date.

        CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
          EXPORTING
            day_in            = v_date
          IMPORTING
            last_day_of_month = v_date
          EXCEPTIONS
            day_in_no_date    = 1
            OTHERS            = 2.

        CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
          EXPORTING
            date             = v_date
            foreign_amount   = gs_data-stprs
            foreign_currency = gs_data-waers
            local_currency   = 'KRW'
            rate             = 0
            type_of_rate     = 'M'
            read_tcurr       = 'X'
          IMPORTING
            exchange_rate    = gs_data-kursf
            local_amount     = gs_data-ttprc
          EXCEPTIONS
            no_rate_found    = 1
            overflow         = 2
            no_factors_found = 3
            no_spread_found  = 4
            derived_2_times  = 5
            OTHERS           = 6.

      ELSE. "원화는 그대로
        gs_data-lwaer = gs_data-waers.
        gs_data-kursf = 1.
        gs_data-ttprc = gs_data-stprs.
      ENDIF.

      "원화단가
      IF gs_data-menge <> 0.
        gs_data-krprc = gs_data-ttprc / gs_data-menge.
      ENDIF.

*      "원화총액
*      gs_data-ttprc = gs_data-krprc * gs_data-menge.
    ENDIF.


    MODIFY gt_data FROM gs_data.

  ENDLOOP.



ENDFORM. " DATA_PROCESSING
