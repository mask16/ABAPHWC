FORM select_data .

  TYPES: BEGIN OF ty_name,
           werks TYPE t001w-werks,
           name1  TYPE t001w-name1,
           lgort TYPE t001l-lgort,
           lgobe TYPE t001l-lgobe,
         END OF ty_name.

  DATA: gt_name TYPE TABLE OF ty_name,
        gs_name TYPE ty_name.


  CLEAR: gt_name.
  SELECT a~werks a~name1 b~lgort b~lgobe
   INTO CORRESPONDING FIELDS OF TABLE gt_name
   FROM t001w AS a
   JOIN t001l AS b ON a~werks = b~werks.

  SORT gt_name BY werks lgort.
  CLEAR: gt_data.
  SELECT a~werks a~lgort a~matnr
         a~grmon a~qualy a~meins a~waers
         a~menge a~netpr a~stext a~ltext
         a~sname a~sales
         a~erdat a~erzet a~ernam
         b~maktx
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zmmt0350 AS a
  LEFT JOIN makt AS b ON a~matnr = b~matnr
                      AND b~spras = sy-langu
  WHERE a~werks IN s_werks
    AND a~lgort IN s_lgort
    AND a~matnr IN s_matnr.

  LOOP AT gt_data INTO gs_data.

    READ TABLE gt_name WITH KEY werks = gs_data-werks
                                      lgort = gs_data-lgort
                                      BINARY SEARCH INTO gs_name.

    IF sy-subrc = 0.
      gs_data-name1 = gs_name-name1.
      gs_data-lgobe = gs_name-lgobe.
    ENDIF.

    gs_data-netwr = gs_data-netpr * gs_data-menge.

    MODIFY gt_data FROM gs_data.

  ENDLOOP.

ENDFORM. " SELECT_DATA
