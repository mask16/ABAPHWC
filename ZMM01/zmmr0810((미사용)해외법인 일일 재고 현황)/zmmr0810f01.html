FORM select_data .

  CLEAR: gt_data.
  SELECT a~bukrs a~lgort a~matnr a~datum
         a~werks a~mapnr
         a~meins a~mzubb a~magbb a~mbwbest
         a~erdat a~erzet a~ernam
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM zmmt0330 AS a
  WHERE a~bukrs IN s_bukrs
    AND a~matnr IN s_matnr
    AND a~datum IN s_datum
    AND a~mapnr IN s_mapnr.


  LOOP AT gt_data INTO gs_data.
    SELECT SINGLE a~butxt a~land1 b~landx
    INTO (gs_data-butxt, gs_data-land1, gs_data-landx)
    FROM zmmt0300 AS a
    LEFT JOIN t005t AS b ON a~land1 = b~land1
                         AND b~spras = sy-langu
    WHERE a~bukrs = gs_data-bukrs.

    SELECT SINGLE lgobe
    INTO (gs_data-lgobe)
    FROM zmmt0310 AS a
    WHERE bukrs  = gs_data-bukrs
      AND lgort = gs_data-lgort.


    SELECT SINGLE maktx
    INTO (gs_data-maktx)
    FROM zmmt0320
    WHERE bukrs = gs_data-bukrs
      AND matnr = gs_data-matnr.

    IF gs_data-mapnr IS NOT INITIAL.
      SELECT SINGLE maktx
      INTO (gs_data-makty)
      FROM makt
      WHERE matnr = gs_data-mapnr
        AND spras = sy-langu.
    ENDIF.

    "생산공장
    IF gs_data-werks IS NOT INITIAL.
      PERFORM read_domain_value
        USING 'ZWERKS_D' gs_data-werks
        CHANGING gs_data-name1.
    ENDIF.

    MODIFY gt_data FROM gs_data.

  ENDLOOP.


ENDFORM. " SELECT_DATA
