      CASE gv_type.
      WHEN 'PR'.
        gt_data-bsart = 'NB'.
        IF  et_tab-knttp = 'A'.
          gt_data-mwskz  ='V2'.

        gt_data-eindt = sy-datum.
      WHEN OTHERS.

        ENDIF.
        gt_data-submi = et_tab-submi.
        gt_data-peinh = et_tab-peinh.

    ENDCASE.

    APPEND gt_data.

  ENDLOOP.

  CALL SCREEN 100.


ENDFUNCTION.
