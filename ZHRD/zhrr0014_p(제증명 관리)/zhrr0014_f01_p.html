FORM FN_CHK_ONLY_SINGLE_MARK USING LT_ITAB LIKE GT_ZHRT9007.

  DATA: L_CNT   TYPE N,
        LS_ITAB LIKE LINE OF LT_ITAB.

  LOOP AT LT_ITAB INTO LS_ITAB WHERE MARK EQ CO_TRUE.
    ADD 1 TO L_CNT.
    IF L_CNT GE 2.
      EXIT.
    ENDIF.
  ENDLOOP.

  CASE SAVE_OK.
    WHEN 'FINS' OR 'FEDI' OR 'FDIS' OR 'FCON'.
      IF L_CNT GE 2.
        MESSAGE S016 DISPLAY LIKE 'E'

        LEAVE TO SCREEN SY-DYNNR.
* 승인(3) 또는 발급(4)상태의 데이타는 변경할 수 없다.
      ELSE.
        IF SAVE_OK = 'FEDI'
        AND ZHRT9007-ISSTATE > 2.
          MESSAGE S016 DISPLAY LIKE 'E'
          WITH '※ 승인또는 발급된 증명서는 변경할 수 없습니다!'.
          LEAVE TO SCREEN SY-DYNNR.
        ENDIF.
      ENDIF.

    WHEN 'FCON' OR 'FPRN'.
      IF L_CNT IS INITIAL.
        MESSAGE S016 DISPLAY LIKE 'E'
        WITH '※ 적절한 데이타를 선택하세요!'.
        LEAVE TO SCREEN SY-DYNNR.
      ENDIF.

  ENDCASE.

FORM FN_GET_GT_ZHRT9007 USING LT_ITAB LIKE GT_ZHRT9007.

  DATA: L_ISSTATE     LIKE ZHRT9007-ISSTATE,
        L_CERTIK      LIKE ZHRT9007-CERTIK,
        L_ORDDAT_LOW  LIKE ZHRT9007-ORDDAT VALUE CO_FRDAT,
        L_ORDDAT_HIGH LIKE ZHRT9007-ORDDAT VALUE CO_LADAT,
        L_PERNR(8) TYPE C.  "LIKE 문을 위해서 CHAR 8로 선언.

*-- 발급상태.
  CASE P_ISSTATE.
    WHEN SPACE OR '1'. "공백이거나, 전체 선택.
      MOVE '%' TO L_ISSTATE.
    WHEN OTHERS.       "신청 또는 승인 또는 발급.
      MOVE P_ISSTATE TO L_ISSTATE.
  ENDCASE.

*-- 신청일자.
*== [ 날짜 ] =================================*
* SPACE ~ SPACE = 0000.00.00 ~ 9999.12.31     *
* SPACE ~ ENDDA = 0000.00.00 ~ ENDDA          *
* BEGDA ~ SPACE = BEGDA ~ 9999.12.31          *
* BEGDA ~ ENDDA = BEGDA ~ ENDDA               *
*=============================================*
  IF     P_BEGDA IS     INITIAL AND P_ENDDA IS     INITIAL.
* 기본 DEFAULT 값 사용.    "◀

  ELSEIF P_BEGDA IS     INITIAL AND P_ENDDA IS NOT INITIAL.
* ENDDA값을 BEGDA로 넘김.
    MOVE P_ENDDA TO : L_ORDDAT_HIGH.

  ELSEIF P_BEGDA IS NOT INITIAL AND P_ENDDA IS     INITIAL.
* BEGDA값을 ENDDA로 넘김.
    MOVE P_BEGDA TO : L_ORDDAT_LOW.

  ELSEIF P_BEGDA IS NOT INITIAL AND P_ENDDA IS NOT INITIAL.
* BEGDA값을 BEGDA로, ENDDA값을 ENDDA로 넘김.
    MOVE P_BEGDA TO : L_ORDDAT_LOW.
    MOVE P_ENDDA TO : L_ORDDAT_HIGH.
  ENDIF.

*-- 증명서 종류.
  MOVE P_CERTIK TO L_CERTIK.

*-- 사원번호.
  CASE P_PERNR.
    WHEN SPACE.  "공백이면 전체.
      MOVE '%' TO L_PERNR.
    WHEN OTHERS.
      MOVE P_PERNR TO L_PERNR.
  ENDCASE.

*-- 발급년도-발급번호.
  RANGES: ZYNKEY FOR ZHRT9007-ZYNKEY.

  IF P_BYEAR IS NOT INITIAL AND P_BNUM IS NOT INITIAL.
    MOVE: 'I'     TO ZYNKEY-SIGN,
          'EQ'    TO ZYNKEY-OPTION,
          P_BYEAR TO ZYNKEY-LOW(4),
          P_BNUM  TO ZYNKEY-LOW+4(4).
    IF P_EYEAR IS NOT INITIAL AND P_ENUM IS NOT INITIAL.
      MOVE: 'BT'    TO ZYNKEY-OPTION,
            P_EYEAR TO ZYNKEY-HIGH(4),
            P_ENUM  TO ZYNKEY-HIGH+4(4).
    ENDIF.

    APPEND ZYNKEY.
  ELSE.
    IF P_EYEAR IS NOT INITIAL AND P_ENUM IS NOT INITIAL.
      MESSAGE S016 DISPLAY LIKE 'E'
      WITH '※ 발급년도-발급번호 시작조건을 입력하세요!'.
    ENDIF.
  ENDIF.


*>>> [ 검색조건 ] <<<.
  IF L_CERTIK IS INITIAL.
    SELECT * FROM ZHRT9007
             INTO TABLE GT_ZHRT9007
             WHERE ISSTATE LIKE L_ISSTATE
               AND ORDDAT BETWEEN L_ORDDAT_LOW AND L_ORDDAT_HIGH
               AND PERNR  LIKE L_PERNR
               AND ZYNKEY IN ZYNKEY
             ORDER BY ORDDAT PERNR.

  ELSE.
    SELECT * FROM ZHRT9007
             INTO TABLE GT_ZHRT9007
             WHERE ISSTATE LIKE L_ISSTATE
               AND CERTIK EQ L_CERTIK
               AND ORDDAT BETWEEN L_ORDDAT_LOW AND L_ORDDAT_HIGH
               AND PERNR  LIKE L_PERNR
               AND ZYNKEY IN ZYNKEY
             ORDER BY ORDDAT PERNR.
  ENDIF.

*-- 신청번호순으로 SORT(DESCENDING).
  SORT GT_ZHRT9007 BY SCNUM DESCENDING.

ENDFORM. " FN_GET_GT_ZHRt9007
