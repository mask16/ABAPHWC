FORM POPUP_MESSAGE_RTN .
  DATA:GT_SELF LIKE SVAL OCCURS 0 WITH HEADER LINE,
       L_MSG      LIKE SM04DIC-POPUPMSG
       VALUE 'Experimental Message',
       L_LEN      TYPE I,
       L_RET      TYPE C.

  REFRESH GT_SELF.
  GT_SELF-TABNAME   = 'SM04DIC'.
  GT_SELF-FIELDNAME = 'POPUPMSG'.
  GT_SELF-FIELDTEXT = 'Message:'.
  APPEND GT_SELF.

  CALL FUNCTION 'POPUP_GET_VALUES'
    EXPORTING
      POPUP_TITLE     = 'Supply the popup message'
    IMPORTING
      RETURNCODE      = L_RET
    TABLES
      FIELDS          = GT_SELF
    EXCEPTIONS
      ERROR_IN_FIELDS = 1
      OTHERS          = 2.
  CHECK L_RET NE 'A'.
  READ TABLE GT_SELF INDEX 1.
  L_MSG = GT_SELF-VALUE.
