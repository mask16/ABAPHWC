FORM SYSTEM_DATA_READ.
  DATA:LT_MSXX  LIKE MSXXLIST OCCURS 0 WITH HEADER LINE,
       LT_INFO  LIKE UINFO    OCCURS 0 WITH HEADER LINE,
       LT_MSWW  LIKE MSXXLIST OCCURS 0 WITH HEADER LINE.
  REFRESH LT_MSXX. CLEAR   LT_MSXX.

* Server list
  CALL FUNCTION 'TH_SERVER_LIST'
    TABLES
      LIST = LT_MSXX.

* User list
  LOOP AT LT_MSXX.
    REFRESH LT_INFO. CLEAR   LT_INFO.

*   CALL FUNCTION 'TH_SYSTEMWIDE_USER_LIST'
*   Host
    CALL FUNCTION 'THUSRINFO'
      DESTINATION LT_MSXX-NAME
      TABLES
        USR_TABL              = LT_INFO
      EXCEPTIONS
        COMMUNICATION_FAILURE = 17
        OTHERS                = 1.

    LOOP AT LT_INFO.
      GT_LIST  = LT_INFO.
      MOVE-CORRESPONDING LT_INFO  TO GT_LIST.
      GT_LIST-APSERVER = LT_MSXX-NAME.
      APPEND GT_LIST. CLEAR GT_LIST.
    ENDLOOP.
  ENDLOOP.

ENDFORM. " SYSTEM_DATA_READ

FORM SYSTEM_DATA_EDIT.
  DATA: L_LENGTH     TYPE I,
        L_IP(15)     TYPE C,
        L_HEX(1)     TYPE X,
        L_INT(3)     TYPE N,
        LS_ADDR      LIKE BAPIADDR3,
        LS_LOGON     LIKE BAPILOGOND,
        LT_RETURN    LIKE BAPIRET2  OCCURS 0 WITH HEADER LINE,
        LT_TEL       LIKE BAPIADTEL OCCURS 0 WITH HEADER LINE,
        LT_MAIL      LIKE BAPIADSMTP OCCURS 0 WITH HEADER LINE.

  LOOP AT GT_LIST.

*   IP

      L_IP             = GT_LIST-HOSTADR.

FORM SYSTEM_USER_EDIT.
  DATA:LT_USR21  LIKE USR21 OCCURS 0 WITH HEADER LINE.

  SELECT *
  INTO   TABLE LT_USR21
  FROM   USR21 CLIENT SPECIFIED
  WHERE NOT MANDT IN ('000','066').

  LOOP AT LT_USR21.
*    GT_USER        = LT_USR21.
    MOVE-CORRESPONDING LT_USR21 TO GT_USER.
    GT_USER-SEQNO  = SY-TABIX.

*   User ADCP
    SELECT SINGLE TEL_NUMBER TEL_EXTENS
                  DEPARTMENT FUNCTION
    INTO CORRESPONDING FIELDS OF GT_USER
    FROM   ADCP  CLIENT SPECIFIED
    WHERE  CLIENT     EQ LT_USR21-MANDT
      AND  PERSNUMBER EQ LT_USR21-PERSNUMBER
      AND  ADDRNUMBER EQ LT_USR21-ADDRNUMBER.
    CONCATENATE GT_USER-TEL_NUMBER
                GT_USER-TEL_EXTENS
           INTO GT_USER-TEL_NUMBER
                SEPARATED BY '-'.

*   User ADRP
    SELECT SINGLE DATE_TO NAME_TEXT
                  PERS_GROUP  SORT1
    INTO CORRESPONDING FIELDS OF GT_USER
    FROM   ADRP CLIENT SPECIFIED
    WHERE  CLIENT     EQ LT_USR21-MANDT
      AND  PERSNUMBER EQ LT_USR21-PERSNUMBER.

    SELECT SINGLE GROUP_TEXT
    INTO   GT_LIST-GROUP_TEXT
    FROM   TSAD8T
    WHERE  PERS_GROUP EQ GT_USER-PERS_GROUP
      AND  LANGU      EQ SY-LANGU.

    SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF GT_USER
    FROM   USR02 CLIENT SPECIFIED
    WHERE  MANDT      EQ LT_USR21-MANDT
      AND  BNAME      EQ LT_USR21-BNAME.

    COLLECT GT_USER. CLEAR GT_USER.
  ENDLOOP.

ENDFORM. " SYSTEM_USER_EDIT
