FUNCTION ZHR_EXCEL_UPLOAD.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" VALUE(FILENAME) TYPE RLGRAP-FILENAME
*" TABLES
*" DATA_TAB
*" EXCEPTIONS
*" UPLOAD_OLE
*"----------------------------------------------------------------------

* UPLOAD 데이터의 보정을 위한 구분자 처리 펑션
* FUNCTION WITH HANDLING CRITERIA THAT ADJUST THE UPLOAD DATA

* Global data declarations

  DATA : LT_TMP LIKE ZSHRC0011 OCCURS 0 WITH HEADER LINE,
         F_CHK, REC(4000), T_VAL(150).
* CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
  call function 'ZHR_EXCEL_TO_INT_TABLE'
    EXPORTING
      FILENAME                = FILENAME
      I_BEGIN_COL             = 1
      I_BEGIN_ROW             = 1
      I_END_COL               = 256
      I_END_ROW               = 65536
    TABLES
      INTERN                  = LT_TMP
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      UPLOAD_ERROR            = 3.
  IF SY-SUBRC <> 0.
    RAISE UPLOAD_OLE.
  ENDIF.
  CLEAR : G_CNT.
  LOOP AT LT_TMP.
    T_VAL = LT_TMP-VALUE.
    DO.
      G_CNT = G_CNT + 1.
      IF G_CNT = LT_TMP-COL.
        IF NOT REC IS INITIAL.
          CONCATENATE REC '|' T_VAL INTO REC.
        ELSE.
          REC = T_VAL.
        ENDIF.
        EXIT.
      ELSE.
        CONCATENATE REC '|' INTO REC.
        CONDENSE REC.
      ENDIF.
    ENDDO.
    CONDENSE REC.
    AT END OF ROW.
      DATA_TAB = REC.
      APPEND DATA_TAB.  CLEAR REC .  CLEAR G_CNT.
    ENDAT.
  ENDLOOP.

ENDFUNCTION.
