*       text : 임시(REC) -> BDC ITAB(ITAB)

FORM CONVERT_TO_ITAB .

  DATA: L_SPLIT1(1000),
        L_SPLIT2(3000),
        L_TLINES TYPE I,
        L_INDEX  TYPE SY-INDEX.

  CHECK REC[] IS NOT INITIAL.

*// UPLOAD 용
  CREATE DATA TBTAB TYPE STANDARD TABLE OF (P_TABN).
  ASSIGN TBTAB->* TO <FS_TABLE>.
  CREATE DATA TBTAB_WA TYPE (P_TABN).
  ASSIGN TBTAB_WA->* TO <FS_WA>.

  L_TLINES = LINES( LT_DD03P ).

  LOOP AT REC.

    SPLIT REC-RECORD AT '|' INTO L_SPLIT1 L_SPLIT2.
    IF SY-SUBRC = 0.
      CLEAR : L_SPLIT1, L_SPLIT2.

**      L_INDEX = 1.  => MANDT 필드로 사용됨
      L_INDEX = 2.
      L_SPLIT2 = REC.
      DO L_TLINES TIMES.
        ASSIGN COMPONENT L_INDEX OF STRUCTURE <FS_WA> TO <FS_FLD2>.
        IF SY-SUBRC = 0.

          SPLIT L_SPLIT2 AT '|' INTO L_SPLIT1 L_SPLIT2.
          IF SY-SUBRC <> 0.
            CLEAR L_SPLIT1.
          ENDIF.

          READ TABLE LT_DD03P INDEX L_INDEX.
          IF SY-SUBRC = 0.
            IF LT_DD03P-LOWERCASE IS INITIAL.
              TRANSLATE L_SPLIT1 TO UPPER CASE.
            ENDIF.
            CASE LT_DD03P-DATATYPE.
              WHEN 'DATS'.
                REPLACE ALL OCCURRENCES OF '.' IN L_SPLIT1 WITH SPACE.
              WHEN 'CURR'.  "금액은 /100 값으로 처리됨
                REPLACE ALL OCCURRENCES OF ',' IN L_SPLIT1 WITH SPACE.
                L_SPLIT1 = L_SPLIT1 / 100.
              WHEN 'LANG'.
                TRANSLATE L_SPLIT1 TO UPPER CASE.
                READ TABLE LT_T002 WITH KEY LAISO = L_SPLIT1.
                IF SY-SUBRC = 0.
                  L_SPLIT1 = LT_T002-SPRAS.
                ENDIF.
            ENDCASE.
          ENDIF.

          <FS_FLD2> = L_SPLIT1.
          L_INDEX = L_INDEX + 1.
        ELSE.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.

    IF NOT <FS_WA> IS INITIAL.
      APPEND <FS_WA> TO <FS_TABLE>.
    ENDIF.

  ENDLOOP.

ENDFORM. " CONVERT_TO_ITAB
