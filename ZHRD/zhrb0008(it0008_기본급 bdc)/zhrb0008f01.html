FORM DATA_UPLOAD .

*# 오브젝트관리자 숨김 -> BDC 처리속도 향상
  DATA: OBJECTMANAGER_OUT VALUE 'X'.
  EXPORT OBJECTMANAGER_OUT TO MEMORY ID 'OBJECTMANAGER_PA30'.

  CLEAR: TOT_CNT, ERR_CNT, SUC_CNT,
         ERR_TAB, ERR_TAB[].


  SORT RECORD BY PERNR BEGDA.

  LOOP AT RECORD.

    &PIMSG ' ' RECORD-PERNR.

    TOT_CNT = TOT_CNT + 1.

    CLEAR: BDCDATA, BDCDATA[], MESSTAB, MESSTAB[].

    ERR_TAB-PERNR = RECORD-PERNR.
    ERR_TAB-BEGDA = RECORD-BEGDA.
*    ERR_TAB-ENDDA = RECORD-ENDDA.

    PERFORM BDC_DYNPRO      USING   'SAPMP50A'    '1000'.
    PERFORM BDC_FIELD       USING   'BDC_OKCODE'  '=INS'.
    PERFORM BDC_FIELD       USING   'RP50G-PERNR' RECORD-PERNR.
    PERFORM BDC_FIELD       USING   'RP50G-TIMR6' 'X'.
    PERFORM BDC_FIELD       USING   'RP50G-BEGDA' RECORD-BEGDA.
    PERFORM BDC_FIELD       USING   'RP50G-ENDDA' RECORD-ENDDA.
    PERFORM BDC_FIELD       USING   'RP50G-CHOIC' '0008'.

    PERFORM BDC_DYNPRO      USING   'MP000800'    '2041'.
    PERFORM BDC_FIELD       USING   'BDC_OKCODE'  '=ENTR'.
    PERFORM BDC_FIELD       USING   'P0008-PREAS' RECORD-PREAS.
    PERFORM BDC_FIELD       USING   'P0008-TRFAR' RECORD-TRFAR.
    PERFORM BDC_FIELD       USING   'P0008-TRFGB' RECORD-TRFGB.
    PERFORM BDC_FIELD       USING   'P0008-TRFGR' RECORD-TRFGR.
    PERFORM BDC_FIELD       USING   'P0008-TRFST' RECORD-TRFST.

    CLEAR: G_INDEX.
    DO 17 TIMES.
      G_SEQNO = SY-INDEX.
      G_INDEX = G_INDEX + 1.

      CONCATENATE 'Q0008-LGART(' G_INDEX ')' INTO G_TGFIELD.
      CONCATENATE 'RECORD-LGA' G_SEQNO INTO G_FIELDNAME.
      ASSIGN (G_FIELDNAME) TO <FS>.
      CHECK NOT <FS> IS INITIAL.
      PERFORM BDC_FIELD USING G_TGFIELD <FS>.

      CONCATENATE 'Q0008-BETRG(' G_INDEX ')' INTO G_TGFIELD.
      CONCATENATE 'RECORD-BET' G_SEQNO INTO G_FIELDNAME.
      ASSIGN (G_FIELDNAME) TO <FS>.
      PERFORM BDC_FIELD USING G_TGFIELD <FS>.

      CONCATENATE 'Q0008-ANZHL(' G_INDEX ')' INTO G_TGFIELD.
      CONCATENATE 'RECORD-ANZ' G_SEQNO INTO G_FIELDNAME.
      ASSIGN (G_FIELDNAME) TO <FS>.
      PERFORM BDC_FIELD USING G_TGFIELD <FS>.

      CONCATENATE 'Q0008-EITXT(' G_INDEX ')' INTO G_TGFIELD.
      CONCATENATE 'RECORD-EIN' G_SEQNO INTO G_FIELDNAME.
      ASSIGN (G_FIELDNAME) TO <FS>.
      PERFORM BDC_FIELD USING G_TGFIELD <FS>.

      CASE G_INDEX.
        WHEN '07'.
          CLEAR: G_INDEX.
          PERFORM BDC_DYNPRO      USING 'MP000800'   '2041'.
          PERFORM BDC_FIELD       USING 'BDC_OKCODE' '=P+'.
      ENDCASE.
    ENDDO.

    PERFORM BDC_DYNPRO      USING 'MP000800'   '2041'.
    PERFORM BDC_FIELD       USING 'BDC_OKCODE' '=UPD'.


    PERFORM BDC_TRANSACTION USING    'PA30' P_MODE
                            CHANGING G_SUBRC.

    IF G_SUBRC  = 0.
      READ TABLE MESSTAB WITH KEY MSGTYP = 'E'
                         TRANSPORTING NO FIELDS.
      IF SY-SUBRC <> 0.
        SUC_CNT = SUC_CNT + 1.
      ELSE.
        ERR_CNT = ERR_CNT + 1.
        LOOP AT MESSTAB.
          CALL FUNCTION 'MESSAGE_TEXT_BUILD'
            EXPORTING
              MSGID               = MESSTAB-MSGID
              MSGNR               = MESSTAB-MSGNR
              MSGV1               = MESSTAB-MSGV1
              MSGV2               = MESSTAB-MSGV2
              MSGV3               = MESSTAB-MSGV3
              MSGV4               = MESSTAB-MSGV4
            IMPORTING
              MESSAGE_TEXT_OUTPUT = ERR_TAB-MSGTEXT
            EXCEPTIONS
              OTHERS              = 1.
          APPEND ERR_TAB. CLEAR ERR_TAB.
        ENDLOOP.
      ENDIF.
    ELSE.
      ERR_CNT = ERR_CNT + 1.
      LOOP AT MESSTAB.
        CALL FUNCTION 'MESSAGE_TEXT_BUILD'
          EXPORTING
            MSGID               = MESSTAB-MSGID
            MSGNR               = MESSTAB-MSGNR
            MSGV1               = MESSTAB-MSGV1
            MSGV2               = MESSTAB-MSGV2
            MSGV3               = MESSTAB-MSGV3
            MSGV4               = MESSTAB-MSGV4
          IMPORTING
            MESSAGE_TEXT_OUTPUT = ERR_TAB-MSGTEXT
          EXCEPTIONS
            OTHERS              = 1.
        APPEND ERR_TAB. CLEAR ERR_TAB.
      ENDLOOP.
    ENDIF.

  ENDLOOP.

  ULINE.
  WRITE:/'※ 총 건 수 : ', TOT_CNT,
        /'※ 성공건수 : ', SUC_CNT,
        /'※ 실패건수 : ', ERR_CNT.
  ULINE.
  WRITE:/ '▶▶▶  ERROR LOG  ◀◀◀'.
  ULINE.

  WRITE:/(10) '[ 사번 ]',
         (10) '[ 시작일 ]',
*         (10) '[ 종료일 ]',
         '[ Error Message ]'.
  ULINE.

  LOOP AT ERR_TAB.
    WRITE:/(10) ERR_TAB-PERNR,
           (10) ERR_TAB-BEGDA,
*           (10) ERR_TAB-ENDDA,
           ERR_TAB-MSGTEXT.
  ENDLOOP.

ENDFORM. " DATA_UPLOAD
