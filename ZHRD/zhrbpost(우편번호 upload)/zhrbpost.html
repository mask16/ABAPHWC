  CLEAR : it_post[], it_post.

  PERFORM upload_data USING locfile.

  LOOP AT record.
    MOVE-CORRESPONDING record TO it_post.
    it_post-mandt = sy-mandt.
    APPEND it_post. CLEAR it_post.
  ENDLOOP.

  DESCRIBE TABLE it_post LINES cnt.
  IF cnt = 0.
    MESSAGE i010(pg) WITH '데이타가 없습니다'..
  ELSE.
    SORT it_post BY seqno zipcode.
    DELETE FROM zhrtpost where state = p_state.
    INSERT zhrtpost FROM TABLE it_post ACCEPTING DUPLICATE KEYS.
    IF sy-subrc = 0.
      COMMIT WORK.
      MESSAGE s016(pg) WITH '우편번호 UPLOAD 성공!'.
    ELSE.
      ROLLBACK WORK.
      MESSAGE e016(pg) WITH '우편번호 UPLOAD 실패!'
                          '시스템 담당자에게 문의하십시오.'.
    ENDIF.
  ENDIF.
