                                   FROM zhrt9001 AS a JOIN zhrt9001t AS b

                 ''     'KEY'                'X',

FORM select_data .
  DATA : l_entry TYPE datum,
             l_btext TYPE btrtx,
             l_ptext TYPE pktxt,
             l_fieldname(30).

  DATA : ls_duration TYPE psen_duration_dec,
              lv_cosy(2)  TYPE n,
              lv_cosm(2)  TYPE n,
              lv_cosd(2)  TYPE n  ,
              lv_year_txt(5),
              lv_mon_txt(6),
              lv_day_txt(6),
              l_wrkd(20),
              lv_rest     TYPE i,
              lv_idx      TYPE i,
              l_usdqta TYPE p2006-anzhl,
              l_rstqta TYPE p2006-anzhl.

  FIELD-SYMBOLS : <fs>.

* 연차정보
  DATA : lv_desta     TYPE ptm_dedstart,
              lv_deend     TYPE ptm_dedend,
              lt_cumul     LIKE bapihrquotacv OCCURS 0 WITH HEADER LINE,
              lt_quota     LIKE bapihrquota OCCURS 0 WITH HEADER LINE.


  CLEAR : t_orgeh, t_orgeh[].

  call function 'ZHRF0001'
    EXPORTING
      i_pernr   = pernr-pernr
      i_keydate = pn-endda
      i_prog    = sy-repid
    IMPORTING
      e_btext   = l_btext
    TABLES
      lt_orgeh  = t_orgeh.

  PERFORM assign_field USING : 'BTEXT'  l_btext.
  IF NOT t_orgeh[] IS INITIAL.
    LOOP AT t_orgeh.
      CONCATENATE 'O_' t_orgeh-scode INTO l_fieldname.
      CONDENSE l_fieldname NO-GAPS.
      PERFORM assign_field USING :  l_fieldname  t_orgeh-orgtx.
    ENDLOOP.
  ELSE.
    DATA : l_orgtx LIKE t527x-orgtx.
    PERFORM assign_field USING :  'ORGEH'   pernr-orgeh.
    SELECT SINGLE orgtx INTO l_orgtx
                                FROM t527x
                               WHERE orgeh = pernr-orgeh
                                   AND endda >= pn-endda
                                   AND begda <= pn-endda
                                   AND sprsl = sy-langu.
    PERFORM assign_field USING :  'ORGTX'   l_orgtx.

  ENDIF.
  PERFORM assign_field USING :  'PERNR' pernr-pernr,
                                                    'ENAME' pernr-ename.

* 사원하위그룹
  SELECT SINGLE ptext INTO l_ptext
                               FROM t503t
                              WHERE sprsl = sy-langu
                                  AND persk = pernr-persk.

  PERFORM assign_field USING : 'PTEXT'  l_ptext.

* 입사일
  CALL FUNCTION 'HR_ENTRY_DATE'
    EXPORTING
      persnr               = pernr-pernr
    IMPORTING
      entrydate            = l_entry
    EXCEPTIONS
      entry_date_not_found = 1
      pernr_not_assigned   = 2
      OTHERS               = 3.

  PERFORM assign_field USING : 'ENTRY'  l_entry.

* 근속일수
  CALL FUNCTION 'HR_SEN_CALCULATE_COMPLETE'
    EXPORTING
      id_pernr                 = pernr-pernr
      id_molga                 = gv_molga
      id_proce                 = '1SVY'
      id_calty                 = '1'
      id_begda                 = l_entry
      id_endda                 = pn-endda
    IMPORTING
      es_duration              = ls_duration
    EXCEPTIONS
      wrong_import_parameter   = 1
      wrong_calculation_type   = 2
      process_not_found        = 3
      process_pstep_not_found  = 4
      no_result_for_calty      = 5
      error_of_other_functions = 6
      OTHERS                   = 7.

  lv_cosy = ls_duration-calyy.
  lv_cosm = ls_duration-calmm.
  lv_cosd = ls_duration-caldd.

  IF ls_duration-calyy IS NOT INITIAL.
    CONCATENATE lv_cosy
                '년' INTO lv_year_txt.
    SHIFT lv_year_txt LEFT DELETING LEADING '0'.
  ENDIF.

  IF ls_duration-calmm IS NOT INITIAL.
    CONCATENATE lv_cosm
                '개월' INTO lv_mon_txt.
    SHIFT lv_mon_txt LEFT DELETING LEADING '0'.
  ENDIF.

  IF ls_duration-caldd IS NOT INITIAL.
    CONCATENATE lv_cosd
                '일' INTO lv_day_txt.
    SHIFT lv_day_txt LEFT DELETING LEADING '0'.
  ENDIF.

  CONCATENATE lv_year_txt lv_mon_txt lv_day_txt INTO l_wrkd.
  CONDENSE l_wrkd NO-GAPS.

  PERFORM assign_field USING :  'WRKSD'   l_wrkd.

* 휴무쿼터
  lv_desta = pn-endda.
  lv_deend = pn-endda.

  CLEAR : lt_quota, lt_quota[], lt_cumul, lt_cumul[].
  CALL FUNCTION 'BAPI_TIMEQUOTA_GETDETAILEDLIST'
    EXPORTING
      employeenumber          = pernr-pernr
      quotaselectionmod       = '1'
      wayofcalculation        = 'D'
      deductbegin             = lv_desta
      deductend               = lv_deend
      keydatefordeduction     = lv_deend
    TABLES
      cumulatedvalues         = lt_cumul
      absencequotareturntable = lt_quota.

  CHECK lt_quota[] IS NOT INITIAL.
  DELETE lt_quota WHERE entitle = 0.

  SORT lt_quota BY quotatype.

  LOOP AT lt_quota.
    PERFORM assign_field USING :  'QTEXT'   lt_quota-quotatext,
                                                      'QBEGDA'  lt_quota-from_date,
                                                      'QENDDA'  lt_quota-to_date,
                                                       'QDESTA'  lt_quota-deductbegin,
                                                       'QDEEND'  lt_quota-deductend,
                                                       'QTNUM'  lt_quota-entitle.

    ADD lt_quota-deduct  TO l_usdqta.   "사용연차
    ADD lt_quota-ordered TO l_usdqta.   "사용연차
    ADD lt_quota-reduced TO l_usdqta.   "사용연차
    l_rstqta = lt_quota-entitle - l_usdqta. "잔여연차

    PERFORM assign_field USING : 'USDQTA'  l_usdqta,
                                                     'RSTQTA' l_rstqta.

    APPEND <fs_wa> TO <fs_table>.
    CLEAR :  l_usdqta, l_rstqta.
  ENDLOOP.
  CLEAR : <fs_wa>.

ENDFORM. " SELECT_DATA
