*&---------------------------------------------------------------------*
*       USER ID로 NAME 찾아오기
*----------------------------------------------------------------------*
*      -->LP_UNAME  USER ID
*      <--LP_NAMET  NAME
*----------------------------------------------------------------------*
FORM NAME_SELECT USING LP_UNAME LIKE SY-UNAME
                  CHANGING LP_NAMET LIKE ADRP-NAME_TEXT.

  DATA: LV_FIRST LIKE ADRP-NAME_FIRST,
        LV_LAST  LIKE ADRP-NAME_LAST.

  SELECT SINGLE
         ADRP~NAME_FIRST
         ADRP~NAME_LAST
   INTO (LV_FIRST,
         LV_LAST)
    FROM USR21
   INNER JOIN ADRP
      ON ADRP~PERSNUMBER EQ USR21~PERSNUMBER
   WHERE USR21~BNAME EQ LP_UNAME.
  IF SY-DBCNT NE 0.
    CONCATENATE LV_LAST LV_FIRST INTO LP_NAMET.
  ELSE.
    LP_NAMET = LP_UNAME.
  ENDIF.

ENDFORM. " NAME_SELECT
