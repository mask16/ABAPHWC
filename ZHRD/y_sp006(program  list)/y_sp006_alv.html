*&---------------------------------------------------------------------*
*       조회변형 DEFAULT SELECT
*----------------------------------------------------------------------*
*      -->LP_VARIA  조회변형
*----------------------------------------------------------------------*
FORM VARIANT_DEFAULT USING LP_VARIA.

  GS_DEF_VARIANT = GS_VARIANT.
  IF NOT LP_VARIA IS INITIAL.
    GS_DEF_VARIANT-VARIANT = LP_VARIA.
  ENDIF.

* 기본 세팅
  CALL FUNCTION 'LVC_VARIANT_DEFAULT_GET'
    EXPORTING
      I_SAVE        = 'A'
    CHANGING
      CS_VARIANT    = GS_DEF_VARIANT
    EXCEPTIONS
      WRONG_INPUT   = 1
      NOT_FOUND     = 2
      PROGRAM_ERROR = 3
      OTHERS        = 4.

  CASE SY-SUBRC.
    WHEN 0.
      LP_VARIA = GS_DEF_VARIANT-VARIANT.
    WHEN 2.
      CLEAR LP_VARIA.
* 사용자별 세팅
      SELECT SINGLE LTDX~VARIANT
        INTO LP_VARIA
        FROM LTDX
       WHERE LTDX~RELID   EQ 'LT'
         AND LTDX~REPORT  EQ GV_PROGRAM
         AND LTDX~ERFNAME EQ SY-UNAME.
  ENDCASE.

ENDFORM. " VARIANT_DEFAULT

      IS_VARIANT    = GS_VARIANT

    IMPORTING

    EXCEPTIONS

      PROGRAM_ERROR = 2
      OTHERS        = 3.
  IF SY-SUBRC NE 0.
    MESSAGE I016 WITH '※ 변형이 존재하지 않습니다!'.
  ENDIF.

  IF GV_VARIANT_EXIT IS INITIAL.
    GS_VARIANT-VARIANT = GS_DEF_VARIANT-VARIANT.
    LP_VARIANT         = GS_DEF_VARIANT-VARIANT.
  ENDIF.

ENDFORM. " VARIANT_HELP

*&      Form  VARIANT_FILL
*&---------------------------------------------------------------------*
*  -->  LP_VARIA  조회변형
*----------------------------------------------------------------------*
FORM VARIANT_FILL USING LP_VARIA LIKE DISVARIANT-VARIANT.

  CLEAR GS_VARIANT.
  IF LP_VARIA IS INITIAL.
    GS_VARIANT-VARIANT = 'STANDARD'.
    GS_VARIANT-REPORT  = GV_REPID.
  ELSE.
    GS_VARIANT-VARIANT = LP_VARIA.
    GS_VARIANT-REPORT  = GV_REPID.
    CALL FUNCTION 'LVC_VARIANT_EXISTENCE_CHECK'
      EXPORTING
        I_SAVE     = 'A'
      CHANGING
        CS_VARIANT = GS_VARIANT
      EXCEPTIONS
        OTHERS     = 01.
    IF SY-SUBRC NE 0.
      MESSAGE E016 WITH '※ 존재하지 않는 변형입니다!'.
    ENDIF.
  ENDIF.

ENDFORM. " VARIANT_FILL
