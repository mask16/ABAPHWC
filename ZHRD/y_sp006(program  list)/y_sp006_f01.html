FORM GET_DATA.

  CLEAR: GT_DATA[], GT_DATA.
  SELECT TRDIR~NAME    "ABAP 프로그램이름
         TRDIRT~TEXT   "프로그램제목
         TRDIR~SUBC    "프로그램유형
         TRDIR~RMAND   "클라이언트
         TRDIR~CNAM    "생성인
         TRDIR~CDAT    "생성일
         TRDIR~UNAM    "최종변경인
         TRDIR~UDAT    "최종변경일
    INTO CORRESPONDING FIELDS OF TABLE GT_DATA
    FROM TRDIR
   INNER JOIN TRDIRT
      ON TRDIRT~NAME  = TRDIR~NAME
     AND TRDIRT~SPRSL = SY-LANGU
      UP TO SP_MAXSE ROWS
   WHERE TRDIR~NAME   IN SS_NAME
     AND TRDIR~NAME   IN GR_NAME
     AND TRDIR~SUBC   IN GR_SUBC
     AND TRDIR~UNAM   IN SS_UNAM.
  IF SY-DBCNT = 0.
    MESSAGE S001(00) WITH '※ DATA 가 없습니다!'.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM. " GET_DATA

FORM ALV_TOP_OF_PAGE.

  DATA: LV_NAME(30),
        LV_MODULE(10),
        LV_SUBC(40).

  CLEAR: GT_LISTHEADER[].
  IF SS_NAME-LOW IS INITIAL.
    LV_NAME = TEXT-003.
  ELSE.
    IF SS_NAME-HIGH IS INITIAL.
      LV_NAME = SS_NAME-LOW.
    ELSE.
      CONCATENATE SS_NAME-LOW '~' SS_NAME-HIGH INTO LV_NAME.
    ENDIF.
  ENDIF.

  GS_LISTHEADER-TYP  = 'S'. "H=Header, S=Selection, A=Action
  GS_LISTHEADER-KEY  = TEXT-014.
  GS_LISTHEADER-INFO = LV_NAME.
  APPEND GS_LISTHEADER TO GT_LISTHEADER.

  CASE GC_X.
    WHEN SR_HR.
      LV_MODULE = TEXT-003.
    WHEN SR_FI.
      LV_MODULE = TEXT-004.
    WHEN SR_CO.
      LV_MODULE = TEXT-005.
    WHEN SR_MM.
      LV_MODULE = TEXT-006.
    WHEN SR_SD.
      LV_MODULE = TEXT-100.
    WHEN SR_PP.
      LV_MODULE = TEXT-007.
  ENDCASE.

  GS_LISTHEADER-TYP  = 'S'. "H=Header, S=Selection, A=Action
  GS_LISTHEADER-KEY  = TEXT-002.
  GS_LISTHEADER-INFO = LV_MODULE.
  APPEND GS_LISTHEADER TO GT_LISTHEADER.

  IF SC_RE EQ GC_X.
    IF LV_SUBC IS INITIAL.
      LV_SUBC = TEXT-009.
    ELSE.
      CONCATENATE LV_SUBC '/ REPORT' INTO LV_SUBC SEPARATED BY SPACE.
    ENDIF.
  ENDIF.

  IF SC_ON EQ GC_X.
    IF LV_SUBC IS INITIAL.
      LV_SUBC = TEXT-010.
    ELSE.
      CONCATENATE LV_SUBC '/ ON-LINE' INTO LV_SUBC SEPARATED BY SPACE.
    ENDIF.
  ENDIF.

  IF SC_IN EQ GC_X.
    IF LV_SUBC IS INITIAL.
      LV_SUBC = TEXT-011.
    ELSE.
      CONCATENATE LV_SUBC '/ INCLUDE' INTO LV_SUBC SEPARATED BY SPACE.
    ENDIF.
  ENDIF.

  GS_LISTHEADER-TYP  = 'S'.
  GS_LISTHEADER-KEY  = TEXT-008.
  GS_LISTHEADER-INFO = LV_SUBC.
  APPEND GS_LISTHEADER TO GT_LISTHEADER.

  CALL FUNCTION 'REUSE_ALV_COMMENTARY_WRITE'
    EXPORTING
      I_LOGO             = 'HPCLOGO'
      IT_LIST_COMMENTARY = GT_LISTHEADER.

ENDFORM. " ALV_TOP_OF_PAGE

FORM USER_COMMAND USING LP_UCOMM LIKE SY-UCOMM
                        RS_SELFIELD TYPE SLIS_SELFIELD.

  CASE LP_UCOMM.
    WHEN '&IC1'.
      CASE RS_SELFIELD-SEL_TAB_FIELD.
        WHEN 'GT_DATA-NAME'.
          READ TABLE GT_DATA INDEX RS_SELFIELD-TABINDEX.
          IF ( GT_DATA-NAME+3(1) EQ 'T' ) OR
             ( GT_DATA-NAME+3(1) EQ 'V' ).
            PERFORM BDC_TRANSACTION_SE11.
          ELSE.
            PERFORM BDC_TRANSACTION_SE80.
          ENDIF.
        WHEN 'GT_DATA-TCODE'.
          READ TABLE GT_DATA INDEX RS_SELFIELD-TABINDEX.
          CHECK GT_DATA-TCODE NE SPACE.
          PERFORM CALL_TRANSACTION.
      ENDCASE.
  ENDCASE.

ENDFORM. " USER_COMMAND

FORM MODIFY_DATA.

  CLEAR GT_DATA.
  LOOP AT GT_DATA.
*   개발클래스
    CALL FUNCTION 'RS_PROGRAM_GET_DEVCLASS'
      EXPORTING
        PROGNAME = GT_DATA-NAME
      IMPORTING
        DEVCLASS = GT_DATA-DEVCL.
*    IF ( GT_DATA-DEVCL EQ '$TMP' ) OR ( GT_DATA-DEVCL(1) EQ 'Z' ).
*    ELSE.
*      DELETE GT_DATA.
*      CONTINUE.
*    ENDIF.

*   프로그램유형
    CASE GT_DATA-SUBC.
      WHEN '1'.
        GT_DATA-STEXT = TEXT-009.
      WHEN 'I'.
        GT_DATA-STEXT = TEXT-011.
      WHEN 'M'.
        GT_DATA-STEXT = TEXT-010.
    ENDCASE.

*   트랜잭션코드
    SELECT SINGLE TSTC~TCODE
      INTO GT_DATA-TCODE
      FROM TSTC
     WHERE TSTC~PGMNA EQ GT_DATA-NAME. " 프로그램이름

*   활성구분
    SELECT COUNT(*)
      INTO SY-DBCNT
      FROM PROGDIR
     WHERE PROGDIR~NAME  EQ GT_DATA-NAME
       AND PROGDIR~STATE EQ GC_STATE.
    IF SY-DBCNT = 0.
      GT_DATA-STATE = TEXT-012.
    ELSE.
      GT_DATA-STATE = TEXT-013.
    ENDIF.

    MODIFY GT_DATA. CLEAR  GT_DATA.
  ENDLOOP.

ENDFORM. " MODIFY_DATA

    FROM TSTC

      ON TSTCT~TCODE = TSTC~TCODE

        LT_DATA-UDAT  = GT_DATA-UDAT.      "최종변경일
      ELSE.

          IMPORTING
            DEVCLASS     = LT_DATA-DEVCL.
        IF ( LT_DATA-DEVCL(3) <> 'ZHR' ) AND
           ( LT_DATA-DEVCL(3) <> 'ZFI' ) AND
