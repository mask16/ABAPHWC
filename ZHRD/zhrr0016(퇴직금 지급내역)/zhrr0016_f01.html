    TABLES
      in_rgdir        = gt_rgdir[]
    EXCEPTIONS
      no_record_found = 1

  CHECK sy-subrc = 0.

  LOOP AT gt_rgdir WHERE srtza = 'A' AND

                                         (  paydt  BETWEEN pn-begda AND pn-endda ) AND

    ENDIF.

    CALL FUNCTION 'PYXX_READ_PAYROLL_RESULT'
      EXPORTING
        clusterid                    = 'KR'

      MOVE gt_pay_result-inter-versc  TO gs_versc.

      MOVE gt_pay_result-nat-krsp1[] TO gt_krsp1[].

              sequencenumber               = tmp_seqnr
            CHANGING
              payroll_result               = gt_pay_result

      PERFORM set_basic_data USING lv_irbda lv_ireda p0538-begda.

  DATA : l_btrtl  TYPE btrtl,
             l_btext TYPE btrtx,
             l_text(20),
             l_fieldname(20).

  CLEAR : t_orgeh, t_orgeh[].

    EXPORTING
      i_pernr   = pernr-pernr

* 급여사유 텍스트
  SELECT SINGLE *
    FROM t52ocrt CLIENT SPECIFIED
    WHERE sprsl = sy-langu AND
         molga = gv_molga AND
         ocrsn = gt_evpdir-ocrsn.

FORM get_ret_amt .
  DATA : tmp_betrg TYPE betrg.
* 퇴직금 / 명예퇴직금 / 소득세 / 주민세 / 국민연금전환금 / 실지급액
  LOOP AT gt_rt.
    CASE gt_rt-lgart.
      WHEN '/R01' OR '/R07'.  "퇴직금 or 퇴직연금 일시금
        PERFORM assign_field USING : 'AMT_01'  gt_rt-betrg.
      WHEN '/R02'.  "명예퇴직금
        PERFORM assign_field USING : 'AMT_02'  gt_rt-betrg.
      WHEN '/R91'.  "소득세
        PERFORM assign_field USING : 'AMT_03'  gt_rt-betrg.
      WHEN '/R92'.  "주민세
        PERFORM assign_field USING : 'AMT_04'  gt_rt-betrg.
      WHEN '/RP1'.  "국민연금전환금
        PERFORM assign_field USING : 'AMT_05'  gt_rt-betrg.
      WHEN '/559'. "실지급액
        READ TABLE gt_rt_0012 WITH KEY lgart = '/559'.
        IF sy-subrc = 0.
          gt_rt-betrg = gt_rt-betrg + gt_rt_0012-betrg.
        ENDIF.
        PERFORM assign_field USING : 'AMT_06'  gt_rt-betrg.
      WHEN '/561'.  "청구 금액
        gt_rt-betrg = gt_rt-betrg * -1.
        PERFORM assign_field USING : 'AMT_06'  gt_rt-betrg.
    ENDCASE.
  ENDLOOP.

  LOOP AT gt_rt_0012 WHERE lgart = '/RO1' OR
                                               lgart = '/RO2'.
    tmp_betrg = tmp_betrg + gt_rt_0012-betrg.
  ENDLOOP.
  tmp_betrg = tmp_betrg * ( -1 ).
  PERFORM assign_field USING : 'AMT_07'  tmp_betrg.

  PERFORM assign_field USING : 'WAERS'  gs_versc-waers,
                                                    'SEQNR'   gt_evpdir-seqnr.

  APPEND <fs_wa> TO <fs_table>.
  CLEAR <fs_wa>.

ENDFORM. " GET_RET_AMT
