FORM read_data .

  PERFORM assign_field USING : 'PERNR' pernr-pernr,
                                                      'ENAME'  pernr-ename.


  CLEAR : t_orgeh, t_orgeh[].

  call function 'ZHRF0001'
    EXPORTING
      i_pernr   = pernr-pernr
      i_keydate = pn-endda
      i_prog    = sy-repid
    IMPORTING
*     e_btrtl   = l_btrtl
*     e_btext   = l_btext
      e_orgeh   = l_orgeh
      e_orgtx   = l_orgtx
    TABLES
      lt_orgeh  = t_orgeh.

  PERFORM assign_field USING : 'ORGEH' l_orgeh,
                                                    'ORGTX '  l_orgtx.

  DATA : l_date(5),
             l_fieldname(30),
             num(2) TYPE n.
  num = 1.

  DO 5 TIMES.
    zdats = pnpbegda(4) - num.
    CONCATENATE zdats '%' INTO l_date.
    UNPACK num TO num.
    SELECT SINGLE aprtf INTO g_aprtf
                 FROM pa0147
              WHERE pernr = pernr-pernr
                   AND begda  LIKE l_date.

    CONCATENATE 'G' num INTO l_fieldname.
    PERFORM assign_field USING : l_fieldname  g_aprtf.
    num = num + 1.
    CLEAR : l_date, zdats, g_aprtf .
  ENDDO.

  APPEND <fs_wa> TO <fs_table>.
  CLEAR <fs_wa>.


ENDFORM. " READ_DATA
