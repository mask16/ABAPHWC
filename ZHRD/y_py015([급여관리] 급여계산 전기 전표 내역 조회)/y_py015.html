                               L_BUKRS STRUCTURE P_BUKRS

        MESSAGE_VAR4      = ERROR_TAB-MSGV4
      IMPORTING
        MESSAGE_TEXT      = ERROR_TAB-TEXT
      EXCEPTIONS
        MESSAGE_NOT_FOUND = 1
        OTHERS            = 2.

                   ERROR_TAB-MSGV3 ERROR_TAB-MSGV4
                   INTO ERROR_TAB-TEXT

    ENDIF.
    MODIFY ERROR_TAB.
