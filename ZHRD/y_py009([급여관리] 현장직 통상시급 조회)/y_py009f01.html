       SCREEN-NAME = '%_PNPSTAT2_%_APP_%-TEXT' OR

       SCREEN-NAME = '%_PNPSTAT2_%_APP_%-VALU_PUSH'.

      SCREEN-INPUT = 0.

      SCREEN-REQUIRED = 1.
