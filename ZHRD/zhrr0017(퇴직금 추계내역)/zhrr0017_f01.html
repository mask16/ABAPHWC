FORM read_kostl .
  CLEAR: gt_csks, gt_csks[].

  SELECT c~kostl c~kosar t~ktext  FROM csks AS c INNER JOIN cskt AS t
                                                          ON c~kokrs = t~kokrs

                                                         AND c~datbi = t~datbi
                                                        INTO TABLE gt_csks
                                                        WHERE c~kokrs = c_kokrs
                                                          AND c~datbi >= gv_endda
                                                          AND c~datab <= gv_endda
                                                          AND t~spras = sy-langu.
  SORT gt_csks BY kostl.

ENDFORM. " READ_KOSTL
