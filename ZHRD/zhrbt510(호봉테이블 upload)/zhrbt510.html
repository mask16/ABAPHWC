REPORT ZHRBT510 MESSAGE-ID zhrm01 NO STANDARD PAGE HEADING LINE-SIZE 255.
*********************************************************************
* TABLES                                                                                                        *
*********************************************************************
TABLES : t510.

*********************************************************************
* DATA                                                                                                            *
*********************************************************************
DATA : BEGIN OF record OCCURS 0,
               molga(002),        " 국가그룹핑
               trfar(002),        " 호봉유형
               trfgb(002),        " 호봉영역
               trfkz(001),        " 취업규칙조항에 대한 ES 그룹핑
               trfgr(008),        " 호봉그룹
               trfst(002),        " 호봉단계
               lgart(004),        " 임금유형
               betrg(015),        " 지급의 임금유형 금액
               begda(008),        " 시작일
               endda(008),        " 종료일
            END OF record.

DATA : it_t510 TYPE TABLE OF t510 WITH HEADER LINE,
            ls_t510 TYPE t510.

CONSTANTS: c_endda TYPE datum VALUE '99991231'.

*********************************************************************
* INCLUDE                                                                                                      *
*********************************************************************
INCLUDE <icon>.
include zhr_common.

*----------------------------------------------------------------------*
*INITIALIZATION.
*----------------------------------------------------------------------*
INITIALIZATION.
  gv_document = 'ZHRINFOT510'.
  func_b-text      = ''.
  func_b-icon_id   = icon_xls.
  func_b-icon_text = 'UPLOAD LAYOUT'.
  sscrfields-functxt_01 = func_b. "Selection screen: Text for pushbuttons

*----------------------------------------------------------------------*
*   At selection screen output                                                                                                  *
*----------------------------------------------------------------------*
AT SELECTION-SCREEN OUTPUT.

  LOOP AT SCREEN.
    CHECK screen-name CP '*P_MODE*'.
    screen-active = 0.
    MODIFY SCREEN.
  ENDLOOP.

*----------------------------------------------------------------------*
*   start of selection                                                 *
*----------------------------------------------------------------------*
START-OF-SELECTION.

  IF dataset IS INITIAL.
    MESSAGE i028.

  ELSE.

    CLEAR :  total_count, error_count, success_count.

    PERFORM upload_data TABLES record.

    DESCRIBE TABLE record LINES total_count.
    CLEAR: it_t510, it_t510[].
    LOOP AT record.
      MOVE-CORRESPONDING record TO it_t510.

      IF record-molga = '41'.
        it_t510-betrg = it_t510-betrg / 100.
      ENDIF.
      APPEND it_t510.

* 이전 정보 확인 => 한계결정
      SELECT SINGLE * INTO CORRESPONDING FIELDS OF ls_t510
        FROM t510
        WHERE molga = record-molga AND
              trfar = record-trfar AND
              trfgb = record-trfgb AND
              trfkz = record-trfkz AND
              trfgr = record-trfgr AND
              trfst = record-trfst AND
              lgart = record-lgart AND
              endda = c_endda.

      IF sy-subrc = 0.
        IF  ls_t510-begda NE record-begda.
          ls_t510-endda = record-begda.
          SUBTRACT 1 FROM ls_t510-endda.
          APPEND ls_t510 TO it_t510.
          CLEAR ls_t510.
        ENDIF.
      ENDIF.

      CLEAR it_t510.
    ENDLOOP.

    MODIFY t510 FROM TABLE it_t510.

    ULINE.

    IF sy-subrc = 0.
      success_count = total_count.
      CLEAR error_count.
    ELSE.
      error_count = total_count.
      CLEAR success_count.
    ENDIF.

    ULINE.
    WRITE:/ '총건수: ', total_count,   '성공건수: ', success_count,
                 '실패건수: ', error_count.

    ULINE.
  ENDIF.
