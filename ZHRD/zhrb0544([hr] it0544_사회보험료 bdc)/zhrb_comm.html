FORM UPLOAD_DATA TABLES RECORD.

  CLEAR: GT_DATA, GT_DATA[].

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME    = P_PATH
      I_BEGIN_COL = 1
      I_BEGIN_ROW = 5
      I_END_COL   = 50
      I_END_ROW   = 65000
    TABLES
      INTERN      = GT_DATA.

  IF SY-SUBRC <> 0.
    MESSAGE I016(PN) WITH '※ File Upload Failed !'.
    STOP.
  ELSE.
    CLEAR: RECORD, RECORD[].
    LOOP AT GT_DATA.
      ASSIGN COMPONENT GT_DATA-COL OF STRUCTURE RECORD TO <FS_REC>.
      <FS_REC> = GT_DATA-VALUE.

      AT END OF ROW.
        APPEND RECORD. CLEAR RECORD.
      ENDAT.
    ENDLOOP.
  ENDIF.

ENDFORM. " UPLOAD_DATA
