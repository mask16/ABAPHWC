  PERFORM SET_VALUE.

  CASE 'X'.
    WHEN R1.  "저장
      PERFORM GET_DATA_SAVE.

    WHEN R2.  "조회
      PERFORM GET_DATA_DISP.
  ENDCASE.

*----------------------------------------------------------------------*
*GET PERNR.
*----------------------------------------------------------------------*
*  PERFORM GET_DATA.

*----------------------------------------------------------------------*
END-OF-SELECTION.
*----------------------------------------------------------------------*
  CASE 'X'.
    WHEN R1.  "저장
      READ TABLE GT_TAB INDEX 1.
      IF SY-SUBRC = 0.
        PERFORM SAVE_DATA.
      ELSE.
        MESSAGE I016 WITH '※ 저장 할 DATA 가 없습니다~!'.
        EXIT.
      ENDIF.

    WHEN R2.  "조회
      READ TABLE GT_DIS INDEX 1.
      IF SY-SUBRC = 0.
        PERFORM DISPLAY_ALV.
      ELSE.
        MESSAGE I016 WITH '※ 선택조건에 맞는 DATA 가 없습니다~!'.
        EXIT.
      ENDIF.
  ENDCASE.
