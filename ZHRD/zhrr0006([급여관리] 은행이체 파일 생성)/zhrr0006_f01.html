        lv_paydate TYPE pc261-paydt,
        lv_ok(1),
        lv_append_ok(1),
        lv_sp_char TYPE fist-searchw,
        lt_rgdir       TYPE STANDARD TABLE OF pc261 WITH HEADER LINE,
        lt_evpdir      LIKE pc261 OCCURS 0 WITH HEADER LINE,
        t500l_wa TYPE t500l,
        lt_pay_result  TYPE pay99_result,
        lv_paydt TYPE d.

  CLEAR :gt_data, lv_append_ok.

  CONCATENATE pn-pabrj pn-pabrp INTO lv_inper.

  CLEAR : ls_tab01, lt_tab01[], ls_bt, ls_tab01,
          lt_rgdir, lt_rgdir[], lt_evpdir, lt_evpdir[],
          lt_pay_result, lv_paydt.

  CALL FUNCTION 'CU_READ_RGDIR'
    EXPORTING
      persnr          = pernr-pernr
    TABLES
      in_rgdir        = lt_rgdir[]
    EXCEPTIONS
      no_record_found = 1
      OTHERS          = 2.

  CHECK sy-subrc = 0.

  LOOP AT lt_rgdir.
    IF lt_rgdir-ocrsn IS INITIAL OR lt_rgdir-ocrsn EQ '0007'.
      CHECK lt_rgdir-inper = lv_inper.
      CHECK lt_rgdir-fpper = lv_inper.
    ELSE.
      CHECK lt_rgdir-ocrsn = p_ocrsn.
      CHECK lt_rgdir-payty = p_payty.
      CHECK lt_rgdir-payid = p_payid.
      CHECK lt_rgdir-bondt = p_bondt.
    ENDIF.

    CALL FUNCTION 'CD_EVALUATION_PERIODS'
      EXPORTING
        bonus_date      = p_bondt
        inper_modif     = lt_rgdir-iperm
        inper           = lt_rgdir-inper
        pay_type        = p_payty
        pay_ident       = p_payid
      TABLES
        rgdir           = lt_rgdir[]
        evpdir          = lt_evpdir[]
      EXCEPTIONS
        no_record_found = 1
        OTHERS          = 2.
  ENDLOOP.

  CLEAR t500l_wa.
  CALL FUNCTION 'HRPY_READ_T500L'
    EXPORTING
      molga          = gv_molga_p
    IMPORTING
      t500l_entry    = t500l_wa
    EXCEPTIONS
      no_entry_found = 1
      OTHERS         = 2.

  CHECK sy-subrc = 0.

  SORT lt_evpdir BY fpper inper seqnr.
  LOOP AT lt_evpdir WHERE srtza EQ 'A'.
    IF lt_evpdir-ocrsn IS INITIAL OR lt_evpdir-ocrsn EQ '0007'.
      CHECK lt_evpdir-fpper EQ lv_inper AND lt_evpdir-inper EQ lv_inper.
    ELSE.
      CHECK lt_evpdir-bondt EQ p_bondt.
      CHECK lt_evpdir-ocrsn EQ p_ocrsn.
    ENDIF.

    CALL FUNCTION 'PYXX_READ_PAYROLL_RESULT'
      EXPORTING
        employeenumber               = pernr-pernr
        sequencenumber               = lt_evpdir-seqnr

      EXCEPTIONS

        import_mismatch_error        = 3
        subpool_dir_full             = 4
        no_read_authority            = 5

      APPEND lt_pay_result TO lt_tab01.
    ENDIF.
  ENDLOOP.


  CHECK lt_tab01[] IS NOT INITIAL.
  SORT lt_tab01 BY evp-fpper evp-inper.
  MODIFY lt_tab01 FROM ls_tab01 TRANSPORTING evp-seqnr WHERE evp-seqnr NE space.

  LOOP AT lt_tab01 INTO ls_tab01.

    lv_paydt = ls_tab01-evp-paydt.

    SORT ls_tab01-inter-bt BY btznr.

    LOOP AT ls_tab01-inter-bt INTO ls_bt.
      CLEAR gt_data.
      MOVE-CORRESPONDING pernr TO gt_data.
      gt_data-ocrsn = p_ocrsn.
      gt_data-ocrsn_txt = gv_ocrsn_txt.

      CHECK ls_bt-lgart EQ '/559'.

*       금액
      ADD ls_bt-betrg TO gt_data-betrg.

*       은행코드
      gt_data-bankl = ls_bt-bankl.

*       은행명
      SELECT SINGLE *
      FROM bnka
      WHERE banks = ls_bt-banks AND
            bankl = gt_data-bankl.
      IF sy-subrc = 0.
        gt_data-banka = bnka-banka.
      ENDIF.

*       계좌번호
      gt_data-bankn = ls_bt-bankn.

      TRANSLATE gt_data-bankn USING '- '.
      CONDENSE gt_data-bankn NO-GAPS.


*       통화
      gt_data-waers = ls_bt-waers.

*       예금주
      IF ls_bt-emftx IS NOT INITIAL.
        gt_data-ename = ls_bt-emftx.
      ENDIF.

      PERFORM get_person_info USING      gt_data-pernr
                                                                 lv_paydt
                                          CHANGING   gt_data-btrtl
                                                              gt_data-btext
                                                              gt_data-orgeh
                                                              gt_data-orgtx.

      APPEND gt_data.
      CLEAR gt_data.
    ENDLOOP.

  ENDLOOP.
ENDFORM. " GET_DATA

FORM field_setting_call USING p_gub
                                                   p_fname
                                                   p_con.

  IF   p_gub =  'S'.
    CLEAR gs_fieldcat.
    gs_fieldcat-fieldname  = p_fname.
    EXIT.
  ENDIF.
  DATA l_col(40).
  FIELD-SYMBOLS <fs>.
  CONCATENATE 'GS_FIELDCAT-' p_fname  INTO l_col.
  ASSIGN      (l_col)         TO       <fs>.
  MOVE         p_con          TO       <fs>.

  IF p_fname = 'SELTEXT_S'.
    gs_fieldcat-colddictxt = 'S'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_L'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.
  ELSEIF p_fname = 'SELTEXT_L'.
    gs_fieldcat-colddictxt = 'L'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.

    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_M'  INTO l_col.
    ASSIGN      (l_col)         TO       <fs>.
    MOVE         p_con          TO       <fs>.
  ELSEIF p_fname = 'SELTEXT_M'.
    gs_fieldcat-colddictxt = 'M'.
    CONCATENATE 'GS_FIELDCAT-' 'SCRTEXT_S'  INTO l_col.
