FORM get_pns_list USING it_p0881 TYPE tt_p0881 iv_is_yea TYPE char1.
  DATA:
    lt_p0881      TYPE STANDARD TABLE OF p0881,
    lt_p0881_tmp  TYPE STANDARD TABLE OF p0881,
    lt_txt        TYPE textpool_table,
    ls_p0881      TYPE p0881,
    lv_ded_amt    TYPE pc207-betrg,
    lv_ded_tot    TYPE pc207-betrg,
    lv_rate       TYPE t511k-kwert,
    lv_limit      TYPE pc207-betrg.
  DATA:
    lv_old_subty  TYPE p0881-subty,
    lv_old_pnsty  TYPE p0881-pnsty.
  FIELD-SYMBOLS:
    <fs_txt>      TYPE textpool.

  CLEAR gt_pns. REFRESH gt_pns.

  lt_p0881 = it_p0881.
  DELETE lt_p0881 WHERE subty NE pkr99_retire_pens
                    AND subty NE pkr99_pens_saving
                    AND subty NE pkr99_house_saving
                    AND subty NE pkr99_lt_aggr_inv_sec
                    AND subty NE pkr99_lt_stock
                    AND subty NE pkr99_invest_assoc
                    AND subty NE pkr99_isa_saving
                    AND subty NE pkr99_lt_aggr_inv_sec_yg
                     .
  CHECK lt_p0881 IS NOT INITIAL.
  SORT lt_p0881 BY subty pnsty.

  lt_p0881_tmp[] = lt_p0881[].
  DELETE lt_p0881_tmp WHERE subty NE pkr99_invest_assoc OR pnsty+1(1) NE '3'.
  IF lt_p0881_tmp IS NOT INITIAL.
    SORT lt_p0881_tmp BY subty ASCENDING pnsty DESCENDING.
    PERFORM re511p USING '41' 'LTVET' tax_endda.
    lv_limit = t511p-betrg.
  ENDIF.

  IF lt_txt IS INITIAL.
    READ TEXTPOOL 'HKRCYED0' INTO lt_txt LANGUAGE form_lg.
  ENDIF.

  LOOP AT lt_p0881 INTO ls_p0881.
    gt_pns-subt = ls_p0881-subty.
    gt_pns-finc = ls_p0881-finco.
    PERFORM re7kr5dt USING ls_p0881-finco tax_begda form_lg
                  CHANGING gt_pns-insn.

    gt_pns-accn = ls_p0881-accno.
    gt_pns-pnsp = ls_p0881-pnsty.
    gt_pns-amtp = ls_p0881-nam01.

    SELECT SINGLE pnstx FROM t7kr5ct INTO gt_pns-pnst
      WHERE infty = ls_p0881-infty
        AND subty = ls_p0881-subty
        AND pnsty = ls_p0881-pnsty
        AND langu = form_lg.

    CASE ls_p0881-subty.
      WHEN pkr99_house_saving.
        IF ls_p0881-pnsty = '02'.
          PERFORM house_saving_text USING ls_p0881-rcbeg
                                          lt_txt
                                 CHANGING gt_pns-pnst.
        ENDIF.
      WHEN pkr99_lt_aggr_inv_sec.
        CLEAR gt_pns-pnsp.
      WHEN pkr99_invest_assoc.
        gt_pns-inyr = tax_begda(4) - ls_p0881-pnsty(1) + 1.
        PERFORM invest_assoc_text USING ls_p0881-pnsty
                                        lt_txt
                               CHANGING gt_pns-pnst.

        PERFORM invest_2_limit USING lt_p0881_tmp
                                     ls_p0881-pnsty
                                     lv_limit
                            CHANGING gt_pns-amtp.
      WHEN pkr99_lt_aggr_inv_sec_yg.
        gt_pns-ptbe = ls_p0881-rcbeg.
        gt_pns-cprd = ls_p0881-lnprd.
      WHEN OTHERS.
    ENDCASE.
    PERFORM amount_to_camount USING gt_pns-amtp CHANGING gt_pns-amnt.

*   Deduction or Tax Credit
    IF ls_p0881-subty NE lv_old_subty OR
      ls_p0881-pnsty NE lv_old_pnsty.
      CLEAR: lv_ded_tot, lv_rate.
      PERFORM pns_rtc_rate   USING ls_p0881-subty ls_p0881-pnsty
                        CHANGING lv_ded_tot lv_rate.
    ENDIF.
    lv_ded_amt = trunc( ls_p0881-nam01 * lv_rate ) / 100.
    IF lv_ded_amt < lv_ded_tot.
      lv_ded_tot = lv_ded_tot - lv_ded_amt.
    ELSE.
      lv_ded_amt = lv_ded_tot.
      lv_ded_tot = 0.
    ENDIF.
    gt_pns-amte = lv_ded_amt.

    IF ls_p0881-subty = pkr99_invest_assoc.
      CLEAR gt_pns-amte.
    ENDIF.

    PERFORM amount_to_camount USING gt_pns-amte CHANGING gt_pns-amtd.

    IF iv_is_yea IS NOT INITIAL.
      IF lv_ded_amt > 0 OR ls_p0881-subty = pkr99_isa_saving.
        APPEND gt_pns.
      ENDIF.
    ELSE.
      IF pns_data-is_beyea IS NOT INITIAL OR
        ( pns_data-is_beyea IS INITIAL AND ( lv_ded_amt > 0
         OR ls_p0881-subty = pkr99_isa_saving ) ).
        gv_pen_sav_indic = 'O'.
        APPEND gt_pns.
      ENDIF.
    ENDIF.

    lv_old_subty = ls_p0881-subty.
    lv_old_pnsty = ls_p0881-pnsty.
    CLEAR gt_pns.
  ENDLOOP.
ENDFORM.

        lv_konst = 'TXCR1'.

    p_rate = t511k-kwert.
  ENDIF.
ENDFORM.
