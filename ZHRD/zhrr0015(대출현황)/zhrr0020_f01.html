FORM select_data .
  DATA : lv_paydt TYPE d.

  CLEAR: gt_rgdir[], gt_rgdir.

* 사번에 대한 급여 내역 READ : PC261
  CALL FUNCTION 'CU_READ_RGDIR'
    EXPORTING
      persnr          = pernr-pernr
    TABLES
      in_rgdir        = gt_rgdir[]
    EXCEPTIONS
      no_record_found = 1
      OTHERS          = 2.
  CHECK sy-subrc EQ 0.


  CLEAR: gt_evpdir, gt_evpdir[].
  LOOP AT gt_rgdir WHERE srtza = 'A' AND ( ocrsn = space OR ocrsn = '0007' ).
    IF gt_rgdir-fpbeg = pnpdisbd AND gt_rgdir-fpend = pnpdised.
      MOVE-CORRESPONDING gt_rgdir TO gt_evpdir.
      APPEND gt_evpdir.
    ENDIF.
  ENDLOOP.

  CHECK gt_evpdir[] IS NOT INITIAL.
  SORT gt_evpdir DESCENDING BY seqnr.

  CLEAR : gt_pay_result_kr, gt_pay_result_kr[],
               gt_v0, gt_v0[],
               gt_rt, gt_rt[].

  LOOP AT gt_evpdir.
* 평가대상의 급여 데이터를 읽는다.
    CLEAR: gt_pay_result.
    CALL FUNCTION 'PYXX_READ_PAYROLL_RESULT'
      EXPORTING
        clusterid                    = 'KR'
        employeenumber               = pernr-pernr
        sequencenumber               = gt_evpdir-seqnr
      CHANGING
        payroll_result               = gt_pay_result_kr
      EXCEPTIONS
        illegal_isocode_or_clusterid = 1
        error_generating_import      = 2
        import_mismatch_error        = 3
        subpool_dir_full             = 4
        no_read_authority            = 5
        no_record_found              = 6
        versions_do_not_match        = 7
        OTHERS                       = 8.

    IF sy-subrc = 0.
      CLEAR :gt_v0, gt_v0[], lv_paydt.
      MOVE gt_pay_result_kr-inter-v0[] TO gt_v0[].
      MOVE gt_pay_result_kr-inter-rt[] TO gt_rt[].
      MOVE gt_pay_result_kr-inter-versc TO gt_versc.
      lv_paydt = gt_evpdir-paydt.
    ENDIF.
  ENDLOOP.

  CHECK gt_v0[] IS NOT INITIAL AND gt_rt[] IS NOT INITIAL.

  PERFORM get_info USING lv_paydt.

ENDFORM. " SELECT_DATA

FORM get_info USING p_paydt.

  LOOP AT gt_v0.
* 소속
    DATA lv_orgtx(25) TYPE c.
    CLEAR lv_orgtx.

    rp-read-infotype pernr-pernr 0001 p0001 pnpdised pnpdised.
    IF pnp-sw-found = 1.
      gs_data-orgeh = p0001-orgeh.

      SELECT SINGLE orgtx
      FROM t527x
      INTO lv_orgtx
      WHERE sprsl = sy-langu
        AND orgeh = p0001-orgeh
        AND begda <= pnpdised
        AND endda => pnpdised.

      IF sy-subrc = 0.
        gs_data-orgeht = lv_orgtx.
      ENDIF.
    ENDIF.


* 사원 번호
    gs_data-pernr = pernr-pernr.

* 성명
    gs_data-ename = pernr-ename.

    CLEAR : p0045, p0045[].
    rp-read-infotype pernr-pernr 0045 p0045 pnpdisbd pnpdised.

    LOOP AT p0045 WHERE subty EQ gt_v0-vinfo(4)
                                  AND  objps EQ gt_v0-vinfo+4(2).
* 대출유형
      gs_data-dlart = p0045-dlart.
      SELECT SINGLE stext INTO gs_data-dtext  FROM t591s
                                                              WHERE sprsl = sy-langu
                                                                 AND infty = '0045'
                                                                 AND subty = gs_data-dlart.

* 대출일
      gs_data-begda = p0045-datbw.

* 대출 금액
      gs_data-darbt = p0045-darbt.

    ENDLOOP.

* 당월 잔액
    READ TABLE gt_rt WITH KEY v0znr = gt_v0-v0znr lgart = '/LLB'.
    IF sy-subrc EQ 0.
      gs_data-thatm = gt_rt-betrg.
    ENDIF.

* 상환일
    gs_data-dday = p_paydt.
*  급여에서의 원금상환액
    LOOP AT gt_rt WHERE  v0znr = gt_v0-v0znr AND ( lgart = '/LRP' OR lgart = '/LEP').
      gs_data-wageb = gs_data-wageb + gt_rt-betrg.
      CLEAR gt_rt.
    ENDLOOP.

* 이자
    READ TABLE gt_rt WITH KEY v0znr = gt_v0-v0znr
                                                    lgart = '/LID'.
    IF sy-subrc = 0.
      gs_data-interb = + gt_rt-betrg.
    ENDIF.

* 외부 상환액
    LOOP AT gt_rt WHERE  v0znr = gt_v0-v0znr AND ( lgart = '/LEE' OR lgart = '/LEX').
      gs_data-etcde = gs_data-etcde  + gt_rt-betrg.
      CLEAR gt_rt.
    ENDLOOP.

* 외부상환 이자
    READ TABLE gt_rt WITH KEY v0znr = gt_v0-v0znr
                                                    lgart = '9LID'.
    IF sy-subrc = 0.
      gs_data-etlid = gt_rt-betrg.
    ENDIF.

* 외부상환일
    CLEAR : p0078, p0078[].
    rp-read-infotype pernr-pernr 0078 p0078 pnpdisbd pnpdised.
    READ TABLE p0078 WITH KEY dlart =  gt_v0-vinfo(4)
                                                  objid = gt_v0-vinfo+4(2)
                                                  subty = '0200'.
    IF sy-subrc = 0.
      gs_data-edate = p0078-begda.
    ENDIF.


* 전 월 잔액
    gs_data-lastm = gs_data-thatm + gs_data-wageb + gs_data-etcde.

* 월상환금액 합계
    gs_data-biltot = gs_data-wageb + gs_data-interb + gs_data-etcde + gs_data-etlid.

* 통화
    gs_data-waers = gt_versc-waers.

    IF gs_data IS NOT INITIAL.
      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDIF.

  ENDLOOP.
ENDFORM. " GET_INFO
