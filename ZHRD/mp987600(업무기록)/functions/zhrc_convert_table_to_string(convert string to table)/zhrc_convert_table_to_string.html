Description: Convert String To Table
FUNCTION ZHRC_CONVERT_TABLE_TO_STRING.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(I_TABLINE_LENGTH) TYPE I
*" EXPORTING
*" REFERENCE(E_STRING) TYPE STRING
*" TABLES
*" IT_TABLE TYPE TABLE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: L_LENGTH TYPE I,
        L_SPACE  TYPE FLAG.

  DATA: LD_CRLF  TYPE ABAP_CR_LF.

  CLASS CL_ABAP_CHAR_UTILITIES DEFINITION LOAD.
  LD_CRLF = CL_ABAP_CHAR_UTILITIES=>CR_LF.

  LOOP AT IT_TABLE.
    IF L_SPACE = 'X'.
      CONCATENATE E_STRING IT_TABLE
             INTO E_STRING SEPARATED BY LD_CRLF.
      CLEAR L_SPACE.
    ELSE.
      CONCATENATE E_STRING IT_TABLE
             INTO E_STRING.
    ENDIF.

    L_LENGTH = I_TABLINE_LENGTH - 1.
    IF IT_TABLE+L_LENGTH = ' '.
      L_SPACE = 'X'.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
