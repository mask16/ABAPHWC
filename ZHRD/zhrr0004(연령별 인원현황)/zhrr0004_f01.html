*----------------------------------------------------------------------*
FORM collect_data USING p_gbdat.
  DATA : mok TYPE i,
             l_fieldname(20).

  FIELD-SYMBOLS : <fs>.

  IF  NOT p_gbdat IS INITIAL.

    CLEAR : no_year, no_month.

    CALL FUNCTION 'HR_SGPBS_YRS_MTHS_DAYS'
      EXPORTING
        beg_da   = p_gbdat
        end_da   = pn-endda
      IMPORTING
        no_month = no_month
        no_year  = no_year.

    IF sy-subrc = 0.
      gt_display-no_year = no_year.

      IF  no_year BETWEEN 20 AND 54.
        mok = no_year DIV 5.
        l_fieldname = mok * 5.
        CONDENSE l_fieldname NO-GAPS.
        CONCATENATE 'P_'  l_fieldname  INTO l_fieldname.
      ELSEIF no_year >= 55.
        l_fieldname = 'P_55'.
      ELSEIF no_year < 20.
        l_fieldname = 'P_10'.
      ENDIF.

      CONDENSE l_fieldname NO-GAPS.
      ASSIGN COMPONENT 'YEAR'  OF STRUCTURE <fs_wa> TO <fs>.
      gt_display-year = <fs> = l_fieldname+2(3).
      ASSIGN COMPONENT  l_fieldname OF STRUCTURE <fs_wa> TO <fs>.
      <fs> = 1.
    ENDIF.

  ELSE.
    ASSIGN COMPONENT 'NOIN'  OF STRUCTURE <fs_wa> TO <fs>.
    <fs> = 1.
  ENDIF.

  ASSIGN COMPONENT 'AVG'  OF STRUCTURE <fs_wa> TO <fs>.
  <fs> = no_year.

  ASSIGN COMPONENT 'P_SUM'  OF STRUCTURE <fs_wa> TO <fs>.
  <fs> = 1.

  COLLECT <fs_wa> INTO <fs_table>.
  CLEAR <fs_wa>.

  gt_display-no_year = no_year.
  APPEND gt_display.
  CLEAR gt_display.

ENDFORM. " COLLECT_DATA

FORM double_p1 USING rs_selfield TYPE slis_selfield.
  FIELD-SYMBOLS : <fs>,
                               <fs2>,
                               <fs3>.

  CLEAR : lt_pernr, lt_pernr[].

  READ TABLE <fs_table> INTO <fs_wa>  INDEX rs_selfield-tabindex.
  CHECK sy-subrc = 0.
************************************************************
  ASSIGN COMPONENT 'P_SUM'  OF STRUCTURE <fs_wa> TO <fs>.
  ASSIGN COMPONENT 'TGBTX' OF STRUCTURE <fs_wa> TO <fs2>.
  ASSIGN COMPONENT 'TRFGR' OF STRUCTURE <fs_wa> TO <fs3>.

  IF rs_selfield-fieldname = 'P_SUM' AND NOT <fs> IS INITIAL.
    LOOP AT gt_display WHERE tgbtx     = <fs2> AND
                                                  trfgr     = <fs3>.
      PERFORM get_pernr USING gt_display-pernr.
    ENDLOOP.
  ELSEIF rs_selfield-fieldname NE 'P_SUM'.
    ASSIGN COMPONENT  rs_selfield-fieldname OF STRUCTURE <fs_wa> TO <fs>.
    CHECK NOT <fs> IS INITIAL.
    LOOP AT gt_display WHERE  tgbtx     = <fs2>
                                      AND trfgr     = <fs3>
                                      AND year = rs_selfield-fieldname+2(2).
      PERFORM get_pernr USING gt_display-pernr.
    ENDLOOP.

  ENDIF.

  IF NOT lt_pernr[] IS INITIAL.
    SUBMIT  zhrdetail     WITH pnppernr-low  IN lt_pernr
                                           AND RETURN.
  ENDIF.
ENDFORM. " DOUBLE_P1
