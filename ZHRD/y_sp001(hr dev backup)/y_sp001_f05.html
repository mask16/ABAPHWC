FORM SCANFORADDITIONALBSPSTUFF USING ILOCBSPAPPL LIKE IBSPAPPL[]
                                      ILOCFUNCTIONS LIKE IFUNCTIONS[]
                                      ILOCCLASSES LIKE ICLASSES[]
                                      VALUE(GETTEXTELEMENTS)
                                      VALUE(GETMESSAGES)
                                      VALUE(GETSCREENS)
                                      VALUE(GETCUSTDICTSTRUCTURES)
                                      VALUE(GETFUNCTIONS)
                                      VALUE(GETINCLUDES)
                                      VALUE(CUSTOMERONLY)
                                      VALUE(CUSTOMERNAMERANGE)
                                      VALUE(RECURSIVEINCLUDESEARCH)
                                      VALUE(RECURSIVEFUNCSEARCH)
                                      VALUE(GETCLASSES)
                                      VALUE(RECURSIVECLASSSEARCH)
                                      VALUE(SOLOCPACKAGE) LIKE SOPACK[].

  DATA: WABSPAPPL  TYPE TBSPAPPL.
  DATA: WAPAGEDETL TYPE TPAGEDETL.
  DATA: MYTABIX TYPE SYTABIX.

  DATA: WACLASS TYPE TCLASS,
        WALINKN TYPE TLINKNAME.

  DATA: IBSPCLASSES
          TYPE STANDARD TABLE OF TCLASS WITH HEADER LINE.

* Best to find all the Pages used in a program first
  IF NOT GETINCLUDES IS INITIAL.
    LOOP AT ILOCBSPAPPL INTO WABSPAPPL.
      MYTABIX = SY-TABIX.
      PERFORM SCANFORPAGESBSPAPPL USING WABSPAPPL-APPLNAME
                                        WABSPAPPL-PAGEDETL[].

*  BSP APPLICATION CLASS GET
      IF NOT GETCLASSES IS INITIAL
       AND WABSPAPPL-APPLCLAS IS NOT INITIAL.
        READ TABLE ILOCCLASSES INTO WACLASS
                        WITH KEY CLSNAME = WABSPAPPL-APPLCLAS.
        IF SY-SUBRC <> 0.
          REFRESH : IBSPCLASSES.
          CLEAR : IBSPCLASSES, WACLASS.

          WACLASS-CLSNAME = WABSPAPPL-APPLCLAS.
*          WACLASS-PROGRAMLINKNAME = WABSPAPPL-APPLNAME.
          APPEND WACLASS TO IBSPCLASSES.

          PERFORM SCANFORADDITIONALCLASSSTUFF
                    USING IBSPCLASSES[]
                          ILOCFUNCTIONS[]
                          GETTEXTELEMENTS
                          GETMESSAGES
                          GETCUSTDICTSTRUCTURES
                          GETFUNCTIONS
                          GETINCLUDES
                          CUSTOMERONLY
                          CUSTOMERNAMERANGE
                          RECURSIVEINCLUDESEARCH
                          RECURSIVEFUNCSEARCH
                          RECURSIVECLASSSEARCH
                          SOLOCPACKAGE[].

          IF IBSPCLASSES[] IS NOT INITIAL.
            LOOP AT IBSPCLASSES INTO WACLASS.
              WALINKN-APPLNAME = WABSPAPPL-APPLNAME.
*              WACLASS-PROGRAMLINKNAME = WABSPAPPL-APPLNAME.
              APPEND WALINKN TO WACLASS-PROGRAMLINKNAME.
              APPEND WACLASS TO ILOCCLASSES.
            ENDLOOP.
          ENDIF.
        ELSE.
          READ TABLE WACLASS-PROGRAMLINKNAME INTO WALINKN
                             WITH KEY APPLNAME = WABSPAPPL-APPLNAME.
          IF SY-SUBRC <> 0.
            WALINKN-APPLNAME = WABSPAPPL-APPLNAME.
            APPEND WALINKN TO WACLASS-PROGRAMLINKNAME.
            MODIFY ILOCCLASSES FROM WACLASS INDEX SY-TABIX.
          ENDIF.
        ENDIF.
      ENDIF.

*     NAVIGATION GET
      PERFORM GET_NAVGRAPH_RTN USING WABSPAPPL-APPLNAME
                            CHANGING WABSPAPPL-INAVGR[].

      MODIFY ILOCBSPAPPL FROM WABSPAPPL INDEX MYTABIX.
    ENDLOOP.
  ENDIF.
*
* Once we have a list of all the includes we need to loop round
* them an select all the other objects
  LOOP AT ILOCBSPAPPL INTO WABSPAPPL.
    MYTABIX = SY-TABIX.
    LOOP AT WABSPAPPL-PAGEDETL INTO WAPAGEDETL.
      IF NOT GETCUSTDICTSTRUCTURES IS INITIAL.

        PERFORM SCANFORTABLES_BSPHANDLER USING WAPAGEDETL-T_HANDLER[]
                                               CUSTOMERONLY
                                               CUSTOMERNAMERANGE
                                               WABSPAPPL-IDICTSTRUCT[].

        PERFORM SCANFORTABLES_BSPATTR USING WAPAGEDETL-T_ATTR[]
                                            CUSTOMERONLY
                                            CUSTOMERNAMERANGE
                                            WABSPAPPL-IDICTSTRUCT[].

        PERFORM SCANFORTABLES_BSPTYPES USING WAPAGEDETL-T_TYPES[]
                                             RECURSIVEINCLUDESEARCH
                                             CUSTOMERONLY
                                             CUSTOMERNAMERANGE
                                             WABSPAPPL-IDICTSTRUCT[]
                                             WABSPAPPL-IINCLUDES[].
      ENDIF.

* Now we have all the program includes and details
* we need to find extra functions
      IF NOT GETFUNCTIONS IS INITIAL.
**        Find any functions defined in the code
        PERFORM SCANFORFUNCTIONS_BSPHANDLER
                                 USING WAPAGEDETL-T_HANDLER[]
                                       WABSPAPPL-APPLNAME
                                       SPACE
                                       SPACE
                                       CUSTOMERONLY
                                       CUSTOMERNAMERANGE
                                       ILOCFUNCTIONS[].
      ENDIF.

    ENDLOOP.

    MODIFY ILOCBSPAPPL FROM WABSPAPPL INDEX MYTABIX.
  ENDLOOP.
* We have a list of all the functions so lets go
* and find details and other function calls
*  PERFORM SCANFORADDITIONALFUNCSTUFF USING ILOCFUNCTIONS[]
*                                           RECURSIVEINCLUDESEARCH
*                                           RECURSIVEFUNCSEARCH
*                                           GETINCLUDES
*                                           GETFUNCTIONS
*                                           GETCUSTDICTSTRUCTURES
*                                           GETMESSAGES
*                                           CUSTOMERONLY
*                                           CUSTOMERNAMERANGE.

ENDFORM. " SCANFORADDITIONALBSPSTUFF

FORM SCANFORPAGESBSPAPPL USING VALUE(P_APPLNAME)
                                   ILOCPAGEDETL LIKE DUMIPAGEDETL[].

  DATA: WAPAGEDETL TYPE TPAGEDETL.

  DATA: PAGE_LIST TYPE O2PAGELIST.
  FIELD-SYMBOLS: <PAGE> TYPE O2PAGATTR.

  DATA: PAGE_KEY   TYPE O2PAGKEY.
  DATA: P_PAGEATTR TYPE O2PAGATTR,
        P_SOURCE   TYPE O2PAGELINE_TABLE,
        P_HANDLER  TYPE O2PAGEVH_TABLETYPE,
        P_ATTR     TYPE O2PAGPAR_TABLETYPE,
        P_TYPE     TYPE RSWSOURCET.

  DATA: PAGE_REF TYPE REF TO CL_O2_API_PAGES.
  DATA: WA_ATTR    TYPE O2PAGPARS,
        WA_CATTR   TYPE TBSPATTR.

  CALL FUNCTION 'O2_GET_ALL_PAGES'
    EXPORTING
      P_APPLNAME = P_APPLNAME
      P_VERSION  = 'A'
    IMPORTING
      P_PAGES    = PAGE_LIST.

  LOOP AT PAGE_LIST ASSIGNING <PAGE>.
    MOVE-CORRESPONDING <PAGE> TO WAPAGEDETL.

    CLEAR : PAGE_KEY, P_PAGEATTR, P_SOURCE,
            P_HANDLER, P_ATTR, P_TYPE.

    PAGE_KEY-APPLNAME = <PAGE>-APPLNAME.
    PAGE_KEY-PAGEKEY = <PAGE>-PAGEKEY.

    CALL METHOD CL_O2_API_PAGES=>LOAD
      EXPORTING
        P_PAGEKEY             = PAGE_KEY
        P_VERSION             = SO2_VERSION_ACTIVE
        P_LANGU               = SY-LANGU
        P_WITH_ALL_TEXTS      = 'X'
        P_LOAD_FOR_GENERATION = 'X'
      IMPORTING
        P_PAGE                = PAGE_REF
      EXCEPTIONS
        OTHERS                = 1.

    IF SY-SUBRC = 0.
      CALL METHOD PAGE_REF->GET_ATTRS
        IMPORTING
          P_ATTRS = P_PAGEATTR
        EXCEPTIONS
          OTHERS  = 1.

      IF P_PAGEATTR-PAGETYPE <> SO2_CONTROLLER.
        CALL METHOD PAGE_REF->GET_PAGE
          IMPORTING
            P_CONTENT = P_SOURCE
          EXCEPTIONS
            OTHERS    = 1.

        CALL METHOD PAGE_REF->GET_PARAMETERS
          IMPORTING
            P_PARAMETERS = P_ATTR
          EXCEPTIONS
            OTHERS       = 1.

        CALL METHOD PAGE_REF->GET_EVENT_HANDLERS
          IMPORTING
            P_EV_HANDLER = P_HANDLER
          EXCEPTIONS
            OTHERS       = 1.

        CALL METHOD PAGE_REF->GET_TYPE_SOURCE
          IMPORTING
            P_SOURCE = P_TYPE
          EXCEPTIONS
            OTHERS   = 1.
      ENDIF.

      WAPAGEDETL-DESCRIPT = P_PAGEATTR-DESCRIPT.

FORM SCANFORLIKEORTYPE_BSP USING ILINES LIKE DUMILINES[]
                                   VALUE(CUSTOMERONLY)
                                   VALUE(CUSTOMERNAMERANGE)
                                   ILOCDICTIONARY LIKE IDICTIONARY[].

  DATA: LT_LINES TYPE STANDARD TABLE OF STRING WITH HEADER LINE.
  DATA: HEAD TYPE STRING.
  DATA: TAIL TYPE STRING.
  DATA: JUNK TYPE STRING.
  DATA: LINETYPE TYPE STRING.
  DATA: LINELENGTH TYPE I VALUE 0.
  DATA: ENDOFLINE TYPE ABAP_BOOL VALUE TRUE.
  DATA: WADICTIONARY TYPE TDICTTABLE.
  DATA: WADICTIONARYCOMPARISON TYPE TDICTTABLE.
  DATA: WALINE TYPE STRING.

  LT_LINES[] = ILINES[].

  LOOP AT LT_LINES INTO WALINE.
*   Find custom tables.
    LINELENGTH = STRLEN( WALINE ).
    IF LINELENGTH > 0.
      IF WALINE(1) = ASTERIX.
        CONTINUE.
      ENDIF.

      TRANSLATE WALINE TO UPPER CASE.

*     Determine the lineType.
      IF ENDOFLINE = TRUE.
        SHIFT WALINE UP TO LIKE.
        IF SY-SUBRC = 0.
          LINETYPE = LIKE.
        ELSE.
          SHIFT WALINE UP TO TYPE.
          IF SY-SUBRC = 0.
            FIND 'BEGIN OF' IN WALINE.
            IF SY-SUBRC <> 0.
              FIND 'END OF' IN WALINE.
              IF SY-SUBRC <> 0.
                FIND 'VALUE' IN WALINE.
                IF SY-SUBRC <> 0.
                  LINETYPE = TYPE.
                ENDIF.
              ENDIF.
            ENDIF.
          ELSE.
            SHIFT WALINE UP TO INCLUDE.
            IF SY-SUBRC = 0.
              SPLIT WALINE AT SPACE INTO JUNK LT_LINES.
            ENDIF.

            SHIFT WALINE UP TO STRUCTURE.
            IF SY-SUBRC = 0.
              LINETYPE = STRUCTURE.
            ELSE.
              CONTINUE.
            ENDIF.
          ENDIF.
        ENDIF.
      ELSE.
        LINETYPE = COMMA.
      ENDIF.

      CASE LINETYPE.
        WHEN LIKE OR TYPE OR STRUCTURE.
*         Work on the appropriate lineType
          SHIFT WALINE UP TO SPACE.
          SHIFT WALINE LEFT DELETING LEADING SPACE.
          IF WALINE CS TABLE.
            SPLIT WALINE AT TABLE INTO HEAD TAIL.
            SPLIT TAIL AT 'OF' INTO HEAD TAIL.
            WALINE = TAIL.
            SHIFT WALINE LEFT DELETING LEADING SPACE.
            REPLACE ALL OCCURRENCES OF 'WITH HEADER LINE'
                    IN WALINE WITH ''.
          ENDIF.

*         Are we only to download SAP dictionary structures.
          IF NOT CUSTOMERONLY IS INITIAL.
            TRY.
                IF WALINE+0(1) = 'Y'
                OR WALINE+0(1) = 'Z'
                OR WALINE CS CUSTOMERNAMERANGE.
                ELSE.
                  LINETYPE = ''.
                  CONTINUE.
                ENDIF.
              CATCH CX_SY_RANGE_OUT_OF_BOUNDS INTO OBJRUNTIMEERROR.
            ENDTRY.
          ENDIF.

          IF WALINE CS COMMA.
            SPLIT WALINE AT COMMA INTO HEAD TAIL.
            IF WALINE CS DASH.
              SPLIT HEAD AT DASH INTO HEAD TAIL.
            ENDIF.
            IF WALINE CS OCCURS.
              SPLIT WALINE AT SPACE INTO HEAD TAIL.
            ENDIF.
          ELSE.
            IF WALINE CS PERIOD.
              SPLIT WALINE AT PERIOD INTO HEAD TAIL.
              IF WALINE CS DASH.
                SPLIT HEAD AT DASH INTO HEAD TAIL.
              ENDIF.
              IF WALINE CS OCCURS.
                SPLIT WALINE AT SPACE INTO HEAD TAIL.
              ENDIF.
            ELSE.
              SPLIT WALINE AT SPACE INTO HEAD TAIL.
              IF WALINE CS DASH.
                SPLIT HEAD AT DASH INTO HEAD TAIL.
              ENDIF.
            ENDIF.
          ENDIF.

          IF NOT HEAD IS INITIAL.
            WADICTIONARY-TABLENAME = HEAD.
*           Don't append the object if we already have it listed
            READ TABLE ILOCDICTIONARY INTO WADICTIONARYCOMPARISON
                       WITH KEY TABLENAME = WADICTIONARY-TABLENAME.
            IF SY-SUBRC <> 0.
              PERFORM FINDTABLEDESCRIPTION
                        USING WADICTIONARY-TABLENAME
                               WADICTIONARY-TABLETITLE.

              PERFORM FINDTABLEDEFINITION
                        USING WADICTIONARY-TABLENAME
                              WADICTIONARY-TABLETITLE
                              WADICTIONARY-ISTRUCTURE[].

*             Only append if the item is a table
*             and not a structure or data element

              IF NOT WADICTIONARY-ISTRUCTURE[] IS INITIAL.
                APPEND WADICTIONARY TO ILOCDICTIONARY.
* 테이블 TYPE 유형구조 추가
                PERFORM FIND_EXTRA_TABLES
                          USING WADICTIONARY-ISTRUCTURE[]
                                ILOCDICTIONARY.
              ELSE.
*=========> TABLE TYPE 감안
                PERFORM FINDTABLETYPEDEFINITION
                          USING WADICTIONARY-TABLENAME
                                WADICTIONARY-TABLETITLE
                                WADICTIONARY-ISTRUCTURE[].
                IF NOT WADICTIONARY-ISTRUCTURE[] IS INITIAL.
                  APPEND WADICTIONARY TO ILOCDICTIONARY.

* 테이블 TYPE 유형구조 추가
                  PERFORM FIND_EXTRA_TABLES
                            USING WADICTIONARY-ISTRUCTURE[]
                                  ILOCDICTIONARY.
                ENDIF.
              ENDIF.

            ENDIF.
            CLEAR WADICTIONARY.
          ENDIF.

          LINETYPE = ''.
      ENDCASE.
    ENDIF.
  ENDLOOP.

ENDFORM. " SCANFORLIKEORTYPE_BSP

FORM SCANFORFUNCTIONS_BSP USING ILINES LIKE DUMILINES[]
                                 VALUE(PROGRAMLINKNAME)
                                 VALUE(RECURSIVEINCLUDES)
                                 VALUE(RECURSIVEFUNCTIONS)
                                 VALUE(CUSTOMERONLY)
                                 VALUE(CUSTOMERNAMERANGE)
                                 ILOCFUNCTIONS LIKE IFUNCTIONS[].

  DATA: ITOKENS TYPE STANDARD TABLE OF STOKES WITH HEADER LINE.
  DATA: ISTATEMENTS TYPE STANDARD TABLE OF SSTMNT WITH HEADER LINE.
  DATA: WATOKENS TYPE STOKES.
  DATA: WAFUNCTION TYPE TFUNCTION.
  DATA: WAFUNCTIONCOMPARISON TYPE TFUNCTION.
  DATA: MAXLINES TYPE I.
  DATA: NEXTLINE TYPE I.
  DATA: SKIPTHISLOOP TYPE ABAP_BOOL.

* Read the program code from the textpool.
  SCAN ABAP-SOURCE ILINES TOKENS INTO ITOKENS
                          WITH INCLUDES
                          STATEMENTS INTO ISTATEMENTS.

  MAXLINES = LINES( ITOKENS ).
  LOOP AT ITOKENS WHERE STR = FUNCTION AND TYPE = 'I'.

    NEXTLINE = SY-TABIX + 1.
    IF NEXTLINE <= MAXLINES.
      READ TABLE ITOKENS INDEX NEXTLINE INTO WATOKENS.

*      Are we only to find customer functions
      SKIPTHISLOOP = FALSE.
      IF NOT CUSTOMERONLY IS INITIAL.
        TRY.
            IF WATOKENS-STR+1(1) = 'Y'
            OR WATOKENS-STR+1(1) = 'Z'
            OR WATOKENS-STR CS CUSTOMERNAMERANGE.
            ELSE.
              SKIPTHISLOOP = TRUE.
            ENDIF.
          CATCH CX_SY_RANGE_OUT_OF_BOUNDS INTO OBJRUNTIMEERROR.
          CLEANUP.
            SKIPTHISLOOP = TRUE.
        ENDTRY.
      ENDIF.

      IF SKIPTHISLOOP = FALSE.
        WAFUNCTION-FUNCTIONNAME = WATOKENS-STR.
        REPLACE ALL OCCURRENCES OF '''' IN WAFUNCTION-FUNCTIONNAME
                                        WITH ' '.
        CONDENSE WAFUNCTION-FUNCTIONNAME.

*        Don't add a function if we alread have it listed.
*        READ TABLE ILOCFUNCTIONS INTO WAFUNCTIONCOMPARISON
*                   WITH KEY FUNCTIONNAME = WAFUNCTION-FUNCTIONNAME
*                            PROGRAMLINKNAME = PROGRAMLINKNAME.
        DATA : L_TABIX TYPE SY-TABIX.
        READ TABLE ILOCFUNCTIONS INTO WAFUNCTIONCOMPARISON
                  WITH KEY FUNCTIONNAME = WAFUNCTION-FUNCTIONNAME.
        IF SY-SUBRC <> 0.
*          Add in the link name if the function is linked to a program
*          WAFUNCTION-PROGRAMLINKNAME = PROGRAMLINKNAME.
          WA_TLPNAME-PNAME = PROGRAMLINKNAME.
          APPEND WA_TLPNAME TO WAFUNCTION-PROGRAMLINKNAME.

*          Don't download functions which are called through
*          an RFC destination
          NEXTLINE = SY-TABIX + 2.
          READ TABLE ITOKENS INDEX NEXTLINE INTO WATOKENS.
          IF WATOKENS-STR <> DESTINATION.

*            Find the function group
            SELECT SINGLE AREA FROM V_FDIR
                          INTO WAFUNCTION-FUNCTIONGROUP
                         WHERE FUNCNAME = WAFUNCTION-FUNCTIONNAME.

            IF SY-SUBRC = 0.
*              Best find the function number as well.
              PERFORM RETRIEVEFUNCTIONDETAIL
                        USING WAFUNCTION-FUNCTIONNAME
                               WAFUNCTION-PROGNAME
                               WAFUNCTION-INCLUDENUMBER
                               WAFUNCTION-FUNCTIONTITLE.

              PERFORM FINDMAINFUNCTIONINCLUDE
                        USING WAFUNCTION-PROGNAME
                              WAFUNCTION-FUNCTIONGROUP
                              WAFUNCTION-INCLUDENUMBER
                              WAFUNCTION-FUNCTIONMAININCLUDE.

              PERFORM FINDFUNCTIONTOPINCLUDE
                        USING WAFUNCTION-PROGNAME
                               WAFUNCTION-FUNCTIONGROUP
                               WAFUNCTION-TOPINCLUDENAME.

*============> 추가사항
              PERFORM SCANFORFUNCTIONINCLUDES
                        USING WAFUNCTION-PROGNAME
                              CUSTOMERONLY
                              CUSTOMERNAMERANGE
                              WAFUNCTION-IINCLUDES[].

*              Find main message class
              PERFORM FINDMAINMESSAGECLASS
                        USING WAFUNCTION-PROGNAME
                               WAFUNCTION-MESSAGECLASS.

              APPEND WAFUNCTION TO ILOCFUNCTIONS.

*              Now lets search a little bit deeper
*              and do a recursive search for other includes
              IF NOT RECURSIVEINCLUDES IS INITIAL.
                PERFORM SCANFORINCLUDEPROGRAMS
                          USING WAFUNCTION-FUNCTIONMAININCLUDE
                                 RECURSIVEINCLUDES
                                 CUSTOMERONLY
                                 CUSTOMERNAMERANGE
                                 WAFUNCTION-IINCLUDES[].
              ENDIF.

*              Now lets search a little bit deeper
*              and do a recursive search for other functions
              IF NOT RECURSIVEFUNCTIONS IS INITIAL.
                PERFORM SCANFORFUNCTIONS
                          USING WAFUNCTION-FUNCTIONMAININCLUDE
                                 PROGRAMLINKNAME  "SPACE
                                 RECURSIVEINCLUDES
                                 RECURSIVEFUNCTIONS
                                 CUSTOMERONLY
                                 CUSTOMERNAMERANGE
                                 ILOCFUNCTIONS[].
              ENDIF.
              CLEAR WAFUNCTION.
            ENDIF.
          ENDIF.
        ELSE.
          L_TABIX = SY-TABIX.
          READ TABLE WAFUNCTIONCOMPARISON-PROGRAMLINKNAME
                     INTO WA_TLPNAME
                     WITH KEY PNAME = PROGRAMLINKNAME.
          IF SY-SUBRC <> 0.
            WA_TLPNAME-PNAME = PROGRAMLINKNAME.
            APPEND WA_TLPNAME TO WAFUNCTIONCOMPARISON-PROGRAMLINKNAME.

            MODIFY ILOCFUNCTIONS FROM WAFUNCTIONCOMPARISON
                                 INDEX L_TABIX.
          ENDIF.
        ENDIF.

        CLEAR WAFUNCTION.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDFORM. " SCANFORFUNCTIONS_BSP
