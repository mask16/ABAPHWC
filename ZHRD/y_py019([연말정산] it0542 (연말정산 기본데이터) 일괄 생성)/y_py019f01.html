FORM INIT_VALUE .

  IF SY-DATUM+4(2) = '12'.
    PNPBEGDA(4) = SY-DATUM(4).
  ELSE.
    PNPBEGDA(4) = SY-DATUM+0(4) - 1.
  ENDIF.

  PNPBEGDA+4(4) = '1231'.

*  PNPTIMED = 'D'.
*  RP-SEL-EIN-AUS-INIT.  "현재원만 선택

  PNPWERKS = 'IEQ'.
  PNPWERKS-LOW = '1000'.
  APPEND PNPWERKS.


*  PNPXABKR = 'K2'.
*
*  PNPABKRS = 'IEQ'.
*  PNPABKRS-LOW = 'K1'. "급여영역
*  APPEND PNPABKRS.
*  PNPABKRS = 'IEQ'.
*  PNPABKRS-LOW = 'K2'. "급여영역
*  APPEND PNPABKRS.

*  DATA: L_YEAR(4) TYPE N.
*
*  IF SY-DATUM+4(2) = '12'.
*    L_YEAR = SY-DATUM(4).
*  ELSE.
*    L_YEAR = SY-DATUM+0(4) - 1.
*  ENDIF.

*  CONCATENATE L_YEAR '0101' INTO S_DATE-LOW.
*  CONCATENATE L_YEAR '1231' INTO S_DATE-HIGH.
  CONCATENATE PNPBEGDA(4) '0101' INTO S_DATE-LOW.
  CONCATENATE PNPBEGDA(4) '1231' INTO S_DATE-HIGH.
  APPEND S_DATE.

ENDFORM. " INIT_VALUE

FORM DATA_UPLOAD .

  &PIMSG PERNR-PERNR PERNR-ENAME.

  CLEAR: BDC_TAB, BDC_TAB[].

  PERFORM DYNPRO USING : 'X'  'SAPMP50A'         '1000',
                         ' '  'RP50G-PERNR'      PERNR-PERNR,
                         ' '  'RP50G-TIMR6'      'X',
                         ' '  'RP50G-BEGDA'      S_DATE-LOW,
                         ' '  'RP50G-ENDDA'      S_DATE-HIGH,
                         ' '  'RP50G-CHOIC'      '0542',
                         ' '  'BDC_OKCODE'       '=INS'.

  PERFORM DYNPRO USING : 'X'  'MP054200'         '2000',
                         ' '  'P0542-PDCID'      'X',  "인적공제변동여부
                         ' '  'BDC_OKCODE'       '=UPD'.

  CLEAR: MESSTAB, MESSTAB[].
  CALL TRANSACTION 'PA30' USING BDC_TAB
                           MODE 'N'
                         UPDATE 'S'
                       MESSAGES INTO MESSTAB.
  IF SY-SUBRC = 0.
  ELSE.
    LOOP AT MESSTAB.
      CALL FUNCTION 'MESSAGE_TEXT_BUILD'
        EXPORTING
          MSGID               = MESSTAB-MSGID
          MSGNR               = MESSTAB-MSGNR
          MSGV1               = MESSTAB-MSGV1
          MSGV2               = MESSTAB-MSGV2
          MSGV3               = MESSTAB-MSGV3
          MSGV4               = MESSTAB-MSGV4
        IMPORTING
          MESSAGE_TEXT_OUTPUT = MSGTEXT
        EXCEPTIONS
          OTHERS              = 1.
      WRITE: /4(100) MSGTEXT.

      WRITE:/2(30) '※ DATA 생성 실패~!',
            '사번 : ',
            PERNR-PERNR COLOR 6.
    ENDLOOP.
  ENDIF.

ENDFORM. " DATA_UPLOAD

FORM DYNPRO USING DYNBEGIN NAME VALUE.

  IF DYNBEGIN = 'X'.
    CLEAR BDC_TAB.
    MOVE: NAME  TO BDC_TAB-PROGRAM,
          VALUE TO BDC_TAB-DYNPRO,
          'X'   TO BDC_TAB-DYNBEGIN.
    APPEND BDC_TAB.
  ELSE.
    CLEAR BDC_TAB.
    MOVE: NAME  TO BDC_TAB-FNAM,
          VALUE TO BDC_TAB-FVAL.
    APPEND BDC_TAB.
  ENDIF.

ENDFORM. " DYNPRO
