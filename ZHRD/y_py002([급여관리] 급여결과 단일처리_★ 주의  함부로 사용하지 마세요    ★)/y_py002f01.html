 FORM GET_CLUSTERID.

         L_CHAR(1)  TYPE C.

   CLEAR: GT_RGDIR, GT_RGDIR[].
   CALL FUNCTION 'CU_READ_RGDIR'
     EXPORTING
       PERSNR          = $PERNR
     TABLES
       IN_RGDIR        = LT_RGDIR
     EXCEPTIONS
       NO_RECORD_FOUND = 1
       OTHERS          = 2.

   IF SY-SUBRC = 0.
     LOOP AT LT_RGDIR.
       MOVE-CORRESPONDING LT_RGDIR TO GT_RGDIR.

       IF GT_RGDIR-FPPER = '000000'.
         CLEAR GT_RGDIR-FPPER.
         L_CHAR = ' '.
       ELSE.
         L_CHAR = '.'.
       ENDIF.
       CONCATENATE GT_RGDIR-FPPER+4(2) L_CHAR GT_RGDIR-FPPER(4)
              INTO GT_RGDIR-NEW_FPPER.

       IF GT_RGDIR-INPER = '000000'.
         CLEAR GT_RGDIR-INPER.
         L_CHAR = ' '.
       ELSE.
         L_CHAR = '.'.
       ENDIF.
       CONCATENATE GT_RGDIR-INPER+4(2) L_CHAR GT_RGDIR-INPER(4)
              INTO GT_RGDIR-NEW_INPER.

       IF NOT GT_RGDIR-REVERSAL IS INITIAL.
         GT_RGDIR-VOID = GT_RGDIR-REVERSAL.
       ENDIF.

       APPEND GT_RGDIR.
     ENDLOOP.
   ENDIF.

 FORM READ_PAYROLL_RESULT USING $PERNR
                                $SEQ.

   DATA: LV_PERNR LIKE PA0001-PERNR,
         LV_SEQNR LIKE PC261-SEQNR.

   LV_PERNR = $PERNR.
   LV_SEQNR = $SEQ.

   CALL FUNCTION 'PYXX_READ_PAYROLL_RESULT'
     EXPORTING
       CLUSTERID          = T500L-RELID
       EMPLOYEENUMBER     = LV_PERNR
       SEQUENCENUMBER     = LV_SEQNR
       FILTER_CUMULATIONS = SPACE
     CHANGING
       PAYROLL_RESULT     = <PAYROLL_RESULT>
     EXCEPTIONS
       OTHERS             = 8.

   IF SY-SUBRC <> 0.
