*&      FORM GET_T591S_TABLE
*&---------------------------------------------------------------------*
*       인포타입 하부유형 LIST 조회
*----------------------------------------------------------------------*
FORM GET_T591S_TABLE TABLES P_TABLE STRUCTURE T591S
                      USING    P_INFTY.

  DATA: T582L   LIKE T582L OCCURS 0 WITH HEADER LINE.
  DATA: L_TABIX LIKE SY-TABIX.

  CLEAR: P_TABLE, P_TABLE[], T582L, T582L[].

* 인포타입 - 국가고유세팅 조회
  SELECT *
    INTO TABLE T582L
    FROM T582L
   WHERE INFTY = P_INFTY
     AND MOLGA = G_MOLGA
     AND STATU = 'E'.

* 인포타입 - 하부유형 조회
  SELECT *
    INTO TABLE P_TABLE
    FROM T591S
   WHERE SPRSL  = SY-LANGU
     AND INFTY  = P_INFTY.

  LOOP AT P_TABLE.
    L_TABIX = SY-TABIX.
    READ TABLE T582L WITH KEY SUBTY = P_TABLE-SUBTY.
    IF SY-SUBRC = 0.
      DELETE P_TABLE INDEX L_TABIX.
    ENDIF.
  ENDLOOP.

ENDFORM. " GET_T591S_TABLE

*&---------------------------------------------------------------------*
*       SCREEN FIELD MODIFY (PNP:HRMX0005 사용시)
*----------------------------------------------------------------------*
FORM MODIFY_SCREEN_PNP .

  LOOP AT SCREEN.
    CASE SCREEN-NAME.
      WHEN 'PNPTIMR1'      OR 'PNPTIMR6'      OR 'PNPTBEG'
        OR 'PNPTIMT6'      OR 'PNPBEGDA'      OR 'PNPENDDA'
        OR 'PNPBEGPS'      OR 'PNPENDPS'      OR '%FZ21091_1000'
        OR '%FBIS111_1000' OR '%FDPS116_1000' OR '%FBIS119_1000'.
        SCREEN-INPUT     = '0'.
        SCREEN-INVISIBLE = '1'.

        MODIFY SCREEN.
    ENDCASE.
  ENDLOOP.

ENDFORM. " MODIFY_SCREEN_PNP
