FORM GET_BATCH_JOB_STATUS USING FP_PROGID FP_STATUS.

*  DATA: l_status TYPE  xfeld.
*
*  DO.
*    CALL FUNCTION 'FKK_AKTIV2_RUN_STATUS_READ'
*      EXPORTING
*        i_aktyp  = 'FPG1'       "액티비티 유형
*        i_laufd  = l_laufd      "Date ID
*        i_laufi  = l_laufi      "identification
*      IMPORTING
*        e_status = l_status.
*
*    CHECK l_status EQ 'X'.
*    EXIT.
*  ENDDO.

*[상태]
*'R'  Active, is being executed.
*'Y'  Ready, the job is released and meets the start requirements.
*     However the job is waiting for a free work process.
*'P'  Scheduled but not yet released.
*'S'  The job is released and is waiting to meet the start requirements.
*     It can then be executed.
*'A'  Terminated.
*'F'  Complete, all job steps were successfully executed.

  DATA: L_JOBNAME   LIKE TBTCO-JOBNAME,
        L_JOBCOUNT  LIKE TBTCO-JOBCOUNT,
        L_STATUS    LIKE TBTCO-STATUS.

  CLEAR: FP_STATUS.

  CONCATENATE FP_PROGID '%' INTO L_JOBNAME.

  DO.
    SELECT SINGLE JOBNAME JOBCOUNT STATUS
      INTO (L_JOBNAME, L_JOBCOUNT, L_STATUS)
      FROM TBTCO
     WHERE JOBNAME LIKE L_JOBNAME.

    CASE L_STATUS.
      WHEN 'A' OR 'F'.
        FP_STATUS = L_STATUS.
        EXIT.
      WHEN OTHERS.
    ENDCASE.
  ENDDO.

ENDFORM. " GET_BATCH_JOB_STATUS
