FORM HELP_FRM .
  CLEAR  :FIELDS,VALUETAB,SEL_VA,DYNPFIELDS,GT_DD03L,DYNPREAD      .
  REFRESH:FIELDS,VALUETAB,SEL_VA,DYNPFIELDS,GT_DD03L,DYNPREAD      .

* ### ## TABLE ## READ
  PERFORM VALUE_READ USING: 'S1_TAB'                              .
  LOOP AT DYNPREAD                                                .
    CASE SY-TABIX                                                  .
      WHEN  1     . S1_TAB = DYNPREAD-FIELDVALUE                    .
    ENDCASE                                                        .
  ENDLOOP                                                         .

* ##### ## FIELD# ### READ(ACTIV ###)
  SELECT *
  INTO CORRESPONDING FIELDS OF TABLE GT_DD03L
  FROM   DD03L
  WHERE  AS4LOCAL  EQ 'A'
  AND    TABNAME   EQ S1_TAB
  AND    FIELDNAME NE '.INCLUDE'                                 .
  SORT GT_DD03L BY POSITION                                      .

* #### MOVE(GT_DD03L -> GT_FORM)
  LOOP AT GT_DD03L                                               .
    SELECT SINGLE DDTEXT INTO VALUETAB-VALUE
    FROM   DD04T
    WHERE  ROLLNAME   EQ GT_DD03L-ROLLNAME
    AND    DDLANGUAGE EQ SY-LANGU                                .
    IF SY-SUBRC NE 0                                             .
      VALUETAB-VALUE = GT_DD03L-FIELDNAME                        .
    ENDIF                                                        .
*
    PERFORM FILL_DATA_ELEMENTS_RTN USING:
    GT_DD03L-FIELDNAME VALUETAB-VALUE
    GT_DD03L-POSITION  GT_DD03L-INTLEN                           .
  ENDLOOP                                                        .

* #### DISPLAY
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      RETFIELD    = 'FLDNM'
      DYNPPROG    = SY-REPID
      DYNPNR      = SY-DYNNR
*     DYNPROFIELD = 'S1_TYP'
      VALUE_ORG   = 'S'
    TABLES
      VALUE_TAB   = GT_FORM.

ENDFORM. " HELP_FRM

FORM SELECTION_CHECK .
  DATA:L_FIL(100) TYPE C, RFLAG                                    .
  SELECT SINGLE * FROM TADIR
  WHERE  PGMID    EQ 'R3TR'
    AND  OBJECT   EQ 'TABL'
    AND  OBJ_NAME EQ S1_TAB                                      .
  IF SY-SUBRC NE 0                                               .
    MESSAGE I001(00) WITH ' Not exist -' S1_TAB ' Table!!'. STOP.
  ENDIF.

  L_FIL  = S1_FIL                                                .
  CASE S1_OPT                                                    .
    WHEN 'I' OR 'M'                                               .
      CALL FUNCTION 'WS_QUERY'
        EXPORTING
          FILENAME = L_FIL
          QUERY    = 'FE'
        IMPORTING
          RETURN   = RFLAG.
      CHECK RFLAG NE '1'                                           .
      MESSAGE E001 WITH ' Not Exist File!!' S1_FIL . STOP          .
    WHEN 'Y' OR 'Z'                                               .
    WHEN 'X'                                                      .
      CHECK S1_TAB(1) NE 'Y'  AND S1_TAB(1) NE 'Z'                .
      CHECK SY-UNAME NE 'HEUS'                                    .
      MESSAGE I001 WITH S1_TAB ' Not CBO Table!!'. STOP           .
    WHEN 'D'                                                      .
  ENDCASE                                                         .
ENDFORM. " SELECTION_CHECK
