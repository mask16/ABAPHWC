*&      FORM  BUILD_HTML
*&---------------------------------------------------------------------*
*       HTML 생성
*----------------------------------------------------------------------*
FORM BUILD_HTML CHANGING P_LENGTH.

  DATA: L_LINE_LEN     TYPE I,
        L_HEADLINE_LEN TYPE I,
        L_TEXT         TYPE SDYDO_TEXT_ELEMENT,
        L_TAB_COUNT    TYPE I.

  CALL METHOD G_DOC->INITIALIZE_DOCUMENT.

  P_LENGTH = 21.

  L_LINE_LEN     = G_DOC->ACT_GUI_PROPERTIES-FONTSIZE + 9.
  L_HEADLINE_LEN = L_LINE_LEN + 10.

  IF L_LINE_LEN GT 16.
    ADD 1 TO L_LINE_LEN.
  ENDIF.

* Fill TOP-Document
  LOOP AT GT_LIST_HEADER INTO GS_LIST_HEADER
                        WHERE TYP = 'H'.
    IF SY-TABIX > 1.
      CALL METHOD G_DOC->NEW_LINE.
    ENDIF.
    L_TEXT = GS_LIST_HEADER-INFO.
    CALL METHOD G_DOC->ADD_TEXT
      EXPORTING
        TEXT      = L_TEXT
        SAP_STYLE = 'HEADING'.
    P_LENGTH = P_LENGTH + L_HEADLINE_LEN.
  ENDLOOP.

  IF SY-SUBRC = 0.
    CALL METHOD G_DOC->NEW_LINE.
    CALL METHOD G_DOC->NEW_LINE.
    P_LENGTH = P_LENGTH + L_LINE_LEN.
  ENDIF.

* Selections
  LOOP AT GT_LIST_HEADER INTO GS_LIST_HEADER
                        WHERE TYP = 'S'.
    IF L_TAB_COUNT IS INITIAL.
      CALL METHOD G_DOC->ADD_TABLE
        EXPORTING
          NO_OF_COLUMNS = 2
          WITH_HEADING  = ''
          WIDTH         = '100%'
          BORDER        = '0'
        IMPORTING
