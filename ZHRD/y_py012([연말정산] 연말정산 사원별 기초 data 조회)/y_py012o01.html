MODULE PICTURE_DISPLAY OUTPUT.

  DATA: L_PER(20).
  DATA: LT_TOAHR LIKE TOAHR OCCURS 0 WITH HEADER LINE.
  DATA: L_DOC_ID TYPE SAEARDOID.
  DATA: H_PIC_CONTAINER TYPE REF TO CL_GUI_CUSTOM_CONTAINER,
        H_PICTURE TYPE REF TO CL_GUI_PICTURE.
  DATA: URL(255).


  CHECK GT_BAS-PERNR IS NOT INITIAL.

*# 사진유무 Check
  CONCATENATE GT_BAS-PERNR '%' INTO L_PER.

  CLEAR: LT_TOAHR, LT_TOAHR[].
  SELECT *
    FROM TOAHR
    INTO CORRESPONDING FIELDS OF TABLE LT_TOAHR
   WHERE SAP_OBJECT = C_OBJ
     AND ARCHIV_ID  = C_ARCID
     AND AR_OBJECT  = C_AROBJ
     AND OBJECT_ID LIKE L_PER.

  IF SY-SUBRC <> 0.
    L_PER = '98761234%'.

    SELECT *
      FROM TOAHR
      INTO CORRESPONDING FIELDS OF TABLE LT_TOAHR
     WHERE SAP_OBJECT = C_OBJ
       AND ARCHIV_ID  = C_ARCID
       AND AR_OBJECT  = C_AROBJ
       AND OBJECT_ID LIKE L_PER.

    SORT LT_TOAHR DESCENDING BY AR_DATE.
    READ TABLE LT_TOAHR INDEX 1.

    L_DOC_ID = LT_TOAHR-ARC_DOC_ID.

  ELSE.
    SORT LT_TOAHR DESCENDING BY AR_DATE.
    READ TABLE LT_TOAHR INDEX 1.

    L_DOC_ID = LT_TOAHR-ARC_DOC_ID.
  ENDIF.

*# Contents Sever 미사용시
  CLEAR: URL.
  CALL FUNCTION 'SCMS_DOC_URL_READ'
    EXPORTING
      STOR_CAT    = SPACE
      CREP_ID     = C_ARCID
      DOC_ID      = L_DOC_ID
      COMP_ID     = 'data'
      DP_URL_ONLY = 'X'
*     LIFETIME    = CNDP_LIFETIME_TRANSACTION
    IMPORTING
      URL         = URL
    EXCEPTIONS
      OTHERS      = 10.

*# Container Free
  IF NOT H_PIC_CONTAINER IS INITIAL.
    CALL METHOD H_PIC_CONTAINER->FREE.
  ENDIF.

  CREATE OBJECT H_PIC_CONTAINER
    EXPORTING
      CONTAINER_NAME = 'JPG'.

  CREATE OBJECT H_PICTURE
    EXPORTING
      PARENT = H_PIC_CONTAINER.

*# Container Size에 맞춰 Display
** Set the display mode to 'normal' (0)
*  CALL METHOD H_PICTURE->SET_DISPLAY_MODE
*    EXPORTING
*      DISPLAY_MODE = CL_GUI_PICTURE=>DISPLAY_MODE_FIT.
**   EXPORTING
**     DISPLAY_MODE = CL_GUI_PICTURE=>DISPLAY_MODE_NORMAL.
  CALL METHOD H_PICTURE->SET_DISPLAY_MODE
    EXPORTING
      DISPLAY_MODE = CL_GUI_PICTURE=>DISPLAY_MODE_STRETCH.

  CALL METHOD H_PICTURE->SET_3D_BORDER
    EXPORTING
      BORDER = 1.  "Display border in 3D format (1=yes, 0=no)

  CALL METHOD H_PICTURE->LOAD_PICTURE_FROM_URL_ASYNC
    EXPORTING
      URL = URL.

ENDMODULE. " PICTURE_DISPLAY OUTPUT
