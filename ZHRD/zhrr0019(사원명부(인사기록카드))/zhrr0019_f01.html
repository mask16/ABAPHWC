FORM get_data .

  DATA : lv_ok VALUE 'X'.

  CLEAR gt_data.
  PERFORM get_basic USING pernr-pernr
                          pn-begda.

  IF so_entda[] IS NOT INITIAL.
    IF gt_data-entrda BETWEEN so_entda-low AND so_entda-high.
    ELSE.
      CLEAR lv_ok.
    ENDIF.
  ENDIF.

  IF so_retda[] IS NOT INITIAL.
    IF gt_data-leavda BETWEEN so_retda-low AND so_retda-high.
    ELSE.
      CLEAR lv_ok.
    ENDIF.
  ENDIF.

  IF s_titel[] IS NOT INITIAL.
    IF gt_data-titel = s_titel-low.
    ELSE.
      CLEAR lv_ok.
    ENDIF.
  ENDIF.
  CHECK lv_ok EQ 'X'.

  APPEND <fs_wa> TO <fs_table>.
  CLEAR <fs_wa>.

ENDFORM. " GET_DATA

FORM get_basic USING p_pernr

  MOVE-CORRESPONDING gs_basic TO gt_data.

  LOOP AT p0022 WHERE begda <= p_keydate.

    "교육기관

    ENDIF.

  IF pnp-sw-found = 1.

    PERFORM get_domain_value USING    p0002-gesch
                             CHANGING gt_data-gtext.
  ENDIF.

  MOVE-CORRESPONDING gt_data TO <fs_wa>.

  CLEAR : t_orgeh, t_orgeh[].
* 조직

    EXPORTING

      i_prog    = sy-repid

      lt_orgeh  = t_orgeh.

      CONDENSE l_fieldname NO-GAPS.
      PERFORM assign_field USING :  l_fieldname  t_orgeh-orgtx.
    ENDLOOP.
  ELSE.
    DATA : l_orgtx LIKE t527x-orgtx.
