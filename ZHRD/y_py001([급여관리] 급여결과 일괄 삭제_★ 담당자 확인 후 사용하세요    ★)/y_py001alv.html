FORM MODIFY_FIELDCAT .

  DATA: L_TABIX   LIKE SY-TABIX.

* ICON
  GS_FIELDCAT-ICON       = GC_X.
  MODIFY GT_FIELDCAT  FROM GS_FIELDCAT TRANSPORTING HOTSPOT ICON
                     WHERE FIELDNAME = 'ZCHEK'
                        OR FIELDNAME = 'RESUT'.

* NO_OUT
  GS_FIELDCAT-NO_OUT     = GC_X.
  MODIFY GT_FIELDCAT  FROM GS_FIELDCAT TRANSPORTING NO_OUT
                     WHERE FIELDNAME = 'HANDLE_C'
                        OR FIELDNAME = 'HANDLE_R'.

* KEY
  GS_FIELDCAT-KEY        = GC_X.
  MODIFY GT_FIELDCAT  FROM GS_FIELDCAT TRANSPORTING KEY
                     WHERE FIELDNAME = 'ZCHEK'
                        OR FIELDNAME = 'RESUT'
                        OR FIELDNAME = 'ERRTX'
                        OR FIELDNAME = 'PERSDATA'
                        OR FIELDNAME = 'ENAME'.

* Text check
  LOOP AT GT_FIELDCAT INTO GS_FIELDCAT.
    L_TABIX = SY-TABIX.
    GS_FIELDCAT-JUST = 'C'.
    CASE GS_FIELDCAT-FIELDNAME.
      WHEN 'ZCHEK'.
        PERFORM FIELDCAT_TEXT  USING    'Check'.
      WHEN 'RESUT'.
        PERFORM FIELDCAT_TEXT  USING    'Result'.
      WHEN 'ERRTX'.
        PERFORM FIELDCAT_TEXT  USING    '에러TEXT'.
      WHEN 'PERSDATA'.
        PERFORM FIELDCAT_TEXT  USING    '사원번호'.
      WHEN 'ENAME'.
        PERFORM FIELDCAT_TEXT  USING    '성명'.
    ENDCASE.
*   MODIFY GT_FIELDCAT
    MODIFY GT_FIELDCAT FROM GS_FIELDCAT INDEX L_TABIX.
  ENDLOOP.

ENDFORM. " modify_fieldcat

FORM SET_FIELDNAME USING P_FNAME
                    CHANGING P_INDEX.

  DATA: LC_FIELD(20).

  FIELD-SYMBOLS: <FS> TYPE ANY.

  ADD 1 TO P_INDEX.

  CLEAR GS_FIELDCAT.

  READ TABLE GT_FIELDCAT INTO GS_FIELDCAT WITH KEY COL_POS = P_INDEX.

  CHECK SY-SUBRC = 0.

  CONCATENATE 'GT_ERROR-' GS_FIELDCAT-FIELDNAME INTO LC_FIELD.
  ASSIGN (LC_FIELD) TO <FS>.
  MOVE P_FNAME TO <FS>.

ENDFORM. " set_fieldname
