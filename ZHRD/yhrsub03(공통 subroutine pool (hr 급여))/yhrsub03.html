FORM GET_RE511K USING P_KONST
                         P_DATUM
                CHANGING P_T511K.

  CLEAR P_T511K.
  SELECT SINGLE *
    INTO P_T511K
    FROM T511K
   WHERE MOLGA = G_MOLGA
     AND KONST = P_KONST
     AND BEGDA <= P_DATUM
     AND ENDDA >= P_DATUM.

ENDFORM. " GET_RE511K
