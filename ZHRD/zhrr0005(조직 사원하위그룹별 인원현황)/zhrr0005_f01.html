FORM get_code_text .

  DATA : lv_molga TYPE molga.
  DATA : lt_parameter TYPE TABLE OF bapiparam WITH HEADER LINE.
  DATA : lt_bapiret2 TYPE TABLE OF bapiret2 WITH HEADER LINE.

  CALL FUNCTION 'BAPI_USER_GET_DETAIL'
    EXPORTING
      username  = sy-uname
    TABLES
      parameter = lt_parameter
      return    = lt_bapiret2.

  LOOP AT lt_parameter WHERE parid = 'UGR'.
  ENDLOOP.
  IF sy-subrc EQ 0.
    lv_molga = lt_parameter-parva.
  ENDIF.

  SELECT DISTINCT persk
  INTO CORRESPONDING FIELDS OF TABLE gt_persk
  FROM t503z
  WHERE molga = lv_molga.

  LOOP AT gt_persk.
    SELECT SINGLE ptext AS pktxt
        INTO CORRESPONDING FIELDS OF gt_persk
        FROM t503t
        WHERE sprsl = sy-langu AND
              persk = gt_persk-persk.
    IF  gt_persk-persk NE '10' AND gt_persk-persk NE '20'.
      gt_persk-persk = '30'.
      gt_persk-pktxt = '현장직'.
    ENDIF.
    MODIFY gt_persk.
    CLEAR gt_persk.
  ENDLOOP.
  SORT gt_persk BY persk.
  DELETE ADJACENT DUPLICATES FROM gt_persk.

ENDFORM. " GET_CODE_TEXT
