            EXPORTING

            IMPORTING

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'

      AT END OF ROW.
        APPEND RECORD. CLEAR RECORD.
      ENDAT.
    ENDLOOP.
  ENDIF.

FORM BDC_RESULT USING P_SUBRC.

  IF P_SUBRC  = 0.
    READ TABLE MESSTAB WITH KEY MSGTYP = 'E'
                       TRANSPORTING NO FIELDS.
    IF SY-SUBRC <> 0.
      SUC_CNT = SUC_CNT + 1.
    ELSE.
      ERR_CNT = ERR_CNT + 1.
      LOOP AT MESSTAB.
        CALL FUNCTION 'MESSAGE_TEXT_BUILD'
          EXPORTING
            MSGID               = MESSTAB-MSGID
            MSGNR               = MESSTAB-MSGNR
            MSGV1               = MESSTAB-MSGV1
            MSGV2               = MESSTAB-MSGV2
            MSGV3               = MESSTAB-MSGV3
            MSGV4               = MESSTAB-MSGV4
          IMPORTING
            MESSAGE_TEXT_OUTPUT = GT_ERR-MSGTEXT
          EXCEPTIONS
            OTHERS              = 1.
        APPEND GT_ERR. CLEAR GT_ERR.
      ENDLOOP.
    ENDIF.
  ELSE.
    ERR_CNT = ERR_CNT + 1.
    LOOP AT MESSTAB.
      CALL FUNCTION 'MESSAGE_TEXT_BUILD'
        EXPORTING
          MSGID               = MESSTAB-MSGID
          MSGNR               = MESSTAB-MSGNR
          MSGV1               = MESSTAB-MSGV1
          MSGV2               = MESSTAB-MSGV2
          MSGV3               = MESSTAB-MSGV3
          MSGV4               = MESSTAB-MSGV4
        IMPORTING
          MESSAGE_TEXT_OUTPUT = GT_ERR-MSGTEXT
        EXCEPTIONS
          OTHERS              = 1.
      APPEND GT_ERR. CLEAR GT_ERR.
    ENDLOOP.
  ENDIF.

ENDFORM. " BDC_RESULT
