REPORT Y_SP011 NO STANDARD PAGE HEADING MESSAGE-ID 00 "398
                LINE-SIZE 120.


TABLES:DD02L.

DATA:G_TABNAME(30) TYPE C VALUE '/1BCDWB/DB',
     ACTION         TYPE SY-UCOMM   VALUE 'ANZE'      ,
     G_AUTH(4)      TYPE C                            ,
     G_DATA_EXIT    LIKE RS38L-NAME VALUE 'UPDA'      .

DATA: MEM_ID(16) VALUE 'TABELLENANZEIGER' .

PARAMETERS:P_TAB LIKE DATABROWSE-TABLENAME MEMORY ID DTB.

AT SELECTION-SCREEN ON VALUE-REQUEST FOR P_TAB .
  PERFORM F4_OBJNAME                                  .

AT SELECTION-SCREEN .
  IF P_TAB(1) <> 'Y' AND P_TAB(1) <> 'Z'              .
    MESSAGE E416(MO) WITH P_TAB                      .
  ENDIF                                               .

  SELECT * FROM  DD02L
  WHERE  TABNAME  EQ P_TAB
    AND  AS4LOCAL EQ 'A'
    AND  TABCLASS NE 'INTTAB'
    AND  TABCLASS NE 'VIEW'
    AND  TABNAME  NE 'APPEND'.
    EXIT                     .
  ENDSELECT                  .
  IF SY-SUBRC NE 0. MESSAGE E402(MO) WITH P_TAB. ENDIF.

*&---------------------------------------------------------------------*
*&      S T A R T  P R O G R A M
*&---------------------------------------------------------------------*
START-OF-SELECTION .

  CASE SY-UNAME.
    WHEN 'AB08' OR 'HEDADM' OR 'HIT01' OR 'HBC01' OR
         'HIT99' OR 'HHR05'.
      CONCATENATE G_TABNAME P_TAB INTO G_TABNAME          .
      DD02L-MAINFLAG = 'X'                                .
      EXPORT ACTION DD02L-MAINFLAG G_AUTH
                    TO MEMORY ID MEM_ID                   .
      SUBMIT (G_TABNAME) VIA SELECTION-SCREEN AND RETURN  .

    WHEN OTHERS.
      MESSAGE I000(OO) WITH '※ 권한이 없습니다!!!'.
      EXIT.
  ENDCASE.


*&---------------------------------------------------------------------*
*&      Form  F4_OBJNAME
*&---------------------------------------------------------------------*
FORM F4_OBJNAME .
  DATA:L_DNUM   LIKE D020S-DNUM                         ,
       L_REPID  LIKE SY-REPID                           .

  L_DNUM = SY-DYNNR.
  L_REPID = SY-REPID.
  CALL FUNCTION 'RS_HELP_HANDLING'
    EXPORTING
      DYNPFIELD                 = 'P_TAB'
      DYNPNAME                  = L_DNUM
      OBJECT                    = 'TB  '
      PROGNAME                  = L_REPID
      SUPPRESS_SELECTION_SCREEN = 'X'.

ENDFORM. " F4_OBJNAME
