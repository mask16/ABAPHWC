FORM user_command USING r_ucomm LIKE sy-ucomm
                                                rs_selfield TYPE  slis_selfield.

  CASE r_ucomm.
    WHEN '&IC1' .

      PERFORM double_p USING  rs_selfield.

    WHEN '&BACK' OR '&EXIT' OR '&CANC'.
      SET SCREEN 0.
  ENDCASE.
ENDFORM. " USER_COMMAND

*       text

*----------------------------------------------------------------------*
FORM double_p USING rs_selfield TYPE slis_selfield.

  FIELD-SYMBOLS : <fs>,
                               <fs2>.

  CLEAR : lt_pernr, lt_pernr[].

  DATA  gs_tab LIKE gt_tab.
  READ TABLE gt_tab  INTO gs_tab  INDEX rs_selfield-tabindex.
  CHECK sy-subrc = 0.
  LOOP AT gt_display WHERE orgeh = gs_tab-orgeh
                                         AND atx = gs_tab-atx.
    CASE   rs_selfield-fieldname .
      WHEN 'A_JO'.
        IF gt_display-a_jo IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN 'B_JO'.
        IF gt_display-b_jo IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN 'C_JO'.
        IF gt_display-c_jo IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN   'NOR'.
        IF gt_display-nor  IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN 'ABSENCE' .  "결근
        IF  gt_display-absence IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN 'PRESENT'.  "현재원.
        IF  gt_display-present  IS NOT INITIAL.
          PERFORM get_pernr USING gt_display-pernr.
        ENDIF.
      WHEN OTHERS.
        PERFORM get_pernr USING gt_display-pernr.
    ENDCASE.
  ENDLOOP.

  IF NOT lt_pernr[] IS INITIAL.
    SUBMIT  zhrdetail     WITH pnppernr-low  IN lt_pernr
                                           AND RETURN.
  ENDIF.
ENDFORM. " DOUBLE_P
