FORM GET_MESSAGE CHANGING P_L_MSG LIKE SM04DIC-POPUPMSG
                          P_RETURNCODE TYPE C.

  DATA: BEGIN OF FIELDS OCCURS 1.
          INCLUDE STRUCTURE SVAL.
  DATA: END OF FIELDS,
        RETURNCODE TYPE C,
        L_FIRST LIKE V_ADRP_CP-NAME_FIRST,
        L_LAST  LIKE V_ADRP_CP-NAME_LAST.


  SELECT SINGLE V~NAME_FIRST V~NAME_LAST
    INTO (L_FIRST, L_LAST )
    FROM USR21 AS U
    JOIN V_ADRP_CP AS V
      ON U~PERSNUMBER = V~PERSNUMBER AND
         U~ADDRNUMBER = V~ADDRNUMBER
   WHERE U~BNAME = SY-UNAME.

  FIELDS-TABNAME = 'SM04DIC'.
  FIELDS-FIELDNAME = 'POPUPMSG'.
  FIELDS-FIELDTEXT = 'Message :'.
  CONCATENATE '  by ' L_LAST L_FIRST
         INTO FIELDS-VALUE SEPARATED BY ' '.
  APPEND FIELDS.

  CALL FUNCTION 'POPUP_GET_VALUES'
    EXPORTING
      POPUP_TITLE = 'Supply the popup message'
    IMPORTING
      RETURNCODE  = P_RETURNCODE
    TABLES
      FIELDS      = FIELDS.

  IF P_RETURNCODE = 'A'.
    EXIT.
  ELSE.
    READ TABLE FIELDS INDEX 1.
    P_L_MSG = FIELDS-VALUE.
  ENDIF.

ENDFORM. " GET_MESSAGE
