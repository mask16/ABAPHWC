FORM user_ok_tc USING p_tc_name TYPE dynfnam
                         p_table_name
                         p_mark_name
                CHANGING p_ok      LIKE sy-ucomm.


*&SPWIZARD: BEGIN OF LOCAL DATA----------------------------------------*
  DATA: l_ok              TYPE sy-ucomm,
        l_offset          TYPE i.
*&SPWIZARD: END OF LOCAL DATA------------------------------------------*

*&SPWIZARD: Table control specific operations                          *
*&SPWIZARD: evaluate TC name and operations                            *
  SEARCH p_ok FOR p_tc_name.
  IF sy-subrc <> 0.
    EXIT.
  ENDIF.
  l_offset = strlen( p_tc_name ) + 1.
  l_ok = p_ok+l_offset.
*&SPWIZARD: execute general and TC specific operations                 *
  CASE l_ok.
    WHEN 'INSR'.                      "insert row
      PERFORM fcode_insert_row USING    p_tc_name
                                        p_table_name.
      CLEAR p_ok.

    WHEN 'DELE'.                      "delete row
      PERFORM fcode_delete_row USING    p_tc_name
                                        p_table_name
                                        p_mark_name.
      CLEAR p_ok.

    WHEN 'P--' OR                     "top of list
         'P-'  OR                     "previous page
         'P+'  OR                     "next page
         'P++'.                       "bottom of list
      PERFORM compute_scrolling_in_tc USING p_tc_name
                                            l_ok.
      CLEAR p_ok.
*
    WHEN 'MARK'.                      "mark all filled lines
      PERFORM fcode_tc_mark_lines USING p_tc_name
                                        p_table_name
                                        p_mark_name   .
      CLEAR p_ok.

    WHEN 'DMRK'.                      "demark all filled lines
      PERFORM fcode_tc_demark_lines USING p_tc_name
                                          p_table_name
                                          p_mark_name .
      CLEAR p_ok.
  ENDCASE.

ENDFORM. " USER_OK_TC

FORM fn_chk_only_single_mark USING lt_itab LIKE gt_zhrt9007.

  DATA: l_cnt   TYPE n,
        ls_itab LIKE LINE OF lt_itab.

  LOOP AT lt_itab INTO ls_itab WHERE mark EQ co_true.

    WHEN 'FINS' OR 'FEDI' OR 'FDIS' OR 'FCON'.

      ELSE.

        AND zhrt9007-isstate > 2.
          MESSAGE s016 DISPLAY LIKE 'E'
          WITH '※ 승인또는 발급된 증명서는 변경할 수 없습니다!'.
          LEAVE TO SCREEN sy-dynnr.
        ENDIF.
      ENDIF.

    WHEN 'FCON' OR 'FPRN'.
      IF l_cnt IS INITIAL.
        MESSAGE s016 DISPLAY LIKE 'E'
        WITH '※ 적절한 데이타를 선택하세요!'.
        LEAVE TO SCREEN sy-dynnr.

FORM fn_get_gt_zhrt9007 USING lt_itab LIKE gt_zhrt9007.

  DATA: l_isstate     LIKE zhrt9007-isstate,
        l_certik      LIKE zhrt9007-certik,
        l_orddat_low  LIKE zhrt9007-orddat VALUE co_frdat,
        l_orddat_high LIKE zhrt9007-orddat VALUE co_ladat,
        l_pernr(8) TYPE c.  "LIKE 문을 위해서 CHAR 8로 선언.

*-- 발급상태.
  CASE p_isstate.
    WHEN space OR '1'. "공백이거나, 전체 선택.
      MOVE '%' TO l_isstate.
    WHEN OTHERS.       "신청 또는 승인 또는 발급.
      MOVE p_isstate TO l_isstate.
  ENDCASE.

*-- 신청일자.
*== [ 날짜 ] =================================*
* SPACE ~ SPACE = 0000.00.00 ~ 9999.12.31     *
* SPACE ~ ENDDA = 0000.00.00 ~ ENDDA          *
* BEGDA ~ SPACE = BEGDA ~ 9999.12.31          *
* BEGDA ~ ENDDA = BEGDA ~ ENDDA               *
*=============================================*
  IF     p_begda IS     INITIAL AND p_endda IS     INITIAL.
* 기본 DEFAULT 값 사용.    "◀

  ELSEIF p_begda IS     INITIAL AND p_endda IS NOT INITIAL.
* ENDDA값을 BEGDA로 넘김.
    MOVE p_endda TO : l_orddat_high.

  ELSEIF p_begda IS NOT INITIAL AND p_endda IS     INITIAL.
* BEGDA값을 ENDDA로 넘김.
    MOVE p_begda TO : l_orddat_low.

  ELSEIF p_begda IS NOT INITIAL AND p_endda IS NOT INITIAL.
* BEGDA값을 BEGDA로, ENDDA값을 ENDDA로 넘김.
    MOVE p_begda TO : l_orddat_low.
    MOVE p_endda TO : l_orddat_high.
  ENDIF.

*-- 증명서 종류.
  MOVE p_certik TO l_certik.

*-- 사원번호.
  CASE p_pernr.
    WHEN space.  "공백이면 전체.
      MOVE '%' TO l_pernr.
    WHEN OTHERS.
      MOVE p_pernr TO l_pernr.
  ENDCASE.

*-- 발급년도-발급번호.
  RANGES: zynkey FOR zhrt9007-zynkey.

  IF p_byear IS NOT INITIAL AND p_bnum IS NOT INITIAL.
    MOVE: 'I'     TO zynkey-sign,
          'EQ'    TO zynkey-option,
          p_byear TO zynkey-low(4),
          p_bnum  TO zynkey-low+4(4).
    IF p_eyear IS NOT INITIAL AND p_enum IS NOT INITIAL.
      MOVE: 'BT'    TO zynkey-option,
            p_eyear TO zynkey-high(4),
            p_enum  TO zynkey-high+4(4).
    ENDIF.

    APPEND zynkey.
  ELSE.
    IF p_eyear IS NOT INITIAL AND p_enum IS NOT INITIAL.
      MESSAGE s016 DISPLAY LIKE 'E'
      WITH '※ 발급년도-발급번호 시작조건을 입력하세요!'.
    ENDIF.
  ENDIF.


*>>> [ 검색조건 ] <<<.
  IF l_certik IS INITIAL.
    SELECT * FROM zhrt9007
             INTO TABLE gt_zhrt9007
             WHERE isstate LIKE l_isstate
               AND orddat BETWEEN l_orddat_low AND l_orddat_high
               AND pernr  LIKE l_pernr
               AND zynkey IN zynkey
             ORDER BY orddat pernr.

  ELSE.
    SELECT * FROM zhrt9007
             INTO TABLE gt_zhrt9007
             WHERE isstate LIKE l_isstate
               AND certik EQ l_certik
               AND orddat BETWEEN l_orddat_low AND l_orddat_high
               AND pernr  LIKE l_pernr
               AND zynkey IN zynkey
             ORDER BY orddat pernr.
  ENDIF.

*-- 신청번호순으로 SORT(DESCENDING).
  SORT gt_zhrt9007 BY scnum DESCENDING.

ENDFORM. " FN_GET_GT_ZHRt9007

*       text

FORM fn_get_new_issnum USING ls_itab LIKE zhrt9007.

* [ 발급번호 채번 RULE ].
* YYYY-NNNN(YYYY:해당연도,NNNN:일괄번호).
* 년도가 바뀌면 일괄번호는 0001 부터 다시 시작한다.

  DATA: l_begyr TYPE sydatum VALUE '00000101',
        l_endyr TYPE sydatum VALUE '99991231'.

  MOVE sy-datum(4) TO : l_begyr(4), l_endyr(4).

  SELECT SINGLE MAX( znum ) FROM zhrt9007
                            INTO ls_itab-znum
                           WHERE orddat BETWEEN l_begyr AND l_endyr.

  MOVE : sy-datum(4) TO ls_itab-zyear.

  ADD 1 TO ls_itab-znum.

  IF ls_itab-znum EQ '0000'.
    ls_itab-znum = '0001'.
  ENDIF.

*-- 발급년도-발급번호 검색조건을 위해서.
  CONCATENATE ls_itab-zyear ls_itab-znum INTO ls_itab-zynkey.

ENDFORM. " FN_GET_NEW_ISSNUM

FORM delete_process USING lt_itab LIKE gt_zhrt9007.
  DATA : lt_032 LIKE zhrt9007 OCCURS 0 WITH HEADER LINE.
  DATA : l_flag.
  DATA : ls_itab LIKE LINE OF lt_itab.

  LOOP AT gt_zhrt9007 INTO ls_itab WHERE mark = 'X'.
    IF ls_itab-isstate <> '2'.
      l_flag = 'X'.
    ELSE.
      MOVE-CORRESPONDING ls_itab TO lt_032.
      APPEND lt_032. CLEAR lt_032.
    ENDIF.
  ENDLOOP.

  IF NOT lt_032[] IS INITIAL.
    DELETE zhrt9007 FROM TABLE lt_032.
    IF sy-subrc = 0.
      IF NOT l_flag IS INITIAL.
        MESSAGE i016 WITH '발급상태가 신청상태인 데이터만 삭제합니다'.
      ELSE.
        MESSAGE i016 WITH '삭제되었습니다'.
      ENDIF.
      COMMIT WORK.
    ELSE.
      MESSAGE i016 WITH '삭제실패 - 관리자 문의'.
      ROLLBACK WORK.
    ENDIF.
  ELSE.
    MESSAGE i016 WITH '발급상태가 신청상태인 데이터만 삭제가능합니다'.
  ENDIF.

ENDFORM. " DELETE_PROCESS
