FORM get_data .
  DATA : lv_molga   TYPE molga,
             pws_result TYPE TABLE OF pdpsp WITH HEADER LINE.
  DATA : tmp_date TYPE datum,
             num TYPE i.


  tmp_date = pn-begda.

  num = pn-endda - pn-begda + 1.

  DO num TIMES.
    gt_data-zpernr = pernr-pernr.
    gt_data-ename = pernr-ename.

    CALL FUNCTION 'HR_COUNTRYGROUPING_GET'
      EXPORTING
        pernr     = gt_data-zpernr
        begda     = tmp_date
        endda     = tmp_date
      IMPORTING
        molga     = lv_molga
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.

    rp-read-infotype gt_data-zpernr 0007 p0007 tmp_date tmp_date.
    CHECK  p0007-zterf = '9'.     "근태평가 하는 사원만

    gt_data-schkz = p0007-schkz.   "근무조

    rp-read-infotype gt_data-zpernr 0001 p0001 tmp_date tmp_date.
* 사원하위그룹
    SELECT SINGLE ptext INTO gt_data-ptext FROM t503t
                                                                 WHERE sprsl = sy-langu
                                                                     AND persk = p0001-persk.
* 공정
    SELECT SINGLE atx INTO gt_data-atx  FROM t542t
                                                          WHERE spras = sy-langu
                                                              AND molga =  lv_molga
                                                               AND ansvh = p0001-ansvh.
* 조직
    call function 'ZHRF0001'
      EXPORTING
        i_pernr   = gt_data-zpernr
        i_keydate = tmp_date
        i_prog    = sy-cprog
      IMPORTING
        e_orgtx   = gt_data-orgtx.

    "근무일정
    CLEAR : pws_result, pws_result[].
    CALL FUNCTION 'HR_READ_TIMEDATA_PSP'
      EXPORTING
        begin_date = tmp_date
        end_date   = tmp_date
        person_id  = gt_data-zpernr
      TABLES
        pers_avail = pws_result.

    READ TABLE pws_result WITH KEY pernr = gt_data-zpernr.
    gt_data-stime = pws_result-begzt.
    gt_data-etime = pws_result-endzt.

    IF ( pws_result-ftkla = c_ftkla_1 OR pws_result-ftkla = c_ftkla_2 ) AND pws_result-tprog CP '*OFF*'.
      gt_data-zpws = c_off.
    ELSE.
      gt_data-zpws = pws_result-tprog.  "일일근무일정
    ENDIF.

    READ TABLE gt_day_ws WITH KEY tprog = gt_data-zpws.
    IF sy-subrc = 0.
      gt_data-stime = gt_day_ws-sobeg.   "계획근무시작시간
      gt_data-etime = gt_day_ws-soend.    "계획근무 종료시간
    ENDIF.

    CASE gt_data-zpws.
      WHEN 'DAY0' OR 'NORM' OR 'NOR2'.
        gt_data-zpws = '주간'.
      WHEN 'NIGT'.
        gt_data-zpws = '야간'.
      WHEN 'OFF' OR 'OFFN'.
        gt_data-zpws = '비번'.
      WHEN 'HOL'.
        gt_data-zpws = '휴일'.
    ENDCASE.

    CLEAR it_itab.
    SELECT  SINGLE *
      INTO CORRESPONDING FIELDS OF  it_itab
*      FROM zhrt0001
      FROM zhrt1010
      WHERE zpernr = gt_data-zpernr
        AND zdats = tmp_date.
    IF sy-subrc = 0.
      MOVE-CORRESPONDING it_itab TO gt_data.

      IF gt_data-reflect = ''.  "전송여부 ('': 미전송 / X: 전송완료)
        gt_data-icon_tra = icon_incomplete.
      ELSEIF gt_data-reflect = 'X'.
        gt_data-icon_tra = icon_checked.
      ENDIF.
    ELSE.
      gt_data-zdats = tmp_date.
    ENDIF.
    APPEND gt_data.
    CLEAR gt_data.
    tmp_date = tmp_date + 1.
  ENDDO.

ENDFORM. " GET_DATA

FORM comment_build USING p_top_of_page TYPE slis_t_listheader.

  DATA : lv_htext(140),
         lv_htext2(100),
         lv_begda(100),
         lv_endda(100),
         lv_datum(100),
         lv_text LIKE t001p-btext.


  WRITE pn-begda TO lv_begda USING EDIT MASK '____.__.__'.
  WRITE pn-endda TO lv_endda USING EDIT MASK '____.__.__'.
  CONCATENATE lv_begda '~' lv_endda INTO lv_htext2 SEPARATED BY space.

  lv_htext = text-t01.

  PERFORM fill_top_of_page USING p_top_of_page
                                 'H' '' lv_htext.

  IF pnpbtrtl-low(2) = '11'.
    pnpbtrtl-low = '1100'.
  ELSEIF pnpbtrtl-low(2) ='12'.
    pnpbtrtl-low = '1200'.
  ENDIF.

  WRITE sy-datum TO lv_datum USING EDIT MASK '____-__-__'.
  CONCATENATE '작업일자 : ' lv_datum INTO lv_datum SEPARATED BY space.

  SELECT SINGLE btext INTO lv_text FROM t001p
                                                 WHERE werks = pnpwerks-low
                                                     AND btrtl = pnpbtrtl-low.

  PERFORM fill_top_of_page USING p_top_of_page
                               'S' '사업장 : ' lv_text.

  PERFORM fill_top_of_page USING p_top_of_page
                                 'S' '조회기간 : ' lv_htext2.

  PERFORM fill_top_of_page USING p_top_of_page
                                 'A' ' ' lv_datum.

ENDFORM. " COMMENT_BUILD
