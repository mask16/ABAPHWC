FORM open_excel .
* 엑셀 Sheet1 열기
  CREATE OBJECT g_application_ole 'excel.application'.
  SET PROPERTY OF g_application_ole 'visible' = 1.

  CALL METHOD OF
      g_application_ole
      'Workbooks'       = g_workbook.
  CALL METHOD OF
      g_workbook
      'Add'.

  CALL METHOD OF
      g_application_ole
      'Worksheets'      = g_sheet
    EXPORTING
      #1                = 1.
  CALL METHOD OF
      g_sheet
      'Activate'.
  SET PROPERTY OF g_sheet 'Name' = 'Sheet1'.


* 엑셀에 내용 찍기
* 1행은 입력할 필드 명세
* 2행은 입력할 값의 포맷
  SORT gt_excel_layout  BY cpos.

  LOOP AT gt_excel_layout.
    CALL METHOD OF
        g_sheet
        'Cells' = g_cells
      EXPORTING
        #1      = gt_excel_layout-cpos.

    SET PROPERTY OF g_cells 'Value' = gt_excel_layout-field.
  ENDLOOP.

  LOOP AT gt_excel_layout.
    CALL METHOD OF
        g_sheet
        'Cells' = g_cells
      EXPORTING
        #1      = 2
        #2      = gt_excel_layout-cpos.

    SET PROPERTY OF g_cells 'NumberFormatLocal' = '@'.
    SET PROPERTY OF g_cells 'Value' = gt_excel_layout-value.
  ENDLOOP.

ENDFORM. " OPEN_EXCEL

*&      Form  GET_FIELD_TEXT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*      -->P_LT_FIELDCAT_LVC  text
*      <--P_L_FTEXT  text
*----------------------------------------------------------------------*
FORM get_field_text USING ps_fieldcat_lvc STRUCTURE lvc_s_fcat
                     CHANGING p_text.

  DATA: l_fldnm   TYPE dd03l-tabname.
  FIELD-SYMBOLS <lfs>.

  CLEAR p_text.

  IF ps_fieldcat_lvc-ref_table IS INITIAL OR ps_fieldcat_lvc-ref_field  IS INITIAL.
    l_fldnm = 'COLTEXT'.

  ELSE.
    CASE ps_fieldcat_lvc-colddictxt.
      WHEN  ''  OR  'R'.
        l_fldnm = 'REPTEXT'.
      WHEN  'S' OR  'M' OR 'L'.
        CONCATENATE 'SCRTEXT_'  ps_fieldcat_lvc-colddictxt  INTO l_fldnm.
      WHEN OTHERS.
        l_fldnm = 'REPTEXT'.
    ENDCASE.
  ENDIF.

  ASSIGN COMPONENT l_fldnm  OF STRUCTURE ps_fieldcat_lvc  TO <lfs>.
  IF <lfs> IS INITIAL.
    l_fldnm = 'REPTEXT'.
    ASSIGN COMPONENT l_fldnm  OF STRUCTURE ps_fieldcat_lvc  TO <lfs>.
  ENDIF.

  p_text  = <lfs>.

  UNASSIGN <lfs>.


ENDFORM. " GET_FIELD_TEXT

  ENDIF.

  CHECK lt_tab[] IS NOT INITIAL.
  SORT lt_tab BY row col.

  DELETE lt_tab WHERE row IS INITIAL AND col IS INITIAL.
  CHECK lt_tab[] IS NOT INITIAL.

  FIELD-SYMBOLS: <fs>.


  LOOP AT lt_tab INTO ls_tab.
    ASSIGN COMPONENT ls_tab-col OF STRUCTURE gs_data TO <fs>.
    <fs> = ls_tab-value.
    AT END OF row.
      APPEND gs_data TO gt_data.
      CLEAR gs_data.
    ENDAT.
  ENDLOOP.

ENDFORM. " UPLOAD_DATA
