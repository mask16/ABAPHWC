FORM PREPARE_INT_TAB TABLES IT_DATA
                            IT_HEADING STRUCTURE LINE.

  CLEAR WA_LINE.
  IF NOT IT_HEADING[] IS INITIAL.
    LOOP AT IT_HEADING.
      CONCATENATE WA_LINE-LINE
                  IT_HEADING-LINE
                  W_TAB
                  INTO WA_LINE-LINE.
      CONDENSE WA_LINE.
    ENDLOOP.
    APPEND WA_LINE TO IT_LINE.
  ENDIF.
  LOOP AT IT_DATA.
    CLEAR WA_LINE.
    DO.
      ASSIGN COMPONENT SY-INDEX OF STRUCTURE IT_DATA TO <FS_FIELD>.
      IF NOT SY-SUBRC IS INITIAL.
        EXIT.
      ENDIF.
      W_FIELD = <FS_FIELD>.
      CONDENSE W_FIELD.
      CONCATENATE WA_LINE-LINE
                  W_FIELD
                  W_TAB
                  INTO WA_LINE-LINE.
      CONDENSE WA_LINE.
    ENDDO.
    APPEND WA_LINE TO IT_LINE.
  ENDLOOP.

ENDFORM. " PREPARE_INT_TAB
