 FORM INIT_COMMENT.

   DATA: DATE1(10) TYPE C,

   ALV_HEADER-TYP  = 'S'.
   ALV_HEADER-KEY  = TEXT-TT9.
   WRITE START_DT DD/MM/YYYY TO DATE1.
   WRITE END_DT DD/MM/YYYY TO DATE2.

               INTO ALV_HEADER-INFO

   APPEND ALV_HEADER.

   ALV_HEADER-TYP  = 'S'.
   ALV_HEADER-KEY  = TEXT-TT1.
