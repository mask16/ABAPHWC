  DATA : FILE_ALREADY_EXISTS TYPE C.

    RAISING DATA_EMPTY.
  ENDIF.

  IF USE_CPBD IS NOT INITIAL.

    ASSIGN W_TAB TO <FS_HEX> TYPE 'X'.

    REFRESH IT_LINE.         "Clip Board 전송 data 생성
    PERFORM PREPARE_INT_TAB TABLES T_DATA

  ENDIF.

  PERFORM CREATE_EXCEL_SHEET USING FILENAME     "Excel File 생성
                                    TABNAME
                                    T_DATA
                                    USE_CPBD
                                    NO_CLOSE
                                    OVERWRITE
                           CHANGING FILE_ALREADY_EXISTS.

  IF T_FORMATOPT[] IS NOT INITIAL.                  "Format 설정
    PERFORM FORMAT_CELLS TABLES T_FORMATOPT
                          USING FILENAME
                                FILE_ALREADY_EXISTS.
  ENDIF.

  IF USE_CPBD IS INITIAL.         "CLIP BOARD 미사용시 Data 전송
    PERFORM SET_TRANS_DATA2 TABLES T_DATA
                                   T_HEADING.
  ENDIF.

  IF NO_CLOSE IS INITIAL.
    PERFORM SAVE_AND_CLOSE USING FILENAME               "File 저장
                                 FILE_ALREADY_EXISTS.
  ENDIF.

ENDFUNCTION.
