FORM GET_DATA_DISP .

  DATA: L_PERSK TYPE PERSK,
        L_ANSVH TYPE ANSVH,
        L_TPROG TYPE TPROG,
        L_SHORT TYPE SHORT_D.

  DATA: LT_PWS TYPE TABLE OF PDPSP WITH HEADER LINE.


  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_DIS
    FROM ZHRT1010
   WHERE ZDATS = P_DATE
     AND ZPERNR IN S_PERNR.

  LOOP AT GT_DIS.

    CLEAR: L_PERSK, L_ANSVH.

    IF GT_DIS-ZPERNR = SPACE.
      DELETE GT_DIS INDEX SY-TABIX.
      CONTINUE.
    ENDIF.

    SELECT SINGLE ORGEH PERSK ANSVH ENAME
      INTO (GT_DIS-ORGEH, L_PERSK, L_ANSVH, GT_DIS-ENAME)
      FROM PA0001
     WHERE PERNR = GT_DIS-ZPERNR
       AND BEGDA <= P_DATE
       AND ENDDA >= P_DATE.

    CALL FUNCTION 'HRWPC_RFC_ORGEH_TEXT_GET'
      EXPORTING
        ORGEH       = GT_DIS-ORGEH
        BEGDA       = P_DATE
        ENDDA       = P_DATE
      IMPORTING
        ORGEH_TEXT1 = L_SHORT   "조직 Short Text (ORG_INTEGRATION = 'X')
        ORGEH_TEXT2 = GT_DIS-ORGTX.  "조직 Full Text

    SELECT SINGLE PTEXT
      INTO GT_DIS-PKTXT  "사원하위그룹
      FROM T503T
     WHERE SPRSL = '3'
       AND PERSK = L_PERSK.

    SELECT SINGLE ATX
      INTO GT_DIS-ATX  "공정
      FROM T542T
     WHERE SPRAS = '3'
       AND MOLGA = '41'
       AND ANSVH = L_ANSVH.

    SELECT SINGLE SCHKZ
      INTO GT_DIS-SCHKZ  "근무조
      FROM PA0007
     WHERE PERNR = GT_DIS-ZPERNR
       AND BEGDA <= P_DATE
       AND ENDDA >= P_DATE.

*// 근무일정
    CLEAR: LT_PWS, LT_PWS[], L_TPROG.
    CALL FUNCTION 'HR_READ_TIMEDATA_PSP'
      EXPORTING
        BEGIN_DATE = P_DATE
        END_DATE   = P_DATE
        PERSON_ID  = GT_DIS-ZPERNR
      TABLES
        PERS_AVAIL = LT_PWS.

*// 일일근무일정
*    IF ( LT_PWS-FTKLA = C_FTKLA_1 OR LT_PWS-FTKLA = C_FTKLA_2 ) AND
*          LT_PWS-TPROG CP '*OFF*'.
*      L_TPROG = C_OFF.  "비번
*    ELSE.
*      L_TPROG = LT_PWS-TPROG.  "일일근무일정
*    ENDIF.
*
*    CASE L_TPROG.
*      WHEN 'DAY0' OR 'NORM' OR 'NOR2'.
*        GT_DIS-TPROG = '주간'.
*      WHEN 'NIGT'.
*        GT_DIS-TPROG = '야간'.
*      WHEN 'OFF' OR 'OFFN'.
*        GT_DIS-TPROG = '비번'.
*      WHEN 'HOL'.
*        GT_DIS-TPROG = '휴일'.
*    ENDCASE.

    READ TABLE LT_PWS INDEX 1.
    READ TABLE GT_TPROG WITH KEY TPROG = LT_PWS-TPROG.
    IF SY-SUBRC = 0.
      GT_DIS-TPROG = LT_PWS-TPROG.
      CASE LT_PWS-TPROG.
        WHEN 'DAY0'.
          CONCATENATE GT_TPROG-TTEXT '(08:00-18:00)'
                 INTO GT_DIS-TPTXT.
        WHEN 'NIGT'.
          CONCATENATE GT_TPROG-TTEXT '(21:00-08:00)'
                 INTO GT_DIS-TPTXT.

      ENDCASE.
    ENDIF.

    IF GT_DIS-STATE = '@MF@'.
      GT_DIS-STKEY = '1'.
    ELSEIF GT_DIS-STATE = '@OJ@'.
      GT_DIS-STKEY = '0'.
    ENDIF.

*    IF GT_DIS-REFLECT = ''.  "전송여부 ('': 미전송 / X: 전송완료)
*      GT_DIS-ICON_TRA = ICON_INCOMPLETE.
*    ELSEIF GT_DIS-REFLECT = 'X'.
*      GT_DIS-ICON_TRA = ICON_CHECKED.
*    ENDIF.

    &PIMSG GT_DIS-ZPERNR GT_DIS-ENAME.

    MODIFY GT_DIS. CLEAR GT_DIS.

  ENDLOOP.

*

    IF GT_TAB-BEG_TIME IS INITIAL AND
       GT_TAB-MIB_TIME IS INITIAL AND
       GT_TAB-END_TIME IS INITIAL.
      GT_TAB-STATE = ICON_STATUS_CRITICAL.  "출퇴근(문제)

    ELSEIF GT_TAB-BEG_TIME IS INITIAL OR
           GT_TAB-END_TIME IS INITIAL.
      GT_TAB-STATE = ICON_STATUS_CRITICAL.  "출퇴근(문제)
    ELSE.
      GT_TAB-STATE = ICON_STATUS_BEST.      "출퇴근(정상)
    ENDIF.

    MOVE-CORRESPONDING GT_TAB TO LT_1010.
    APPEND LT_1010. CLEAR GT_TAB.
  ENDLOOP.

  IF LT_1010[] IS NOT INITIAL.
    MODIFY ZHRT1010 FROM TABLE LT_1010.
    COMMIT WORK.
    MESSAGE S016 WITH '※ 출퇴근 DATA 가 저장되었습니다~!'.
  ENDIF.

ENDFORM. " SAVE_DATA
